def triton_matmul_outgoing(left, right):
    # left, right shape [B,N,N,H]
    B, N, _, H = left.shape
    device = left.device
    # allocate output
    out = torch.empty_like(left, device=device)

    # Get pointers as integer address using .data_ptr()
    left_ptr = left
    right_ptr = right
    out_ptr = out

    # Compute strides (C-contiguous)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Use triton launch
    import math
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid_m = math.ceil(N / BLOCK_M)
    grid_n = math.ceil(N / BLOCK_N)
    grid_bh = B * H   # number of combined batch*hidden dimension

    # Define kernel
    @triton.jit
    def matmul_outgoing_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        pid_bh = tl.program_id(0)
        pid_tile = tl.program_id(1)

        # compute batch and hidden from pid_bh
        b = pid_bh // H
        h = pid_bh % H

        # tile indices
        num_tiles_m = tl.cdiv(N, BLOCK_M)
        # compute tile row and col
        pid_m = pid_tile // tl.cdiv(N, BLOCK_N)
        pid_n = pid_tile % tl.cdiv(N, BLOCK_N)

        # offset within matrix
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        # Initialize accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over K dimension tiles
        for k in range(0, tl.cdiv(N, BLOCK_K)):
            k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
            # mask for out-of-bounds
            mask_a = (offs_m < N)[:, None] & (k_offset < N)[None, :]
            mask_b = (k_offset < N)[:, None] & (offs_n < N)[None, :]

            # pointers for A (left)
            a_ptrs = left_ptr + \
                b * stride_l_b + \
                h * stride_l_h + \
                offs_m[:, None] * stride_l_i + \
                k_offset[None, :] * stride_l_k

            # pointers for B (right transposed)
            b_ptrs = right_ptr + \
                b * stride_r_b + \
                h * stride_r_h + \
                k_offset[:, None] * stride_r_k + \
                offs_n[None, :] * stride_r_j

            a = tl.load(a_ptrs, mask=mask_a, other=0.0)
            b = tl.load(b_ptrs, mask=mask_b, other=0.0)
            acc += tl.dot(a, b)

        # Write back result
        c_ptrs = out_ptr + \
            b * stride_o_b + \
            h * stride_o_h + \
            offs_m[:, None] * stride_o_i + \
            offs_n[None, :] * stride_o_j

        mask_c = (offs_m < N)[:, None] & (offs_n < N)[None, :]
        tl.store(c_ptrs, acc, mask=mask_c)

    # Launch kernel
    grid = (grid_bh, grid_m * grid_n)
    matmul_outgoing_kernel[grid](
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    return out

"""
custom_kernel(data) -> Tensor

Implements the forward pass of the TriMul “outgoing” operator from AlphaFold3.
The kernel fuses the expensive contraction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batch of N×N matrix multiplications (A @ Bᵀ) for each hidden
dimension d.  The contraction is performed with a custom Triton kernel that
processes tiles of size (BLOCK_M, BLOCK_N) and reduces over K in blocks.
All other operations (LayerNorms, Linear projections and gating) are
executed with PyTorch functional calls.
"""

import math
import torch
import triton
import triton.language as tl


def _triton_matmul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication for the TriMul contraction:
        out[b, i, j, h] = left[b, i, k, h] * right[b, j, k, h] summed over k.
    left/right: [B, N, N, H] (contiguous, float32).
    Returns out of the same shape.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides (in elements) of the contiguous tensors.
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – tuned for A200 GPUs.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Grid: (batch*hidden, tiles_m * tiles_n)
    grid_bh = B * H
    grid_m = math.ceil(N / BLOCK_M)
    grid_n = math.ceil(N / BLOCK_N)
    grid = (grid_bh, grid_m * grid_n)

    @triton.jit
    def _kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        # program IDs
        pid_bh = tl.program_id(0)    # combined batch and hidden dimension
        pid_tile = tl.program_id(1)  # tile index in the (i, j) plane

        # Decode batch and hidden index
        b = pid_bh // H
        h = pid_bh % H

        # Decode tile row / col
        num_tiles_m = tl.cdiv(N, BLOCK_M)
        num_tiles_n = tl.cdiv(N, BLOCK_N)
        pid_m = pid_tile // num_tiles_n
        pid_n = pid_tile % num_tiles_n

        # Offsets inside the matrix
        offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        # Accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over K dimension
        num_k_tiles = tl.cdiv(N, BLOCK_K)
        for k_tile in range(0, num_k_tiles):
            offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)

            # Masks for out‑of‑bounds loads
            mask_a = (offs_i < N)[:, None] & (offs_k < N)[None, :]
            mask_b = (offs_k < N)[:, None] & (offs_j < N)[None, :]

            # Pointers for A = left[b, i, k, h]
            a_ptrs = left_ptr + \
                b * stride_l_b + \
                h * stride_l_h + \
                offs_i[:, None] * stride_l_i + \
                offs_k[None, :] * stride_l_k

            # Pointers for Bᵀ = right[b, j, k, h]ᵀ  (shape K×N)
            b_ptrs = right_ptr + \
                b * stride_r_b + \
                h * stride_r_h + \
                offs_k[:, None] * stride_r_k + \
                offs_j[None, :] * stride_r_j

            a = tl.load(a_ptrs, mask=mask_a, other=0.0)
            b = tl.load(b_ptrs, mask=mask_b, other=0.0)

            # Block‑wise matmul
            acc += tl.dot(a, b)

        # Write result to out[b, i, j, h]
        c_ptrs = out_ptr + \
            b * stride_o_b + \
            h * stride_o_h + \
            offs_i[:, None] * stride_o_i + \
            offs_j[None, :] * stride_o_j

        mask_c = (offs_i < N)[:, None] & (offs_j < N)[None, :]
        tl.store(c_ptrs, acc, mask=mask_c)

    # Launch
    _kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of TriMul (outgoing) using a Triton kernel for the core
    contraction.  Arguments:
        data = (x, mask, weights, config)
            x       : [B, N, N, dim]  torch.Tensor (float32, CUDA)
            mask    : [B, N, N]       torch.Tensor (float32 or bool)
            weights : dict of model parameters (all float32 CUDA tensors)
            config  : dict with keys "dim" and "hidden_dim"
    Returns:
        out : [B, N, N, dim] torch.Tensor (float32, CUDA)
    """
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------ LayerNorm on input ------------------
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------ Linear projections ------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------ Apply mask (if present) ------------------
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------ Gating ---------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguity for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------------------ Core contraction (Triton) -------------
    out = _triton_matmul_outgoing(left, right)   # [B,N,N,H]

    # ------------------ Output LayerNorm -----------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------ Output gating and final linear -------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

        hidden = cfg["hidden_dim"]
