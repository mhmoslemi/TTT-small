def tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    b = tl.program_id(0)  # batch*hidden dim
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_n = offs_n < N

def tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing Triangle Multiplicative Update.
    Steps:
    1. Layer‑norm on the input.
    2. Linear projections (left/right) and gated versions.
    3. Batched matrix‑multiply (L @ Rᵀ) via a Triton kernel.
    4. Layer‑norm + gating on the hidden representation.
    5. Final linear projection back to the original channel dimension.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # step 1: layer norm
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # step 2: projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    if not config.get("nomask", False):
        # mask is assumed to be 0/1 float
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape  # B,N,N,hidden_dim
    # reshape for batched GEMM: (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch*hidden dimension
    batch_h = left_perm.shape[0] * left_perm.shape[1]  # B * hidden_dim
    left_mat = left_perm.view(batch_h, N, N)
    right_mat = right_perm.view(batch_h, N, N)

    out_mat = torch.empty_like(left_mat)

    # launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        left_mat.stride(0),
        left_mat.stride(1),
        left_mat.stride(2),
        right_mat.stride(0),
        right_mat.stride(2),  # stride for K (original column)
        right_mat.stride(1),  # stride for N (original row)
        out_mat.stride(0),
        out_mat.stride(1),
        out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to [B, N, N, hidden_dim]
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # step 4: hidden layer norm + gating
    mean_h = out_hidden.mean(dim=-1, keepdim=True)
    var_h = ((out_hidden - mean_h) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out_hidden - mean_h) / torch.sqrt(var_h + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]
    out_norm = out_norm * out_gate

    # step 5: final linear projection back to dim
    out = F.linear(out_norm, weights["to_out.weight"])

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    A_ptr,                     # *[batch_h, M, K]  left matrix
    B_ptr,                     # *[batch_h, N, K]  right matrix (will be read as KxN)
    C_ptr,                     # *[batch_h, M, N]  output
    M, N, K,                   # matrix sizes (int32 scalars)
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (note: B is interpreted as KxN)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel computing  C = A @ Bᵀ  for many batches.

    A has shape (batch_h, M, K)   – row‑major.
    B has shape (batch_h, N, K)   – we treat it as (batch_h, K, N) by swapping
                                   the K‑ and N‑strides, thus performing A @ Bᵀ.
    """
    pid_b = tl.program_id(0)               # batch*hidden index
    pid_m = tl.program_id(1)               # block row index
    pid_n = tl.program_id(2)               # block column index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the current block
    a_ptrs = A_ptr + pid_b * stride_Ab \
                     + offs_m[:, None] * stride_Am \
                     + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * stride_Bb \
                     + offs_k[:, None] * stride_Bk \
                     + offs_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)        # (BLOCK_M, BLOCK_N)

        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    c_ptrs = C_ptr + pid_b * stride_Cb \
                     + offs_m[:, None] * stride_Cm \
                     + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update.

    Steps
    -----
    1. Layer‑norm the input tensor.
    2. Linear projections `left` and `right` and their sigmoid gates.
    3. Optional masking of the projected tensors.
    4. Batched matrix multiplication `out = left @ rightᵀ` via a Triton kernel.
    5. Layer‑norm + sigmoid gate on the hidden output.
    6. Final linear projection back to the original channel dimension.

    Input
    -----
    data = (input_tensor, mask_tensor, weights, config)
        input_tensor : [B, N, N, dim]   (float32)
        mask_tensor  : [B, N, N]        (0/1 mask, may be ignored)
        weights      : dict of model weights (see function body)
        config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Output
    ------
    Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------
    # 1) Layer‑norm on the input
    # ------------------------------------------------------------
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]       # shape [B,N,N,dim]

    # ------------------------------------------------------------
    # 2) Linear projections
    # ------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------
    # 3) Optional masking (applied before gating)
    # ------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(dtype).unsqueeze(-1)    # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------
    # 4) Compute gates
    # ------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------
    # 5) Core triangular multiplication (batched GEMM)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------
    B, N, _, _ = left.shape                     # B, N, N, hidden_dim
    # reshape to (B, hidden_dim, N, N) and flatten batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    batch_h = B * hidden_dim
    left_mat = left_perm.view(batch_h, N, N)    # [batch_h, M, K]
    right_mat = right_perm.view(batch_h, N, N)  # [batch_h, N, K] (will be read as KxN)

    out_mat = torch.empty_like(left_mat)        # [batch_h, M, N]

    # launch Triton kernel
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,                     # M
        N,                     # N
        N,                     # K
        left_mat.stride(0),    # stride_Ab
        left_mat.stride(1),    # stride_Am
        left_mat.stride(2),    # stride_Ak
        right_mat.stride(0),   # stride_Bb
        right_mat.stride(2),   # stride_Bk (original K‑stride)
        right_mat.stride(1),   # stride_Bn (original N‑stride)
        out_mat.stride(0),     # stride_Cb
        out_mat.stride(1),     # stride_Cm
        out_mat.stride(2),     # stride_Cn
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------
    # 6) Hidden‑dim layer norm + gating
    # ------------------------------------------------------------
    mean_h = out_hidden.mean(dim=-1, keepdim=True)
    var_h = ((out_hidden - mean_h) ** 2).mean(dim=-1, keepdim=True)
    out = (out_hidden - mean_h) / torch.sqrt(var_h + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]
    out = out * out_gate                               # gating

    # ------------------------------------------------------------
    # 7) Final linear projection back to original dim
    # ------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
