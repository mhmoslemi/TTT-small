"""
TriMul (outgoing) implementation.

The algorithm follows the reference PyTorch module:
1) Layer‑norm the input.
2) Linear projections and gating (all done with PyTorch).
3) Apply mask (if present) and element‑wise gating.
4) Core O(N³) contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication per hidden channel.
   It is implemented with a custom Triton kernel that tiles the
   multiplication over the (i,j) axes while iterating over k.
5) Second layer‑norm, out‑gate multiplication and final linear projection.

The heavy O(N³) part is accelerated by the Triton kernel, while the
remaining lightweight operations stay in PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


# ---------------------------------------------------------------
# Triton kernel: batched GEMM over (i,j) for every hidden channel d
# ---------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    left_ptr,                     # const float16*  [B, N, N, H]
    right_ptr,                    # const float16*  [B, N, N, H]
    out_ptr,                      #       float32*  [B, N, N, H]
    B, N, H,                      # int32
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # int64
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # int64
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # int64
    BLOCK_M: tl.constexpr,        # tile size for i
    BLOCK_N: tl.constexpr,        # tile size for j
    BLOCK_K: tl.constexpr,        # tile size for k
):
    pid_i = tl.program_id(0)      # tile index along i
    pid_j = tl.program_id(1)      # tile index along j
    pid_z = tl.program_id(2)      # combined batch*hidden index

    # derive batch and hidden indices
    b = pid_z // H
    h = pid_z % H

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: shape (BLOCK_M, BLOCK_K)  -> (i, k)
        a_ptrs = (left_ptr
                  + b * stride_l_b
                  + h * stride_l_h
                  + offs_i[:, None] * stride_l_i
                  + offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)               # half precision
        # right: shape (BLOCK_K, BLOCK_N) -> (k, j)
        b_ptrs = (right_ptr
                  + b * stride_r_b
                  + h * stride_r_h
                  + offs_k[:, None] * stride_r_k
                  + offs_j[None, :] * stride_r_j)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)               # half precision

        # dot product in half, cast to fp32 for accumulation
        acc += tl.dot(a, b).to(tl.float32)

    # store result
    c_ptrs = (out_ptr
              + b * stride_o_b
              + h * stride_o_h
              + offs_i[:, None] * stride_o_i
              + offs_j[None, :] * stride_o_j)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def trimul_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication per hidden channel using Triton.
    Both `left` and `right` must be of shape [B, N, N, H] and dtype float16.
    Returns a tensor of shape [B, N, N, H] and dtype float32.
    """
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, N2, H = left.shape
    assert N == N2, "Input must be square in the 2nd and 3rd dimensions."

    # output in float32 (higher precision for the downstream layernorm)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tunable block sizes (work well for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = ((N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            B * H)

    trimul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,

        stride_o_h,

        BLOCK_K=BLOCK_K,
        num_warps=8,   # sufficient parallelism on B200
    )

    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm ----------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ---------- Linear projections ----------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- Mask ----------
    if mask is None or config.get("nomask", False):
        # No mask – skip masking step
        pass
    else:
        # mask shape [B,N,N]; broadcast over hidden dim
        mask_f = mask.to(dtype=x.dtype, device=device).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to half for kernel ----------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # ---------- Core contraction (TriMul) ----------
    out_hidden = trimul_matmul_triton(left_h, right_h)   # [B,N,N,hidden_dim] float32

    # ---------- Second LayerNorm ----------
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ---------- Out gate ----------
    out_hidden = out_hidden * out_gate

    # ---------- Final projection ----------
    out = torch.nn.functional.linear(out_hidden, weights["to_out.weight"])  # [B,N,N,dim]

    return out
