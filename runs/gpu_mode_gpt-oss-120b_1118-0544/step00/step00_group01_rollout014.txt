"""
TriMul forward implementation using Triton for the heavy contraction.
The kernel computes, for each batch and hidden dimension d:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication of left with the transpose of right.
All other operations (LayerNorm, projections, gates, final LayerNorm and linear)
are performed in PyTorch. The Triton kernel is tiled over the sequence dimension
to achieve high memory reuse.
"""
import torch
import triton
import triton.language as tl
import math

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # dimensions
    stride_lb, stride_li, stride_lk, stride_lh,  # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,  # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,  # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """ Compute out = left @ right^T over the K dimension. """
    # 3‑D grid: (block_i, block_j, batch*head)
    pid_i = tl.program_id(0)  # block row (i)
    pid_j = tl.program_id(1)  # block col (j)
    pid_bh = tl.program_id(2) # combined batch and hidden index

    # decode batch and hidden dimension
    b = pid_bh // H
    h = pid_bh % H

    # offsets for the i, j and k dimensions
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds i / j
    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of tiles over K
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(num_k_tiles):
        k_offset = tk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ----- load left tile A[i,k] (shape M×K) -----
        a_ptrs = left_ptr \
                 + b * stride_lb \
                 + i_offset[:, None] * stride_li \
                 + k_offset[None, :] * stride_lk \
                 + h * stride_lh
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ----- load right tile B_T[k,j] (shape K×N) -----
        # right is stored as (b, j, k, h); we need the transpose on (k,j)
        b_ptrs = right_ptr \
                 + b * stride_rb \
                 + k_offset[:, None] * stride_rk \
                 + j_offset[None, :] * stride_rj \
                 + h * stride_rh
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # dot product for this K‑tile
        acc += tl.dot(a, b)   # (M,N)

    # ----- store result tile -----
    out_ptrs = out_ptr \
               + b * stride_ob \
               + i_offset[:, None] * stride_oi \
               + j_offset[None, :] * stride_oj \
               + h * stride_oh
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton implementation of the tri‑mul contraction.
    left,right: shape [B, N, N, H] (float32, contiguous)
    returns:   out   : shape [B, N, N, H] (float32)
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), dtype=left.dtype, device=left.device)

    # strides in number of elements
    l_str = left.stride()
    r_str = right.stride()
    o_str = out.stride()

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        l_str[0], l_str[1], l_str[2], l_str[3],
        r_str[0], r_str[1], r_str[2], r_str[3],
        o_str[0], o_str[1], o_str[2], o_str[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing version) using a Triton kernel
    for the contraction. Returns a tensor of shape [bs, seq_len, seq_len, dim].
    """
    # unpack inputs
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Helper: LayerNorm (last dimension) using supplied weight/bias
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # Projections
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]    # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight) # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Mask handling
    # ------------------------------------------------------------------
    if mask_tensor is None or nomask:
        mask = torch.ones(
            (x.shape[0], x.shape[1], x.shape[2]),
            dtype=dtype,
            device=device,
        )
    else:
        mask = mask_tensor.to(dtype)

    mask = mask.unsqueeze(-1)   # [B, N, N, 1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Gate values (sigmoid of linear transforms of the normalized input)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core contraction via Triton
    # ------------------------------------------------------------------
    out = tri_mul_triton(left, right)   # [B, N, N, hidden_dim]

    # Apply out‑gate
    out = out * out_gate

    # ------------------------------------------------------------------
    # Output LayerNorm and projection back to dim
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim]

    return out
