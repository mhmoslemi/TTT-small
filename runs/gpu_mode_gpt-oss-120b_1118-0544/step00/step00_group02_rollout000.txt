def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # using torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(
            input_tensor,
            normalized_shape=(dim,),
            weight=norm_weight,
            bias=norm_bias
        )
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Masking
    if mask is not None and not config.get("nomask", False):
        mask_ = mask.unsqueeze(-1).type_as(left)  # shape B,N,N,1
        left = left * mask_
        right = right * mask_
    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate
    # Core multiplication using Triton
    B, N, _, _ = left.shape  # shape B,N,N,hidden_dim
    # allocate out
    out = torch.empty_like(left)  # same shape dtype float32
    # prepare strides
    left_stride_b, left_stride_i, left_stride_k, left_stride_h = left.stride()
    right_stride_b, right_stride_i, right_stride_k, right_stride_h = right.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_h = out.stride()
    # note for right we need stride for i and k (i is first N, k is second N)
    # it's okay, we treat same.
    # set block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (B * hidden_dim, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    tri_mul_matmul_kernel[grid](
        left, right, out,
        left_stride_b, left_stride_i, left_stride_k, left_stride_h,
        right_stride_b, right_stride_i, right_stride_k, right_stride_h,
        out_stride_b, out_stride_i, out_stride_j, out_stride_h,
        B, N, hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K

    # After that, out is shape B,N,N,hidden_dim

    out = torch.nn.functional.layer_norm(

        bias=weights["to_out_norm.bias"]
    )

    # Final projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    # Pointers to the batch/head slice
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # Compute offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, N, BLOCK_K):
        # offsets for k
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left_block = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right block transposed: right block shape (BLOCK_N, BLOCK_K) after transpose
        # Original right indices: (j, k); we want (k, j). So use i_stride for j dimension and k_stride for k.
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_i
        right_block = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(left_block, right_block)  # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Write back
    out_ptrs = out_base + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul outgoing implementation with Triton acceleration.

Algorithm:
1. LayerNorm the input over the channel dimension.
2. Linear projections: left_proj, right_proj (dim → hidden_dim).
3. (Optional) Apply pairwise mask to the projected tensors.
4. Compute gating vectors with sigmoid and multiply element‑wise.
5. Core operation: out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d].
   This is performed as a batched GEMM: for each (batch, head) pair,
   compute C = A @ Bᵀ where A=left[...,d] and B=right[...,d].
   The GEMM is implemented in a Triton kernel (block‑wise matmul).
6. LayerNorm over the hidden dimension, apply output gate, and final linear
   projection back to the original channel dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for batched matrix multiplication with transposition.
# Computes out[b, i, j, h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    # strides for left: (batch, i, k, hidden)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for right: (batch, i, k, hidden)
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # strides for out: (batch, i, j, hidden)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)          # combined batch * hidden index
    pid_m = tl.program_id(1)           # block row index for output
    pid_n = tl.program_id(2)           # block col index for output

    # Resolve batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # Base pointers for this (batch, hidden) slice
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # Offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator for the output block
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block (M x K)
        left_ptrs = left_base + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left_block = tl.load(left_ptrs,
                             mask=mask_m[:, None] & mask_k[None, :],
                             other=0.0).to(tl.float32)

        # Load right block transposed (K x N) -> we need right[j,k] -> (k,j) layout
        # right indices: (batch, i=row=j, k=col=k, hidden)
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_i
        right_block = tl.load(right_ptrs,
                              mask=mask_k[:, None] & mask_n[None, :],
                              other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(left_block, right_block)  # (M,K) @ (K,N) -> (M,N)

    # Write the result block back to global memory
    out_ptrs = out_base + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Performs the outgoing TriangleMultiplicativeUpdate (TriMul) forward pass.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict with all linear/LayerNorm parameters
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if mask is not None and not nomask:
        # mask: [B,N,N] -> unsqueeze to broadcast over hidden_dim
        mask_f = mask.to(dtype).unsqueeze(-1)       # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ---------- Core TriMul via Triton ----------
    B, N, _, _ = left.shape
    out = torch.empty_like(left)   # [B,N,N,hidden_dim]

    # Strides (in elements) for each tensor
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Block sizes – chosen to fit typical B200 capacity
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Launch grid: (batch*head, row_blocks, col_blocks)
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_matmul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        B, N, hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Output LayerNorm, gating, and final projection ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to original dim

    return out
