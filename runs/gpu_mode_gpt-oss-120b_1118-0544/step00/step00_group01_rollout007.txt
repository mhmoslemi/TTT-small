import torch
import triton
import triton.language as tl

def custom_kernel(data):
    """TriMul outgoing forward pass with Triton fused batched matmul.

    Implements:
        - LayerNorm on input.
        - Linear projections and gating.
        - Elementwise mask (if provided).
        - Batched matrix multiplication across the `k` dimension for each hidden channel
          using a custom Triton kernel.
        - Output LayerNorm, gating, and final linear projection.
    All computations are performed in FP16 inside the kernel for speed, with final
    result cast back to FP32.
    """
    # Unpack input
    input_tensor, mask, weights, config = data
    # Ensure contiguous
    x = input_tensor
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]  # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)    # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if needed
    if not nomask and mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Prepare tensors for Triton batched matmul
    # left: [B,N,N,H]; right: [B,N,N,H]; compute out = sum_k left[...,i,k,d] * right[...,j,k,d]
    # Transform to batched matrix multiply per (batch, hidden) pair.
    # left_perm = left.permute(0,3,1,2) -> [B,H,N,N]
    # right_T = right.permute(0,2,1,3) (swap i,j) -> [B,N,N,H]
    # right_perm = right_T.permute(0,3,1,2) -> [B,H,N,N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_T = right.permute(0, 2, 1, 3).contiguous()
    right_perm = right_T.permute(0, 3, 1, 2).contiguous()

    # Flatten batch and hidden dimension
    total_batches = B * hidden_dim
    A = left_perm.view(total_batches, N, N).to(torch.float16)
    B_ = right_perm.view(total_batches, N, N).to(torch.float16)

    # Allocate output
    C = torch.empty_like(A, dtype=torch.float16)

    # Determine block sizes (tuned for GPU)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute strides (in elements) for A, B, C
    stride_ab = A.stride(0)  # batch stride
    stride_am = A.stride(1)  # M stride
    stride_ak = A.stride(2)  # K stride

    stride_bb = B_.stride(0)
    stride_bk = B_.stride(1)
    stride_bn = B_.stride(2)

    stride_cb = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    # Kernel definition
    @triton.jit
    def batched_matmul_kernel(
        a_ptr, b_ptr, c_ptr,
        M, N, K,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        batch_id = tl.program_id(0)
        pid_m = tl.program_id(1)
        pid_n = tl.program_id(2)

        # Compute offsets
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        # Initialize accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Pointers to the start of each batch
        a_base = a_ptr + batch_id * stride_ab
        b_base = b_ptr + batch_id * stride_bb
        c_base = c_ptr + batch_id * stride_cb

        # Loop over K dimension
        for k in range(0, tl.cdiv(K, BLOCK_K)):
            # Compute pointers for this tile
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

            a_ptrs = a_base + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
            b_ptrs = b_base + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

            # Load with mask
            a = tl.load(a_ptrs,
                        mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                        other=0.0)
            b = tl.load(b_ptrs,
                        mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                        other=0.0)

            # Dot product accumulate
            acc += tl.dot(a, b)

        # Write back result
        c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
        tl.store(c_ptrs, acc.to(tl.float16),
                 mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

    # Launch kernel
    grid = (total_batches,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        A, B_, C,
        N, N, N,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B,N,N,hidden_dim]
    out = C.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm, gate, and final linear projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,),
                                          weight=to_out_norm_weight,
                                          bias=to_out_norm_bias,
                                          eps=eps)
    out = out * out_gate
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return out

"""
TriMul (outgoing) forward pass with a fused Triton batched‑matmul.

The heavy part of the algorithm is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batch of `hidden_dim` independent N×N matrix multiplications.
We implement this contraction with a custom Triton kernel that processes
`B * hidden_dim` matrices in parallel, using FP16 arithmetic with
FP32 accumulation. All other operations (LayerNorms, linear projections,
gates, and final linear) are performed with PyTorch for simplicity.
"""
import torch
import triton
import triton.language as tl

def custom_kernel(data):
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → ignore mask

    # ---------- LayerNorm on input ----------
    eps = 1e-5
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]

    # ---------- Linear projections ----------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if not nomask and mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gating ----------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare tensors for Triton batched matmul ----------
    # left_perm: [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right_T swaps the second and third axes, then bring hidden dim forward
    right_T = right.permute(0, 2, 1, 3).contiguous()
    right_perm = right_T.permute(0, 3, 1, 2).contiguous()

    B, N = left_perm.shape[0], left_perm.shape[2]   # B, N
    total_batches = B * hidden_dim

    # Flatten batch×hidden_dim dimension and cast to fp16 for the kernel
    A = left_perm.view(total_batches, N, N).to(torch.float16)
    B_ = right_perm.view(total_batches, N, N).to(torch.float16)
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides (in element counts)
    stride_ab = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_bb = B_.stride(0)
    stride_bk = B_.stride(1)
    stride_bn = B_.stride(2)

    stride_cb = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    # ----------- Triton kernel (batched matmul) -----------
    @triton.jit
    def batched_matmul_kernel(
        a_ptr, b_ptr, c_ptr,
        M, N, K,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        batch_id = tl.program_id(0)
        pid_m = tl.program_id(1)
        pid_n = tl.program_id(2)

        # Offsets within the tile
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        # Base pointers for this batch
        a_batch = a_ptr + batch_id * stride_ab
        b_batch = b_ptr + batch_id * stride_bb
        c_batch = c_ptr + batch_id * stride_cb

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k in range(0, tl.cdiv(K, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

            a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
            b_ptrs = b_batch + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

            a = tl.load(a_ptrs,
                        mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                        other=tl.float16(0.0))
            b = tl.load(b_ptrs,
                        mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                        other=tl.float16(0.0))

            acc += tl.dot(a, b)

        c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
        tl.store(c_ptrs, acc.to(tl.float16),
                 mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

    # Tuned block sizes (adapted for typical B200 memory hierarchy)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        A,
        B_,
        C,
        N, N, N,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Restore original layout ----------
    out = (
        C.to(torch.float32)
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B,N,N,hidden_dim]

    # ---------- Output normalization, gating and final projection ----------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # -> [B,N,N,dim]

    return out
