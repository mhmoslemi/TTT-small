Implementation details:

We rely on typical matmul kernel pattern: we have three loops over K dimension, block sizes defined. We'll treat each block representing a sub-matrix of dimensions (BLOCK_M,N?) Standard matmul shapes: A [M,K], B [K,N] produce C [M,N]. But we have to multiply across K with elementwise product per hidden block: For each hidden h in block, we compute the matrix multiplication. So we can treat each hidden block as separate channels processed in parallel: we can load left and right for each h and compute dot product across K.

    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h, # left shape [B,N,N,H]
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):

    # Let's compute block indices:
    total_tiles_i = tl.cdiv(N, BLOCK_M)
    total_tiles_j = tl.cdiv(N, BLOCK_N)
    total_tiles_h = tl.cdiv(H, BLOCK_H)

    mask_j = j_offset < N
    mask_h = h_offset < H

def triangular_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_j, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    i_tile = tl.program_id(0)
    j_tile = tl.program_id(1)
    # Offsets
    i_offset = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    h_offset = tl.arange(0, BLOCK_H)
    # Mask for boundaries
    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_h = h_offset < H
    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # loops over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N
        # load left slice: shape (BLOCK_M, BLOCK_K, BLOCK_H)
        # pointer: left_ptr + i_offset[:,None,None]*stride_left_i + k_offset[None,:,None]*stride_left_k + h_offset[None,None,:]*stride_left_h
        left_ptrs = left_ptr + i_offset[:, None, None] * stride_left_i + k_offset[None, :, None] * stride_left_k + h_offset[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0).to(tl.float32)

        # load right slice: shape (BLOCK_K, BLOCK_N, BLOCK_H)
        # pointer: right_ptr + k_offset[:, None, None]*stride_right_k + j_offset[None, :, None]*stride_right_j + h_offset[None, None, :]*stride_right_h
        right_ptrs = right_ptr + k_offset[:, None, None] * stride_right_k + j_offset[None, :, None] * stride_right_j + h_offset[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0).to(tl.float32)
        # Now compute dot product for each h j block: we want sum over k dimension.
        # Equivalent to matrix multiplication left (M,K) * right (K,N) for each h.

        # For each h index:
        for h in range(0, BLOCK_H):
            # left_h: (M,K)
            left_h = left[:, :, h]
            # right_h: (K,N)
            right_h = right[:, :, h]
            acc[:, :, h] += tl.dot(left_h, right_h)

    # store result
    out_ptrs = out_ptr + i_offset[:, None, None] * stride_out_i + j_offset[None, :, None] * stride_out_j + h_offset[None, None, :] * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

"""
TriMul (outgoing) forward implementation using Triton for the
pairwise reduction (sum_k left[i,k,:] * right[j,k,:]).
The heavy reduction is performed with a custom Triton kernel,
while the surrounding LayerNorms, linear projections and gating
are handled with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def triu_mul_outgoing_kernel(
    left_ptr,           # POINTER: left tensor slice  [N, N, H_block]
    right_ptr,          # POINTER: right tensor slice [N, N, H_block]
    out_ptr,            # POINTER: output slice       [N, N, H_block]
    N,                  # SEQ_LEN (int)
    H,                  # CURRENT hidden block size (int)
    stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_j, stride_r_k, stride_r_h,   # right strides
    stride_o_i, stride_o_j, stride_o_h,   # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute for a tile (i_tile, j_tile) of size BLOCK_M x BLOCK_N:
        out[i, j, h] = Σ_k left[i, k, h] * right[j, k, h]
    for h in the current hidden block (size BLOCK_H, masked by H).
    """
    pid_i = tl.program_id(0)  # tile over i dimension
    pid_j = tl.program_id(1)  # tile over j dimension

    # Offsets for the tile
    i_base = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    h_base = tl.arange(0, BLOCK_H)

    # Boundary masks
    mask_i = i_base < N
    mask_j = j_base < N
    mask_h = h_base < H

    # Accumulator for the tile [M,N,H]
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # Loop over k dimension in blocks
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_base < N

        # Load left[i, k, h] -> shape (M, K, H)
        left_ptrs = (
            left_ptr
            + i_base[:, None, None] * stride_l_i
            + k_base[None, :, None] * stride_l_k
            + h_base[None, None, :] * stride_l_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # Load right[j, k, h] transposed to (K, N, H)
        right_ptrs = (
            right_ptr
            + k_base[:, None, None] * stride_r_k
            + j_base[None, :, None] * stride_r_j
            + h_base[None, None, :] * stride_r_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # Perform dot product for each hidden slice
        for h in range(0, BLOCK_H):
            # Only compute for valid h (masked later on store)
            left_h = left[:, :, h]   # (M, K)
            right_h = right[:, :, h] # (K, N)
            acc[:, :, h] += tl.dot(left_h, right_h)

    # Store results with masking
    out_ptrs = (
        out_ptr
        + i_base[:, None, None] * stride_o_i
        + j_base[None, :, None] * stride_o_j
        + h_base[None, None, :] * stride_o_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        output tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Config parameters
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- LayerNorm on input -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ----- Linear projections -------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ----- Mask handling ------------------------------------------------------
    if nomask or mask is None:
        mask_ = torch.ones_like(input_tensor[..., 0], dtype=dtype, device=device)
    else:
        mask_ = mask.to(dtype)

    mask_ = mask_.unsqueeze(-1)  # [B, N, N, 1]

    # ----- Gating -------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply mask and gates (element‑wise)
    left = left * mask_ * left_gate
    right = right * mask_ * right_gate

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape

    # ----- Triton reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] -----
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Kernel block configuration
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 32  # compile‑time maximum hidden block size

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N )

    for b in range(B):
        left_b = left[b]   # [N, N, H]
        right_b = right[b] # [N, N, H]
        out_b = out[b]     # [N, N, H]

        for h_start in range(0, hidden_dim, BLOCK_H):
            cur_h = min(BLOCK_H, hidden_dim - h_start)

            # Slice the current hidden block (still contiguous in memory)
            left_slice = left_b[..., h_start:h_start + BLOCK_H]
            right_slice = right_b[..., h_start:h_start + BLOCK_H]
            out_slice = out_b[..., h_start:h_start + BLOCK_H]

            # Strides (in elements) for the sliced tensors
            stride_l_i, stride_l_k, stride_l_h = left_slice.stride()
            stride_r_j, stride_r_k, stride_r_h = right_slice.stride()
            stride_o_i, stride_o_j, stride_o_h = out_slice.stride()

            # Launch Triton kernel
            triu_mul_outgoing_kernel[grid](
                left_slice,
                right_slice,
                out_slice,
                N,
                cur_h,
                stride_l_i,
                stride_l_k,
                stride_l_h,
                stride_r_j,
                stride_r_k,
                stride_r_h,
                stride_o_i,
                stride_o_j,
                stride_o_h,
                BLOCK_M=BLOCK_M,
                BLOCK_N=BLOCK_N,
                BLOCK_K=BLOCK_K,
                BLOCK_H=BLOCK_H,
            )

    # ----- Output LayerNorm and gating ----------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B, N, N, hidden_dim]

    # Apply output gate
    out = out * out_gate

    # ----- Final linear projection --------------------------------------------
    to_out_weight = weights["to_out.weight"]
    output = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return output
