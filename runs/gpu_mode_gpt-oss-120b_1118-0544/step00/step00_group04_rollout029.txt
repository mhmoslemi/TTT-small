import torch
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # get config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if provided
    if mask is not None and mask.numel() > 0:
        mask_unsq = mask.unsqueeze(-1).type_as(x)
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --- Triton reduction ---
    B, N, _, _ = left.shape

    # Ensure contiguous layout for kernel
    left = left.contiguous()
    right = right.contiguous()

    # Allocate output after reduction (hidden dimension)
    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # Compute strides (in elements)
    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out_hidden.stride()

    # Choose block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)

    _tri_mul_kernel[grid](
        left,
        right,
        out_hidden,
        B,
        N,
        hidden_dim,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # --- End Triton reduction ---

    # LayerNorm on reduced hidden representation
    out_hidden_norm = torch.nn.functional.layer_norm(out_hidden, (hidden_dim,), weight=to_out_norm_weight,
                                                     bias=to_out_norm_bias, eps=eps)
    # Apply out gate
    out_hidden_norm = out_hidden_norm * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out_hidden_norm, to_out_weight)

    return out

@triton.jit
def _tri_mul_kernel(
    left_ptr,              # *pointer to left tensor [B,N,N,hidden_dim] (float32)
    right_ptr,             # *pointer to right tensor [B,N,N,hidden_dim] (float32)
    out_ptr,               # *pointer to output tensor [B,N,N,hidden_dim] (float32)
    B: tl.int32,
    N: tl.int32,
    hidden_dim: tl.int32,
    stride_l_b: tl.int32, stride_l_i: tl.int32, stride_l_k: tl.int32, stride_l_d: tl.int32,
    stride_r_b: tl.int32, stride_r_j: tl.int32, stride_r_k: tl.int32, stride_r_d: tl.int32,
    stride_o_b: tl.int32, stride_o_i: tl.int32, stride_o_j: tl.int32, stride_o_d: tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Triton kernel implementing the core triangular multiplicative update:
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d].
    The kernel processes each (batch, hidden_dim) pair as an independent GEMM
    with shapes (N, N), using tiling over M, N and K dimensions.
    """
    # program ids
    pid_i = tl.program_id(axis=0)   # block row index (i)
    pid_j = tl.program_id(axis=1)   # block column index (j)
    pid_hd = tl.program_id(axis=2)  # combined batch and hidden dimension

    # decode batch and hidden index
    batch_id = pid_hd // hidden_dim
    d = pid_hd % hidden_dim

    # compute block start offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # pointers for left block (M x K)
        left_ptrs = (
            left_ptr
            + batch_id * stride_l_b
            + i_offset[:, None] * stride_l_i
            + k_offset[None, :] * stride_l_k
            + d * stride_l_d
        )
        # pointers for right block transposed (K x N)
        right_ptrs = (
            right_ptr
            + batch_id * stride_r_b
            + k_offset[:, None] * stride_r_k
            + j_offset[None, :] * stride_r_j
            + d * stride_r_d
        )

        # load with mask
        left_block = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right_block = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Cast to float32 (if inputs are float16)
        left_block = left_block.to(tl.float32)
        right_block = right_block.to(tl.float32)

        # GEMM accumulation
        acc += tl.dot(left_block, right_block)

    # store result
    out_ptrs = (
        out_ptr
        + batch_id * stride_o_b
        + i_offset[:, None] * stride_o_i
        + j_offset[None, :] * stride_o_j
        + d * stride_o_d
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The kernel computes the core O(N³) reduction:
    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]

All other operations (LayerNorm, linear projections, gating, masking,
and the final output projection) are performed with PyTorch.
The heavy reduction is off‑loaded to a tiled Triton GEMM kernel
that processes each (batch, hidden_dim) pair as an independent
matrix multiplication (left @ rightᵀ) using 64×64×64 tiles.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask: torch.Tensor or None, shape [B, N, N] (optional)
            - weights: dict with the following (torch.Tensor) entries
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys "dim" and "hidden_dim"
    Returns:
        output torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------ #
    # unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------ #
    # retrieve model weights
    # ------------------------------------------------------------------ #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------ #
    # 1️⃣ Layer‑norm on the input pair representation
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------ #
    # 2️⃣ Linear projections (float32) -> cast to float16 for the kernel
    # ------------------------------------------------------------------ #
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------ #
    # 3️⃣ Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------ #
    if mask is not None and mask.numel() > 0:
        mask_broadcast = mask.unsqueeze(-1).type_as(left)  # [B, N, N, 1]
        left = left * mask_broadcast
        right = right * mask_broadcast

    # ------------------------------------------------------------------ #
    # 4️⃣ Gating (sigmoid) – keep in float32 for numerical stability
    # ------------------------------------------------------------------ #
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()  # later used in float32

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # cast to float16 before the massive reduction (saves memory)
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------ #
    # 5️⃣ Triton kernel for the O(N³) reduction
    # ------------------------------------------------------------------ #
    out_hidden = torch.empty(
        (B, N, N, hidden_dim), dtype=torch.float32, device=device
    )  # result of the einsum

    # stride information (in elements, not bytes)
    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out_hidden.stride()

    # tile sizes – experimentally good for the supported N (≤1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out_hidden,
        B,
        N,
        hidden_dim,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------ #
    # 6️⃣ Layer‑norm on the hidden representation, gating and final proj.
    # ------------------------------------------------------------------ #
    out_hidden_norm = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out_hidden_norm = out_hidden_norm * out_gate  # out_gate is float32

    # final linear back to original dimension
    out = F.linear(out_hidden_norm, to_out_weight)

    return out


@triton.jit
def _tri_mul_kernel(
    left_ptr,               # *pointer to left tensor (float16)
    right_ptr,              # *pointer to right tensor (float16)
    out_ptr,                # *pointer to output tensor (float32)
    B: tl.int32,
    N: tl.int32,
    hidden_dim: tl.int32,
    # strides for left: (batch, i, k, d)
    stride_l_b: tl.int32,
    stride_l_i: tl.int32,
    stride_l_k: tl.int32,
    stride_l_d: tl.int32,
    # strides for right: (batch, j, k, d)
    stride_r_b: tl.int32,
    stride_r_j: tl.int32,
    stride_r_k: tl.int32,
    stride_r_d: tl.int32,
    # strides for out: (batch, i, j, d)
    stride_o_b: tl.int32,
    stride_o_i: tl.int32,
    stride_o_j: tl.int32,
    stride_o_d: tl.int32,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] using
    a tiled GEMM.  Each program instance handles a (batch, hidden_dim)
    pair and a 2‑D tile of the i‑j plane.
    """
    pid_i = tl.program_id(axis=0)   # tile row (i)
    pid_j = tl.program_id(axis=1)   # tile column (j)
    pid_hd = tl.program_id(axis=2)  # combined batch & hidden index

    batch_id = pid_hd // hidden_dim
    d = pid_hd % hidden_dim

    # block offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # pointers for a (left) block: shape (M, K)
        left_ptrs = (
            left_ptr
            + batch_id * stride_l_b
            + i_offset[:, None] * stride_l_i
            + k_offset[None, :] * stride_l_k
            + d * stride_l_d
        )
        # pointers for b (right) block transposed: shape (K, N)
        right_ptrs = (
            right_ptr
            + batch_id * stride_r_b
            + k_offset[:, None] * stride_r_k
            + j_offset[None, :] * stride_r_j
            + d * stride_r_d
        )

        # guarded loads (other=0 ensures out‑of‑bounds contribute nothing)
        left_block = tl.load(
            left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0
        )
        right_block = tl.load(
            right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0
        )

        # convert to float32 for accumulation
        left_block = left_block.to(tl.float32)
        right_block = right_block.to(tl.float32)

        # GEMM accumulate: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left_block, right_block)

    # store the computed tile
    out_ptrs = (
        out_ptr
        + batch_id * stride_o_b
        + i_offset[:, None] * stride_o_i
        + j_offset[None, :] * stride_o_j
        + d * stride_o_d
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
