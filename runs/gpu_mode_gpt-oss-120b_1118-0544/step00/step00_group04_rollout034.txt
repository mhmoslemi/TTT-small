@triton.jit
def mask_mul_kernel(src_ptr, mask_ptr, dst_ptr,
                    B, N, H,
                    stride_src_b, stride_src_i, stride_src_k, stride_src_h,
                    stride_mask_b, stride_mask_i, stride_mask_k,

    i = rem // N

def custom_kernel(data):
   input_tensor, mask, weights, config = data

   dim = config["dim"]
   hidden_dim = config["hidden_dim"]
   nomask = config.get("nomask", True)

   norm_weight = weights["norm.weight"]
   norm_bias = weights["norm.bias"]
   left_proj_weight = weights["left_proj.weight"]
   right_proj_weight = weights["right_proj.weight"]
   left_gate_weight = weights["left_gate.weight"]
   right_gate_weight = weights["right_gate.weight"]
   out_gate_weight = weights["out_gate.weight"]
   to_out_norm_weight = weights["to_out_norm.weight"]
   to_out_norm_bias = weights["to_out_norm.bias"]
   to_out_weight = weights["to_out.weight"]

   # LayerNorm
   x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

   # Projections
   left = torch.nn.functional.linear(x_norm, left_proj_weight)  # no bias
   right = torch.nn.functional.linear(x_norm, right_proj_weight)

   if not nomask:
       # mask may be bool, convert to float
       mask_f = mask.to(dtype=left.dtype) if mask is not None else torch.ones_like(left[...,0])
       left = apply_mask_triton(left, mask_f)
       right = apply_mask_triton(right, mask_f)
   # Gates
   left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
   right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
   out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

   # Apply gates
   left = left * left_gate
   right = right * right_gate

   # Reduce via batched GEMM
   B, N, N2, H = left.shape
   left_perm = left.permute(0, 3, 1, 2).contiguous()
   right_perm = right.permute(0, 3, 1, 2).contiguous()
   left_flat = left_perm.view(B * H, N, N)
   right_flat = right_perm.view(B * H, N, N)
   out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

"""
custom_kernel for the TriMul (outgoing) update.

Algorithm
---------
1) Layer‑norm the input tensor `x` over the last dimension using the provided
   `norm.{weight,bias}`.
2) Project the normalized tensor into the hidden dimension with two linear
   layers (`left_proj`, `right_proj`), both without bias.
3) If a mask is supplied (i.e. config["nomask"] is False) multiply the two
   projected tensors by the mask.  This step is performed by a small Triton
   kernel that loads a full hidden‑dim slice for each (batch, i, k) location,
   multiplies by the scalar mask value and stores the result.
4) Compute three gate tensors by applying a sigmoid to three more linear
   transforms of the normalized input (`left_gate`, `right_gate`, `out_gate`).
5) Apply the gates element‑wise to the masked projections.
6) Reduce over the third sequence axis `k`.  The contraction
      out[...,i,j,d] = Σ_k left[...,i,k,d] * right[...,j,k,d]
   is equivalent to a batched matrix multiplication for each hidden channel:
      left_d @ right_dᵀ   (where d indexes the hidden dimension).
   We perform this as a single `torch.bmm` on a view of shape
   (B·H, N, N) → (B·H, N, N) → (B·H, N, N) and reshape back to
   (B, N, N, H).
7) Apply a second layer‑norm (`to_out_norm.{weight,bias}`) on the reduced
   tensor, multiply by the `out_gate` tensor, and finally linearly project
   back to the original channel dimension using `to_out.weight`.

The only non‑trivial operation off‑loaded to Triton is the mask‑multiplication
(step 3), which demonstrates a fused element‑wise kernel while the
computationally dominant O(N³) reduction is handled by cuBLAS via
`torch.bmm`, giving high performance on the B200.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: multiply a [B,N,N,H] tensor by a broadcasted mask [B,N,N]
# ----------------------------------------------------------------------
@triton.jit
def mask_mul_kernel(src_ptr, mask_ptr, dst_ptr,
                    B, N, H,
                    stride_src_b, stride_src_i, stride_src_k, stride_src_h,
                    stride_mask_b, stride_mask_i, stride_mask_k,
                    BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    k = rem % N

    # Load the scalar mask value for this (b,i,k) entry
    mask_offset = mask_ptr + b * stride_mask_b + i * stride_mask_i + k * stride_mask_k
    mask_val = tl.load(mask_offset)

    # Base pointers for source and destination slices (size H)
    src_base = src_ptr + b * stride_src_b + i * stride_src_i + k * stride_src_k
    dst_base = dst_ptr + b * stride_src_b + i * stride_src_i + k * stride_src_k

    offs = tl.arange(0, BLOCK_H)
    mask_h = offs < H

    src = tl.load(src_base + offs * stride_src_h, mask=mask_h, other=0.0)
    dst = src * mask_val
    tl.store(dst_base + offs * stride_src_h, dst, mask=mask_h)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Apply a broadcasted mask of shape [B,N,N] to a tensor of shape [B,N,N,H]
    using a Triton kernel. Returns a new tensor (same dtype/device).
    """
    B, N, N2, H = tensor.shape
    assert N == N2, "Tensor must be square in the two sequence dimensions."
    tensor = tensor.contiguous()
    mask = mask.contiguous().to(dtype=tensor.dtype)

    out = torch.empty_like(tensor)

    # Strides (in elements, not bytes)
    stride_src_b, stride_src_i, stride_src_k, stride_src_h = tensor.stride()
    stride_mask_b, stride_mask_i, stride_mask_k = mask.stride()

    grid = (B * N * N,)
    BLOCK_H = 128  # hidden_dim ≤ 128 in all provided configs
    mask_mul_kernel[grid](
        tensor,
        mask,
        out,
        B, N, H,
        stride_src_b, stride_src_i, stride_src_k, stride_src_h,
        stride_mask_b, stride_mask_i, stride_mask_k,
        BLOCK_H=BLOCK_H,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward pass (outgoing version).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be None)
        - weights      : dict of model parameters
        - config       : dict with keys "dim", "hidden_dim", and "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,H]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3) Optional mask (handled by Triton)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask can be bool/int; convert to same dtype as tensors
        mask_f = mask.to(dtype=left.dtype)
        left = apply_mask_triton(left, mask_f)
        right = apply_mask_triton(right, mask_f)

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Reduction over the third sequence axis (batched GEMM)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_flat = left_perm.view(B * H, N, N)            # [B*H, N, N]
    right_flat = right_perm.view(B * H, N, N)          # [B*H, N, N]

    # batched matrix multiplication: (B*H, N, N) @ (B*H, N, N).T  -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Final LayerNorm, out_gate, and output projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
