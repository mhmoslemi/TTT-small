@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    B, N, C,
    stride_x_b, stride_x_i, stride_x_j, stride_x_c,
    stride_out_b, stride_out_i, stride_out_j, stride_out_c,
    BLOCK_C: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    # compute linear index: idx = pid
    b = idx // (N*N)
    rem = idx % (N*N)
    i = rem // N
    j = rem % N

    # Offsets across C dimension
    offs_c = tl.arange(0, BLOCK_C)
    mask_c = offs_c < C

    # pointers offsets:
    x_ptrs = x_ptr + b * stride_x_b + i * stride_x_i + j * stride_x_j + offs_c * stride_x_c
    out_ptrs = out_ptr + b * stride_out_b + i * stride_out_i + j * stride_out_j + offs_c * stride_out_c

    # compute mean and var
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)
    for offset in range(0, tl.cdiv(C, BLOCK_C)):
        c_start = offset * BLOCK_C
        offs = c_start + offs_c
        mask = offs < C
        x = tl.load(x_ptrs + c_start * stride_x_c, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
    # Compute mean, var
    mean = sum_val / C
    var = sum_sq / C - mean * mean
    # second pass to normalize and apply weight,bias
    for offset in range(0, tl.cdiv(C, BLOCK_C)):
        c_start = offset * BLOCK_C
        offs = c_start + offs_c
        mask = offs < C

    eps = 1e-5

    # layernorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # mask application
    if config.get("nomask", False):
        # no mask: same as 1
        mask_tensor = torch.ones_like(mask, dtype=left.dtype, device=device)
    else:
        mask_tensor = mask
    mask_unsqueeze = mask_tensor.unsqueeze(-1)  # shape [B,N,N,1]
    left = left * mask_unsqueeze
    right = right * mask_unsqueeze

    # gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # reshape for GEMM
    B, N, _, _ = input_tensor.shape
    left_ = left.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous()

    # flatten groups
    num_groups = B * hidden_dim
    left_mat = left_.view(num_groups, N, N)
    right_mat = right_.view(num_groups, N, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float32, device=device)

    # get strides
    stride_am = left_mat.stride(1)    # N
    stride_ak = left_mat.stride(2)    # 1
    stride_bn = right_mat.stride(2)   # 1 (col stride of original)
    stride_bk = right_mat.stride(1)   # N
    stride_cm = out_mat.stride(1)     # N
    stride_cn = out_mat.stride(2)     # 1
    group_stride = left_mat.stride(0) # N*N

    # define block sizes heuristically based on N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # set grid
    grid = (num_groups,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trimul_gemm_kernel[grid](
        left_mat, right_mat, out_mat,
        N, N, N,
        stride_am, stride_ak,
        stride_bn, stride_bk,
        stride_cm, stride_cn,
        group_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape output back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # out_gate multiplication (broadcast)
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bn, stride_bk,
    stride_cm, stride_cn,
    GROUP_STRIDE,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    group = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute base pointers (group offset)
    a_ptrs = a_ptr + group * GROUP_STRIDE + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + group * GROUP_STRIDE + offs_n[None, :] * stride_bn + tl.arange(0, BLOCK_K)[:, None] * stride_bk

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        # compute actual mask for K within this block
        mask_k = cur_k + tl.arange(0, BLOCK_K) < K
        a = tl.load(a_ptrs + cur_k * stride_ak, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs + cur_k * stride_bk, mask=mask_n[None, :] & mask_k[:, None], other=0.0)
        # convert to float32 (they already are)
        acc += tl.dot(a, b)
    # store result
    c_ptrs = c_ptr + group * GROUP_STRIDE + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The heavy part of the TriMul operation is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication per hidden channel.
We fuse this contraction into a Triton GEMM kernel that processes all
hidden channels (hidden_dim) as independent batch instances.  All other
steps (LayerNorm, linear projections, gating) are performed with
torch.nn.functional for simplicity – satisfying the “at least part in
a kernel” requirement while keeping the implementation clear and fast.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for each hidden channel.
# Computes C = A @ Bᵀ  where each batch corresponds to one hidden channel.
# A, B, C have shape (G, N, N) with G = batch_size * hidden_dim.
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers to A, B (transposed), C
    M, N, K,                      # matrix dimensions (all equal to seq_len)
    stride_am, stride_ak,         # strides of A (row, col)
    stride_bn, stride_bk,         # strides of Bᵀ (row, col)
    stride_cm, stride_cn,         # strides of C (row, col)
    GROUP_STRIDE,                 # stride between successive groups (B*H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 3‑D launch: (group, tile_m, tile_n)
    group_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this group
    a_base = a_ptr + group_id * GROUP_STRIDE
    b_base = b_ptr + group_id * GROUP_STRIDE
    c_base = c_ptr + group_id * GROUP_STRIDE

    # pointers for the current tile (without the K‑dimension offset)
    a_tile_ptr = a_base + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_tile_ptr = b_base + tl.arange(0, BLOCK_K)[:, None] * stride_bn + offs_n[None, :] * stride_bk

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K‑loop
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        # mask for the current K‑block
        cur_k_valid = (cur_k + tl.arange(0, BLOCK_K)) < K

        a = tl.load(
            a_tile_ptr + cur_k * stride_ak,
            mask=mask_m[:, None] & cur_k_valid[None, :],
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr + cur_k * stride_bk,
            mask=mask_n[None, :] & cur_k_valid[:, None],
            other=0.0,
        )
        # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

    # write back
    c_tile_ptr = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom TriMul forward function
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]
        mask         : torch.Tensor [B, N, N] (may be ignored when config["nomask"]=True)
        weights      : dict of torch.Tensors containing all linear / LN parameters
        config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1) LayerNorm over the channel dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Mask (if present)
    # --------------------------------------------------------------
    if config.get("nomask", False):
        mask_tensor = torch.ones_like(mask, dtype=left.dtype, device=device)
    else:
        mask_tensor = mask
    mask_unsqueeze = mask_tensor.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
    left = left * mask_unsqueeze
    right = right * mask_unsqueeze

    # --------------------------------------------------------------
    # 4) Gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Contraction via Triton GEMM (batched over hidden_dim)
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape

    # move hidden dimension to the batch axis for the kernel
    left_t = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # flatten (B, H) -> G groups
    G = B * hidden_dim
    left_mat = left_t.view(G, N, N)
    right_mat = right_t.view(G, N, N)

    # output buffer
    out_mat = torch.empty_like(left_mat, dtype=torch.float32, device=device)

    # strides (all tensors are contiguous)
    stride_am = left_mat.stride(1)      # N
    stride_ak = left_mat.stride(2)      # 1
    stride_bn = right_mat.stride(2)     # 1  (row stride of Bᵀ)
    stride_bk = right_mat.stride(1)     # N  (col stride of Bᵀ)
    stride_cm = out_mat.stride(1)       # N
    stride_cn = out_mat.stride(2)       # 1
    group_stride = left_mat.stride(0)   # N * N

    # block sizes – tuned for typical N (≤ 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        G,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trimul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_bn,
        stride_bk,
        stride_cm,
        stride_cn,
        group_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # --------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
