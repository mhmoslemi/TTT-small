def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_idx, # Not needed as pointer offset.
    M, N, K,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program ID across grid = pid (0th dimension)
    # It enumerates block indices across M and N.
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    # Also have batch dimension: we will launch kernel with separate grid for each batch (num_blocks_batch). Or combine via program id in 1D.
    # Simpler: We can launch kernel per batch (running many blocks across M,N) using grid = (num_blocks_m * num_blocks_n, batch_size)
    # But Triton doesn't directly support 2D grid (pid axis up to 3). We can use axis=0 for block index, axis=1 for batch (tidy).
    # So we will use program_id(0) for blockM_N tile, program_id(1) for batch (i.e., batch index).
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1) # batch index (0 <= pid1 < batch)

    block_m = pid0 // num_blocks_n
    block_n = pid0 % num_blocks_n

    # compute offsets
    # row offsets for A: batch start + block_m * BLOCK_M + arange
    # col offsets for B: batch start + block_n * BLOCK_N + arange
    # k offset loops

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # pointer offsets with strides
    a_ptrs = A_ptr + pid1 * stride_batch_a + offs_m[:, None] * stride_m_a + offs_k[None, :] * stride_k_a
    b_ptrs = B_ptr + pid1 * stride_batch_b + offs_k[:, None] * stride_k_b + offs_n[None, :] * stride_n_b

    # Use accumulator in float32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):

        # advance pointers
        a_ptrs += BLOCK_K * stride_k_a
        b_ptrs += BLOCK_K * stride_k_b

    # write to C
    c_ptrs = C_ptr + pid1 * stride_batch_c + offs_m[:, None] * stride_m_c + offs_n[None, :] * stride_n_c
    tl.store(c_ptrs, acc.to(tl.float16) if C_ptr.dtype == tl.float16 else acc, mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    eps = 1e-5

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), norm_weight, norm_bias, eps=eps)

    # Cast to half
    x_fp16 = x.half()

    # Linear projections: prepare fp16 weights
    left_proj_w = weights["left_proj.weight"].to(x_fp16.dtype)
    right_proj_w = weights["right_proj.weight"].to(x_fp16.dtype)

    left = torch.nn.functional.linear(x_fp16, left_proj_w)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_fp16, right_proj_w)

    # mask: shape [B,N,N] maybe bool or float. Convert to same dtype
    if mask is None:
        mask_fp16 = torch.ones_like(left[..., :1], dtype=x_fp16.dtype)
    else:
        mask_fp16 = mask.unsqueeze(-1).to(x_fp16.dtype)  # broadcast
        # treat mask values 0/1.

    left = left * mask_fp16
    right = right * mask_fp16

    # Gates
    left_gate_w = weights["left_gate.weight"].to(x_fp16.dtype)
    right_gate_w = weights["right_gate.weight"].to(x_fp16.dtype)
    out_gate_w = weights["out_gate.weight"].to(x_fp16.dtype)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # Prepare for GEMM: reshape to (B*hidden_dim, N, N) and transpose right for multiplication
    B, N, _, _ = left.shape
    left_ = left.permute(0, 3, 1, 2).contiguous()  # B, hidden_dim, N, N
    right_ = right.permute(0, 3, 1, 2).contiguous()
    # transpose right_ on last two dims
    right_T = right_.permute(0, 1, 3, 2).contiguous()

    # flatten batch*hidden_dim

def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,

    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,

    pid_batch = tl.program_id(1)

    # compute block indices
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid_mn // num_blocks_n
    block_n = pid_mn % num_blocks_n

    # compute offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # pointer offsets with strides
    a_ptrs = A_ptr + pid_batch * stride_batch_a + offs_m[:, None] * stride_m_a + offs_k[None, :] * stride_k_a
    b_ptrs = B_ptr + pid_batch * stride_batch_b + offs_k[:, None] * stride_k_b + offs_n[None, :] * stride_n_b

Triton kernel that works on FP16 inputs and accumulates in FP32.
The final result is cast back to FP32 for the remaining layers.

    M, N, K,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Computes C = A @ Bᵀ  for many independent batches (B·D).

    A : (batch, M, K)   (FP16)
    B : (batch, N, K)   (FP16)   – note we pass B already transposed so that
                                 the K‑dimension is the contiguous one.
    C : (batch, M, N)   (FP16, stored as FP16 from FP32 accumulator)
    """
    pid_mn = tl.program_id(0)   # tile index for (M,N)
    pid_batch = tl.program_id(1)  # which batch (including hidden dimension)

    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid_mn // num_blocks_n
    block_n = pid_mn % num_blocks_n

    # Offsets of the current tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers for the current tile
    a_ptrs = A_ptr + pid_batch * stride_batch_a \
                     + offs_m[:, None] * stride_m_a \
                     + offs_k[None, :] * stride_k_a
    b_ptrs = B_ptr + pid_batch * stride_batch_b \
                     + offs_k[:, None] * stride_k_b \
                     + offs_n[None, :] * stride_n_b

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k_tiles):
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0, dtype=tl.float16)
        acc += tl.dot(a, b)               # dot does FP16×FP16 → FP32 accumulation

        # advance K dimension
        a_ptrs += BLOCK_K * stride_k_a
        b_ptrs += BLOCK_K * stride_k_b

    # Write result back as FP16
    c_ptrs = C_ptr + pid_batch * stride_batch_c \
                     + offs_m[:, None] * stride_m_c \
                     + offs_n[None, :] * stride_n_c
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), norm_weight, norm_bias, eps=eps
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy path
    # ------------------------------------------------------------------
    x_h = x.half()   # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(
        x_h, weights["left_proj.weight"].to(x_h.dtype)
    )  # [B, N, N, hidden_dim] fp16
    right = torch.nn.functional.linear(
        x_h, weights["right_proj.weight"].to(x_h.dtype)
    )  # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 4. Mask (if present)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_h = mask_tensor.unsqueeze(-1).to(x_h.dtype)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x_h, weights["left_gate.weight"].to(x_h.dtype)
        )
    )  # [B, N, N, hidden_dim] fp16
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x_h, weights["right_gate.weight"].to(x_h.dtype)
        )
    )  # [B, N, N, hidden_dim] fp16
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x_h, weights["out_gate.weight"].to(x_h.dtype)
        )
    )  # [B, N, N, hidden_dim] fp16

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Prepare tensors for batched GEMM: treat each (batch, hidden_dim)
    #    as an independent matrix multiplication.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # shape -> (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # transpose the second N dim of right to get (B, hidden_dim, N, K) where K=N
    right_t = right_perm.permute(0, 1, 3, 2).contiguous()

    # flatten batch*hidden_dim
    batch_hidden = B * hidden_dim
    left_view = left_perm.view(batch_hidden, N, N)       # (BH, M, K)
    right_view = right_t.view(batch_hidden, N, N)       # (BH, N, K) where K=N

    # output buffer (fp16, will be cast back)
    out_view = torch.empty_like(left_view)

    # ------------------------------------------------------------------
    # 7. Launch Triton kernel for C = A @ Bᵀ  (A: (BH,M,K), B: (BH,N,K))
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_x = ((N + BLOCK_M - 1) // BLOCK_M) * ((N + BLOCK_N - 1) // BLOCK_N)
    grid = (grid_x, batch_hidden)   # (tiles over M,N), (batch dimension)

    # strides (in elements, not bytes)
    sba = left_view.stride(0)
    sma = left_view.stride(1)
    sk_a = left_view.stride(2)

    sbb = right_view.stride(0)
    sk_b = right_view.stride(1)   # K‑dimension stride
    sn_b = right_view.stride(2)

    sbc = out_view.stride(0)
    sm_c = out_view.stride(1)
    sn_c = out_view.stride(2)

    _batched_matmul_kernel[grid](
        left_view,
        right_view,
        out_view,
        M=N,
        N=N,
        K=N,
        stride_batch_a=sba,
        stride_m_a=sma,
        stride_k_a=sk_a,
        stride_batch_b=sbb,
        stride_k_b=sk_b,
        stride_n_b=sn_b,
        stride_batch_c=sbc,
        stride_m_c=sm_c,
        stride_n_c=sn_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 8. Reshape back to (B, N, N, hidden_dim) and cast to FP32
    # ------------------------------------------------------------------
    out = (
        out_view.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .float()
    )  # [B, N, N, hidden_dim] FP32

    # ------------------------------------------------------------------
    # 9. Output LayerNorm on hidden_dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 10. Apply the outgoing gate (cast to float32)
    # ------------------------------------------------------------------
    out = out * out_gate.float()

    # ------------------------------------------------------------------
    # 11. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
