    x, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # some maybe config includes "nomask" etc.
    # copy to device if needed? Already on device.

    # Normalization
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Linear projections
    left_proj = F.linear(x_norm, weights["left_proj.weight"])
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    # mask application (if not nomask)
    if mask is not None and not config.get("nomask", False):
         mask = mask.to(x_norm.dtype).unsqueeze(-1)  # shape [B,N,N,1]
         left_proj = left_proj * mask
         right_proj = right_proj * mask

    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Convert to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute and reshape for kernel
    B, N, _, _ = left_fp16.shape
    # left shape [B,N,N,hidden_dim]
    left_mat = left_fp16.permute(0,3,1,2).contiguous()  # [B, hidden_dim, N, N]
    right_mat = right_fp16.permute(0,3,1,2).contiguous()

    BH = B * hidden_dim
    left_mat = left_mat.view(BH, N, N)
    right_mat = right_mat.view(BH, N, N)

    out_mat = torch.empty((BH, N, N), device=x.device, dtype=torch.float32)

    # Determine block sizes based on N
    # Choose 64 for M,N; and K block same.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # If N is small maybe adjust but fine.

    # Compute strides (elements)
    stride_a_batch = N * N
    stride_a_row = N
    stride_a_col = 1

    stride_b_batch = N * N
    stride_b_row = N   # row stride (original i dimension)
    stride_b_col = 1   # column stride (k dimension)

    stride_c_batch = N * N
    stride_c_row = N
    stride_c_col = 1

    # Launch kernel

"""
TriMul outgoing variant (AlphaFold3) with a Triton kernel for the
contraction:

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

All LayerNorms, linear projections and gating are performed with
PyTorch; the expensive O(N³) einsum is replaced by a batched GEMM
kernel that computes A @ Bᵀ for each hidden‑dim slice.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched multiplication C = A @ Bᵀ.
    A and B have shape (batch, M, K) and (batch, N, K) respectively.
    The result C has shape (batch, M, N).
    """
    pid_batch = tl.program_id(0)   # which (batch * hidden_dim) matrix
    pid_m = tl.program_id(1)       # tile over rows   (M)
    pid_n = tl.program_id(2)       # tile over cols   (N)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K in BLOCK_K chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A block: shape (BLOCK_M, BLOCK_K)
        a_offs = a_batch \
                 + offs_m[:, None] * stride_a_row \
                 + cur_k[None, :] * stride_a_col
        a = tl.load(a_offs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B block transposed on the fly: we need Bᵀ of shape (BLOCK_K, BLOCK_N)
        # Original B layout is (N, K); Bᵀ is accessed as (K, N)
        b_offs = b_batch \
                 + cur_k[:, None] * stride_b_col \
                 + offs_n[None, :] * stride_b_row
        b = tl.load(b_offs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # write result
    c_offs = c_batch \
             + offs_m[:, None] * stride_c_row \
             + offs_n[None, :] * stride_c_col
    tl.store(c_offs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (tensors on same device)
            - config: dict containing at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm ----------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ---------- Mask (if any) ----------
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ---------- Apply gates ----------
    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare tensors for the Triton GEMM ----------
    # Cast to half to reduce memory traffic (accumulation stays fp32)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Permute to [B, hidden_dim, N, N] and flatten batch*hidden_dim
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (float32, accumulator type)
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # ----------- Kernel launch parameters -----------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides in number of elements (not bytes)
    stride_a_batch = N * N
    stride_a_row = N
    stride_a_col = 1

    stride_b_batch = N * N
    stride_b_row = N          # original B row stride (i dimension)
    stride_b_col = 1          # original B column stride (k dimension)

    stride_c_batch = N * N
    stride_c_row = N
    stride_c_col = 1

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # ---------- Reshape back to original 4‑D layout ----------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,hidden_dim]

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
