def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_i = tl.program_id(0)  # i index
    pid_j = tl.program_id(1)  # j index
    pid_z = tl.program_id(2)  # batch* hidden block index

    # compute batch index and hidden block start
    blocks_per_batch = tl.cdiv(H, BLOCK_H)
    batch = pid_z // blocks_per_batch
    h_block = pid_z % blocks_per_batch
    h_start = h_block * BLOCK_H

    # bounds
    i = pid_i
    j = pid_j
    i_valid = i < N
    j_valid = j < N

    # compute hidden valid mask
    h_offsets = tl.arange(0, BLOCK_H)
    h_valid = (h_start + h_offsets) < H

    # accumulator
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    # loop over k
    for k in range(0, N, BLOCK_K):
        # compute end of block
        cur_k = tl.minimum(k + BLOCK_K, N)
        # iterate over each kk in [k, cur_k)
        for kk in range(k, cur_k):
            kk_valid = kk < N
            # compute pointers for loading a block of h values
            left_ptrs = left_ptr + batch * stride_left_b + i * stride_left_i + kk * stride_left_j + (h_start + h_offsets) * stride_left_h
            right_ptrs = right_ptr + batch * stride_right_b + j * stride_right_i + kk * stride_right_j + (h_start + h_offsets) * stride_right_h

            right = tl.load(right_ptrs, mask = j_valid & kk_valid & h_valid, other=0.0)

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_i = tl.program_id(0)  # i index
    pid_j = tl.program_id(1)  # j index
    pid_z = tl.program_id(2)  # combined batch and hidden block

    # Determine batch idx and hidden block offset
    num_h_blocks = (H + BLOCK_H - 1) // BLOCK_H
    batch = pid_z // num_h_blocks
    h_block = pid_z % num_h_blocks
    h_start = h_block * BLOCK_H

    i = pid_i
    j = pid_j

    i_valid = i < N
    j_valid = j < N

    h_offsets = tl.arange(0, BLOCK_H)
    h_valid = (h_start + h_offsets) < H

    # accumulator per hidden channel
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    for k in range(0, N, BLOCK_K):
        # iterate over each element within the k block
        for kk_offset in range(0, BLOCK_K):
            kk = k + kk_offset
            kk_valid = kk < N

            # compute pointers
            left_ptrs = left_ptr + batch * stride_left_b + i * stride_left_i + kk * stride_left_j + (h_start + h_offsets) * stride_left_h
            right_ptrs = right_ptr + batch * stride_right_b + j * stride_right_i + kk * stride_right_j + (h_start + h_offsets) * stride_right_h

            left = tl.load(left_ptrs, mask = i_valid & kk_valid & h_valid, other = 0.0)
            right = tl.load(right_ptrs, mask = j_valid & kk_valid & h_valid, other = 0.0)

            acc += left * right

    # store result
    out_ptrs = out_ptr + batch * stride_out_b + i * stride_out_i + j * stride_out_j + (h_start + h_offsets) * stride_out_h

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    eps = 1e-5

    # layernorm on dim
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps
    )

    # linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # mask
    if not config.get("nomask", False) and mask_tensor is not None:
        mask = mask_tensor.to(dtype=x_norm.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # contraction via Triton
    B, N, _, _ = left.shape
    out = torch.empty_like(left, dtype=left.dtype, device=device)  # shape [B,N,N,hidden_dim]

    # kernel launch
    BLOCK_K = 32
    BLOCK_H = 8  # choose 8, handle hidden_dim not divisible by 8 via mask

    num_h_blocks = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    grid = (N, N, B * num_h_blocks)

    left_stride = left.stride()
    right_stride = right.stride()
    out_stride = out.stride()

    tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        left_stride[0], left_stride[1], left_stride[2], left_stride[3],
        right_stride[0], right_stride[1], right_stride[2], right_stride[3],
        out_stride[0], out_stride[1], out_stride[2], out_stride[3],
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # apply to_out_norm layernorm over hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps
    )

    # multiply by out_gate (elementwise)
    # Use a Triton kernel for elementwise multiplication to satisfy requirement:
    # but we could also do torch multiplication. Since we already used Triton for contraction,
    # maybe not needed. We'll just use torch pointwise for simplicity.

    out = out * out_gate

    # final linear
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output

"""
TriMul "outgoing" update implemented with a custom Triton kernel.

The heavy operation is the 4‑D contraction:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
where `left` and `right` have shape (B, N, N, H) (H = hidden_dim).
The kernel assigns one thread to a specific output element (i, j) together
with a small block of hidden channels (BLOCK_H).  Each thread iterates over
the k‑dimension in blocks of size BLOCK_K, loads a vector of length BLOCK_H
from `left` and `right`, multiplies element‑wise and accumulates the result.
After the reduction the thread writes the H‑block of the output.

All other tensor operations (LayerNorm, linear projections, gates) are
performed with regular PyTorch kernels; only the O(N³·H) contraction is
offloaded to Triton.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, H,                                      # tensor sizes
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    num_h_blocks,                                 # H split into blocks
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr, # compile‑time constants
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_i = tl.program_id(0)      # i index
    pid_j = tl.program_id(1)      # j index
    pid_z = tl.program_id(2)      # combined batch & hidden‑block index

    # -------------------------------------------------------------
    # Decode batch and hidden‑block from pid_z
    batch = pid_z // num_h_blocks
    h_block = pid_z % num_h_blocks
    h_start = h_block * BLOCK_H                     # first hidden index for this thread

    # -------------------------------------------------------------
    i = pid_i
    j = pid_j

    i_ok = i < N
    j_ok = j < N

    h_offsets = tl.arange(0, BLOCK_H)
    h_ok = (h_start + h_offsets) < H

    # accumulator for this (i,j) element over a BLOCK_H slice
    acc = tl.zeros([BLOCK_H], dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over k in blocks of size BLOCK_K
    for k in range(0, N, BLOCK_K):
        # inner loop over each element inside the block (unrolled)
        for kk in range(0, BLOCK_K):
            cur_k = k + kk
            k_ok = cur_k < N

            # pointers to left[b,i,cur_k,:] and right[b,j,cur_k,:]
            left_ptrs = left_ptr + batch * stride_left_b \
                        + i * stride_left_i \
                        + cur_k * stride_left_j \
                        + (h_start + h_offsets) * stride_left_h
            right_ptrs = right_ptr + batch * stride_right_b \
                        + j * stride_right_i \
                        + cur_k * stride_right_j \
                        + (h_start + h_offsets) * stride_right_h

            left = tl.load(left_ptrs,
                           mask = i_ok & k_ok & h_ok,
                           other = 0.0)
            right = tl.load(right_ptrs,
                            mask = j_ok & k_ok & h_ok,
                            other = 0.0)

            acc += left * right

    # -------------------------------------------------------------
    # Write the computed BLOCK_H slice back to memory
    out_ptrs = out_ptr + batch * stride_out_b \
               + i * stride_out_i \
               + j * stride_out_j \
               + (h_start + h_offsets) * stride_out_h
    tl.store(out_ptrs, acc, mask = i_ok & j_ok & h_ok)


def custom_kernel(data):
    """
    Forward pass of the TriMul "outgoing" module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    if not config.get("nomask", False) and mask_tensor is not None:
        mask = mask_tensor.to(dtype=x_norm.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid after a linear)
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Contraction via Triton kernel (out = Σ_k left * right)
    B, N, _, _ = left.shape
    out = torch.empty_like(left)       # shape [B, N, N, hidden_dim]

    # Kernel launch parameters
    BLOCK_K = 32
    BLOCK_H = 8                        # can be tuned; must divide hidden_dim via masking
    num_h_blocks = (hidden_dim + BLOCK_H - 1) // BLOCK_H

    grid = (N, N, B * num_h_blocks)    # (i, j, batch·hblock)

    # strides for pointer arithmetic
    l_str = left.stride()
    r_str = right.stride()
    o_str = out.stride()

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        l_str[0],
        l_str[1],
        l_str[2],
        l_str[3],
        r_str[0],
        r_str[1],
        r_str[2],
        r_str[3],
        o_str[0],
        o_str[1],
        o_str[2],
        o_str[3],
        num_h_blocks,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 6. Output LayerNorm over hidden_dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # 7. Multiply by the output gate (element‑wise)
    out = out * out_gate

    # 8. Final linear projection back to dim
    output = F.linear(out, weights["to_out.weight"])

    return output
