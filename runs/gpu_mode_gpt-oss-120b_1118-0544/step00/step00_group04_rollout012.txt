def batched_gemm_kernel(A, B, C,
                        M, N, K,
                        stride_am, stride_ak,
                        stride_bk, stride_bn,
                        stride_cm, stride_cn,
                        batch_stride_a, batch_stride_b, batch_stride_c,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
   pid_batch = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # compute offsets

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = config.get("eps", 1e-5)
    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            normalized_shape=(dim,),
                                            weight=weights["norm.weight"],
                                            bias=weights["norm.bias"],
                                            eps=eps)

    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask (if any)
    if not nomask and mask is not None:
        mask_ = mask.to(input_tensor.dtype).unsqueeze(-1)
        left = left * mask_
        right = right * mask_
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Prepare for reduction
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # B, H, N, N
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # B, H, N, N (transposed)

    B, H, N, _ = left_perm.shape

    # Flatten batch and head
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Output placeholder
    out_flat = torch.empty_like(left_flat)  # same dtype and device

    # Compute strides (in elements)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    batch_stride_a = left_flat.stride(0)

    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)
    batch_stride_b = right_flat.stride(0)

    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    batch_stride_c = out_flat.stride(0)

    M = N
    K = N
    # Launch kernel
    import triton
    import triton.language as tl

    @triton.jit
    def batched_gemm_kernel(A, B, C,
                            M, N, K,
                            stride_am, stride_ak,
                            stride_bk, stride_bn,
                            stride_cm, stride_cn,
                            batch_stride_a, batch_stride_b, batch_stride_c,
                            BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
        pid_batch = tl.program_id(0)
        pid_m = tl.program_id(1)
        pid_n = tl.program_id(2)

        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        mask_m = offs_m < M
        mask_n = offs_n < N

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k in range(0, K, BLOCK_K):
            offs_k = k + tl.arange(0, BLOCK_K)
            mask_k = offs_k < K

            a_ptrs = A + pid_batch * batch_stride_a + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
            b_ptrs = B + pid_batch * batch_stride_b + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

            a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
            b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

            acc += tl.dot(a, b)

        c_ptrs = C + pid_batch * batch_stride_c + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
        tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    # Choose block dimensions
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H, (M + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape output back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,H)

    # Apply to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps
    )

    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul forward (outgoing) implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B, N, N, C_in]).
2. Linear projections → `left` and `right` (shape [B, N, N, H]).
3. Optional masking (broadcast on the last dim) and per‑position sigmoid gates.
4. Reduce over the sequence dimension `k`:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication `left @ rightᵀ` for every head `d`.
   The reduction is performed in a Triton kernel (`batched_gemm_kernel`).
5. Layer‑norm the reduced tensor (over the hidden dimension `H`), multiply
   with `out_gate` (computed from the normalized input) and apply the final
   linear projection back to the original channel dimension `C_in`.

Only the heavy reduction (step 4) is executed on the GPU with Triton; all
other operations use PyTorch for simplicity and correctness.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B) where A, B, C have shape
#   (batch, M, K), (batch, K, N), (batch, M, N) respectively.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)          # which matrix in the batch
    pid_m = tl.program_id(1)              # block row
    pid_n = tl.program_id(2)              # block col

    # --- compute offsets for the current tile --------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --- main k‑loop ---------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers with batch offset
        a_ptrs = A + pid_batch * batch_stride_a + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = B + pid_batch * batch_stride_b + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # matrix multiply for the tile
        acc += tl.dot(a, b)

    # --- write back -----------------------------------------------------------
    c_ptrs = C + pid_batch * batch_stride_c + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the "outgoing" TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C_in] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool or float) – may be None
            - weights_dict : mapping from weight name to torch.Tensor
            - config_dict  : contains at least "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C_in] (float32)
    """
    x, mask, weights, config = data

    B, N, _, C_in = x.shape
    dim = config["dim"]               # equals C_in
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (same as nn.LayerNorm(dim))
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])    # -> [B, N, N, H]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (applied after projection, before gating)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f = mask.to(x.dtype).unsqueeze(-1)            # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Prepare tensors for the reduction (B x H independent matmuls)
    #    left  : [B, N, N, H] -> [B, H, N, N]
    #    right : [B, N, N, H] -> [B, H, N, N] (transposed on the last two axes)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()       # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()     # [B, H, N, N] (k‑axis transposed)

    # Flatten batch & head dimensions for the kernel
    left_flat = left_perm.view(B * hidden_dim, N, N)        # (B*H, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)      # (B*H, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # 6) Compute strides in element units (required by Triton)
    # ------------------------------------------------------------------
    stride_am = left_flat.stride(1)       # stride between rows (M) -> K
    stride_ak = left_flat.stride(2)       # stride between cols (K) -> 1
    batch_stride_a = left_flat.stride(0)

    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)
    batch_stride_b = right_flat.stride(0)

    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    batch_stride_c = out_flat.stride(0)

    M = N
    K = N
    # ------------------------------------------------------------------
    # 7) Launch Triton kernel (batched GEMM)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # batch * heads
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                       # typical for fp32 GEMM
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 9) Post‑reduction LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
