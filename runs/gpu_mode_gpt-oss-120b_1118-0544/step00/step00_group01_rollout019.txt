    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(dim=-1, keepdim=True)) / torch.sqrt(input_tensor.var(dim=-1, unbiased=False, keepdim=True) + eps)

    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

        right = right * mask

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_dim, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # pid dims
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # batch offset
    batch_offset = pid_batch * stride_a_batch

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # compute offsets for A
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        a_ptrs = a_ptr + batch_offset + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=(offs_m < M) & (offs_k < K), other=0.0)
        a = a.to(tl.float32)

        # compute offsets for B
        # B is stored transposed, dimensions: (batch, K, N)
        b_ptrs = b_ptr + batch_offset + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        b = b.to(tl.float32)

    # store result

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

Algorithm:
1. Layer‑norm the input tensor over the last dimension.
2. Linear projections (left/right) and gated linear layers (left_gate,
   right_gate, out_gate) are applied with a sigmoid non‑linearity.
3. Optional pairwise mask is broadcasted and applied to left/right.
4. The core “tri‑mul” operation:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is performed as a batched matrix multiplication using a fused Triton
   kernel (each (batch × hidden) slice is a separate GEMM of size N×N).
5. Layer‑norm over the hidden dimension, gating with out_gate,
   and a final linear projection back to the original channel dimension.
All heavy compute (step 4) is done in the Triton kernel; the surrounding
operations use PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_dim, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   where A∈(batch, M, K), B∈(batch, K, N)."""
    pid_batch = tl.program_id(0)   # which (batch*hidden) matrix
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile col

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Global batch offset
    batch_off = pid_batch * stride_a_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers for A (batch, M, K)
        a_ptrs = a_ptr + batch_off + offs_m[:, None] * stride_a_m + k_off[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (k_off[None, :] < K),
                    other=0.0).to(tl.float32)

        # Pointers for B (batch, K, N) – note that B is already transposed
        b_ptrs = b_ptr + batch_off + k_off[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(k_off[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = c_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(c_ptr.dtype),
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the tri‑multiplicative update using the Triton kernel.
    Input tensors have shape [B, N, N, H] (float32/float16 on CUDA).
    Returns a tensor of the same shape.
    """
    assert left.is_cuda and right.is_cuda, "Triton kernel requires CUDA tensors"
    B, N, _, H = left.shape

    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # transpose right for A @ B.T pattern
    right_t = right_flat.transpose(1, 2)   # shape (B*H, N, N) with swapped strides

    out = torch.empty_like(left_flat)

    # Tune tile sizes – 64 is a good default for fp32 on most GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left_flat,
        right_t,
        out,
        B * H, N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_t.stride(0), right_t.stride(1), right_t.stride(2),
        out.stride(0), out.stride(1), out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    data = (input_tensor, mask, weights, config)
    Returns tensor of shape [bs, seq_len, seq_len, dim].
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm over channels (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(dtype)  # shape [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gated scalars
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core tri‑multiplicative update (batched GEMM via Triton)
    # ------------------------------------------------------------------
    out = tri_mul_triton(left, right)          # shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gating and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
