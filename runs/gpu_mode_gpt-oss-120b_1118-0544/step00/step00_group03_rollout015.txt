```
@triton.jit
def apply_gate_mask_kernel(left_ptr, right_ptr,
                           left_gate_ptr, right_gate_ptr,
                           mask_ptr,
                           hidden_dim: tl.constexpr,
                           numel: tl.constexpr,
                           BLOCK_SIZE: tl.constexpr,
                           HAS_MASK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_g = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_g = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    left = left * left_g
    right = right * right_g
    if HAS_MASK:
        mask_idx = offs // hidden_dim
        mask_v = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0) # broadcast mask else 1
        left = left * mask_v
        right = right * mask_v
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype  # assume float32

    # unpack weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]  # shape [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm: across dim dimension
    # Using torch.nn.functional.layer_norm
    # shape for normalized: [B,N,N,dim] means normalized over last dim.
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            normalized_shape=(dim,),
                                            weight=norm_weight,
                                            bias=norm_bias,
                                            eps=1e-5)

    # Linear projections
    # using torch.nn.functional.linear for efficiency
    # input shape: (..., dim)
    # weight shape: (out_features, in_features)
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Gate projections
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # Apply mask and gates via Triton kernel
    # if mask is None (nomask=True), we set mask = torch.ones([...], dtype=x_norm.dtype, device=device)
    if mask is None:
        HAS_MASK = False
    else:
        HAS_MASK = True
        # ensure mask dtype float (1 for true, 0 for false)
        mask = mask.to(dtype=dtype, device=device) # mask broadcast shape [B,N,N]
    # ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    # flatten
    left_flat = left.view(-1)
    right_flat = right.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)

    if HAS_MASK:
        mask_flat = mask.view(-1).contiguous()
    else:
        mask_flat = torch.empty(0, device=device, dtype=dtype)  # placeholder

    # launch Triton kernel
    numel = left_flat.numel()
    BLOCK_SIZE = 1024
    grid = lambda meta: ( (numel + meta['BLOCK_SIZE'] - 1) // meta['BLOCK_SIZE'], )
    apply_gate_mask_kernel[grid](left_flat, right_flat,
                                 left_gate_flat, right_gate_flat,
                                 mask_flat,
                                 hidden_dim,
                                 numel,
                                 BLOCK_SIZE=BLOCK_SIZE,
                                 HAS_MASK=HAS_MASK)

    # Now left and right have been updated
    # Compute out using batched matmul
    B, N, _, _ = left.shape  # left shape still [B,N,N,hidden_dim]
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2)  # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 1, 2)
    # reshape for batched matmul
    left_mat = left_perm.reshape(B * hidden_dim, N, N)
    right_mat = right_perm.reshape(B * hidden_dim, N, N)
    # matmul: left_mat @ right_mat.transpose(-2,-1)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # shape (B*hidden_dim, N, N)
    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,hidden_dim)

    # Apply to_out_norm (LayerNorm over hidden_dim)
    out = torch.nn.functional.layer_norm(out,
                                         normalized_shape=(hidden_dim,),
                                         weight=to_out_norm_weight,
                                         bias=to_out_norm_bias,
                                         eps=1e-5)
    # Apply out_gate (broadcast)
    out = out * out_gate

    # Linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)  # shape (B,N,N,dim)

    return out
```

import torch
import triton
import triton.language as tl

@triton.jit
def apply_gate_mask_kernel(left_ptr, right_ptr,
                           left_gate_ptr, right_gate_ptr,
                           mask_ptr,
                           hidden_dim: tl.constexpr,
                           numel,
                           BLOCK_SIZE: tl.constexpr,
                           HAS_MASK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    # mask for out-of-bounds
    mask = offs < numel

    # load values
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_g = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_g = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    left = left * left_g
    right = right * right_g

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # if out-of-range treat as 1
        left = left * mask_val
        right = right * mask_val

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

"""
custom_kernel for the TriMul “outgoing” update.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension (`dim`).
2. Linear projections to the hidden dimension for `left` and `right`.
3. Compute three sigmoid gates from the same normalized input:
   * left_gate, right_gate (apply to the projected tensors)
   * out_gate   (applied after the core update)
4. Fuse the element‑wise multiplication of the projections with the
   corresponding gates and the optional pair‑wise mask using a tiny
   Triton kernel (`apply_gate_mask_kernel`).  This kernel runs on a
   flattened view of the tensors and therefore avoids extra Python
   loops.
5. The core “TriMul” operation is
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is exactly a batch matrix‑multiply L·Rᵀ for every hidden
   channel `d`.  We reshape the tensors to (B·hidden_dim, N, N) and
   perform a batched GEMM (`torch.bmm`), which is heavily optimized
   on the GPU.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and finally linearly project back to the original feature dimension
   (`dim`).

The only part that is explicitly written in Triton is step 4; all
other heavy linear algebra is delegated to PyTorch’s cuBLAS‑backed
implementations for optimal speed.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse gate multiplication and (optional) mask application
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    left_ptr,               # *float32, [numel] – flattened left tensor
    right_ptr,              # *float32, [numel] – flattened right tensor
    left_gate_ptr,          # *float32, [numel] – flattened left_gate
    right_gate_ptr,         # *float32, [numel] – flattened right_gate
    mask_ptr,               # *float32, [mask_numel] – flattened mask (if any)
    hidden_dim: tl.constexpr,
    numel: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < numel

    # load values (out‑of‑bounds loads are masked)
    l = tl.load(left_ptr + offs, mask=valid, other=0.0)
    r = tl.load(right_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # multiply by gates
    l = l * lg
    r = r * rg

    # optional mask (broadcast over the hidden dimension)
    if HAS_MASK:
        # each group of `hidden_dim` consecutive elements belongs to the same (b,i,j)
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)
        l = l * m
        r = r * m

    # store the results back in‑place
    tl.store(left_ptr + offs, l, mask=valid)
    tl.store(right_ptr + offs, r, mask=valid)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              - input_tensor: torch.Tensor of shape [B, N, N, dim]
              - mask_tensor:  torch.Tensor of shape [B, N, N] (may be ignored)
              - weights_dict: mapping from weight name to torch.Tensor
              - config_dict: contains "dim", "hidden_dim", and "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False)   # True -> respect provided mask

    # ------------------------------------------------------------------
    # layer‑norm on the input (over the last dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]  # [hidden_dim, dim]

    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight) # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Fuse gate multiplication (+ optional mask) with Triton
    # ------------------------------------------------------------------
    # Ensure contiguous storage for flattening
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    left_flat = left.view(-1)
    right_flat = right.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)

    if use_mask:
        # mask shape [B,N,N] -> float (0/1) and flatten
        mask_f = mask.to(dtype=dtype, device=device)
        mask_flat = mask_f.contiguous().view(-1)
    else:
        # dummy tensor (won’t be read because HAS_MASK=False)
        mask_flat = torch.empty(0, dtype=dtype, device=device)

    numel = left_flat.numel()
    BLOCK_SIZE = 1024
    grid = lambda meta: ((numel + meta["BLOCK_SIZE"] - 1) // meta["BLOCK_SIZE"],)

    apply_gate_mask_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        hidden_dim,
        numel,
        BLOCK_SIZE=BLOCK_SIZE,
        HAS_MASK=use_mask,
    )
    # After the kernel, `left` and `right` already contain the gated (and masked) values.

    # ------------------------------------------------------------------
    # Core TriMul: batch matrix multiply over the kth dimension
    #   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left.shape == (B,N,N,hidden_dim)

    # reshape to (B*hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # L @ Rᵀ for each hidden channel
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden_dim, N, N)

    # reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output processing
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]   # [dim, hidden_dim]

    # layer‑norm over hidden_dim
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    # multiply by out_gate
    out = out * out_gate

    # final linear projection back to `dim`
    out = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
