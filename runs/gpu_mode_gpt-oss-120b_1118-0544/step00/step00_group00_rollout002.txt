@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rj, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(2)
    b = pid_bh // H
    h = pid_bh % H

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J

    # Offsets
    i_offsets = i0 + tl.arange(0, BLOCK_I)
    j_offsets = j0 + tl.arange(0, BLOCK_J)

    i_mask = i_offsets < N
    j_mask = j_offsets < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for k_offset in range(0, N, BLOCK_K):
        k_offsets = k_offset + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # Load left tile: shape (BLOCK_I, BLOCK_K)
        left_ptrs = left_ptr + b * stride_lb + h * stride_lh + i_offsets[:, None] * stride_li + k_offsets[None, :] * stride_lk
        left = tl.load(left_ptrs, mask = i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # Load right tile transposed: shape (BLOCK_K, BLOCK_J)
        right_ptrs = right_ptr + b * stride_rb + h * stride_rh + k_offsets[:, None] * stride_ri + j_offsets[None, :] * stride_rj
        right = tl.load(right_ptrs, mask = k_mask[:, None] & j_mask[None, :], other=0.0).to(tl.float32)

        # dot: left (BLOCK_I, BLOCK_K) x right (BLOCK_K, BLOCK_J)
        acc += tl.dot(left, right)

    # Store output
    out_ptrs = out_ptr + b * stride_ob + h * stride_oh + i_offsets[:, None] * stride_oi + j_offsets[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask = i_mask[:, None] & j_mask[None, :])

def custom_kernel(data):
   x, mask, weights, config = data

   dim = config["dim"]
   hidden_dim = config["hidden_dim"]
   eps = 1e-5

   # x shape B,N,N,dim
   B, N, _, _ = x.shape

   # LayerNorm
   # compute mean var across last dim
   mean = x.mean(dim=-1, keepdim=True)
   var = x.var(dim=-1, unbiased=False, keepdim=True)
   x_norm = (x - mean) / torch.sqrt(var + eps)
   x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]
   # Might reshape weight/bias shape [dim]; broadcast.

   # Projections
   left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"]) # shape B,N,N,hidden_dim
   right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

   # Mask handling
   if mask is not None and not config.get("nomask", True):
       mask_exp = mask.unsqueeze(-1)  # shape B,N,N,1
       left = left * mask_exp
       right = right * mask_exp
   else:
       # if no mask, just treat as ones.
       pass

   # Gates
   left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
   right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
   out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

   # Apply gates
   left = left * left_gate
   right = right * right_gate

   # Convert to half for kernel
   left_h = left.to(torch.float16).contiguous()
   right_h = right.to(torch.float16).contiguous()

   # Prepare output tensor
   out_h = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=x.device)

   # Get strides
   stride_lb, stride_li, stride_lk, stride_lh = left_h.stride()
   stride_rb, stride_ri, stride_rj, stride_rh = right_h.stride()
   stride_ob, stride_oi, stride_oj, stride_oh = out_h.stride()

   # Launch kernel
   BLOCK_I = 32
   BLOCK_J = 32
   BLOCK_K = 64

"""
TriMul forward implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the last channel dimension.
2. Apply two linear projections (left/right) and three gating linear layers.
3. Mask the projected tensors if a mask is provided.
4. Fuse gating with the projected tensors.
5. Compute the core 4‑D einsum:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   using a Triton kernel that tiles the i, j dimensions and reduces over k.
   Each kernel block processes a (BLOCK_I × BLOCK_J) tile for a fixed batch
   and hidden channel `h`. The reduction over `k` is performed in blocks of
   `BLOCK_K`, accumulating in FP32 for numeric stability.
6. Layer‑norm over the hidden dimension, multiply by the output gate, and
   finally linearly project back to the original channel dimension.

The heavy pairwise reduction (step 5) runs on the GPU via Triton,
while the surrounding linear, mask, and norm operations are performed in PyTorch.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # *float16
    right_ptr,              # *float16
    out_ptr,                # *float32
    B, N, H,                # int32
    stride_lb, stride_li, stride_lk, stride_lh,   # int32 strides for left
    stride_rb, stride_ri, stride_rj, stride_rh,   # int32 strides for right
    stride_ob, stride_oi, stride_oj, stride_oh,   # int32 strides for out
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    # program ids
    pid_i = tl.program_id(0)                # tile over i dimension
    pid_j = tl.program_id(1)                # tile over j dimension
    pid_z = tl.program_id(2)                # combined batch * hidden
    b = pid_z // H
    h = pid_z % H

    # start indices of the tile
    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J

    # range vectors
    i_range = i0 + tl.arange(0, BLOCK_I)
    j_range = j0 + tl.arange(0, BLOCK_J)
    i_mask = i_range < N
    j_mask = j_range < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k dimension
    for k_offset in range(0, N, BLOCK_K):
        k_range = k_offset + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # ----- load left[b, i, k, h] -----
        # shape (BLOCK_I, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + h * stride_lh
            + i_range[:, None] * stride_li
            + k_range[None, :] * stride_lk
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        )
        left = left.to(tl.float32)

        # ----- load right[b, j, k, h] transposed to (k, j) -----
        # shape (BLOCK_K, BLOCK_J)
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + h * stride_rh
            + k_range[:, None] * stride_ri
            + j_range[None, :] * stride_rj
        )
        right = tl.load(
            right_ptrs,
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0,
        )
        right = right.to(tl.float32)

        # dot product over k -> accumulate
        acc += tl.dot(left, right)  # (BLOCK_I, BLOCK_J)

    # ----- store result -----
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + h * stride_oh
        + i_range[:, None] * stride_oi
        + j_range[None, :] * stride_oj
    )
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul module using a Triton kernel for the core reduction.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be ignored)
            - weights: dict of model weights
            - config: dict with keys 'dim', 'hidden_dim', 'nomask' (bool)
    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # Compute mean/var over channel dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b  # broadcast over last dim

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Mask (if present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape [B, N, N] -> [B, N, N, 1]
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gates
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the sum over k
    # ------------------------------------------------------------------
    # Convert to fp16 for speed, keep a copy for accumulation in fp32
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, _ = left_h.shape  # N should be same as before

    # Output buffer for the intermediate hidden representation
    out_h = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Extract strides (in elements, not bytes)
    stride_lb, stride_li, stride_lk, stride_lh = left_h.stride()
    stride_rb, stride_ri, stride_rj, stride_rh = right_h.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out_h.stride()

    # Kernel block sizes (tuned for typical B200 memory)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        B * hidden_dim,
    )

    # Launch kernel
    tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_ri,
        stride_rj,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # LayerNorm over hidden_dim
    mean_out = out_h.mean(dim=-1, keepdim=True)
    var_out = out_h.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out_h - mean_out) / torch.sqrt(var_out + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # Apply output gate (shape broadcastable)
    out_norm = out_norm * out_gate

    # Final linear projection back to original dim
    out = torch.nn.functional.linear(out_norm, to_out_w)
    return out
