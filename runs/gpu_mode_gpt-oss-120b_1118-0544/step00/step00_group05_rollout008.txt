Define the kernel:

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Abatch, stride_Ar, stride_Ac,
    stride_Bbatch, stride_Br, stride_Bc,
    stride_Cbatch, stride_Cr, stride_Cc,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ACC_TYPE: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=ACC_TYPE)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        # compute pointers
        a_ptr = A_ptr + batch_id * stride_Abatch + offs_m[:, None] * stride_Ar + offs_k[None, :] * stride_Ac
        b_ptr = B_ptr + batch_id * stride_Bbatch + offs_k[:, None] * stride_Br + offs_n[None, :] * stride_Bc
        # load with mask for out-of-bounds
        a = tl.load(a_ptr, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptr, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)
    # write back
    c_ptr = C_ptr + batch_id * stride_Cbatch + offs_m[:, None] * stride_Cr + offs_n[None, :] * stride_Cc
    tl.store(c_ptr, acc, mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

"""
custom_kernel: Highly optimized TriMul (outgoing) forward pass.

Algorithm
---------
1. LayerNorm over the innermost channel dimension of the input tensor.
2. Two linear projections (left/right) without bias.
3. Optional mask multiplication (broadcasted over the channel dim).
4. Gated activations (sigmoid of three independent linear layers) and element‑wise gating.
5. Core N³ operation:  out[i,j,:] = Σ_k left[i,k,:] * right[j,k,:].
   This is a batch of matrix multiplications  C = A @ Bᵀ  where
   A = left[..., i, k, :]  (shape B×H×N×N) and
   Bᵀ = right[..., k, j, :] (shape B×H×N×N).
   We flatten the batch×hidden dimension and implement the batched matmul
   with a custom Triton kernel (half‑precision compute, FP32 accumulation).
6. LayerNorm over the hidden dimension, gating by out_gate, and final linear
   projection back to the original channel dimension.

Only the N³ reduction (step 5) is performed in Triton; all other
operations use PyTorch for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for batched matmul C = A @ B  (A: [B, M, K], B: [B, K, N])
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Abatch, stride_Ar, stride_Ac,
    stride_Bbatch, stride_Br, stride_Bc,
    stride_Cbatch, stride_Cr, stride_Cc,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ACC_TYPE: tl.constexpr
):
    """Compute batched GEMM using tiling. All tensors are assumed
    contiguous after any required transposes."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets of the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_batch_ptr = A_ptr + batch_id * stride_Abatch
    b_batch_ptr = B_ptr + batch_id * stride_Bbatch
    c_batch_ptr = C_ptr + batch_id * stride_Cbatch

    # Accumulator in higher precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=ACC_TYPE)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_Ar + offs_k[None, :] * stride_Ac
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_Br + offs_n[None, :] * stride_Bc
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Block matrix multiply and accumulate
        acc += tl.dot(a, b)

    # Write the result back
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_Cr + offs_n[None, :] * stride_Cc
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matmul C = A @ B using the Triton kernel above.
    A: (batch, M, K)  float16
    B: (batch, K, N)  float16
    Returns C: (batch, M, N) float16
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    # Allocate output
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Tunable tile sizes (chosen to balance occupancy and shared memory)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        A_ptr=A,
        B_ptr=B,
        C_ptr=C,
        M=M,
        N=N,
        K=K,
        stride_Abatch=A.stride(0),
        stride_Ar=A.stride(1),
        stride_Ac=A.stride(2),
        stride_Bbatch=B.stride(0),
        stride_Br=B.stride(1),
        stride_Bc=B.stride(2),
        stride_Cbatch=C.stride(0),
        stride_Cr=C.stride(1),
        stride_Cc=C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        ACC_TYPE=tl.float32,
        num_warps=4,
    )
    return C


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool) – may be all‑ones
        - weights_dict : mapping from parameter name to torch.Tensor
        - config_dict  : contains 'dim', 'hidden_dim', and optional 'nomask'
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x_norm, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask may be bool; cast to match dtype
        mask_exp = mask.to(x.dtype).unsqueeze(-1)        # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of linear layers)
    # --------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core N³ reduction via Triton (batched matmul)
    # --------------------------------------------------------------
    # Cast to half for the kernel (saves bandwidth)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # right needs to be transposed on the two sequence axes: (j, k) -> (k, j)
    right_h_t = right_h.permute(0, 2, 1, 3).contiguous()  # [B, N, N, H] with axes (k, j)

    B, N_seq, _, H = left_h.shape  # left shape: [B, N, N, H]

    # Bring hidden dimension to batch dimension for the matmul
    A_mat = left_h.permute(0, 3, 1, 2).contiguous().view(B * H, N_seq, N_seq)   # [B*H, M, K]
    B_mat = right_h_t.permute(0, 3, 1, 2).contiguous().view(B * H, N_seq, N_seq)  # [B*H, K, N]

    # Perform batched matmul with Triton
    C_mat = triton_batched_matmul(A_mat, B_mat)  # [B*H, M, N] half precision

    # Reshape back to [B, N, N, H]
    out_h = C_mat.view(B, H, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # Cast back to float32 for the remaining FP32 ops
    out = out_h.to(torch.float32)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate  # element‑wise gate

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # final projection to original dim

    return out
```
