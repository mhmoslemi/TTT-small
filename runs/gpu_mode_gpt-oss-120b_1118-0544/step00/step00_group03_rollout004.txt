@triton.jit
def tri_mul_out_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H, # hidden_dim
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_head = tl.program_id(0)
    # decode batch and hidden dim from pid_head
    b = pid_head // H
    h = pid_head % H

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out of bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left: shape (BLOCK_M, BLOCK_K)
        # left pointers: left_ptr + b*stride_l_b + h*stride_l_h + offs_m[:,None]*stride_l_i + k_offset[None,:]*stride_l_k
        left_ptrs = left_ptr + b*stride_l_b + h*stride_l_h + offs_m[:, None] * stride_l_i + k_offset[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right: need right[j,k] where right shape b,j,k,h. Since we need right transposed, we load right as (BLOCK_N, BLOCK_K) where j is output column, k is reduction index.
        # right ptrs: right_ptr + b*stride_r_b + h*stride_r_h + offs_n[:, None] * stride_r_j + k_offset[None, :] * stride_r_k
        right_ptrs = right_ptr + b*stride_r_b + h*stride_r_h + offs_n[:, None] * stride_r_j + k_offset[None, :] * stride_r_k
        right = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right) # shape (BLOCK_M, BLOCK_N)
    # store
    out_ptrs = out_ptr + b*stride_o_b + h*stride_o_h + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']; hidden_dim = config['hidden_dim']
    device = input_tensor.device

    # layernorm
    norm_weight = weights['norm.weight']; norm_bias = weights['norm.bias']
    # torch.nn.functional layer_norm: normalized shape = dim
    # Input shape [B,N,N,dim]; we need to normalize along last dim
    # using weight, bias
    eps = 1e-5
    # compute mean var per location:
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # projections
    left_proj_weight = weights['left_proj.weight']  # shape [hidden_dim, dim]
    right_proj_weight = weights['right_proj.weight'] # shape [hidden_dim, dim]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask if provided (size [B,N,N])
    if mask is not None:
        mask_unsqueeze = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_unsqueeze
        right = right * mask_unsqueeze

    # gates
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # apply gates to left/right
    left = left * left_gate
    right = right * right_gate

    # ensure dtype: use float16 for kernel if possible:
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    # allocate output
    out = torch.empty((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2], hidden_dim), dtype=torch.float16, device=device)

    # launch kernel
    # compute strides:
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Determine block sizes
    BLOCK_M = 64 # rows
    BLOCK_N = 64 # cols
    BLOCK_K = 64 # reduction block size

    grid = (input_tensor.shape[0] * hidden_dim,  # heads
            (input_tensor.shape[1] + BLOCK_M - 1) // BLOCK_M,
            (input_tensor.shape[2] + BLOCK_N - 1) // BLOCK_N)

    tri_mul_out_kernel[grid](
        left, right, out,
        input_tensor.shape[0], input_tensor.shape[1], hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # convert back to float32 for further layers
    out = out.to(torch.float32)

    # to_out_norm (LayerNorm over hidden_dim)
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    eps_norm = 1e-5
    # Normalize per (..., hidden_dim)
    mean = out.mean(dim=-1, keepdim=True)
    var = ((out - mean) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean) / torch.sqrt(var + eps_norm)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # apply out_gate
    out_norm = out_norm * out_gate

    # final linear to dim
    to_out_weight = weights['to_out.weight']    # shape [dim, hidden_dim]
    out_final = torch.nn.functional.linear(out_norm, to_out_weight)  # bias None

    return out_final

def custom_kernel(data):
    """
    Triton implementation of the outgoing Triangle Multiplicative Update (TriMul).
    Performs LayerNorm, linear projections, gating, and a batched matrix multiplication
    over the sequence dimension using a Triton kernel, followed by a second LayerNorm
    and output projection.
    """
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    # using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections without bias
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if present
    if not nomask and mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # Allocate output tensor for inner multiplication
    out_h = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Strides
    stride_l = left_h.stride()  # (b,i,k,d)
    stride_r = right_h.stride()
    stride_o = out_h.stride()

    # Block size heuristics; may tune
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_out_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        hidden_dim,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to float32 for subsequent layernorm and linear
    out = out_h.to(torch.float32)

    # Second LayerNorm (hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply output gate
    out = out * out_gate

    # Final projection to dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # no bias

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) kernel.

This implementation follows the PyTorch reference but accelerates the
core O(N^3) contraction using a custom Triton kernel.  The steps are:

1. Layer‑norm the input tensor (B,N,N,dim).
2. Linear projections to a hidden dimension (hidden_dim) for left/right.
3. Optional mask application.
4. Gated projections (sigmoid‑gated linear layers).
5. Batched matrix multiplication:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   (equivalently left @ rightᵀ for each hidden head) performed by Triton.
6. Second layer‑norm over the hidden dimension.
7. Output gate (sigmoid) and final linear projection back to dim.

Only the heavy contraction (step 5) runs on the GPU via Triton; all
other ops use PyTorch which is cheap compared to the N³ work.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def tri_mul_out_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_head = tl.program_id(0)   # (batch * hidden_dim) dimension
    pid_m = tl.program_id(1)      # block row   (i)
    pid_n = tl.program_id(2)      # block col   (j)

    # decode batch and hidden index
    b = pid_head // H
    h = pid_head % H

    # off‑sets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left block: shape (BLOCK_M, BLOCK_K) -----
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + offs_m[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
        )
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ----- right block transposed: shape (BLOCK_K, BLOCK_N) -----
        # right is indexed as (b, j, k, h); we need (k, j)
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + offs_k[:, None] * stride_r_k
            + offs_n[None, :] * stride_r_j
        )
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # store the result (still FP32, will be cast later)
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + offs_m[:, None] * stride_o_i
        + offs_n[None, :] * stride_o_j
    )
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

    Forward pass of the outgoing TriMul operator.

        Output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """

    device = input_tensor.device

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input layer‑norm
    # ------------------------------------------------------------
    eps = 1e-5
    x_norm = F.layer_norm(
        input_tensor,

        eps=eps,
    )   # [B,N,N,dim], float32

    # ------------------------------------------------------------

    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate

    # ------------------------------------------------------------
    # 5. Batched matrix multiplication via Triton
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------
    # Cast to float16 for the kernel (memory & bandwidth reduction)
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out_h = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # strides (in elements, not bytes) for each tensor
    stride_l = left_h.stride()   # (b, i, k, d)
    stride_r = right_h.stride()
    stride_o = out_h.stride()

    # block sizes – these work well for the target hardware
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # one program per (batch, hidden) pair
        (N + BLOCK_M - 1) // BLOCK_M,       # rows
        (N + BLOCK_N - 1) // BLOCK_N,       # cols
    )

    tri_mul_out_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        hidden_dim,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 for the remaining ops
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------
    # 6. Second layer‑norm over hidden_dim
    # ------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------
    # 7. Output gate and final linear projection
    # ------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # final [B,N,N,dim]

    return out
```
