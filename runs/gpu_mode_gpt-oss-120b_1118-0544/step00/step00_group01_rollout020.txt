@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_ri, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)  # i block
    pid_j = tl.program_id(1)  # j block
    pid_bd = tl.program_id(2)  # combined batch*hidden
    b = pid_bd // hidden_dim
    d = pid_bd % hidden_dim

    # compute start offsets
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # create offsets for i and j within block
    offs_i = i_offset + tl.arange(0, BLOCK_M)[:, None]  # shape (BLOCK_M, 1)
    offs_j = j_offset + tl.arange(0, BLOCK_N)[None, :]  # shape (1, BLOCK_N)

    # mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i & mask_j

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)[:, None]  # (BLOCK_K,1)
        # compute k mask
        mask_k = cur_k < N

        # left: shape (BLOCK_M, BLOCK_K) indices (i,k)
        left_ptrs = left_ptr + b*stride_lb + d*stride_ld + offs_i * stride_li + cur_k * stride_lk
        left = tl.load(left_ptrs, mask=mask_i & mask_k, other=0.0).to(tl.float32)

        # right: need shape (BLOCK_K, BLOCK_N) (k,j)
        right_ptrs = right_ptr + b*stride_rb + d*stride_rd + cur_k * stride_rk + offs_j * stride_rj
        right = tl.load(right_ptrs, mask=mask_k & mask_j, other=0.0).to(tl.float32)

        # dot product
        # left: (BLOCK_M, BLOCK_K), right: (BLOCK_K, BLOCK_N)
        acc += tl.dot(left, right)

    # write out
    out_ptrs = out_ptr + b*stride_ob + d*stride_od + offs_i * stride_oi + offs_j * stride_oj
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_ij)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm - x_norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # compute mean var across last dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # linear projections
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])  # shape [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # mask handling
    if config.get("nomask", False):
        mask_tensor = torch.ones((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]), dtype=input_tensor.dtype, device=device)
    else:
        mask_tensor = mask.to(input_tensor.dtype)
    mask_exp = mask_tensor.unsqueeze(-1)  # shape [B,N,N,1]

    left = left_proj * mask_exp
    right = right_proj * mask_exp

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # cast to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output
    out = torch.empty((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2], hidden_dim), dtype=torch.float32, device=device)

    # get strides
    stride_bl, stride_bi, stride_bk, stride_bd = left_fp16.stride()
    stride_rl, stride_ri, stride_rk, stride_rd = right_fp16.stride()
    stride_ol, stride_oi, stride_oj, stride_od = out.stride()

    # Define block sizes
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # Number of blocks
    grid = (
        (input_tensor.shape[1] + BLOCK_M - 1) // BLOCK_M,
        (input_tensor.shape[2] + BLOCK_N - 1) // BLOCK_N,
        input_tensor.shape[0] * hidden_dim,
    )

    # launch kernel
    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        input_tensor.shape[0],
        input_tensor.shape[1],  # N
        stride_bl,
        stride_bi,
        stride_bk,
        stride_bd,
        stride_rl,
        stride_ri,
        stride_rk,
        stride_rd,
        stride_ol,
        stride_oi,
        stride_oj,
        stride_od,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_out) * torch.rsqrt(var_out + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # apply out_gate
    out_norm = out_norm * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"]
    final = torch.nn.functional.linear(out_norm, to_out_weight)  # shape [B,N,N,dim]

    return final

import triton
import triton.language as tl

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_ri, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    hidden_dim: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)

    b = pid_bd // hidden_dim
    d = pid_bd % hidden_dim

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # Ranges
    i_range = i_base + tl.arange(0, BLOCK_M)
    j_range = j_base + tl.arange(0, BLOCK_N)

    # Masks for output
    mask_i = i_range < N
    mask_j = j_range < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_offset in range(0, N, BLOCK_K):
        k_range = k_offset + tl.arange(0, BLOCK_K)

        # masks for k
        mask_k = k_range < N

        # Load left tile (i,k)
        offs_i = i_range[:, None]  # (M,1)
        offs_k = k_range[None, :]  # (1,K)
        left_ptrs = left_ptr + b * stride_lb + d * stride_ld + offs_i * stride_li + offs_k * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile (k,j) transposed
        offs_k_r = k_range[:, None]  # (K,1)
        offs_j = j_range[None, :]    # (1,N)
        right_ptrs = right_ptr + b * stride_rb + d * stride_rd + offs_k_r * stride_rk + offs_j * stride_rj
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Accumulate dot product
        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_ptr + b * stride_ob + d * stride_od + i_range[:, None] * stride_oi + j_range[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_ij)

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy O(N³) part:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is implemented as a batched, per‑channel GEMM using a custom Triton kernel.
All other operations (LayerNorm, Linear, gating, final LayerNorm/Linear) are
performed in PyTorch.  The kernel works on fp16 left/right tensors for
bandwidth efficiency and accumulates in fp32.  It supports arbitrary batch
size (B≤2) and hidden dimension (hidden_dim≤384) by launching a 3‑D grid:
  (i‑blocks, j‑blocks, batch*hidden_dim).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr,               # *fp16  [B,N,N,HD]
    right_ptr,              # *fp16  [B,N,N,HD]
    out_ptr,                # *fp32  [B,N,N,HD]
    B, N,                   # runtime scalars
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_ri, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    hidden_dim: tl.constexpr,                     # compile‑time
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)          # block row (i)
    pid_j = tl.program_id(1)          # block col (j)
    pid_bd = tl.program_id(2)         # batch * hidden_dim

    b = pid_bd // hidden_dim          # batch index
    d = pid_bd % hidden_dim           # hidden channel

    # ------------------- offsets for the output tile -------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    i_range = i_base + tl.arange(0, BLOCK_M)          # (M,)
    j_range = j_base + tl.arange(0, BLOCK_N)          # (N,)

    mask_i = i_range < N
    mask_j = j_range < N
    mask_ij = mask_i[:, None] & mask_j[None, :]       # (M,N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- reduction over k -------------------
    for k_offset in range(0, N, BLOCK_K):
        k_range = k_offset + tl.arange(0, BLOCK_K)    # (K,)
        mask_k = k_range < N

        # ----- left tile: (i, k) -----
        # shape (M, K)
        offs_i = i_range[:, None]                     # (M,1)
        offs_k = k_range[None, :]                     # (1,K)
        left_ptrs = (left_ptr
                     + b * stride_lb
                     + d * stride_ld
                     + offs_i * stride_li
                     + offs_k * stride_lk)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ----- right tile transposed: (k, j) -----
        # shape (K, N)
        offs_k_r = k_range[:, None]                   # (K,1)
        offs_j = j_range[None, :]                     # (1,N)
        right_ptrs = (right_ptr
                      + b * stride_rb
                      + d * stride_rd
                      + offs_k_r * stride_rk
                      + offs_j * stride_ri)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # ----- accumulate dot product over k -----
        acc += tl.dot(left, right)   # (M,N)

    # ------------------- write out -------------------
    out_ptrs = (out_ptr
                + b * stride_ob
                + d * stride_od
                + i_range[:, None] * stride_oi
                + j_range[None, :] * stride_oj)
    tl.store(out_ptrs, acc, mask=mask_ij)


def custom_kernel(data):
    """Forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] torch.Tensor (float32)
            - mask:        [B, N, N]    torch.Tensor (bool/float)
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm on dim ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_weight + norm_bias  # shape [B,N,N,dim]

    # ---------- Linear projections ----------
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B,N,N,HD]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ---------- Mask handling ----------
    if config.get("nomask", False):
        mask_tensor = torch.ones(input_tensor.shape[:3], dtype=dtype, device=device)
    else:
        mask_tensor = mask.to(dtype)
    mask_exp = mask_tensor.unsqueeze(-1)  # [B,N,N,1]

    left = left_proj * mask_exp
    right = right_proj * mask_exp

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to fp16 for the heavy kernel ----------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ---------- Allocate output (before final linear) ----------
    B, N, _, _ = input_tensor.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ---------- Strides (in elements) ----------
    stride_lb, stride_li, stride_lk, stride_ld = left_fp16.stride()
    stride_rb, stride_ri, stride_rk, stride_rd = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # ---------- Triton launch parameters ----------
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- to_out_norm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # ---------- Apply out gate ----------
    out_norm = out_norm * out_gate

    # ---------- Final linear back to dim ----------
    final = F.linear(out_norm, weights["to_out.weight"])  # [B,N,N,dim]

    return final
