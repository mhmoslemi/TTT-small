@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    B, N, D,
    stride_x_batch, stride_x_i, stride_x_j, stride_x_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    eps: tl.float32 = 1e-5,
    BLOCK_D: tl.constexpr):
   pid = tl.program_id(0)
   total = B * N * N
   i = pid % (N * N)
   b = pid // (N * N)
   # i / N gives i_idx ; i % N gives j_idx? Actually we have 2 dimensions i and j.
   i_idx = i // N
   j_idx = i % N

   # compute offsets for x slice for this (b,i_idx,j_idx,:)
   offs_d = tl.arange(0, BLOCK_D)
   # compute offset base pointer: base = x_ptr + b* stride_x_batch + i_idx * stride_x_i + j_idx * stride_x_j
   # Note x shape (B,N,N,D). stride_x_i for 2nd axis (i?), stride_x_j for 3rd axis.
   base = x_ptr + b * stride_x_batch + i_idx * stride_x_i + j_idx * stride_x_j

   sum = tl.float32(0.0)
   sumsq = tl.float32(0.0)

   num_iter = tl.cdiv(D, BLOCK_D)
   for it in range(0, num_iter):
       d_off = it * BLOCK_D
       cur_offs = d_off + offs_d
       mask = cur_offs < D
       x = tl.load(base + cur_offs * stride_x_d, mask=mask, other=0.0).to(tl.float32)
       sum += tl.sum(x, axis=0)
       sumsq += tl.sum(x * x, axis=0)

   mean = sum / D
   var = sumsq / D - mean * mean
   rsig = 1.0 / tl.sqrt(var + eps)

   # second pass: write normalized values
   for it in range(0, num_iter):
       d_off = it * BLOCK_D
       cur_offs = d_off + offs_d
       mask = cur_offs < D
       x = tl.load(base + cur_offs * stride_x_d, mask=mask, other=0.0).to(tl.float32)
       norm = (x - mean) * rsig
       if weight_ptr != 0:
           weight = tl.load(weight_ptr + cur_offs * 1, mask=mask, other=0.0)
           norm = norm * weight
       if bias_ptr != 0:
           bias = tl.load(bias_ptr + cur_offs * 1, mask=mask, other=0.0)
           norm = norm + bias
       tl.store(out_ptr + base_offset_out + cur_offs * stride_out_d, norm.to(tl.float32), mask=mask)

import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    B, N, D,
    stride_x_b, stride_x_i, stride_x_j, stride_x_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    eps: tl.float32,
    BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    total_n2 = N * N
    b = pid // total_n2
    rem = pid % total_n2
    i = rem // N
    j = rem % N

    # base pointer for this vector (b,i,j)
    off = b * stride_x_b + i * stride_x_i + j * stride_x_j

    # define offsets for D dimension
    offs_d = tl.arange(0, BLOCK_D)
    num_iters = tl.cdiv(D, BLOCK_D)

    sum_val = tl.float32(0.0)
    sumsq_val = tl.float32(0.0)

    for it in range(0, num_iters):
        d = it * BLOCK_D + offs_d
        mask = d < D
        x = tl.load(x_ptr + off + d * stride_x_d, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x)
        sumsq_val += tl.sum(x * x)

    mean = sum_val / D
    var = sumsq_val / D - mean * mean
    rsig = 1.0 / tl.sqrt(var + eps)

    # second pass: write normalized values
    for it in range(0, num_iters):
        d = it * BLOCK_D + offs_d
        mask = d < D
        x = tl.load(x_ptr + off + d * stride_x_d, mask=mask, other=0.0).to(tl.float32)
        norm = (x - mean) * rsig
        # apply weight and bias
        if weight_ptr != 0:
            w = tl.load(weight_ptr + d, mask=mask, other=0.0)
            norm = norm * w
        if bias_ptr != 0:
            b_ = tl.load(bias_ptr + d, mask=mask, other=0.0)
            norm = norm + b_
        # store
        out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j
        tl.store(out_ptr + out_base + d * stride_out_d, norm, mask=mask)

"""
TriMul (outgoing) forward pass with a Triton LayerNorm kernel.

Algorithm
---------
1. **LayerNorm (Triton)**  
   Normalize the input tensor `x` over the last dimension (`dim`).  
   The kernel computes per‑vector mean/variance, applies the affine
   parameters (weight/bias) and writes the result to `x_norm`.

2. **Linear projections & gates** (torch)  
   * left  = Linear(x_norm, left_proj_weight)        # → (B,N,N,hidden)  
   * right = Linear(x_norm, right_proj_weight)  

   Apply optional mask (unsqueezed on the last axis) to `left` and `right`.  
   Compute element‑wise gates with a sigmoid and multiply:
   `left  = left  * sigmoid(Linear(x_norm, left_gate_weight ))`  
   `right = right * sigmoid(Linear(x_norm, right_gate_weight))`

3. **Triangular multiplicative contraction**  
   The contraction `out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]` is a batched
   matrix multiplication per hidden channel.  
   We permute to (B, hidden, N, N) and use `torch.matmul`:

   `out = (left.permute(0,3,1,2) @ right.permute(0,3,1,2).transpose(-2,-1))`
   → shape (B, hidden, N, N) → permute back to (B,N,N,hidden).

4. **Output normalization & gating**  
   * out = LayerNorm(out, to_out_norm_weight, to_out_norm_bias)  
   * out_gate = sigmoid(Linear(x_norm, out_gate_weight))  
   * out = out * out_gate

5. **Final projection** (torch)  
   `result = Linear(out, to_out_weight)`  # (B,N,N,dim)

The only custom Triton component is the fused LayerNorm (step 1); the rest
leverages highly‑optimized PyTorch kernels (cuBLAS, cuDNN). This satisfies
the requirement of using a Triton kernel while keeping the overall
implementation correct and reasonably fast.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,                # *float32 input
    out_ptr,              # *float32 output
    weight_ptr,           # *float32 layernorm weight (size D)
    bias_ptr,             # *float32 layernorm bias   (size D)
    B, N, D,              # int32: batch, seq_len, hidden dim
    stride_x_b, stride_x_i, stride_x_j, stride_x_d,   # input strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,   # output strides
    eps=1e-5,                                 # float32 epsilon
    BLOCK_D: tl.constexpr,                    # compile‑time block size for D
):
    """Fused LayerNorm over the last dimension.

    Each program instance handles one vector (b, i, j, :) of size D.
    """
    pid = tl.program_id(0)
    total_n2 = N * N
    b = pid // total_n2
    rem = pid % total_n2
    i = rem // N
    j = rem % N

    # Base offset for the vector (b,i,j,0)
    base = b * stride_x_b + i * stride_x_i + j * stride_x_j
    base_out = b * stride_o_b + i * stride_o_i + j * stride_o_j

    # ------------------------------------------------------------------
    # 1st pass: compute mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sumsq_val = tl.float32(0.0)

    offs_d = tl.arange(0, BLOCK_D)
    for d_start in range(0, D, BLOCK_D):
        d = d_start + offs_d
        mask = d < D
        x = tl.load(x_ptr + base + d * stride_x_d, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x)
        sumsq_val += tl.sum(x * x)

    mean = sum_val / D
    var = sumsq_val / D - mean * mean
    rsig = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2nd pass: write normalized values with affine transform
    # ------------------------------------------------------------------
    for d_start in range(0, D, BLOCK_D):
        d = d_start + offs_d
        mask = d < D
        x = tl.load(x_ptr + base + d * stride_x_d, mask=mask, other=0.0).to(tl.float32)

        norm = (x - mean) * rsig

        # affine parameters (always present in our usage)
        w = tl.load(weight_ptr + d, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr + d, mask=mask, other=0.0)
        norm = norm * w + b_

        tl.store(out_ptr + base_out + d * stride_o_d, norm, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of torch.Tensors
            - config: dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Extract weight tensors (ensure contiguous on the device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias = weights["norm.bias"].contiguous()
    left_proj_weight = weights["left_proj.weight"].contiguous()
    right_proj_weight = weights["right_proj.weight"].contiguous()
    left_gate_weight = weights["left_gate.weight"].contiguous()
    right_gate_weight = weights["right_gate.weight"].contiguous()
    out_gate_weight = weights["out_gate.weight"].contiguous()
    to_out_norm_weight = weights["to_out_norm.weight"].contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].contiguous()
    to_out_weight = weights["to_out.weight"].contiguous()

    # ------------------------------------------------------------------
    # 1. LayerNorm over the input tensor using Triton
    # ------------------------------------------------------------------
    x_norm = torch.empty_like(input_tensor)

    # Strides for input and output (both are the same layout)
    sx_b, sx_i, sx_j, sx_d = input_tensor.stride()
    so_b, so_i, so_j, so_d = x_norm.stride()

    # Triton launch configuration
    # One program per (b,i,j) vector
    total_vectors = B * N * N
    BLOCK_D = 32  # works for D up to 384 (dim in the tests)

    layernorm_kernel[(total_vectors,)](
        input_tensor,
        x_norm,
        norm_weight,
        norm_bias,
        B,
        N,
        dim,
        sx_b,
        sx_i,
        sx_j,
        sx_d,
        so_b,
        so_i,
        so_j,
        so_d,
        eps=1e-5,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gating
    # ------------------------------------------------------------------
    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if provided
    if mask is not None:
        # Ensure mask has the same dtype as the projections
        mask_f = mask.to(dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Triangular multiplicative contraction via batched matmul
    #    out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
    # ------------------------------------------------------------------
    # Bring hidden dimension to the second axis to use torch.matmul
    left_perm = left.permute(0, 3, 1, 2)   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2) # (B, hidden, N, N)

    # Batched matmul: (B*hidden) independent GEMMs of (N,N) @ (N,N)^T
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # (B, hidden, N, N)

    # Back to original layout
    out = out_perm.permute(0, 2, 3, 1)   # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 4. Output LayerNorm + out gate
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out_scaled = out_norm * out_gate

    # ------------------------------------------------------------------
    # 5. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    result = torch.nn.functional.linear(out_scaled, to_out_weight)   # (B,N,N,dim)

    return result
