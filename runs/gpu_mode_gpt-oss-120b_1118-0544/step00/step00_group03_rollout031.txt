def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, 
    GROUP_M: tl.constexpr):
    # Determine block indices
    batch = tl.program_id(0) // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid = tl.program_id(0) % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    # indices along M and N dims
    num_m_blocks = tl.cdiv(M, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    block_i = pid // num_n_blocks
    block_j = pid % num_n_blocks
    # offsets
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # pointer for start of this batch
    a_batch_ptr = a_ptr + batch * batch_stride_a
    b_batch_ptr = b_ptr + batch * batch_stride_b
    c_batch_ptr = c_ptr + batch * batch_stride_c

    # load A tile
    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_batch_ptr + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, K, BLOCK_K):

        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)

def custom_kernel(data):
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl
    
    input_tensor, mask, weights, config = data

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Apply LayerNorm: (bs, seq_len, seq_len, dim) over last dim.
    # Use torch.nn.functional.layer_norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj = F.linear(x, weights["left_proj.weight"])  # shape [bs, seq_len, seq_len, hidden_dim]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if mask is not None:
        # mask shape [bs, seq_len, seq_len]
        mask = mask.unsqueeze(-1)  # [bs, seq_len, seq_len, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Now heavy operation: compute out = sum_k left_{i,k,d} * right_{j,k,d} across k (seq_len)
    # left shape: [bs, seq_len, seq_len, hidden_dim] -> reorder to [bs, hidden_dim, seq_len, seq_len]
    # Using batch*hidden as big batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # B, H, N, N
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose second-last and last dimensions of right_perm for multiplication
    right_perm_t = right_perm.transpose(2, 3).contiguous()  # B, H, N, N (as transpose)
    B, H, N, _ = left_perm.shape
    # flatten batch and hidden dimension
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm_t.view(B * H, N, N)

    # Convert to half for speed
    left_half = left_flat.to(torch.float16)
    right_half = right_flat.to(torch.float16)

    # Run Triton gemm
    out_half = triton_gemm(left_half, right_half, N)
    # out_half shape [B*H, N, N]
    # reshape back
    out = out_half.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out = out.to(torch.float32)

    # to_out_norm layernorm over hidden_dim axis
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # layernorm over last dim (hidden_dim)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate
    out = out * out_gate

    # final linear to output dim
    out = F.linear(out, weights["to_out.weight"])

    return out
```

import triton.language as tl

def triton_gemm(A, B, N):
    # A, B shape [batch, M, K] where M=K=N.
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), device=A.device, dtype=A.dtype)

    # compute strides in elements
    a_stride_batch, a_stride_m, a_stride_k = A.stride()
    b_stride_batch, b_stride_k, b_stride_n = B.stride()
    c_stride_batch, c_stride_m, c_stride_n = C.stride()

    # define kernel param values
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = lambda META: (triton.cdiv(M, META['BLOCK_M']) * triton.cdiv(N, META['BLOCK_N']) * batch,)

    @triton.jit
    def gemm_kernel(
        a_ptr, b_ptr, c_ptr,
        a_batch_stride, a_m_stride, a_k_stride,
        b_batch_stride, b_k_stride, b_n_stride,
        c_batch_stride, c_m_stride, c_n_stride,
        M, N, K,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
        ):
        pid = tl.program_id(0)
        # compute batch index
        # each batch uses full grid of MxN blocks, total blocks per batch = num_m_blocks * num_n_blocks
        num_m = tl.cdiv(M, BLOCK_M)
        num_n = tl.cdiv(N, BLOCK_N)
        batch = pid // (num_m * num_n)
        block_idx = pid % (num_m * num_n)
        block_i = block_idx // num_n

def custom_kernel(data):
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    # get dims
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Projections
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    if mask is not None and not config.get("nomask", False):
        mask = mask.unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Reorder for gemm
    left = left.permute(0, 3, 1, 2).contiguous()   # B, H, N, N
    right = right.permute(0, 3, 1, 2).contiguous() # B, H, N, N

    # Transpose right for dot product
    right_t = right.transpose(2, 3).contiguous()   # B, H, N, N

    B, H, N, _ = left.shape

    # flatten batch and head dim
    left_flat = left.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)

    # convert to half for fast compute
    left_half = left_flat.to(torch.float16)
    right_half = right_flat.to(torch.float16)

    # gemm kernel
    out_half = batched_gemm(left_half, right_half, N)

    # reshape back
    out = out_half.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # free memory
    del left_half, right_half, left_flat, right_flat, left, right, right_t

    # to_out_norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # apply out_gate
    out = out * out_gate

    # final linear
    out = F.linear(out, weights["to_out.weight"])

    return out
```

def batched_gemm(A, B, N):
    """
    Compute C = A @ B where A,B shape (batch, N, N) and B is already transposed.
    Uses Triton kernel for batched GEMM.
    Returns C of same dtype as A.
    """
    # A,B are torch.Tensor on cuda.
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert M == K == K2 == N
    # Output
    C = torch.empty((batch, M, N), device=A.device, dtype=A.dtype)

    a_batch_stride, a_m_stride, a_k_stride = A.stride()
    b_batch_stride, b_k_stride, b_n_stride = B.stride()
    c_batch_stride, c_m_stride, c_n_stride = C.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # compute grid
    grid = (batch * ((M + BLOCK_M - 1)//BLOCK_M) * ((N + BLOCK_N - 1)//BLOCK_N),)

    # launch kernel
    batched_gemm_kernel[grid](
        A, B, C,
        a_batch_stride, a_m_stride, a_k_stride,
        b_batch_stride, b_k_stride, b_n_stride,
        c_batch_stride, c_m_stride, c_n_stride,
        M, N, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    return C
```

def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    a_batch_stride, a_m_stride, a_k_stride,
    b_batch_stride, b_k_stride, b_n_stride,
    c_batch_stride, c_m_stride, c_n_stride,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    # Compute total number of blocks per batch
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    # total blocks per batch
    total_blocks = num_m * num_n
    batch_idx = pid // total_blocks
    block_idx = pid % total_blocks
    block_i = block_idx // num_n
    block_j = block_idx % num_n

    # offsets
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute pointers for this batch
    a_ptrs = a_ptr + batch_idx * a_batch_stride + offs_m[:, None] * a_m_stride + tl.arange(0, BLOCK_K)[None, :] * a_k_stride
    b_ptrs = b_ptr + batch_idx * b_batch_stride + tl.arange(0, BLOCK_K)[:, None] * b_k_stride + offs_n[None, :] * b_n_stride

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        cur_k = k
        # load A and B
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (tl.arange(0, BLOCK_K) < K - cur_k)[None, :], other=0.0)

"""
Triton implementation of the outgoing TriangleMultiplicative Update (TriMul).

The kernel computes the core O(N³) reduction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
by reshaping the 4‑D tensors to (B·H, N, N) and executing a batched
matrix multiplication with a custom Triton GEMM kernel.  All surrounding
LayerNorms and linear projections are performed with PyTorch for
readability, while the heavy reduction uses Triton for speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B) where A,B have shape
# (batch, M, K) and (batch, K, N) respectively.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    a_batch_stride, a_m_stride, a_k_stride,
    b_batch_stride, b_k_stride, b_n_stride,
    c_batch_stride, c_m_stride, c_n_stride,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # number of blocks per matrix dimension
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    total_blocks = num_m * num_n

    batch_idx = pid // total_blocks
    block_idx = pid % total_blocks
    block_i = block_idx // num_n
    block_j = block_idx % num_n

    # Offsets within the M and N dimensions
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers for the current batch
    a_ptrs = a_ptr + batch_idx * a_batch_stride + \
             offs_m[:, None] * a_m_stride + tl.arange(0, BLOCK_K)[None, :] * a_k_stride
    b_ptrs = b_ptr + batch_idx * b_batch_stride + \
             tl.arange(0, BLOCK_K)[:, None] * b_k_stride + offs_n[None, :] * b_n_stride

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    for cur_k in range(0, K, BLOCK_K):
        k_remaining = K - cur_k
        mask_k = tl.arange(0, BLOCK_K) < k_remaining

        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & (offs_n < N)[None, :],
                    other=0.0)

        # a, b are already FP32 (or will be cast automatically)
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑block
        a_ptrs += BLOCK_K * a_k_stride
        b_ptrs += BLOCK_K * b_k_stride

    # Write the result back to C
    c_ptrs = c_ptr + batch_idx * c_batch_stride + \
             offs_m[:, None] * c_m_stride + offs_n[None, :] * c_n_stride
    tl.store(c_ptrs,
             acc,
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])


def batched_gemm_fp32(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B using Triton.
    A: (batch, M, K)  float32
    B: (batch, K, N)  float32
    Returns C of shape (batch, M, N) float32.
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be on CUDA"
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match"

    C = torch.empty((batch, M, N), device=A.device, dtype=torch.float32)

    # Strides (in elements)
    a_batch_stride, a_m_stride, a_k_stride = A.stride()
    b_batch_stride, b_k_stride, b_n_stride = B.stride()
    c_batch_stride, c_m_stride, c_n_stride = C.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch
        * ((M + BLOCK_M - 1) // BLOCK_M)
        * ((N + BLOCK_N - 1) // BLOCK_N),
    )

    batched_gemm_kernel[grid](
        A, B, C,
        a_batch_stride, a_m_stride, a_k_stride,
        b_batch_stride, b_k_stride, b_n_stride,
        c_batch_stride, c_m_stride, c_n_stride,
        M, N, K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriangleMultiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, dim]
            - mask_tensor  : torch.Tensor of shape [B, N, N] (or None)
            - weights_dict : dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : dict containing at least "hidden_dim" and optionally "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on the input channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (input_tensor.shape[-1],),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask
    # ------------------------------------------------------------------
    nomask = config.get("nomask", False)
    if mask is not None and not nomask:
        # Ensure mask has same dtype for multiplication
        mask_ = mask.unsqueeze(-1).to(left_proj.dtype)  # [B,N,N,1]
        left_proj = left_proj * mask_
        right_proj = right_proj * mask_

    # ------------------------------------------------------------------
    # Gate computations (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the O(N³) reduction
    #   left  : [B, N, N, H] -> permute to [B, H, N, N] (i,k)
    #   right : [B, N, N, H] -> permute to [B, H, N, N] (k,j)
    #   Then batch*H GEMMs are performed.
    # ------------------------------------------------------------------
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # permute and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, i]
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # [B, H, k, j]

    # flatten batch and head dimensions for GEMM
    left_flat = left_perm.view(B * hidden_dim, N, N)    # [B*H, N, N]
    right_flat = right_perm.view(B * hidden_dim, N, N)  # [B*H, N, N]

    # ------------------------------------------------------------------
    # Batched GEMM via Triton (float32)
    # ------------------------------------------------------------------
    out_flat = batched_gemm_fp32(left_flat, right_flat)  # [B*H, N, N]

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
