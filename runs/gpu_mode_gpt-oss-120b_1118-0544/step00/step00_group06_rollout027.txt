@triton.jit
def trimul_kernel(left_ptr, right_ptr, out_ptr,
                  B, N, H,
                  stride_left_batch, stride_left_i, stride_left_k,
                  stride_right_batch, stride_right_j, stride_right_k,
                  stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    # compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_base = 0

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load A tile left[batch, i, k:k+BLOCK_K]
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask out of bounds
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # compute pointers for A
        a_ptrs = left_ptr + pid_batch * stride_left_batch \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # compute B tile (right_T)
        # B element: right[batch, k, j] (since transposed)
        b_ptrs = right_ptr + pid_batch * stride_right_batch \
                 + offs_k[:, None] * stride_right_j \
                 + offs_j[None, :] * stride_right_k

        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)
        # accumulate
        acc += tl.dot(a, b)  # a shape MxK, b shape KxN
    # store result
    out_ptrs = out_ptr + pid_batch * stride_out_batch \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config['dim']
    hidden_dim = config['hidden_dim']

    # LayerNorm on dim
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

    # Linear projections
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])

    # Apply mask if needed
    if mask is not None and not config.get('nomask', False):
        # mask may be bool or float; convert to same dtype
        mask = mask.to(dtype=dtype)
        mask = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gating
    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights['out_gate.weight']).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute core contraction using Triton kernel
    # reshape for kernel
    B, N, _, _ = left.shape
    # left shape (B,N,N,hidden_dim)
    left_flat = left.permute(0,3,1,2).contiguous()
    right_flat = right.permute(0,3,1,2).contiguous()
    # flatten batch and hidden_dim
    batch_h = left_flat.shape[0] * left_flat.shape[1]
    left_flat = left_flat.view(batch_h, N, N)
    right_flat = right_flat.view(batch_h, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat, dtype=left_flat.dtype, device=device)

    # Determine block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # compute grid
    grid = (batch_h,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # strides
    stride_a_batch = left_flat.stride(0)
    stride_a_i = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_i = right_flat.stride(1)  # row stride (original)
    stride_b_k = right_flat.stride(2)  # column stride (original)

    stride_c_batch = out_flat.stride(0)
    stride_c_i = out_flat.stride(1)
    stride_c_j = out_flat.stride(2)

    # launch kernel
    trimul_kernel[grid](
        left_flat, right_flat, out_flat,
        batch_h, N,
        stride_a_batch, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_i, stride_b_k,
        stride_c_batch, stride_c_i, stride_c_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back
    out_hidden = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1)  # [B,N,N,hidden_dim]

    # Post processing
    out = torch.nn.functional.layer_norm(out_hidden, (hidden_dim,),
                                         weight=weights['to_out_norm.weight'],
                                         bias=weights['to_out_norm.bias'])
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights['to_out.weight'])

    # ensure output dtype float32
    out = out.to(torch.float32)
    return out

import triton
import triton.language as tl

@triton.jit
def trimul_kernel(A_ptr, B_ptr, C_ptr,
                  batch, N,
                  stride_a_batch, stride_a_i, stride_a_k,
                  stride_b_batch, stride_b_i, stride_b_k,
                  stride_c_batch, stride_c_i, stride_c_j,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Compute C[b,i,j] = sum_k A[b,i,k] * B[b,k,j] for batched matrices.
    A and B are stored in row-major layout (batch, rows, cols).
    This kernel treats B as transposed on-the-fly to match the original TriMul
    contraction ``left[i,k] * right[j,k]``.
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-bounds rows/cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile [M, K]
        a_ptrs = A_ptr + pid_batch * stride_a_batch \
                 + offs_m[:, None] * stride_a_i \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile [K, N] (B is transposed on‑the‑fly)
        # B original layout (batch, row=j, col=k)
        # we need element B_T[k, j] = B[j, k]
        b_ptrs = B_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = C_ptr + pid_batch * stride_c_batch \
             + offs_m[:, None] * stride_c_i \
             + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

The input tensor x ∈ ℝ^{B×N×N×C} is first layer‑normalized, linearly projected
to a hidden dimension H and gated.  The core contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

is a batched matrix multiplication for each (b,h) pair.  We fuse this
O(N³·H) operation in a Triton kernel that processes blocks of the i‑ and
j‑dimensions while iterating over tiled k‑chunks.  The kernel works on the
flattened batch dimension B·H and uses row‑major layout; the right operand
is accessed transposed on‑the‑fly to match the required ``left[i,k] *
right[j,k]`` pattern.

All other operations (LayerNorm, linear projections, sigmoids, final
normalisation and linear projection) are performed with PyTorch; the
kernel handles the heavy tensor contraction.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    batch, N,                                 # dimensions
    stride_a_batch, stride_a_i, stride_a_k,   # strides for A (left)
    stride_b_batch, stride_b_i, stride_b_k,   # strides for B (right, original layout)
    stride_c_batch, stride_c_i, stride_c_j,   # strides for C (output)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[b,i,j] = Σ_k A[b,i,k] * B[b,k,j] for a batch of matrices.
    A and B are stored in row‑major order (batch, rows, cols).  B is
    accessed transposed (B_T[k,j] = B[j,k]) to implement the TriMul
    contraction.
    """
    pid_batch = tl.program_id(0)   # batch index (B * H)
    pid_m = tl.program_id(1)       # tile row (i)
    pid_n = tl.program_id(2)       # tile column (j)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows/cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- Load A tile (M × K) ----
        a_ptrs = A_ptr + pid_batch * stride_a_batch \
                 + offs_m[:, None] * stride_a_i \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # ---- Load B tile (K × N) ----
        # B original layout is (batch, row=j, col=k).  We need B_T[k,j].
        b_ptrs = B_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # ---- Accumulate product ----
        acc += tl.dot(a, b)   # a: (M,K), b: (K,N) → (M,N)

    # ---- Store the result ----
    c_ptrs = C_ptr + pid_batch * stride_c_batch \
             + offs_m[:, None] * stride_c_i \
             + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensor containing all model weights
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Cast weights to the same dtype as the input (prefer fp32 for correctness)
    # ------------------------------------------------------------------
    def cast(t):
        return t.to(dtype) if t.dtype != dtype else t

    norm_w = cast(weights["norm.weight"])
    norm_b = cast(weights["norm.bias"])
    left_proj_w = cast(weights["left_proj.weight"])
    right_proj_w = cast(weights["right_proj.weight"])
    left_gate_w = cast(weights["left_gate.weight"])
    right_gate_w = cast(weights["right_gate.weight"])
    out_gate_w = cast(weights["out_gate.weight"])
    to_out_norm_w = cast(weights["to_out_norm.weight"])
    to_out_norm_b = cast(weights["to_out_norm.bias"])
    to_out_w = cast(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # LayerNorm on the input channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_w, bias=norm_b
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask = mask.to(dtype).unsqueeze(-1)                # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating scalars
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton contraction: out_hidden[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Rearrange to (B, H, N, N) and view as (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(-1, N, N)      # shape (B*H, N, N)
    right_flat = right_perm.view(-1, N, N)    # shape (B*H, N, N)

    # Allocate output tensor
    out_flat = torch.empty_like(left_flat)

    # Choose block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions: (batch, row_tiles, col_tiles)
    grid = (
        left_flat.shape[0],                      # B * H
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Strides for the flattened tensors (row‑major)
    stride_a_batch = left_flat.stride(0)
    stride_a_i = left_flat.stride(1)   # row stride (i)
    stride_a_k = left_flat.stride(2)   # column stride (k)

    stride_b_batch = right_flat.stride(0)
    stride_b_i = right_flat.stride(1)  # original row stride (j)
    stride_b_k = right_flat.stride(2)  # original column stride (k)

    stride_c_batch = out_flat.stride(0)
    stride_c_i = out_flat.stride(1)
    stride_c_j = out_flat.stride(2)

    # Launch the kernel
    trimul_kernel[grid](
        left_flat, right_flat, out_flat,
        left_flat.shape[0], N,
        stride_a_batch, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_i, stride_b_k,
        stride_c_batch, stride_c_i, stride_c_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out_hidden = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # Post‑processing: normalization, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out_hidden = out_hidden * out_gate
    out = torch.nn.functional.linear(out_hidden, to_out_w)

    # Ensure final output is float32 as required
    out = out.to(torch.float32)
    return out
