def triton_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    N: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)   # batch * hidden
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # base pointer for this batch
    a_ptr = a_ptr + pid * batch_stride_a
    b_ptr = b_ptr + pid * batch_stride_b
    c_ptr = c_ptr + pid * batch_stride_c

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # mask for boundary
        mask_m = offs_m < N
        mask_n = offs_n < N
        mask_k = offs_k < N

        # Load A block
        a_offs = offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptr + a_offs, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)

        # Load B block (transposed)
        # B indices: (k, n) due to transposition
        b_offs = offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptr + b_offs, mask=mask_k[:, None] & mask_n[None, :], other=0.0, dtype=tl.float16)

        # dot product
        acc += tl.dot(a, b)

        # Increment pointers not needed, using offsets each iteration
    # Store result
    c_offs = offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptr + c_offs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # get weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # note: no bias terms for linear layers.

    B, N, _, _ = input_tensor.shape   # shape [B, N, N, dim]

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), norm_weight, norm_bias, eps=eps)

    # projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask
    if mask is None:
        mask_tensor = torch.ones((B, N, N), dtype=x.dtype, device=x.device)
    else:
        mask_tensor = mask
    mask_tensor = mask_tensor.unsqueeze(-1)  # shape [B,N,N,1]

    left = left * mask_tensor
    right = right * mask_tensor

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Convert to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # permute and reshape for kernel
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()  # (B, hidden_dim, N, N)
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()
    # reshape to (B*hidden_dim, N, N)
    batch_hidden = left_perm.shape[0] * left_perm.shape[1]
    left_mat = left_perm.view(-1, N, N)
    right_mat = right_perm.view(-1, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat, dtype=torch.float32)

    # strides
    a_batch_stride = left_mat.stride(0)
    a_stride_m = left_mat.stride(1)
    a_stride_k = left_mat.stride(2)

    b_batch_stride = right_mat.stride(0)
    # For transposed B: we will use stride_bk = 1, stride_bn = N
    # Note: b_stride_k and b_stride_n are scalar values we pass as constants.
    # b_stride_k = 1, b_stride_n = N

    c_batch_stride = out_mat.stride(0)
    c_stride_m = out_mat.stride(1)
    c_stride_n = out_mat.stride(2)

    # block size selection, tune for N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch_hidden,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    triton_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        a_batch_stride,
        a_stride_m,
        a_stride_k,
        b_batch_stride,
        1,  # stride_bk (transposed: row stride)
        N,  # stride_bn
        c_batch_stride,
        c_stride_m,
        c_stride_n,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape output back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # LayerNorm to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), to_out_norm_weight, to_out_norm_bias, eps=eps)
    # Apply out_gate
    out = out * out_gate
    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
custom_kernel: Triton‑accelerated forward pass of the TriMul (outgoing) module.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections → left, right (no bias) and three gate projections.
3. Apply optional mask, then element‑wise multiply left/right with their gates.
4. Reduce over the third sequence dimension:
   For each batch and hidden channel d,
   compute out[b,d,i,j] = sum_k left[b,d,i,k] * right[b,d,j,k].
   This is a batched matrix multiplication of shape (N,N) @ (N,N).T.
   The reduction is performed by a Triton kernel that treats each (b,d)
   pair as an independent matrix‑multiply, using FP16 inputs and FP32
   accumulation (tensor‑core friendly).
5. Layer‑norm the reduction result, multiply by the output‑gate,
   and finally project back to the original channel dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel for the core reduction (batched GEMM with B transposed)
# --------------------------------------------------------------------------- #
@triton.jit
def _triton_gemm_kernel(
    a_ptr,                # FP16 left  matrix (batch, i, k)
    b_ptr,                # FP16 right matrix (batch, k, j)   (transposed view)
    c_ptr,                # FP32 output   (batch, i, j)
    batch_stride_a,       # stride to next batch in A
    stride_am, stride_ak,  # row‑stride and col‑stride inside A
    batch_stride_b,       # stride to next batch in B
    stride_bk, stride_bn,  # k‑stride and n‑stride inside B (NOTE: transposed!)
    batch_stride_c,       # stride to next batch in C
    stride_cm, stride_cn,  # row‑stride and col‑stride inside C
    N: tl.constexpr,      # sequence length
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # index over batch*hidden_dim
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile column

    # Offsets within the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for the output tile (handles out‑of‑bounds)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the current batch matrix
    a_batch_ptr = a_ptr + pid_batch * batch_stride_a
    b_batch_ptr = b_ptr + pid_batch * batch_stride_b
    c_batch_ptr = c_ptr + pid_batch * batch_stride_c

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a block from A (shape BLOCK_M x BLOCK_K)
        a_offs = offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_batch_ptr + a_offs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Load a block from B (shape BLOCK_K x BLOCK_N) – B is accessed transposed
        b_offs = offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_batch_ptr + b_offs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Accumulate the dot product (FP16 * FP16 → FP32)
        acc += tl.dot(a, b)

    # Store the result tile
    c_offs = offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_batch_ptr + c_offs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Host‑side entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor:  torch.Tensor of shape [B, N, N] (float 0/1) or None
            - weights_dict: mapping from layer names to torch.Tensor parameters
            - config_dict: contains keys "dim", "hidden_dim", and optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------- #
    # Retrieve weights
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    B, N, _, _ = input_tensor.shape  # input shape = [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------- #
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------- #
    # 2. Projections (no bias) and mask handling
    # ------------------------------------------------------------------- #
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    if config.get("nomask", False) or mask is None:
        mask_tensor = torch.ones((B, N, N), dtype=x.dtype, device=x.device)
    else:
        mask_tensor = mask.to(dtype=x.dtype, device=x.device)

    mask_tensor = mask_tensor.unsqueeze(-1)          # [B,N,N,1]
    left = left * mask_tensor
    right = right * mask_tensor

    # ------------------------------------------------------------------- #
    # 3. Gate projections and application
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 4. Core reduction via Triton (batched GEMM with B transposed)
    # ------------------------------------------------------------------- #
    # Cast to FP16 for tensor‑core friendly compute
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Move hidden dimension to the batch axis and make matrices contiguous
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()

    # Collapse batch & channel into a single leading dimension
    left_mat = left_perm.view(-1, N, N)   # shape (B*hidden_dim, N, N)
    right_mat = right_perm.view(-1, N, N)

    # Output buffer (FP32)
    out_mat = torch.empty_like(left_mat, dtype=torch.float32)

    # Strides (in elements)
    a_batch_stride = left_mat.stride(0)
    a_stride_m = left_mat.stride(1)   # row‑stride = N
    a_stride_k = left_mat.stride(2)   # col‑stride = 1

    b_batch_stride = right_mat.stride(0)
    # For the transposed view of B we swap row/col strides:
    b_stride_k = 1                     # stride for the k‑dimension after transpose
    b_stride_n = N                     # stride for the n‑dimension after transpose

    c_batch_stride = out_mat.stride(0)
    c_stride_m = out_mat.stride(1)    # row‑stride = N
    c_stride_n = out_mat.stride(2)    # col‑stride = 1

    # Block sizes – can be tuned, 64 works well for N up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        left_mat.shape[0],                          # batch * hidden_dim
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _triton_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        a_batch_stride,
        a_stride_m,
        a_stride_k,
        b_batch_stride,
        b_stride_k,
        b_stride_n,
        c_batch_stride,
        c_stride_m,
        c_stride_n,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Re‑shape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 5. Post‑processing (LayerNorm, out‑gate, final Linear)
    # ------------------------------------------------------------------- #
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate                       # apply the output gate
    out = F.linear(out, to_out_weight)         # project back to original dim

    return out
