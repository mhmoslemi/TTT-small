import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H, d_offset,
    stride_i_left, stride_k_left, stride_d_left,
    stride_k_right, stride_j_right, stride_d_right,
    stride_i_out, stride_j_out, stride_d_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, D_BLOCK: tl.constexpr
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col

    # offsets for block
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = d_offset + tl.arange(0, D_BLOCK)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, D_BLOCK), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile [M, K, D]
        left_ptrs = left_ptr + (offs_i[:, None, None] * stride_i_left
                               + offs_k[None, :, None] * stride_k_left
                               + offs_d[None, None, :] * stride_d_left)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                       other=0.0).to(tl.float32)  # shape (M, K, D)

        # load right tile transposed [K, N, D]
        right_ptrs = right_ptr + (offs_k[:, None, None] * stride_k_right
                                 + offs_j[None, :, None] * stride_j_right
                                 + offs_d[None, None, :] * stride_d_right)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
                        other=0.0).to(tl.float32)  # shape (K, N, D)

        # multiply-accumulate per d
        for d in range(0, D_BLOCK):
            # optionally guard with mask_d[d]; but th load already zeroed out-of-range
            left_slice = left[:, :, d]   # (M, K)
            right_slice = right[:, :, d] # (K, N)
            acc[:, :, d] += tl.dot(left_slice, right_slice)

    # store result
    out_ptrs = out_ptr + (offs_i[:, None, None] * stride_i_out
                          + offs_j[None, :, None] * stride_j_out
                          + offs_d[None, None, :] * stride_d_out)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

def custom_kernel(data):
    """Computes the TriMul forward pass using a Triton kernel to
    accelerate the core N^2 x N reduction.
    ..."""
    import torch
    import triton
    import triton.language as tl
    import math

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = torch.float16

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Cast to fp16 for projection
    x_half = x.to(dtype)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"].to(dtype)
    right_proj_weight = weights["right_proj.weight"].to(dtype)
    left = torch.nn.functional.linear(x_half, left_proj_weight)  # shape [B,N,N,H]
    right = torch.nn.functional.linear(x_half, right_proj_weight)

    # Gates
    left_gate_weight = weights["left_gate.weight"].to(dtype)
    right_gate_weight = weights["right_gate.weight"].to(dtype)
    out_gate_weight = weights["out_gate.weight"].to(dtype)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_half, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_half, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_half, out_gate_weight))

    # Apply mask (if any)
    if mask is None or config.get("nomask", True):
        mask_f = torch.ones_like(input_tensor[...,0], dtype=dtype, device=device)
    else:
        mask_f = mask.to(dtype)
    mask_f = mask_f.unsqueeze(-1)  # [B,N,N,1]

    left = left * mask_f
    right = right * mask_f

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

    batch_size, seq_len, _, _ = left.shape

    # Prepare output tensor
    out = torch.empty_like(left)  # fp16

    # Triton kernel parameters
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64
    D_BLOCK = 8  # compile-time constant

    # Triton kernel definition
    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        N, H, d_offset,
        stride_i_left, stride_k_left, stride_d_left,
        stride_k_right, stride_j_right, stride_d_right,
        stride_i_out, stride_j_out, stride_d_out,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
        BLOCK_K: tl.constexpr, D_BLOCK: tl.constexpr
    ):
        pid_m = tl.program_id(0)  # i block
        pid_n = tl.program_id(1)  # j block

        offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        offs_d = d_offset + tl.arange(0, D_BLOCK)

        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_d = offs_d < H

        acc = tl.zeros((BLOCK_M, BLOCK_N, D_BLOCK), dtype=tl.float32)

        # loop over reduction dimension k
        for k in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # left tile [M, K, D]
            left_ptrs = left_ptr + (
                offs_i[:, None, None] * stride_i_left +
                offs_k[None, :, None] * stride_k_left +
                offs_d[None, None, :] * stride_d_left
            )
            left = tl.load(left_ptrs,
                           mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                           other=0.0).to(tl.float32)

            # right tile transposed [K, N, D]
            right_ptrs = right_ptr + (
                offs_k[:, None, None] * stride_k_right +
                offs_j[None, :, None] * stride_j_right +
                offs_d[None, None, :] * stride_d_right
            )
            right = tl.load(right_ptrs,
                            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
                            other=0.0).to(tl.float32)

            # dot for each d
            for d in range(0, D_BLOCK):
                # Note: out-of-range d entries are already zero‑masked
                left_slice = left[:, :, d]   # (M, K)
                right_slice = right[:, :, d] # (K, N)
                acc[:, :, d] += tl.dot(left_slice, right_slice)

        # write out
        out_ptrs = out_ptr + (
            offs_i[:, None, None] * stride_i_out +
            offs_j[None, :, None] * stride_j_out +
            offs_d[None, None, :] * stride_d_out
        )
        tl.store(out_ptrs,
                 acc.to(tl.float16),
                 mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

    # launch kernel for each batch and d block
    for b in range(batch_size):
        left_b = left[b]
        right_b = right[b]
        out_b = out[b]

        stride_i_left = left_b.stride(0)
        stride_k_left = left_b.stride(1)
        stride_d_left = left_b.stride(2)

        stride_k_right = right_b.stride(1)
        stride_j_right = right_b.stride(0)
        stride_d_right = right_b.stride(2)

        stride_i_out = out_b.stride(0)
        stride_j_out = out_b.stride(1)
        stride_d_out = out_b.stride(2)

        grid = (math.ceil(seq_len / BLOCK_M), math.ceil(seq_len / BLOCK_N))
        for d_start in range(0, hidden_dim, D_BLOCK):
            # launch kernel
            tri_mul_kernel[grid](
                left_b, right_b, out_b,
                seq_len, hidden_dim, d_start,
                stride_i_left, stride_k_left, stride_d_left,
                stride_k_right, stride_j_right, stride_d_right,
                stride_i_out, stride_j_out, stride_d_out,
                BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
                BLOCK_K=BLOCK_K, D_BLOCK=D_BLOCK,
                num_warps=4  # maybe tune
            )

    # out now fp16; convert to fp32
    out_fp32 = out.to(torch.float32)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]

import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) operator.
    The heavy N³ reduction `einsum('...i k d, ...j k d -> ...i j d')` is
    executed in a blocked fashion by a custom Triton kernel, while
    LayerNorms and linear projections are performed with PyTorch.
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    dtype = torch.float16

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (per‑pair feature dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections + gates (all in fp16)
    # ------------------------------------------------------------------
    x_h = x.to(dtype)

    left_proj_w = weights["left_proj.weight"].to(dtype)
    right_proj_w = weights["right_proj.weight"].to(dtype)

    left = torch.nn.functional.linear(x_h, left_proj_w)   # [B, N, N, hidden]
    right = torch.nn.functional.linear(x_h, right_proj_w)

    left_gate_w = weights["left_gate.weight"].to(dtype)
    right_gate_w = weights["right_gate.weight"].to(dtype)
    out_gate_w = weights["out_gate.weight"].to(dtype)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w))

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the last dimension)
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", True):
        mask_f = torch.ones_like(input_tensor[..., 0], dtype=dtype, device=device)
    else:
        mask_f = mask.to(dtype)
    mask_f = mask_f.unsqueeze(-1)  # [B, N, N, 1]

    left = left * mask_f
    right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous tensors for simple pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape  # H == hidden_dim
    # output of the heavy einsum
    out = torch.empty_like(left)  # fp16 tensor

    # ------------------------------------------------------------------
    # 5. Triton kernel: compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64
    D_BLOCK = 8  # compile‑time block size for the hidden dimension

    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        N, H, d_offset,
        stride_i_left, stride_k_left, stride_d_left,
        stride_k_right, stride_j_right, stride_d_right,
        stride_i_out, stride_j_out, stride_d_out,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
        BLOCK_K: tl.constexpr, D_BLOCK: tl.constexpr
    ):
        pid_m = tl.program_id(0)  # i‑block
        pid_n = tl.program_id(1)  # j‑block

        offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        offs_d = d_offset + tl.arange(0, D_BLOCK)

        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_d = offs_d < H

        # accumulator per (i, j, d) element in the tile
        acc = tl.zeros((BLOCK_M, BLOCK_N, D_BLOCK), dtype=tl.float32)

        # reduction over k
        for k in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # left tile: shape (M, K, D)
            left_ptrs = left_ptr + (
                offs_i[:, None, None] * stride_i_left +
                offs_k[None, :, None] * stride_k_left +
                offs_d[None, None, :] * stride_d_left
            )
            left_tile = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                other=0.0
            ).to(tl.float32)

            # right tile (transposed) : shape (K, N, D)
            right_ptrs = right_ptr + (
                offs_k[:, None, None] * stride_k_right +
                offs_j[None, :, None] * stride_j_right +
                offs_d[None, None, :] * stride_d_right
            )
            right_tile = tl.load(
                right_ptrs,
                mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
                other=0.0
            ).to(tl.float32)

            # multiply‑accumulate per d slice
            for d in range(0, D_BLOCK):
                # Each iteration works on a slice (M, K) and (K, N)
                left_slice = left_tile[:, :, d]    # (M, K)
                right_slice = right_tile[:, :, d]  # (K, N)
                acc[:, :, d] += tl.dot(left_slice, right_slice)

        # write results
        out_ptrs = out_ptr + (
            offs_i[:, None, None] * stride_i_out +
            offs_j[None, :, None] * stride_j_out +
            offs_d[None, None, :] * stride_d_out
        )
        tl.store(
            out_ptrs,
            acc.to(tl.float16),
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        )

    # launch kernel for each batch element and each hidden‑dim block
    for b in range(B):
        left_b = left[b]
        right_b = right[b]
        out_b = out[b]

        stride_i_left = left_b.stride(0)
        stride_k_left = left_b.stride(1)
        stride_d_left = left_b.stride(2)

        stride_k_right = right_b.stride(1)
        stride_j_right = right_b.stride(0)
        stride_d_right = right_b.stride(2)

        stride_i_out = out_b.stride(0)
        stride_j_out = out_b.stride(1)
        stride_d_out = out_b.stride(2)

        grid = (math.ceil(N / BLOCK_M), math.ceil(N / BLOCK_N))
        for d_start in range(0, hidden_dim, D_BLOCK):
            tri_mul_kernel[grid](
                left_b, right_b, out_b,
                N, hidden_dim, d_start,
                stride_i_left, stride_k_left, stride_d_left,
                stride_k_right, stride_j_right, stride_d_right,
                stride_i_out, stride_j_out, stride_d_out,
                BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
                BLOCK_K=BLOCK_K, D_BLOCK=D_BLOCK,
                num_warps=4
            )

    # ------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_normed = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )

    out_normed = out_normed * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"]
    result = torch.nn.functional.linear(out_normed, to_out_w)

    return result
