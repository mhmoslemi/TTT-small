import torch
import torch.nn.functional as F
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) implemented with
    Torch for preprocessing (LayerNorms, linear projections, gating) and a
    Triton kernel for the cubic reduction.
    The kernel computes, for each batch and hidden dimension d:
        out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    which is equivalent to a batched matrix multiplication
    left[b,d] @ right[b,d].T.
    """
    # Unpack data
    if len(data) == 4:
        input_tensor, mask, weights, config = data
    else:
        # mask may be omitted; the third element is weights
        input_tensor, weights, config = data
        mask = None

    device = input_tensor.device
    dtype = input_tensor.dtype

    # Extract configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    eps = 1e-5

    # ------------------------------------------------------------
    # 1) LayerNorm on input
    # ------------------------------------------------------------
    x = F.layer_norm(input_tensor, normalized_shape=dim,
                     weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------
    # 3) Optional mask applied to projections (broadcast)
    # ------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------
    # 5) Core cubic reduction via Triton (batched gemm)
    # ------------------------------------------------------------
    # reshape tensors to [B, hidden_dim, N, N] for easier tiling
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, D, N, _ = left_perm.shape   # N == N

    # Allocate output buffer
    out_perm = torch.empty_like(left_perm)   # same dtype, same shape

    # Strides (in elements, not bytes)
    stride_a_batch = left_perm.stride(0)
    stride_a_m = left_perm.stride(2)   # i dimension
    stride_a_k = left_perm.stride(3)   # k dimension

    stride_b_batch = right_perm.stride(0)
    stride_b_n = right_perm.stride(2)  # j dimension (will be used as column)
    stride_b_k = right_perm.stride(3)  # k dimension (row in transposed view)

    stride_c_batch = out_perm.stride(0)
    stride_c_m = out_perm.stride(2)    # i dimension
    stride_c_n = out_perm.stride(3)    # j dimension

    # Kernel launch configuration
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64

    grid = (B * D,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    @triton.jit
    def triton_trimul_kernel(
            a_ptr, b_ptr, c_ptr,
            BATCH, M, N, K,
            stride_a_batch, stride_a_m, stride_a_k,
            stride_b_batch, stride_b_n, stride_b_k,
            stride_c_batch, stride_c_m, stride_c_n,
            BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
        """Compute C[b,i,j] = sum_k A[b,i,k] * B[b,j,k] for each batch."""
        batch_idx = tl.program_id(0)
        pid_m = tl.program_id(1)
        pid_n = tl.program_id(2)

        # Compute tile offsets
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        # Initialize accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over K dimension
        num_k_blocks = tl.cdiv(K, BLOCK_K)
        for k in range(0, num_k_blocks):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

            # Pointers for A and B (B is accessed as transposed)
            a_ptrs = a_ptr + batch_idx * stride_a_batch \
                     + offs_m[:, None] * stride_a_m \
                     + offs_k[None, :] * stride_a_k

            b_ptrs = b_ptr + batch_idx * stride_b_batch \
                     + offs_k[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n

            # Bounds checking
            mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
            mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

            a = tl.load(a_ptrs, mask=mask_a, other=0.0)
            b = tl.load(b_ptrs, mask=mask_b, other=0.0)

            acc += tl.dot(a, b)

        # Write the result
        c_ptrs = c_ptr + batch_idx * stride_c_batch \
                 + offs_m[:, None] * stride_c_m \
                 + offs_n[None, :] * stride_c_n

        mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
        tl.store(c_ptrs, acc, mask=mask_c)

    # launch
    triton_trimul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B * D, N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Bring output back to original layout [B, N, N, hidden_dim]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------
    # 6) Post‑norm, gating and final linear projection
    # ------------------------------------------------------------
    out = F.layer_norm(out, normalized_shape=hidden_dim,
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate

    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) – AlphaFold3.

This implementation follows the reference PyTorch code but off‑loads the
cubic reduction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

to a Triton kernel.  The reduction is equivalent to a batched matrix
multiplication `left[b,d] @ right[b,d].T` for each hidden dimension `d`.
All other operations (LayerNorm, linear projections, gating) are performed
with PyTorch for simplicity.

The kernel processes a batch of size `B*hidden_dim`, each computing a
`NxN` matrix product in tiles of size (BLOCK_M, BLOCK_N, BLOCK_K).  The
accumulator uses float32 for numerical stability; the final output is
returned as float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the cubic reduction (batched GEMM with transposed B)
# ----------------------------------------------------------------------
@triton.jit
def _triton_trimul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k] for every batch `b`.
    A has shape [BATCH, M, K]; B is interpreted as [BATCH, K, N] (i.e. the
    original tensor with its last two dimensions swapped).  The result C
    has shape [BATCH, M, N].
    """
    batch_idx = tl.program_id(0)          # BATCH dimension
    pid_m = tl.program_id(1)              # tile row
    pid_n = tl.program_id(2)              # tile column

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K sized chunks
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers for A (row‑major) and B (treated as transposed)
        a_ptrs = a_ptr + batch_idx * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k

        b_ptrs = b_ptr + batch_idx * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        # In‑bounds masks
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # Block‑wise dot product
        acc += tl.dot(a, b)

    # Write back C
    c_ptrs = c_ptr + batch_idx * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


# ----------------------------------------------------------------------
# Public entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Execute the outgoing TriMul operator.

    Args:
        data: tuple containing
            - input_tensor (torch.Tensor) : [B, N, N, dim]
            - mask         (torch.Tensor) : [B, N, N]  (may be None)
            - weights      (dict)          : model parameters
            - config       (dict)          : {"dim": int,
                                             "hidden_dim": int,
                                             "nomask": bool, ...}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments (mask may be omitted)
    # ------------------------------------------------------------------
    if len(data) == 4:
        input_tensor, mask, weights, config = data
    else:  # mask omitted
        input_tensor, weights, config = data
        mask = None

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Configuration & weight extraction
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=dim,
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        if mask is None:
            mask = torch.ones(x.shape[:3], dtype=dtype, device=device)
        mask = mask.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core cubic reduction – Triton batched GEMM
    # ------------------------------------------------------------------
    # Rearrange to [B, hidden_dim, N, N] for easier tiling
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, D, N, _ = left_perm.shape                     # D == hidden_dim

    out_perm = torch.empty_like(left_perm)           # [B, D, N, N]

    # Strides (in element units, not bytes)
    stride_a_batch = left_perm.stride(0)
    stride_a_m     = left_perm.stride(2)   # i dimension
    stride_a_k     = left_perm.stride(3)   # k dimension

    stride_b_batch = right_perm.stride(0)
    stride_b_n     = right_perm.stride(2)   # j dimension (column in output)
    stride_b_k     = right_perm.stride(3)   # k dimension (row in transposed view)

    stride_c_batch = out_perm.stride(0)
    stride_c_m     = out_perm.stride(2)    # i dimension
    stride_c_n     = out_perm.stride(3)    # j dimension

    # Tile sizes – chosen to fit in shared memory and registers on an A200
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64

    grid = (B * D,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _triton_trimul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B * D, N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Restore original layout: [B, N, N, hidden_dim]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Post‑norm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=hidden_dim,
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
