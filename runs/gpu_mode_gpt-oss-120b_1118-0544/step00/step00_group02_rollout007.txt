def batched_matmul_triton(A, B):
    # A shape (B*H, M, K), B shape (B*H, N, K) (N=M)
    # compute C = A @ B^T
    assert A.shape[0] == B.shape[0]
    batch = A.shape[0]
    M = A.shape[1]
    K = A.shape[2]
    N = B.shape[1]  # should equal M

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)
    # compute strides
    a_ptr = A
    b_ptr = B
    c_ptr = C

    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_bm = B.stride(1)
    stride_bn = B.stride(2)

    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

"""TriMul (outgoing) forward pass with a custom Triton kernel.

The kernel computes a batched matrix multiplication:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
where `left` and `right` are the projected and gated pairwise
representations (shape [B, N, N, H]).

The implementation follows the PyTorch reference but moves the
N³ reduction into a Triton kernel. All other lightweight operations
(LayerNorm, linear projections, gating) are done with PyTorch for
convenience.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: batched matmul A @ B^T  (A,B: [B, M, K])
# -------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                        # matrix sizes
    stride_az, stride_am, stride_ak,# strides for A: (batch, M, K)
    stride_bz, stride_bm, stride_bn,# strides for B: (batch, N, K)
    stride_cz, stride_cm, stride_cn,# strides for C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_z = tl.program_id(2)   # batch (B*H)

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this batch
    a_batch = a_ptr + pid_z * stride_az
    b_batch = b_ptr + pid_z * stride_bz
    c_batch = c_ptr + pid_z * stride_cz

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile [BLOCK_M, BLOCK_K]
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B^T tile [BLOCK_K, BLOCK_N] (B is [N, K])
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_bn + offs_n[None, :] * stride_bm),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Promote to fp32, dot, accumulate
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # write back
    c = acc
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ B^T for tensors of shape [batch, M, K] and [batch, N, K].
    Returns C of shape [batch, M, N] (float32, same dtype as inputs).
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, _ = B.shape
    assert K == B.shape[2]

    # Output tensor
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Strides (in elements)
    stride_az, stride_am, stride_ak = A.stride()
    stride_bz, stride_bm, stride_bn = B.stride()
    stride_cz, stride_cm, stride_cn = C.stride()

    # Tune block sizes (good defaults for typical GPUs)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bm,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    eps = 1e-5

    # Basic shapes
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", True)

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Optional masking (mask shape [B,N,N] -> broadcast to [...,1])
    # ------------------------------------------------------------------
    if use_mask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Prepare tensors for batched matmul: combine batch and hidden dim
    # ------------------------------------------------------------------
    # left/right: [B, N, N, H] -> [B*H, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # Core O(N^3) reduction via Triton: out_flat = left_flat @ right_flat^T
    # ------------------------------------------------------------------
    out_flat = _batched_matmul_triton(left_flat, right_flat)  # [B*H, N, N]

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B,N,N,dim]

    return out
