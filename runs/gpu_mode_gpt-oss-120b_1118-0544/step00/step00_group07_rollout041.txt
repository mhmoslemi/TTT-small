    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)
    # Compute start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    # Offsets for dimension i and j
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    mask_i = i_range < N
    mask_j = j_range < N

import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    # Starting indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    i_idx = i0 + tl.arange(0, BLOCK_M)
    j_idx = j0 + tl.arange(0, BLOCK_N)
    d_idx = tl.arange(0, BLOCK_D)

    # boundary check
    mask_i = i_idx < N
    mask_j = j_idx < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    num_tiles_k = tl.cdiv(N, BLOCK_K)

    for tk in range(num_tiles_k):
        k0 = tk * BLOCK_K
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # compute offsets
        # left shape: (B, N, N, H) -> (i, k, d)
        offs_left = (pid_b * stride_left_b
                     + i_idx[:, None, None] * stride_left_i
                     + k_idx[None, :, None] * stride_left_k
                     + d_idx[None, None, :] * stride_left_d)
        # load left tile, shape (M, K, D)
        left = tl.load(left_ptr + offs_left,
                       mask=mask_i[:, None, None] & mask_k[None, :, None],
                       other=0.0)
        # right shape (B, N, N, H) -> (j, k, d)
        offs_right = (pid_b * stride_right_b
                      + j_idx[:, None, None] * stride_right_j
                      + k_idx[None, :, None] * stride_right_k
                      + d_idx[None, None, :] * stride_right_d)
        right = tl.load(right_ptr + offs_right,
                        mask=mask_j[:, None, None] & mask_k[None, :, None],
                        other=0.0)

        # Loop over k within tile
        # using for loop unrolled manually for performance?
        for kk in range(BLOCK_K):
            # mask for kk, but we can rely on zeros for invalid k already
            left_vec = left[:, kk, :]   # (M, D)
            right_vec = right[:, kk, :] # (N, D)
            # outer product broadcast across i and j
            # left_vec[:, None, :] shape (M, 1, D)
            # right_vec[None, :, :] shape (1, N, D)
            # broadcast to (M, N, D)
            acc += left_vec[:, None, :] * right_vec[None, :, :]
    # Store result
    offs_out = (pid_b * stride_out_b
                + i_idx[:, None, None] * stride_out_i
                + j_idx[None, :, None] * stride_out_j
                + d_idx[None, None, :] * stride_out_d)
    tl.store(out_ptr + offs_out,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None])
```

TriMul (outgoing) implementation.

The heavy N³ contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is performed by a custom Triton kernel that tiles the i, j and k dimensions
as well as the hidden dimension d.  All surrounding LayerNorms, linear
projections and gating are handled in PyTorch for simplicity.

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    d_offset,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].

    Tiling:
      * i (rows)  -> BLOCK_M
      * j (cols)  -> BLOCK_N
      * k (sum)   -> BLOCK_K
      * d (hidden)-> BLOCK_D (processed slice)
    """
    pid_i = tl.program_id(0)   # tile over i
    pid_j = tl.program_id(1)   # tile over j
    pid_b = tl.program_id(2)   # batch index

    # start indices of the tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    i_idx = i0 + tl.arange(0, BLOCK_M)
    j_idx = j0 + tl.arange(0, BLOCK_N)
    d_idx = d_offset + tl.arange(0, BLOCK_D)

    mask_i = i_idx < N
    mask_j = j_idx < N
    mask_d = d_idx < H

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    num_tiles_k = tl.cdiv(N, BLOCK_K)
    for tk in range(num_tiles_k):
        k0 = tk * BLOCK_K
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # ----- load left[b,i,k,d] -----

            pid_b * stride_left_b
            + i_idx[:, None, None] * stride_left_i
            + k_idx[None, :, None] * stride_left_k
            + d_idx[None, None, :] * stride_left_d

            left_ptr + offs_left,

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]  (float32)
            - mask: [B, N, N] (bool/float) – may be ignored if config["nomask"] is True
            - weights: dict of model parameters (LayerNorm, Linear weights, etc.)
            - config: dict containing "dim", "hidden_dim", "nomask", ...

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # -------------------------------------------------
    # 1. Input LayerNorm
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )  # [B, N, N, dim]

    # -------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]    # (hidden_dim, dim)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # -------------------------------------------------
    # 3. Optional mask (applied before gating)
    # -------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------
    # 4. Gating (sigmoid)
    # -------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5. Core N^3 contraction via Triton
    # -------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.zeros_like(left)  # [B, N, N, hidden_dim]

    # Tile sizes – tuned for typical B200 GPUs
    BLOCK_M = 16   # i‑axis
    BLOCK_N = 16   # j‑axis
    BLOCK_K = 32   # k‑axis
    BLOCK_D = 32   # hidden‑dim slice

    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N

    # Strides for pointer arithmetic
    s_left_b, s_left_i, s_left_k, s_left_d = left.stride()
    s_right_b, s_right_j, s_right_k, s_right_d = right.stride()
    s_out_b, s_out_i, s_out_j, s_out_d = out.stride()

    # Process hidden dimension in slices of BLOCK_D
    for d_start in range(0, hidden_dim, BLOCK_D):
        tri_mul_kernel[
            (grid_i, grid_j, B)
        ](
            left,
            right,
            out,
            B,
            N,
            hidden_dim,
            s_left_b,
            s_left_i,
            s_left_k,
            s_left_d,
            s_right_b,
            s_right_j,
            s_right_k,
            s_right_d,
            s_out_b,
            s_out_i,
            s_out_j,
            s_out_d,
            d_start,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # -------------------------------------------------
    # 6. Post‑processing: LayerNorm, out‑gate, final projection
    # -------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )
    out = out * out_gate                         # broadcasted multiplication

    to_out_weight = weights["to_out.weight"]     # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)  # back to [B,N,N,dim]

    return out
