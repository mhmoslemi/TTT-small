@triton.jit
def triangular_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_bl, stride_il, stride_kl, stride_dl,
    stride_br, stride_jr, stride_kr, stride_dr,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids: m = tl.program_id(0) ; n = tl.program_id(1) ; task = tl.program_id(2)
    task = tl.program_id(2)
    b = task // hidden_dim
    d = task % hidden_dim

    # base pointer offsets for this batch and channel
    left_base = left_ptr + b * stride_bl + d * stride_dl
    right_base = right_ptr + b * stride_br + d * stride_dr
    out_base = out_ptr + b * stride_ob + d * stride_od

    # Offsets for block coordinates
    offs_m = m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = n * BLOCK_N + tl.arange(0, BLOCK_N)

    # create masks for boundaries
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        # compute actual k indices
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile: shape (BLOCK_M, BLOCK_K)
        # left[i,k] base offset: left_base + offs_m[:, None] * stride_il + offs_k[None, :] * stride_kl
        left_ptrs = left_base + (offs_m[:, None] * stride_il) + (offs_k[None, :] * stride_kl)
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load right tile transposed: we need right[k,j]
        # right base offset: right_base + offs_k[:, None] * stride_kr + offs_n[None, :] * stride_jr (since column stride is j)
        right_ptrs = right_base + (offs_k[:, None] * stride_kr) + (offs_n[None, :] * stride_jr)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # accumulate
        acc += tl.dot(left, right)  # left (M,K) dot right (K,N) => (M,N)
        # note: tl.dot expects shapes (M,K) and (K,N). They accept float32 and float16 matches.

    # store result to out: out[i,j] = acc (M,N)
    out_ptrs = out_base + (offs_m[:, None] * stride_oi) + (offs_n[None, :] * stride_oj)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    N,
    hidden_dim,
    stride_bl,
    stride_il,
    stride_kl,
    stride_dl,
    stride_br,
    stride_kr,
    stride_jr,
    stride_dr,
    stride_ob,
    stride_oi,
    stride_oj,
    stride_od,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    block_m = tl.program_id(0)  # i block
    block_n = tl.program_id(1)  # j block
    task = tl.program_id(2)     # batch*hidden_dim index

    # derive batch and channel
    b = task // hidden_dim
    d = task % hidden_dim

    # base pointers for this batch & channel
    left_base = left_ptr + b * stride_bl + d * stride_dl
    right_base = right_ptr + b * stride_br + d * stride_dr
    out_base = out_ptr + b * stride_ob + d * stride_od

    # offsets for this block
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (i,k)
        left_ptrs = left_base + (offs_m[:, None] * stride_il) + (offs_k[None, :] * stride_kl)
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right tile transposed (k,j)
        # Note: we access right as (k, j) which in memory corresponds to stride_kr for k and stride_jr for j
        right_ptrs = right_base + (offs_k[:, None] * stride_kr) + (offs_n[None, :] * stride_jr)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(left, right)  # (M,K) @ (K,N) -> (M,N)

    # write result
    out_ptrs = out_base + (offs_m[:, None] * stride_oi) + (offs_n[None, :] * stride_oj)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    norm_bias = weights["norm.bias"]

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [bs,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask handling
    if mask_tensor is not None:
        mask_exp = mask_tensor.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    # Allocate output tensor
    out = torch.empty((bs, N, N, hidden_dim), dtype=left.dtype, device=device)

    # Get strides (in elements)
    stride_bl = left.stride(0)
    stride_il = left.stride(1)
    stride_kl = left.stride(2)
    stride_dl = left.stride(3)

    stride_br = right.stride(0)
    # right's stride for k (dim2):
    stride_kr = right.stride(2)  # k dimension
    stride_jr = right.stride(1)  # j dimension
    stride_dr = right.stride(3)

    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_od = out.stride(3)

    # Triton launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (

        triton.cdiv(N, BLOCK_N),

        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # wait
    torch.cuda.synchronize()

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # apply out_gate
    out_norm = out_norm * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"]
    final = torch.nn.functional.linear(out_norm, to_out_weight)

    return final

"""
TriMul (outgoing) forward implementation.

The heavy O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a custom Triton kernel that treats each (batch, channel) pair as an
independent matrix‑multiply:  left (N×N) @ right.T (N×N).  The kernel tiles the
output (i, j) and the reduction dimension k using BLOCK_M, BLOCK_N and BLOCK_K
tiles and accumulates in float32.  All other operations (LayerNorms, linear
projections, gating, final linear) are done with PyTorch.

Inputs:
    data = (input_tensor, mask_tensor, weights, config)
        input_tensor: [B, N, N, dim]  (float32)
        mask_tensor : [B, N, N] or None
        weights     : dict of model parameters
        config      : dict with "dim" and "hidden_dim"

Output:
    Tensor of shape [B, N, N, dim] (float32)
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------- #
# Triton kernel for the core contraction (outgoing TriMul)
# --------------------------------------------------------------------- #
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,               # const float*   (B,N,N,hidden)
    right_ptr,              # const float*   (B,N,N,hidden)
    out_ptr,                # float*         (B,N,N,hidden)
    N,                      # i32  sequence length
    hidden_dim,             # i32  hidden dimension
    stride_bl, stride_il, stride_kl, stride_dl,   # strides for left
    stride_br, stride_kr, stride_jr, stride_dr,   # strides for right (note: kr = k‑stride, jr = j‑stride)
    stride_ob, stride_oi, stride_oj, stride_od,   # strides for out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # -----------------------------------------------------------------
    # program IDs: block_i (axis 0), block_j (axis 1), task = batch*hidden (axis 2)
    # -----------------------------------------------------------------
    block_i = tl.program_id(0)
    block_j = tl.program_id(1)
    task = tl.program_id(2)

    # decode batch index b and hidden channel d from task
    b = task // hidden_dim
    d = task % hidden_dim

    # base pointers for this (b,d)
    left_base  = left_ptr  + b * stride_bl + d * stride_dl
    right_base = right_ptr + b * stride_br + d * stride_dr
    out_base   = out_ptr   + b * stride_ob + d * stride_od

    # tile coordinates
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator for the output tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of tiles along the reduction (k) dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for k_tile in range(0, num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------- load left tile (i,k) -----------------
        # address: left_base + i*stride_il + k*stride_kl
        left_ptrs = left_base + (offs_i[:, None] * stride_il) + (offs_k[None, :] * stride_kl)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # ----------- load right tile transposed (k,j) -----------------
        # we need right[k,j] == original right[j,k]
        # address: right_base + k*stride_kr + j*stride_jr
        right_ptrs = right_base + (offs_k[:, None] * stride_kr) + (offs_j[None, :] * stride_jr)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # dot-product accumulation
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # ----------- write the output tile -----------------
    out_ptrs = out_base + (offs_i[:, None] * stride_oi) + (offs_j[None, :] * stride_oj)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

    # unpack inputs
    # -------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -------------------------------------------------------------
    # 1. Input LayerNorm
    # -------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, N, N, dim]

    # -------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # -------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # -------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # -------------------------------------------------------------
    # 4. Gating
    # -------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # 5. Core contraction with Triton
    # -------------------------------------------------------------
    # Ensure tensors are contiguous for stride‑based addressing
    left  = left.contiguous()
    right = right.contiguous()
    out   = torch.empty((bs, N, N, hidden_dim), dtype=dtype, device=device)

    # stride values in *elements* (not bytes)
    stride_bl = left.stride(0)
    stride_il = left.stride(1)
    stride_kl = left.stride(2)
    stride_dl = left.stride(3)

    stride_br = right.stride(0)
    # right is accessed as (k,j) -> stride for k is dim 2, for j is dim 1
    stride_kr = right.stride(2)   # k‑stride
    stride_jr = right.stride(1)   # j‑stride
    stride_dr = right.stride(3)

    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_od = out.stride(3)

    # Tile sizes – chosen to fit well on B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # blocks along i
        triton.cdiv(N, BLOCK_N),          # blocks along j
        bs * hidden_dim,                  # one program per (batch, channel)
    )

    tri_mul_outgoing_kernel[grid](
        left, right, out,
        N, hidden_dim,
        stride_bl, stride_il, stride_kl, stride_dl,
        stride_br, stride_kr, stride_jr, stride_dr,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # -------------------------------------------------------------
    # 6. Output LayerNorm and final gating
    # -------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out_norm = out_norm * out_gate

    # -------------------------------------------------------------
    # 7. Final linear projection back to dim
    # -------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]
    result = torch.nn.functional.linear(out_norm, to_out_weight)   # [B,N,N,dim]

    return result
```
