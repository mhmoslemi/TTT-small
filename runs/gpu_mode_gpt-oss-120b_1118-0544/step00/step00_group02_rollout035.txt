@triton.jit
def mask_gate_kernel(
    x_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, D,
    stride_xb, stride_xi, stride_xk, stride_xd,
    stride_gateb, stride_gatei, stride_gatek, stride_gated,
    stride_outb, stride_outi, stride_outk, stride_outd,
    stride_maskb, stride_maski, stride_maskk,
    BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    b = pid // (N * N)
    rem = pid - b * (N * N)
    i = rem // N
    k = rem - i * N

    # offsets
    offs_d = tl.arange(0, BLOCK_D)

    # cond for valid channels
    mask = offs_d < D

    x_ptrs = x_ptr + b * stride_xb + i * stride_xi + k * stride_xk + offs_d * stride_xd

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    bs, N, N_, dim = input_tensor.shape
    assert N == N_

    hidden_dim = config["hidden_dim"]

    # We need to optionally handle missing mask: if mask is None, set mask = torch.ones(bs,N,N, device=input_tensor.device, dtype=input_tensor.dtype)
    if mask is None:
        mask = torch.ones(bs, N, N, device=input_tensor.device, dtype=input_tensor.dtype)

    # LN
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5

    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Determine hidden_dim from left_proj shape
    _, _, _, hidden_dim = left_proj.shape

    # apply mask and gating using Triton
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # run kernel for left
    _launch_mask_gate(left_proj, left_gate, mask, left)

    # run kernel for right
    _launch_mask_gate(right_proj, right_gate, mask, right)

    # O(N^3) multiplication: treat hidden_dim as channel dimension
    # reshape
    # left: [B,N,N,Dh] -> [B*Dh, N, N] after permute
    left_reshape = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)

    # compute
    out_mat = torch.bmm(left_reshape, right_reshape.transpose(1, 2))  # shape B*Dh, N, N

    # reshape back
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,hidden_dim]

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # multiply out_gate
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

def mask_gate_kernel(x_ptr, gate_ptr, mask_ptr, out_ptr,
                     B, N, D,
                     stride_xb, stride_xi, stride_xk, stride_xd,
                     stride_gateb, stride_gatei, stride_gatek, stride_gated,
                     stride_outb, stride_outi, stride_outk, stride_outd,
                     stride_maskb, stride_maski, stride_maskk,
                     channel_offset: tl.constexpr,
                     BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    b = pid // (N * N)
    rem = pid - b * N * N
    i = rem // N
    k = rem - i * N

    # base offset for tensors
    base_x = b * stride_xb + i * stride_xi + k * stride_xk
    base_gate = b * stride_gateb + i * stride_gatei + k * stride_gatek
    base_out = b * stride_outb + i * stride_outi + k * stride_outk
    mask_offset = b * stride_maskb + i * stride_maski + k * stride_maskk

    # Load scalar mask
    mask_val = tl.cast(tl.load(mask_ptr + mask_offset), tl.float32)

    offs = tl.arange(0, BLOCK_D)
    channel_idx = channel_offset + offs
    mask_channel = channel_idx < D

    out = x * g * mask_val

    tl.store(out_ptr + base_out + channel_idx * stride_outd, out, mask=mask_channel)

"""
TriMul (outgoing) forward pass implemented with Triton.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Linear projections (left_proj, right_proj) and three gating
   projections (left_gate, right_gate, out_gate) are computed with
   torch.nn.functional.linear (no bias).
3. An elementwise Triton kernel fuses:   out = proj * mask * gate
   for the left and right tensors separately.  The kernel iterates over
   (batch, i, k) tiles and processes up to 32 channels per program.
4. The core O(N³) operation is a batched matrix multiplication:
       out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
   This is performed efficiently with torch.bmm on tensors reshaped to
   (B*D, N, N).
5. A second Layer‑norm (to_out_norm) is applied, multiplied by the
   out‑gate, and finally projected back to the original channel dimension
   with a linear layer.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def mask_gate_kernel(
    proj_ptr,          # left_proj or right_proj
    gate_ptr,          # corresponding gate tensor
    mask_ptr,          # mask tensor (no channel dim)
    out_ptr,           # destination tensor (same shape as proj)
    B, N, D,           # runtime dimensions
    stride_proj_b, stride_proj_i, stride_proj_k, stride_proj_d,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_d,
    stride_out_b,  stride_out_i,  stride_out_k,  stride_out_d,
    stride_mask_b, stride_mask_i, stride_mask_k,
    channel_offset: tl.constexpr,   # start index of the channel block
    BLOCK_D: tl.constexpr,          # number of channels processed (≤32)
):
    """Apply proj * mask * gate element‑wise for a block of channels."""
    pid = tl.program_id(0)
    # decode (b, i, k) from the 1‑D program id
    total_ik = N * N
    b = pid // total_ik
    rem = pid - b * total_ik
    i = rem // N
    k = rem - i * N

    # base offsets for the tensors (without the channel component)
    base_proj = b * stride_proj_b + i * stride_proj_i + k * stride_proj_k
    base_gate = b * stride_gate_b + i * stride_gate_i + k * stride_gate_k
    base_out  = b * stride_out_b  + i * stride_out_i  + k * stride_out_k
    mask_off  = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k

    # scalar mask value (0/1) – cast to float for multiplication
    mask_val = tl.cast(tl.load(mask_ptr + mask_off), tl.float32)

    # channel offsets inside this block
    offs = tl.arange(0, BLOCK_D)
    ch_idx = channel_offset + offs
    ch_mask = ch_idx < D

    # load vectors, apply mask and gate, store result
    proj = tl.load(proj_ptr + base_proj + ch_idx * stride_proj_d,
                   mask=ch_mask, other=0.0)
    gate = tl.load(gate_ptr + base_gate + ch_idx * stride_gate_d,
                   mask=ch_mask, other=0.0)

    out = proj * gate * mask_val
    tl.store(out_ptr + base_out + ch_idx * stride_out_d,
             out, mask=ch_mask)


def _apply_mask_gate(proj, gate, mask, out):
    """
    Fuse projection, mask and gate using Triton.
    proj, gate, out: tensors of shape [B, N, N, D]
    mask: tensor of shape [B, N, N] (broadcasted over D)
    """
    B, N, _, D = proj.shape
    # strides (in elements, not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_out = out.stride()
    s_mask = mask.stride()

    block = 32  # max threads per program (one warp)
    total = B * N * N
    grid = (total,)

    for off in range(0, D, block):
        cur = min(block, D - off)
        mask_gate_kernel[grid](
            proj_ptr=proj,
            gate_ptr=gate,
            mask_ptr=mask,
            out_ptr=out,
            B=B,
            N=N,
            D=D,
            stride_proj_b=s_proj[0],
            stride_proj_i=s_proj[1],
            stride_proj_k=s_proj[2],
            stride_proj_d=s_proj[3],
            stride_gate_b=s_gate[0],
            stride_gate_i=s_gate[1],
            stride_gate_k=s_gate[2],
            stride_gate_d=s_gate[3],
            stride_out_b=s_out[0],
            stride_out_i=s_out[1],
            stride_out_k=s_out[2],
            stride_out_d=s_out[3],
            stride_mask_b=s_mask[0],
            stride_mask_i=s_mask[1],
            stride_mask_k=s_mask[2],
            channel_offset=off,
            BLOCK_D=cur,
            num_warps=1,
        )
        torch.cuda.synchronize()


def custom_kernel(data):
    """
    TriMul forward (outgoing) implementation.
    Input:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Projections and gates (linear, no bias)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Ensure mask is float (0/1) and on the same device
    if mask is None:
        mask = torch.ones(bs, seq_len, seq_len, device=input_tensor.device, dtype=input_tensor.dtype)
    mask = mask.to(input_tensor.dtype)

    # ------------------------------------------------------------------
    # 3. Fuse mask & gates with Triton
    # ------------------------------------------------------------------
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    _apply_mask_gate(left_proj, left_gate, mask, left)
    _apply_mask_gate(right_proj, right_gate, mask, right)

    # free intermediate tensors to lower peak memory
    del left_proj, left_gate, right_proj, right_gate

    # ------------------------------------------------------------------
    # 4. Core O(N³) multiplication via batched GEMM
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    # reshape to (B*D, N, N) matrices
    left_mat = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, seq_len, seq_len)
    right_mat = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, seq_len, seq_len)

    # batched matmul (left * right^T)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
