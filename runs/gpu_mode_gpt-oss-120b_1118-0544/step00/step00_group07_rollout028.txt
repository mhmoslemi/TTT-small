def custom_kernel(data):
    # unpack
    x, mask, weights, config = data
    # get dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]  # shape [dim, hidden_dim]

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # optional mask: shape [B,N,N] possibly None? We have mask param.
    if mask is not None:
        mask_ex = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_ex
        right = right * mask_ex

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Rearrange for kernel
    # permute to (B, hidden_dim, N, N) then view
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left.shape
    left_flat = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)

    # Convert to half for kernel compute
    left_flat_h = left_flat.to(torch.float16)
    right_flat_h = right_flat.to(torch.float16)

    out_flat = torch.empty_like(left_flat_h)  # same dtype

    # Launch Triton kernel
    launch_tri_mul(left_flat_h, right_flat_h, out_flat, block_m=64, block_n=64, block_k=32)

    # Convert out back to float32
    out = out_flat.to(torch.float32)
    # reshape back
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,H)

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # gate
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, to_out_weight)  # shape (B,N,N,dim)

    return out
```

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,   # pointers
    N,                             # sequence length
    B_H,                           # batch * hidden
    stride_bl, stride_il, stride_kl,
    stride_br, stride_jr, stride_kr,
    stride_bo, stride_io, stride_jo,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute global offsets for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # guard against overflow
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch*hidden slice
    left_base = left_ptr + pid_b * stride_bl
    right_base = right_ptr + pid_b * stride_br
    out_base = out_ptr + pid_b * stride_bo

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M, K)
        left_ptrs = left_base + offs_m[:, None] * stride_il + offs_k[None, :] * stride_kl
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load right tile transposed (K, N)
        # original right is (j,k); we need (k,j) for matrix multiply
        right_ptrs = right_base + offs_k[:, None] * stride_kr + offs_n[None, :] * stride_jr
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product across K
        acc += tl.dot(left, right)   # left: (M,K), right: (K,N) -> (M,N)

    # store result
    out_ptrs = out_base + offs_m[:, None] * stride_io + offs_n[None, :] * stride_jo
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    This implementation fuses the core O(N^3) contraction using a custom Triton kernel.
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left = F.linear(x_norm, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x_norm, right_proj_weight)

    # Apply mask if present
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_
        right = right * mask_

    # Gates (float32 for sigmoid stability)
    left_gate = F.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = F.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = F.linear(x_norm, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Rearrange to (B, hidden, N, N) and flatten batch*hidden for the kernel
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left.shape
    left_flat = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)

    # Cast to half for kernel computation
    left_half = left_flat.to(torch.float16)
    right_half = right_flat.to(torch.float16)
    out_half = torch.empty_like(left_half)

    # Triton kernel launch
    _tri_mul_launch(left_half, right_half, out_half, BLOCK_M=64, BLOCK_N=64, BLOCK_K=32)

    # Convert back to float32 and reshape to original layout
    out = out_half.to(torch.float32)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # Post‑processing LayerNorm over hidden dimension
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to original dimensionality
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    return out

"""
TriMul (outgoing) forward pass with a fused O(N³) contraction.

The kernel computes, for each batch × hidden channel,
    out[i, j] = Σₖ left[i, k] * right[j, k]

which is equivalent to the einsum
    out[..., i, j, d] = Σₖ left[..., i, k, d] * right[..., j, k, d].

All surrounding LayerNorms, linear projections and gates are performed in
PyTorch; the heavy contraction is implemented in Triton with tiling over
the sequence dimension.
"""
import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    N, B_H,                               # sequence length, batch*hidden
    stride_bl, stride_il, stride_kl,      # left strides: batch, i, k
    stride_br, stride_jr, stride_kr,      # right strides: batch, j, k
    stride_bo, stride_io, stride_jo,      # out strides: batch, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch * hidden channel
    pid_m = tl.program_id(1)   # output row tile
    pid_n = tl.program_id(2)   # output col tile

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this batch*hidden slice
    left_base  = left_ptr  + pid_b * stride_bl
    right_base = right_ptr + pid_b * stride_br
    out_base   = out_ptr   + pid_b * stride_bo

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: (M, K)
        left_ptrs = left_base + offs_m[:, None] * stride_il + offs_k[None, :] * stride_kl
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # Load right tile transposed: (K, N)
        # original right shape is (batch, j, k); we need (k, j)
        right_ptrs = right_base + offs_k[:, None] * stride_kr + offs_n[None, :] * stride_jr
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        # Accumulate dot product over K
        acc += tl.dot(left, right)   # (M, N)

    # Write the result
    out_ptrs = out_base + offs_m[:, None] * stride_io + offs_n[None, :] * stride_jo
    tl.store(out_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def _launch_tri_mul(left: torch.Tensor, right: torch.Tensor, out: torch.Tensor,
                    BLOCK_M=64, BLOCK_N=64, BLOCK_K=32):
    """Launch the Triton kernel for the tiled contraction."""
    B_H, N, _ = left.shape
    grid = (B_H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _tri_mul_kernel[grid](
        left, right, out,
        N, B_H,
        left.stride(0), left.stride(1), left.stride(2),
        right.stride(0), right.stride(1), right.stride(2),
        out.stride(0), out.stride(1), out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (see code)
            - config: dict containing 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------------------- unpack weights ----------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ---------------------- input LayerNorm ----------------------
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight,
                          bias=norm_bias, eps=eps)

    # ---------------------- linear projections ------------------
    left  = F.linear(x_norm, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # ---------------------- optional mask ----------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x_norm.dtype)  # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ---------------------- gates (sigmoid) -------------------
    left_gate  = F.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = F.linear(x_norm, right_gate_weight).sigmoid()
    out_gate   = F.linear(x_norm, out_gate_weight).sigmoid()

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # ---------------------- prepare tensors for kernel -------
    # Move hidden dimension to the second axis for efficient flattening
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left.shape

    # Collapse B and H into a single batch dimension
    left_flat  = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)

    # Cast to half-precision for the kernel (fast WMMA path)
    left_half  = left_flat.to(torch.float16)
    right_half = right_flat.to(torch.float16)
    out_half   = torch.empty_like(left_half)

    # ---------------------- Triton contraction ---------------
    _launch_tri_mul(left_half, right_half, out_half,
                    BLOCK_M=64, BLOCK_N=64, BLOCK_K=32)

    # Convert back to FP32 and restore original layout
    out = out_half.to(torch.float32)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ---------------------- post‑processing -------------------
    # LayerNorm over hidden dimension
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    # Output gate
    out = out * out_gate

    # Final linear projection back to `dim`
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
