def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # read dim etc.
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    B,N,N2, c = input_tensor.shape

    # Convert to float16 for speed
    x = input_tensor
    # LayerNorm
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    # x_norm = (x - mean)/std * norm_weight + norm_bias.
    # We need per-last-dim LayerNorm over dim.
    # Use torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)
    # Linear projections
    left_proj_weight = weights['left_proj.weight']  # shape (hidden_dim, dim)
    right_proj_weight = weights['right_proj.weight']
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    # mask
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)  # shape (B,N,N,1)
        left = left * mask
        right = right * mask

    # gating

    right_gate = torch.nn.functional.linear(x_norm, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights['out_gate.weight']).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Cast to float16 for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Prepare output tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # compute strides
    stride_b_l, stride_i_l, stride_k_l, stride_d_l = left.stride()
    stride_b_r, stride_i_r, stride_k_r, stride_d_r = right.stride()
    stride_b_o, stride_i_o, stride_j_o, stride_d_o = out.stride()

    # block size
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)

        stride_b_r, stride_i_r, stride_k_r, stride_d_r,
        stride_b_o, stride_i_o, stride_j_o, stride_d_o,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
    )

    out = out.to(torch.float32)

    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

The algorithm follows the AlphaFold2 TriangularMultiplicativeUpdateOutgoing:
1. Layer‑norm the input (last dim).
2. Linear projections to a hidden dimension (left/right) and apply an optional mask.
3. Gated projection (sigmoid gates) on left/right.
4. Core computation: for each hidden slice d,
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   This is a batched matrix multiplication left @ rightᵀ.
   The reduction is performed with a Triton kernel that tiles the
   i, j and k dimensions and processes several hidden channels per launch.
5. Apply a second layer‑norm, element‑wise output gate and a final linear
   projection back to the original dimension.

The Triton kernel tiles i‑j blocks (BLOCK_M×BLOCK_N), k‑tiles (BLOCK_K)
and processes BLOCK_D hidden channels together.  All arithmetic is done
in float16 (Tensor‑core friendly) while accumulation uses float32 for
numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_b_l, stride_i_l, stride_k_l, stride_d_l,
    stride_b_r, stride_i_r, stride_k_r, stride_d_r,
    stride_b_o, stride_i_o, stride_j_o, stride_d_o,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_m = tl.program_id(0)  # i‑block
    pid_n = tl.program_id(1)  # j‑block
    pid_s = tl.program_id(2)  # batch * hidden‑slice

    # --------------------
    # Compute batch index and hidden‑slice start
    num_hidden_slices = (D + BLOCK_D - 1) // BLOCK_D
    b = pid_s // num_hidden_slices
    hidden_slice = pid_s - b * num_hidden_slices
    d_start = hidden_slice * BLOCK_D

    # --------------------
    # Offsets for i, j and d dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = d_start + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # Accumulator for the output tile
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over k dimension in tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # --------------------
        # Load left tile: shape (BLOCK_M, BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr + \
            b * stride_b_l + \
            offs_i[:, None, None] * stride_i_l + \
            offs_k[None, :, None] * stride_k_l + \
            offs_d[None, None, :] * stride_d_l
        left_mask = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0, dtype=tl.float16).to(tl.float32)

        # --------------------
        # Load right tile (j‑dimension): shape (BLOCK_N, BLOCK_K, BLOCK_D)
        right_ptrs = right_ptr + \
            b * stride_b_r + \
            offs_j[:, None, None] * stride_i_r + \
            offs_k[None, :, None] * stride_k_r + \
            offs_d[None, None, :] * stride_d_r
        right_mask = mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0, dtype=tl.float16).to(tl.float32)

        # --------------------
        # Compute per‑channel dot products: out[i,j,d] += Σ_k left[i,k,d] * right[j,k,d]
        # Loop over the hidden channels inside the tile
        for d in range(0, BLOCK_D):
            # left[:, :, d] : (BLOCK_M, BLOCK_K)
            # right[:, :, d]: (BLOCK_N, BLOCK_K)
            acc[:, :, d] += tl.dot(left[:, :, d], right[:, :, d], trans_b=True)

    # --------------------
    # Store the accumulated block
    out_ptrs = out_ptr + \
        b * stride_b_o + \
        offs_i[:, None, None] * stride_i_o + \
        offs_j[None, :, None] * stride_j_o + \
        offs_d[None, None, :] * stride_d_o
    out_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=out_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, N2, _ = input_tensor.shape
    assert N == N2, "Input must be square in the sequence dimensions."

    # ----------------------------------------------------------------------
    # LayerNorm (first)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ----------------------------------------------------------------------
    # Linear projections (left/right) – bias is zero in the reference code
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----------------------------------------------------------------------
    # Optional mask (broadcast over hidden_dim)
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------------
    # Gating (sigmoid)
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # ----------------------------------------------------------------------
    # Core Triton kernel: compute Σ_k left[...,k,d] * right[...,k,d]
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Strides for left/right/out (in elements, not bytes)
    stride_b_l, stride_i_l, stride_k_l, stride_d_l = left.stride()
    stride_b_r, stride_i_r, stride_k_r, stride_d_r = right.stride()
    stride_b_o, stride_i_o, stride_j_o, stride_d_o = out.stride()

    # Tiling parameters – chosen to fit well on typical GPUs
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * ((hidden_dim + BLOCK_D - 1) // BLOCK_D),
    )

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        bs,
        N,
        hidden_dim,
        stride_b_l,
        stride_i_l,
        stride_k_l,
        stride_d_l,
        stride_b_r,
        stride_i_r,
        stride_k_r,
        stride_d_r,
        stride_b_o,
        stride_i_o,
        stride_j_o,
        stride_d_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ----------------------------------------------------------------------
    # Continue in PyTorch (float32 path)
    out = out.to(torch.float32)

    # Output LayerNorm, gating and final linear projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)

    return out
