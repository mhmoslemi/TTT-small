def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    dtype = torch.float16

    # Cast inputs and weights to dtype
    x = input_tensor.to(dtype)
    if mask_tensor is None or config.get("nomask", True):
        mask = torch.ones((bs, seq_len, seq_len), dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(dtype)

    # Load weights
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias = weights["norm.bias"].to(dtype)
    left_proj_weight = weights["left_proj.weight"].to(dtype)
    right_proj_weight = weights["right_proj.weight"].to(dtype)
    left_gate_weight = weights["left_gate.weight"].to(dtype)
    right_gate_weight = weights["right_gate.weight"].to(dtype)
    out_gate_weight = weights["out_gate.weight"].to(dtype)
    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(dtype)
    to_out_weight = weights["to_out.weight"].to(dtype)

    # LayerNorm on input dim (dim)
    eps = 1e-5
    # Using functional layer_norm: normalized_shape = (dim,)
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections and gates
    # For efficiency we may want to use torch.nn.functional.linear. It supports half.
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape B,N,N,hidden_dim
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))

    # Apply mask and gates via Triton kernel
    # Flatten tensors for kernel
    total = bs * seq_len * seq_len * hidden_dim

    # Output tensors for left and right after mask+gate
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Ensure mask flattened
    mask_flat = mask.reshape(-1)  # shape B*N*N

    # Launch kernel
    BLOCK_SIZE = 1024
    grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

    # left

    # right

                                 in_ptr=right_proj,
                                 gate_ptr=right_gate,
                                 mask_ptr=mask_flat,
                                 B=bs,
                                 N=seq_len,
                                 H=hidden_dim,
                                 TOTAL=total,
                                 BLOCK_SIZE=BLOCK_SIZE,
                                 num_warps=4)

    right_f = right.permute(0, 3, 1, 2).contiguous()

@triton.jit
def fused_mask_gate_kernel(out_ptr, in_ptr, gate_ptr, mask_ptr,
                           B, N, H, TOTAL,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL

    # Load x and gate values
    x = tl.load(in_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    mask_idx = offs // H   # floor division
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # Compute output
    out = x * g * m

    tl.store(out_ptr + offs, out, mask=mask)

import torch
import triton
import triton.language as tl

@triton.jit
def fused_mask_gate_kernel(out_ptr, in_ptr, gate_ptr, mask_ptr,
        B, N, H,
        TOTAL,
        BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL

    # Load input, gate and mask values (half precision)
    x = tl.load(in_ptr + offs, mask=mask, other=tl.float16(0.0))
    g = tl.load(gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))

    out = x * g * m
    tl.store(out_ptr + offs, out, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul module.
    Steps:
      1. Layer‑norm the input.
      2. Linear projections to hidden_dim.
      3. Compute sigmoid gates.
      4. Fuse mask, projection and gate with a custom Triton kernel.
      5. Batched matrix multiplication per hidden channel (left @ rightᵀ).
      6. Layer‑norm the result, apply the output gate and final linear.
    All heavy element‑wise work (mask × gate × proj) is done by a Triton kernel.
    The final linear projection returns fp32 output.
    """
    # Unpack
    input_tensor, mask_tensor, weights, config = data
    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = torch.float16  # use half precision for speed

    # Prepare mask (if missing or nomask → all ones)
    nomask = config.get("nomask", True)
    if mask_tensor is None or nomask:
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(dtype)

    # Cast inputs and weights to half precision
    x = input_tensor.to(dtype)

    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias   = weights["norm.bias"].to(dtype)

    left_proj_weight  = weights["left_proj.weight"].to(dtype)
    right_proj_weight = weights["right_proj.weight"].to(dtype)

    left_gate_weight  = weights["left_gate.weight"].to(dtype)
    right_gate_weight = weights["right_gate.weight"].to(dtype)

    out_gate_weight   = weights["out_gate.weight"].to(dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(dtype)

    to_out_weight = weights["to_out.weight"].to(dtype)

    eps = 1e-5

    # 1. LayerNorm over the last channel dimension (C)
    x_norm = torch.nn.functional.layer_norm(
        x, (C,), weight=norm_weight, bias=norm_bias, eps=eps
    )

    # 2. Linear projections (no bias)
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)    # [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    # 3. Gates (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))

    # 4. Fuse mask, projection and gate using Triton
    total = B * N * N * hidden_dim
    BLOCK_SIZE = 1024
    grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

    # Flatten mask to 1‑D (B*N*N)
    mask_flat = mask.reshape(-1)

    # Allocate output tensors for the gated & masked values
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Launch kernel for left
    fused_mask_gate_kernel[grid](
        out_ptr=left,
        in_ptr=left_proj,
        gate_ptr=left_gate,
        mask_ptr=mask_flat,
        B=B, N=N, H=hidden_dim,
        TOTAL=total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    # Launch kernel for right
    fused_mask_gate_kernel[grid](
        out_ptr=right,
        in_ptr=right_proj,
        gate_ptr=right_gate,
        mask_ptr=mask_flat,
        B=B, N=N, H=hidden_dim,
        TOTAL=total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # 5. Batched matrix multiplication per hidden channel:
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    This is equivalent to (B, hidden_dim, N, N) @ (B, hidden_dim, N, N)^T
    left_f = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_f = right.permute(0, 3, 1, 2).contiguous()

    out_f = torch.matmul(left_f, right_f.transpose(-2, -1))  # (B, H, N, N)
    out = out_f.permute(0, 2, 3, 1).contiguous()            # (B, N, N, H)

    # 6. Output layer‑norm (hidden_dim)
    out_norm = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps
    )

    # 7. Output gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # 8. Apply gate and final linear projection back to `dim`
    out = out_norm * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)

    # Return fp32 tensor as required by the interface
    return out.to(torch.float32)

def fused_mask_gate_kernel(
    out_ptr, in_ptr, gate_ptr, mask_ptr,
    B, N, H,
    TOTAL,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse element‑wise mask, projection and gate:
        out = in * gate * mask

    Tensors are viewed as 1‑D contiguous buffers.
    Each hidden channel (size H) shares the same mask entry,
    therefore mask index = offset // H.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL

    # Load values (half precision)
    x = tl.load(in_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    # mask index repeats every H elements
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    out = x * g * m
    tl.store(out_ptr + offs, out, mask=mask)
