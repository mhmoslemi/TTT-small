def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm
    eps=1e-5
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # optional mask
    nomask = config.get("nomask", True)
    if not nomask:
        # mask shape (bs, N, N)
        mask_exp = mask.unsqueeze(-1).type_as(left)
        left = left * mask_exp
        right = right * mask_exp

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Contraction using triton
    out_mul = torch.empty_like(left)  # shape B,N,N,hidden_dim

    # Launch kernel
    launch_trimul_kernel(left, right, out_mul)

    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(out_mul, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    out_norm = out_norm * out_gate

    # final linear
    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out

"""
custom_kernel: TriMul (outgoing) forward implementation

The kernel fuses the O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
for all batches `b` and hidden channels `d`.  The contraction is expressed as a batch
matrix multiplication L_d @ R_dᵀ for each hidden dimension `d`.  A Triton kernel
computes this using tiled GEMM (block M×N×K) with half‑precision loads and
float‑32 accumulation.  All other operations (LayerNorm, Linear projections,
sigmoids, and the final output projection) are performed in PyTorch.

The algorithm:
1. Layer‑norm the input.
2. Linear projections to `hidden_dim`.
3. (Optional) mask the projected tensors.
4. Compute three gating tensors via sigmoid(linear(x)).
5. Apply left/right gates.
6. Use Triton to compute the pairwise contraction.
7. Layer‑norm the contracted tensor, apply output gate and final linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel for the core contraction (outgoing TriMul)
# -------------------------------------------------------------
@triton.jit
def _triu_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # tiling over i, j, (batch,hidden)
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)
    tiles_per_pair = num_tiles_i * num_tiles_j            # tiles for one (b,h)
    pair_idx   = pid // tiles_per_pair                     # flattened (b,h) index
    tile_idx   = pid % tiles_per_pair
    i_tile = tile_idx // num_tiles_j
    j_tile = tile_idx % num_tiles_j

    b = pair_idx // H
    h = pair_idx % H

    i_off = i_tile * BLOCK_M
    j_off = j_tile * BLOCK_N

    # out‑of‑bounds masks for the tile
    i_valid = i_off + tl.arange(0, BLOCK_M) < N
    j_valid = j_off + tl.arange(0, BLOCK_N) < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k_off in range(0, N, BLOCK_K):
        k_valid = k_off + tl.arange(0, BLOCK_K) < N

        # ----------- load left tile (i, k) -------------
        i_idx = i_off + tl.arange(0, BLOCK_M)[:, None]
        k_idx = k_off + tl.arange(0, BLOCK_K)[None, :]
        left_addr = (
            left_ptr
            + b * stride_left_b
            + i_idx * stride_left_i
            + k_idx * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(left_addr,
                       mask=i_valid[:, None] & k_valid[None, :],
                       other=0.0,
                       dtype=tl.float16)
        left = left.to(tl.float32)

        # ----------- load right tile (k, j) ------------
        j_idx = j_off + tl.arange(0, BLOCK_N)[None, :]
        # note: for the transposed view we index right as (b, j, k, h)
        right_addr = (
            right_ptr
            + b * stride_right_b
            + j_idx * stride_right_i    # stride of the second N dim
            + k_idx * stride_right_k    # stride of the third N dim
            + h * stride_right_h
        )
        right = tl.load(right_addr,
                        mask=k_valid[:, None] & j_valid[None, :],
                        other=0.0,
                        dtype=tl.float16)
        right = right.to(tl.float32)

        # ----------- accumulate dot product -------------
        acc += tl.dot(left, right)   # (M,K)·(K,N) → (M,N)

    # ----------- store result (back to FP16) ----------
    i_idx = i_off + tl.arange(0, BLOCK_M)[:, None]
    j_idx = j_off + tl.arange(0, BLOCK_N)[None, :]
    out_addr = (
        out_ptr
        + b * stride_out_b
        + i_idx * stride_out_i
        + j_idx * stride_out_j
        + h * stride_out_h
    )
    tl.store(out_addr,
             acc.to(tl.float16),
             mask=i_valid[:, None] & j_valid[None, :])


def _launch_triu_contract(left: torch.Tensor, right: torch.Tensor, out: torch.Tensor):
    """
    Launch the Triton contraction kernel.
    left / right / out must be contiguous and have shape [B, N, N, H].
    """
    assert left.is_contiguous() and right.is_contiguous() and out.is_contiguous()
    B, N, _, H = left.shape
    # block sizes – chosen to give good occupancy on the B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # strides (in elements)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # total number of program instances
    tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    tiles_j = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = B * H * tiles_i * tiles_j

    grid = (total_tiles,)

    _triu_contract_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_ri,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # enough warps for good latency hiding
        num_stages=2,
    )


# -------------------------------------------------------------
# Entry point required by the evaluator
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (may be None)
        - weights      : dict of named weight tensors
        - config       : dict containing 'dim', 'hidden_dim', 'nomask' flag

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input (last dimension = dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections to hidden dimension
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask (applied before gating)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # ensure same dtype for broadcasting
        m = mask.to(left.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * m
        right = right * m

    # -----------------------------------------------------------------
    # 4) Gates (sigmoid of linear layers)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 5) Apply left/right gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6) Cast to FP16 for the heavy contraction
    # -----------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # output buffer for the contraction (FP16)
    out_mul_fp16 = torch.empty_like(left_fp16)

    # -----------------------------------------------------------------
    # 7) Triton contraction: out_mul[b,i,j,h] = Σ_k left[b,i,k,h]*right[b,j,k,h]
    # -----------------------------------------------------------------
    _launch_triu_contract(left_fp16, right_fp16, out_mul_fp16)

    # Cast back to FP32 for the remaining FP32 ops
    out_mul = out_mul_fp16.to(torch.float32)          # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 8) Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out_mul,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate                      # broadcast over hidden dim
    out = F.linear(out_norm, weights["to_out.weight"])  # back to [B, N, N, dim]

    return out
