    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # layer norm
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])  # shape [..., hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask apply if not nomask
    if not config.get("nomask", False):

        left = left * mask
        right = right * mask

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # now compute tri mul using Triton
    # ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # shape [B,N,N,hidden_dim]

    # obtain strides
    B, N, _, _ = left.shape  # left shape [B,N,N,hidden_dim]

    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        B * hidden_dim,  # program id axis 0

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    d = pid0 % D
    b = pid0 // D

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # create offsets for i and j within block
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # create mask for valid i, j
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        # compute offsets for left: (i, k, d)
        left_ptrs = left_ptr + b*stride_left_b + d*stride_left_d + offs_i[:, None] * stride_left_i + (k0 + offs_k)[None, :] * stride_left_k
        # compute offsets for right: (j, k, d)
        right_ptrs = right_ptr + b*stride_right_b + d*stride_right_d + offs_j[:, None] * stride_right_i + (k0 + offs_k)[None, :] * stride_right_k

        # mask for k dimension inside load: need to ensure k0+offs_k < N
        mask_k = (k0 + offs_k) < N

        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        acc += tl.dot(left, right, trans_b=True)  # (M,K)@(K,N) -> (M,N)

    # store result
    out_ptrs = out_ptr + b*stride_out_b + d*stride_out_d + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_ij)

"""
TriMul "outgoing" implementation with Triton acceleration.

The heavy O(N³·C) part (einsum '... i k d, ... j k d -> ... i j d') is
compiled as a Triton kernel that performs a batched matrix multiplication
for each hidden channel `d`. The rest of the forward pass (LayerNorms,
Linear projections, gating) is done with PyTorch ops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, D,                                    # sizes: batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]"""
    pid0 = tl.program_id(0)   # combines batch and hidden channel
    pid_i = tl.program_id(1)  # block of output rows   (i direction)
    pid_j = tl.program_id(2)  # block of output cols   (j direction)

    # Decode batch and hidden dimension from pid0
    d = pid0 % D
    b = pid0 // D

    # Starting offsets of the current block
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Offsets within the block
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for boundary handling
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        # --- Load left block (M x K) ---
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + d * stride_left_d
            + offs_i[:, None] * stride_left_i
            + (k0 + offs_k)[None, :] * stride_left_k
        )
        mask_k = (k0 + offs_k) < N
        mask_left = mask_i[:, None] & mask_k[None, :]
        left = tl.load(left_ptrs, mask=mask_left, other=0.0).to(tl.float32)

        # --- Load right block (N x K) ---
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + d * stride_right_d
            + offs_j[:, None] * stride_right_i
            + (k0 + offs_k)[None, :] * stride_right_k
        )
        mask_right = mask_j[:, None] & mask_k[None, :]
        right = tl.load(right_ptrs, mask=mask_right, other=0.0).to(tl.float32)

        # Accumulate dot product: left (M,K) @ right.T (K,N) -> (M,N)
        acc += tl.dot(left, right, trans_b=True)

    # Store the result
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + d * stride_out_d
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc, mask=mask_ij)


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.
    Arguments:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Config parameters
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----- 1) Input LayerNorm -----
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ----- 2) Linear projections (no bias) -----
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ----- 3) Optional mask -----
    if not nomask:
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ----- 4) Gating -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----- 5) Core TriMul via Triton -----
    # Ensure contiguity for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, _ = left.shape
    out = torch.empty_like(left)  # [B,N,N,hidden_dim]

    # Strides (in elements, not bytes)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        B * hidden_dim,                              # program ID axis 0 (batch * hidden)
        (N + BLOCK_M - 1) // BLOCK_M,                # axis 1 – i blocks
        (N + BLOCK_N - 1) // BLOCK_N,                # axis 2 – j blocks
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()  # ensure kernel completion

    # ----- 6) Output LayerNorm, gating, final projection -----
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # final projection to [dim]
    return out
