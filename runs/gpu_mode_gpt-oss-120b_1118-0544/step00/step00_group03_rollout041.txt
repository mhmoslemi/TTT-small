def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    B,N,N_,dim = input_tensor.shape; assert N==N_
    hidden_dim = config['hidden_dim']

    # Device
    device = input_tensor.device
    dtype = input_tensor.dtype
    
    # LayerNorm on dim
    # norm_weight, norm_bias shape (dim,)
    # compute normalized input: (x - mean) / sqrt(var + eps) * weight + bias
    # Instead of using torch.nn.LayerNorm default eps=1e-5.
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # compute linear projections:
    # left_proj = linear with weight shape (hidden_dim, dim) (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # compute gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # apply gating
    left = left * left_gate
    right = right * right_gate

    # apply mask on left/right? We'll do in kernel.

    # Compute core einsum using Triton kernel
    out_hidden = torch.empty((B,N,N,hidden_dim), dtype=torch.float32, device=device)

    # Determine whether we need mask
    apply_mask = not config.get("nomask", False) and mask_tensor is not None

    # If mask tensor is not provided or we don't apply, we can pass dummy
    if not apply_mask:
        dummy_mask = torch.empty((1,), dtype=torch.float32, device=device)
        mask_ptr = dummy_mask
    else:
        mask_ptr = mask_tensor  # shape (B,N,N)

    # Get strides (in elements)
    left_stride = left.stride()
    right_stride = right.stride()
    out_stride = out_hidden.stride()
    if apply_mask:
        mask_stride = mask_tensor.stride()
    else:
        mask_stride = (0,0,0)  # placeholder

    # Launch kernel
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_D = 32
    # Must ensure these constants not exceed dimension.
    # If hidden_dim not divisible by BLOCK_D we still handle by masking.

    grid_i = (N + BLOCK_I - 1) // BLOCK_I
    grid_j = (N + BLOCK_J - 1) // BLOCK_J
    num_d_blocks = (hidden_dim + BLOCK_D - 1) // BLOCK_D
    grid_z = B * num_d_blocks

    trilmul_outgoing_kernel[grid_i, grid_j, grid_z](
        left, right, out_hidden,
        mask_ptr,
        B, N, hidden_dim,
        left_stride[0], left_stride[1], left_stride[2], left_stride[3],
        right_stride[0], right_stride[1], right_stride[2], right_stride[3],
        out_stride[0], out_stride[1], out_stride[2], out_stride[3],
        mask_stride[0] if apply_mask else 0,
        mask_stride[1] if apply_mask else 0,
        mask_stride[2] if apply_mask else 0,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_D=BLOCK_D,
        APPLY_MASK=apply_mask,
    )

    # Now out_hidden is computed.

    # Apply to_out_norm (layernorm) and out_gate multiplication, then final linear.
    out_hidden_norm = torch.nn.functional.layer_norm(out_hidden, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    out = out_hidden_norm * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

@triton.jit
def trilmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    mask_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK_D: tl.constexpr,
    APPLY_MASK: tl.constexpr,
):
    i = tl.program_id(0)  # i index
    j = tl.program_id(1)  # j index
    b = tl.program_id(2)  # batch index

    # d range
    d_offs = tl.arange(0, BLOCK_D)

    # guard for i,j < N, etc.
    i_valid = i < N
    j_valid = j < N

    # Compute offset base for left, right, out.
    # left[i,k,d]: pointer: left_ptr + b*stride_left_b + i*stride_left_i + k*stride_left_k + d_offs*stride_left_d
    # right[j,k,d]: pointer: right_ptr + b*stride_right_b + j*stride_right_j + k*stride_right_k + d_offs*stride_right_d

    acc = tl.zeros((BLOCK_D,), dtype=tl.float32)

    for k in range(0, N):
        k_valid = k < N
        # load left vector
        left_offset = b*stride_left_b + i*stride_left_i + k*stride_left_k + d_offs*stride_left_d
        left = tl.load(left_ptr + left_offset, mask=i_valid & k_valid & (d_offs < D), other=0.0).to(tl.float32)
        right_offset = b*stride_right_b + j*stride_right_j + k*stride_right_k + d_offs*stride_right_d
        right = tl.load(right_ptr + right_offset, mask=j_valid & k_valid & (d_offs < D), other=0.0).to(tl.float32)

        if APPLY_MASK:
            # mask_i
            mask_i_offset = b*stride_mask_b + i*stride_mask_i + k*stride_mask_k
            mask_i = tl.load(mask_ptr + mask_i_offset, mask=i_valid & k_valid, other=0.0).to(tl.float32)
            left = left * mask_i

            # mask_j
            mask_j_offset = b*stride_mask_b + j*stride_mask_i + k*stride_mask_k
            mask_j = tl.load(mask_ptr + mask_j_offset, mask=j_valid & k_valid, other=0.0).to(tl.float32)
            right = right * mask_j

        acc += left * right

    # Store out
    out_offset = b*stride_out_b + i*stride_out_i + j*stride_out_j + d_offs*stride_out_d
    tl.store(out_ptr + out_offset, acc, mask=i_valid & j_valid & (d_offs < D))

def custom_kernel(data):
    """TriMul outgoing implementation using Triton kernel for the core reduction.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates (mask will be applied inside kernel)
    left = left * left_gate
    right = right * right_gate

    # Ensure tensors are contiguous for correct strides
    left = left.contiguous()
    right = right.contiguous()

    # Prepare output tensor
    out_hidden = torch.empty((left.shape[0], left.shape[1], left.shape[2], hidden_dim), dtype=torch.float32, device=device)

    # Determine mask handling
    apply_mask = (not config.get("nomask", False)) and (mask_tensor is not None)
    if not apply_mask:
        # dummy mask (won't be read)
        dummy_mask = torch.empty((1,), dtype=torch.float32, device=device)
        mask_tensor = dummy_mask
        mask_stride = (0,0,0)
    else:
        mask_tensor = mask_tensor.contiguous()
        mask_stride = mask_tensor.stride()

    # Get strides
    left_stride = left.stride()
    right_stride = right.stride()
    out_stride = out_hidden.stride()
    if apply_mask:
        mask_stride = mask_tensor.stride()
    else:
        mask_stride = (0,0,0)

    # Kernel launch
    Bsz, N, _, _ = left.shape

    GRID_I = (N + BLOCK_I - 1) // BLOCK_I
    GRID_J = (N + BLOCK_J - 1) // BLOCK_J
    # Since we compute one output element per (i,j) program, we set BLOCK_I=1? Actually we use per program for each i and j.
    # We'll use program_id directly as i,j (no blocking). So grid_i = N, grid_j = N.
    grid = (N, N, Bsz)  # each program for (i,j,b)

    trilmul_einsum_kernel[grid](
        left,
        right,
        out_hidden,
        mask_tensor,
        Bsz,
        N,
        hidden_dim,
        left_stride[0], left_stride[1], left_stride[2], left_stride[3],

        mask_stride[1] if apply_mask else 0,
        mask_stride[2] if apply_mask else 0,
        BLOCK_D=hidden_dim,
        APPLY_MASK=int(apply_mask),
    )

    # Continue with to_out_norm, gating, final linear

        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],

    )

def trilmul_einsum_kernel(
    left_ptr, right_ptr, out_ptr, mask_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK_D: tl.constexpr,
    APPLY_MASK: tl.constexpr,
):

    j = tl.program_id(1)
    b = tl.program_id(2)

    d_idx = tl.arange(0, BLOCK_D)

    i_valid = i < N
    j_valid = j < N
    d_valid = d_idx < D

    # accumulator
    acc = tl.zeros((BLOCK_D,), dtype=tl.float32)

The heavy reduction `einsum('... i k d, ... j k d -> ... i j d')` is
performed by a custom Triton kernel that computes, for each output
position (b, i, j, d),

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

The rest of the module (LayerNorm, Linear projections, gating) is
implemented in PyTorch. The kernel operates on one output element
(i, j) per program (grid entry) and loops over the reduction axis k.
Masking is fused into the kernel when required.

import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel for the core pairwise multiplication (the einsum)
# ----------------------------------------------------------------------------- #
@triton.jit
def trilmul_einsum_kernel(
    left_ptr, right_ptr, out_ptr, mask_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK_D: tl.constexpr,
    APPLY_MASK: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].

    One program works on a single (i,j,b) element and iterates over k.
    """
    i = tl.program_id(0)               # seq index i
    j = tl.program_id(1)               # seq index j
    b = tl.program_id(2)               # batch index

    # range over hidden dimension (compile‑time block size)
    d_idx = tl.arange(0, BLOCK_D)

    # bounds checks
    i_valid = i < N
    j_valid = j < N
    d_valid = d_idx < D

    # accumulator for the hidden dimension
    acc = tl.zeros((BLOCK_D,), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, N):
        # ----- load left[b, i, k, d] -----
        left_offset = (
            b * stride_left_b +
            i * stride_left_i +
            k * stride_left_k +
            d_idx * stride_left_d
        )
        left = tl.load(left_ptr + left_offset,
                       mask=i_valid & d_valid,
                       other=0.0).to(tl.float32)

        # ----- load right[b, j, k, d] -----
        right_offset = (
            b * stride_right_b +
            j * stride_right_j +
            k * stride_right_k +
            d_idx * stride_right_d
        )
        right = tl.load(right_ptr + right_offset,
                        mask=j_valid & d_valid,
                        other=0.0).to(tl.float32)

        # ----- optional mask multiplication -----
        if APPLY_MASK:
            # mask for left (b,i,k)
            mask_i_offset = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k
            mask_i = tl.load(mask_ptr + mask_i_offset,
                             mask=i_valid,
                             other=0.0).to(tl.float32)
            left = left * mask_i

            # mask for right (b,j,k)
            mask_j_offset = b * stride_mask_b + j * stride_mask_i + k * stride_mask_k
            mask_j = tl.load(mask_ptr + mask_j_offset,
                             mask=j_valid,
                             other=0.0).to(tl.float32)
            right = right * mask_j

        # accumulate product over k
        acc += left * right

    # ----- write result -----
    out_offset = (
        b * stride_out_b +
        i * stride_out_i +
        j * stride_out_j +
        d_idx * stride_out_d
    )
    tl.store(out_ptr + out_offset,
             acc,
             mask=i_valid & j_valid & d_valid)


# ----------------------------------------------------------------------------- #
# Public entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # = c_z in the paper, always 128 in tests

    # --------------------------------------------------------------------- #
    # 1. LayerNorm on the input (dim‑wise)
    # --------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------------- #
    # 2. Linear projections
    # --------------------------------------------------------------------- #
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    # 3. Gates (sigmoid of linear)
    # --------------------------------------------------------------------- #
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply elementwise gates (mask will be fused later)
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for correct stride handling
    left = left.contiguous()
    right = right.contiguous()

    # --------------------------------------------------------------------- #
    # 4. Core reduction via Triton kernel
    # --------------------------------------------------------------------- #
    B, N, _, _ = left.shape                     # batch, sequence length
    D = hidden_dim                               # hidden dimension for the reduction

    # Output tensor of the reduction (shape [B, N, N, D])
    out_hidden = torch.empty((B, N, N, D), dtype=torch.float32, device=device)

    # Decide whether we need to apply the mask inside the kernel
    apply_mask = (not config.get("nomask", False)) and (mask_tensor is not None)
    if not apply_mask:
        # Dummy mask – never read because APPLY_MASK=0
        mask_tensor = torch.empty((1,), dtype=torch.float32, device=device)
        mask_stride = (0, 0, 0)
    else:
        mask_tensor = mask_tensor.contiguous()
        mask_stride = mask_tensor.stride()

    # Gather strides (in elements, not bytes)
    left_stride = left.stride()      # (B, N, N, D)
    right_stride = right.stride()
    out_stride = out_hidden.stride()
    mask_stride_vals = mask_stride

    # Launch kernel: one program per (i, j, b) element
    grid = (N, N, B)

    trilmul_einsum_kernel[grid](
        left,
        right,
        out_hidden,
        mask_tensor,
        B,
        N,
        D,
        left_stride[0], left_stride[1], left_stride[2], left_stride[3],
        right_stride[0], right_stride[1], right_stride[2], right_stride[3],
        out_stride[0], out_stride[1], out_stride[2], out_stride[3],
        mask_stride_vals[0] if apply_mask else 0,
        mask_stride_vals[1] if apply_mask else 0,
        mask_stride_vals[2] if apply_mask else 0,
        BLOCK_D=hidden_dim,               # compile‑time constant
        APPLY_MASK=int(apply_mask),       # constexpr bool
    )

    # --------------------------------------------------------------------- #
    # 5. Output projection
    # --------------------------------------------------------------------- #
    # LayerNorm on hidden dimension
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Apply out_gate
    out_hidden = out_hidden * out_gate

    # Final linear projection back to original dim
    out = torch.nn.functional.linear(out_hidden, weights["to_out.weight"])

    return out
