Define kernel:

```
@triton.jit
def tri_mul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    total_instances = B * H
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    # Compute instance idx, tile indices
    instance_idx = pid // (num_blocks_m * num_blocks_n)
    tile_id = pid % (num_blocks_m * num_blocks_n)
    block_m = tile_id // num_blocks_n
    block_n = tile_id % num_blocks_n

    # Compute b and h from instance_idx
    b = instance_idx // H
    h = instance_idx % H

    # Offsets for i, j, k
    i_offset = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    # compute pointer ranges, also mask out-of-bounds
    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_k = k_offset < N    # we'll use in loop but in loading we can compute pointer condition.

    # compute base pointers for left and right for this instance (b,h)
    left_base = left_ptr + b * stride_left_batch + h * stride_left_d
    right_base = right_ptr + b * stride_right_batch + h * stride_right_d
    out_base = out_ptr + b * stride_out_batch + h * stride_out_d

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k blocks
    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_blocks_k):
        # compute k start offset
        kb_start = kb * BLOCK_K
        k_off = kb_start + k_offset

        # optional mask for k
        mask_k = k_off < N

        # Load A (left) block: shape [BLOCK_M, BLOCK_K]
        a_ptrs = left_base + i_offset[:, None] * stride_left_i + k_off[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B (right) block transposed: shape [BLOCK_K, BLOCK_N]
        b_ptrs = right_base + k_off[:, None] * stride_right_k + j_offset[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) * (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Write out
    out_ptrs = out_base + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) kernel.

This implementation follows the AlphaFold3 TriMul module:
1. Layer‑norm the input tensor.
2. Linear projections (left/right) and gating weights (sigmoid).
3. Apply mask and gates to the projected tensors.
4. Compute the expensive contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a custom Triton kernel that performs a batched matrix‑multiply
   (A @ Bᵀ) for each (batch, hidden) pair.
5. Apply a second LayerNorm, multiply by the out‑gate and a final
   linear projection back to the original channel dimension.

Only the contraction (step 4) is executed on Triton; all other
operations use PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_contract_kernel(
    left_ptr,            # *float32
    right_ptr,           # *float32
    out_ptr,             # *float32
    B, N, H,             # int32 scalars
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # total number of (batch, hidden) instances
    total_instances = B * H
    # how many tiles per instance in the i‑j plane
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    tiles_per_inst = num_blocks_m * num_blocks_n

    # decode program id → instance, tile‑coords
    inst_id = pid // tiles_per_inst
    tile_id = pid % tiles_per_inst
    block_m = tile_id // num_blocks_n
    block_n = tile_id % num_blocks_n

    # batch & hidden indices for this instance
    b = inst_id // H
    h = inst_id % H

    # offsets for the current tile
    i_offset = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # base pointers for this (b,h) slice
    left_base = left_ptr + b * stride_left_batch + h * stride_left_d
    right_base = right_ptr + b * stride_right_batch + h * stride_right_d
    out_base = out_ptr + b * stride_out_batch + h * stride_out_d

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_blocks_k):
        k_start = kb * BLOCK_K
        k_off = k_start + k_offset
        mask_k = k_off < N

        # Load A = left[b,i,k,h]   shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + i_offset[:, None] * stride_left_i + k_off[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load Bᵀ = right[b,j,k,h]   shape (BLOCK_K, BLOCK_N)   (transposed on‑the‑fly)
        b_ptrs = right_base + k_off[:, None] * stride_right_k + j_offset[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (M,K)·(K,N) → (M,N)

    # Store the computed tile
    out_ptrs = out_base + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated TriMul forward (outgoing).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor:  torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict: mapping of weight names to torch.Tensors
            - config_dict: contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ----------------------------------------------------------------------
    # unpack arguments
    # ----------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ----------------------------------------------------------------------
    # Layer‑norm on the input
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]

    # ----------------------------------------------------------------------
    # Linear projections (no bias)
    # ----------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ----------------------------------------------------------------------
    # Optional masking
    # ----------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [B, N, N, 1]
        mask_exp = mask.to(dtype).unsqueeze(-1)
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # ----------------------------------------------------------------------
    # Gating (sigmoid)
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    )

    # ----------------------------------------------------------------------
    # Apply gates (mask already applied if present)
    # ----------------------------------------------------------------------
    left = left_proj * left_gate          # [B, N, N, H]
    right = right_proj * right_gate       # [B, N, N, H]

    # ----------------------------------------------------------------------
    # Triton contraction: out_contract[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ----------------------------------------------------------------------
    B, N, _, _ = x.shape
    # allocate output tensor
    out_contract = torch.empty_like(left)  # same shape [B, N, N, H]

    # Choose block sizes (tuned for typical N up to 1024, H≈128)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides for left/right/out (in elements)
    s_left = left.stride()    # (sb, si, sk, sh)
    s_right = right.stride()  # (sb, sj, sk, sh)
    s_out = out_contract.stride()  # (sb, si, sj, sh)

    # Compute launch grid
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_instances = B * hidden_dim
    total_programs = total_instances * num_blocks_m * num_blocks_n

    # Launch the kernel
    tri_mul_contract_kernel[total_programs](
        left,
        right,
        out_contract,
        B,
        N,
        hidden_dim,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------------
    # Post‑processing: LayerNorm → out‑gate → final projection
    # ----------------------------------------------------------------------
    # LayerNorm over hidden_dim
    out = torch.nn.functional.layer_norm(
        out_contract,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Multiply by the out‑gate
    out = out * out_gate

    # Final linear projection back to `dim`
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
