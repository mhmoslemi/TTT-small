def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, n, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # LayerNorm of input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    
    # Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight) # shape[bs,n,n,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if not nomask
    if mask is not None and not config.get("nomask", False):
        mask_unsq = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))
    
    left = left * left_gate
    right = right * right_gate

    # Compute einsum using Triton kernel
    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # shape [bs, n, n, hidden_dim]

    # call kernel
    _tri_einsum_outgoing_kernel(left, right, out)
    # define kernel separately
    
    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    
    out = out_norm * out_gate
    
    # final linear
    to_out_weight = weights["to_out.weight"]
    final_out = torch.nn.functional.linear(out, to_out_weight)
    
    return final_out
```

```
@triton.jit
def tri_einsum_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
   pid_i = tl.program_id(axis=0)
   pid_j = tl.program_id(axis=1)
   pid_bh = tl.program_id(axis=2)

   # number of hidden blocks
   num_h_blocks = tl.cdiv(H, BLOCK_H)
   b = pid_bh // num_h_blocks
   h_block = pid_bh % num_h_blocks

   # base indices
   i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
   j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
   k_base = tl.arange(0, BLOCK_K)
   h = h_block * BLOCK_H + tl.arange(0, BLOCK_H)

   # masks for bounds
   mask_i = i < N
   mask_j = j < N
   mask_h = h < H

   # accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

   for kk in range(0, N, BLOCK_K):
       k = kk + k_base
       mask_k = k < N

       # compute masks for load
       mask_left = mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]
       mask_right = mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]

       # compute pointers
       # left: [B, N, N, H] with dims (b,i,k,h)
       left_ptrs = left_ptr + b*stride_left_b + i[:, None, None] * stride_left_i + k[None, :, None] * stride_left_k + h[None, None, :] * stride_left_h
       left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0).to(tl.float32)

       # right: [B, N, N, H] with dims (b,j,k,h)
       right_ptrs = right_ptr + b*stride_right_b + j[:, None, None] * stride_right_j + k[None, :, None] * stride_right_k + h[None, None, :] * stride_right_h
       right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0).to(tl.float32)

       # loop over k dimension within the tile, performing outer product per element
       # vectorized: for each kk in BLOCK_K compute outer product across i,j
       # using broadcasting
       # left_tile shape (BLOCK_M, BLOCK_K, BLOCK_H)
       # right_tile shape (BLOCK_N, BLOCK_K, BLOCK_H)
       # we compute for each kk: acc += left[:, kk, :] (M,H) * right[:, kk, :] (N,H) outer product

       # loop manually
       for k_offset in range(0, BLOCK_K):
           # optional: break if k_offset+kk >= N? But mask already zero
           l = left_tile[:, k_offset, :]   # (M,H)
           r = right_tile[:, k_offset, :]  # (N,H)
           # broadcast outer product: (M,1,H) * (1,N,H) -> (M,N,H)
           acc += l[:, None, :] * r[None, :, :]
   # end for k tiles

   # store result
   out_ptrs = out_ptr + b*stride_out_b + i[:, None, None] * stride_out_i + j[None, :, None] * stride_out_j + h[None, None, :] * stride_out_h
   store_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
   tl.store(out_ptrs, acc, mask=store_mask)
```

"""
TriMul outgoing kernel (AlphaFold3).

This implementation follows the forward pass of the reference PyTorch
TriMul module:
    1. LayerNorm on the input.
    2. Linear projections (left/right) and gating.
    3. Core computation:
          out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
       which is a batch of matrix multiplications per hidden channel.
       The heavy sum‑over‑k is performed by a custom Triton kernel.
    4. LayerNorm on the hidden dimension, gating and final linear projection.

The Triton kernel processes a tile of (i, j, d) at a time:
    - BLOCK_M x BLOCK_N output positions (i, j)
    - BLOCK_H hidden channels
    - K‑dimension is reduced in blocks of size BLOCK_K.
All out‑of‑bounds loads are masked to zero, therefore the kernel works
for any sequence length N and hidden size H.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the batched Einsum:
# out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # problem size
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    pid_i = tl.program_id(axis=0)          # tile row   (i)
    pid_j = tl.program_id(axis=1)          # tile col   (j)
    pid_bh = tl.program_id(axis=2)         # combined batch & hidden block

    # ------------------------------------------------------------------
    # Decode batch index and hidden‑block index from pid_bh
    # ------------------------------------------------------------------
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    b = pid_bh // num_h_blocks               # batch id
    h_block = pid_bh % num_h_blocks           # hidden‑block id

    # ------------------------------------------------------------------
    # Offsets for this tile
    # ------------------------------------------------------------------
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_base = tl.arange(0, BLOCK_K)           # will be shifted each iteration
    h_off = h_block * BLOCK_H + tl.arange(0, BLOCK_H)

    # ------------------------------------------------------------------
    # Boundary masks
    # ------------------------------------------------------------------
    mask_i = i_off < N
    mask_j = j_off < N
    mask_h = h_off < H

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K in blocks
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        k = k_start + k_base
        mask_k = k < N

        # ----- left tile -----
        # shape (BLOCK_M, BLOCK_K, BLOCK_H)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i_off[:, None, None] * stride_left_i
            + k[None, :, None] * stride_left_k
            + h_off[None, None, :] * stride_left_h
        )
        mask_left = mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]
        left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0).to(tl.float32)

        # ----- right tile -----
        # shape (BLOCK_N, BLOCK_K, BLOCK_H)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + j_off[:, None, None] * stride_right_j
            + k[None, :, None] * stride_right_k
            + h_off[None, None, :] * stride_right_h
        )
        mask_right = mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]
        right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0).to(tl.float32)

        # ------------------------------------------------------------------
        # Accumulate: for each kk within the K‑tile compute outer product
        # ------------------------------------------------------------------
        for kk in range(0, BLOCK_K):
            # (BLOCK_M, BLOCK_H) and (BLOCK_N, BLOCK_H)
            left_vec  = left_tile[:, kk, :]          # M x H
            right_vec = right_tile[:, kk, :]         # N x H
            # outer product across i and j, keep hidden dim
            # left_vec[:, None, :] -> M x 1 x H
            # right_vec[None, :, :] -> 1 x N x H
            acc += left_vec[:, None, :] * right_vec[None, :, :]

    # ------------------------------------------------------------------
    # Store result with proper masking
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i_off[:, None, None] * stride_out_i
        + j_off[None, :, None] * stride_out_j
        + h_off[None, None, :] * stride_out_h
    )
    mask_out = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


# ----------------------------------------------------------------------
# Helper to launch the Triton kernel
# ----------------------------------------------------------------------
def _tri_mul_outgoing_triton(left: torch.Tensor, right: torch.Tensor, out: torch.Tensor):
    """
    Launch `tri_mul_outgoing_kernel` on the given tensors.
    All tensors must be contiguous and reside on the same CUDA device.
    """
    assert left.is_cuda and right.is_cuda and out.is_cuda
    B, N, _, H = left.shape

    # Tile sizes – tuned for typical N (256‑1024) and H=128
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # dim 0 -> i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # dim 1 -> j‑tiles
        B * ((H + BLOCK_H - 1) // BLOCK_H)    # dim 2 -> batch × hidden‑blocks
    )

    (s_b, s_i, s_k, s_h) = left.stride()
    (rs_b, rs_i, rs_k, rs_h) = right.stride()
    (os_b, os_i, os_j, os_h) = out.stride()

    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=s_b,
        stride_left_i=s_i,
        stride_left_k=s_k,
        stride_left_h=s_h,
        stride_right_b=rs_b,
        stride_right_j=rs_i,   # second axis of right (i/j)
        stride_right_k=rs_k,
        stride_right_h=rs_h,
        stride_out_b=os_b,
        stride_out_i=os_i,
        stride_out_j=os_j,
        stride_out_h=os_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a custom Triton kernel
    for the heavy sum‑over‑k operation.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [bs, seq_len, seq_len, dim]
        - mask         : torch.Tensor of shape [bs, seq_len, seq_len] (optional)
        - weights      : dict of model parameters
        - config       : dict containing at least "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Projections
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # broadcast mask to last dim and convert to same dtype as tensors
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core einsum computed by Triton
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous for correct stride extraction
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)   # shape [B, N, N, hidden_dim]

    _tri_mul_outgoing_triton(left, right, out)

    # ------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm, out‑gate, final linear)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    final_out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return final_out
