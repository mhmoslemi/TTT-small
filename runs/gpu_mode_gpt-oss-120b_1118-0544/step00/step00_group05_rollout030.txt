    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Get program IDs
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute pointers to start of batch
    a_batch_ptr = a_ptr + pid_batch * batch_stride_a
    b_batch_ptr = b_ptr + pid_batch * batch_stride_b
    c_batch_ptr = c_ptr + pid_batch * batch_stride_c

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A atomic block
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0).to(tl.float32)

        # Load B block transposed: we need B[k, n] where original B has rows=j, cols=k.
        # So we load using row index = offs_k (k) and col index = offs_n (j)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

        # Increment pointers (actually we recompute each iteration)
        # Not needed because we recompute pointers inside loop
    # Store result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    if mask_tensor is not None:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Contraction using triton kernel
    B, N, _, _ = left.shape
    # reshape for kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # B, H, N, N
    right_perm = right.permute(0, 3, 1, 2).contiguous() # B, H, N, N

    left_flat = left_perm.view(-1, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(-1, N, N)

    out_flat = torch.empty_like(left_flat)  # shape (B*H, N, N)

    # Retrieve strides
    batch_dim = left_flat.shape[0]
    stride_a_batch = left_flat.stride(0)
    stride_am = left_flat.stride(1)  # M stride
    stride_ak = left_flat.stride(2)  # K stride
    stride_b_batch = right_flat.stride(0)
    stride_bk = right_flat.stride(2) # for K row - we will use as row stride for transposed
    stride_bn = right_flat.stride(1) # for N col stride for transposed
    stride_c_batch = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # constants for block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # launch kernel
    grid = (batch_dim, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    batch_matmul_transpose_kernel[grid](
        left_flat, right_flat, out_flat,
        stride_a_batch, stride_am, stride_ak,
        stride_b_batch, stride_bk, stride_bn,
        stride_c_batch, stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Reshape output back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape B,N,N,hidden_dim

    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
custom_kernel: Triton‑accelerated outgoing TriMul.

Algorithm
---------
1. Layer‑normalize the [B, N, N, D] input (D = dim).
2. Linear projections → left/right tensors of shape [B, N, N, H] (H = hidden_dim).
3. (Optional) apply a binary mask broadcast on the last dimension.
4. Gating: sigmoid(linear(x_norm)) per projection and an output gate.
5. Contract over the intermediate sequence dimension k:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   Implemented as a batched GEMM (left @ rightᵀ) using a Triton kernel.
6. Layer‑norm over the hidden dimension, apply the output gate and a final
   linear projection back to dim.
The kernel handles a batch of independent GEMMs (B·H of size N×N) and
stores the result in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batch‑wise GEMM where the second operand is transposed on‑the‑fly.
    Computes:
        C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]   (B is accessed as Bᵀ)
    All tensors are 3‑D: (batch, rows, cols)."""
    pid_batch = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block column

    # Offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the first element of the current batch
    a_batch = a_ptr + pid_batch * batch_stride_a
    b_batch = b_ptr + pid_batch * batch_stride_b
    c_batch = c_ptr + pid_batch * batch_stride_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # Load B tile transposed: we need B[k, j] which in memory is B[j, k]
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Accumulate (M x K)·(K x N) → (M x N)
        acc += tl.dot(a, b)

    # Write back the result tile
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Triton‑implemented forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              - input_tensor: torch.Tensor of shape [B, N, N, D]
              - mask_tensor: torch.Tensor of shape [B, N, N] (binary)
              - weights_dict: mapping of weight names to torch.Tensors
              - config_dict: contains "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32, on the same device as inputs)
    """
    # ----- unpack ------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    # ----- LayerNorm on input ------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, D]

    # ----- Linear projections ------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])        # [B, N, N, H]

    # ----- Optional mask ------------------------------------------------------
    if mask is not None and not cfg.get("nomask", False):
        mask_f = mask.to(left.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----- Gating -------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----- Batched GEMM (contract over k) ------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim

    # Rearrange to (B, H, N, N) and flatten batch*hidden for the kernel
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_t.view(-1, N, N)    # [B*H, N, N]
    right_flat = right_t.view(-1, N, N)  # [B*H, N, N]
    out_flat = torch.empty_like(left_flat)  # result placeholder

    # Strides (int) for the kernel
    batch_stride_a = left_flat.stride(0)
    stride_am = left_flat.stride(1)   # row stride (M)
    stride_ak = left_flat.stride(2)   # col stride (K)

    batch_stride_b = right_flat.stride(0)
    stride_bk = right_flat.stride(2)  # stride for K when accessed as rows (original col stride)
    stride_bn = right_flat.stride(1)  # stride for N when accessed as cols (original row stride)

    batch_stride_c = out_flat.stride(0)
    stride_cm = out_flat.stride(1)    # row stride (M)
    stride_cn = out_flat.stride(2)    # col stride (N)

    # Block sizes – tuned for typical N (≤1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        left_flat.shape[0],                         # batch*hidden
        (N + BLOCK_M - 1) // BLOCK_M,               # rows
        (N + BLOCK_N - 1) // BLOCK_N,               # cols
    )

    _batched_matmul_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_stride_a,
        stride_am,
        stride_ak,
        batch_stride_b,
        stride_bk,
        stride_bn,
        batch_stride_c,
        stride_cm,
        stride_cn,
        N,          # M
        N,          # N
        N,          # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    torch.cuda.synchronize()  # ensure kernel completion before further ops

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ----- Output layer‑norm, gate and final linear -------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, D]

    return out
