    M, N, K,

def layernorm_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    B, N, C,
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_ob, stride_oi, stride_oj, stride_oc,
    eps: tl.constexpr,

    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)  # flatten over (b,i,j) maybe 1D
    total = B * N * N
    pid0 = pid // (N * N)
    pid1 = (pid % (N * N)) // N
    pid2 = (pid % (N * N)) % N
    # Actually simpler: create offsets.

    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # compute offsets for C
    offs_c = tl.arange(0, C)
    x = tl.load(x_ptr + b * stride_xb + i * stride_xi + j * stride_xj + offs_c * stride_xc, mask=offs_c < C, other=0.0)  # shape (C)
    # compute mean, var
    mean = tl.sum(x, axis=0) / C
    var = tl.sum((x - mean) * (x - mean), axis=0) / C
    inv_std = 1.0 / tl.sqrt(var + eps)
    y = (x - mean) * inv_std
    if weight_ptr is not None:
        w = tl.load(weight_ptr + offs_c, mask=offs_c < C, other=1.0)
        y = y * w
    if bias_ptr is not None:
        b_tb = tl.load(bias_ptr + offs_c, mask=offs_c < C, other=0.0)
        y = y + b_tb
    tl.store(out_ptr + b * stride_ob + i * stride_oi + j * stride_oj + offs_c * stride_oc, y)

TriMul (outgoing) forward implemented with a custom Triton kernel for the
core O(N³) reduction. The kernel performs, for each batch and hidden channel,
a batched matrix multiplication:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

# -----------------------------------------------------------------------------
# Triton kernel: batch*hidden‑dim independent matrix multiplication.
# Each program (block) computes a tile of the output matrix (i,j) for a
# single (batch, hidden) pair.
# -----------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # *float32, shape [B,N,N,H]
    right_ptr,           # *float32, shape [B,N,N,H]
    out_ptr,             # *float32, shape [B,N,N,H]
    M, N, K,             # matrix sizes (M=N rows, N=cols, K=reduce dim)
    stride_am, stride_ak,   # left: row‑stride (i), col‑stride (k)
    stride_bk, stride_bn,   # right (transposed): row‑stride (k), col‑stride (j)
    stride_cm, stride_cn,   # out: row‑stride (i), col‑stride (j)
    BLOCK_M: tl.constexpr,  # tile rows
    BLOCK_N: tl.constexpr,  # tile cols
    BLOCK_K: tl.constexpr,  # reduction tile
):
    pid_i = tl.program_id(0)           # tile index over rows (i)
    pid_j = tl.program_id(1)           # tile index over cols (j)
    pid_z = tl.program_id(2)           # matrix index = batch * hidden + hidden_id

    # -------------------------------------------------------------------------
    # Offsets for the output tile (i,j)
    # -------------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # -------------------------------------------------------------------------
    # Base pointer for the current matrix (batch * hidden + hidden_id)
    # The matrix index advances by one element in the hidden dimension, thus
    # we can simply add pid_z to the original pointer.
    # -------------------------------------------------------------------------
    base_left  = left_ptr  + pid_z
    base_right = right_ptr + pid_z
    base_out   = out_ptr   + pid_z

    # -------------------------------------------------------------------------
    # Accumulator for the tile
    # -------------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------------------
    # Loop over the reduction dimension K in BLOCK_K sized chunks
    # -------------------------------------------------------------------------
    num_k = tl.cdiv(K, BLOCK_K)
    for k_iter in range(num_k):
        k_offset = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = base_left + (offs_i[:, None] * stride_am
                              + k_offset[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Right tile (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = base_right + (k_offset[:, None] * stride_bk
                               + offs_j[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # GEMM on the tile
        acc += tl.dot(a, b)

    # -------------------------------------------------------------------------
    # Write the result back, respecting the output bounds
    # -------------------------------------------------------------------------
    c_ptrs = base_out + (offs_i[:, None] * stride_cm
                         + offs_j[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# -----------------------------------------------------------------------------
# Main entry point used by the evaluator
# -----------------------------------------------------------------------------
def custom_kernel(data):
    """
    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (float or bool)
        - weights      : dict of torch.Tensors
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # -----------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B,N,N,dim]

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # Gating linear layers + sigmoid
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # -----------------------------------------------------------------
    # Apply mask (if present) and gating
    # -----------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)  # [B,N,N,1]
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Make sure the tensors are contiguous for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # -----------------------------------------------------------------
    # Allocate output tensor for the core reduction
    # -----------------------------------------------------------------
    out = torch.empty_like(left, dtype=dtype, device=device)  # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # Compute strides for the Triton kernel (elements, not bytes)
    # left/right shape: [B, N, N, H]
    # stride order is (B, i, k, d)
    stride_b, stride_i, stride_k, stride_d = left.stride()
    # For our GEMM we need:
    #   A (left)  : row‑stride = stride_i , col‑stride = stride_k
    #   B (right) : row‑stride = stride_k , col‑stride = stride_i  (transposed)
    #   C (out)   : same row/col stride as A
    stride_am = stride_i
    stride_ak = stride_k
    stride_bk = stride_k
    stride_bn = stride_i
    stride_cm = stride_i
    stride_cn = stride_i

    # -----------------------------------------------------------------
    # Kernel launch configuration
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_mats = B * hidden_dim  # number of independent (batch, hidden) matrices
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        total_mats,
    )

    # Launch the Triton kernel
    tri_mul_kernel[grid](
        left,
        right,
        out,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    torch.cuda.synchronize()

    # -----------------------------------------------------------------
    # Post‑processing: LayerNorm over hidden_dim, out‑gate and final linear
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # broadcast over last dim

    # Final linear projection back to original dimension
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
