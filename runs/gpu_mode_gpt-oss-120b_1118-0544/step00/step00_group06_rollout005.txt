"""
TriMul outgoing kernel (AlphaFold3 style)

This implementation follows the PyTorch reference but accelerates the O(N³) core
operation using a fused Triton kernel. The kernel computes, for each batch and
hidden channel d,

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which is equivalent to a batched matrix multiplication
C = A @ Bᵀ with A = left[..., :, :, d] and B = right[..., :, :, d].

The kernel is tiled over the i (BLOCK_M) and j (BLOCK_N) dimensions and
accumulates over k (BLOCK_K).  Batch‑hidden indices are processed in the third
grid dimension, so the same kernel launches all B·hidden_dim independent
matrix multiplications in parallel.

All other operations (LayerNorms, Linear projections, sigmoids, masking) are
performed with PyTorch for simplicity.  The kernel works with float32 inputs
(and can be adapted to fp16/bf16 by casting before the kernel call).
"""

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr,            # *f32  left tensor   [B, N, N, H]
    right_ptr,           # *f32  right tensor  [B, N, N, H]
    out_ptr,             # *f32  output tensor [B, N, N, H]
    B, N, H,             # i32   dimensions
    # strides for left: (batch, i, k, hidden)
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    # strides for right: (batch, j, k, hidden)
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    # strides for out: (batch, i, j, hidden)
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # tile over i
    pid_n = tl.program_id(1)          # tile over j
    pid_bh = tl.program_id(2)         # combined batch * hidden index

    # Split pid_bh into batch and hidden index
    batch_idx = pid_bh // H
    h_idx = pid_bh % H

    # Offsets in the i/j dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + batch_idx * stride_left_batch  + h_idx * stride_left_h
    right_base = right_ptr + batch_idx * stride_right_batch + h_idx * stride_right_h
    out_base   = out_ptr   + batch_idx * stride_out_batch   + h_idx * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for kb in range(num_blocks_k):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers for the current k‑tile
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j

        # Load with bounds masking
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float32)

        # Accumulate product over k
        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask_tensor: torch.Tensor [B, N, N] (may be None)
            - weights_dict: mapping name -> torch.Tensor containing all model weights
            - config_dict: contains "dim", "hidden_dim" (and optionally others)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32 on the same device)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----- LayerNorm on input -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ----- Linear projections (no bias) ---------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----- Optional masking ----------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_ = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ----- Gating -------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ----- Prepare for Triton kernel -----------------------------------------
    B, N, N2, _ = left.shape
    assert N == N2, "Input must be square in the two sequence dimensions"
    # Ensure computation in float32 (Triton kernel operates in float32)
    left = left.to(torch.float32)
    right = right.to(torch.float32)

    out = torch.empty((B, N, N, hidden_dim), device=left.device, dtype=torch.float32)

    # Strides (in elements)
    ls = left.stride()
    rs = right.stride()
    os = out.stride()
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h = ls
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h = rs
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h = os

    # Tile sizes – chosen to fit well on B200 (A100) shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_batch,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_batch,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_batch,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Post‑processing ----------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    out = out * out_gate  # gate applied element‑wise

    # Final linear projection back to 'dim'
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
