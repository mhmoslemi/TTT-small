@triton.jit
def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k, # right needs to be transposed: we store right in original layout (B,N,N) but we will treat as transposed on the fly using different strides.
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program id = batch index * (grid_m * grid_n) + block_m * grid_n + block_n
    # Use linearized pid to compute batch, block_m, block_n.
    # Pid_total = tl.program_id(0)
    pid = tl.program_id(0)
    batch_dim = BATCH
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    # find block indices
    batch = pid // (num_m * num_n)
    block_idx = pid % (num_m * num_n)
    block_m = block_idx // num_n
    block_n = block_idx % num_n

    # offsets for each block
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # guard against out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offsets for K dimension
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # load left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + batch * stride_left_batch + (offs_m[:, None] * stride_left_m) + (k_offset[None, :] * stride_left_k)
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right block transposed: we need right[batch, n, k] (original right is [batch, n, k]), we want right_T[k, n] for multiplication.
        # right layout: batch stride + n*stride_right_n + k*stride_right_k (since original shape (batch, N, N)).
        # For multiplication we need right_T: element at (k, n) = right[batch, n, k]
        right_ptrs = right_ptr + batch * stride_right_batch + (k_offset[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_n)   # Wait need shape (BLOCK_K, BLOCK_N)
        # Actually we want right_T block (K,N) -> use k_offset as rows, offs_n as cols.
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(left, right)

    # write back
    out_ptrs = out_ptr + batch * stride_out_batch + (offs_m[:, None] * stride_out_m) + (offs_n[None, :] * stride_out_n)
    tl.store(out_ptrs, acc.to(tl.float16) if out_ptr.dtype == tl.float16 else acc, mask=mask_m[:, None] & mask_n[None, :])

import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_left_m, stride_left_k,
    stride_right_k, stride_right_n,
    stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load left sub-block (M x K)
        left_ptrs = left_ptr + batch * (M * K) + offs_m[:, None] * stride_left_m + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right sub-block (K x N)
        # right currently is transposed (K,N)
        right_ptrs = right_ptr + batch * (K * N) + k_offset[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # Write back
    out_ptrs = out_ptr + batch * (M * N) + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Optimized outgoing Triangle Multiplicative Update (TriMul).

The kernel computes the heavy O(N³·H) contraction
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
for every batch `b` and hidden channel `h`.  The contraction is expressed as a
batched matrix multiplication of shape (B·H, N, N) × (B·H, N, N)ᵀ and is
implemented in Triton with tiling (BLOCK_M × BLOCK_N) and a K‑tile loop.
All other operations (LayerNorm, linear projections, gating, final
normalisation and projection) are performed in PyTorch.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ Bᵀ) where A,B have shape [Btotal, N, N]
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr,              # left  matrix (float16)  [Btotal, M, K]
    b_ptr,              # right matrix transposed (float16) [Btotal, K, N]
    c_ptr,              # output matrix (float16) [Btotal, M, N]
    M, N, K,            # matrix dimensions (int)
    stride_a_m, stride_a_k,    # strides for A
    stride_b_k, stride_b_n,    # strides for B (already transposed)
    stride_c_m, stride_c_n,    # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Accumulator (float32 for precision)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile   (M x K)
        a_ptrs = a_ptr + batch_id * (M * K) \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile   (K x N)   (B is already transposed)
        b_ptrs = b_ptr + batch_id * (K * N) \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back the result (cast back to fp16)
    # --------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * (M * N) \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

    hidden_dim = cfg["hidden_dim"]

    # LayerNorm on the channel dimension
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(

        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # Apply mask if present (mask shape [B, N, N])
    # --------------------------------------------------------------
    if mask is not None:

        right_proj = right_proj * mask_f

    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # Apply gates
    # --------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Cast to fp16 for the heavy O(N³) contraction
    # --------------------------------------------------------------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    B, N, _, H = left_f16.shape  # H == hidden_dim

    # ------------------------------------------------------------------
    # Reshape to a 2‑D batch where each hidden channel becomes a separate batch
    # ------------------------------------------------------------------

    left_flat = left_f16.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_flat = right_f16.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()

    # Right matrix must be transposed for the matmul (B*H, K, N)
    right_t = right_flat.transpose(1, 2).contiguous()

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    BLOCK_M = 64 if N >= 128 else 32
    BLOCK_N = BLOCK_M
    BLOCK_K = 64 if N >= 128 else 32

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # --------------------------------------------------------------
    # Strides (in element units, not bytes)
    # --------------------------------------------------------------
    stride_a_m = left_flat.stride(1)   # row stride (N)
    stride_a_k = left_flat.stride(2)   # column stride (1)

    stride_b_k = right_t.stride(0)     # row stride (N)  <-- K dimension
    stride_b_n = right_t.stride(1)     # column stride (1)

    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # --------------------------------------------------------------
    # Launch Triton kernel
    # --------------------------------------------------------------
    batched_matmul_kernel[grid](
        left_flat,
        right_t,
        out_flat,
        N, N, N,

        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # Reshape back to [B, N, N, H] and convert to float32 for the

    # --------------------------------------------------------------

    )

    # --------------------------------------------------------------
