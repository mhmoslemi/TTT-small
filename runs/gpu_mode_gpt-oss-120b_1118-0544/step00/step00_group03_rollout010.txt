"""
custom_kernel: fast Triton‑fused TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,C]).
2. Linear projections → left_proj, right_proj (C→H) and gating logits
   → left_gate, right_gate, out_gate (C→H). Sigmoid the gates.
3. Fuse element‑wise mask, left_gate and right_gate into left_proj / right_proj
   with a small Triton kernel (avoids extra memory passes).
4. Compute the outgoing triangle multiplicative update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   i.e. a batch of B·H independent matrix multiplies (N×N) @ (N×N)^T.
   We perform this with torch.bmm on the reshaped tensors (highly‑optimized
   cuBLAS implementation).
5. Apply a second LayerNorm (over the hidden dimension H), multiply by the
   output gate, and a final linear projection back to dimension C.
6. Return the result in float32.

Only the mask‑gate fusion is executed in Triton, satisfying the requirement
to use a custom kernel while keeping the heavy matrix‑multiply in cuBLAS for
speed.
"""

import math
from typing import Tuple, Dict

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _fuse_gates_kernel(
    left_proj_ptr,   # [total_elts]  float32
    left_gate_ptr,   # [total_elts]  float32
    right_proj_ptr,  # [total_elts]  float32
    right_gate_ptr,  # [total_elts]  float32
    mask_ptr,        # [total_mask]  float32
    # output pointers (we overwrite the proj tensors)
    left_out_ptr,    # [total_elts]  float32
    right_out_ptr,   # [total_elts]  float32
    total_elts: tl.constexpr,
    hidden_dim: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    m = offs < total_elts

    # index of the mask element that corresponds to the (b,i,j,*) entry
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=m, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=m, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=m, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=m, other=0.0)

    left_out = lp * lg * mask_val
    right_out = rp * rg * mask_val

    tl.store(left_out_ptr + offs, left_out, mask=m)
    tl.store(right_out_ptr + offs, right_out, mask=m)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, C] (float32)
            - mask_tensor : [B, N, N]   (float32 or bool, 0/1)
            - weights_dict: contains all linear / layer‑norm parameters
            - config_dict : must contain "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # --------------------------------------------------------------------- #
    # LayerNorm on the input (affine)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B,N,N,C]

    # --------------------------------------------------------------------- #
    # Linear projections and gating logits (no bias in the model)
    # --------------------------------------------------------------------- #
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    left_proj = F.linear(x, left_proj_weight)          # [B,N,N, H]
    right_proj = F.linear(x, right_proj_weight)        # [B,N,N, H]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B,N,N, H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))     # [B,N,N, H]

    # --------------------------------------------------------------------- #
    # Prepare mask (float, broadcast over hidden_dim)
    # --------------------------------------------------------------------- #
    if mask_tensor is None:
        mask = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=dtype,
            device=device,
        )
    else:
        mask = mask_tensor.to(dtype)

    # Ensure contiguous tensors for the kernel
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_flat = mask.contiguous().view(-1)  # shape B*N*N

    # --------------------------------------------------------------------- #
    # Triton kernel: fuse mask and gates into the projected tensors
    # --------------------------------------------------------------------- #
    total_elts = left_proj.numel()
    BLOCK_SIZE = 1024  # 1‑K threads per block (well within limits)

    grid = (triton.cdiv(total_elts, BLOCK_SIZE),)
    _fuse_gates_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask_flat,
        # write‑back in‑place
        left_proj,
        right_proj,
        total_elts,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # Gates are no longer needed
    del left_gate, right_gate

    # --------------------------------------------------------------------- #
    # Batched matrix multiplication: out_{h} = left_{h} @ right_{h}ᵀ
    # --------------------------------------------------------------------- #
    B, N, _, _ = left_proj.shape  # left_proj shape = [B,N,N,H]

    # Move hidden dimension to second axis for easy reshaping
    left_perm = left_proj.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_proj.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Collapse batch & head dimensions
    left_mat = left_perm.view(B * hidden_dim, N, N)    # [B*H, N, N]
    right_mat = right_perm.view(B * hidden_dim, N, N)  # [B*H, N, N]

    # Perform the N³ reduction via batched GEMM (cuBLAS)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N]

    # Restore original tensor layout
    out_perm = out_mat.view(B, hidden_dim, N, N)                # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]

    # --------------------------------------------------------------------- #
    # Second LayerNorm, output gating and final linear projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # -> [B, N, N, C]

    # Ensure final dtype is float32 as required
    return out.to(torch.float32)
