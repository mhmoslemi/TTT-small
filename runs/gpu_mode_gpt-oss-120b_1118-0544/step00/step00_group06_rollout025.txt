def triton_matmul_transpose(A, B):
    # A: (M,N,K)
    # B: (M,N,K)
    # returns C: (M,N,N) where C = A @ B^T
    M, N, K = A.shape
    BLOCK_M = 64  # tile for i dimension
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), M)
    out = torch.empty((M, N, N), dtype=A.dtype, device=A.device)
    matmul_transpose_kernel[grid](
        A_ptr=A, B_ptr=B, C_ptr=out,
        M=M, N=N, K=K,
        stride_am=A.stride(0), stride_ai=A.stride(1), stride_ak=A.stride(2),
        stride_bm=B.stride(0), stride_bj=B.stride(1), stride_bk=B.stride(2),
        stride_cm=out.stride(0), stride_ci=out.stride(1), stride_cj=out.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    return out

    # we can use torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

```python
"""TriMul forward (outgoing) implementation.
Computes per‑batch, per‑hidden‑dim matrix multiplication A @ Bᵀ
(where A and B are the left/right projected pairwise representations).
The O(N³) contraction is expressed as a batched GEMM and executed
with a Triton kernel for high performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel: computes C[m,i,j] = sum_k A[m,i,k] * B[m,j,k]
# A, B have shape (M, N, K) where K == N.
# C has shape (M, N, N).
@triton.jit
def tri_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(2)  # merged batch‑hidden index
    pid_i = tl.program_id(0)  # tile row
    pid_j = tl.program_id(1)  # tile column

    # Compute block offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for boundary handling
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < K

    # Pointers to the start of the current tile
    a_ptrs = A_ptr + pid_m * stride_am + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_m * stride_bm + offs_j[:, None] * stride_bj + offs_k[None, :] * stride_bk

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k):
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

"""
TriMul (outgoing) forward implementation.

The most expensive part of the TriMul module is the contraction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which can be rewritten as a batched matrix multiplication
C = A @ Bᵀ for each (batch, hidden) slice, where
A and B have shape (N, N) and the hidden dimension is usually 128.
The kernel below computes this batched GEMM with Triton,
while the surrounding LayerNorm, linear projections and gating are
performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_matmul_kernel(
    A_ptr,               # pointer to left tensor  (M, N, K)
    B_ptr,               # pointer to right tensor (M, N, K)
    C_ptr,               # pointer to output     (M, N, N)
    M, N, K,             # dimensions
    stride_am, stride_ai, stride_ak,   # A strides
    stride_bm, stride_bj, stride_bk,   # B strides
    stride_cm, stride_ci, stride_cj,   # C strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(2)  # merged batch‑hidden index
    pid_i = tl.program_id(0)  # tile row
    pid_j = tl.program_id(1)  # tile column

    # Block coordinates
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Boundary masks
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < K

    # Pointers to the current tile
    a_ptrs = A_ptr + pid_m * stride_am + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_m * stride_bm + offs_j[:, None] * stride_bj + offs_k[None, :] * stride_bk

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k):
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        acc += tl.dot(a, b)               # sum over K dimension
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the result
    c_ptrs = C_ptr + pid_m * stride_cm + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
