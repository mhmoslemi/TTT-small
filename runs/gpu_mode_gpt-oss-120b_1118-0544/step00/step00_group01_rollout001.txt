    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,

    BLOCK_D: tl.constexpr,
    eps: tl.constexpr,
):

    pid_tmp = pid

    pid_tmp //= D
    b = pid_tmp % B
    pid_tmp //= B

    block_n = pid_tmp // tl.cdiv(N, BLOCK_M)

    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

def custom_kernel(data):
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Retrieve weights as torch.tensors
    norm_w = weights["norm.weight"]; norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]; to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]
    # LayerNorm
    eps = 1e-5
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b, eps=eps)

    # Projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # shape B,N,N,hidden
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # Apply mask if provided
    if not config.get("nomask", True):
        # mask shape [B,N,N]; unsqueeze last dim
        mask_ = mask.unsqueeze(-1).type_as(left_proj)
        left_proj = left_proj * mask_
        right_proj = right_proj * mask_
    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Compute out = left @ right^T per hidden dim using Triton
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Get strides
    # left stride: (stride_b, stride_i, stride_k, stride_d)
    stride_lb = left.stride(0)
    stride_li = left.stride(1)
    stride_lk = left.stride(2)
    stride_ld = left.stride(3)

    stride_rb = right.stride(0)
    stride_rj = right.stride(1)   # j dimension
    stride_rk = right.stride(2)
    stride_rd = right.stride(3)

    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_od = out.stride(3)

    # Define block sizes.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # compute grid dims
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * hidden_dim * grid_m * grid_n

    # Launch kernel
    tri_mul_gemm_kernel[(total_programs,)](
        left, right, out,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Apply to_out_norm
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps)

    out_gated = out_norm * out_gate

    # Final linear projection to dim
    out_final = torch.nn.functional.linear(out_gated, to_out_w)  # shape B,N,N,dim

    return out_final

    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,

    grid_m = tl.cdiv(N, BLOCK_M)
    grid_n = tl.cdiv(N, BLOCK_N)
    # decode pid
    tmp = pid
    d = tmp % D
    tmp = tmp // D
    b = tmp % B
    tmp = tmp // B
    block_m = tmp % grid_m
    block_n = tmp // grid_m

    # offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right[j,k] -> need transpose: we load right as (k,j)

        acc += tl.dot(left, right)  # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # store

"""
TriMul "outgoing" forward pass.

Implements the heavy quadratic N‑N multiplication (left @ rightᵀ)
via a custom Triton kernel.  The remaining linear projections,
LayerNorms and sigmoids are performed with PyTorch for simplicity.
The kernel computes, for each batch b and hidden dimension d,
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication L₍d₎ @ R₍d₎ᵀ over the sequence axis.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------
# Triton kernel: batched GEMM across sequence dimension for each hidden dim.
# --------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,                      # pointers
    B, N, D,                                           # dimensions
    stride_lb, stride_li, stride_lk, stride_ld,        # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,        # right strides
    stride_ob, stride_oi, stride_oj, stride_od,        # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # compute grid sizes
    grid_m = tl.cdiv(N, BLOCK_M)
    grid_n = tl.cdiv(N, BLOCK_N)

    # decode program id into (b, d, block_m, block_n)
    tmp = pid
    d = tmp % D
    tmp = tmp // D
    b = tmp % B
    tmp = tmp // B
    block_m = tmp % grid_m
    block_n = tmp // grid_m                     # will be < grid_n

    # offsets within the current tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension (the sequence axis summed over)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)   left[i,k]
        left_ptrs = (left_ptr
                     + b * stride_lb
                     + d * stride_ld
                     + offs_m[:, None] * stride_li
                     + offs_k[None, :] * stride_lk)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile (transposed while loading): shape (BLOCK_K, BLOCK_N)   rightᵀ[k,j]
        right_ptrs = (right_ptr
                      + b * stride_rb
                      + d * stride_rd
                      + offs_k[:, None] * stride_rk
                      + offs_n[None, :] * stride_rj)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        # matrix multiply‑accumulate
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # store result tile to out[b,i,j,d]
    out_ptrs = (out_ptr
                + b * stride_ob
                + d * stride_od
                + offs_m[:, None] * stride_oi
                + offs_n[None, :] * stride_oj)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------
# Entry point used by the evaluator.
# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] on the same device.
    """
    # ----------------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------------------------------------------------
    # Load weights
    # ----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ----------------------------------------------------------------
    # LayerNorm on the input (dim‑wise)
    # ----------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b, eps=eps)

    # ----------------------------------------------------------------
    # Linear projections (no bias)
    # ----------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_w)

    # ----------------------------------------------------------------
    # Optional masking
    # ----------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape: [B,N,N]; broadcast over hidden dim
        mask_f = mask.unsqueeze(-1).type_as(left_proj)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ----------------------------------------------------------------
    # Gating (sigmoid)
    # ----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ----------------------------------------------------------------
    # Compute the quadratic term via Triton GEMM
    # ----------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides for the kernel
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # Tile sizes (tuned for A100 / B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * hidden_dim * grid_m * grid_n

    # Launch Triton kernel
    tri_mul_gemm_kernel[(total_programs,)](
        left, right, out,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ----------------------------------------------------------------
    # Post‑processing: norm over hidden dim, gating, final linear proj.
    # ----------------------------------------------------------------
    out_norm = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w,
                            bias=to_out_norm_b, eps=eps)

    out_gated = out_norm * out_gate

    # Final linear projection back to original dim
    final_out = F.linear(out_gated, to_out_w)   # [B,N,N,dim]

    return final_out
