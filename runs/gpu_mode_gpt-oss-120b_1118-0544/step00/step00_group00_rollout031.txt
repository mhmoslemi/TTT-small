    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Get weights as torch.nn.Parameter torch tensors (float32)
    # LayerNorm
    eps = 1e-5 # typical
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # Linear projections as linear: torch.nn.functional.linear(x_norm, weight, bias=None)
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

        mask_tensor = mask
    # Expand mask to hidden dimension
    mask_unsq = mask_tensor.unsqueeze(-1)
    left = left * mask_unsq
    right = right * mask_unsq

    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate
    # Cast to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    # Allocate output in half
    out = torch.empty_like(left, dtype=torch.float16)  # shape [B,N,N,hidden_dim]

    # We can compute in a loop over d in steps of D_BLOCK=1 if simplicity.

import torch
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) operation.

    The heavy part is the einsum '... i k d, ... j k d -> ... i j d',
    which we implement as a batch of matrix multiplications per hidden channel:
        out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask handling
    if config.get("nomask", False):
        mask = torch.ones_like(mask_tensor, dtype=x.dtype, device=x.device)
    else:
        mask = mask_tensor.to(x.dtype)
    mask = mask.unsqueeze(-1)  # [B,N,N,1]

    left = left * mask
    right = right * mask

    # Gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for the heavy matmul
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float16)  # [B,N,N,H]

    # Triton kernel definition
    @triton.jit
    def matmul_tile_kernel(
        out_ptr, a_ptr, b_ptr,
        B, N,
        stride_out_batch, stride_out_i, stride_out_j,
        stride_a_batch, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_j,
        BLOCK_M: tl.constexpr,
        BLOCK_N: tl.constexpr,
        BLOCK_K: tl.constexpr,
    ):
        pid_b = tl.program_id(0)
        pid_m = tl.program_id(1)
        pid_n = tl.program_id(2)

        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        mask_m = offs_m < N
        mask_n = offs_n < N

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k in range(0, N, BLOCK_K):
            cur_k = k + tl.arange(0, BLOCK_K)
            mask_k = cur_k < N

            # A tile: shape (BLOCK_M, BLOCK_K)
            a_ptrs = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_a_i + cur_k[None, :] * stride_a_k
            a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

            # B tile transposed: shape (BLOCK_K, BLOCK_N)
            b_ptrs = b_ptr + pid_b * stride_b_batch + cur_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_j
            b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

            acc += tl.dot(a, b)

        out_ptrs = out_ptr + pid_b * stride_out_batch + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
        tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    # Choose tile sizes (tuned for typical hardware)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    # Loop over hidden dimension
    for d in range(H):
        a_slice = left[..., d]
        b_slice = right[..., d]
        c_slice = out[..., d]

        stride_out_batch, stride_out_i, stride_out_j = c_slice.stride()
        stride_a_batch, stride_a_i, stride_a_k = a_slice.stride()
        stride_b_batch, stride_b_i, stride_b_k = b_slice.stride()

        # In the kernel we need stride for B's column dimension (j) which is stride_b_i.
        matmul_tile_kernel[grid](
            out_ptr=c_slice,
            a_ptr=a_slice,
            b_ptr=b_slice,
            B=B,
            N=N,
            stride_out_batch=stride_out_batch,
            stride_out_i=stride_out_i,
            stride_out_j=stride_out_j,
            stride_a_batch=stride_a_batch,
            stride_a_i=stride_a_i,
            stride_a_k=stride_a_k,
            stride_b_batch=stride_b_batch,
            stride_b_k=stride_b_k,   # row stride for k
            stride_b_j=stride_b_i,   # column stride for j
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

    # Convert back to fp32 for following layers
    out = out.to(torch.float32)

    # Final LayerNorm, gating and linear projection
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # broadcast over last dim
    output = torch.nn.functional.linear(out, weights["to_out.weight"])
    return output
```

TriMul (outgoing) forward pass with Triton acceleration.

The heavy computation is the einsum:
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

We treat each hidden channel `d` as an independent matrix multiplication:
    out_slice = left_slice @ right_slice.T
where left_slice and right_slice have shape (B, N, N).  The multiplication

import triton

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, cfg_dict)
            - input_tensor: [B, N, N, dim] float32
            - mask_tensor : [B, N, N]   (float or bool)
            - weights_dict: mapping of weight names to tensors (float32)
            - cfg_dict    : contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """

    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    # 1. Layer‑norm on the input
    # ------------------------------------------------------------------

        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask (if any)
    # ------------------------------------------------------------------
    if cfg.get("nomask", False):
        mask = torch.ones_like(mask_tensor, dtype=x.dtype, device=x.device)
    else:
        mask = mask_tensor.to(x.dtype)
    mask = mask.unsqueeze(-1)                     # [B, N, N, 1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"]
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"]
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy pair‑wise reduction via Triton
    #    Compute out_slice = left_slice @ right_slice.T for each hidden dim.
    # ------------------------------------------------------------------
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    B, N, _, H = left.shape               # H == hidden_dim
    out = torch.empty_like(left, dtype=torch.float16)   # placeholder for result

    @triton.jit
    def _matmul_tile_kernel(
        out_ptr, a_ptr, b_ptr,
        B, N,
        stride_out_batch, stride_out_i, stride_out_j,
        stride_a_batch, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_j,
        BLOCK_M: tl.constexpr,
        BLOCK_N: tl.constexpr,
        BLOCK_K: tl.constexpr,
    ):
        pid_b = tl.program_id(0)                     # batch axis
        pid_m = tl.program_id(1)                     # i (row) axis
        pid_n = tl.program_id(2)                     # j (col) axis

        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        mask_m = offs_m < N
        mask_n = offs_n < N

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k in range(0, N, BLOCK_K):
            cur_k = k + tl.arange(0, BLOCK_K)
            mask_k = cur_k < N

            # A tile: (BLOCK_M, BLOCK_K)
            a_ptrs = a_ptr + pid_b * stride_a_batch \
                    + offs_m[:, None] * stride_a_i \
                    + cur_k[None, :] * stride_a_k
            a = tl.load(a_ptrs,
                        mask=mask_m[:, None] & mask_k[None, :],
                        other=0.0)

            # B tile (transposed): (BLOCK_K, BLOCK_N)
            b_ptrs = b_ptr + pid_b * stride_b_batch \
                    + cur_k[:, None] * stride_b_k \
                    + offs_n[None, :] * stride_b_j
            b = tl.load(b_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

            acc += tl.dot(a, b)

        out_ptrs = out_ptr + pid_b * stride_out_batch \
                   + offs_m[:, None] * stride_out_i \
                   + offs_n[None, :] * stride_out_j
        tl.store(out_ptrs,
                 acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_n[None, :])

    # Tile sizes (tuned for typical RTX/A100 style GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # Loop over hidden dimension – each slice is a separate GEMM
    for d in range(H):
        a_slice = left[..., d]          # [B, N, N] view
        b_slice = right[..., d]         # [B, N, N] view
        c_slice = out[..., d]           # [B, N, N] view (output)

        # Strides for each tensor
        stride_out_batch, stride_out_i, stride_out_j = c_slice.stride()
        stride_a_batch, stride_a_i, stride_a_k = a_slice.stride()
        stride_b_batch, stride_b_i, stride_b_k = b_slice.stride()

        # Kernel launch
        _matmul_tile_kernel[grid](
            out_ptr=c_slice,
            a_ptr=a_slice,
            b_ptr=b_slice,
            B=B,
            N=N,
            stride_out_batch=stride_out_batch,
            stride_out_i=stride_out_i,
            stride_out_j=stride_out_j,
            stride_a_batch=stride_a_batch,
            stride_a_i=stride_a_i,
            stride_a_k=stride_a_k,
            stride_b_batch=stride_b_batch,
            stride_b_k=stride_b_k,      # row‑stride for k in B (already H‑size)
            stride_b_j=stride_b_i,      # column‑stride for j in B
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

    # ------------------------------------------------------------------
    # 6. Convert back to fp32 for the remaining ops
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 7. Post‑processing: LayerNorm, gating, final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, weights["to_out.weight"])
    return output
```
