**Kernel design details:**

We need to compute out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d] (over k).

Pseudo-code:

```
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    bh = tl.program_id(0)  # combined batch and hidden
    b = bh // H
    d = bh % H
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute i and j offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # Create range vectors
    i_range = i_base + tl.arange(0, BLOCK_M)
    j_range = j_base + tl.arange(0, BLOCK_N)

    # mask i/j out-of-bounds
    i_mask = i_range < N
    j_mask = j_range < N

    # offsets for left: (i,k) dimension = i_range[:,None] and k_range[None,:]
    # offsets for right: (j,k) dimension transposed: we want (k,j)
    # We'll compute pointer addresses accordingly.
    # Initiate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        k_range = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # left address: b*stride_lb + d*stride_ld + i_range*stride_li + k_range*stride_lk
        left_ptrs = left_ptr + b*stride_lb + d*stride_ld + i_range[:,None]*stride_li + k_range[None,:]*stride_lk
        # right address: b*stride_rb + d*stride_rd + k_range[:,None]*stride_rk + j_range[None,:]*stride_rj
        # Notice we treat k as first dimension, j as second, to have shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + b*stride_rb + d*stride_rd + k_range[:,None]*stride_rk + j_range[None,:]*stride_rj

        # load with mask
        left = tl.load(left_ptrs, mask=i_mask[:,None] & k_mask[None,:], other=0.0)
        right = tl.load(right_ptrs, mask=k_mask[:,None] & j_mask[None,:], other=0.0)

        # Multiply and accumulate: left is (M,K), right is (K,N)
        acc += tl.dot(left, right)   # result (M,N)

    # Store output
    out_ptrs = out_ptr + b*stride_ob + d*stride_od + i_range[:,None]*stride_oi + j_range[None,:]*stride_oj
    tl.store(out_ptrs, acc, mask=i_mask[:,None] & j_mask[None,:])
```

def trimul_kernel_launcher(left, right):
    B, N, N2, H = left.shape
    out = torch.empty_like(left, dtype=left.dtype)   # shape [B,N,N,H]
    # Compute strides
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()
    # Convert to int64
    # Launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B*H, (N+BLOCK_M-1)//BLOCK_M, (N+BLOCK_N-1)//BLOCK_N)
    trimul_kernel[grid](
        left_ptr=left, right_ptr=right, out_ptr=out,
        B=B, N=N, H=H,
        stride_lb= stride_lb, stride_li= stride_li, stride_lk= stride_lk, stride_ld= stride_ld,
        stride_rb= stride_rb, stride_rj= stride_rj, stride_rk= stride_rk, stride_rd= stride_rd,
        stride_ob= stride_ob, stride_oi= stride_oi, stride_oj= stride_oj, stride_od= stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )
    return out

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm x
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    eps = 1e-5
    # Compute per-feature mean and var across last dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_w + norm_b

    # Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask application
    if not config.get("nomask", False):
        mask = mask.to(dtype)  # ensure dtype
        mask = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask
    else:
        # assume mask all ones (no effect)
        pass

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # TriMul contraction with Triton kernel
    out = trimul_kernel_launcher(left, right)  # shape [B,N,N,hidden]

    # to_out_norm:
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # compute LN over hidden_dim
    # shape last dim dim hidden_dim
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # gating
    out_norm = out_norm * out_gate

"""
custom_kernel for the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,C]).
2. Project the normalized tensor to a hidden dimension H with two linear layers
   (left_proj, right_proj).  The resulting tensors have shape [B,N,N,H].
3. Apply an optional pairwise mask (broadcasted on the hidden dimension).
4. Compute three sigmoid gates (left_gate, right_gate, out_gate) from the
   normalized tensor and apply left/right gates element‑wise.
5. Compute the core contraction:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication over the sequence dimension.
   This O(N³) operation is implemented in a Triton kernel that tiles the i‑,
   j‑ and k‑axes (BLOCK_M × BLOCK_N × BLOCK_K) and processes each (b,d) pair
   independently.
6. Layer‑norm the result over the hidden dimension, multiply by the out_gate,
   and apply a final linear projection back to the original feature dimension C.
7. Return the tensor of shape [B,N,N,C].

The Triton kernel is deliberately simple: it loads blocks of the left and
right tensors, performs a dot‑product accumulation, and stores the block of the
output.  All strides are passed explicitly, allowing the kernel to operate on
contiguous tensors of shape (B, N, N, H) without requiring any additional layout
transformations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the O(N³) contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,           # pointers
    B, N, H,                               # runtime sizes
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides (B, i, k, d)
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides (B, j, k, d)
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides (B, i, j, d)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    bh = tl.program_id(0)                 # combined batch * hidden index
    b = bh // H                           # batch index
    d = bh % H                            # hidden index

    pid_i = tl.program_id(1)              # block row (i)
    pid_j = tl.program_id(2)              # block column (j)

    # ------------------------------------------------------------------
    # Offsets for i and j dimensions
    # ------------------------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)

    i_mask = i_range < N
    j_mask = j_range < N

    # accumulator for the block (M, N)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k dimension in blocks
    # ------------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_iter * BLOCK_K
        k_range = k0 + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # left:   (i, k)
        left_offset = (
            b * stride_lb
            + d * stride_ld
            + i_range[:, None] * stride_li
            + k_range[None, :] * stride_lk
        )
        # right:  (k, j) – note we treat k as the leading dimension
        right_offset = (
            b * stride_rb
            + d * stride_rd
            + k_range[:, None] * stride_rk
            + j_range[None, :] * stride_rj
        )

        left = tl.load(left_ptr + left_offset,
                       mask=(i_mask[:, None] & k_mask[None, :]),
                       other=0.0)
        right = tl.load(right_ptr + right_offset,
                        mask=(k_mask[:, None] & j_mask[None, :]),
                        other=0.0)

        # (M, K) @ (K, N) -> (M, N)
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # Store the computed block into the output tensor
    # ------------------------------------------------------------------
    out_offset = (
        b * stride_ob
        + d * stride_od
        + i_range[:, None] * stride_oi
        + j_range[None, :] * stride_oj
    )
    tl.store(out_ptr + out_offset,
             acc,
             mask=(i_mask[:, None] & j_mask[None, :]))
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # Ensure mask is float for multiplication
        mask_f = mask.to(dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction via Triton
    # ------------------------------------------------------------------
    out = trimul_kernel_launcher(left, right)   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6) Output layer norm, out gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    output = F.linear(out, weights["to_out.weight"])

    return output
```
