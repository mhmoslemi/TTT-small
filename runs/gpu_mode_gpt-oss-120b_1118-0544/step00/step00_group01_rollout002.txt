"""
custom_kernel: Triton-accelerated TriMul (outgoing) forward pass.

The heavy O(N³) contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T
using a Triton kernel that processes one (batch, hidden) plane at a time.
All other ops (LayerNorm, linear projections, gating, final LayerNorm & linear)
are performed with PyTorch functions for simplicity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to left, right, output
    B, N, H,                           # batch, seq_len, hidden
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B.T for each (batch, hidden) slice.
    A: (B, N, N, H)   – indexed as (b, i, k, h)
    B: (B, N, N, H)   – indexed as (b, j, k, h)
    C: (B, N, N, H)   – indexed as (b, i, j, h)
    """
    # 3‑D grid: (batch*hidden, block_i, block_j)
    bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Decompose bh into batch and hidden indices
    b = bh // H
    h = bh % H

    # Compute base offsets for this (b, h)
    batch_offset_a = b * stride_a_batch
    batch_offset_b = b * stride_b_batch
    batch_offset_c = b * stride_c_batch
    hidden_offset = h * stride_a_h   # stride_a_h == stride_b_h == stride_c_h (contiguous)

    # Offsets within the N dimension
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A block (shape BLOCK_M x BLOCK_K)
        a_ptrs = a_ptr + batch_offset_a + offs_i[:, None] * stride_a_i \
                 + k_offset[None, :] * stride_a_k + hidden_offset
        a = tl.load(a_ptrs,
                    mask=(offs_i[:, None] < N) & (k_offset[None, :] < N),
                    other=0.0).to(tl.float32)

        # Load B block transposed (shape BLOCK_K x BLOCK_N)
        b_ptrs = b_ptr + batch_offset_b + k_offset[:, None] * stride_b_k \
                 + offs_j[None, :] * stride_b_j + hidden_offset
        b = tl.load(b_ptrs,
                    mask=(k_offset[:, None] < N) & (offs_j[None, :] < N),
                    other=0.0).to(tl.float32)

        # Matrix multiply‑accumulate
        acc += tl.dot(a, b)   # a: (M,K), b: (K,N) → (M,N)

    # Store result
    c_ptrs = c_ptr + batch_offset_c + offs_i[:, None] * stride_c_i \
             + offs_j[None, :] * stride_c_j + hidden_offset
    tl.store(c_ptrs, acc,
             mask=(offs_i[:, None] < N) & (offs_j[None, :] < N))


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              - input_tensor: torch.Tensor of shape [B, N, N, dim]
              - mask_tensor:  torch.Tensor of shape [B, N, N] (may be dummy)
              - weights_dict: mapping of weight names to torch.Tensor
              - config_dict: contains "dim", "hidden_dim" and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    # Extract config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- LayerNorm on input ----------
    # weights for the first LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    # Normalize over the last dim (feature dimension)
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"] # (hidden_dim, dim)

    left = F.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight) # [B,N,N,hidden_dim]

    # ---------- Apply mask (if present) ----------
    if not nomask:
        # mask should broadcast on hidden dim
        mask = mask.to(dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gating ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ---------- Triton kernel: contraction ----------
    # Allocate output tensor for the gated multiplication
    out = torch.empty_like(left)   # same shape [B,N,N,hidden_dim]

    # Compute strides (int64)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    B, N, _, H = left.shape  # H == hidden_dim

    # Choose block sizes; tune empirically for the target GPU
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # reduction block

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left,            # a_ptr
        right,           # b_ptr
        out,             # c_ptr
        B, N, H,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )
    # Ensure kernel completion
    torch.cuda.synchronize(device)

    # ---------- Output LayerNorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    # Multiply by out gate
    out = out * out_gate

    # ---------- Final linear projection ----------
    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    output = F.linear(out, to_out_weight)     # bias=False per definition

    return output
