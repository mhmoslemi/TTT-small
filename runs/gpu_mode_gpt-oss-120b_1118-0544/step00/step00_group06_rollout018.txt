"""
TriMul (outgoing) forward pass with a Triton kernel for the
core 4‑D tensor contraction:
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

The kernel treats each (batch, hidden) pair as a separate
matrix‑multiply (N×N · N×Nᵀ) and runs a batched GEMM in
tiles (BLOCK_M, BLOCK_N, BLOCK_K). All other operations
(LayerNorm, Linear, Sigmoid gating) are performed with
standard PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the einsum part
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # batch, seq_len, hidden
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # block row (i)
    pid_n = tl.program_id(1)          # block col (j)
    pid_bh = tl.program_id(2)         # combined batch * hidden index

    # ------------------------------------------------------------------
    # decode combined batch*hidden index
    # ------------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension tiles
    # ------------------------------------------------------------------
    for k_idx in range(0, NUM_K):
        k_offset = k_idx * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + (
            b * stride_left_b
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + h * stride_left_h
        )
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)

        # right tile transposed: we need right[b, j, k, h] as (k, j)
        # shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + (
            b * stride_right_b
            + offs_k[:, None] * stride_right_k   # k dimension as rows
            + offs_n[None, :] * stride_right_j   # j dimension as cols
            + h * stride_right_h
        )
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)

        # matrix multiplication tile
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # write out result
    # ------------------------------------------------------------------
    c_ptrs = out_ptr + (
        b * stride_out_b
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(c_ptrs, acc,
              mask=(mask_m[:, None] & mask_n[None, :]))


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the "outgoing" TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, [B, N, N, dim]
        - mask_tensor  : torch.Tensor or None, [B, N, N]
        - weights_dict : dict of torch.Tensors (see keys in the prompt)
        - config_dict  : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on input (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias   # broadcast over last dim

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]    # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x_norm, left_proj_weight)        # [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(dtype=dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = F.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = F.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = F.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core einsum via Triton kernel
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)          # shape [B, N, N, hidden_dim]
    out = out.contiguous()

    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tiling parameters
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 16
    NUM_K = (N + BLOCK_K - 1) // BLOCK_K

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_K=NUM_K,
    )

    # ------------------------------------------------------------------
    # LayerNorm on hidden dimension (hidden_dim)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    mean_h = out.mean(dim=-1, keepdim=True)
    var_h = ((out - mean_h) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_h) / torch.sqrt(var_h + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # ------------------------------------------------------------------
    # Output gating and final linear projection
    # ------------------------------------------------------------------
    out_gated = out_norm * out_gate
    to_out_weight = weights["to_out.weight"]          # (dim, hidden_dim)
    output = F.linear(out_gated, to_out_weight)      # [B, N, N, dim]

    return output
