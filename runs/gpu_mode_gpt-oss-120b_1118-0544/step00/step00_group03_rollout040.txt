@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(0)  # batch
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    pid_d = tl.program_id(3)

    # compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)  # inner loop
    
    d_start = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    # masks
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_d = d_start < D

    # pointers with strides
    # left: shape (B,N,N,D) where dimensions ordering: b, i, k, d?
    # Actually left has shape (B,N,N,D) with i as second dim, k as third dim, D last.
    # So indexes: left_ptr + b*stride_left_b + i*stride_left_i + k*stride_left_k + d*stride_left_d
    # right: shape (B,N,N,D) with j as second dim, k as third dim, d last.
    # out: shape (B,N,N,D) (i,j,d).

    # Compute pointers
    left_ptrs = left_ptr + pid_b * stride_left_b \
               + (offs_m[:, None, None] * stride_left_i) \
               + (offs_k[None, :, None] * stride_left_k) \
               + (d_start[None, None, :] * stride_left_d)

    right_ptrs = right_ptr + pid_b * stride_right_b \
                + (offs_n[:, None, None] * stride_right_j) \
                + (offs_k[None, :, None] * stride_right_k) \
                + (d_start[None, None, :] * stride_right_d)

    # Initialize accumulator (BLOCK_M, BLOCK_N, BLOCK_D)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load left and right for current k tile
        lo = tl.load(left_ptrs, mask=mask_m[:, None, None] & (offs_k[None, :] < N) & mask_d[None, None, :], other=0.0)
        ro = tl.load(right_ptrs, mask=mask_n[:, None, None] & (offs_k[None, :] < N) & mask_d[None, None, :], other=0.0)
        # lo shape (M,K,D), ro shape (N,K,D). We need dot over K dimension for each D: left[i,k,d] * right[j,k,d]
        # Use tl.dot for each d: we can loop over d? Could use vectorized dot broadcasting.
        # However, tl.dot doesn't support broadcasting on third dimension. So we need to loop over d slice (BLOCK_D) or use tl.dot per d.
        # Simplify: loop over d in Python? Not possible inside kernel. Use for loop over d dimension (small). BLOCK_D can be small (e.g., 1 or 2)
        for d_idx in range(BLOCK_D):
            # left slice: (M,K)
            lo_d = lo[:, :, d_idx]
            ro_d = ro[:, :, d_idx]
            # Compute dot
            acc[:, :, d_idx] += tl.dot(lo_d, ro_d) # left MxK dot right N x K.T -> shape MxN

        # advance pointers for next K tile
        left_ptrs += BLOCK_K * stride_left_k
        right_ptrs += BLOCK_K * stride_right_k

    # store out
    out_ptrs = out_ptr + pid_b * stride_out_b \
               + (offs_m[:, None, None] * stride_out_i) \
               + (offs_n[None, :, None] * stride_out_j) \
               + (d_start[None, None, :] * stride_out_d)

    tl.store(out_ptrs, acc, mask=mask_m[:, None, None] & mask_n[None, :, None] & mask_d[None, None, :])
```

def custom_kernel(data):
    input, mask, weights, config = data
    bs, N, _, dim = input.shape
    hidden_dim = config['hidden_dim']
    # layernorm on input: using weights['norm.weight'] and bias
    # We can implement LayerNorm via torch.nn.functional.layer_norm: torch.nn.functional.layer_norm(input, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)
    # Could also compute manually with mean and var across dim via torch.
    # Note: We can also implement a custom Triton kernel for layernorm as an example.
    # To incorporate Triton, we will implement a kernel for LayerNorm (norm) and maybe for the main multiplication.
    # Provide both implementations.

    # Steps:
    # 1. LayerNorm using pytorch (or Triton)
    x_norm = torch.nn.functional.layer_norm(input, (dim,), weight=norm_weight, bias=norm_bias)
    # 2. Linear projections using torch.nn.functional.linear
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N, hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    # 3. Masking:
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_
    # 4. Gating using linear + sigmoid
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))
    left = left * left_gate
    right = right * right_gate
    # 5. Compute multiplication using custom Triton kernel compute_out = trmul_einsum(left, right)
    # left,right shape [B,N,N,hidden_dim]
    out = trmul_einsum_kernel(left, right)  # result same shape
    # 6. to_out_norm (LayerNorm over hidden_dim)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    # 7. gating
    out = out * out_gate
    # 8. final projection
    out = torch.nn.functional.linear(out, to_out_weight)
    return out

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # maybe check if mask is None; but they always pass mask
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # layernorm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    
    # projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)       # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    
    # mask
    if mask_tensor is not None:
        mask = mask_tensor.to(left.dtype)
        mask = mask.unsqueeze(-1)  # B,N,N,1
        left = left * mask
        right = right * mask
    else:
        # no mask: treat as ones
        pass
    
    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))
    
    left = left * left_gate
    right = right * right_gate
    
    # compute core out using Triton kernel
    out = trmul_einsum_triton(left, right)  # shape B,N,N,hidden_dim
    
    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    
    # final linear
    out = torch.nn.functional.linear(out, to_out_weight)
    return out

def trmul_einsum_triton(left, right):
    # assert left.shape == right.shape
    B, N, N2, D = left.shape
    assert N == N2
    out = torch.empty_like(left)  # same shape
    # Choose block sizes based on GPU constraints
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8
    for b in range(B):
        l = left[b].contiguous()
        r = right[b].contiguous()
        o = out[b]
        grid = ( (N + BLOCK_M - 1)//BLOCK_M,
                 (N + BLOCK_N - 1)//BLOCK_N,
                 (D + BLOCK_D - 1)//BLOCK_D )

            l, r, o,

            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,

@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, D,
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_j, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)
    
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N
    d_offset = pid_d * BLOCK_D
    
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_d = d_offset + tl.arange(0, BLOCK_D)
    
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D
    
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)
    
    # Loop over K dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_iter * BLOCK_K
        cur_k = offs_k + k_base
        mask_k = cur_k < N
        
        # Load left tile (M,K,D)
        left = tl.load(
            left_ptr + (offs_i[:, None, None] * stride_left_i
                        + cur_k[None, :, None] * stride_left_k
                        + offs_d[None, None, :] * stride_left_d),
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0
        )
        # Load right tile (K,N,D) - note swapped axes compared to right's shape (j,k,d) -> we need (k,j,d)
        right = tl.load(
            right_ptr + (cur_k[:, None, None] * stride_right_k
                         + offs_j[None, :, None] * stride_right_j
                         + offs_d[None, None, :] * stride_right_d),
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
            other=0.0
        )
        # Compute dot for each d
        for d in range(BLOCK_D):
            # left[:, :, d] shape (M,K), right[:, :, d] shape (K,N)
            acc[:, :, d] = acc[:, :, d] + tl.dot(left[:, :, d], right[:, :, d])
        # Loop ends
    # Store result
    out_ptrs = out_ptr + (offs_i[:, None, None] * stride_out_i
                         + offs_j[None, :, None] * stride_out_j
                         + offs_d[None, None, :] * stride_out_d)
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

"""
TriMul (outgoing) implementation with a custom Triton kernel.

The heavy part of TriMul is the einsum
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

which is a batch of D independent matrix multiplications
    left_d @ right_d^T    (N×N  @  N×N → N×N) for each hidden channel d.

We implement this reduction in Triton using a tiled GEMM style kernel.
Each program processes a tile of the (i,j) output plane for a small
subset of hidden channels (BLOCK_D).  The kernel loops over the k
dimension in blocks (BLOCK_K) and accumulates the result in registers.
All remaining operations (LayerNorm, linear projections, gates) are
performed with standard PyTorch calls.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, D,
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_j, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Compute out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d].

    left  : (N, N, D)  indexed as (i, k, d)
    right : (N, N, D)  indexed as (j, k, d)
    out   : (N, N, D)  indexed as (i, j, d)
    """
    pid_i = tl.program_id(0)   # block index along i
    pid_j = tl.program_id(1)   # block index along j
    pid_d = tl.program_id(2)   # block index along hidden dim

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    d_base = pid_d * BLOCK_D

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_d = d_base + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # accumulator [M, N, D]
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k_start = kb * BLOCK_K
        cur_k = offs_k + k_start
        mask_k = cur_k < N

        # left tile: shape (M, K, D)
        left = tl.load(
            left_ptr
            + offs_i[:, None, None] * stride_left_i
            + cur_k[None, :, None] * stride_left_k
            + offs_d[None, None, :] * stride_left_d,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # right tile: shape (K, N, D)  (k dimension first, then j)
        right = tl.load(
            right_ptr
            + cur_k[:, None, None] * stride_right_k
            + offs_j[None, :, None] * stride_right_j
            + offs_d[None, None, :] * stride_right_d,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # dot for each hidden channel slice
        for d in range(BLOCK_D):
            acc[:, :, d] = acc[:, :, d] + tl.dot(left[:, :, d], right[:, :, d])

    # store result
    out_ptrs = (
        out_ptr
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_d[None, None, :] * stride_out_d
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


def trmul_einsum_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched (B,N,N,D) → (B,N,N,D) multiplication using the Triton kernel above.
    """
    B, N, N2, D = left.shape
    assert N == N2, "pairwise dimensions must match"
    out = torch.empty_like(left)

    # Choose tile sizes – these work well on an A200 (SM80)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8  # should divide hidden_dim (128, 384, …)

    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_d = (D + BLOCK_D - 1) // BLOCK_D
    grid = (grid_i, grid_j, grid_d)

    # Triton expects contiguous tensors for predictable strides
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    for b in range(B):
        trmul_kernel[
            grid
        ](
            left[b],
            right[b],
            out[b],
            N,
            D,
            stride_left_i=left.stride(0),
            stride_left_k=left.stride(1),
            stride_left_d=left.stride(2),
            stride_right_j=right.stride(0),
            stride_right_k=right.stride(1),
            stride_right_d=right.stride(2),
            stride_out_i=out.stride(0),
            stride_out_j=out.stride(1),
            stride_out_d=out.stride(2),
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )
    return out


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ----- unpack weights -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----- 1. LayerNorm on the input -----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ----- 2. Linear projections to hidden dimension -----
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----- 3. Apply pair mask (if present) -----
    if mask_tensor is not None:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ----- 4. Gating values -----
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----- 5. Core multiplicative update (einsum) with Triton -----
    out = trmul_einsum_triton(left, right)   # [B,N,N,hidden_dim]

    # ----- 6. Output LayerNorm, gating, and final projection -----
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # back to dim

    return out
