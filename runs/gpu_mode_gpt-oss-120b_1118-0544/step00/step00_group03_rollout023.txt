def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Get weights as described

    # LayerNorm: using torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask handling
    if mask_tensor is not None:
        # ensure mask dtype same as x (float)
        mask_f = mask_tensor.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # Gate computations
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Core einsum
    # out shape [B,N,N,hidden_dim]
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # to_out_norm (LayerNorm on hidden_dim)
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply out_gate using Triton kernel
    out = triton_mul(out, out_gate)   # out_gate shape same

    # Final linear
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

import triton

import math

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, n_elements, BLOCK: tl.constexpr):

    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def triton_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Element‑wise multiplication of two tensors using a tiny Triton kernel."""
    assert a.shape == b.shape, "shapes must match"
    # flatten to 1‑D for the kernel
    a_flat = a.contiguous().view(-1)
    b_flat = b.contiguous().view(-1)
    out_flat = torch.empty_like(a_flat)
    total = a_flat.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](a_flat, b_flat, out_flat, total, BLOCK=BLOCK)
    return out_flat.view_as(a)

    Triton‑accelerated implementation of the outgoing TriMul operator.
    The heavy pairwise multiplication (the einsum) is performed with
    batched GEMM (via torch.bmm) and the final element‑wise gating uses
    a small Triton kernel.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # ------------------------------------------------------------------
    # unpack weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    right_gate_weight = weights["right_gate.weight"]

    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    # LayerNorm on the input (last dimension = dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    # Linear projections to hidden_dim (no bias)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    out = torch.nn.functional.linear(out, to_out_weight)

import torch
import triton
import triton.language as tl


@triton.jit
def _elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, n_elements, BLOCK: tl.constexpr):
    """Simple Triton kernel that computes out = a * b element‑wise."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _triton_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Fuse element‑wise multiplication with a tiny Triton kernel."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication."
    a_flat = a.contiguous().view(-1)
    b_flat = b.contiguous().view(-1)
    out_flat = torch.empty_like(a_flat)
    total = a_flat.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    _elementwise_mul_kernel[grid](a_flat, b_flat, out_flat, total, BLOCK=BLOCK)
    return out_flat.view_as(a)


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.
    Steps:
      1. Layer‑norm the input.
      2. Linear projections to ``hidden_dim`` (left/right).
      3. Optional mask application.
      4. Gating (sigmoid of linear projections) and gate‑modulation.
      5. Core O(N³) reduction via a batched GEMM (torch.bmm) on the
         hidden dimension.
      6. Layer‑norm on the hidden dimension.
      7. Element‑wise out‑gate multiplication via a small Triton kernel.
      8. Final linear projection back to ``dim``.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # unpack model weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape

    # 1. Input LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # 2. Linear projections (no bias)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # 3. Optional mask (broadcast over hidden dimension)
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # 4. Gates + modulation
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core pairwise multiplication: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM over the hidden dimension.
    #    Reshape to (B*H, N, N) and use torch.bmm.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    L = left_perm.view(B * hidden_dim, N, N)            # [B*H, N, N]
    R = right_perm.view(B * hidden_dim, N, N)           # [B*H, N, N]

    out_mat = torch.bmm(L, R.transpose(1, 2))          # [B*H, N, N]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # 7. Apply out‑gate using a Triton kernel (element‑wise multiply)
    out = _triton_mul(out, out_gate)

    # 8. Final linear projection back to original dimension
    out = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
