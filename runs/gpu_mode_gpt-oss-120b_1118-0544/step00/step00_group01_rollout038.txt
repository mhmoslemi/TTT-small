out[b,i,j,d] = sum_{k} left[b,i,k,d] * right[b,j,k,d].

But must implement at least part of the operations in a Triton kernel. So we need to place the contraction operation in a Triton kernel. In addition, to achieve performance, we need to tune block sizes. Also we need to handle both cases: mask may be provided or not. But we can apply the mask before projections.

We'll define:

@triton.jit
def tri_mul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    pid_g = tl.program_id(axis=2)
    # compute batch and hidden index
    b = pid_g // H
    h = pid_g % H

    # Compute offsets for rows and cols
    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute K offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for left: shape [BLOCK_M, BLOCK_K]
        a_ptrs = left_ptr + b * stride_l_b + h * stride_l_h + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        # load left tile
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # pointers for right: shape [BLOCK_K, BLOCK_N] (since we need B_T)
        b_ptrs = right_ptr + b * stride_r_b + h * stride_r_h + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_i
        # load right tile
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product across K dimension: sum over k: accumulator
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

    # Write back if within bounds
    out_ptrs = out_ptr + b * stride_o_b + h * stride_o_h + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # ensure dtype is float32
    # get weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, left_proj_weight)  # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask handling
    if mask is None:
        mask_tensor = torch.ones_like(left[..., :1], dtype=left.dtype, device=left.device)  # shape B,N,N,1
    else:
        mask_tensor = mask.to(left.dtype).unsqueeze(-1)  # shape [B,N,N,1]
    left = left * mask_tensor
    right = right * mask_tensor

    # gating
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # cast to half for kernel
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    B, N, _, _ = left_half.shape
    # allocate output tensor for contraction
    out_contr = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_half.device)

    # Compute strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_half.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_half.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_contr.stride()

    # kernel block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             B * hidden_dim )

    # launch kernel
    tri_mul_contract_kernel[grid](
        left_half,
        right_half,
        out_contr,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Apply second layernorm on hidden dim
    out_norm = torch.nn.functional.layer_norm(out_contr, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply by out_gate and final linear projection
    out_norm = out_norm * out_gate
    out = torch.nn.functional.linear(out_norm, to_out_weight)

    return out
```

"""
TriMul (outgoing) forward pass implemented with a Triton kernel.

Algorithm:
1. Layer‑normalize the input tensor.
2. Apply two linear projections (left/right) and a mask if provided.
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Multiply the gated projections element‑wise.
5. Contract the gated tensors over the intermediate sequence dimension k:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication (for each batch and hidden channel)
   performed by a tiled Triton kernel (BLOCK_M × BLOCK_N tiles, reduction over k).
6. Apply a second LayerNorm on the hidden dimension, multiply by the out‑gate,
   and finally project back to the original dimension.
The kernel works on half‑precision inputs (to save bandwidth) and accumulates
in FP32 for numerical stability.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    pid_g = tl.program_id(axis=2)                  # combined batch & hidden index

    # Decode batch (b) and hidden channel (h) from the combined id
    b = pid_g // H
    h = pid_g % H

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis k in blocks
    num_tiles_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_tiles_k):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers and loads for left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + b * stride_lb + h * stride_lh \
                 + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Pointers and loads for right (transposed on the fly): shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + b * stride_rb + h * stride_rh \
                 + offs_k[:, None] * stride_rk + offs_n[None, :] * stride_ri
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Matrix‑multiply‑accumulate
        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # Write the result back to global memory
    out_ptrs = out_ptr + b * stride_ob + h * stride_oh \
               + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing version).

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model weight tensors
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32, on CUDA).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is None:
        mask_tensor = torch.ones_like(left[..., :1], dtype=left.dtype, device=left.device)
    else:
        mask_tensor = mask.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
    left = left * mask_tensor
    right = right * mask_tensor

    # ------------------------------------------------------------------
    # 4. Compute gates from the normalized input
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Apply gates to the projected tensors
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Contract over the intermediate sequence dimension k using Triton
    # ------------------------------------------------------------------
    # Cast projections to half‑precision for the kernel (uses fp16 loads)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape
    out_contr = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_h.device)

    # Strides (in elements) for each tensor
    stride_lb, stride_li, stride_lk, stride_lh = left_h.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right_h.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out_contr.stride()

    # Tiling parameters – tuned for typical GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    tri_mul_contract_kernel[grid](
        left_h,
        right_h,
        out_contr,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Second LayerNorm over the hidden dimension, out‑gate and final projection
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_contr,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out_norm = out_norm * out_gate
    out = torch.nn.functional.linear(out_norm, to_out_weight)  # [B,N,N,C]

    return out
