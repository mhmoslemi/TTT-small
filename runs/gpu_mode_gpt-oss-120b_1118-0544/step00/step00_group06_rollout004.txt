def custom_kernel(data):
    x, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    eps = 1e-5

    # layer norm on input
    # compute mean / var over last dim
    # use torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(x, normalized_shape=(dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    if not config.get("nomask", True):
        # apply mask mask shape [B,N,N], broadcast to last dim
        mask = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # compute core trimul
    out = triton_tri_mul(left, right)   # returns tensor shape [B,N,N,hidden_dim] float32

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,),
                                         weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
```

"""
custom_kernel: Triton-accelerated outgoing Triangle Multiplicative Update (TriMul)

The kernel performs the expensive O(N³) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left/right are projected and gated versions of the input.
All surrounding linear layers, LayerNorms and sigmoid gates are handled in
PyTorch; the core multiplication is fused into a Triton GEMM that treats the
hidden dimension as an extra batch axis.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # batch, sequence length, hidden dim
    stride_lb, stride_li, stride_lk, stride_lh, # left strides (B, i, k, h)
    stride_rb, stride_rj, stride_rk, stride_rh, # right strides (B, j, k, h)
    stride_ob, stride_oi, stride_oj, stride_oh, # out strides (B, i, j, h)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]  (h = hidden)
    Grid:
        axis 0 : block over (i, j) tiles (i_block * num_j_blocks + j_block)
        axis 1 : combined (batch, hidden) index = b * H + h
    """
    pid0 = tl.program_id(0)   # tile id over i/j
    pid1 = tl.program_id(1)   # batch*hidden idx

    # decode batch and hidden index
    h = pid1 % H
    b = pid1 // H

    # number of tiles in each dimension
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    i_block = pid0 // num_j
    j_block = pid0 % num_j

    # offsets within the tile
    i_offset = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_block * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_offset < N
    j_mask = j_offset < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension tiles
    for k_tile in range(tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # left block: shape (BLOCK_M, BLOCK_K)  -> indices (i, k)
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + i_offset[:, None] * stride_li
            + k_offset[None, :] * stride_lk
            + h * stride_lh
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        )

        # right block transposed: shape (BLOCK_K, BLOCK_N)  -> indices (k, j)
        # We reinterpret right as (k, j) by swapping strides.
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + k_offset[:, None] * stride_rk
            + j_offset[None, :] * stride_rj
            + h * stride_rh
        )
        right = tl.load(
            right_ptrs,
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0,
        )

        # GEMM accumulation
        acc += tl.dot(left, right)

    # store the result
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + i_offset[:, None] * stride_oi
        + j_offset[None, :] * stride_oj
        + h * stride_oh
    )
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton wrapper for the O(N³) TriMul core.
    left/right: [B, N, N, H] (float32, contiguous)
    Returns: out of same shape and dtype.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides (as Python ints)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Tile sizes – a trade‑off between occupancy and shared memory usage.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (i*j tiles, batch*hidden)
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_i * grid_j, B * H)

    _tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_lh=stride_lh,
        stride_rb=stride_rb,
        stride_rj=stride_rj,
        stride_rk=stride_rk,
        stride_rh=stride_rh,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_oh=stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using Triton for the O(N³) reduction.

    Arguments
    ---------
    data : tuple
        (input, mask, weights, config) where
            - input : torch.Tensor [B, N, N, dim]
            - mask  : torch.Tensor [B, N, N] (may be omitted if nomask=True)
            - weights : dict of torch.Tensor containing all model parameters
            - config  : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as input)
    """
    # Unpack
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------------------
    # 1. Input LayerNorm
    # ---------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ---------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ---------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ---------------------------------------------------------------------
    # 3. Optional mask (broadcasted over hidden dim)
    # ---------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_broadcast = mask.unsqueeze(-1).to(dtype)
        left = left * mask_broadcast
        right = right * mask_broadcast

    # ---------------------------------------------------------------------
    # 4. Gates (sigmoid after linear)
    # ---------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------------------
    # 5. Core TriMul reduction (Triton)
    # ---------------------------------------------------------------------
    # Ensure contiguous tensors for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = _tri_mul_triton(left, right)  # shape [B, N, N, hidden_dim]

    # ---------------------------------------------------------------------
    # 6. Output LayerNorm
    # ---------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ---------------------------------------------------------------------
    # 7. Output gating and final projection
    # ---------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
