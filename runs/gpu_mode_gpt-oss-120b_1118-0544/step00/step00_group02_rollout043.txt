    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 8

        cur_h = h_end - h_start

        left_slice = left[..., h_start:h_end]   # shape B,N,N,cur_h

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_j_o, stride_h_o,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # base indices
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # range offsets
    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    # masks for bounds
    mask_i = i < N
    mask_j = j < N
    mask_h = h < H

    # compute base offsets for batch
    batch_offset_l = pid_b * stride_b_l
    batch_offset_r = pid_b * stride_b_r
    batch_offset_o = pid_b * stride_b_o

    # accumulator in float32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over K dimension
    for k_offset in range(0, N, BLOCK_K):
        # calculate actual K size (for mask)
        cur_k = tl.minimum(N - k_offset, BLOCK_K)
        # compute k indices
        k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # load left block (I,K,H)
        left_ptrs = left_ptr + batch_offset_l \
            + i[:, None, None] * stride_i_l \
            + k[None, :, None] * stride_k_l \
            + h[None, None, :] * stride_h_l
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0).to(tl.float32)

        # load right block transposed (K,J,H)
        right_ptrs = right_ptr + batch_offset_r \
            + k[:, None, None] * stride_k_r \
            + j[None, :, None] * stride_i_r \
            + h[None, None, :] * stride_h_r
        right = tl.load(right_ptrs, mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0).to(tl.float32)

        # compute dot for each h
        for hh in range(0, BLOCK_H):
            # compute dot of left (I,K) and right (K,J)
            # left slice shape (BLOCK_I, BLOCK_K)
            # right slice shape (BLOCK_K, BLOCK_J)
            l = left[:, :, hh]   # shape (I,K)
            r = right[:, :, hh]   # shape (K,J)
            # accumulate dot
            acc[:, :, hh] += tl.dot(l, r)

    # store out
    out_ptrs = out_ptr + batch_offset_o \
        + i[:, None, None] * stride_i_o \
        + j[None, :, None] * stride_j_o \
        + h[None, None, :] * stride_h_o
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    device = input_tensor.device

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # cast to half for lower memory
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # mask
    if not nomask:
        # mask may be bool; convert to same dtype as left
        mask_f = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid().to(torch.float16)
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid().to(torch.float16)
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()  # keep float32 for later

    left = left * left_gate
    right = right * right_gate

    # Compute triangular multiplication using Triton
    out = tri_mul_triton(left, right)  # returns float32

    # Apply final LayerNorm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # Gate and linear projection
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out

def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_j_o, stride_h_o,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # start indices
    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J

    # offsets within tile
    i = i0 + tl.arange(0, BLOCK_I)
    j = j0 + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    # bounds masks
    mask_i = i < N
    mask_j = j < N
    mask_h = h < H

    # batch offset
    batch_off_l = pid_b * stride_b_l
    batch_off_r = pid_b * stride_b_r
    batch_off_o = pid_b * stride_b_o

    # accumulator float32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

The kernel computes, for each batch b, hidden channel h and pair (i, j):
    out[b, i, j, h] = sum_{k=0}^{N-1} left[b, i, k, h] * right[b, j, k, h]

All LayerNorms, linear projections and gates are performed with PyTorch,
while the O(N³) reduction is fused in Triton.  The implementation works for
any batch size (1‑2), sequence length up to 1024 and hidden dimensions that
fit the block size (default 8).  Memory usage is reduced by storing the
projected tensors in fp16; the reduction accumulates in fp32 and the final
output is returned in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_j_o, stride_h_o,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    # 3‑D grid: (i_tile, j_tile, batch)
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # tile origins
    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J

    # thread‑local indices
    i = i0 + tl.arange(0, BLOCK_I)
    j = j0 + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    # bounds masks
    mask_i = i < N
    mask_j = j < N
    mask_h = h < H

    # batch offsets
    off_b_l = pid_b * stride_b_l
    off_b_r = pid_b * stride_b_r
    off_b_o = pid_b * stride_b_o

    # accumulator (float32)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the reduction dimension k
    # -----------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left tile: (i, k, h)  – original layout
        left_ptrs = left_ptr + off_b_l \
            + i[:, None, None] * stride_i_l \
            + k[None, :, None] * stride_k_l \
            + h[None, None, :] * stride_h_l
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # right tile transposed: (k, j, h)  – we load with k as the leading dim
        right_ptrs = right_ptr + off_b_r \
            + k[:, None, None] * stride_k_r \
            + j[None, :, None] * stride_i_r \
            + h[None, None, :] * stride_h_r
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # per‑channel matrix multiply: (i,k) @ (k,j) -> (i,j)
        for hh in range(0, BLOCK_H):
            l = left[:, :, hh]          # (BLOCK_I, BLOCK_K)
            r = right[:, :, hh]         # (BLOCK_K, BLOCK_J)
            acc[:, :, hh] += tl.dot(l, r)

    # -----------------------------------------------------------------
    # Write the result (float32)
    # -----------------------------------------------------------------
    out_ptrs = out_ptr + off_b_o \
        + i[:, None, None] * stride_i_o \
        + j[None, :, None] * stride_j_o \
        + h[None, None, :] * stride_h_o
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton implementation of the outgoing triangle multiplicative update.
    `left` and `right` must be [B, N, N, H] tensors of dtype torch.float16.
    Returns a float32 tensor of the same shape.
    """
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # -----------------------------------------------------------------
    # Block configurations – adjust if needed for a different GPU
    # -----------------------------------------------------------------
    BLOCK_I = 16   # tile size along i
    BLOCK_J = 16   # tile size along j
    BLOCK_K = 32   # reduction tile size
    BLOCK_H = 8    # number of hidden channels processed together

    # Strides for the three tensors
    stride_b_l, stride_i_l, stride_k_l, stride_h_l = left.stride()
    stride_b_r, stride_i_r, stride_k_r, stride_h_r = right.stride()
    stride_b_o, stride_i_o, stride_j_o, stride_h_o = out.stride()

    # Process the hidden dimension in blocks of BLOCK_H
    for h_start in range(0, H, BLOCK_H):
        cur_h = min(BLOCK_H, H - h_start)

        left_slice = left[..., h_start : h_start + cur_h]
        right_slice = right[..., h_start : h_start + cur_h]
        out_slice = out[..., h_start : h_start + cur_h]

        # launch grid (i_tiles, j_tiles, batch)
        grid = (
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J,
            B,
        )

        tri_mul_kernel[grid](
            left_slice,
            right_slice,
            out_slice,
            B,
            N,
            cur_h,
            stride_b_l,
            stride_i_l,
            stride_k_l,
            stride_h_l,
            stride_b_r,
            stride_i_r,
            stride_k_r,
            stride_h_r,
            stride_b_o,
            stride_i_o,
            stride_j_o,
            stride_h_o,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_H=cur_h,
            num_warps=4,
        )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask : torch.Tensor [B, N, N] (may be None)
        - weights : dict of model parameters
        - config : dict containing "dim", "hidden_dim", "nomask"
    Returns
    -------
    torch.Tensor : [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias) -> fp16 for memory saving
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16) # [B, N, N, H]

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; convert to same dtype as left/right
        mask_f = mask.to(left.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid) – cast to fp16 to stay in the same precision
    # -----------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid().to(torch.float16)
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid().to(torch.float16)
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()  # keep fp32 for later

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Triangular multiplicative update (outgoing) via Triton
    # -----------------------------------------------------------------
    out = tri_mul_triton(left, right)   # [B, N, N, H]  float32

    # -----------------------------------------------------------------
    # 6. Final LayerNorm, gating and output projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # back to original dim

    return out
