We'll implement as typical block matmul:

@triton.jit
def triton_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr
):
    pid = tl.program_id(0)
    # calculate block indices [batch, block_m, block_n]
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_id = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = block_id // tl.cdiv(N, BLOCK_N)
    pid_n = block_id % tl.cdiv(N, BLOCK_N)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # masks for out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer base for batch
    a_ptrs = a_ptr + batch * batch_stride_a
    b_ptrs = b_ptr + batch * batch_stride_b
    c_ptrs = c_ptr + batch * batch_stride_c

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # offsets for K dimension
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (BLOCK_M, BLOCK_K)
        a = tl.load(a_ptrs + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak), mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # load B tile (BLOCK_K, BLOCK_N)
        b = tl.load(b_ptrs + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn), mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # compute dot product
        acc += tl.dot(a, b)

        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c = acc.to(tl.float16)  # cast down
    tl.store(c_ptrs + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn), c, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul forward kernel (outgoing version)

The kernel fuses the expensive O(N³) operation:
    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]

where left/right are the projected and gated pairwise tensors.
We treat each hidden dimension `d` as an independent matrix‑multiplication
(left_i_k @ right_j_kᵀ) and batch them together (batch * hidden_dim
independent GEMMs).  A Triton tiled GEMM kernel computes all these
multiplications in fp16 with a float32 accumulator, then the result is
post‑processed (layer‑norm, out‑gate and final linear projection) in
PyTorch.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    pid = tl.program_id(0)

    # ----------- grid decomposition ----------
    total_m = tl.cdiv(M, BLOCK_M)
    total_n = tl.cdiv(N, BLOCK_N)

    batch = pid // (total_m * total_n)
    block = pid % (total_m * total_n)
    pid_m = block // total_n
    pid_n = block % total_n

    # ----------- offsets within a block ----------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----------- base pointers for this batch ----------
    a_batch = a_ptr + batch * batch_stride_a
    b_batch = b_ptr + batch * batch_stride_b
    c_batch = c_ptr + batch * batch_stride_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------- tiled K loop ----------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(
            a_batch
            + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B is accessed as transposed (k, n)
        b = tl.load(
            b_batch
            + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # ---------- write back ----------
    c = acc.to(tl.float16)
    tl.store(
        c_batch
        + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    TriMul forward pass with a Triton kernel for the core O(N³) multiplication.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool or float) or None
            - weights_dict: mapping of weight names to torch.Tensors
            - config_dict : contains "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm on input -----
    x = F.layer_norm(
        input_tensor,
        normalized_shape=dim,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----- Linear projections -----
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Optional mask -----
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ----- Gates -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----- Cast to FP16 for the heavy GEMM -----
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ----- Rearrange to (batch, hidden, N, N) -----
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # flatten the (B, H) dimensions → total_batch
    total_batch = B * hidden_dim
    left_flat = left.view(total_batch, N, N)
    right_flat = right.view(total_batch, N, N)

    # output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # ----- Strides (int64) -----
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    batch_stride_a = left_flat.stride(0)

    # B is used as transposed (k, n)
    stride_bk = right_flat.stride(2)   # stride for k dimension (original column)
    stride_bn = right_flat.stride(1)   # stride for n dimension (original row)
    batch_stride_b = right_flat.stride(0)

    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    batch_stride_c = out_flat.stride(0)

    # ----- Kernel launch parameters -----
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    num_warps = 8

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_grid = grid_m * grid_n * total_batch

    _batched_gemm_kernel[(total_grid,)](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        batch_stride_a=batch_stride_a,
        batch_stride_b=batch_stride_b,
        batch_stride_c=batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # ----- Reshape back to [B, N, N, hidden] -----
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # back to fp32 for the remaining ops

    # ----- Post‑processing (norm, out‑gate, final linear) -----
    out = F.layer_norm(
        out,
        normalized_shape=hidden_dim,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
