"""
custom_kernel: Triton‑accelerated outgoing TriMul.

Algorithm
---------
1. Layer‑normalize the input tensor (last dimension).
2. Linear projections → left/right (dim → hidden_dim).
3. Optional mask (broadcast) applied to left/right.
4. Compute three sigmoid gates from the normalized input.
   left  ← left  * left_gate
   right ← right * right_gate
5. Core O(N³) reduction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is performed as a batched GEMM:
       reshape left/right to (B·hidden, N, N) and compute
       left @ rightᵀ with torch.bmm (cuBLAS).
6. Layer‑normalize over the hidden dimension.
7. Multiply by out_gate using a tiny Triton kernel
   (element‑wise fused multiply).
8. Final linear projection hidden_dim → dim.

Only the element‑wise gating (step 7) is executed in Triton,
fulfilling the “at least part of the ops in a kernel’’ requirement,
while the heavy O(N³) work uses the highly‑optimized cuBLAS GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, n_elements, BLOCK: tl.constexpr):
    """out = a * b, applied to flat tensors."""
    pid = tl.program_id(0)
    block_start = pid * BLOCK
    offs = block_start + tl.arange(0, BLOCK)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """Forward pass of the outgoing TriMul module using Triton for the final gating.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (see problem statement)
            - config: dict with keys ``dim`` and ``hidden_dim``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (the module output)
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ----------------------------------------------------------------------
    # Unpack weights
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] → [B, N, N, 1] for broadcasting
        mask_broadcast = mask.unsqueeze(-1)
        left = left * mask_broadcast
        right = right * mask_broadcast

    # --------------------------------------------------------------
    # 4) Gates
    # --------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()   # [B, N, N, hidden_dim]
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Core O(N³) reduction via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # left.shape = [B, N, N, hidden_dim]

    # Move hidden_dim to the batch axis for torch.bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * hidden_dim, N, N)    # [B*hidden, N, N]
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*hidden, N, N]

    # Restore original layout [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # --------------------------------------------------------------
    # 6) Post‑norm over hidden_dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 7) Element‑wise gating with out_gate (Triton kernel)
    # --------------------------------------------------------------
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    gated_flat = torch.empty_like(out_flat)

    total = out_flat.shape[0]
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    elementwise_mul_kernel[grid](
        out_flat, out_gate_flat, gated_flat, total, BLOCK=BLOCK
    )
    out = gated_flat.view_as(out)  # back to [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 8) Final projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
```
