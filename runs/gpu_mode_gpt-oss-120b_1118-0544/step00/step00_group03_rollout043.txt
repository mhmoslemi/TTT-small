@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    num_i = tl.cdiv(N, BLOCK_M)
    b = pid // num_i
    i_block = pid % num_i
    j_block = tl.program_id(1)

    i_start = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_start = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    d_base = tl.arange(0, BLOCK_D)

    # Masks for out-of-bounds for i and j
    i_mask = i_start < N
    j_mask = j_start < N
    d_mask = d_base < hidden

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over k dimension in tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k_start = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        # Mask for k dimension within tile
        k_mask = k_start < N

        # Load left tile: shape (BLOCK_M, BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr + b*stride_lb + i_start[:, None, None]*stride_li + k_start[None, :, None]*stride_lk + d_base[None, None, :]*stride_ld
        left_tile = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :], other=0.0)

        # Load right tile: shape (BLOCK_N, BLOCK_K, BLOCK_D)
        right_ptrs = right_ptr + b*stride_rb + j_start[:, None, None]*stride_rj + k_start[None, :, None]*stride_rk + d_base[None, None, :]*stride_rd
        right_tile = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :], other=0.0)

        # Compute outer product for each k in tile
        for kk_idx in range(0, BLOCK_K):
            # Check if this k index is valid
            valid = k_mask[kk_idx]
            # left slice
            l = left_tile[:, kk_idx, :]  # (M, D)
            r = right_tile[:, kk_idx, :] # (N, D)

            # if not valid: skip
            # We can mask multiplication with valid
            # Broadcast l and r
            prod = l[:, None, :] * r[None, :, :]  # (M,N,D)
            # Apply i, j masks as well
            prod = prod * tl.where(i_mask[:, None, None] & j_mask[None, :, None] & valid, 1.0, 0.0)
            acc += prod

    # Store result
    out_ptrs = out_ptr + b*stride_ob + i_start[:, None, None]*stride_oi + j_start[None, :, None]*stride_oj + d_base[None, None, :]*stride_od
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :])
```

    x, mask, weights, config = data
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    # extract weights:
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]; right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # [B,N,N,hidden]

        left = left * mask_f
        right = right * mask_f

    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    right = right * right_gate

    # Cast to float16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output placeholder
    out_fp16 = torch.empty_like(left_fp16)

    # strides
    stride_lb, stride_li, stride_lk, stride_ld = left_fp16.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_fp16.stride()

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_D = 16
    BLOCK_K = 32

    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N

def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_D: tl.constexpr, BLOCK_K: tl.constexpr
):

    num_i = tl.cdiv(N, BLOCK_M)
    b = pid // num_i
    i_block = pid % num_i
    j_block = tl.program_id(1)

    i_start = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_start = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    d_idx = tl.arange(0, BLOCK_D)

    i_mask = i_start < N
    j_mask = j_start < N
    d_mask = d_idx < hidden

def custom_kernel(data):
    import torch, triton, triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # optionally get other config

    # Step 1: LayerNorm
    norm_weight, norm_bias = weights["norm.weight"], weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Step 2: Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape B,N,N,hidden
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Step 3: Mask
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # shape B,N,N,1
        left = left * mask
        right = right * mask

    # Step 4: Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Prepare output fp16 tensor
    out_fp16 = torch.empty_like(left_fp16)  # shape B,N,N,hidden

    # Strides
    stride_lb, stride_li, stride_lk, stride_ld = left_fp16.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_fp16.stride()

    B, N = left_fp16.shape[0], left_fp16.shape[1]
    # Kernel block sizes
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_D = 16
    BLOCK_K = 32

    # Compute grid dimensions
    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    num_d = (hidden_dim + BLOCK_D - 1) // BLOCK_D
    grid = (B * num_i, num_j, num_d)

    # Launch kernel
    tri_mul_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_D=BLOCK_D, BLOCK_K=BLOCK_K
    )

    # Convert back to float32
    out = out_fp16.to(torch.float32)

    # Step 5: Output normalization and projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
```

"""
TriMul (outgoing) forward implementation using Triton for the core
pairwise reduction.

Algorithm:
1. Layer‑norm the input tensor.
2. Linear projections (left/right) and optional masking.
3. Compute gating scalars (sigmoid of linear projections) and apply.
4. Compute the einsum
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   with a custom Triton kernel that tiles the (i,j,d) output and
   iterates over k in blocks, accumulating in fp32.
5. Apply a second LayerNorm, multiply by the output gate and a final
   linear projection back to the original channel dimension.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_D: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    # number of tiles in each dimension
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    num_d = tl.cdiv(H, BLOCK_D)

    # decode batch, i‑tile, j‑tile and d‑tile indices
    batch = pid0 // num_i
    i_tile = pid0 % num_i
    j_tile = pid1
    d_tile = pid2

    # start indices of the tile
    i_start = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_start = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    d_start = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)

    # bounds masks
    i_mask = i_start < N
    j_mask = j_start < N
    d_mask = d_start < H

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over k dimension in blocks of size BLOCK_K
    cur_k = 0
    while cur_k < N:
        k_idx = cur_k + tl.arange(0, BLOCK_K)
        k_mask = k_idx < N

        # -----------------------------------------------------------------
        # Load a tile of left: shape (M, K, D)
        left_ptrs = (
            left_ptr
            + batch * stride_lb
            + i_start[:, None, None] * stride_li
            + k_idx[None, :, None] * stride_lk
            + d_start[None, None, :] * stride_ld
        )
        left_tile = tl.load(
            left_ptrs,
            mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
            other=0.0,
        )

        # Load a tile of right transposed: shape (K, N, D)
        right_ptrs = (
            right_ptr
            + batch * stride_rb
            + k_idx[:, None, None] * stride_rk
            + j_start[None, :, None] * stride_rj
            + d_start[None, None, :] * stride_rd
        )
        right_tile = tl.load(
            right_ptrs,
            mask=k_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :],
            other=0.0,
        )
        # -----------------------------------------------------------------

        # accumulate dot products for each d in the block
        for d in range(0, BLOCK_D):
            a = left_tile[:, :, d]          # (M, K)
            b = right_tile[:, :, d]         # (K, N)
            # tl.dot returns the same dtype as inputs (fp32 here)
            acc[:, :, d] += tl.dot(a, b).to(tl.float32)

        cur_k += BLOCK_K

    # store the result
    out_ptrs = (
        out_ptr
        + batch * stride_ob
        + i_start[:, None, None] * stride_oi
        + j_start[None, :, None] * stride_oj
        + d_start[None, None, :] * stride_od
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float32),
        mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul)
    using a Triton kernel for the heavy einsum reduction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C]
            - mask: torch.Tensor [B, N, N] (or None)
            - weights: dict of model parameters (LayerNorm, Linear weights)
            - config: dict with at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )  # [B,N,N,dim]

    # 2) Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # 3) Optional mask
    if mask is not None and mask.numel() > 0:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # 4) Gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core einsum via Triton kernel
    B, N = left.shape[0], left.shape[1]

    # Prepare output tensor
    out_core = torch.empty_like(left)

    # Strides (in elements)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_core.stride()

    # Blocking parameters (tuned for the B200 GPU)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_D = 16
    BLOCK_K = 32

    # Grid dimensions
    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    num_d = (hidden_dim + BLOCK_D - 1) // BLOCK_D
    grid = (B * num_i, num_j, num_d)

    # Launch kernel
    tri_mul_kernel[grid](
        left,
        right,
        out_core,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_D=BLOCK_D,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6) Post‑processing: second LayerNorm, output gate, final projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out = torch.nn.functional.layer_norm(
        out_core, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)  # back to [B,N,N,dim]

    return out
