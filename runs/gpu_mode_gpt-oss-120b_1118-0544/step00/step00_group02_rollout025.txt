"""
TriMul (outgoing) forward pass.

The heavy reduction
    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
is executed by a custom Triton kernel that performs, for each batch
and hidden dimension, a matrix multiplication
    left[b, d] @ right[b, d].T .
All other operations (LayerNorms, linear projections, gating,
masking, output normalization and projection) are carried out with
PyTorch. The kernel works on tensors of shape [B*H, N, N] where
H = hidden_dim and computes the result in blocks (default 64×64×32).
"""

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_j, stride_right_k,
    stride_out_bh, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication for the TriMul reduction.

    left  : [BH, N, N]   (i, k)
    right : [BH, N, N]   (j, k) – we read it as (k, j) to get a dot product.
    out   : [BH, N, N]   (i, j)
    """
    bh   = tl.program_id(0)          # index over batch * hidden_dim
    pid_i = tl.program_id(1)         # block row index
    pid_j = tl.program_id(2)         # block col index

    # coordinate ranges for this block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block (i, k)
        left_ptrs = left_ptr + bh * stride_left_bh \
                    + offs_i[:, None] * stride_left_i \
                    + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right block transposed (k, j)
        right_ptrs = right_ptr + bh * stride_right_bh \
                     + offs_k[:, None] * stride_right_k \
                     + offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # write the result
    out_ptrs = out_ptr + bh * stride_out_bh \
                + offs_i[:, None] * stride_out_i \
                + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple(input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm on the input channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional pair mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_broadcast = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_broadcast
        right = right * mask_broadcast

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the Triton kernel: shape (BH, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # move hidden dim to the second axis and make it contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_view = left_perm.view(BH, N, N)
    right_view = right_perm.view(BH, N, N)

    # allocate output buffer
    out_view = torch.empty_like(left_view)

    # strides (in elements, not bytes)
    stride_left_bh, stride_left_i, stride_left_k = left_view.stride()
    stride_right_bh, stride_right_j, stride_right_k = right_view.stride()
    stride_out_bh, stride_out_i, stride_out_j = out_view.stride()

    # ------------------------------------------------------------------
    # Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    tri_mul_kernel[grid](
        left_view,
        right_view,
        out_view,
        N,
        stride_left_bh,
        stride_left_i,
        stride_left_k,
        stride_right_bh,
        stride_right_j,
        stride_right_k,
        stride_out_bh,
        stride_out_i,
        stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_view.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
