    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,

    pid_n = tl.program_id(1)
    pid_z = tl.program_id(2)

    num_hidden = D
    b = pid_z // num_hidden
    d = pid_z % num_hidden

    # Compute start offsets
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # Ranges

    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    right_base = right_ptr + b * stride_r_b + d * stride_r_d
    out_base = out_ptr + b * stride_o_b + d * stride_o_d

        kk = k0 + tl.arange(0, BLOCK_K)

    )

    # Projections
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # Apply mask if provided and not nomask
    if not config.get("nomask", False) and mask is not None:
        mask = mask.to(x.dtype)
        mask = mask.unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Cast to half for the reduction kernel (reduces memory bandwidth)
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    BLOCK_N = 64
    BLOCK_K = 32

        (seq_len + BLOCK_N - 1) // BLOCK_N,

    )

    r_strides = right.stride()

        o_strides[3],

"""
custom_kernel: Outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor over its last dimension (dim).
2. Linear projections (no bias) → left_proj, right_proj  (shape [..., hidden_dim]).
3. Compute three gate vectors from the same normalized input and apply sigmoid.
4. Optionally mask left/right projections (mask shape [B,N,N]).
5. Apply the left/right gates element‑wise.
6. Reduce over the middle sequence dimension k:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This reduction is a batched matrix multiplication per hidden channel.
   It is implemented in Triton as a GEMM where each program computes a
   (BLOCK_M×BLOCK_N) tile for a single batch element and a single hidden
   channel (the third program‑id axis encodes batch*hidden_dim).
7. Layer‑norm over the hidden dimension, multiply by the output‑gate,
   then a final linear projection back to dim.
All heavy reduction work (step 6) runs on GPU via a Triton kernel, while
the inexpensive point‑wise operations use PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tr_mul_gemm_one_channel_kernel(
    left_ptr,          # const half*
    right_ptr,         # const half*
    out_ptr,           # float*
    B, N, D,           # batch, seq_len, hidden_dim (int)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_z = tl.program_id(2)          # batch * hidden_dim index

    # ----- decode batch index and hidden channel -----
    hidden = D
    b = pid_z // hidden                # batch index
    d = pid_z % hidden                 # hidden channel

    # ----- tile start coordinates -----
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)    # row indices within the tile
    j = j0 + tl.arange(0, BLOCK_N)    # col indices within the tile

    mask_i = i < N
    mask_j = j < N

    # ----- base pointers for this batch & channel -----
    left_base  = left_ptr  + b * stride_l_b + d * stride_l_d
    right_base = right_ptr + b * stride_r_b + d * stride_r_d
    out_base   = out_ptr   + b * stride_o_b + d * stride_o_d

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)

    for k in range(0, num_k_blocks):
        k0 = k * BLOCK_K
        kk = k0 + tl.arange(0, BLOCK_K)
        mask_k = kk < N

        # ----- load left tile: (M, K) -----
        left_ptrs = left_base + (i[:, None] * stride_l_i) + (kk[None, :] * stride_l_k)
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)          # half -> promoted automatically in dot

        # ----- load right tile: (K, N) -----
        right_ptrs = right_base + (j[None, :] * stride_r_i) + (kk[:, None] * stride_r_k)
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)         # half

        # ----- accumulate -----
        acc += tl.dot(left_tile, right_tile)   # (M,N) += (M,K)·(K,N) in FP32

    # ----- store result -----
    out_ptrs = out_base + (i[:, None] * stride_o_i) + (j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask         : torch.Tensor [B, N, N] (bool/float) or None
        weights      : dict of model parameters (torch.Tensors)
        config       : dict with at least 'dim', 'hidden_dim' and optional 'nomask', 'eps'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask = mask.to(input_tensor.dtype).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Reduce over the middle dimension (Trimeric multiplication)
    #    Implemented as batched GEMM per hidden channel via Triton.
    # ------------------------------------------------------------------
    # Cast to half for bandwidth‑limited kernel
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # Output tensor (float32)
    out_hidden = torch.empty((B, N, N, hidden_dim),
                             dtype=torch.float32,
                             device=input_tensor.device)

    # Triton launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over i
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over j
        B * hidden_dim                  # combined batch & channel dimension
    )

    # strides (as Python ints)
    s_l = left.stride()
    s_r = right.stride()
    s_o = out_hidden.stride()

    tr_mul_gemm_one_channel_kernel[
        grid
    ](
        left,
        right,
        out_hidden,
        B,
        N,
        hidden_dim,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7. Output layer‑norm, gate, and final linear projection back to dim
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(out_hidden,
                            normalized_shape=(hidden_dim,),
                            weight=weights["to_out_norm.weight"],
                            bias=weights["to_out_norm.bias"],
                            eps=eps)

    out = out_norm * out_gate
    output = F.linear(out, weights["to_out.weight"])
    return output
