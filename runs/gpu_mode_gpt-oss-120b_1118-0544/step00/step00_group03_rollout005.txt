"""
tri_mul_outgoing.py

Implements the “outgoing” TriangleMultiplicativeUpdate (TriMul) forward pass
used in AlphaFold‑3. The heavy O(N³) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a custom Triton kernel that computes a batched
matrix‑multiply for each (batch, hidden) pair (left @ rightᵀ). All other
operations (LayerNorms, linear projections, gating) are done with PyTorch
operations for simplicity and numerical stability.

Supported input shapes:
    B ∈ {1,2}, N ∈ {128,…,1024}, dim ∈ {128,384}, hidden_dim = 128
The kernel works for any N, batch size and hidden dimension (passed as a
runtime argument). The final output is float32.
"""
import torch
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: computes per‑channel matrix product
#   C[b,d,i,j] = Σ_k A[b,d,i,k] * B[b,d,j,k]   (B is read transposed)
# where A = left, B = right.
# -------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # problem size
    B, M, N, K,
    # strides for A (left)
    stride_ab, stride_ai, stride_ak, stride_ad,
    # strides for B (right, transposed in‑kernel)
    stride_bb, stride_bk, stride_bj, stride_bd,
    # strides for C (output)
    stride_cb, stride_ci, stride_cj, stride_cd,
    # hidden dimension (channels per batch)
    hidden_dim,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ---------------------------------------------------------
    # Decode the combined batch index (batch * hidden_dim)
    # ---------------------------------------------------------
    pid = tl.program_id(0)               # linear index over (B * hidden_dim)
    batch_idx = pid // hidden_dim        # original batch index
    d = pid % hidden_dim                 # channel index inside the batch

    # base pointers for this (batch, channel)
    a_base = a_ptr + batch_idx * stride_ab + d * stride_ad
    b_base = b_ptr + batch_idx * stride_bb + d * stride_bd
    c_base = c_ptr + batch_idx * stride_cb + d * stride_cd

    # ---------------------------------------------------------
    # 2‑D grid indices for the output matrix tiles
    # ---------------------------------------------------------
    pid_i = tl.program_id(1)  # tile row index
    pid_j = tl.program_id(2)  # tile column index

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------
    # Loop over K dimension in blocks
    # ---------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile (transposed): shape (BLOCK_K, BLOCK_N)
        # Row stride = stride_bk (k dimension), column stride = stride_bj (j dimension)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b)

    # ---------------------------------------------------------
    # Write back the result
    # ---------------------------------------------------------
    c_ptrs = c_base + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------
# Entry point used by the test harness
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing Triangle Multiplicative Update.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask_tensor : torch.Tensor [B, N, N] (may be None)
            - weights_dict: dict of model weight tensors
            - config_dict : dict with keys "dim" and "hidden_dim"
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]  # always 128 in the provided tests

    # ---------------------------------------------------------
    # 1) Input LayerNorm
    # ---------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ---------------------------------------------------------
    # 2) Linear projections (no bias)
    # ---------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------
    # 3) Optional mask (broadcasted over the hidden dimension)
    # ---------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_f = mask.unsqueeze(-1).to(dtype)    # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------------------------------------------------------
    # 4) Gating (sigmoid of linear projections)
    # ---------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 5) Core O(N³) reduction using Triton
    # ---------------------------------------------------------
    # Allocate output tensor for the intermediate representation
    out_inter = torch.empty_like(left)  # shape [B,N,N,hidden_dim]

    # Strides (in elements) for the input tensors
    stride_ab, stride_ai, stride_ak, stride_ad = left.stride()   # (B,N,N,hidden_dim)
    stride_bb, stride_bi, stride_bk, stride_bd = right.stride()
    stride_cb, stride_ci, stride_cj, stride_cd = out_inter.stride()

    # For the right‑hand side we need the K‑row stride and J‑column stride
    # (i.e. we treat right as transposed inside the kernel)
    stride_bj = stride_bi   # column (j) uses the original i‑stride
    # stride_bk already the K‑row stride (original k‑stride)

    # Block sizes – tuned for V100/A100 style GPUs (B200 works similarly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (batch*hidden_dim, ceil(N/BM), ceil(N/BN))
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch kernel
    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out_inter.data_ptr(),
        B, N, N, N,
        stride_ab, stride_ai, stride_ak, stride_ad,
        stride_bb, stride_bk, stride_bj, stride_bd,
        stride_cb, stride_ci, stride_cj, stride_cd,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # ---------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_inter,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out_norm = out_norm * out_gate
    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out
