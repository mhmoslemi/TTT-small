def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # unpack dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype  # assume float32

    # Get weights as torch tensors (already on device)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1. LayerNorm on input across dim
    # Use torch.nn.functional.layer_norm: input shape (B,N,N,dim), normalized shape (dim,)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias
    )

    # 2. Linear projections
    # Using torch.nn.functional.linear: input shape (..., dim) -> output (..., hidden_dim)
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape (B,N,N,hidden)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)  # same shape
    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Apply mask if present
    if mask is not None:
        # mask shape (B,N,N); unsqueeze last dim
        mask_f = mask.to(dtype).unsqueeze(-1)  # broadcast
        left = left * mask_f
        right = right * mask_f

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # At this stage left/right are (B,N,N,hidden)
    # We'll call Triton kernel to compute out = left @ right^T over k dimension.

    # Ensure left and right contiguous for efficient loads (they are contiguous)
    left = left.contiguous()
    right = right.contiguous()

    # Prepare output tensor
    out = torch.empty_like(left)  # (B,N,N,hidden) dtype = same as left

    # Launch kernel
    # Compute strides
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # If hidden_dim not a multiple of block, we can treat hidden dim as batch dimension - an index used for kernel; width dimension M,N are both N.
    # Use kernel grid:
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # define grid:
    batch = input_tensor.shape[0]
    N = input_tensor.shape[1] # seq_len
    num_gemm = batch * hidden_dim
    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             num_gemm)

    # call kernel
    tri_mul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=batch,
        N=N,
        H=hidden_dim,
        stride_l_batch=stride_l_batch,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_h=stride_l_h,
        stride_r_batch=stride_r_batch,
        stride_r_j=stride_r_j,
        stride_r_k=stride_r_k,
        stride_r_h=stride_r_h,
        stride_o_batch=stride_o_batch,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_h=stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_blocks_k=(N + BLOCK_K - 1) // BLOCK_K, # maybe not needed
    )
    # Note: we need to pass num_blocks_k? Actually we can compute inside kernel via N and BLOCK_K.

    # After kernel compute out.

    # 3. to_out_norm: layernorm over hidden dim
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # 4. apply out_gate (shape [B,N,N,hidden])
    out = out * out_gate

    # 5. final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)  # shape (B,N,N,dim)

    return out

The operator first applies layer‑norm and linear projections with gating.
The costly part is the 3‑tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication:
for each batch `b` and hidden channel `d`,
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T

The kernel below computes this batched GEMM using Triton, handling arbitrary
sequence lengths up to 1024 and hidden dimensions up to 384. The rest of the
forward pass (layer‑norms, linear layers, sigmoids) is performed with PyTorch
operations for simplicity and correctness.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # batch, sequence length, hidden dim
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,   # right strides
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: out = left @ right.T over the k dimension."""
    pid_i = tl.program_id(0)           # tile row (i)
    pid_j = tl.program_id(1)           # tile col (j)
    pid_g = tl.program_id(2)           # batch * hidden index

    # decode batch and hidden index
    batch_idx = pid_g // H
    hidden_idx = pid_g % H

    # start offsets for the tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # range vectors
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute batch and hidden offsets for each tensor
    off_l_batch = batch_idx * stride_l_batch
    off_r_batch = batch_idx * stride_r_batch
    off_o_batch = batch_idx * stride_o_batch

    off_l_hidden = hidden_idx * stride_l_h
    off_r_hidden = hidden_idx * stride_r_h
    off_o_hidden = hidden_idx * stride_o_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_blocks_k):
        k0 = k_iter * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (M x K) -----
        a_ptrs = (left_ptr
                  + off_l_batch
                  + offs_i[:, None] * stride_l_i
                  + offs_k[None, :] * stride_l_k
                  + off_l_hidden)
        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & mask_k[None, :]),
                    other=0.0,
                    dtype=tl.float32)

        # ----- load right tile (K x N) -----
        # we need right indexed as (k, j) = right[b, j, k, hidden]
        b_ptrs = (right_ptr
                  + off_r_batch
                  + offs_k[:, None] * stride_r_k
                  + offs_j[None, :] * stride_r_j
                  + off_r_hidden)
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_j[None, :]),
                    other=0.0,
                    dtype=tl.float32)

        # ----- multiply‑accumulate -----
        acc += tl.dot(a, b)

    # write result
    c_ptrs = (out_ptr
              + off_o_batch
              + offs_i[:, None] * stride_o_i
              + offs_j[None, :] * stride_o_j
              + off_o_hidden)
    tl.store(c_ptrs, acc,
             mask=(mask_i[:, None] & mask_j[None, :]))


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul (outgoing) module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    torch.set_grad_enabled(False)

    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gates
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B,N,N,H]

    # mask handling
    if mask is not None and mask.numel() > 0:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core O(N³) contraction using Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # [B,N,N,H]

    # strides for tensors
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # block sizes (tuned for B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # grid: (tiles_i, tiles_j, batch*hidden)
    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,
        (seq_len + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )

    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        bs,
        seq_len,
        hidden_dim,
        stride_l_batch,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_batch,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 4. Output normalization, gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
