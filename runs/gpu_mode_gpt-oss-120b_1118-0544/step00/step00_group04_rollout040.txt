def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # indices
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    offs_d = tl.arange(0, BLOCK_D)
    # masks
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointers for left and right tiles
        # left tile shape (BLOCK_M, BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr + pid_b * stride_l_b \
            + offs_i[:, None, None] * stride_l_i \
            + offs_k[None, :, None] * stride_l_k \
            + offs_d[None, None, :] * stride_l_d

        # right tile shape (BLOCK_N, BLOCK_K, BLOCK_D)
        right_ptrs = right_ptr + pid_b * stride_r_b \
            + offs_j[:, None, None] * stride_r_i \
            + offs_k[None, :, None] * stride_r_k \
            + offs_d[None, None, :] * stride_r_d

        # load
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0).to(tl.float32)
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0).to(tl.float32)

        # compute per d dot product
        # Loop over d
        for d in range(0, BLOCK_D):
            # left slice (M,K)
            left_slice = left[:, :, d]   # (M, K)
            right_slice = right[:, :, d] # (N, K)
            # dot left_slice @ right_slice.T  -> (M,N)
            acc[:, :, d] += tl.dot(left_slice, right_slice, trans_b=True)
    # store output: shape (M,N,D)
    out_ptrs = out_ptr + pid_b * stride_o_b \
        + offs_i[:, None, None] * stride_o_i \
        + offs_j[None, :, None] * stride_o_j \
        + offs_d[None, None, :] * stride_o_d

    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Get weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Normalize input
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Project
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    mul_mask = mask.unsqueeze(-1)  # shape [B,N,N,1]
    left = left * mul_mask
    right = right * mul_mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Allocate output tensor
    B, N, _, H = left_h.shape
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left_h.device)

    # Launch Triton kernel
    # choose block sizes based on N and H
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 32  # adjust if hidden_dim not multiple of BLOCK_D
    grid = (B, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    # Compute strides as ints
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left_h.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right_h.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    triton_tri_mul_kernel[grid](
        left_h, right_h, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_d,
        stride_r_b, stride_r_i, stride_r_k, stride_r_d,
        stride_o_b, stride_o_i, stride_o_j, stride_o_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D
    )
    # Now out is [B,N,N,H]

    # Apply final LayerNorm and linear
    # to_out_norm (LayerNorm over hidden_dim)
    out_norm = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)
    out_norm = out_norm * out_gate  # broadcast over hidden_dim
    output = torch.nn.functional.linear(out_norm, to_out_weight)   # shape [B,N,N,dim]

    return output

    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

"""
TriMul forward (outgoing) implementation with Triton acceleration.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (B, N, N, C).
2. Linear projections (left/right) and gating linear layers (left, right, out).
   Apply the optional mask and the sigmoid gates.
3. Core O(N³·H) update:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is implemented as a fused Triton kernel that
   - tiles the i‑ and j‑dimensions (BLOCK_M × BLOCK_N),
   - iterates over the reduction dimension k in blocks of size BLOCK_K,
   - processes the hidden dimension H in blocks of size BLOCK_D.
   Each (i,j,d) element is accumulated in FP32 while the inputs are read as
   FP16 to reduce memory bandwidth.
4. Layer‑norm the hidden‑dim output, multiply by the out‑gate,
   and project back to the original feature dimension C.

The kernel is deliberately simple and uses only element‑wise loads,
broadcasted masks and tl.dot (matrix‑multiply) for the inner product.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, H,                                     # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,  # left strides
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,  # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,  # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Kernel computing the einsum:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # indices within a tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # number of blocks in the reduction and hidden dimensions
    num_k = tl.cdiv(N, BLOCK_K)
    num_d = tl.cdiv(H, BLOCK_D)

    # -------------------------------------------------------------------------
    # Loop over hidden‑dim blocks
    # -------------------------------------------------------------------------
    for d_iter in range(num_d):
        d_off = d_iter * BLOCK_D
        offs_d = d_off + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H

        # accumulator for the current (i,j,d‑block)
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # -----------------------------------------------------------------
        # Reduction over k
        # -----------------------------------------------------------------
        for k_iter in range(num_k):
            k_off = k_iter * BLOCK_K
            offs_k = k_off + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # -------------------- load left tile -------------------------
            left_ptrs = (
                left_ptr
                + pid_b * stride_l_b
                + offs_i[:, None, None] * stride_l_i
                + offs_k[None, :, None] * stride_l_k
                + offs_d[None, None, :] * stride_l_d
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                other=0.0,
            ).to(tl.float32)   # (M, K, D)

            # -------------------- load right tile ------------------------
            right_ptrs = (
                right_ptr
                + pid_b * stride_r_b
                + offs_j[:, None, None] * stride_r_i
                + offs_k[None, :, None] * stride_r_k
                + offs_d[None, None, :] * stride_r_d
            )
            right = tl.load(
                right_ptrs,
                mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                other=0.0,
            ).to(tl.float32)   # (N, K, D)

            # -------------------- dot per hidden dim --------------------
            # Compile‑time loop over the BLOCK_D channels
            for d in range(0, BLOCK_D):
                # (M, K) · (K, N)ᵀ  -> (M, N)
                acc[:, :, d] += tl.dot(
                    left[:, :, d],        # (M, K)
                    right[:, :, d],       # (N, K)
                    trans_b=True
                )
        # -----------------------------------------------------------------
        # Store the completed (i,j,d‑block)
        # -----------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_o_b
            + offs_i[:, None, None] * stride_o_i
            + offs_j[None, :, None] * stride_o_j
            + offs_d[None, None, :] * stride_o_d
        )
        tl.store(
            out_ptrs,
            acc,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
        )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32 or bool). May be None.
            - weights: dict of model weight tensors (see function body)
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # Unpack weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # Step 1: LayerNorm over the feature dimension `dim`
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # Step 2: Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # Optional mask (broadcast over hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ext = mask.unsqueeze(-1).to(left.dtype)          # [B, N, N, 1]
        left = left * mask_ext
        right = right * mask_ext

    # -----------------------------------------------------------------
    # Step 3: Gating (sigmoid of linear maps)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Convert to half‑precision for the heavy Triton kernel
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # Triton launch configuration
    # -----------------------------------------------------------------
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 32

    grid = (B, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    # Compute strides (as Python int) – they match the layout of contiguous tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left_h.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right_h.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    _tri_mul_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_d,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_d,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # -----------------------------------------------------------------
    # Step 4: final LayerNorm + out‑gate + projection back to `dim`
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out_norm = out_norm * out_gate  # broadcast over hidden_dim
    output = F.linear(out_norm, to_out_weight)   # [B, N, N, dim]

    return output
```
