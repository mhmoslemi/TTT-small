@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute mask for M,N within bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offsets for k
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        # mask for k dimension
        mask_k = offs_k < K

        # Pointers
        a_ptrs = a_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_batch * stride_b_batch + offs_n[:, None] * stride_b_n + offs_k[None, :] * stride_b_k

        # Load tiles with masks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # Convert to fp16 then dot accumulation
        a = a.to(tl.float16)
        b = b.to(tl.float16)
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = c_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

Note: tl.dot expects inputs shape (BLOCK_M, BLOCK_K) and (BLOCK_K, BLOCK_N) -> returns (BLOCK_M, BLOCK_N). It operates on fp16 and accumulates in fp32.

import torch
import triton
import triton.language as tl

# Triton kernel for batched matrix multiplication (A @ B^T)
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask for out-of-bounds in M/N dimensions
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for A (M x K) and B (N x K) interpreted as B^T (K x N)
        a_ptrs = a_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_batch * stride_b_batch + offs_n[:, None] * stride_b_n + offs_k[None, :] * stride_b_k

        # Load with masks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Convert to FP16 for dot
        a = a.to(tl.float16)
        b = b.to(tl.float16)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton-accelerated TriMul forward pass (outgoing variant).

    This implementation:
    - Layer‑normalises the input.
    - Applies linear projections and multiplicative gates.
    - Uses a custom Triton kernel to compute the core tensor contraction:
          out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
      which is a batched matrix‑multiplication (left @ rightᵀ) for every
      hidden channel d.
    - Applies a second LayerNorm, an output gate and a final linear
      projection back to the original channel dimension.
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = config.get("eps", 1e-5)

    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # broadcast weight/bias
    norm_weight = norm_weight.view(1, 1, 1, -1)
    norm_bias = norm_bias.view(1, 1, 1, -1)

    # compute mean/var over last dimension
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias  # shape [B,N,N,dim]

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ---------- Mask ----------
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)                # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gates ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ---------- Convert to half and reshape for batched GEMM ----------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # shape    (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    batch_h = batch_size * hidden_dim
    left = left.view(batch_h, seq_len, seq_len)
    right = right.view(batch_h, seq_len, seq_len)

    # Output buffer
    out = torch.empty(batch_h, seq_len, seq_len, dtype=torch.float32, device=device)

    # Strides (all tensors are contiguous)
    stride_a_batch, stride_a_m, stride_a_k = left.stride()
    stride_b_batch, stride_b_n, stride_b_k = right.stride()
    stride_c_batch, stride_c_m, stride_c_n = out.stride()

    # Block sizes (tuned for the B200, keep them ≤ 64 for register pressure)
    BLOCK_M = min(64, seq_len)
    BLOCK_N = min(64, seq_len)
    BLOCK_K = min(64, seq_len)

    grid = (batch_h,
            (seq_len + BLOCK_M - 1) // BLOCK_M,
            (seq_len + BLOCK_N - 1) // BLOCK_N)

    batched_matmul_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        M=seq_len,
        N=seq_len,
        K=seq_len,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_n=stride_b_n,
        stride_b_k=stride_b_k,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Reshape back to [B,N,N,hidden] ----------
    out = out.view(batch_size, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # ---------- Second LayerNorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"].view(1, 1, 1, -1)
    to_out_norm_bias = weights["to_out_norm.bias"].view(1, 1, 1, -1)

    mean2 = out.mean(dim=-1, keepdim=True)
    var2 = ((out - mean2) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean2) / torch.sqrt(var2 + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # ---------- Output gate ----------
    out_norm = out_norm * out_gate

    # ---------- Final projection ----------
    to_out_weight = weights["to_out.weight"]
    final = torch.nn.functional.linear(out_norm, to_out_weight)  # [B,N,N,dim]

    return final

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑normalize the input tensor x ∈ ℝ^{B×N×N×C}.
2. Project x to a hidden dimension H (left/right) with two independent
   Linear layers (no bias).  Apply the optional mask.
3. Compute multiplicative gates (left/right/out) from x, apply sigmoid,
   and multiply left/right by their gates.
4. The core contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix multiplication left @ rightᵀ for each hidden
   channel h.  We reshape the tensors to (B·H, N, N) and execute a
   custom Triton kernel that performs the GEMM in FP16 while accumulating
   in FP32.
5. Reshape the result back to (B,N,N,H), apply a second LayerNorm
   over H, multiply by the out‑gate, and finally project back to C
   with a linear layer.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # batch * hidden index
    pid_m = tl.program_id(1)              # tile row
    pid_n = tl.program_id(2)              # tile column

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A : (M, K)   Bᵀ : (K, N)  →  C : (M, N)
        a_ptrs = a_ptr + pid_batch * stride_a_batch + \
                 offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_batch * stride_b_batch + \
                 offs_n[:, None] * stride_b_n + offs_k[None, :] * stride_b_k

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        acc += tl.dot(a, b)

    c_ptrs = c_ptr + pid_batch * stride_c_batch + \
             offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operation.
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = config.get("eps", 1e-5)

    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension

    # ---------- first LayerNorm ----------
    norm_w = weights["norm.weight"].view(1, 1, 1, -1)
    norm_b = weights["norm.bias"].view(1, 1, 1, -1)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b                     # shape [B,N,N,C]

    # ---------- linear projections ----------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- optional mask ----------
    if not config.get("nomask", False) and mask is not None:
        m = mask.to(dtype).unsqueeze(-1)    # [B,N,N,1]
        left = left * m
        right = right * m

    # ---------- gates ----------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- prepare for batched GEMM ----------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape to (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out = torch.empty(B * H, N, N, dtype=torch.float32, device=device)

    # strides for contiguous tensors
    stride_a_batch, stride_a_m, stride_a_k = left.stride()
    stride_b_batch, stride_b_n, stride_b_k = right.stride()
    stride_c_batch, stride_c_m, stride_c_n = out.stride()

    # block sizes (bounded by 64 for register pressure)
    BLOCK_M = min(64, N)
    BLOCK_N = min(64, N)
    BLOCK_K = min(64, N)

    grid = (
        B * H,                                 # batch × hidden
        (N + BLOCK_M - 1) // BLOCK_M,          # tiles in M dimension
        (N + BLOCK_N - 1) // BLOCK_N,          # tiles in N dimension
    )

    batched_matmul_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_n=stride_b_n,
        stride_b_k=stride_b_k,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- reshape back ----------
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ---------- second LayerNorm ----------
    to_out_norm_w = weights["to_out_norm.weight"].view(1, 1, 1, -1)
    to_out_norm_b = weights["to_out_norm.bias"].view(1, 1, 1, -1)

    mean2 = out.mean(dim=-1, keepdim=True)
    var2 = ((out - mean2) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean2) / torch.sqrt(var2 + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # ---------- output gate ----------
    out_norm = out_norm * out_gate

    # ---------- final projection ----------
    final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])   # [B,N,N,C]

    return final
```
