def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    pid_b = tl.program_id(1)
    pid_h = tl.program_id(2)

    # compute tile indices for i and j
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    i_tile = pid // num_j
    j_tile = pid % num_j

    # base indices
    i_base = i_tile * BLOCK_I
    j_base = j_tile * BLOCK_J

    # create range offsets
    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)
    # masks for in-range
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulate
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k blocks
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers to left block (i,k)
        left_ptrs = left_ptr + pid_b * stride_left_b + pid_h * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # pointers to right block transposed (k,j)
        right_ptrs = right_ptr + pid_b * stride_right_b + pid_h * stride_right_h + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot product over k dimension
        acc += tl.dot(left, right)  # left (BLOCK_I, BLOCK_K), right (BLOCK_K, BLOCK_J)

    # store results
    out_ptrs = out_ptr + pid_b * stride_out_b + pid_h * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward implementation with Triton acceleration.

The heavy O(N³·hidden_dim) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a custom Triton kernel that tiles the (i, j) output plane
and reduces over the shared k dimension using a blocked matrix‑multiply style
dot product. All other operations (LayerNorm, linear projections, gating,
final LayerNorm and projection) are done with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids: tile over (i,j) plane, then batch, then hidden
    pid_ij = tl.program_id(0)
    pid_b = tl.program_id(1)
    pid_h = tl.program_id(2)

    # number of tiles in each dimension
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)

    i_tile = pid_ij // num_j
    j_tile = pid_ij % num_j

    # base offsets for this tile
    i_base = i_tile * BLOCK_I
    j_base = j_tile * BLOCK_J

    # offsets within the tile
    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)

    # masks for out‑of‑bounds handling
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator for the partial sum over k
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k in BLOCK_K sized chunks
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------------------------------------------------
        # Load left block  [i, k]
        # -------------------------------------------------
        left_ptrs = (
            left_ptr
            + pid_b * stride_left_b
            + pid_h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # -------------------------------------------------
        # Load right block transposed  [k, j]  (i.e. right[j, k] -> (k, j))
        # -------------------------------------------------
        right_ptrs = (
            right_ptr
            + pid_b * stride_right_b
            + pid_h * stride_right_h
            + offs_k[:, None] * stride_right_k   # k dimension as rows
            + offs_j[None, :] * stride_right_i   # j dimension as columns
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # dot over the reduction dimension k
        acc += tl.dot(left, right)   # (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J) -> (BLOCK_I, BLOCK_J)

    # -------------------------------------------------
    # Store the result tile back to global memory
    # -------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_out_b
        + pid_h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated TriMul (outgoing) forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [bs, N, N, dim]
            - mask:        torch.Tensor [bs, N, N] (may be all‑ones)
            - weights: dict of model parameters
            - config:  dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [bs, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape

    # ----- LayerNorm ---------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [bs, N, N, dim]

    # ----- Linear projections -------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # [bs, N, N, hidden_dim]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ----- Apply mask (if any) -----------------------------------------------
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)               # [bs, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ----- Gating -------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [bs, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ----- Apply gates --------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ----- Rearrange to [B, hidden_dim, N, N] ---------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # ----- Allocate output tensor ---------------------------------------------
    out = torch.empty_like(left)  # [B, H, N, N]

    # ----- Strides -----------------------------------------------------------
    stride_left_b, stride_left_h, stride_left_i, stride_left_k = left.stride()
    stride_right_b, stride_right_h, stride_right_i, stride_right_k = right.stride()
    stride_out_b, stride_out_h, stride_out_i, stride_out_j = out.stride()

    # ----- Triton kernel launch -----------------------------------------------
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    num_i_tiles = (N + BLOCK_I - 1) // BLOCK_I
    num_j_tiles = (N + BLOCK_J - 1) // BLOCK_J
    grid = (num_i_tiles * num_j_tiles, bs, hidden_dim)

    _trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_h=stride_left_h,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_right_b=stride_right_b,
        stride_right_h=stride_right_h,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_out_b=stride_out_b,
        stride_out_h=stride_out_h,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # ----- Bring back to original layout --------------------------------------
    out = out.permute(0, 2, 3, 1).contiguous()   # [bs, N, N, hidden_dim]

    # ----- Final LayerNorm, gating and projection -------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])  # [bs, N, N, dim]

    return output
