def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Possibly config also has "nomask"? We'll get it from config.

    # Device
    device = input_tensor.device

    # Layernorm
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections (no bias)
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask handling
    if not config.get("nomask", True):
        # mask shape [B,N,N], unsqueeze last dim
        mask_expanded = mask.unsqueeze(-1).to(x.dtype)
        left_proj = left_proj * mask_expanded
        right_proj = right_proj * mask_expanded

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Prepare for triton reduction
    B, N, _, _ = left.shape  # shape [B,N,N,hidden]
    # Permute to [B, hidden, N, N] and reshape batch
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Combine batch and hidden dims
    batch = B * hidden_dim
    left_reshaped = left_perm.view(batch, N, N)
    right_reshaped = right_perm.view(batch, N, N)

    # Output placeholder
    out_reshaped = torch.empty_like(left_reshaped)

    # Launch Triton kernel
    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides
    stride_am = left_reshaped.stride(1)  # N
    stride_ak = left_reshaped.stride(2)  # 1
    batch_stride_a = left_reshaped.stride(0)  # N*N

    stride_bk = right_reshaped.stride(2)  # 1
    stride_bn = right_reshaped.stride(1)  # N
    batch_stride_b = right_reshaped.stride(0)  # N*N

    stride_cm = out_reshaped.stride(1)  # N
    stride_cn = out_reshaped.stride(2)  # 1
    batch_stride_c = out_reshaped.stride(0)  # N*N

    # compute grid
    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch
    batched_gemm_kernel[grid](
        left_reshaped,
        right_reshaped,
        out_reshaped,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        batch_stride_a=batch_stride_a,
        batch_stride_b=batch_stride_b,
        batch_stride_c=batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back
    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,hidden]

    # Final layernorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

import torch.nn.functional as F

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B^T for many independent batches.
    A shape: (batch, M, K)
    B shape: (batch, N, K)      (stored as (batch, N, K) but accessed as transposed)
    C shape: (batch, M, N)
    """
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Coordinates of the output tile.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds.
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch.
    a_base = a_ptr + batch * batch_stride_a
    b_base = b_ptr + batch * batch_stride_b
    c_base = c_ptr + batch * batch_stride_c

    # Pointers to the tiles that will be loaded in the K loop.
    a_ptrs = a_base + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_base + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

        # advance pointers by BLOCK_K in K dimension
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

def custom_kernel(data):
    """
    Triton‑accelerated implementation of the outgoing TriMul operator.
    The heavy O(N³) contraction is performed by a batched GEMM kernel that
    evaluates   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
    All other linear layers are executed with PyTorch functional ops.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (over the last dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating vectors (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction using Triton: out = Σ_k left[...,k,:] * right[...,k,:]
    #     This is implemented as many independent GEMMs of shape (N,N) per hidden channel.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape is [B,N,N,hidden]
    # reshape to (B, hidden, N, N) then fuse batch and hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch = B * hidden_dim
    left_flat = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)

    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # 7. Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=left_flat.stride(1),
        stride_ak=left_flat.stride(2),
        stride_bk=right_flat.stride(2),
        stride_bn=right_flat.stride(1),
        stride_cm=out_flat.stride(1),
        stride_cn=out_flat.stride(2),
        batch_stride_a=left_flat.stride(0),
        batch_stride_b=right_flat.stride(0),
        batch_stride_c=out_flat.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Final LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 9. Multiply by output gate and project back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,        # A: (M, K)
    stride_bk, stride_bn,       # B (transposed): (K, N)
    stride_cm, stride_cn,       # C: (M, N)
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute many independent matrix multiplications:
        C_batch = A_batch @ B_batch.T
    where
        A_batch : (M, K)
        B_batch : (N, K)   (stored as (N, K) but accessed as transposed)
        C_batch : (M, N)

    The tensors are laid out as
        a_ptr : (batch, M, K)
        b_ptr : (batch, N, K)
        c_ptr : (batch, M, N)
    """
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # -------------------------------------------------------------
    # Tile coordinates
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------------------------------------------------------------
    # Base pointers for the current batch
    # -------------------------------------------------------------
    a_base = a_ptr + batch * batch_stride_a
    b_base = b_ptr + batch * batch_stride_b
    c_base = c_ptr + batch * batch_stride_c

    # Pointers that will be stepped in the K‑loop
    a_ptrs = a_base + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_base + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)          # fp16 -> fp32 accumulation

        # advance to the next K‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # -------------------------------------------------------------
    # Write back the tile
    # -------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul.
    The heavy O(N³) contraction is performed by a batched GEMM kernel that
    evaluates   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
    All other linear layers are executed with ordinary PyTorch functional ops.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # -------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------------------
    # 2. Linear projections (bias=False in the reference model)
    # -------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    # -------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------------------
    # 4. Gating vectors (sigmoid)
    # -------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -------------------------------------------------------------
    # 5. Apply gates
    # -------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # 6. Core O(N³) contraction via Triton.
    #    Treat each hidden channel as an independent batch:
    #       C_d = L_d @ R_d^T   (L_d,R_d : N×N)
    # -------------------------------------------------------------
    B, N, _, _ = left.shape                     # left shape: [B, N, N, hidden]
    # Rearrange to [B, hidden, N, N] and fuse batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    total_batches = B * hidden_dim

    left_flat = left_perm.view(total_batches, N, N)
    right_flat = right_perm.view(total_batches, N, N)

    out_flat = torch.empty_like(left_flat)

    # -------------------------------------------------------------
    # 7. Triton launch configuration
    # -------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=left_flat.stride(1),   # = N
        stride_ak=left_flat.stride(2),   # = 1
        stride_bk=right_flat.stride(2),  # = 1 (k dimension)
        stride_bn=right_flat.stride(1),  # = N (n dimension)
        stride_cm=out_flat.stride(1),    # = N
        stride_cn=out_flat.stride(2),    # = 1
        batch_stride_a=left_flat.stride(0),
        batch_stride_b=right_flat.stride(0),
        batch_stride_c=out_flat.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------
    # 8. Final LayerNorm over the hidden dimension
    # -------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------------------
    # 9. Multiply by output gate and project back to original dim
    # -------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
