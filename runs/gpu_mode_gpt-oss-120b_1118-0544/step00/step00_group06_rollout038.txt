@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # mask of valid indices
    i_mask = i_offset < N
    j_mask = j_offset < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # Pointers for left and right
        left_ptrs = left_ptr + pid_b * stride_lb \
            + i_offset[:, None, None] * stride_li \
            + k_offset[None, :, None] * stride_lk \
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_lh

        right_ptrs = right_ptr + pid_b * stride_rb \
            + j_offset[:, None, None] * stride_rj \
            + k_offset[None, :, None] * stride_rk \
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_rh

        left_tile = tl.load(left_ptrs, mask = (i_mask[:, None, None] & k_mask[None, :, None]), other=0.0) # shape (BLOCK_I, BLOCK_K, BLOCK_H)
        right_tile = tl.load(right_ptrs, mask = (j_mask[:, None, None] & k_mask[None, :, None]), other=0.0) # shape (BLOCK_J, BLOCK_K, BLOCK_H)

        # Compute multiplication and sum over k
        # Expand dims for broadcasting
        left_tile_exp = left_tile[:, :, None, :]   # (I, K, 1, H)
        right_tile_exp = right_tile[None, :, :, :] # (1, K, J, H)

        mul = left_tile_exp * right_tile_exp  # (I, K, J, H)
        # sum over K axis
        acc += tl.sum(mul, axis=1)  # (I, J, H)
    # Store result
    out_ptrs = out_ptr + pid_b * stride_ob \
        + i_offset[:, None, None] * stride_oi \
        + j_offset[None, :, None] * stride_oj \
        + tl.arange(0, BLOCK_H)[None, None, :] * stride_oh

    tl.store(out_ptrs, acc, mask = (i_mask[:, None, None] & j_mask[None, :, None]))

def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # offsets
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    k = tl.arange(0, BLOCK_K)

    # mask for valid i/j
    i_mask = i < N
    j_mask = j < N

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K + k
        k_mask = k_offset < N

        # load left tile (I,K,H)
        left_ptrs = left_ptr + pid_b * stride_lb \
            + i[:, None, None] * stride_li \
            + k_offset[None, :, None] * stride_lk \
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_lh
        left_mask = i_mask[:, None, None] & k_mask[None, :, None]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0)

        # load right tile (K,J,H) (note k first)
        right_ptrs = right_ptr + pid_b * stride_rb \
            + k_offset[:, None, None] * stride_rk \
            + j[None, :, None] * stride_rj \
            + tl.arange(0, BLOCK_H)[None, None, :] * stride_rh
        right_mask = j_mask[None, :, None] & k_mask[:, None, None]   # shape (K,J,1) broadcast
        right = tl.load(right_ptrs, mask=right_mask, other=0.0)

        # For each hidden channel slice
        for h in range(0, BLOCK_H):
            # left_sub: (I,K)
            left_sub = left[:, :, h]
            # right_sub: (K,J)
            right_sub = right[:, :, h]
            # matrix multiply (I,J)
            acc[:, :, h] += tl.dot(left_sub, right_sub)
    # store output
    out_ptrs = out_ptr + pid_b * stride_ob \
        + i[:, None, None] * stride_oi \
        + j[None, :, None] * stride_oj \
        + tl.arange(0, BLOCK_H)[None, None, :] * stride_oh

    out_mask = i_mask[:, None, None] & j_mask[None, :, None]
    tl.store(out_ptrs, acc, mask=out_mask)

"""
TriMul "outgoing" kernel (AlphaFold3) using Triton.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension (dim).
2. Linear projections (left/right) from dim → hidden_dim (no bias).
3. Apply an optional mask (broadcast over hidden_dim).
4. Compute per‑channel gate values (sigmoid of linear projections) and
   multiply them with the projected tensors.
5. Core operation: for each batch, i, j and hidden channel h
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply per channel and is implemented
   in a Triton kernel that tiles the (i,j) output plane and the
   reduction axis k.  The hidden dimension is processed in small
   blocks (BLOCK_H) to keep the kernel compile‑time shapes static.
6. Layer‑norm the intermediate result (hidden_dim) and apply the
   output gate (sigmoid of another linear projection).
7. Final linear projection hidden_dim → dim.
All heavy arithmetic (step 5) runs on the GPU via Triton, while
the surrounding linear/normalisation steps are performed with
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr,          # *fp16  [B, I, K, H]
    right_ptr,         # *fp16  [B, J, K, H]
    out_ptr,           # *fp32  [B, I, J, H]
    B, N,              # int32
    stride_lb, stride_li, stride_lk, stride_lh,   # int64 left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # int64 right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # int64 out strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # ------------------------------------------------------------------
    #  Tile offsets
    # ------------------------------------------------------------------
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    k = tl.arange(0, BLOCK_K)
    h = tl.arange(0, BLOCK_H)

    i_valid = i < N
    j_valid = j < N

    # accumulator for a (BLOCK_I, BLOCK_J, BLOCK_H) tile
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Loop over k dimension in blocks
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k_off = kk * BLOCK_K + k
        k_valid = k_off < N

        # --------------------------------------------------------------
        # Load left tile  : shape (I, K, H)
        # --------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + pid_b * stride_lb
            + i[:, None, None] * stride_li
            + k_off[None, :, None] * stride_lk
            + h[None, None, :] * stride_lh
        )
        left_mask = i_valid[:, None, None] & k_valid[None, :, None]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0)

        # --------------------------------------------------------------
        # Load right tile : shape (K, J, H)   (K is first dim for dot)
        # --------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + pid_b * stride_rb
            + k_off[:, None, None] * stride_rk
            + j[None, :, None] * stride_rj
            + h[None, None, :] * stride_rh
        )
        right_mask = k_valid[:, None, None] & j_valid[None, :, None]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0)

        # --------------------------------------------------------------
        # Per‑channel matmul:  left[I,K] @ right[K,J]  -> (I,J)
        # --------------------------------------------------------------
        for hh in range(0, BLOCK_H):
            left_h = left_tile[:, :, hh]   # (I, K)
            right_h = right_tile[:, :, hh] # (K, J)
            acc[:, :, hh] += tl.dot(left_h, right_h)

    # ------------------------------------------------------------------
    #  Store the result
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_ob
        + i[:, None, None] * stride_oi
        + j[None, :, None] * stride_oj
        + h[None, None, :] * stride_oh
    )
    out_mask = i_valid[:, None, None] & j_valid[None, :, None]
    tl.store(out_ptrs, acc, mask=out_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool or float)
            - weights: dict of model weights (torch.Tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------- 1. LayerNorm -------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # ------------------- 2. Projections (float32) -------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------- 3. Optional mask -------------------------
    if not nomask:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------- 4. Gates (sigmoid) -----------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # keep float32 for later

    # Apply gates
    left = left.to(torch.float16) * left_gate
    right = right.to(torch.float16) * right_gate

    # Ensure tensors are contiguous for stride extraction
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape

    # ------------------- 5. Core TriMul via Triton -----------------
    # Output accumulator (float32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # Tuning constants
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 16  # hidden_dim is a multiple of 16 (128)

    # Grid over i‑tiles, j‑tiles, batch
    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        B,
    )

    # Process the hidden dimension in fixed blocks
    for h_start in range(0, hidden_dim, BLOCK_H):
        left_slice = left[..., h_start : h_start + BLOCK_H]
        right_slice = right[..., h_start : h_start + BLOCK_H]
        out_slice = out[..., h_start : h_start + BLOCK_H]

        # Extract strides (need int64 for Triton)
        l_strides = left_slice.stride()
        r_strides = right_slice.stride()
        o_strides = out_slice.stride()

        trimul_kernel[
            grid
        ](
            left_slice,
            right_slice,
            out_slice,
            B,
            N,
            l_strides[0],
            l_strides[1],
            l_strides[2],
            l_strides[3],
            r_strides[0],
            r_strides[1],
            r_strides[2],
            r_strides[3],
            o_strides[0],
            o_strides[1],
            o_strides[2],
            o_strides[3],
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )

    # ------------------- 6. Post‑processing ------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate  # channel‑wise gating (float32)
    output = F.linear(out, to_out_weight)  # final projection back to dim

    return output
