"""
TriMul (outgoing) forward implementation.

The heavy O(N³) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is realized as a batched matrix multiplication (per hidden dimension) using a
custom Triton kernel.  All other operations (LayerNorm, linear projections,
gating, masking) are performed with standard PyTorch ops.  The kernel processes
each (batch * hidden) slice independently and accumulates in FP32 for numerical
stability while the inputs/outputs are stored in FP16 to reduce memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the triple reduction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # const float16 *
    right_ptr,              # const float16 *
    out_ptr,                # float16 *
    N,                      # sequence length (int)
    stride_left_c, stride_left_i, stride_left_k,   # strides for left: (C, i, k)
    stride_right_c, stride_right_i, stride_right_k, # strides for right: (C, i, k) – we will use transposed view
    stride_out_c, stride_out_i, stride_out_j,      # strides for out: (C, i, j)
    BLOCK_M: tl.constexpr,  # tile size for i dimension
    BLOCK_N: tl.constexpr,  # tile size for j dimension
    BLOCK_K: tl.constexpr,  # tile size for k (reduction) dimension
):
    pid_c = tl.program_id(2)          # index over (batch * hidden_dim)
    pid_m = tl.program_id(0)          # i‑tile
    pid_n = tl.program_id(1)          # j‑tile

    # i / j coordinates for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left block (i, k) -----
        a_ptrs = left_ptr \
            + pid_c * stride_left_c \
            + offs_m[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)

        # ----- load right block transposed: we need (k, j) where right_T[k,j] = right[j,k]
        #   original right has strides (C, i=row=j, k=col=k)
        #   transposed view swaps the i/k strides
        b_ptrs = right_ptr \
            + pid_c * stride_right_c \
            + offs_k[:, None] * stride_right_k  \
            + offs_n[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)

        # dot product over the reduction axis
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) in FP32

    # write back, cast to fp16
    out_ptrs = out_ptr \
        + pid_c * stride_out_c \
        + offs_m[:, None] * stride_out_i \
        + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=(mask_m[:, None] & mask_n[None, :]))


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask         : torch.Tensor[B, N, N] (float32 or bool)
        - weights      : dict of torch.Tensor containing all linear / layernorm weights
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor[B, N, N, dim] (float32)
    """
    torch.cuda.set_device(0)  # ensure we are on GPU

    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # shortcuts for weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    # mean and var across dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias   # broadcast over [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape after linear: [B, N, N, hidden_dim]
    left = F.linear(x_norm, left_proj_weight)      # (out_features=hidden_dim, in_features=dim)
    right = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] -> [B,N,N,1] and broadcast
        mask = mask.to(x_norm.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for Triton kernel
    # ------------------------------------------------------------------
    # Move hidden dimension forward and make it contiguous
    # left_permuted : [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B = left_perm.shape[0]
    N = left_perm.shape[2]          # sequence length
    C_total = B * hidden_dim        # combined batch‑hidden dimension

    # reshape to (C_total, N, N)
    left_flat = left_perm.view(C_total, N, N).contiguous()
    right_flat = right_perm.view(C_total, N, N).contiguous()

    # cast to FP16 for the kernel (still keep a FP32 copy for possible debugging)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # output buffer
    out_fp16 = torch.empty_like(left_fp16)

    # ------------------------------------------------------------------
    # Compute strides (in element counts, not bytes)
    # ------------------------------------------------------------------
    stride_left_c, stride_left_i, stride_left_k = left_fp16.stride()
    stride_right_c, stride_right_i, stride_right_k = right_fp16.stride()
    stride_out_c, stride_out_i, stride_out_j = out_fp16.stride()

    # ------------------------------------------------------------------
    # Triton kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles over j
        C_total                         # batch * hidden
    )

    # launch
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N,
        stride_left_c, stride_left_i, stride_left_k,
        stride_right_c, stride_right_i, stride_right_k,
        stride_out_c, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # Back to original layout and dtype
    # ------------------------------------------------------------------
    # out_fp16 shape: (C_total, N, N)
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # subsequent ops are in FP32

    # ------------------------------------------------------------------
    # Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_out) * torch.rsqrt(var_out + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # apply output gate
    out_norm = out_norm * out_gate

    # final linear projection back to dim
    out_final = F.linear(out_norm, to_out_weight)  # (out_features=dim)

    return out_final
```
