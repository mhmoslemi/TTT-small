@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_d,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    # compute batch index and hidden dim index
    b = pid0 // H
    d = pid0 % H

    # i and j block offsets
    i_offset = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for i/j boundary
    mask_i = i_offset < N
    mask_j = j_offset < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # pointers for left block: shape (M,K)
        a_ptrs = left_ptr + b * stride_l_batch + i_offset[:, None] * stride_l_i + k_offset[None, :] * stride_l_k + d * stride_l_d
        # load left tile
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # pointers for right transposed block: shape (K,N)
        b_ptrs = right_ptr + b * stride_r_batch + k_offset[:, None] * stride_r_i + j_offset[None, :] * stride_r_k + d * stride_r_d
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b)
    
    # write back
    c_ptrs = out_ptr + b * stride_o_batch + i_offset[:, None] * stride_o_i + j_offset[None, :] * stride_o_j + d * stride_o_d
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

TriMul "outgoing" forward implementation.

The heavy O(N³) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a custom Triton kernel.  The rest of the
TriMul pipeline (LayerNorms, Linear projections, gating and the final
linear projection) is done in PyTorch.  The kernel treats each
(batch, hidden) pair as an independent GEMM (left @ rightᵀ) and
processes the output in tiles of size BLOCK_M×BLOCK_N with a
BLOCK_K reduction loop.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel that computes the O(N³) reduction for a single (b,d) pair.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,               # tensor pointers
    B, N, H,                                    # runtime sizes
    stride_l_batch, stride_l_i, stride_l_j, stride_l_d,
    stride_r_batch, stride_r_i, stride_r_j, stride_r_d,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    Input tensors are 4‑D [B, N, N, H] (H == hidden_dim).
    The kernel processes a tile (BLOCK_M, BLOCK_N) of the (i,j) plane
    for a specific (b, d) combination.
    """
    pid0 = tl.program_id(0)   # encodes (b, d)
    pid1 = tl.program_id(1)   # i‑tile index
    pid2 = tl.program_id(2)   # j‑tile index

    # Decode batch and hidden‑dim index.
    b = pid0 // H
    d = pid0 %  H

    # Offsets for the i and j tiles.
    i_offset = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # Accumulator in FP32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k.
    # --------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ------- Load a tile of left: shape (BLOCK_M, BLOCK_K) -------
        # left[b, i_offset, k_offset, d]
        a_ptrs = left_ptr \
            + b * stride_l_batch \
            + d * stride_l_d \
            + i_offset[:, None] * stride_l_i \
            + k_offset[None, :] * stride_l_j
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # ------- Load a tile of right transposed (k, j) -------
        # right_T[k_offset, j_offset] == right[b, j_offset, k_offset, d]
        b_ptrs = right_ptr \
            + b * stride_r_batch \
            + d * stride_r_d \
            + k_offset[:, None] * stride_r_i \
            + j_offset[None, :] * stride_r_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # ------- Accumulate the outer product -------
        acc += tl.dot(a, b)   # (M,K)·(K,N) -> (M,N)

    # --------------------------------------------------------------
    # Store the result.
    # --------------------------------------------------------------
    c_ptrs = out_ptr \
        + b * stride_o_batch \
        + d * stride_o_d \
        + i_offset[:, None] * stride_o_i \
        + j_offset[None, :] * stride_o_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (normalize over `dim`)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if not config.get("nomask", True) and mask is not None:
        # mask shape: [B, N, N] -> broadcast over hidden_dim
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy O(N³) reduction using Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # Tile sizes – chosen to fit typical B200 resources.
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # Grid: (batch*hidden, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Strides for pointer arithmetic
    stride_l_batch, stride_l_i, stride_l_j, stride_l_d = left.stride()
    stride_r_batch, stride_r_i, stride_r_j, stride_r_d = right.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d = out.stride()

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_batch,
        stride_l_i,
        stride_l_j,
        stride_l_d,
        stride_r_batch,
        stride_r_i,
        stride_r_j,
        stride_r_d,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7) Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_gated = out_norm * out_gate
    output = F.linear(out_gated, to_out_weight)   # [B, N, N, dim]

    return output
