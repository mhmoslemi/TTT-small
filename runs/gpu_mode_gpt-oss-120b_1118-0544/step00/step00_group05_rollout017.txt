"""
TriMul (outgoing) implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input `x` (dim).
2. Linear projections → `left` & `right` (hidden_dim).
3. Optional mask multiplication (broadcast over hidden dim).
4. Gating: compute sigmoid(linear(x)) for left, right and output.
   Apply left/right gates.
5. Core operation: for each batch and hidden channel compute
   out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h].
   This is a batched matrix‑multiply where the second operand is
   transposed on the `k` dimension.  Implemented with a Triton GEMM
   kernel (block‑wise, FP16 input, FP32 accumulation).
6. Layer‑norm over `hidden_dim`, multiply by output gate.
7. Final linear projection back to `dim`.

The Triton kernel handles the expensive O(N³·hidden_dim) step, while
the remaining cheap operations are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the einsum: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def triton_trimul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    B, N, H,                                 # dimensions
    stride_ah, stride_ai, stride_ak, stride_ad,   # left strides
    stride_bh, stride_bi, stride_bk, stride_bd,   # right strides
    stride_ch, stride_ci, stride_cj, stride_cd,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids: (i‑block, j‑block, bh-index)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # decode batch and hidden‑channel from pid_bh
    b_idx = pid_bh // H               # batch index
    h_idx = pid_bh % H                # hidden channel index

    # offsets for the i and j tiles
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load a tile:  left[b, i, k, h]  (M × K) -----
        a_ptrs = (
            a_ptr
            + b_idx * stride_ah
            + h_idx * stride_ad
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=(mask_i[:, None] & mask_k[None, :]),
            other=0.0,
        ).to(tl.float32)

        # ----- load b tile:  right[b, j, k, h]  (N × K) -----
        b_ptrs = (
            b_ptr
            + b_idx * stride_bh
            + h_idx * stride_bd
            + offs_j[:, None] * stride_bi
            + offs_k[None, :] * stride_bk
        )
        b = tl.load(
            b_ptrs,
            mask=(mask_j[:, None] & mask_k[None, :]),
            other=0.0,
        ).to(tl.float32)

        # accumulate M×N block
        acc += tl.dot(a, b)   # dot over K, result (M,N)

    # ----- write back the result -----
    c_ptrs = (
        c_ptr
        + b_idx * stride_ch
        + h_idx * stride_cd
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=(mask_i[:, None] & mask_j[None, :]))


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (float32) or None
            - weights_dict: mapping from weight names to tensors
            - config_dict: contains "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), norm_weight, norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional mask (mask shape [B,N,N])
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 and make contiguous for the Triton kernel
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape
    # output tensor (FP16) to be filled by the kernel
    out_fp16 = torch.empty_like(left_fp16)

    # ------------------------------------------------------------------
    # Strides (in elements) for Triton
    # ------------------------------------------------------------------
    a_stride = left_fp16.stride()   # (B, i, k, h)
    b_stride = right_fp16.stride()
    c_stride = out_fp16.stride()

    # Launch parameters
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    triton_trimul_kernel[
        grid
    ](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        a_stride[0],
        a_stride[1],
        a_stride[2],
        a_stride[3],
        b_stride[0],
        b_stride[1],
        b_stride[2],
        b_stride[3],
        c_stride[0],
        c_stride[1],
        c_stride[2],
        c_stride[3],
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # Continue the pipeline in FP32
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # Output layer norm (over hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), to_out_norm_weight, to_out_norm_bias, eps=eps)

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to dim
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # shape [B, N, N, dim]

    return out
