"""
TriMul outgoing implementation with Triton acceleration.

The heavy O(N³) reduction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is executed as a batched matrix multiplication per hidden channel:
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T
using a custom Triton kernel (half‑precision inputs, fp32 accumulation).
All other operations (LayerNorm, Linear projections, gating) are performed
with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                     # matrix sizes (all equal to seq_len)
    stride_am, stride_ak,         # stride for A (left): row i, col k
    stride_bk, stride_bn,         # stride for B (right) transposed: row k, col j
    stride_cm, stride_cn,         # stride for C (output): row i, col j
    stride_a_batch, stride_a_d,   # batch and hidden‑dim strides for A
    stride_b_batch, stride_b_d,   # batch and hidden‑dim strides for B
    stride_c_batch, stride_c_d,   # batch and hidden‑dim strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    hidden_dim: tl.int32,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_z = tl.program_id(2)   # combined batch * hidden index

    # decode batch and hidden channel from pid_z
    batch = pid_z // hidden_dim
    d = pid_z % hidden_dim

    # base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch * stride_a_batch + d * stride_a_d
    b_base = b_ptr + batch * stride_b_batch + d * stride_b_d
    c_base = c_ptr + batch * stride_c_batch + d * stride_c_d

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[i,k]
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B[k,j] (right matrix transposed on‑the‑fly)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # accumulate dot product over the K tile
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # write result
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass for the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input, mask, weights, config)
            input  – tensor [B, N, N, dim] (float32)
            mask   – tensor [B, N, N] (bool or float) or None
            weights – dict with LayerNorm, Linear, and gating parameters
            config – dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----- LayerNorm on the last dimension -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ----- Linear projections -----
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Optional mask -----
    if not nomask:
        # mask broadcasted over hidden channels
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----- Gating -----
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- Cast to half for the kernel (saving bandwidth) -----
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape

    # ----- Allocate output tensor (float32) -----
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # ----- Strides for Triton kernel -----
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d = left_h.stride()
    stride_b_batch, stride_b_i, stride_b_k, stride_b_d = right_h.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_d = out.stride()

    # A is left (i rows, k columns)
    stride_am = stride_a_i          # row stride (i)
    stride_ak = stride_a_k          # col stride (k)

    # B is right transposed: we need row stride = k, col stride = j (original i)
    stride_bk = stride_b_k          # row stride (k)
    stride_bn = stride_b_i          # col stride (j)

    stride_cm = stride_c_i          # row stride for output (i)
    stride_cn = stride_c_j          # col stride for output (j)

    # ----- Tile sizes -----
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    # Launch Triton kernel
    tri_mul_outgoing_kernel[grid](
        left_h,
        right_h,
        out,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_a_batch=stride_a_batch,
        stride_a_d=stride_a_d,
        stride_b_batch=stride_b_batch,
        stride_b_d=stride_b_d,
        stride_c_batch=stride_c_batch,
        stride_c_d=stride_c_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        hidden_dim=hidden_dim,
        num_warps=4,
    )

    # ----- Post‑processing: LayerNorm, gating, final linear -----
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_norm = out_norm * out_gate

    final = F.linear(out_norm, weights["to_out.weight"])
    return final
