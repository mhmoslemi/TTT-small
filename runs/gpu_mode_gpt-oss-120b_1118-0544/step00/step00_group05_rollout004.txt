@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden,
    stride_bl, stride_il, stride_kl, stride_dl,
    stride_br, stride_ir, stride_kr, stride_dr,
    stride_bo, stride_io, stride_ko, stride_do,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
   # each program processes a tile (m,n) (i block rows and j block cols) for a particular batch*hidden index
   pid_z = tl.program_id(2)  # combined batch*hidden
   batch_idx = pid_z // hidden
   d_idx = pid_z % hidden

   # compute tile offsets
   pid_m = tl.program_id(0)  # block row i
   pid_n = tl.program_id(1)  # block col j

   # start offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

   # mask for out-of-bounds
   mask_m = offs_m < N
   mask_n = offs_n < N

   # loading left tiles
   # We need to iterate over k dimension in BLOCK_K sized chunks.
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for k in range(0, tl.cdiv(N, BLOCK_K)):
       # compute k offset
       k_base = k * BLOCK_K
       offs_k = k_base + tl.arange(0, BLOCK_K)
       mask_k = offs_k < N

       # compute pointer offsets for left (left[b, i, k, d])
       left_ptrs = left_ptr + batch_idx * stride_bl + d_idx * stride_dl + offs_m[:, None] * stride_il + offs_k[None, :] * stride_kl
       # load left with masking
       left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

       # compute pointer offsets for right (right[b, j, k, d])
       # Here we need right indexed by j row (block col) and same k
       right_ptrs = right_ptr + batch_idx * stride_br + d_idx * stride_dr + offs_n[:, None] * stride_ir + offs_k[None, :] * stride_kr
       # load
       right = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

       # Multiply and sum over k dimension: acc += left * right
       # shape (BLOCK_M, BLOCK_N) after dot
       # Since left and right share k dimension on axis 1 (BLOCK_K), we can compute tl.dot
       # However tl.dot expects inner dimension to align: (M, K) and (N, K). Actually dot expects shape (M, K) and (N, K) then compute sum_k a[m,k] * b[n,k] -> (M,N).
       # That's exactly what we need.
       acc += tl.dot(left, right)

   # store result to out
   out_ptrs = out_ptr + batch_idx * stride_bo + d_idx * stride_do + offs_m[:, None] * stride_io + offs_n[None, :] * stride_ko
   tl.store(out_ptrs, acc.to(tl.float16) if tl.dtype(out_ptr) == tl.float16 else acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    # dims
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # layernorm on x
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]  # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape (B,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask
        right = right * mask

    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

    # allocate out tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=device)

    # compute strides
    stride_bl, stride_il, stride_kl, stride_dl = left.stride()
    stride_br, stride_ir, stride_kr, stride_dr = right.stride()
    stride_bo, stride_io, stride_ko, stride_do = out.stride()

    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32  # k tile size
    # compile kernel
    grid = ( (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N, B*hidden_dim)
    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_bl, stride_il, stride_kl, stride_dl,
        stride_br, stride_ir, stride_kr, stride_dr,
        stride_bo, stride_io, stride_ko, stride_do,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Now apply to_out_norm (layernorm on hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # shape (B,N,N,dim)

    return out

"""
TriMul (outgoing) forward kernel.

The heavy part of the TriMul operation is the 3‑way tensor contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
where `left` and `right` are projected and gated versions of the input.
For each batch and hidden channel `d` this is exactly a dense matrix
multiplication of size (N×N).  The Triton kernel below performs a fused
batched GEMM over the combined batch*hidden dimension, using a
standard tiled algorithm (BLOCK_M × BLOCK_N output tiles, accumulating
over K in steps of BLOCK_K).  The rest of the TriMul pipeline
(LayerNorm, linear projections, gating, and final linear) is executed
with PyTorch for simplicity.

The kernel returns a float32 tensor, which is then layer‑normed,
gated, and projected back to the original feature dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the einsum part
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,            # pointers
    B, N, H,                                 # batch, seq_len, hidden_dim
    stride_bl, stride_il, stride_kl, stride_dl,   # left strides
    stride_br, stride_ir, stride_kr, stride_dr,   # right strides
    stride_bo, stride_io, stride_ko, stride_do,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    pid_z = tl.program_id(2)          # combined batch*hidden index
    pid_m = tl.program_id(0)          # tile row (i)
    pid_n = tl.program_id(1)          # tile col (j)

    # Decode batch and hidden channel from pid_z
    batch_idx = pid_z // H
    hidden_idx = pid_z % H

    # Offsets for the i/j dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for boundary handling
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers to left[b,i,k,h] and right[b,j,k,h]
        left_ptrs = (left_ptr +
                     batch_idx * stride_bl +
                     hidden_idx * stride_dl +
                     offs_m[:, None] * stride_il +
                     offs_k[None, :] * stride_kl)
        right_ptrs = (right_ptr +
                      batch_idx * stride_br +
                      hidden_idx * stride_dr +
                      offs_n[:, None] * stride_ir +
                      offs_k[None, :] * stride_kr)

        # Load with masking, promote to FP32
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
        right = tl.load(right_ptrs,
                        mask=mask_n[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate dot product over K
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_K)·(BLOCK_N, BLOCK_K)^T -> (BLOCK_M, BLOCK_N)

    # Store the result
    out_ptrs = (out_ptr +
                batch_idx * stride_bo +
                hidden_idx * stride_do +
                offs_m[:, None] * stride_io +
                offs_n[None, :] * stride_ko)

    mask_out = mask_m[:, None] & mask_n[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


# ----------------------------------------------------------------------
# Custom Triton‑based TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (dim -> hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask handling
    # ------------------------------------------------------------------
    if mask_tensor is None or config.get("nomask", True):
        mask = torch.ones((B, N, N), dtype=x.dtype, device=device)
    else:
        mask = mask_tensor.to(x.dtype)
    mask = mask.unsqueeze(-1)   # [B,N,N,1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid) and gating application
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # Triton kernel: compute the einsum contraction
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides
    stride_bl, stride_il, stride_kl, stride_dl = left.stride()
    stride_br, stride_ir, stride_kr, stride_dr = right.stride()
    stride_bo, stride_io, stride_ko, stride_do = out.stride()

    # Tiling parameters (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * hidden_dim                  # batch * hidden
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_bl,
        stride_il,
        stride_kl,
        stride_dl,
        stride_br,
        stride_ir,
        stride_kr,
        stride_dr,
        stride_bo,
        stride_io,
        stride_ko,
        stride_do,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm, gating, final Linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                         # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out
