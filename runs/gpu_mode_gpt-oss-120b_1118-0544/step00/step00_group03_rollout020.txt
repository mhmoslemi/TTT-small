"""
TriMul outgoing update (AlphaFold3) with a fused batched GEMM Triton kernel.

The forward pass performs:
1) LayerNorm on the 4‑D input.
2) Linear projections (`left`, `right`) and gated projections.
3) Core O(N³) operation:
   out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
   which is a batch of H (=hidden_dim) independent matrix multiplications:
   C = A @ Bᵀ   with  A∈(M=N,K=N), B∈(N=N,K=N) per hidden channel.
   The kernel fuses all batch*hidden channels.
4) LayerNorm + output gate + final linear projection.

Only the heavy matrix multiplication (step 3) is implemented in Triton;
all other ops use PyTorch for simplicity and correctness.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_abatch,
    stride_bk, stride_bn, stride_bbatch,
    stride_cm, stride_cn, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B  where A is (BH, M, K) and B is (BH, K, N)."""
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block column
    pid_z = tl.program_id(2)  # batch * hidden index

    # ----------------------------------------------------------------------
    # Compute offsets for the current tile
    # ----------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the start of the tile (k = 0)
    a_ptr_base = a_ptr + pid_z * stride_abatch + offs_m[:, None] * stride_am
    b_ptr_base = b_ptr + pid_z * stride_bbatch + offs_n[None, :] * stride_bn

    # ----------------------------------------------------------------------
    # Accumulate over K with a for‑loop
    # ----------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M x K) and B tile (K x N)
        a_ptrs = a_ptr_base + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr_base + offs_k[:, None] * stride_bk

        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # ----------------------------------------------------------------------
    # Write the tile back to C
    # ----------------------------------------------------------------------
    c_ptrs = c_ptr + pid_z * stride_cbatch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Args:
        data: Tuple of (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (float32 or bool)
            - weights_dict: mapping with all required weight & bias tensors
            - config_dict: must contain "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ----------------------------------------------------------------------
    # Unpack arguments
    # ----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------------------
    # Weight shortcuts
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    eps = 1e-5

    # ----------------------------------------------------------------------
    # 1) LayerNorm on input
    # ----------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ----------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ----------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)    # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)  # (B,N,N,hidden_dim)

    # ----------------------------------------------------------------------
    # 3) Apply optional mask before gating
    # ----------------------------------------------------------------------
    if not nomask:
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)          # (B,N,N,1)
        left = left * mask_unsq
        right = right * mask_unsq

    # ----------------------------------------------------------------------
    # 4) Gating
    # ----------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5) Core TriMul (batched GEMM) – Triton kernel
    # ----------------------------------------------------------------------
    # Convert to half precision for the kernel (still accurate enough)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Permute to shape (B, H, N, N) where:
    #   left : (i, k) → (M=N, K=N)
    #   right: (j, k) → we need (K=N, N=j) → transpose last two dims
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # (B, H, M, K)
    right_perm = right_h.permute(0, 3, 2, 1).contiguous() # (B, H, K, N)

    # Collapse batch + hidden dimension
    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, N, N).contiguous()    # (BH, M, K)
    right_flat = right_perm.view(BH, N, N).contiguous()  # (BH, K, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)  # (BH, M, N)

    # Strides (in elements)
    stride_am = left_flat.stride(1)        # stride for M
    stride_ak = left_flat.stride(2)        # stride for K
    stride_abatch = left_flat.stride(0)    # stride for batch*hidden

    stride_bk = right_flat.stride(1)       # stride for K
    stride_bn = right_flat.stride(2)       # stride for N
    stride_bbatch = right_flat.stride(0)   # stride for batch*hidden

    stride_cm = out_flat.stride(1)         # stride for M
    stride_cn = out_flat.stride(2)         # stride for N
    stride_cbatch = out_flat.stride(0)     # stride for batch*hidden

    # Tunable block sizes – work well for the provided dimensions
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        BH,
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_am, stride_ak, stride_abatch,
        stride_bk, stride_bn, stride_bbatch,
        stride_cm, stride_cn, stride_cbatch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Convert back to fp32 for the remaining layers
    out = out_flat.to(torch.float32)

    # Restore original 4‑D layout: (B, N, N, hidden_dim)
    out = out.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------------
    # 6) Output LayerNorm, gating and projection
    # ----------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # (B,N,N,dim)

    return out
