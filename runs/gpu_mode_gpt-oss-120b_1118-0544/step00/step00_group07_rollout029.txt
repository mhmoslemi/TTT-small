    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_bnhc, stride_left_nhc, stride_left_hc, stride_left_c,
    stride_right_bnhc, stride_right_nhc, stride_right_hc, stride_right_c,
    stride_out_bnhc, stride_out_nhc, stride_out_hc, stride_out_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Determine indices
    bh = tl.program_id(0)   # flattened b*H + h
    b = bh // H
    h = bh % H

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    j_mask = j_offset < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Compute batch and hidden dims from flattened program id.
    ph = tl.program_id(0)  # 0: flattened B*H
    b = ph // H
    h = ph % H

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Offsets for i and j
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets arrays
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # Mask out-of-range
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Compute pointers for left: shape (i,k)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k + \
            h * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Compute pointers for right: shape (j,k)
        # we need right transposed to allow dot with left. We'll load right as (j,k) and use trans_b=True.
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            offs_j[:, None] * stride_right_i + \
            offs_k[None, :] * stride_right_k + \
            h * stride_right_h
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # dot with transpose of right (shape (i,k) @ (j,k)^T => (i,j))
        # tl.dot(A, B, trans_b=True) does A @ B^T
        acc += tl.dot(left, right, trans_b=True)

    # Store result
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j + \
        h * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

def custom_kernel(data):
    """TriMul outgoing update: compute gated pairwise interaction via Triton optimized
    matrix multiplication over the sequence dimension.

    The kernel performs the core einsum:
        out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    where left/right are linear projections of the input with masks and gates.
    All other operations (LayerNorm, Linear projections, sigmoid gates) are done with
    PyTorch for simplicity.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Extract config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------
    # LayerNorm over last dimension (dim)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # -------------------------------------------------
    # Linear projections (no bias)
    # -------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -------------------------------------------------
    # Mask (optional)
    # -------------------------------------------------
    if nomask or mask_tensor is None:
        mask = torch.ones_like(mask_tensor, dtype=dtype, device=device) if mask_tensor is not None else torch.ones(x.shape[:3], dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(dtype)
    # broadcast mask to hidden dim
    mask = mask.unsqueeze(-1)   # [B,N,N,1]
    left = left * mask
    right = right * mask

    # -------------------------------------------------
    # Gates: sigmoid(linear(x))
    # -------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # Core TriMul via Triton
    # -------------------------------------------------
    B, N, _, _ = left.shape
    # Ensure tensors are contiguous
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)   # shape [B,N,N,hidden_dim]

    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Define block sizes (tune if needed)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # launch kernel
    grid = (B * hidden_dim, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # -------------------------------------------------
    # Out layernorm, gating, final linear
    # -------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # shape [B,N,N,dim]

    return out

def tri_mul_kernel(

    B, N, H,

    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    out = F.linear(out, to_out_weight)   # [B,N,N,dim]

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy O(N³) part
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is realized with a Triton kernel that performs a batched matrix
multiplication for every hidden channel *h*.  All surrounding LayerNorms,
linear projections and sigmoid gates are carried out with PyTorch.
The kernel tiles the (i,j) output plane (BLOCK_M × BLOCK_N) and reduces
over k in chunks (BLOCK_K), using an inner `tl.dot` for fast accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    # (b, h) is flattened into the first program‑id dimension
    bh = tl.program_id(0)
    b = bh // H
    h = bh % H

    pid_i = tl.program_id(1)  # block index over i
    pid_j = tl.program_id(2)  # block index over j

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, h]
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k + \
            h * stride_left_h
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right[b, j, k, h]
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            offs_j[:, None] * stride_right_i + \
            offs_k[None, :] * stride_right_k + \
            h * stride_right_h
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )

        # acc += left @ rightᵀ  (tl.dot supports trans_b flag)
        acc += tl.dot(left, right, trans_b=True)

    # write back
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j + \
        h * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights, config)
            - input_tensor: [B, N, N, dim] torch.Tensor
            - mask_tensor:  [B, N, N]   torch.Tensor (optional, may be None)
            - weights: dict of all linear / norm parameters (no bias on Lin)
            - config: dictionary containing at least "dim", "hidden_dim"
              and optionally "nomask" (bool).

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (over last dim = dim)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)    # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3) Optional mask
    # -----------------------------------------------------------------
    if nomask or mask_tensor is None:
        mask = torch.ones(x.shape[:3], dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(dtype)
    mask = mask.unsqueeze(-1)               # [B,N,N,1] for broadcasting
    left = left * mask
    right = right * mask

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid of linear projections of the normalized input)
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core TriMul operation – Triton
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)   # shape [B,N,N,hidden_dim]

    # strides in elements (torch returns strides in elements, not bytes)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Tiling parameters – chosen to fit typical B200 resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                         # flattened (batch, hidden)
        (N + BLOCK_M - 1) // BLOCK_M,           # i blocks
        (N + BLOCK_N - 1) // BLOCK_N,           # j blocks
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)      # [B,N,N,dim]

    return out
