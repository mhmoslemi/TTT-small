    j_start = pid_n * BLOCK_N

"""
TriMul "outgoing" kernel (AlphaFold3 style) implemented with Triton.

The heavy O(N³·H) reduction:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

is performed by a tiled Triton kernel (block‑matrix multiply) where the
hidden dimension H is tiled as a third grid dimension.  The rest of the
TriMul pipeline (LayerNorm, linear projections, gating, final
normalization and projection) is carried out in PyTorch.  The kernel
works for batch size 1‑2, sequence length up to 1024 and hidden size
≤128.  Mixed‑precision is used: inputs to the kernel are stored as
float16, accumulation is float32 and the final hidden tensor is stored
as float32.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,        # *half   (N, N, H)   – left projection for a single batch
    right_ptr,       # *half   (N, N, H)   – right projection for a single batch
    out_ptr,         # *float  (N, N, H)   – output hidden tensor for a single batch
    # sizes
    N, H,
    # strides (in elements, not bytes)
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_i, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_m = tl.program_id(0)          # block index over i (rows)
    pid_n = tl.program_id(1)          # block index over j (cols)
    pid_h = tl.program_id(2)          # block index over hidden dimension

    # ------------------------------------------------------------------
    # Tile offsets
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N
    h_start = pid_h * BLOCK_H

    i_offset = i_start + tl.arange(0, BLOCK_M)
    j_offset = j_start + tl.arange(0, BLOCK_N)
    h_offset = h_start + tl.arange(0, BLOCK_H)

    i_mask = i_offset < N
    j_mask = j_offset < N
    h_mask = h_offset < H

    # accumulator for the (i, j, h) block
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_offset = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # left tile: (M, K, H)
        left_ptrs = left_ptr + (
            i_offset[:, None, None] * stride_left_i +
            k_offset[None, :, None] * stride_left_k +
            h_offset[None, None, :] * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :],
            other=0.0
        ).to(tl.float32)

        # right tile, transposed in the K dimension: (K, N, H)
        # we load using k as the first axis so that a dot product can be used.
        right_ptrs = right_ptr + (
            k_offset[:, None, None] * stride_right_i +     # k now indexes the former 'i' axis
            j_offset[None, :, None] * stride_right_k +     # j indexes the former 'k' axis
            h_offset[None, None, :] * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=k_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
            other=0.0
        ).to(tl.float32)

        # Dot for each hidden slice inside the tile
        for dh in range(0, BLOCK_H):
            # l: (M, K), r: (K, N)
            l = left[:, :, dh]
            r = right[:, :, dh]
            acc[:, :, dh] += tl.dot(l, r)

    # ------------------------------------------------------------------
    # Write result back
    out_ptrs = out_ptr + (
        i_offset[:, None, None] * stride_out_i +
        j_offset[None, :, None] * stride_out_j +
        h_offset[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :]
    )
    # ------------------------------------------------------------------


def custom_kernel(data):
    """
    Perform the forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) – may be ignored if config["nomask"] is True
            - weights: dict with all linear / layer‑norm parameters (torch.Tensors)
            - config: dict with keys "dim", "hidden_dim", and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    torch.cuda.empty_cache()  # ensure a clean launch environment
    with torch.no_grad():
        device = input_tensor.device
        B, N, _, dim = input_tensor.shape
        hidden_dim = config["hidden_dim"]
        eps = 1e-5

        # ------------------------------------------------------------------
        # Extract weights (all are stored as torch.float32 tensors)
        norm_weight = weights["norm.weight"]
        norm_bias = weights["norm.bias"]
        left_proj_weight = weights["left_proj.weight"]
        right_proj_weight = weights["right_proj.weight"]
        left_gate_weight = weights["left_gate.weight"]
        right_gate_weight = weights["right_gate.weight"]
        out_gate_weight = weights["out_gate.weight"]
        to_out_norm_weight = weights["to_out_norm.weight"]
        to_out_norm_bias = weights["to_out_norm.bias"]
        to_out_weight = weights["to_out.weight"]

        # ------------------------------------------------------------------
        # 1. Input LayerNorm
        x = torch.nn.functional.layer_norm(
            input_tensor,
            (dim,),
            weight=norm_weight,
            bias=norm_bias,
            eps=eps,
        )  # [B, N, N, C]

        # 2. Linear projections to hidden dimension (no bias)
        left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, H]
        right = torch.nn.functional.linear(x, right_proj_weight)

        # 3. Apply mask if required (mask broadcast over hidden dim)
        if not config.get("nomask", False):
            mask_f = mask.float().unsqueeze(-1)                 # [B, N, N, 1]
            left = left * mask_f
            right = right * mask_f

        # 4. Gates (sigmoid activation)
        left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
        right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
        out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

        # 5. Apply gating to projections
        left = left * left_gate
        right = right * right_gate

        # ------------------------------------------------------------------
        # 6. Compute the expensive O(N³·H) reduction with Triton
        # Cast to half for bandwidth‑limited loads
        left_h = left.to(torch.float16)
        right_h = right.to(torch.float16)

        # Allocate output hidden tensor (float32 accumulator)
        out_hidden = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

        # Block sizes (tuned for typical A100‑like GPUs)
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32
        BLOCK_H = 8

        # Strides for the non‑batch tensors (elements, not bytes)
        s_left_i = left_h.stride(1)   # stride over i (first N)
        s_left_k = left_h.stride(2)   # stride over k (second N)
        s_left_h = left_h.stride(3)   # stride over hidden
        s_right_i = right_h.stride(1)
        s_right_k = right_h.stride(2)
        s_right_h = right_h.stride(3)
        s_out_i = out_hidden.stride(1)
        s_out_j = out_hidden.stride(2)
        s_out_h = out_hidden.stride(3)

        # Launch one kernel per batch (B ≤ 2)
        grid = lambda: (
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            (hidden_dim + BLOCK_H - 1) // BLOCK_H,
        )
        for b in range(B):
            trimul_outgoing_kernel[grid()](
                left_h[b].data_ptr(),
                right_h[b].data_ptr(),
                out_hidden[b].data_ptr(),
                N,
                hidden_dim,
                s_left_i,
                s_left_k,
                s_left_h,
                s_right_i,
                s_right_k,
                s_right_h,
                s_out_i,
                s_out_j,
                s_out_h,
                BLOCK_M=BLOCK_M,
                BLOCK_N=BLOCK_N,
                BLOCK_K=BLOCK_K,
                BLOCK_H=BLOCK_H,
            )

        # ------------------------------------------------------------------
        # 7. Final LayerNorm + gating + output projection
        out_norm = torch.nn.functional.layer_norm(
            out_hidden,
            (hidden_dim,),
            weight=to_out_norm_weight,
            bias=to_out_norm_bias,
            eps=eps,
        )  # [B, N, N, H]

        out_gated = out_norm * out_gate
        output = torch.nn.functional.linear(out_gated, to_out_weight)  # -> [B, N, N, dim]

        return output
