```python
import torch
import triton
import triton.language as tl

# Triton kernel
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    # shape
    B, N, H,
    # strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # const
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    EPS: tl.constexpr = 1e-5,
):
    pid = tl.program_id(0)  # combined batch*hidden
    pid_i = tl.program_id(1)  # block row i
    pid_j = tl.program_id(2)  # block col j

    # map pid to batch and hidden index
    b = pid // H
    h = pid % H

    # base offset for this (b,h)
    offset_batch = b * stride_left_b + h * stride_left_h

    # compute offsets within N
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # create mask for out-of-range i/j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # offset for out_ptr for storing results
    out_ptrs = out_ptr + offset_batch + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile shape (M, K)
        left_ptrs = left_ptr + offset_batch + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile shape (K, N) where indices are (k, j)
        right_ptrs = right_ptr + offset_batch + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(left, right)

    # store result
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # get weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm over last dim dim
    # Compute mean and var
    x = input_tensor
    # flatten last dim for mean/var
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # (B,N,N,hidden)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    if not nomask:
        # mask shape (B,N,N). Expand.
        mask_exp = mask.unsqueeze(-1).to(dtype)  # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

    # Allocate output for TriMul
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=device)

    # launch triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # compute strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    grid = (B * hidden_dim, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Out_norm (layernorm over hidden_dim)
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var = ((out - out_mean) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - out_mean) * torch.rsqrt(out_var + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    out_norm = out_norm * out_gate

    # Final linear
    output = torch.nn.functional.linear(out_norm, to_out_weight)

    return output

"""
custom_kernel for the "outgoing" Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the last dimension.
2. Linear projections (left/right) from dim → hidden_dim.
3. Apply optional mask and per‑position gates (sigmoid of linear layers).
4. Core computation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply for every hidden slice.
   A Triton kernel performs the N×N GEMM for each (batch, hidden) pair,
   loading tiles of left and the transposed right tensor and accumulating
   with `tl.dot`.  The kernel works on float32 for accuracy.
5. Layer‑norm the intermediate result (hidden_dim), apply the output gate,
   and a final linear projection back to dim.

The Triton kernel handles the heavy O(B·hidden·N³) work, while the
surrounding linear / norm / gate logic is expressed in PyTorch.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul core
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # B=batch, N=seq_len, H=hidden_dim
    # strides of the three tensors (B,N,N,H)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid = tl.program_id(0)   # combined batch*hidden index
    pid_i = tl.program_id(1) # block row (i)
    pid_j = tl.program_id(2) # block column (j)

    # decode batch and hidden index from pid
    b = pid // H
    h = pid % H

    # base offset for this (b,h) slice
    batch_off = b * stride_left_b + h * stride_left_h

    # tile offsets in the sequence dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointer to the output tile
    out_ptrs = out_ptr + batch_off + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (M x K) -----
        left_ptrs = left_ptr + batch_off + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        mask_left = mask_i[:, None] & mask_k[None, :]
        left = tl.load(left_ptrs, mask=mask_left, other=0.0).to(tl.float32)

        # ----- load right tile (K x N) -----
        # right is logically transposed: we need (k, j) order
        right_ptrs = right_ptr + batch_off + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        mask_right = mask_k[:, None] & mask_j[None, :]
        right = tl.load(right_ptrs, mask=mask_right, other=0.0).to(tl.float32)

        # ----- accumulate -----
        acc += tl.dot(left, right)

    # store the result
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

            - mask: torch.Tensor[B, N, N] (bool or float) – may be ignored if config["nomask"]==True
            - weights: dict of model weights (see keys in the skeleton)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor[B, N, N, dim] – the TriMul output.
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x = input_tensor
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x_norm, left_proj_weight)   # (B,N,N,hidden)
    right = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional masking (mask shape B,N,N)
    # ------------------------------------------------------------------
    if not nomask:
        # ensure mask dtype matches computation
        mask_f = mask.to(dtype).unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = F.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = F.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = F.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core TriMul (batched GEMM) via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=device)

    # tile sizes (tuned for B200 GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides (PyTorch is row‑major: last dim fastest)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    grid = (
        B * hidden_dim,                     # combined batch*hidden
        (N + BLOCK_M - 1) // BLOCK_M,       # i‐blocks
        (N + BLOCK_N - 1) // BLOCK_N,       # j‐blocks
    )

    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,        # reasonable for the B200 architecture
    )

    # ------------------------------------------------------------------
    # LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var = ((out - out_mean) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - out_mean) * torch.rsqrt(out_var + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # ------------------------------------------------------------------
    # Output gating and final linear projection back to dim
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, to_out_weight)   # (B,N,N,dim)

    return output
