@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Determine batch_hidden index
    pid = tl.program_id(0)
    bh = pid
    b = bh // H
    h = bh % H
    
    # Determine tile indices
    tile_i = tl.program_id(1)
    tile_j = tl.program_id(2)
    
    # Offsets
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    
    # Keep within bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    
    # Initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N
        
        # left pointer: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b*stride_left_b + offs_i[:,None]*stride_left_i + offs_k[None,:]*stride_left_k + h*stride_left_h
        left_val = tl.load(left_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0)
        
        # right pointer: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + b*stride_right_b + offs_k[:,None]*stride_right_k + offs_j[None,:]*stride_right_j + h*stride_right_h
        right_val = tl.load(right_ptrs, mask=mask_k[:,None] & mask_j[None,:], other=0.0)
        
        # Compute dot
        acc += tl.dot(left_val, right_val)
    
    # Store result
    out_ptrs = out_ptr + b*stride_out_b + offs_i[:,None]*stride_out_i + offs_j[None,:]*stride_out_j + h*stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:,None] & mask_j[None,:])
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # Determine dims
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # Get weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # ----- LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    
    # ----- Linear Projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [bs,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)
    
    # ----- Mask
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # shape [bs,N,N,1]
        left = left * mask
        right = right * mask
    # else no mask
    
    # ----- Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    
    left = left * left_gate
    right = right * right_gate
    
    # ----- TriMul contraction via Triton
    # Output shape: [bs,N,N,hidden_dim]
    out = torch.empty_like(left)  # allocate
    
    # Define kernel
    import triton
    import triton.language as tl
    
    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        pid_bh = tl.program_id(0)
        b = pid_bh // H
        h = pid_bh % H
        pid_i = tl.program_id(1)
        pid_j = tl.program_id(2)
        # offsets
        offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_i = offs_i < N
        mask_j = offs_j < N
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        # Loop over k
        for kk in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N
            # Load left tile (M,K)
            left_ptrs = left_ptr + b*stride_l_b + offs_i[:,None]*stride_l_i + offs_k[None,:]*stride_l_k + h*stride_l_h
            left = tl.load(left_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0)
            # Load right tile (K,N) transposed (k,j)
            right_ptrs = right_ptr + b*stride_r_b + offs_k[:,None]*stride_r_k + offs_j[None,:]*stride_r_j + h*stride_r_h
            right = tl.load(right_ptrs, mask=mask_k[:,None] & mask_j[None,:], other=0.0)
            # Accumulate
            acc += tl.dot(left, right)
        # Store output
        out_ptrs = out_ptr + b*stride_o_b + offs_i[:,None]*stride_o_i + offs_j[None,:]*stride_o_j + h*stride_o_h
        tl.store(out_ptrs, acc, mask=mask_i[:,None] & mask_j[None,:])
    
    # Determine strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()
    
    # Choose block sizes based on N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # Compute grid dimensions
    grid = (
        bs * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    # launch kernel
    tri_mul_kernel[grid](
        left, right, out,
        bs, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    
    # ----- Output normalization and gating
    # to_out_norm: layer norm over hidden_dim
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate  # elementwise gating
    # final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)
    
    return out
```

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

The kernel computes, for each batch `b` and hidden head `h` (hidden_dim),
the matrix product:

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

which is equivalent to the Einstein sum
`einsum('... i k d, ... j k d -> ... i j d')`.  The kernel
processes a tile of rows (`BLOCK_M`) and columns (`BLOCK_N`) at a time
and iterates over the reduction dimension `k` in blocks (`BLOCK_K`),
using a fused `tl.dot` for high throughput.  The surrounding operations
(LayerNorm, linear projections, gates, post‑norm and final projection)
are performed with PyTorch for simplicity.

The implementation supports optional masking and works for any
batch size `B∈{1,2}`, sequence length `N≤1024`, input dimension `dim`
(128 or 384) and hidden dimension `hidden_dim` (commonly 128).
"""

import torch
import triton
import triton.language as tl

@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_bh = tl.program_id(0)                 # combines batch and hidden
    b = pid_bh // H
    h = pid_bh % H
    pid_i = tl.program_id(1)                  # row block
    pid_j = tl.program_id(2)                  # column block

    # block offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = (left_ptr
                     + b * stride_l_b
                     + offs_i[:, None] * stride_l_i
                     + offs_k[None, :] * stride_l_k
                     + h * stride_l_h)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile transposed: shape (BLOCK_K, BLOCK_N)
        # original right layout is (b, j, k, h); we load as (k, j)
        right_ptrs = (right_ptr
                      + b * stride_r_b
                      + offs_k[:, None] * stride_r_k
                      + offs_j[None, :] * stride_r_j
                      + h * stride_r_h)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # fused dot product over the K dimension
        acc += tl.dot(left, right)   # (M,K) dot (K,N) -> (M,N)

    # write out tile
    out_ptrs = (out_ptr
                + b * stride_o_b
                + offs_i[:, None] * stride_o_i
                + offs_j[None, :] * stride_o_j
                + h * stride_o_h)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward of the outgoing TriMul operation.

    Args:
        data: Tuple containing
            - input_tensor : torch.Tensor of shape [B, N, N, dim]
            - mask_tensor  : torch.Tensor of shape [B, N, N] or None
            - weights       : dict of weight tensors
            - config        : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ----------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # shape [B, N, N, dim]

    # ----------------------------------------------------------------------
    # 2) Linear projections to hidden dimension
    # ----------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------------
    # 3) Optional elementwise mask
    # ----------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------------
    # 4) Gating
    # ----------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for kernel indexing
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape

    # ----------------------------------------------------------------------
    # 5) Triangular multiplicative update via Triton (outgoing version)
    # ----------------------------------------------------------------------
    out = torch.empty_like(left)   # [B, N, N, hidden_dim]

    # strides in *elements* (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Block sizes – tuned for Hopper B200 (can be adjusted)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # batch * hidden head dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # number of row blocks
        (N + BLOCK_N - 1) // BLOCK_N,       # number of column blocks
    )

    _tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------------
    # 6) Post‑multiplication LayerNorm, gating and final projection
    # ----------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
