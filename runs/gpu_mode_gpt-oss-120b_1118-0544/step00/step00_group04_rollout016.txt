@triton.jit
def fused_mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    B, N, H, total_elements,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elements
    # compute mask index and offsets
    # Since left/right have shape (B,N,N,H) contiguous, we can map linear index to mask index: idx_mask = offs // H
    mask_idx = offs // H
    # load elements
    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask)
    left = left * mask_val * left_gate
    right = right * mask_val * right_gate
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    B: tl.int32, N: tl.int32, H: tl.int32, total_elements: tl.int32,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements
    # compute mask index (broadcast across H)
    mask_idx = offs // H
    # load values
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
    # apply mask and gates
    left = left * m * left_gate
    right = right * m * right_gate
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    """
    Triton-accelerated forward for the outgoing TriMul operator.
    Performs LayerNorm, linear projections, gating, masked elementwise fusion,
    batched matrix multiplication (L @ R^T) and final projection.
    """
    x, mask, weights, config = data
    device = x.device
    # config params
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Cast everything to float16 for compute
    x = x.to(torch.float16)
    # ---------- First LayerNorm ----------
    eps = 1e-5
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Flatten for linear layers
    B, N, _, _ = x.shape  # x shape [B, N, N, dim]
    x_flat = x.view(-1, dim)  # (B*N*N, dim)

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left = F.linear(x_flat, left_proj_weight).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, right_proj_weight).view(B, N, N, hidden_dim)

    # Gates (sigmoid)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_flat, left_gate_weight)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_weight)).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_weight)).view(B, N, N, hidden_dim)

    # Prepare mask
    if nomask:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(torch.float16)
    mask_tensor = mask_tensor.contiguous().view(-1)  # flatten to (B*N*N)

    # Ensure contiguous tensors for Triton kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # Fuse mask and gates using Triton
    total = B * N * N * hidden_dim
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total, BLOCK_SIZE),)
    fused_mask_gate_kernel[grid](
        left, right, mask_tensor,
        left_gate, right_gate,
        B, N, hidden_dim, total,
        BLOCK=BLOCK_SIZE,
    )

    # Batched matrix multiplication: out = left @ right^T for each hidden channel
    # reshape to (B*hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(-1, N, N)                # (B*hidden_dim, N, N)
    right_mat = right_perm.view(-1, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden_dim, N, N)

    # reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Second LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    # Apply out gate
    out = out * out_gate

    # Final linear projection to original dim
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out_flat = out.view(-1, hidden_dim)
    out = F.linear(out_flat, to_out_weight).view(B, N, N, dim)

    # Cast back to original dtype (float32)
    return out.to(x.dtype)
```

"""
Optimized outgoing TriMul (Triangular Multiplicative Update) forward pass.

The kernel fuses mask application and gating (left/right) into a single
element‑wise Triton kernel.  All dense linear operations are performed with
torch.nn.functional.linear in FP16 which maps to cuBLAS GEMM.  After the
gating step a batched matrix multiplication computes
    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
i.e. a per‑head GEMM L @ Rᵀ.  The result is then layer‑normalised,
gated by an output gate and projected back to the original dimension.

The implementation works for any batch size (≤2), sequence length (≤1024),
and supports optional masking.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    B: tl.int32, N: tl.int32, H: tl.int32, total: tl.int32,
    BLOCK: tl.constexpr
):
    """Apply mask and element‑wise gates to left/right tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # mask index is shared across the hidden dimension H
    mask_idx = offs // H

    # Load values (use 0.0 for out‑of‑bounds threads)
    left  = tl.load(left_ptr  + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    l_gate = tl.load(left_gate_ptr  + offs, mask=mask, other=0.0)
    r_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # Fuse mask and gates
    left  = left  * m * l_gate
    right = right * m * r_gate

    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input, mask, weights, config)
            - input: torch.Tensor [bs, N, N, dim] (float32)
            - mask:  torch.Tensor [bs, N, N] (bool/float) – may be all‑ones.
            - weights: dict with keys matching the PyTorch implementation.
            - config: dict with at least "dim", "hidden_dim", optionally "nomask".

    Returns:
        torch.Tensor of shape [bs, N, N, dim] (same dtype as input).
    """
    # Unpack arguments
    inp, mask, weights, config = data
    device = inp.device
    orig_dtype = inp.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Cast to FP16 for all heavy compute
    # ------------------------------------------------------------------
    x = inp.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. First LayerNorm (per last dimension)
    # ------------------------------------------------------------------
    eps = 1e-5
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b, eps=eps)

    # ------------------------------------------------------------------
    # 3. Linear projections and gate logits (flattened GEMM)
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)

    # Projection weights
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x_flat, left_proj_w).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, right_proj_w).view(B, N, N, hidden_dim)

    # Gate weights
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_flat, left_gate_w)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w)).view(B, N, N, hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4. Fuse mask & gates with a Triton kernel
    # ------------------------------------------------------------------
    if nomask:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(torch.float16)

    mask_flat = mask_tensor.contiguous().view(-1)               # (B*N*N)

    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total = B * N * N * hidden_dim
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    fused_mask_gate_kernel[grid](
        left, right, mask_flat,
        left_gate, right_gate,
        B, N, hidden_dim, total,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5. Batched matrix multiplication: out = left @ rightᵀ   (per hidden dim)
    # ------------------------------------------------------------------
    # Reshape to (B*hidden_dim, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Second LayerNorm
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w,
                       bias=to_out_norm_b, eps=eps)

    # ------------------------------------------------------------------
    # 7. Output gating and final projection
    # ------------------------------------------------------------------
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = F.linear(out.view(-1, hidden_dim), to_out_w).view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 8. Cast back to original dtype
    # ------------------------------------------------------------------
    return out.to(orig_dtype)
