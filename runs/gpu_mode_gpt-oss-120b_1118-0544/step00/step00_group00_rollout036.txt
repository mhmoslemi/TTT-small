    N, hidden_dim,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr

    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for boundary
    mask_m = offs_m < N
    mask_n = offs_n < N

    # loop over hidden dims
    for d_start in range(0, hidden_dim, BLOCK_D):
        cur_hidden = tl.minimum(BLOCK_D, hidden_dim - d_start)

            mask_k = offs_k < N

            mask_b = mask_k[:, None] & mask_n[None, :]

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True) # Not used else.

    # layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    if mask is not None:
        mask_f = mask.unsqueeze(-1).type_as(left) # float32
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # cast to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # create output tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    left_stride_b, left_stride_i, left_stride_j, left_stride_d = left.stride()
    right_stride_b, right_stride_i, right_stride_j, right_stride_d = right.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_d = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * hidden_dim, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, hidden_dim,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication for the outgoing triangle update.
    Computes out[b,i,j,hd] = sum_k left[b,i,k,hd] * right[b,j,k,hd].
    The `right` tensor is accessed transposed (right[k,j]).

    The kernel is launched with grid:
    (B*hidden_dim, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    where program_id(0) encodes (batch, hidden_dim) pair.
    """
    pid0 = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # decode batch and hidden index
    batch_id = pid0 // hidden_dim
    hid = pid0 % hidden_dim

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        mask_a = mask_m[:, None] & mask_k[None, :]
        mask_b = mask_k[:, None] & mask_n[None, :]

        # pointers
        a_ptr = left_ptr + batch_id * stride_left_batch \
                 + (offs_m[:, None] * stride_left_i) \
                 + (offs_k[None, :] * stride_left_k) \
                 + hid * stride_left_d

        b_ptr = right_ptr + batch_id * stride_right_batch \
                 + (offs_k[:, None] * stride_right_i) \
                 + (offs_n[None, :] * stride_right_k) \
                 + hid * stride_right_d

        acc += tl.dot(a, b).to(tl.float32)

               + (offs_n[None, :] * stride_out_j) \
               + hid * stride_out_d

"""
TriMul outgoing update (AlphaFold3)

The forward pass consists of:
1. LayerNorm on the 4‑D input.
2. Linear projections to a hidden dimension (left/right) and three sigmoid gates.
3. Optional element‑wise masking.
4. Element‑wise gating of the projected tensors.
5. Core computation:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication of each hidden slice:
       out_h = left_h @ right_hᵀ
   This N³ reduction is implemented in a custom Triton kernel.
6. LayerNorm on the hidden result, scaling by the out‑gate,
   and a final linear projection back to the original dimension.
All heavy reductions are performed in FP16 (inputs) with FP32 accumulation.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, hidden_dim,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute the outgoing triangle multiplication:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Launch grid:
        (B * hidden_dim,  ceil(N / BLOCK_M),  ceil(N / BLOCK_N))
    program_id(0) encodes (batch, hidden) pair, program_id(1) -> i‑tile,
    program_id(2) -> j‑tile.
    """
    pid0 = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Decode batch index and hidden channel index
    batch_id = pid0 // hidden_dim
    hid = pid0 % hidden_dim

    # Tile offsets for the output matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Reduction over the K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        mask_a = mask_m[:, None] & mask_k[None, :]
        mask_b = mask_k[:, None] & mask_n[None, :]

        # Left operand: left[b, i, k, h]
        a_ptr = left_ptr + batch_id * stride_left_batch \
                + (offs_m[:, None] * stride_left_i) \
                + (offs_k[None, :] * stride_left_k) \
                + hid * stride_left_d

        # Right operand transposed: we need right[b, k, j, h] which corresponds to
        # original right[b, j, k, h] accessed with swapped indices.
        b_ptr = right_ptr + batch_id * stride_right_batch \
                + (offs_k[:, None] * stride_right_i) \
                + (offs_n[None, :] * stride_right_k) \
                + hid * stride_right_d

        a = tl.load(a_ptr, mask=mask_a, other=0.0).to(tl.float16)
        b = tl.load(b_ptr, mask=mask_b, other=0.0).to(tl.float16)

        # Dot product over K, accumulate in FP32
        acc += tl.dot(a, b).to(tl.float32)

    # Write the tile back to memory
    out_ptrs = out_ptr + batch_id * stride_out_batch \
               + (offs_m[:, None] * stride_out_i) \
               + (offs_n[None, :] * stride_out_j) \
               + hid * stride_out_d
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N]  (or None)
        - weights      : dict of weight tensors
        - config       : dict containing "dim", "hidden_dim", "nomask", ...

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- 1) Input LayerNorm ----------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ---------- 2) Linear projections ----------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- 3) Optional masking ----------
    if not nomask and mask is not None:
        mask_f = mask.to(input_tensor.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ---------- 4) Gates ----------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # Cast to FP16 for the heavy N³ kernel
    left = left.contiguous().to(torch.float16)
    right = right.contiguous().to(torch.float16)

    B, N, _, _ = left.shape

    # ---------- 5) Core TriMul (batched GEMM) ----------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # Strides for pointer arithmetic inside the kernel
    l_stride = left.stride()   # (batch, i, k, d)
    r_stride = right.stride()
    o_stride = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                      # batch * hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,        # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N         # j‑tiles
    )

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        N,
        hidden_dim,
        l_stride[0], l_stride[1], l_stride[2], l_stride[3],
        r_stride[0], r_stride[1], r_stride[2], r_stride[3],
        o_stride[0], o_stride[1], o_stride[2], o_stride[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- 6) Output LayerNorm, gating and final projection ----------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate                     # element‑wise gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
