@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)

    b = pid_hb // H
    h = pid_hb % H

    # Offsets for i, j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_valid = offs_i < N
    j_valid = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for block_k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = block_k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        k_valid = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left_mask = i_valid[:, None] & k_valid[None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # Load right tile (transposed): shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right_mask = k_valid[:, None] & j_valid[None, :]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

"""
Optimized Triton implementation of the “outgoing” Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor `x` over its last dimension.
2. Compute left/right projections and their logistic gates (all in FP32).
3. Apply the optional mask and the gates:   left = left * left_gate,   right = right * right_gate.
4. Cast `left` and `right` to FP16 and compute the core tensor contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batched matrix‑multiply `L @ Rᵀ` for each head `d`.
   The contraction is performed by a fused Triton kernel that processes a
   tile of (BLOCK_M × BLOCK_N) output elements while looping over the K‑dimension.
5. Convert the FP16 result back to FP32, apply the output layer‑norm,
   the output gate and the final linear projection back to `dim`.

The kernel operates on a flattened batch‑head dimension (batch * hidden_dim)
to avoid any synchronisation across heads.  All boundary tiles are masked
to handle arbitrary sequence lengths.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # tile row
    pid_j = tl.program_id(1)  # tile col
    pid_hb = tl.program_id(2) # batch*head index

    # decode batch and head
    b = pid_hb // H
    h = pid_hb % H

    # global offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_valid = offs_i < N
    j_valid = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for pid_k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = pid_k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        k_valid = offs_k < N

        # ----- load left tile (M x K) -----
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
        )
        left_mask = i_valid[:, None] & k_valid[None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0)   # FP16

        # ----- load right tile transposed (K x N) -----
        # we need right[j, k] -> we load as (k, j) to match GEMM semantics
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_i
        )
        right_mask = k_valid[:, None] & j_valid[None, :]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0)  # FP16

        # accumulate (FP16 inputs, FP32 accumulation)
        acc += tl.dot(left, right)   # result is FP32

    # ----- store result (convert back to FP16) -----
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    out_mask = i_valid[:, None] & j_valid[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=out_mask)


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """Forward pass of the outgoing TriMul operator using a fused Triton kernel."""
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (FP32)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1]
        mask_broadcast = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_broadcast
        right = right * mask_broadcast

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) and gating of projections
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the heavy contraction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    B, N, _, H = left_fp16.shape  # B,N,N,hidden_dim

    # ------------------------------------------------------------------
    # 6. Allocate output buffer (FP16) and launch Triton kernel
    # ------------------------------------------------------------------
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # Strides (in elements) for each tensor
    s_l_b, s_l_i, s_l_k, s_l_h = left_fp16.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right_fp16.stride()
    s_o_b, s_o_i, s_o_k, s_o_h = out_fp16.stride()   # note: third stride corresponds to the j‑dimension

    # Tunable block sizes – chosen to give good occupancy on A100/B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,      # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,      # j‑tiles
        B * H,                             # batch × head
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        H,
        s_l_b, s_l_i, s_l_k, s_l_h,
        s_r_b, s_r_i, s_r_k, s_r_h,
        s_o_b, s_o_i, s_o_k, s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # 7. Convert back to FP32 and finish the remaining lightweight ops
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # Output layer‑norm over hidden_dim
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Apply output gate
    out = out * out_gate
    # Final linear projection back to `dim`
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
