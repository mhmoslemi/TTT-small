def layernorm_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr,
                     N_vectors, C, eps,
                     stride_xn, stride_xc,
                     stride_yn, stride_yc,
                     stride_wn, stride_wc,
                     stride_bn, stride_bc):
  pid = tl.program_id(0)
  if pid >= N_vectors: return

  # Compute base pointer for this row
  x_row_ptr = x_ptr + pid * stride_xn
  y_row_ptr = y_ptr + pid * stride_yn

  # Compute sum and sum_sq
  sum_val = tl.zeros([1], dtype=tl.float32)
  sum_sq = tl.zeros([1], dtype=tl.float32)

  # Loop over C in blocks
  for c in range(0, C, BLOCK_SIZE):
    cur_off = c + tl.arange(0, BLOCK_SIZE)
    mask = cur_off < C
    x = tl.load(x_row_ptr + cur_off * stride_xc, mask=mask, other=0.0).to(tl.float32)
    sum_val += tl.sum(x, axis=0)
    sum_sq += tl.sum(x * x, axis=0)

  mean = sum_val / C
  var = sum_sq / C - mean * mean
  rstd = tl.rsqrt(var + eps)

  # Normalize and store
  for c in range(0, C, BLOCK_SIZE):
    cur_off = c + tl.arange(0, BLOCK_SIZE)
    mask = cur_off < C
    x = tl.load(x_row_ptr + cur_off * stride_xc, mask=mask, other=0.0).to(tl.float32)
    norm = (x - mean) * rstd
    # apply weight and bias if provided
    if weight_ptr != 0:
        w = tl.load(weight_ptr + cur_off * stride_wc, mask=mask, other=1.0).to(tl.float32)
        norm = norm * w
    if bias_ptr != 0:
        b = tl.load(bias_ptr + cur_off * stride_bc, mask=mask, other=0.0).to(tl.float32)
        norm = norm + b
    tl.store(y_row_ptr + cur_off * stride_yc, norm.to(tl.float32), mask=mask)

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, dim_ = input_tensor.shape
    assert dim_ == dim

    eps = 1e-5

    # Define layernorm using Triton
    def triton_layernorm(x, weight, bias):
        # x shape [B, N, N, C]
        B,N,_ ,C = x.shape
        total = B*N*N
        # flatten
        x_flat = x.reshape(total, C)
        out_flat = torch.empty_like(x_flat)
        # compute strides
        stride_x0, stride_x1 = x_flat.stride()
        stride_y0, stride_y1 = out_flat.stride()
        # weight/bias are 1D of size C (maybe need view)
        w = weight
        b = bias
        # pass pointers
        # call kernel
        grid = (total,)
        BLOCK_SIZE = 128 # compile-time
        layernorm_kernel[grid](
            x_flat, out_flat, w, b,
            total, C,

            stride_x0, stride_x1,
            stride_y0, stride_y1,
            w.stride(0), 1,
            b.stride(0), 1,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # reshape back
        return out_flat.view(B,N,N,C)

@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    N_vec, C,
    eps,
    stride_xn, stride_xc,
    stride_yn, stride_yc,
    stride_wn, stride_bn,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    if pid >= N_vec:
        return

    # compute pointers for this row
    x_row_ptr = x_ptr + pid * stride_xn
    y_row_ptr = y_ptr + pid * stride_yn

    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    # loop over C
    for offset in range(0, C, BLOCK_SIZE):
        cur = offset + tl.arange(0, BLOCK_SIZE)
        mask = cur < C
        # load as the dtype of the pointer
        x = tl.load(x_row_ptr + cur * stride_xc, mask=mask, other=0.0)
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=0)
        sum_sq += tl.sum(x_f32 * x_f32, axis=0)

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second loop storing normalized values
    for offset in range(0, C, BLOCK_SIZE):
        cur = offset + tl.arange(0, BLOCK_SIZE)
        mask = cur < C

        # reload
        x = tl.load(x_row_ptr + cur * stride_xc, mask=mask, other=0.0)
        x_f32 = x.to(tl.float32)

        norm = (x_f32 - mean) * rstd

        # weight and bias
        w = tl.load(weight_ptr + cur * stride_wn, mask=mask, other=1.0)
        b = tl.load(bias_ptr + cur * stride_bn, mask=mask, other=0.0)
        w_f32 = w.to(tl.float32)
        b_f32 = b.to(tl.float32)

        out = norm * w_f32 + b_f32
        # cast back to original dtype
        out_cast = out.to(x.dtype)  # this will cast to same dtype as input (float16 or float32)
        tl.store(y_row_ptr + cur * stride_yc, out_cast, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    # retrieve dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # layernorm on input
    x = custom_layernorm(input_tensor, weights["norm.weight"], weights["norm.bias"], eps=eps)

    # linear projections
    # these weight tensors are shape (hidden_dim, dim)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask if needed
    if not config.get("nomask", False):
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask_f

    # gating
    out_gated = out_norm * out_gate

    # final linear to dim
    out_final = torch.nn.functional.linear(out_gated, weights["to_out.weight"])

    return out_final

"""
TriMul (outgoing) forward pass with custom Triton LayerNorm.

The implementation follows the reference PyTorch module but replaces the
LayerNorm operations with a fused Triton kernel.  The steps are:

1. Layer‑norm the input tensor (shape [B,N,N,D]).
2. Linear projections to hidden dimension (left/right) and gated linear
   layers (left_gate, right_gate, out_gate).
3. Optional element‑wise masking.
4. Element‑wise multiplication with gates.
5. Batched matrix multiplication over the sequence dimension:
   for each hidden channel h:
       out_h = left_h @ right_h^T   (shape [B,N,N])
   This is realized with a single torch.bmm over the flattened batch.
6. Layer‑norm of the intermediate result.
7. Gating with out_gate.
8. Final linear projection back to the original dimension.

Only the LayerNorm is implemented in Triton – this satisfies the
requirement to use a custom kernel while keeping the heavy O(N³) reduction
efficient by leveraging cuBLAS.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # input & output pointers
    weight_ptr, bias_ptr,       # LayerNorm weight & bias
    n_vec, c,                   # number of vectors, channel dim
    eps,                        # epsilon
    stride_xn, stride_xc,       # strides for input (row, channel)
    stride_yn, stride_yc,       # strides for output
    stride_wn, stride_bn,       # strides for weight & bias (always 1)
    BLOCK_SIZE: tl.constexpr,   # compile‑time block size
):
    """Fused LayerNorm (float32) for a 2‑D view [n_vec, c].

    Each program processes one row (one vector of length c).  The kernel
    computes mean/variance in FP32, normalizes, then applies the affine
    parameters.
    """
    pid = tl.program_id(0)
    if pid >= n_vec:
        return

    # Pointers to the start of the row
    x_row = x_ptr + pid * stride_xn
    y_row = y_ptr + pid * stride_yn

    # ------------------------------------------------------------------
    # 1) Compute mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    for offset in range(0, c, BLOCK_SIZE):
        cur = offset + tl.arange(0, BLOCK_SIZE)
        mask = cur < c
        # Load a block of the row (dtype follows the pointer, float32 here)
        x = tl.load(x_row + cur * stride_xc, mask=mask, other=0.0)
        xf = x.to(tl.float32)
        sum_val += tl.sum(xf, axis=0)
        sum_sq  += tl.sum(xf * xf, axis=0)

    mean = sum_val / c
    var  = sum_sq / c - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2) Normalize and apply affine transformation
    # ------------------------------------------------------------------
    for offset in range(0, c, BLOCK_SIZE):
        cur  = offset + tl.arange(0, BLOCK_SIZE)
        mask = cur < c
        x = tl.load(x_row + cur * stride_xc, mask=mask, other=0.0)
        xf = x.to(tl.float32)

        # (x - mean) / sqrt(var + eps)
        norm = (xf - mean) * rstd

        # Load weight / bias (default weight=1, bias=0 if needed)
        w = tl.load(weight_ptr + cur * stride_wn, mask=mask, other=1.0)
        b = tl.load(bias_ptr   + cur * stride_bn, mask=mask, other=0.0)
        wf = w.to(tl.float32)
        bf = b.to(tl.float32)

        out = norm * wf + bf        # still FP32
        tl.store(y_row + cur * stride_yc, out, mask=mask)


def triton_layernorm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """
    Apply LayerNorm (affine) using the Triton kernel.
    x   : [B, N, N, C] (float32, contiguous)
    weight, bias: [C] (float32)
    returns tensor of same shape as x.
    """
    assert x.is_cuda and weight.is_cuda and bias.is_cuda
    B, N, _, C = x.shape
    total_vec = B * N * N

    # Flatten to [total_vec, C]
    x_flat = x.reshape(total_vec, C).contiguous()
    y_flat = torch.empty_like(x_flat)

    # Strides for the flattened tensors
    stride_xn, stride_xc = x_flat.stride()
    stride_yn, stride_yc = y_flat.stride()
    stride_wn = weight.stride(0)
    stride_bn = bias.stride(0)

    BLOCK_SIZE = 128  # works for all C in the test set

    grid = (total_vec,)
    layernorm_kernel[grid](
        x_flat, y_flat,
        weight, bias,
        total_vec, C,
        eps,
        stride_xn, stride_xc,
        stride_yn, stride_yc,
        stride_wn, stride_bn,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return y_flat.view(B, N, N, C)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, S, S, D] (float32, CUDA)
        - mask         : torch.Tensor [B, S, S] (float32 or bool, CUDA)
        - weights      : dict of torch.Tensors containing all linear/LN params
        - config       : dict with keys "dim", "hidden_dim", and "nomask"

    Returns
    -------
    torch.Tensor [B, S, S, D]  (float32, CUDA)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1) Input LayerNorm (custom Triton kernel)
    # --------------------------------------------------------------
    x_norm = triton_layernorm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Optional mask
    # --------------------------------------------------------------
    if not config.get("nomask", False):
        mask_exp = mask.to(x_norm.dtype).unsqueeze(-1)   # [B, S, S, 1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4) Gating
    # --------------------------------------------------------------
    left_gate  = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Core TriMul reduction via batched matrix multiplication
    # --------------------------------------------------------------
    B, S, _, _ = input_tensor.shape
    # reshape to (B*hidden_dim, S, S) for torch.bmm
    left_flat  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, S, S)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, S, S)

    # out_flat: (B*hidden_dim, S, S) = left_flat @ right_flat^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, hidden_dim, S, S).permute(0, 2, 3, 1)   # [B, S, S, H]

    # --------------------------------------------------------------
    # 6) Output LayerNorm (custom Triton kernel)
    # --------------------------------------------------------------
    out_norm = triton_layernorm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 7) Output gating
    # --------------------------------------------------------------
    out_gated = out_norm * out_gate

    # --------------------------------------------------------------
    # 8) Final linear projection back to original dimension
    # --------------------------------------------------------------
    final = F.linear(out_gated, weights["to_out.weight"])

    return final
