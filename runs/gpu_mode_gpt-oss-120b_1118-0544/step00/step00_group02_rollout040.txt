Define kernel:

```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_i = tl.program_id(0)  # i block
    pid_j = tl.program_id(1)  # j block
    b = tl.program_id(2)      # batch index

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # allocate accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # Loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K

        # create offsets
        i = tl.arange(0, BLOCK_I)
        j = tl.arange(0, BLOCK_J)
        k = tl.arange(0, BLOCK_K)
        d = tl.arange(0, BLOCK_D)

        i_idx = i_base + i
        j_idx = j_base + j
        k_idx = k_base + k
        d_idx = d

        # masks for bounds
        mask_i = i_idx < N
        mask_j = j_idx < N
        mask_k = k_idx < N
        mask_d = d_idx < D

        # left: shape (BLOCK_I, BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr \
            + b * stride_left_b \
            + i_idx[:, None, None] * stride_left_i \
            + k_idx[None, :, None] * stride_left_k \
            + d_idx[None, None, :] * stride_left_d
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

        # right transposed: shape (BLOCK_K, BLOCK_J, BLOCK_D)
        right_ptrs = right_ptr \
            + b * stride_right_b \
            + j_idx[None, :, None] * stride_right_j \
            + k_idx[:, None, None] * stride_right_k \
            + d_idx[None, None, :] * stride_right_d
        right = tl.load(right_ptrs, mask=mask_j[None, :, None] & mask_k[:, None, None] & mask_d[None, None, :], other=0.0)

        # accumulate dot products per channel
        for d_off in tl.static_range(BLOCK_D):
            # left slice (i,k)
            left_d = left[:, :, d_off]   # (BLOCK_I, BLOCK_K)
            # right slice (k,j)
            right_d = right[:, :, d_off] # (BLOCK_K, BLOCK_J)
            acc[:, :, d_off] += tl.dot(left_d, right_d)  # (BLOCK_I, BLOCK_J)
    # after reduction, store result

        + b * stride_out_b \
        + i_idx[:, None, None] * stride_out_i \
        + j_idx[None, :, None] * stride_out_j \
        + d_idx[None, None, :] * stride_out_d

"""
TriMul Outgoing implementation with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input.
2. Linear projections (left, right) and gating (left_gate, right_gate,
   out_gate) – done in PyTorch.
3. Apply the mask (if any) and gates to left/right.
4. Compute the heavy “einsum”:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is exactly a batched matrix multiplication
        out_flat = left_flat @ right_flat.T
   where left_flat and right_flat are the tensors reshaped to
   (B*D, N, N) with the second operand transposed.
   The multiplication is performed by a custom Triton kernel that
   tiles the M, N and K dimensions (BLOCK_M×BLOCK_N×BLOCK_K).
5. Layer‑norm the result, multiply by out_gate and a final linear
   projection – also in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B) where each batch corresponds to
# a (batch * hidden_dim) entry.  A shape (B*D, M, K), B shape (B*D, K, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    batch, M, N, K,                    # sizes
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch index (B*D)

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-range rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M×K)
        a_ptrs = (
            A_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B tile (K×N) – B is already transposed
        b_ptrs = (
            B_ptr
            + pid_b * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate
        acc += tl.dot(a, b)

    # Write back C
    c_ptrs = (
        C_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entrypoint callable from the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    `data` is a tuple (input, mask, weights, config) where:
        input : torch.Tensor [B, N, N, dim]
        mask  : torch.Tensor [B, N, N]  (may be None)
        weights: dict of model parameters
        config : dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim]
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. LayerNorm on the input
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Mask handling (if nomask -> all ones)
    # --------------------------------------------------------------
    if config.get("nomask", False) or mask is None:
        mask_tensor = torch.ones((batch, N, N), dtype=dtype, device=device)
    else:
        mask_tensor = mask.to(dtype)

    mask_tensor = mask_tensor.unsqueeze(-1)  # [B,N,N,1]
    left = left * mask_tensor
    right = right * mask_tensor

    # --------------------------------------------------------------
    # 4. Gating
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core TriMul (Batched GEMM via Triton)
    # --------------------------------------------------------------
    # Rearrange tensors to (B, D, N, N) and flatten batch*D
    left_perm = left.permute(0, 3, 1, 2).contiguous()        # (B, D, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()      # (B, D, N, N)  (transposed on last two axes)

    BHD = batch * hidden_dim                                   # effective batch size for GEMM

    left_flat = left_perm.view(BHD, N, N)      # (B*D, M=N, K=N)
    right_flat = right_perm.view(BHD, N, N)    # (B*D, K=N, N=N)

    out_flat = torch.empty_like(left_flat)    # (B*D, N, N)

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N

    batched_gemm_kernel[
        (grid_m, grid_n, BHD)
    ](
        left_flat,
        right_flat,
        out_flat,
        BHD,
        N,
        N,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(batch, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output layer‑norm, final gating and linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
