@triton.jit
def tril_mul_reduce_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, K,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    D_START: tl.constexpr, D_SIZE: tl.constexpr
):
    # B, N, K same (K = N)
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute i_start, j_start
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # offsets for M,N
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Offsets for d
    offs_d = tl.arange(0, D_SIZE) + D_START
    # mask_d = offs_d < hidden_dim maybe not needed

    # initialize accumulator
    # shape (BLOCK_M, BLOCK_N, D_SIZE)
    acc = tl.zeros((BLOCK_M, BLOCK_N, D_SIZE), dtype=tl.float32)

    # loop over k blocks
    for k_base in range(0, K, BLOCK_K):
        # offset for k
        k_offset = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # calculate pointer offsets for left and right
        # left: indexes (b, i, k, d)
        left_ptrs = left_ptr + pid_b * stride_bl \
            + offs_i[:, None, None] * stride_il \
            + k_offset[None, :, None] * stride_kl \
            + offs_d[None, None, :] * stride_hl
        # right: indexes (b, j, k, d), note right shape is (B,N,N,H). We need to index (b, j, k, d)
        right_ptrs = right_ptr + pid_b * stride_br \
            + offs_j[:, None, None] * stride_ir \
            + k_offset[None, :, None] * stride_kr \
            + offs_d[None, None, :] * stride_hr

        # load left and right tiles
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & (offs_d[None, None, :] < D_SIZE), other=0.0) # Actually D_SIZE always valid
        right_tile = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & (offs_d[None, None, :] < D_SIZE), other=0.0)

        # Multiply and accumulate across k dimension:
        # left_tile shape (M,K,D), right_tile shape (N,K,D) but we need shape (M,K,N,D) for product.
        # To compute sum over k, we want for each d: acc[i,j,d] += sum_k left[i,k,d] * right[j,k,d]
        # We can compute using outer product per k for each d: left_tile[:,k,d].unsqueeze(1) * right_tile[:,k,d].unsqueeze(0)
        # Equivalent to: for each k, compute outer = left_tile[:, k, d][:, None] * right_tile[:, k, d][None, :]
        # We'll accumulate across k and d.

        # Use manual loop over k within block
        for kk in range(0, BLOCK_K):
            # Note: we need to check that k_base+kk < K
            k_valid = (k_base + kk) < K
            if not k_valid:
                continue
            # Load slice per channel
            left_vec = left_tile[:, kk, :]   # (M,D)
            right_vec = right_tile[:, kk, :]  # (N,D)
            # Compute outer product for each d: left_vec[:, d] * right_vec[:, d].T => (M,N) per d
            # We can use broadcasting: left_vec[:, None, :] * right_vec[None, :, :] -> (M,N,D)
            # In Triton we can use tl.broadcast? We can do: prod = left_vec[:, None, :] * right_vec[None, :, :]
            prod = left_vec[:, None, :] * right_vec[None, :, :]  # shape (M,N,D)
            acc += prod  # broadcasts

    # store result
    out_ptrs = out_ptr + pid_b * stride_bo \
        + offs_i[:, None, None] * stride_io \
        + offs_j[None, :, None] * stride_jo \
        + offs_d[None, None, :] * stride_ho

    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & (offs_d[None, None, :] < D_SIZE))

    # left, right shape (B,N,N,H). Left dtype float32.
    B,N,N_,H = left.shape
    assert N == N_
    # compute output shape.
    out = torch.empty_like(left)  # allocate output
    # compute strides:
    left_strides = left.stride()
    right_strides = right.stride()
    out_strides = out.stride()
    # stride in elements not bytes
    # Launch grid:
    grid = (B, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)
    # Host loop over d blocks
    for d_start in range(0, H, BLOCK_D):

        # compile kernel with D_START and D_SIZE as constexpr
        trilu_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            out_ptr=out,

            K=N,
            stride_bl=left_strides[0],
            stride_il=left_strides[1],
            stride_kl=left_strides[2],
            stride_hl=left_strides[3],

            stride_ir=right_strides[1],

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Retrieve weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # layernorm on input
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask
    nomask = config.get("nomask", False)
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # shape (...,1)
        left = left * mask
        right = right * mask

    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate

    out = triton_trimul_reduce(left, right)

    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # apply out_gate
    out = out * out_gate

    # final projection
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

@triton.jit
def tril_mul_reduce_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, K,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    D_START: tl.constexpr, D_SIZE: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    offs_d = tl.arange(0, D_SIZE) + D_START

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, D_SIZE), dtype=tl.float32)

    for k_base in range(0, K, BLOCK_K):
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # compute pointer offsets
        left_ptrs = left_ptr + pid_b * stride_bl \
                    + offs_i[:, None, None] * stride_il \
                    + offs_k[None, :, None] * stride_kl \
                    + offs_d[None, None, :] * stride_hl

        right_ptrs = right_ptr + pid_b * stride_br \
                     + offs_j[:, None, None] * stride_ir \
                     + offs_k[None, :, None] * stride_kr \
                     + offs_d[None, None, :] * stride_ho

        # load tiles
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0)

        # iterate over K inside block
        for kk in range(0, BLOCK_K):
            # optional: check if k_valid, but load already masked; we can include mask on kk but not needed.
            left_vec = left_tile[:, kk, :]   # shape (BLOCK_M, D_SIZE)
            right_vec = right_tile[:, kk, :] # shape (BLOCK_N, D_SIZE)
            # outer product across i and j dimensions for each d

            prod = left_vec[:, None, :] * right_vec[None, :, :]  # shape (M,N,D)
            acc += prod

                + offs_i[:, None, None] * stride_io \
                + offs_j[None, :, None] * stride_jo \
                + offs_d[None, None, :] * stride_ho

    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None])

"""
TriMul (outgoing) forward with a custom Triton kernel for the
O(N³·C_hidden) reduction.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections → left, right (shape [B,N,N,C_hidden]).
3. Optional mask multiplication.
4. Gating: sigmoid(linear(x)) applied element‑wise to left/right.
5. Reduction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a Triton kernel that tiles the (i,j) plane and
   processes a small chunk of the hidden dimension at a time.
   The kernel accummulates a (BLOCK_M, BLOCK_N, D_BLOCK) tile in
   registers, loops over K‑blocks and over the K dimension inside the
   block, performing an outer‑product for every k.
6. Layer‑norm on the hidden dimension,
   multiply by an output gate, and a final linear projection back to
   the original dimension.

Only the reduction step is performed on the GPU with Triton; all other
operations use regular PyTorch (which is already highly optimized).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton reduction kernel
# ----------------------------------------------------------------------
@triton.jit
def tril_mul_reduce_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, K,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    D_START: tl.constexpr, D_SIZE: tl.constexpr
):
    """
    Compute a tile of the TriMul reduction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    Parameters
    ----------
    left_ptr, right_ptr, out_ptr : pointers to the tensors
    B, N, K                     : batch size, sequence length, reduction length
    stride_*                    : strides of the tensors (in elements)
    BLOCK_M, BLOCK_N, BLOCK_K   : tiling sizes
    D_START, D_SIZE             : start index and size of the hidden‑dim block
    """
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    offs_d = tl.arange(0, D_SIZE) + D_START

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator: (M, N, D_BLOCK) in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, D_SIZE), dtype=tl.float32)

    for k_base in range(0, K, BLOCK_K):
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ------------------- load tiles -------------------
        left_ptrs = left_ptr + pid_b * stride_bl \
                    + offs_i[:, None, None] * stride_il \
                    + offs_k[None, :, None] * stride_kl \
                    + offs_d[None, None, :] * stride_hl

        right_ptrs = right_ptr + pid_b * stride_br \
                     + offs_j[:, None, None] * stride_ir \
                     + offs_k[None, :, None] * stride_kr \
                     + offs_d[None, None, :] * stride_hr

        left_tile = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None],
            other=0.0
        )
        right_tile = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None],
            other=0.0
        )
        # ------------------- inner reduction over k -------------------
        for kk in range(0, BLOCK_K):
            # left_vec: (M, D), right_vec: (N, D)
            left_vec = left_tile[:, kk, :]   # (M, D_SIZE)
            right_vec = right_tile[:, kk, :] # (N, D_SIZE)

            # outer product per hidden channel:
            # (M,1,D) * (1,N,D) -> (M,N,D)
            prod = left_vec[:, None, :] * right_vec[None, :, :]
            acc += prod

    # ------------------- store result -------------------
    out_ptrs = out_ptr + pid_b * stride_bo \
               + offs_i[:, None, None] * stride_io \
               + offs_j[None, :, None] * stride_jo \
               + offs_d[None, None, :] * stride_ho

    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None]
    )


def triton_trimul_reduce(left: torch.Tensor, right: torch.Tensor,
                         BLOCK_M=32, BLOCK_N=32, BLOCK_K=32, BLOCK_D=8):
    """
    Wrapper that launches `tril_mul_reduce_kernel` for the full hidden dimension.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides are in elements (torch's stride)
    stride_bl, stride_il, stride_kl, stride_hl = left.stride()
    stride_br, stride_ir, stride_kr, stride_hr = right.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out.stride()

    grid = (B, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    for d_start in range(0, H, BLOCK_D):
        d_size = min(BLOCK_D, H - d_start)
        tril_mul_reduce_kernel[grid](
            left, right, out,
            B, N, N,
            stride_bl, stride_il, stride_kl, stride_hl,
            stride_br, stride_ir, stride_kr, stride_hr,
            stride_bo, stride_io, stride_jo, stride_ho,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            D_START=d_start,
            D_SIZE=d_size,
            num_warps=4,
        )
    return out


# ----------------------------------------------------------------------
# Entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    O(N³·C_hidden) reduction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] on the same device as the input.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) O(N³·C_hidden) reduction via Triton
    # ------------------------------------------------------------------
    out = triton_trimul_reduce(left, right)

    # ------------------------------------------------------------------
    # 6) Output layer‑norm, output gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # back to [B,N,N,dim]

    return out
