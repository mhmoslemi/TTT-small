import torch
import triton
import triton.language as tl

# Mask apply kernel
@triton.jit
def mask_mul_kernel(left_ptr, right_ptr, mask_ptr,
                    total_elements, hidden_dim,
                    BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elements
    # compute index for mask (flatten over B,N,N)
    idx0 = offs // hidden_dim
    # load mask
    m = tl.load(mask_ptr + idx0, mask=mask)
    # load left and right
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    # multiply
    l = l * m
    r = r * m
    # store
    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)

def apply_mask_triton(left, right, mask):
    """Apply mask (shape [B,N,N]) to left/right tensors (shape [B,N,N,hidden_dim]) in-place."""
    if mask is None:
        return
    # Ensure contiguous and same dtype
    left = left.contiguous()
    right = right.contiguous()
    mask = mask.contiguous().to(left.dtype)
    B, N, _, hidden_dim = left.shape
    total = B * N * N * hidden_dim
    BLOCK = 1024
    grid = ( (total + BLOCK - 1) // BLOCK, )
    mask_mul_kernel[grid](left, right, mask,
                          total, hidden_dim,
                          BLOCK=BLOCK)

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) module.
    Performs LayerNorm, linear projections, masking (via Triton), gating,
    a batched matrix‑multiply reduction, final LayerNorm, gating and output projection.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------
    # LayerNorm on input
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------
    # Linear projections (left, right) without bias
    # -------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # Apply mask (if any) using Triton kernel
    # -------------------------------------------------
    # The config may indicate there is no mask (nomask=True)
    if not config.get("nomask", False) and mask is not None:
        apply_mask_triton(left, right, mask)

    # -------------------------------------------------
    # Gating
    # -------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # Core reduction: out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
    # Implemented via batched matrix multiplication.
    # -------------------------------------------------
    # reshape for batched matmul
    # left: [B,N,N,hidden_dim] -> [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: same permute, then transpose last two dims
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    right_t = right_perm.permute(0, 1, 3, 2).contiguous()

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat)  # (B*H, N, N)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # -------------------------------------------------
    # Output layer norm, gating and projection
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass with a Triton kernel for the mask application.
The heavy O(N³) reduction is performed via a batched matrix multiplication
(torch.bmm), while the mask‑multiply step is fused in a custom Triton kernel.
All other operations (LayerNorm, Linear projections, gating, final projection)
use PyTorch for simplicity and correctness.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _mask_mul_kernel(
    left_ptr,            # *float32
    right_ptr,           # *float32
    mask_ptr,            # *float32
    total_elements,      # int64 total number of elements in left/right
    hidden_dim,          # int64 size of the hidden dimension
    BLOCK: tl.constexpr, # compile‑time block size
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # map flat offset to mask index (ignore hidden dimension)
    mask_idx = offs // hidden_dim

    m = tl.load(mask_ptr + mask_idx, mask=mask)               # mask value (0 or 1)
    l = tl.load(left_ptr + offs, mask=mask)                  # left value
    r = tl.load(right_ptr + offs, mask=mask)                 # right value

    l = l * m
    r = r * m

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


def _apply_mask_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """In‑place mask multiplication using Triton.  left/right: [B,N,N,H], mask: [B,N,N]."""
    if mask is None:
        return
    # Ensure contiguous tensors and matching dtype
    left = left.contiguous()
    right = right.contiguous()
    mask = mask.contiguous().to(left.dtype)

    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024  # reasonable block size for 32‑bit values
    grid = ( (total + BLOCK - 1) // BLOCK, )
    _mask_mul_kernel[grid](
        left,
        right,
        mask,
        total,
        H,
        BLOCK=BLOCK,
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul module (outgoing variant).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with at least "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) Input layer‑norm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Apply optional mask (Triton kernel)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        _apply_mask_triton(left, right, mask)

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise gating
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core reduction: out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
    #    Implemented as a batched matrix multiply (bmm) over hidden_dim.
    # -----------------------------------------------------------------
    # Cast to half‑precision for the heavy matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape for batched matmul
    # left: [B,N,N,H] -> [B,H,N,N]   (i,k)   last two dims are (i,k)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    # right: same permutation then transpose last two dims to get (k,j)
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    right_t = right_perm.permute(0, 1, 3, 2).contiguous()

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)      # (B*H, N, N) with axes (i,k)
    right_flat = right_t.view(B * H, N, N)       # (B*H, N, N) with axes (k,j)

    # batched matrix multiply
    out_flat = torch.bmm(left_flat, right_flat)  # (B*H, N, N) -> (i,j) per hidden dim
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # back to [B,N,N,H]
    out = out.to(torch.float32)  # cast back for the following float‑32 ops

    # -----------------------------------------------------------------
    # 6) Final LayerNorm, gating, and projection back to `dim`
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
