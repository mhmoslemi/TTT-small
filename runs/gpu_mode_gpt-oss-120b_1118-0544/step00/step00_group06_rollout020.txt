def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            (dim,),
                                            weight=norm_weight,
                                            bias=norm_bias)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Masks
    if mask is not None:
        mask = mask.to(left_proj.dtype).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Convert to fp16 for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Prepare output tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # Triton kernel launch
    import triton
    import triton.language as tl

    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute strides
    ls = left.stride()
    rs = right.stride()
    os = out.stride()

    # Launch kernel with grid
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)
    trimul_matmul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        hidden_dim=hidden_dim,
        stride_l_b=ls[0],
        stride_l_i=ls[1],
        stride_l_k=ls[2],
        stride_l_d=ls[3],
        stride_r_b=rs[0],
        stride_r_i=rs[1],
        stride_r_j=rs[2],
        stride_r_d=rs[3],
        stride_o_b=os[0],
        stride_o_i=os[1],
        stride_o_j=os[2],
        stride_o_d=os[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_stages=2,
        num_warps=4,
    )
    # after kernel, out has shape (B, N, N, hidden_dim) (float32)

    # Apply to_out_norm and out_gate, final projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out,
                                              (hidden_dim,),
                                              weight=to_out_norm_weight,
                                              bias=to_out_norm_bias)

    out = out_norm * out_gate

    # Final linear to dim
    to_out_weight = weights["to_out.weight"]
    out_final = torch.nn.functional.linear(out, to_out_weight)  # shape (B,N,N,dim)

    return out_final

"""
TriMul forward (outgoing) fused kernel.

The kernel computes the core Tensor contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
which is equivalent to a batched matrix multiplication
    out = left @ rightᵀ
for each hidden channel *d*.  The surrounding LayerNorms,
Linear projections and gating are performed with PyTorch,
while the O(N³·C) contraction is executed by a custom Triton
kernel (blocked GEMM).  The implementation works for any batch
size (1‑2), sequence length up to 1024 and hidden dimensions
up to 384.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_j, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM for the TriMul contraction.

    left  shape: (B, N, N, hidden_dim)   – indexed as (b, i, k, d)
    right shape: (B, N, N, hidden_dim)   – indexed as (b, j, k, d)
    The kernel treats the k‑dimension as the reduction axis and
    computes out[b, i, j, d] = Σₖ left[b,i,k,d] * right[b,j,k,d].
    """
    pid_m = tl.program_id(0)            # tile row   index (i)
    pid_n = tl.program_id(1)            # tile col   index (j)
    pid_h = tl.program_id(2)            # batch * hidden channel index

    # Decompose pid_h into batch and hidden channel
    b = pid_h // hidden_dim
    d = pid_h % hidden_dim

    # Offsets for the M and N dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---------- load left[i, k, d] ----------
        # left[b, i, k, d] with strides (b,i,k,d)
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + d * stride_l_d
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # ---------- load right_T[k, j, d] (i.e. right[j, k, d]) ----------
        # We treat the second N‑axis of right as the column (j) and the
        # third N‑axis as the reduction axis (k).
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + d * stride_r_d
            + offs_k[:, None] * stride_r_k   # k dimension
            + offs_j[None, :] * stride_r_j   # j dimension
        )
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # Cast to FP32 for the dot product (Tensor‑core friendly)
        left = left.to(tl.float32)
        right = right.to(tl.float32)

        # Accumulate the matrix product block
        acc += tl.dot(left, right)

    # ---------- store the result ----------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + d * stride_o_d
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, [B, N, N, dim]
        - mask         : torch.Tensor or None, [B, N, N]
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict with keys "dim", "hidden_dim"
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    import torch.nn.functional as F

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ---------- LayerNorm on the input ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections ----------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ---------- Apply mask (if provided) ----------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ---------- Convert to FP16 for the heavy contraction ----------
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # ---------- Prepare strides for Triton ----------
    ls = left.stride()   # (B, i, k, d)
    rs = right.stride()
    os = out.stride()    # (B, i, j, d)

    # Block sizes (tuned for B200, keep them modest)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    trimul_matmul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        hidden_dim=hidden_dim,
        stride_l_b=ls[0],
        stride_l_i=ls[1],
        stride_l_k=ls[2],
        stride_l_d=ls[3],
        stride_r_b=rs[0],
        stride_r_j=rs[1],           # original second N‑axis (j)
        stride_r_k=rs[2],           # original third N‑axis (k)
        stride_r_d=rs[3],
        stride_o_b=os[0],
        stride_o_i=os[1],
        stride_o_j=os[2],
        stride_o_d=os[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )

    # ---------- Second LayerNorm and output gating ----------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out_norm * out_gate

    # ---------- Final projection back to dim ----------
    out_final = F.linear(out, weights["to_out.weight"])

    return out_final
