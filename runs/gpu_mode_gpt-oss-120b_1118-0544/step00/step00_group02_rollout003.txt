@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0)  # combined batch*hidden dimension
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    # compute start indices
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    i_mask = i_off < N
    j_mask = j_off < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # Compute offsets for left and right blocks
        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + i_off[:,None]*stride_left_i + k_off[None,:]*stride_left_k
        right_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + j_off[:,None]*stride_right_j + k_off[None,:]*stride_right_k

        # load, applying mask
        left_vals = tl.load(left_ptrs, mask=i_mask[:,None] & k_mask[None,:], other=0.0).to(tl.float32)
        right_vals = tl.load(right_ptrs, mask=j_mask[:,None] & k_mask[None,:], other=0.0).to(tl.float32)

        acc += tl.dot(left_vals, right_vals)   # shape (BLOCK_M, BLOCK_N)
    
    # Store result
    out_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + i_off[:,None]*stride_out_i + j_off[None,:]*stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16) if out_ptr.dtype == tl.float16 else acc, mask=i_mask[:,None] & j_mask[None,:])
```

"""
TriMul "outgoing" update implemented with a Triton kernel for the heavy
einsum reduction.

The kernel computes, for each batch `b` and hidden channel `h`:

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

where `left` and `right` are the gated, masked projections of the input.
The reduction over `k` is performed as a batched GEMM using tiling.
All other operations (LayerNorm, linear projections, gating, final
normalisation and projection) are done in PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    left_ptr,            # *float32 left  [B, N, N, H]
    right_ptr,           # *float32 right [B, N, N, H]
    out_ptr,             # *float32 out   [B, N, N, H]
    B, N, H,             # int32 scalar dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)   # combined batch * hidden
    pid1 = tl.program_id(1)   # block row (i)
    pid2 = tl.program_id(2)   # block col (j)

    # Decode batch and hidden index
    b = pid0 // H
    h = pid0 % H

    # ------------------------------
    # offsets for the M (i) and N (j) dimensions
    # ------------------------------
    i_off = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = (N + BLOCK_K - 1) // BLOCK_K

    for kk in range(num_k):
        k_off = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # left: shape (BLOCK_M, BLOCK_K) -> indices (i, k)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + i_off[:, None] * stride_left_i
            + k_off[None, :] * stride_left_k
        )
        left_vals = tl.load(
            left_ptrs,
            mask=(i_mask[:, None] & k_mask[None, :]),
            other=0.0,
        ).to(tl.float32)

        # right: shape (BLOCK_K, BLOCK_N) -> indices (k, j)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + k_off[:, None] * stride_right_k
            + j_off[None, :] * stride_right_i
        )
        right_vals = tl.load(
            right_ptrs,
            mask=(k_mask[:, None] & j_mask[None, :]),
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(left_vals, right_vals)   # (M,K)·(K,N) → (M,N)

    # store the result
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + i_off[:, None] * stride_out_i
        + j_off[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


# --------------------------------------------------------------------------- #
# Entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul "outgoing" operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (may be ignored)
        - weights      : dict of torch.Tensors with the model parameters
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------- #
    # unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # LayerNorm on the input (dim axis)
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------- #
    # Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left_proj_weight  = weights["left_proj.weight"]   # (H, dim)
    right_proj_weight = weights["right_proj.weight"]  # (H, dim)

    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_weight) # (B,N,N,H)

    # ------------------------------------------------------------------- #
    # Apply mask if present
    # ------------------------------------------------------------------- #
    if not config.get("nomask", True):
        # mask: (B,N,N) -> (B,N,N,1)
        mask_f = mask.to(dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------- #
    # Gating (sigmoid) and gated multiplication
    # ------------------------------------------------------------------- #
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # Ensure contiguity for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------- #
    # Triton kernel for the einsum reduction
    # ------------------------------------------------------------------- #
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tunable tile sizes – 64 works well for the given hardware & sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,            # batch * hidden
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
    )

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # Post‑processing: to_out_norm, gating, final linear projection
    # ------------------------------------------------------------------- #
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]   # (dim, hidden_dim)

    # LayerNorm on hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to original dim
    out = torch.nn.functional.linear(out, to_out_weight)   # (B,N,N,dim)

    return out
