@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # decode batch and hidden
    b = pid_bh // H
    h = pid_bh % H

    # base pointers for this batch/h
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # tile ranges
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    # Create masks
    mask_i = i_range < N
    mask_j = j_range < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kb * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # Load A tile (i,k)
        a_ptrs = left_base + i_range[:, None] * stride_left_i + k_range[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B_T tile (k,j) - note we need to index right[j,k]
        b_ptrs = right_base + k_range[:, None] * stride_right_k + j_range[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Convert to float32 (if they are not)
        acc += tl.dot(a, b)

    # Store result
    out_ptrs = out_base + i_range[:, None] * stride_out_i + j_range[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    # Expect D == dim.

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # confirm shapes matches.

    # get weight tensors
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    import torch.nn.functional as F

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left = F.linear(x, left_proj_weight)   # shape B,N,N,hidden
    right = F.linear(x, right_proj_weight)

    # Apply mask if provided
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(input_tensor.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    left = left.contiguous()

    H_val = hidden_dim

    BLOCK_K = 64

"""
TriMul forward implementation with a fused Triton kernel.

The heavy O(N³·H) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed by a custom Triton kernel that tiles the i‑j axes
(Block_M×Block_N) and iterates over k (Block_K).  The kernel is
batched over the product of batch and hidden dimension (B·H) so that each
(hidden-channel, batch) pair is computed independently, avoiding any
synchronisation across channels.

All other operations (LayerNorms, Linear projections, sigmoids,
masking, final LayerNorm and output Linear) are executed in PyTorch.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    # Pointers
    left_ptr, right_ptr, out_ptr,
    # Compile‑time constants
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    # Strides (in elements)
    stride_left_b: tl.int64, stride_left_i: tl.int64, stride_left_k: tl.int64, stride_left_h: tl.int64,
    stride_right_b: tl.int64, stride_right_i: tl.int64, stride_right_k: tl.int64, stride_right_h: tl.int64,
    stride_out_b: tl.int64, stride_out_i: tl.int64, stride_out_j: tl.int64, stride_out_h: tl.int64,
    # Tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)               # tile index over i
    pid_j = tl.program_id(1)               # tile index over j
    pid_bh = tl.program_id(2)              # combined batch * hidden index

    # Decode batch and hidden channel
    b = pid_bh // H
    h = pid_bh % H

    # Base pointers for this (b, h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Start indices of the tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Ranges
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    # Boundary masks for i/j
    mask_i = i_range < N
    mask_j = j_range < N

    # Accumulator for the C tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k):
        k_start = kk * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # Load A tile: shape (BLOCK_M, BLOCK_K) → left[b, i, k, h]
        a_ptrs = left_base + i_range[:, None] * stride_left_i + k_range[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile transposed: shape (BLOCK_K, BLOCK_N) → right[b, j, k, h]
        # Access pattern right[j, k] = right_base + k*stride_k + j*stride_i
        b_ptrs = right_base + k_range[:, None] * stride_right_k + j_range[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Matrix multiplication for this k‑tile
        acc += tl.dot(a, b)

    # Store the result tile
    out_ptrs = out_base + i_range[:, None] * stride_out_i + j_range[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, dim]
            - mask: torch.Tensor or None, shape [B, N, N]
            - weights: dict of weight tensors
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor[B, N, N, dim] (float32, on the same device as input)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Load model parameters onto the correct device/dtype
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=dtype)
    norm_bias   = weights["norm.bias"].to(device=device, dtype=dtype)

    left_proj_weight  = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_weight = weights["right_proj.weight"].to(device=device, dtype=dtype)

    left_gate_weight  = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_weight = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_weight   = weights["out_gate.weight"].to(device=device, dtype=dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device=device, dtype=dtype)

    to_out_weight = weights["to_out.weight"].to(device=device, dtype=dtype)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on the hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask = mask.to(dtype=dtype, device=device).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear on the normalized input)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Fuse the O(N³) contraction with a Triton kernel
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # shape [B, N, N, hidden]

    # Strides in elements (not bytes)
    s_b_l, s_i_l, s_k_l, s_h_l = left.stride()
    s_b_r, s_i_r, s_k_r, s_h_r = right.stride()
    s_b_o, s_i_o, s_j_o, s_h_o = out.stride()

    # Tile sizes (tuned for typical B200 memory hierarchy)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_bh = B * hidden_dim

    # Launch the kernel
    tri_mul_kernel[(grid_i, grid_j, grid_bh)](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        s_b_l, s_i_l, s_k_l, s_h_l,
        s_b_r, s_i_r, s_k_r, s_h_r,
        s_b_o, s_i_o, s_j_o, s_h_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm, output gate and final Linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    output = F.linear(out, to_out_weight)   # final shape [B, N, N, dim]

    return output
