    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_l_b, stride_l_i, stride_l_k,
    stride_r_b, stride_r_j, stride_r_k,
    stride_o_b, stride_o_i, stride_o_j,

    num_pid_n = tl.cdiv(N, BLOCK_N)

"""
TriMul forward (outgoing) with a Triton kernel for the heavy O(N³) reduction.

The Triton kernel computes, for each hidden channel d,
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]    (i,j,k ∈ [0, N))

i and j are tiled with BLOCK_M × BLOCK_N threads, k is reduced in
BLOCK_K‑sized chunks. The kernel is launched once per hidden channel
(d) – this keeps the register pressure low while still providing a
substantial speed‑up over a pure‑Python / torch.einsum implementation.
All remaining operations (layer‑norms, linear projections, gating) are
performed with PyTorch.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def triton_trimul_kernel(
    left_ptr,   # const void*  (B,N,N,H)
    right_ptr,  # const void*  (B,N,N,H)
    out_ptr,    #       void*  (B,N,N,H)
    B: tl.int64,
    N: tl.int64,
    # strides for left
    stride_l_b: tl.int64,
    stride_l_i: tl.int64,
    stride_l_k: tl.int64,
    stride_l_d: tl.int64,
    # strides for right
    stride_r_b: tl.int64,
    stride_r_j: tl.int64,
    stride_r_k: tl.int64,
    stride_r_d: tl.int64,
    # strides for out
    stride_o_b: tl.int64,
    stride_o_i: tl.int64,
    stride_o_j: tl.int64,
    stride_o_d: tl.int64,
    # compile‑time constants
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    D_IDX: tl.constexpr,  # hidden‑channel index processed by this launch
):
    pid_b = tl.program_id(2)          # batch dimension
    pid_i = tl.program_id(0)          # tile row
    pid_j = tl.program_id(1)          # tile col

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers (point to the beginning of the tile, without the k offset)
    a_base = (
        left_ptr
        + pid_b * stride_l_b
        + offs_i[:, None] * stride_l_i
        + D_IDX * stride_l_d
    )
    b_base = (
        right_ptr
        + pid_b * stride_r_b
        + offs_j[None, :] * stride_r_j
        + D_IDX * stride_r_d
    )

    # accumulate in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tile in range(num_k_tiles):
        cur_k = tile * BLOCK_K + offs_k
        k_mask = cur_k < N

        # load A[i, k]   -> shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_base + cur_k[None, :] * stride_l_k,
            mask=mask_i[:, None] & k_mask[None, :],
            other=0.0,
        )
        # load B[k, j]   -> shape (BLOCK_K, BLOCK_N)  (note the transposed view)
        b = tl.load(
            b_base + cur_k[:, None] * stride_r_k,
            mask=k_mask[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # store result
    c_ptr = (
        out_ptr
        + pid_b * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + D_IDX * stride_o_d
    )
    tl.store(
        c_ptr,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor or None, shape [B, N, N]
        - weights_dict : dict of torch.Tensor, contains all model weights
        - config_dict  : dict, must contain "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1. unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, N2, _ = input_tensor.shape
    assert N == N2, "input must be square in the sequence dimensions"

    # ------------------------------------------------------------------
    # 2. LayerNorm on the input (no bias → bias can be None)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 4. Optional masking
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5. Gating
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core TriMul contraction using Triton
    # ------------------------------------------------------------------
    # Ensure contiguous memory (required for correct stride handling)
    left = left.contiguous()
    right = right.contiguous()
    core = torch.empty_like(left)   # will hold [B,N,N,hidden_dim]

    # Strides (int64)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_d = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = core.stride()

    # Blocking parameters (tuned for typical B200 memory hierarchy)
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B,                     # batch dimension
    )

    # Launch one kernel per hidden channel
    for d in range(hidden_dim):
        triton_trimul_kernel[grid](
            left_ptr=left.data_ptr(),
            right_ptr=right.data_ptr(),
            out_ptr=core.data_ptr(),
            B=B,
            N=N,
            stride_l_b=int(stride_l_b),
            stride_l_i=int(stride_l_i),
            stride_l_k=int(stride_l_k),
            stride_l_d=int(stride_l_d),
            stride_r_b=int(stride_r_b),
            stride_r_j=int(stride_r_j),
            stride_r_k=int(stride_r_k),
            stride_r_d=int(stride_r_d),
            stride_o_b=int(stride_o_b),
            stride_o_i=int(stride_o_i),
            stride_o_j=int(stride_o_j),
            stride_o_d=int(stride_o_d),
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            D_IDX=d,
        )
    # Synchronize to make sure the kernel is finished before proceeding
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7. Post‑processing: norm → gate → final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    core = torch.nn.functional.layer_norm(
        core,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    core = core * out_gate  # apply output gate

    to_out_weight = weights["to_out.weight"]      # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(core, to_out_weight)   # final projection

    return out
