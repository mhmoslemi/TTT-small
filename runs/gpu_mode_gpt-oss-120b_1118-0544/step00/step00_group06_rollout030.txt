```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute indexes
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_i * num_j)
    pid_rem = pid % (num_i * num_j)
    block_i = pid_rem // num_j
    block_j = pid_rem % num_j

    # offsets within the block
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = tl.arange(0, BLOCK_D)
    # masks for boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < BLOCK_D   # always true but keep for consistency

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over k dimension tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointers
        left_ptrs = left_ptr \
            + batch * stride_left_b \
            + offs_i[:, None, None] * stride_left_i \
            + offs_k[None, :, None] * stride_left_k \
            + offs_d[None, None, :] * stride_left_d

        right_ptrs = right_ptr \
            + batch * stride_right_b \
            + offs_j[:, None, None] * stride_right_j \
            + offs_k[None, :, None] * stride_right_k \
            + offs_d[None, None, :] * stride_right_d

        # load tiles
        left_tile = tl.load(left_ptrs,
            mask=(mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]),
            other=0.0)
        right_tile = tl.load(right_ptrs,
            mask=(mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]),
            other=0.0)

        # inner loop over k within tile
        for k_inner in range(0, BLOCK_K):
            # Because values outside actual N are zeroed by mask, we can ignore k validity
            l = left_tile[:, k_inner, :]   # (M, D)
            r = right_tile[:, k_inner, :]  # (N, D)
            # broadcast multiply
            acc += l[:, None, :] * r[None, :, :]   # (M, N, D)

    # write back
    out_ptrs = out_ptr \
        + batch * stride_out_b \
        + offs_i[:, None, None] * stride_out_i \
        + offs_j[None, :, None] * stride_out_j \
        + offs_d[None, None, :] * stride_out_d

    tl.store(out_ptrs,
        acc,
        mask=(mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]))
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Ensure tensors are on GPU
    device = input_tensor.device
    # LayerNorm on dim
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Mask handling
    if mask is None:
        mask = torch.ones((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]), dtype=input_tensor.dtype, device=device)
    mask_exp = mask.unsqueeze(-1)  # (B, N, N, 1)
    left = left * mask_exp
    right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape

    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # compute strides
    stride_left = left_fp16.stride()
    stride_right = right_fp16.stride()
    stride_out = out_hidden.stride()

    # block sizes
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64
    BLOCK_D_MAX = 32

    # Grid dimensions
    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * num_i * num_j,)

    # Process hidden dimension in chunks
    for d_start in range(0, hidden_dim, BLOCK_D_MAX):
        d_end = min(d_start + BLOCK_D_MAX, hidden_dim)
        d_size = d_end - d_start

        left_sub = left_fp16[..., d_start:d_end]   # shape B,N,N,d_size
        right_sub = right_fp16[..., d_start:d_end]
        out_sub = out_hidden[..., d_start:d_end]

        trimul_kernel[grid](
            left_sub,
            right_sub,
            out_sub,
            B,

            right_sub.stride(0), right_sub.stride(1), right_sub.stride(2), right_sub.stride(3),
            out_sub.stride(0), out_sub.stride(1), out_sub.stride(2), out_sub.stride(3),
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=d_size,

    out = out_hidden_norm * out_gate

    # Final linear projection to dim

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

The kernel computes, for each batch `b`, feature `d` and pair of positions
`i, j`:

    out[b, i, j, d] = sum_{k=0}^{N-1} left[b, i, k, d] * right[b, j, k, d]

The surrounding operations (LayerNorm, Linear projections, sigmoid gates,
masking, and final projection) are performed with PyTorch.  The heavy
O(N³·C) contraction is off‑loaded to Triton for speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr,                # *fp16  (B,N,N,D_block)
    right_ptr,               # *fp16
    out_ptr,                 # *fp32  (B,N,N,D_block)
    B, N,                    # runtime sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,   # tile size for i dimension
    BLOCK_N: tl.constexpr,   # tile size for j dimension
    BLOCK_K: tl.constexpr,   # tile size for reduction k
    BLOCK_D: tl.constexpr,   # tile size for hidden dimension (D_block)
):
    pid = tl.program_id(0)
    # grid layout: (batch, block_i, block_j)
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    batch = pid // (num_i * num_j)
    pid_rem = pid % (num_i * num_j)
    block_i = pid_rem // num_j
    block_j = pid_rem % num_j

    # Offsets for the current tile
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < BLOCK_D  # always true but keeps shape broadcastable

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over k dimension in tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers to the current k‑tile (i,k,d) and (j,k,d)
        left_ptrs = left_ptr + \
            batch * stride_left_b + \
            offs_i[:, None, None] * stride_left_i + \
            offs_k[None, :, None] * stride_left_k + \
            offs_d[None, None, :] * stride_left_d

        right_ptrs = right_ptr + \
            batch * stride_right_b + \
            offs_j[:, None, None] * stride_right_i + \
            offs_k[None, :, None] * stride_right_k + \
            offs_d[None, None, :] * stride_right_d

        # Load tiles (zero for out‑of‑bounds locations)
        left_tile = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )
        right_tile = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # Accumulate the outer product over the K dimension inside the tile
        # (M,N,D) += sum_k left[i,k,d] * right[j,k,d]
        for k_inner in range(0, BLOCK_K):
            l = left_tile[:, k_inner, :]          # (M, D)
            r = right_tile[:, k_inner, :]         # (N, D)
            # broadcasting gives (M, N, D)
            acc += l[:, None, :] * r[None, :, :]

    # Write the result
    out_ptrs = out_ptr + \
        batch * stride_out_b + \
        offs_i[:, None, None] * stride_out_i + \
        offs_j[None, :, None] * stride_out_j + \
        offs_d[None, None, :] * stride_out_d

    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


def custom_kernel(data):
    """
    Entry point for the TriMul forward pass.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights: dict with all Linear / LayerNorm parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- 1. Input LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ---------- 2. Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- 3. Optional mask ----------
    if mask is None:
        mask = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=input_tensor.dtype,
            device=input_tensor.device,
        )
    mask_exp = mask.unsqueeze(-1)  # (B,N,N,1)
    left = left * mask_exp
    right = right * mask_exp

    # ---------- 4. Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- 5. Core O(N³·C) contraction (Triton) ----------
    # Cast to fp16 for bandwidth, accumulate in fp32
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape
    # Output tensor (float32)
    out_hidden = torch.empty(
        (B, N, N, hidden_dim), dtype=torch.float32, device=left_fp16.device
    )

    # Strides (in elements, not bytes)
    stride_l = left_fp16.stride()    # (b,i,k,d)
    stride_r = right_fp16.stride()
    stride_o = out_hidden.stride()

    # Tile sizes (tuned for typical B200 memory hierarchy)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64
    BLOCK_D_MAX = 32  # max number of hidden channels per kernel launch

    # Grid: one program for each (batch, i‑tile, j‑tile)
    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * num_i * num_j,)

    # Process hidden channels in chunks of BLOCK_D_MAX
    for d_start in range(0, hidden_dim, BLOCK_D_MAX):
        d_end = min(d_start + BLOCK_D_MAX, hidden_dim)
        d_size = d_end - d_start

        left_sub = left_fp16[..., d_start:d_end]
        right_sub = right_fp16[..., d_start:d_end]
        out_sub = out_hidden[..., d_start:d_end]

        trimul_kernel[
            grid
        ](
            left_sub,
            right_sub,
            out_sub,
            B,
            N,
            # left strides
            stride_l[0],
            stride_l[1],
            stride_l[2],
            stride_l[3],
            # right strides
            stride_r[0],
            stride_r[1],
            stride_r[2],
            stride_r[3],
            # out strides
            stride_o[0],
            stride_o[1],
            stride_o[2],
            stride_o[3],
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=d_size,
        )
    # Ensure kernel completion before next ops
    torch.cuda.synchronize()

    # ---------- 6. Output LayerNorm ----------
    out_hidden_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ---------- 7. Output gating ----------
    out = out_hidden_norm * out_gate

    # ---------- 8. Final linear projection ----------
    output = F.linear(out, weights["to_out.weight"])

    return output
