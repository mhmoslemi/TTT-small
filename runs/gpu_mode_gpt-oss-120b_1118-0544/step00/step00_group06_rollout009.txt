"""
TriMul forward (outgoing) implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor X ∈ ℝ^{B×N×N×C}.
2. Linear projections (no bias):
   L = X·W_left_projᵀ      # shape (B,N,N,H)
   R = X·W_right_projᵀ
3. Optional mask M ∈ {0,1}^{B×N×N}:  L,R ← L,R ⊙ M[:, :, :, None]
4. Gating:
   gL = σ(X·W_left_gateᵀ)   ;   L ← L ⊙ gL
   gR = σ(X·W_right_gateᵀ)  ;   R ← R ⊙ gR
   gO = σ(X·W_out_gateᵀ)    # shape (B,N,N,H)
5. Core contraction (the O(N³) part) :
   out[b,i,j,h] = Σ_k L[b,i,k,h] * R[b,j,k,h]
   This is a batched matrix multiplication: for each batch‑gate pair
   we compute L_h @ R_hᵀ.  The computation is performed by a custom Triton
   kernel that tiles the output (M×N) and iterates over the reduction axis K.
6. Layer‑norm over the hidden dimension H, apply the output gate and a final
   linear projection back to C channels.

Only the expensive contraction (step 5) is written in Triton; all other
operations use standard PyTorch kernels for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    N,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiply C = A @ Bᵀ where A,B have shape (batch, N, N)."""
    batch_id = tl.program_id(0)
    pid_i = tl.program_id(1)   # block row
    pid_j = tl.program_id(2)   # block col

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointer to the output tile
    out_ptrs = (C_ptr + batch_id * stride_c_batch
                + offs_i[:, None] * stride_c_row
                + offs_j[None, :] * stride_c_col)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A[i, k]
        a_ptrs = (A_ptr + batch_id * stride_a_batch
                  + offs_i[:, None] * stride_a_row
                  + offs_k[None, :] * stride_a_col)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load Bᵀ[k, j]  (i.e. B[j, k])
        b_ptrs = (B_ptr + batch_id * stride_b_batch
                  + offs_k[:, None] * stride_b_col
                  + offs_j[None, :] * stride_b_row)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M×BLOCK_K)·(BLOCK_K×BLOCK_N) → (BLOCK_M×BLOCK_N)

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the contraction out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a Triton kernel.  left/right shape: (B,N,N,H).
    Returns out with the same shape.
    """
    B, N, _, H = left.shape
    # (B, H, N, N) → (batch, N, N) where batch = B*H
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(-1, N, N)
    right_flat = right_perm.view(-1, N, N)
    batch = left_flat.shape[0]

    out_flat = torch.empty_like(left_flat)

    # Strides (in elements, not bytes) for the flattened tensors
    stride_a_batch, stride_a_row, stride_a_col = left_flat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_flat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_flat.stride()

    # Tile sizes – tuned for typical B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4  # a reasonable default
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, C]
            - mask: torch.Tensor or None, shape [B, N, N]
            - weights: dict of torch.Tensors with keys matching the PyTorch
              reference implementation
            - config: dict with at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    x, mask, weights, config = data
    B, N, _, C = x.shape
    H = config["hidden_dim"]   # hidden dimension (often 128)

    # ---------- LayerNorm on input ----------
    x = F.layer_norm(
        x,
        (C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if mask is not None:
        m = mask.to(left.dtype).unsqueeze(-1)       # (B,N,N,1)
        left = left * m
        right = right * m

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # (B,N,N,H)

    left = left * left_gate
    right = right * right_gate

    # ---------- Core contraction (Triton) ----------
    out = _tri_mul_triton(left, right)   # (B,N,N,H)

    # ---------- Output normalization, gate, final linear ----------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
