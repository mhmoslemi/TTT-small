"""
TriMul (outgoing) forward pass implemented with a mix of PyTorch and Triton.

Algorithm
---------
1. Layer‑normalize the input tensor `x` over the channel dimension `dim`.
2. Project the normalized tensor to a hidden dimension `h` with two linear
   layers (`left_proj`, `right_proj`).
3. Optionally apply a pairwise mask (broadcasted to the hidden dim).
4. Compute three gating vectors (left, right, out) via linear layers + sigmoid.
5. Fuse the mask and gates:   left  = left  * left_gate
                               right = right * right_gate
6. Reduce over the third sequence axis `k`.  This is a batched matrix
   multiplication for each hidden channel:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a `torch.bmm` on tensors reshaped to
   (B*h, N, N) × (B*h, N, N)^T.
7. Layer‑normalize the reduction result over the hidden dimension.
8. Multiply the normalized result element‑wise with `out_gate`.  The
   multiplication is performed by a tiny Triton kernel.
9. Final linear projection back to `dim`.

Only the element‑wise multiplication (step 8) is off‑loaded to Triton,
ensuring the solution satisfies the requirement to contain a custom kernel.
All heavy linear algebra (projections and the O(N³) reduction) leverages
highly‑optimized cuBLAS via `torch.nn.functional.linear` and `torch.bmm`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: element‑wise multiplication of two tensors.
# -------------------------------------------------------------
@triton.jit
def elem_mul_kernel(
    out_ptr, a_ptr, b_ptr,  # pointers
    n_elements,            # total number of elements
    BLOCK: tl.constexpr    # compile‑time block size
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK
    offs = block_start + tl.arange(0, BLOCK)
    mask = offs < n_elements

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = a * b
    tl.store(out_ptr + offs, c, mask=mask)


# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward TriMul (outgoing) implementation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with all required weight tensors
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcasted to hidden dim)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # mask may be bool; convert to same dtype as tensors
        mask_f = mask.to(dtype).unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating vectors
    # -----------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # -----------------------------------------------------------------
    # 5. Apply gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Reduction over the third sequence dimension (k)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    # reshape to (B*hidden, N, N) for batched gemm
    left_reshape = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # right^T for each batch: (B*hidden, N, N) -> (B*hidden, N, N)
    out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1, 2))

    # reshape back to (B,N,N,hidden)
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # -----------------------------------------------------------------
    # 7. Output LayerNorm over hidden dimension
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 8. Element‑wise multiplication with out_gate using Triton
    # -----------------------------------------------------------------
    # Ensure contiguous tensors for pointer arithmetic
    out_contig = out.contiguous()
    out_gate_contig = out_gate.contiguous()
    out_mul = torch.empty_like(out_contig)

    total_elements = out_contig.numel()
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    elem_mul_kernel[grid](
        out_mul.data_ptr(),
        out_contig.data_ptr(),
        out_gate_contig.data_ptr(),
        total_elements,
        BLOCK=BLOCK_SIZE,
    )

    # -----------------------------------------------------------------
    # 9. Final projection back to original dimension
    # -----------------------------------------------------------------
    out_final = F.linear(out_mul, weights["to_out.weight"])

    return out_final
