TriMul (outgoing) implementation with a small Triton kernel.

The kernel fuses the element‑wise mask, left/right gates and the linear
projections (already computed) into the final left/right tensors that are
used by the expensive `einsum` core.  The rest of the forward pass (layer‑norm,
gates, the N³ reduction and the final linear) is performed with native PyTorch
operations for simplicity and correctness.

The kernel processes one (batch, i, j) entry per program and iterates over the
hidden dimension in blocks of `BLOCK_H`.  It multiplies:

    left_out = left_proj * left_gate * mask
    right_out = right_proj * right_gate * mask

mask is optional; when `nomask` is True the mask is ignored (treated as all‑ones).

The overall workflow:
1. Layer‑norm the input.
2. Linear projections for left/right and their gates (torch kernels).
3. Apply mask & gates with the Triton kernel.
4. Compute the O(N³) reduction via `torch.einsum`.
5. Layer‑norm, gate and final linear to obtain the output.
"""

    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,

    stride_out_right_b, stride_out_right_i, stride_out_right_j, stride_out_right_h,
    # Compile‑time constants
    BLOCK_H: tl.constexpr,
    HAVE_MASK: tl.constexpr,
):

    # Compute the (b, i, j) coordinates for this program.
    # --------------------------------------------------------------------- #
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    rem = pid % (N * N)

    j = rem % N

            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model weight tensors
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ----------------------- unpack inputs ----------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------- LayerNorm -------------------------- #
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ----------------------- Linear projections ------------------- #
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ----------------------- Gates (pre‑sigmoid) ----------------- #
    left_gate_raw = torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    right_gate_raw = torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    out_gate_raw = torch.nn.functional.linear(x_norm, weights["out_gate.weight"])

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)           # [B, N, N, hidden_dim]

    # ----------------------- Prepare mask ------------------------ #
    B, N, _, _ = input_tensor.shape
    if nomask or mask is None:
        have_mask = 0
        # dummy mask tensor (won’t be read because HAVE_MASK=0)
        mask_tensor = torch.empty((0,), device=device, dtype=input_tensor.dtype)
    else:
        have_mask = 1
        mask_tensor = mask.to(dtype=input_tensor.dtype, device=device)

    # ----------------------- Allocate output buffers ------------ #
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    # ----------------------- Compute strides --------------------- #
    # Input / output tensors are contiguous in the default layout:
    # [B, N, N, H]
    stride_left_b, stride_left_i, stride_left_j, stride_left_h = left_proj.stride()
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h = left_gate.stride()
    stride_right_b, stride_right_i, stride_right_j, stride_right_h = right_proj.stride()
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h = right_gate.stride()
    stride_mask_b, stride_mask_i, stride_mask_j = mask_tensor.stride() if have_mask else (0, 0, 0)

    stride_out_left_b, stride_out_left_i, stride_out_left_j, stride_out_left_h = left_out.stride()
    stride_out_right_b, stride_out_right_i, stride_out_right_j, stride_out_right_h = right_out.stride()

    # ----------------------- Launch Triton kernel -------------- #
    total_elements = B * N * N
    grid = (total_elements,)

    BLOCK_H = 32  # reasonable block size for the hidden dimension

    apply_mask_gate_kernel[grid](
        left_proj, left_gate,
        right_proj, right_gate,
        mask_tensor,
        left_out, right_out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_j, stride_left_h,
        stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
        stride_right_b, stride_right_i, stride_right_j, stride_right_h,
        stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
        stride_mask_b, stride_mask_i, stride_mask_j,
        stride_out_left_b, stride_out_left_i, stride_out_left_j, stride_out_left_h,
        stride_out_right_b, stride_out_right_i, stride_out_right_j, stride_out_right_h,
        BLOCK_H=BLOCK_H,
        HAVE_MASK=have_mask,
        num_warps=4,
    )
    # ------------------------------------------------------------ #
    # Core N³ reduction (einsum)                                 #
    # ------------------------------------------------------------ #
    # left_out, right_out : [B, N, N, hidden_dim]
    out = torch.einsum('b i k d, b j k d -> b i j d', left_out, right_out)

    # ------------------------------------------------------------ #
    # Post‑processing: norm, gating and final linear              #
    # ------------------------------------------------------------ #
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_gated = out_norm * out_gate
    output = torch.nn.functional.linear(out_gated, weights["to_out.weight"])

    return output
