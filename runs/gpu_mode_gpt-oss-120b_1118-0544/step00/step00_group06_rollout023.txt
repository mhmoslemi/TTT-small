def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # layer norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # shape: (hidden_dim, dim)
    left = torch.nn.functional.linear(x_norm, left_proj_weight) # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    
    # mask
    if mask is not None:
        # mask shape (B,N,N) bool or float? Expect float
        mask_exp = mask.unsqueeze(-1).to(dtype)  # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # contraction using triton
    out_hidden = tri_mul_contraction(left, right, bs, N, hidden_dim)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_hidden_norm = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    # gating
    out_hidden_norm = out_hidden_norm * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out_hidden_norm, to_out_weight)  # (B,N,N,dim)
    return out
```

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    b = pid // H
    h = pid % H

    # Compute indices
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left load
        left_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right load (j dimension)
        right_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + offs_j[:, None] * stride_right_i + offs_k[None, :] * stride_right_k
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # right_T shape (BLOCK_K, BLOCK_N)
        right_T = tl.trans(right)
        acc += tl.dot(left, right_T)
    # End loop

    # Write out
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j

    custom Triton kernel that treats each hidden dimension as an independent matrix multiplication.
    All other operations (LayerNorm, Linear projections, gates) are performed with PyTorch functional
    calls.
    """

    hidden_dim = config["hidden_dim"]

    right_proj_weight = get("right_proj.weight")
    left_gate_weight = get("left_gate.weight")
    right_gate_weight = get("right_gate.weight")
    out_gate_weight = get("out_gate.weight")
    to_out_norm_weight = get("to_out_norm.weight")
    to_out_norm_bias = get("to_out_norm.bias")
    to_out_weight = get("to_out.weight")

    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)
    right = torch.nn.functional.linear(x, right_proj_weight)

    nomask = config.get("nomask", False)
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    )

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out_hidden, to_out_weight)

"""
TriMul (outgoing) forward pass with a Triton kernel for the heavy O(N³) contraction.
All lightweight operations (LayerNorm, Linear projections, gates) are performed with
PyTorch, while the core contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM using a custom Triton kernel.
"""
import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 3‑D grid:
    #   axis 0 -> (batch * hidden)   -> pid
    #   axis 1 -> i‑tile (rows)      -> pid_m
    #   axis 2 -> j‑tile (cols)      -> pid_n
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    b = pid // H                # batch index
    h = pid % H                 # hidden‑dim index

    # Offsets for the i and j tiles
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator for the inner product over k
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)  -> indices (i, k)
        left_ptrs = left_ptr + b * stride_left_b + h * stride_left_h \
                    + offs_i[:, None] * stride_left_i \
                    + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right block loaded as (BLOCK_K, BLOCK_N) i.e. (k, j)
        # This layout permits a direct tl.dot with left.
        right_ptrs = right_ptr + b * stride_right_b + h * stride_right_h \
                     + offs_k[:, None] * stride_right_k \
                     + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # Store the resulting tile
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def tri_mul_contraction(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Computes the O(N³) contraction using the Triton kernel.
    Arguments:
        left, right: tensors of shape (B, N, N, H) and dtype float32.
    Returns:
        out: tensor of shape (B, N, N, H), same dtype/device.
    """
    B, N, _, H = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )
    return out

# ----------------------------------------------------------------------
# Entry point following the required signature
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing variant) fully on‑GPU.
    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor of shape [B, N, N] or None
        - weights_dict : mapping of all LayerNorm/Linear weight tensors
        - config_dict  : contains "dim", "hidden_dim" and optional "nomask"
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Helper to move weights to the right device/dtype
    def get_weight(name):
        return weights[name].to(device=device, dtype=dtype)

    # ----- Load all parameters ------------------------------------------------
    norm_weight = get_weight("norm.weight")
    norm_bias = get_weight("norm.bias")
    left_proj_weight = get_weight("left_proj.weight")
    right_proj_weight = get_weight("right_proj.weight")
    left_gate_weight = get_weight("left_gate.weight")
    right_gate_weight = get_weight("right_gate.weight")
    out_gate_weight = get_weight("out_gate.weight")
    to_out_norm_weight = get_weight("to_out_norm.weight")
    to_out_norm_bias = get_weight("to_out_norm.bias")
    to_out_weight = get_weight("to_out.weight")

    # ----- Input LayerNorm ----------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ----- Linear projections -------------------------------------------------
    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # ----- Optional mask -------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ----- Gates ---------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----- Heavy contraction (i‑k × j‑k) ---------------------------------------
    out_hidden = tri_mul_contraction(left, right)  # (B,N,N,hidden_dim)

    # ----- Output LayerNorm ----------------------------------------------------
    out_hidden = F.layer_norm(out_hidden,
                               normalized_shape=(hidden_dim,),
                               weight=to_out_norm_weight,
                               bias=to_out_norm_bias,
                               eps=1e-5)

    # ----- Output gate ---------------------------------------------------------
    out_hidden = out_hidden * out_gate

    # ----- Final projection ----------------------------------------------------
    out = F.linear(out_hidden, to_out_weight)   # (B,N,N,dim)

    return out
