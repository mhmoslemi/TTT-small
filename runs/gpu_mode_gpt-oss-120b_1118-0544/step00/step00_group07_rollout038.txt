```
import torch
import triton
import triton.language as tl
import torch.nn.functional as F

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # layer norm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # projections
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])
    # gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))
    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate
    # mask
    if mask is not None:
        mask = mask.unsqueeze(-1) # shape [B,N,N,1]
        left = left * mask
        right = right * mask
    # make right_T (swap N dims)
    right_T = right.permute(0, 2, 1, 3).contiguous()
    # flatten batch and hidden_dim
    B, N, _, H = left.shape
    # reshape to (B * H, N, N)
    left_flat = left.reshape(B * H, N, N)
    right_flat = right_T.reshape(B * H, N, N)
    # allocate output
    out_flat = torch.empty_like(left_flat)
    # launch kernel
    triton_matmul(left_flat, right_flat, out_flat, 
                  left_flat.shape[0], left_flat.shape[1], left_flat.shape[2],
                  left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
                  right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
                  out_flat.stride(0), out_flat.stride(1), out_flat.stride(2))
    # reshape output back
    out = out_flat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous() # shape [B,N,N,H]
    # layer norm on output
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate
    # final linear
    out = F.linear(out, weights["to_out.weight"])
    return out
```

```
@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          BATCH, M, N, K,
                          stride_batch_A, stride_m_A, stride_k_A,
                          stride_batch_B, stride_k_B, stride_n_B,
                          stride_batch_C, stride_m_C, stride_n_C,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program ids
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute starting indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        # offsets for k tile
        offs_k = k + tl.arange(0, BLOCK_K)

        # compute pointers
        a_ptrs = A_ptr + batch_id * stride_batch_A + offs_m[:, None] * stride_m_A + offs_k[None, :] * stride_k_A
        b_ptrs = B_ptr + batch_id * stride_batch_B + offs_k[:, None] * stride_k_B + offs_n[None, :] * stride_n_B

        # load tiles with mask
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(a, b)

    # write back
    c_ptrs = C_ptr + batch_id * stride_batch_C + offs_m[:, None] * stride_m_C + offs_n[None, :] * stride_n_C
    tl.store(c_ptrs, acc, mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])
```

def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over k
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

"""
TriMul forward kernel (outgoing variant) for AlphaFold3.
----------------------------------------------------
The heavy O(N³) computation
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is realised as a batched GEMM using a Triton kernel.  The input tensor
[x] is first LayerNormed, linearly projected, gated and masked (all in
PyTorch).  The projections are rearranged so that the hidden dimension
becomes a batch dimension, yielding matrices of shape (B*H, N, N).  The
Triton kernel computes (B*H) independent matrix multiplications
A @ Bᵀ where Bᵀ represents the transposed second pair of sequence
indices.  The result is reshaped back, LayerNormed again, gated and
final‑projected to the original feature dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel.
    Computes C = A @ B where A is (batch, M, K) and B is (batch, K, N).
    All tensors are assumed to be contiguous in the batch dimension.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers with proper strides
        a_ptrs = (
            A_ptr
            + batch_id * stride_Ab
            + offs_m[:, None] * stride_Am
            + offs_k[None, :] * stride_Ak
        )
        b_ptrs = (
            B_ptr
            + batch_id * stride_Bb
            + offs_k[:, None] * stride_Bk
            + offs_n[None, :] * stride_Bn
        )

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)

    # Write out result
    c_ptrs = (
        C_ptr
        + batch_id * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul(A: torch.Tensor, B: torch.Tensor, C: torch.Tensor,
                    BLOCK_M=64, BLOCK_N=64, BLOCK_K=64, num_warps=4):
    """Launch the Triton batched GEMM kernel.
    A, B, C must be contiguous 3‑D tensors of shape (batch, M, K),
    (batch, K, N) and (batch, M, N) respectively.
    """
    assert A.is_contiguous() and B.is_contiguous() and C.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_matmul_kernel[grid](
        A_ptr=A,
        B_ptr=B,
        C_ptr=C,
        M=M,
        N=N,
        K=K,
        stride_Ab=A.stride(0),
        stride_Am=A.stride(1),
        stride_Ak=A.stride(2),
        stride_Bb=B.stride(0),
        stride_Bk=B.stride(1),
        stride_Bn=B.stride(2),
        stride_Cb=C.stride(0),
        stride_Cm=C.stride(1),
        stride_Cn=C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operation.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim] torch.Tensor (float32)
        - mask        : [B, N, N] torch.Tensor (bool/float) or None
        - weights     : dict of named parameters required by the module
        - config      : dict containing at least `dim` and `hidden_dim`
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Pre‑processing in PyTorch (LayerNorm, Projections, Gates, Mask)
    # -----------------------------------------------------------------
    # LayerNorm over the feature dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # Linear projections (no bias as per the reference)
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # Gates (sigmoid after linear)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left_proj * left_gate      # [B, N, N, H]
    right = right_proj * right_gate   # [B, N, N, H]

    # Optional mask handling
    if mask is None:
        mask = torch.ones(
            input_tensor.shape[:3],
            dtype=input_tensor.dtype,
            device=input_tensor.device,
        )
    mask = mask.to(input_tensor.dtype).unsqueeze(-1)  # [B, N, N, 1]
    left = left * mask
    right = right * mask

    # ---------------------------------------------------------------
    # 2. Rearrange tensors so that the hidden dimension becomes batch
    # ---------------------------------------------------------------
    # Bring hidden dim to second axis and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i, k)
    # For the right side we need the transposed pair (k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k, j)

    # Cast to fp16 for the heavy O(N³) compute
    A = left_perm.to(torch.float16)
    B = right_perm.to(torch.float16)

    Bsz, H, N, _ = A.shape
    batch = Bsz * H

    # Flatten batch*hidden into a single leading dimension
    A_flat = A.view(batch, N, N)
    B_flat = B.view(batch, N, N)
    C_flat = torch.empty_like(A_flat, dtype=torch.float16)

    # ---------------------------------------------------------------
    # 3. Triton batched GEMM (core TriMul)
    # ---------------------------------------------------------------
    _batched_matmul(A_flat, B_flat, C_flat,
                    BLOCK_M=64, BLOCK_N=64, BLOCK_K=64, num_warps=4)

    # Reshape back to [B, H, N, N] and permute to original layout
    out_hidden = C_flat.view(Bsz, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out_hidden = out_hidden.to(torch.float32)

    # ---------------------------------------------------------------
    # 4. Post‑processing (LayerNorm, output gate, final linear)
    # ---------------------------------------------------------------
    out = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
