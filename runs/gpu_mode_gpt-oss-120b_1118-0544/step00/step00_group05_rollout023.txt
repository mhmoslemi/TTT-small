    input_tensor, mask, weights, config = data
    # parse config
    dim = config["dim"]; hidden = config["hidden_dim"]; B = input_tensor.shape[0]; N = input_tensor.shape[1]; # dims: x shape [B,N,N,dim]
    # Get weights
    # Norm: use layernorm manually: out = (x - mean) / sqrt(var+eps) * weight + bias. need epsilon.
    # Could use F.layer_norm.

    # Use torch module functions for speed: torch.nn.functional.layer_norm

    x = input_tensor
    eps = 1e-5
    # x shape [B,N,N,dim]; to normalize over last dim we can view as shape [B*N*N, dim] flatten? but F.layer_norm works on any shape.
    # Need weight and bias: norm_weight, norm_bias
    x = F.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # compute left and right projections
    # Linear: matmul over last dim: left = torch.einsum('bnqd,hd->bnqh', x, left_proj_weight.T) basically x @ W.T
    left = torch.nn.functional.linear(x, left_proj_weight, bias=None)  # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight, bias=None)

    # mask handling
    if mask is not None:

        left = left * mask_ext
        right = right * mask_ext
    else:
        # no mask, do nothing (or mask = 1)

        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_li_b, stride_li_i, stride_li_k, stride_li_h,
        stride_rj_b, stride_rj_j, stride_rj_k, stride_rj_h,
        stride_oi_b, stride_oi_i, stride_oi_j, stride_oi_h,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
    ):
    pid = tl.program_id(axis=0)
    total_blocks_m = tl.cdiv(N, BLOCK_M)
    total_blocks_n = tl.cdiv(N, BLOCK_N)
    num_blocks = total_blocks_m * total_blocks_n
    b = pid // num_blocks
    block_id = pid % num_blocks
    pid_m = block_id // total_blocks_n
    pid_n = block_id % total_blocks_n

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_j = offs_j < N

"""
TriMul outgoing kernel (AlphaFold3).

The forward pass consists of:
1. Layer‑norm over the input last dimension.
2. Two linear projections (left/right) followed by elementwise mask and gating.
3. Core operation:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication per hidden channel:
       out_d = L_d @ R_d.T
   Implemented in Triton for the reduction over the sequence dimension K.
4. Layer‑norm over the hidden dimension, gating and final linear projection.
All tensors stay on the GPU; the heavy K‑reduction is performed in a custom
Triton kernel that processes an (i, j) tile for every (batch, hidden) pair.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # fp16*   [B, N, N, H]
    right_ptr,         # fp16*   [B, N, N, H]
    out_ptr,           # fp32*   [B, N, N, H]
    B, N, H,
    stride_li_b, stride_li_i, stride_li_k, stride_li_h,
    stride_rj_b, stride_rj_j, stride_rj_k, stride_rj_h,
    stride_oi_b, stride_oi_i, stride_oi_j, stride_oi_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # 2‑D grid over (i, j)
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    b = pid // blocks_per_batch                # batch index
    block_id = pid % blocks_per_batch
    pid_m = block_id // num_blocks_n           # tile row
    pid_n = block_id % num_blocks_n            # tile col

    # indices of the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # iterate over hidden channels
    h = 0
    while h < H:
        # accumulator for a single (i, j) tile of a given hidden channel
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # reduction over the K dimension
        k = 0
        while k < N:
            cur_k = k + tl.arange(0, BLOCK_K)
            mask_k = cur_k < N

            # left tile   shape (M, K)
            left_ptrs = (
                left_ptr
                + b * stride_li_b
                + offs_i[:, None] * stride_li_i
                + cur_k[None, :] * stride_li_k
                + h * stride_li_h
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None] & mask_k[None, :],
                other=0.0,
            )  # fp16

            # right tile  shape (K, N) – we load it transposed so that tl.dot works
            right_ptrs = (
                right_ptr
                + b * stride_rj_b
                + cur_k[:, None] * stride_rj_k
                + offs_j[None, :] * stride_rj_j
                + h * stride_rj_h
            )
            right = tl.load(
                right_ptrs,
                mask=mask_k[:, None] & mask_j[None, :],
                other=0.0,
            )  # fp16, shape (K, N)

            # matrix multiply for this hidden channel
            acc += tl.dot(left, right).to(tl.float32)

            k += BLOCK_K

        # write the result for this hidden channel
        out_ptrs = (
            out_ptr
            + b * stride_oi_b
            + offs_i[:, None] * stride_oi_i
            + offs_j[None, :] * stride_oi_j
            + h * stride_oi_h
        )
        tl.store(
            out_ptrs,
            acc,
            mask=mask_i[:, None] & mask_j[None, :],
        )
        h += 1


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict, contains at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    import math

    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim], float32

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]       # (hidden, dim)
    right_proj_weight = weights["right_proj.weight"]     # (hidden, dim)

    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight) # [B,N,N,hidden]

    # ---------- Mask ----------
    if mask is None or config.get("nomask", False):
        # no masking -> just a tensor of ones (no effect)
        mask_f = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask_f = mask.to(dtype)  # keep same dtype as x
    mask_f = mask_f.unsqueeze(-1)  # [B,N,N,1]
    left = left * mask_f
    right = right * mask_f

    # ---------- Gates ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Core TriMul (einsum) ----------
    # Cast to half for the kernel (speed) – the kernel accumulates in fp32
    left_half = left.to(torch.float16).contiguous()
    right_half = right.to(torch.float16).contiguous()

    # allocate output of the reduction: float32
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # strides (in elements)
    s_li_b, s_li_i, s_li_k, s_li_h = left_half.stride()
    s_rj_b, s_rj_j, s_rj_k, s_rj_h = right_half.stride()
    s_oi_b, s_oi_i, s_oi_j, s_oi_h = out.stride()

    # block sizes – tuned for typical B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # launch configuration
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_blocks = B * grid_m * grid_n
    grid = (total_blocks,)

    _trimul_outgoing_kernel[grid](
        left_half,
        right_half,
        out,
        B,
        N,
        hidden,
        s_li_b,
        s_li_i,
        s_li_k,
        s_li_h,
        s_rj_b,
        s_rj_j,
        s_rj_k,
        s_rj_h,
        s_oi_b,
        s_oi_i,
        s_oi_j,
        s_oi_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------- Output processing ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]  # (dim, hidden)

    # layer‑norm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    # gating
    out = out * out_gate
    # final linear back to original dim
    out = F.linear(out, to_out_weight)  # bias is None

    return out
