"""
TriMul "outgoing" forward pass implementation.

The heavy part of the TriMul operator is the contraction
    out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]
which is a batched matrix multiplication for each hidden channel `h`.
We fuse the batch and hidden dimensions and run a Triton kernel that
computes many (i, j) blocks while iterating over the summation dimension `k`.
All other operations (LayerNorm, linear projections, gating, final
LayerNorm + output linear) are performed in PyTorch for simplicity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for each (batch, hidden) pair.
# ----------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # output strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = left @ right^T for each (batch, hidden) slice."""
    pid_z = tl.program_id(2)   # combined batch*hidden index
    pid_m = tl.program_id(0)   # i block
    pid_n = tl.program_id(1)   # j block

    total = B * H
    b = pid_z // H
    h = pid_z % H

    # Offsets for i, j, k dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator init
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension (the summation axis)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # --------------------------------------------------------------- #
        # Load left block: shape (BLOCK_M, BLOCK_K)  -> left[i, k]
        # --------------------------------------------------------------- #
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
            + h * stride_l_h
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # --------------------------------------------------------------- #
        # Load right block transposed: shape (BLOCK_K, BLOCK_N) -> right_T[k, j]
        # --------------------------------------------------------------- #
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + offs_k[:, None] * stride_r_k      # k as row index
            + offs_j[None, :] * stride_r_j      # j as column index
            + h * stride_r_h
        )
        right_T = tl.load(right_ptrs,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0)

        # Matrix multiplication accumulate
        acc += tl.dot(left, right_T)   # (M,K) @ (K,N) -> (M,N)

    # --------------------------------------------------------------- #
    # Write back
    # --------------------------------------------------------------- #
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + h * stride_o_h
    )
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------------- #
# Main entry point as required by the problem statement.
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul "outgoing" operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask (optional), weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask        : torch.Tensor of shape [B, N, N] (may be omitted or None)
        - weights     : dict of parameter tensors
        - config      : dict containing {"dim", "hidden_dim", "nomask", ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    # Unpack arguments
    # --------------------------------------------------------------------- #
    if len(data) == 4:
        x, mask, weights, config = data
    else:  # mask omitted
        x, weights, config = data
        mask = None

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------------- #
    # Extract weight tensors
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------------- #
    eps = 1e-5
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # --------------------------------------------------------------------- #
    # Linear projections (no bias)
    # --------------------------------------------------------------------- #
    # Shape after projection: [B, N, N, hidden_dim]
    left = F.linear(x_norm, left_proj_weight)   # (hidden_dim, dim) weight
    right = F.linear(x_norm, right_proj_weight)

    # --------------------------------------------------------------------- #
    # Gating projections + sigmoid
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # --------------------------------------------------------------------- #
    # Apply mask if present; mask shape [B, N, N]
    # --------------------------------------------------------------------- #
    if not nomask and mask is not None:
        # Ensure mask is float and broadcastable
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # -> [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f
    # If nomask=True, mask is effectively all ones, so nothing to do.

    # --------------------------------------------------------------------- #
    # Apply elementwise gates
    # --------------------------------------------------------------------- #
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for Triton kernels
    left = left.contiguous()
    right = right.contiguous()

    # --------------------------------------------------------------------- #
    # Triton kernel to compute the triple contraction
    # --------------------------------------------------------------------- #
    B, N, _, _ = left.shape  # N already known

    # Allocate output tensor for the einsum result (float32)
    out = torch.empty((B, N, N, hidden_dim),
                      dtype=left.dtype,
                      device=left.device)

    # Compute grid dimensions
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    # Launch Triton kernel
    tri_mul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_l_b=left.stride(0),
        stride_l_i=left.stride(1),
        stride_l_k=left.stride(2),
        stride_l_h=left.stride(3),
        stride_r_b=right.stride(0),
        stride_r_j=right.stride(1),
        stride_r_k=right.stride(2),
        stride_r_h=right.stride(3),
        stride_o_b=out.stride(0),
        stride_o_i=out.stride(1),
        stride_o_j=out.stride(2),
        stride_o_h=out.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    # Post‑processing: to_out_norm, gating, final linear
    # --------------------------------------------------------------------- #
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    # Apply out_gate elementwise (same shape as out)
    out = out * out_gate

    # Final linear projection back to dim
    output = F.linear(out, to_out_weight)   # (dim, hidden_dim) weight

    return output
