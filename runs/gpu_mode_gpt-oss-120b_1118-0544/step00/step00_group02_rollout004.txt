"""
TriMul forward (outgoing) implementation.

The heavy part of the operation is the triplet multiplication:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which can be seen as a batched matrix multiplication of shape
    (B*H, N, N) @ (B*H, N, N).T
The kernel fuses the reduction over k using a tiled GEMM
implemented in Triton, while all surrounding LayerNorms,
Linear projections, and gating are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton kernel for the core sum‑over‑k multiplication
# ------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,        # pointers
    B, N, H,                             # batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where left : (B,N,N,H) and right : (B,N,N,H).
    The kernel treats each (b,h) pair as an independent GEMM.
    """
    pid_m = tl.program_id(0)   # combined (batch*head)*grid_m
    pid_n = tl.program_id(1)   # grid_n

    grid_m = tl.cdiv(N, BLOCK_M)

    # Decode (batch, head) and the tile indices within M dimension
    batch_head = pid_m // grid_m               # 0 .. B*H-1
    tile_i = pid_m % grid_m                    # 0 .. grid_m-1
    b = batch_head // H                         # batch index
    h = batch_head % H                          # head (hidden_dim) index

    # Offsets for the output tile (i,j)
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: (M, K)
        left_ptrs = left_base + \
                    offs_i[:, None] * stride_left_i + \
                    offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # Load right block transposed: (K, N)
        # We need right[k, j] → rows = k, cols = j
        right_ptrs = right_base + \
                     offs_k[:, None] * stride_right_k + \
                     offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # Accumulate dot product
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # Store result
    out_ptrs = out_base + \
               offs_i[:, None] * stride_out_i + \
               offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])

# Custom kernel entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] on the same device as input.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper to fetch weight tensors (ensure correct device / dtype)
    # ------------------------------------------------------------------
    def get(name):
        t = weights[name]
        # move to correct device/dtype if necessary
        if t.device != device:
            t = t.to(device)
        if t.dtype != dtype:
            t = t.to(dtype)
        return t

    # ------------------------------------------------------------------
    # LayerNorm on input (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=get("norm.weight"),
        bias=get("norm.bias"),
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, get("left_proj.weight"))     # [B, N, N, H]
    right = F.linear(x, get("right_proj.weight"))   # [B, N, N, H]

    # ------------------------------------------------------------------
    # Apply mask if present and not nomask
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask shape: [B, N, N]  -> broadcast on hidden dim
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid) and apply gates
    # ------------------------------------------------------------------
    left_gate = F.linear(x, get("left_gate.weight")).sigmoid()   # [B,N,N,H]
    right_gate = F.linear(x, get("right_gate.weight")).sigmoid()
    out_gate = F.linear(x, get("out_gate.weight")).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape  # _, _ = N, hidden_dim

    # ------------------------------------------------------------------
    # Allocate output tensor for the core multiplication
    # ------------------------------------------------------------------
    out_core = torch.empty_like(left)   # shape [B, N, N, H]

    # ------------------------------------------------------------------
    # Extract strides (in element counts)
    # ------------------------------------------------------------------
    s_left = left.stride()     # (b, i, k, h)
    s_right = right.stride()   # (b, j, k, h)
    s_out = out_core.stride()  # (b, i, j, h)

    # ------------------------------------------------------------------
    # Triton kernel launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    # Combine batch*head dimension with grid_m
    total_batch_head = B * hidden_dim
    grid = (grid_m * total_batch_head, grid_n)

    _trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_core,
        B=int(B),
        N=int(N),
        H=int(hidden_dim),
        stride_left_b=int(s_left[0]),
        stride_left_i=int(s_left[1]),
        stride_left_k=int(s_left[2]),
        stride_left_h=int(s_left[3]),
        stride_right_b=int(s_right[0]),
        stride_right_j=int(s_right[1]),
        stride_right_k=int(s_right[2]),
        stride_right_h=int(s_right[3]),
        stride_out_b=int(s_out[0]),
        stride_out_i=int(s_out[1]),
        stride_out_j=int(s_out[2]),
        stride_out_h=int(s_out[3]),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # reasonable default for the B200
    )

    # ------------------------------------------------------------------
    # Post‑processing: normalization, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_core,
        (hidden_dim,),
        weight=get("to_out_norm.weight"),
        bias=get("to_out_norm.bias"),
    )
    out = out * out_gate
    out = F.linear(out, get("to_out.weight"))   # projects back to dim

    return out
