@triton.jit
def triu_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    mask_ptr,  # optional mask may be null pointer? We'll handle.
    mask_batch_stride, mask_i_stride, mask_j_stride,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)  # combine batch*hidden dimension
    # decode batch and hidden indices:
    batch_idx = pid_b // hidden
    d_idx = pid_b % hidden

    pid_i = tl.program_id(0)  # i block index
    pid_j = tl.program_id(1)  # j block index

    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Guard for out-of-bounds
    i_mask = i_offset < N
    j_mask = j_offset < N

    # Initialize accumulator to zero (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute offsets for left and right
    # base pointer per batch, d fixed
    left_base = left_ptr + batch_idx * stride_left_batch + d_idx * stride_left_d
    right_base = right_ptr + batch_idx * stride_right_batch + d_idx * stride_right_d

    # Loop over k dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # Load left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i_offset[:, None] * stride_left_i + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # Load right block as transposed: we need shape (BLOCK_K, BLOCK_N)
        # right_base + j_offset[None, :] * stride_right_i + k_offset[:, None] * stride_right_k
        right_ptrs = right_base + k_offset[:, None] * stride_right_k + j_offset[None, :] * stride_right_i
        right = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0).to(tl.float32)

        # Compute dot
        acc += tl.dot(left, right)  # left (M,K) dot right (K,N) -> (M,N)

    # Apply mask if present: mask shape [B,N,N], same for each d.
    if mask_ptr is not None:
        mask_base = mask_ptr + batch_idx * mask_batch_stride
        mask_ptrs = mask_base + i_offset[:, None] * mask_i_stride + j_offset[None, :] * mask_j_stride
        mask = tl.load(mask_ptrs, mask=i_mask[:, None] & j_mask[None, :], other=0.0).to(tl.float32)
        acc *= mask

    # Store result
    out_base = out_ptr + batch_idx * stride_out_batch + d_idx * stride_out_d
    out_ptrs = out_base + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

The kernel computes, for each batch b and hidden channel h,
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where left and right are the linear projections of the normalized input,
optionally masked and gated.  The computation is a batched matrix multiply
(left @ rightᵀ) performed with Triton tiling (BLOCK_M × BLOCK_N tiles,
reduced over K in BLOCK_K steps).  All other operations (layer‑norm,
linear projections, sigmoid gates) are carried out in PyTorch.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for one batch * hidden channel slice.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # decode which batch and hidden channel this program works on
    pid = tl.program_id(2)
    batch_idx = pid // H
    hid = pid % H

    # base pointers for this (batch, hidden) slice
    left_base = left_ptr + batch_idx * stride_l_batch + hid * stride_l_h
    right_base = right_ptr + batch_idx * stride_r_batch + hid * stride_r_h
    out_base = out_ptr + batch_idx * stride_o_batch + hid * stride_o_h

    # tile indices for the i (row) and j (col) dimensions
    i_offset = tl.program_id(0) * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = tl.program_id(1) * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_offset < N
    j_mask = j_offset < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k in BLOCK_K steps
    for k_base in range(0, N, BLOCK_K):
        k_offset = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # left block  (M x K)
        left_ptrs = left_base + i_offset[:, None] * stride_l_i + k_offset[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0).to(tl.float32)

        # right block transposed -> (K x N)
        right_ptrs = right_base + k_offset[:, None] * stride_r_k + j_offset[None, :] * stride_r_i
        right = tl.load(right_ptrs,
                        mask=k_mask[:, None] & j_mask[None, :],
                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)      # (M,N) += (M,K) @ (K,N)

    # write back
    out_ptrs = out_base + i_offset[:, None] * stride_o_i + j_offset[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _tri_mul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication over the third axis:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    left/right shape: [B, N, N, H] (contiguous, float32)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # make sure data are contiguous for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # strides (in element counts)
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # tiling parameters (tuned for typical N <= 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H
    )
    _tri_mul_kernel[grid](
        left, right, out,
        N, H,
        s_l_b, s_l_i, s_l_k, s_l_h,
        s_r_b, s_r_i, s_r_k, s_r_h,
        s_o_b, s_o_i, s_o_j, s_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    return out

# ----------------------------------------------------------------------
# Main entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments:
        data: tuple (input, mask, weights, config)
            - input: torch.Tensor of shape [B, N, N, dim]
            - mask:  torch.Tensor of shape [B, N, N] (may be None or empty)
            - weights: dict of model parameters (see skeleton)
            - config: dict with at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (mask is broadcast on the hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # mask may be bool; convert to float
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating tensors (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core TriMul einsum implemented as a fused Triton kernel
    # ------------------------------------------------------------------
    out = _tri_mul_outgoing(left, right)   # shape [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Output normalization, gating and final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
