@triton.jit
def batched_matmul_T_kernel(
    A_ptr, B_ptr, C_ptr,
    N,
    stride_az, stride_ai, stride_aj,
    stride_bz, stride_bi, stride_bj,
    stride_cz, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program_id for batch, block row, block col
    batch_id = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # Compute row and col offsets
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Compute offsets for A block (i, k)
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        a_ptrs = A_ptr + batch_id * stride_az + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_aj
        b_ptrs = B_ptr + batch_id * stride_bz + offs_k[:, None] * stride_bi + offs_j[None, :] * stride_bj
        # Load blocks with masks for out-of-range
        a = tl.load(a_ptrs, mask=(offs_i < N)[:, None] & (offs_k < N)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < N)[:, None] & (offs_j < N)[None, :], other=0.0)
        # Multiply accumulate
        acc += tl.dot(a, b)
    # Write back
    c_ptrs = C_ptr + batch_id * stride_cz + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=(offs_i < N)[:, None] & (offs_j < N)[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_T_kernel(A_ptr, B_ptr, C_ptr,
                            N,
                            stride_az, stride_ai, stride_aj,
                            stride_bz, stride_bi, stride_bj,
                            stride_cz, stride_ci, stride_cj,
                            BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)

        # Load A block (M, K)
        a_ptrs = A_ptr + batch_id * stride_az + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_aj
        a = tl.load(a_ptrs,
                    mask=(offs_m < N)[:, None] & (offs_k < N)[None, :],
                    other=0.0)

        # Load B_T block (K, N) from B using transposed access
        b_ptrs = B_ptr + batch_id * stride_bz + offs_k[:, None] * stride_bi + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=(offs_k < N)[:, None] & (offs_n < N)[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # Store
    c_ptrs = C_ptr + batch_id * stride_cz + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs,
             acc,
             mask=(offs_m < N)[:, None] & (offs_n < N)[None, :])

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the outgoing TriMul operation.
    The kernel performs LayerNorm, linear projections, gating, and an
    O(N^3) pairwise multiplication using a batched matrix multiplication
    implemented in Triton. The rest of the operations are done with
    standard PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # optional nomask: config.get("nomask", False)
    nomask = config.get("nomask", False)
    
    # LayerNorm (affine)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    
    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # Compute linear maps
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)
    
    # Apply mask if present
    if mask is not None and not nomask:
        # Expect mask shape [B,N,N]; cast to float
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f
    
    # Gates (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))
    
    # Apply gating
    left = left * left_gate
    right = right * right_gate
    
    # Reshape for batched matmul
    B, N, _, _ = left.shape  # B,N,N,H
    # Move hidden dim to batch dimension
    left_ = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_ = right.permute(0, 3, 1, 2).contiguous()
    # flatten batch * hidden_dim
    left_flat = left_.reshape(-1, N, N)
    right_flat = right_.reshape(-1, N, N)
    
    # Batched matmul using Triton
    out_flat = triton_batch_matmul_T(left_flat, right_flat)   # [B*H, N, N]
    
    # Restore shape [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    
    # Output layernorm (affine)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    
    # Apply out_gate
    out = out * out_gate
    
    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_weight)   # shape [B,N,N,dim]
    
    return out

"""
Outgoing TriMul (TriMul) forward pass via Triton.

Algorithm
---------
1. Layer‑norm the input tensor (affine) over the last dimension `dim`.
2. Linear projections (no bias) to hidden dimension `hidden_dim`:
   left = W_left_proj·x, right = W_right_proj·x.
3. (Optional) Apply the pairwise mask to `left` and `right`.
4. Compute three sigmoid gates from the normalized input and apply:
   left  *= sigmoid(W_left_gate·x)
   right *= sigmoid(W_right_gate·x)
   out_gate = sigmoid(W_out_gate·x)   # used later.
5. The core O(N³) operation:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   This is a batched matrix multiplication of many N×N matrices
   with the second operand transposed.  We implement it with a
   Triton kernel that processes all (batch * hidden_dim) matrices
   in parallel.
6. Reshape the result back to [B,N,N,hidden_dim] and apply a second
   LayerNorm (affine) over the hidden dimension.
7. Multiply by `out_gate` and project back to `dim` with a final linear.
8. Return the tensor of shape [B,N,N,dim].

Only the heavy N‑dimensional multiplication (step 5) is performed in Triton;
all other operations use native PyTorch for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ  where A,B are [B, N, N] matrices.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_T_kernel(
    A_ptr, B_ptr, C_ptr,
    N,
    stride_az, stride_ai, stride_aj,
    stride_bz, stride_bi, stride_bj,
    stride_cz, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)               # combined (batch * hidden) index
    pid_m = tl.program_id(1)                  # output block row
    pid_n = tl.program_id(2)                  # output block column

    # Offsets within the output matrix C
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator for the output block
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K (also N)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)

        # A block: rows = offs_m, cols = offs_k
        a_ptrs = (A_ptr + batch_id * stride_az
                  + offs_m[:, None] * stride_ai
                  + offs_k[None, :] * stride_aj)
        a = tl.load(a_ptrs,
                    mask=(offs_m < N)[:, None] & (offs_k < N)[None, :],
                    other=0.0)

        # Bᵀ block: rows = offs_k (original K dimension), cols = offs_n
        b_ptrs = (B_ptr + batch_id * stride_bz
                  + offs_k[:, None] * stride_bi
                  + offs_n[None, :] * stride_bj)
        b = tl.load(b_ptrs,
                    mask=(offs_k < N)[:, None] & (offs_n < N)[None, :],
                    other=0.0)

        # Accumulate the matrix product
        acc += tl.dot(a, b)

    # Write the result back to C
    c_ptrs = (C_ptr + batch_id * stride_cz
              + offs_m[:, None] * stride_ci
              + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs,
             acc,
             mask=(offs_m < N)[:, None] & (offs_n < N)[None, :])

def _triton_batch_matmul_T(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched (A @ Bᵀ) for tensors of shape [B, N, N] using Triton.
    Both A and B must be float32 and contiguous.
    Returns C of the same shape.
    """
    assert A.shape == B.shape
    batch, N, _ = A.shape
    C = torch.empty_like(A)

    # Strides in elements (not bytes)
    stride_Az, stride_Ai, stride_Aj = A.stride()
    stride_Bz, stride_Bi, stride_Bj = B.stride()
    stride_Cz, stride_Ci, stride_Cj = C.stride()

    # Tunable block sizes (work well on Nvidia Hopper/B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _batched_matmul_T_kernel[grid](
        A, B, C,
        N,
        stride_Az, stride_Ai, stride_Aj,
        stride_Bz, stride_Bi, stride_Bj,
        stride_Cz, stride_Ci, stride_Cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C

# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of pre‑trained weight tensors
        - config       : dict containing at least "dim" and "hidden_dim".
                         May also contain "nomask" (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (`dim`)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) and their application
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via batched matmul (Triton)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, hidden_dim

    # Move hidden dimension to the batch axis for the kernel
    left_ = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_ = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch*hidden into a single batch dimension
    left_flat = left_.reshape(-1, N, N)   # [B*H, N, N]
    right_flat = right_.reshape(-1, N, N)

    # Triton batched matmul (A @ Bᵀ)
    out_flat = _triton_batch_matmul_T(left_flat, right_flat)   # [B*H, N, N]

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (affine) over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 7. Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                     # gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
