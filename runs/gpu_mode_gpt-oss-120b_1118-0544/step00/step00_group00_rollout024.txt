@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_Am, stride_Ak,
                          stride_Bk, stride_Bn,
                          stride_Cm, stride_Cn,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # A: (batch, M, K)
    # B: (batch, K, N)
    # C: (batch, M, N)
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute base pointers for this batch
    A_ptrs = A_ptr + batch * stride_Ab + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    B_ptrs = B_ptr + batch * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # accumulators
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles A and B
        a = tl.load(A_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0,BLOCK_K) < K), other=0.0)
        b = tl.load(B_ptrs, mask=(k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        # dot
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)
        # advance pointers
        A_ptrs += BLOCK_K * stride_Ak
        B_ptrs += BLOCK_K * stride_Bk
    # write back
    C_ptrs = C_ptr + batch * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(C_ptrs, acc.to(tl.float16) if acc.dtype==tl.float32 else acc, mask=mask_m[:, None] & mask_n[None, :])

import torch
import triton
import triton.language as tl

@triton.jit
def batch_matmul_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_Ab, stride_Am, stride_Ak,
                        stride_Bb, stride_Bk, stride_Bn,
                        stride_Cb, stride_Cm, stride_Cn,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """Batched matrix multiplication C = A @ B (where B is already transposed).

    A: (batch, M, K)
    B: (batch, K, N)
    C: (batch, M, N)
    All tensors are in row-major (contiguous) layout.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets for rows and cols
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # load A tile (M x K)
        a_ptr = A_ptr + batch_id * stride_Ab + offs_m[:, None] * stride_Am + cur_k[None, :] * stride_Ak
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B tile (K x N)
        b_ptr = B_ptr + batch_id * stride_Bb + cur_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_ptr = C_ptr + batch_id * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Steps:
    1. Layer‑norm the 4‑D input.
    2. Linear projections (left/right) + optional mask.
    3. Compute three sigmoid gates and apply them.
    4. Core O(N³) reduction implemented as a batched matmul:
       out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
       which is equivalent to left @ rightᵀ for each hidden channel.
    5. Layer‑norm + output gate on the hidden result.
    6. Final linear projection back to ``dim``.
    """
    import torch
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Layer norm over the feature dimension (last axis)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape (B,N,N,dim)

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.to(left.dtype)
        mask = mask.unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid) and application
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core reduction via batched matrix multiplication in Triton
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    total_batches = B * hidden_dim

    # Cast to half for kernel bandwidth
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Right needs to be transposed on the two N axes
    # Shape after transpose: (B, hidden_dim, N, N)
    right_h_T = right_h.permute(0, 3, 2, 1).contiguous()

    # Reshape to (total_batches, N, N) where each batch corresponds to a hidden channel
    left_flat = left_h.permute(0, 3, 1, 2).contiguous().view(total_batches, N, N)
    right_flat = right_h_T.view(total_batches, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=left_flat.device)

    # strides (in elements)
    stride_Ab = left_flat.stride(0)
    stride_Am = left_flat.stride(1)
    stride_Ak = left_flat.stride(2)

    stride_Bb = right_flat.stride(0)
    stride_Bk = right_flat.stride(1)
    stride_Bn = right_flat.stride(2)

    stride_Cb = out_flat.stride(0)
    stride_Cm = out_flat.stride(1)
    stride_Cn = out_flat.stride(2)

    # Block sizes – tuned for the B200 memory hierarchy.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batch_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        stride_Ab,
        stride_Am,
        stride_Ak,
        stride_Bb,
        stride_Bk,
        stride_Bn,
        stride_Cb,
        stride_Cm,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B,N,N,hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # -----------------------------------------------------------------
    # 6. Output layer‑norm, output gate and final projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out_norm = out_norm * out_gate

    to_out_weight = weights["to_out.weight"]
    final = torch.nn.functional.linear(out_norm, to_out_weight)  # (B,N,N,dim)

    return final

"""
Triton‑accelerated forward pass for the outgoing TriMul operator.

Algorithm
---------
1. Layer‑norm the 4‑D input over the feature dimension.
2. Linear projections (left/right) without bias, followed by optional mask.
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Apply gates to the projected tensors.
5. Core O(N³) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a batched matrix multiplication
   left @ rightᵀ for each hidden channel h.  Implemented with a
   custom Triton kernel that processes (B·hidden) independent matrix
   multiplications in half‑precision.
6. Layer‑norm over the hidden dimension, apply the output gate and
   project back to the original feature dimension.
All final results are returned as float32.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def batch_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication C = A @ B  (B is already transposed).

    A: (batch, M, K)   – row‑major
    B: (batch, K, N)   – row‑major (here B is the transposed right tensor)
    C: (batch, M, N)
    """
    batch_id = tl.program_id(0)      # which hidden‑channel / batch we are processing
    pid_m   = tl.program_id(1)      # tile of rows
    pid_n   = tl.program_id(2)      # tile of cols

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (better precision for the sum)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptr = A_ptr + batch_id * stride_Ab \
                + offs_m[:, None] * stride_Am \
                + cur_k[None, :] * stride_Ak
        a = tl.load(a_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile: (BLOCK_K, BLOCK_N)
        b_ptr = B_ptr + batch_id * stride_Bb \
                + cur_k[:, None] * stride_Bk \
                + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # write the result back (store as fp16 to save bandwidth)
    c_ptr = C_ptr + batch_id * stride_Cb \
            + offs_m[:, None] * stride_Cm \
            + offs_n[None, :] * stride_Cn
    tl.store(c_ptr,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑based forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights: dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # 1. Layer‑norm over the feature dimension
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # (B, N, N, dim)

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if mask is None:
        mask = torch.ones((B, N, N), dtype=left.dtype, device=left.device)
    mask = mask.to(left.dtype).unsqueeze(-1)    # (B,N,N,1)
    left = left * mask
    right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid) – computed from the normalized input (no mask)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core reduction using a batched matmul kernel
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    -> for each hidden channel h we compute left_h @ right_hᵀ
    # -----------------------------------------------------------------
    total_batches = B * hidden_dim

    # Cast to half‑precision for the kernel
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Transpose the two N axes of the right tensor
    right_h_T = right_h.permute(0, 3, 2, 1).contiguous()   # (B, hidden, N, N)

    # Reshape to (total_batches, N, N) – each hidden channel becomes an independent batch
    left_flat  = left_h.permute(0, 3, 1, 2).contiguous().view(total_batches, N, N)
    right_flat = right_h_T.view(total_batches, N, N)

    # Allocate output in half‑precision
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=left_flat.device)

    # Strides (in elements) – required by the kernel
    stride_Ab = left_flat.stride(0)
    stride_Am = left_flat.stride(1)
    stride_Ak = left_flat.stride(2)

    stride_Bb = right_flat.stride(0)
    stride_Bk = right_flat.stride(1)
    stride_Bn = right_flat.stride(2)

    stride_Cb = out_flat.stride(0)
    stride_Cm = out_flat.stride(1)
    stride_Cn = out_flat.stride(2)

    # Block sizes – tuned for the B200 architecture
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batch_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore the original shape (B, N, N, hidden)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # -----------------------------------------------------------------
    # 6. Output layer‑norm, output gate, final projection back to `dim`
    # -----------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate

    final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])   # (B, N, N, dim)

    return final
