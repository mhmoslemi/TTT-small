@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bc = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    batch = pid_bc // H
    channel = pid_bc % H

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    i_offset = i_start + tl.arange(0, BLOCK_M)
    j_offset = j_start + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k_tiles):
        k_start = kk * BLOCK_K
        k_offset = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left: [B, i, k, h] (i rows, k cols)
        left_ptrs = left_ptr + batch * stride_left_b + i_offset[:, None] * stride_left_i + k_offset[None, :] * stride_left_k + channel * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right transposed: we want [B, k, j, h]
        right_ptrs = right_ptr + batch * stride_right_b + k_offset[:, None] * stride_right_i + j_offset[None, :] * stride_right_j + channel * stride_right_h
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)  # left (M,K), right (K,N) -> (M,N)

    # store
    out_ptrs = out_ptr + batch * stride_out_b + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j + channel * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h] for each batch and channel.
    Grid dimensions:
        axis 0 : batch * hidden_dim  (select batch and channel)
        axis 1 : tile row index i
        axis 2 : tile column index j
    """
    pid_bc = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and channel
    batch = pid_bc // H
    channel = pid_bc - batch * H

    # start indices for the i/j tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # row/col offsets within the tile
    i_off = i_start + tl.arange(0, BLOCK_M)
    j_off = j_start + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # accumulator for the output tile (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    k_tile = tl.int32(0)
    while k_tile < num_k_tiles:
        k_start = k_tile * BLOCK_K

"""
TriMul (outgoing) forward pass accelerated with Triton.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×C}.
2. Linear projections (bias‑free) produce left/right ∈ ℝ^{B×N×N×H}.
3. Optional pairwise mask is applied, then per‑position gates
   (sigmoid(linear(x))) are multiplied into left/right.
4. Core operation:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   i.e. for each channel h we compute a matrix multiplication
   L_h @ R_hᵀ where L_h,R_h ∈ ℝ^{N×N}.
   This is implemented in a Triton kernel tiled over (i,j) and
   summed over k.
5. The result (shape B×N×N×H) is layer‑normed, multiplied by an
   output gate and finally projected back to the original channel
   dimension C.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,                # const *T
    right_ptr,               # const *T
    out_ptr,                 # *T
    B, N, H,                 # runtime int32
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Grid layout:
        axis 0 : batch * hidden_dim  (selects (batch, channel))
        axis 1 : tile row index i
        axis 2 : tile column index j
    """
    pid_bc = tl.program_id(0)          # batch * hidden
    pid_i = tl.program_id(1)           # tile row
    pid_j = tl.program_id(2)           # tile column

    # Decode batch index and hidden channel
    batch = pid_bc // H
    channel = pid_bc - batch * H

    # Tile start positions
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets inside the tile
    i_off = i_start + tl.arange(0, BLOCK_M)
    j_off = j_start + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # Accumulator for the (BLOCK_M, BLOCK_N) output tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k in tiles of size BLOCK_K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_off = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left[b, i, k, channel]   (shape BLOCK_M × BLOCK_K)
        left_ptrs = (
            left_ptr
            + batch * stride_left_b
            + i_off[:, None] * stride_left_i
            + k_off[None, :] * stride_left_k
            + channel * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right[b, j, k, channel]   we need it transposed -> (k, j)
        right_ptrs = (
            right_ptr
            + batch * stride_right_b
            + k_off[:, None] * stride_right_i   # k plays role of original i
            + j_off[None, :] * stride_right_j   # j stays as original j
            + channel * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # Perform the matmul for this k‑tile
        acc += tl.dot(left, right)   # (M,K) × (K,N) → (M,N)

    # Write the accumulated tile to global memory
    out_ptrs = (
        out_ptr
        + batch * stride_out_b
        + i_off[:, None] * stride_out_i
        + j_off[None, :] * stride_out_j
        + channel * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [bs, N, N, dim]
        - mask         : torch.Tensor of shape [bs, N, N] (may be None)
        - weights      : dict of torch.Tensors containing the model weights
        - config       : dict with keys "dim", "hidden_dim", "nomask" ...

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, N, N, dim] (float32, same device as input).
    """
    input_tensor, mask, weights, config = data

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----- LayerNorm on the input -----
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----- Linear projections (bias=False) -----
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----- Optional mask (broadcast over the hidden dimension) -----
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)               # [bs, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----- Gating ----------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----- Triton kernel for the core multiplicative update -----
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)   # shape [bs, N, N, hidden_dim]

    # Strides (already in element units)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_j, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tile sizes – chosen to give good occupancy on B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        bs * hidden_dim,                     # axis 0 : batch * channel
        (N + BLOCK_M - 1) // BLOCK_M,        # axis 1 : i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,        # axis 2 : j‑tiles
    )

    tri_mul_kernel[
        grid
    ](
        left,
        right,
        out,
        bs,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_j,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Post‑processing -----
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
