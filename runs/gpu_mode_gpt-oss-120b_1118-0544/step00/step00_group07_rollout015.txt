import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_m = tl.program_id(2)

    # compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # pointers
        a_ptrs = left_ptr + pid_m * stride_left_batch + (offs_i[:, None] * stride_left_i) + (cur_k[None, :] * stride_left_k)
        b_ptrs = right_ptr + pid_m * stride_right_batch + (cur_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_j)

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_N)

    # store
    out_ptrs = out_ptr + pid_m * stride_out_batch + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass fused with a Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over its channel dimension.
2. Linear projections to a hidden dimension (left / right) and to three
   gating tensors (left_gate, right_gate, out_gate).  The gates are
   passed through a sigmoid.
3. Optional binary mask is applied to the projected left/right tensors.
4. Gates are fused with the projected tensors (element‑wise product).
5. Core “triangle‑multiplicative” update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   For each batch and hidden‑head this is a matrix multiplication
   `left[b,d] @ right_T[b,d]`.  The multiplication is performed by a
   custom Triton kernel that batches the (B·hidden_dim) independent
   matmuls.
6. Layer‑norm the result, multiply by `out_gate` and a final linear
   projection back to the original channel dimension.

The Triton kernel implements a tiled batched GEMM (no bias, no
activation) with fp16 inputs and fp32 accumulation.  All other
operations are handled in PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication for the TriMul core.

    Computes, for each batch index `m` (0 ≤ m < M):
        out[m, i, j] = Σ_k left[m, i, k] * right[m, k, j]

    Arguments:
        left_ptr, right_ptr, out_ptr: pointers to the input / output buffers.
        M, N, K: dimensions (M = B*hidden_dim, N = seq_len, K = seq_len).
        stride_*: strides (in elements) for each tensor.
    """
    pid_i = tl.program_id(0)           # tile row
    pid_j = tl.program_id(1)           # tile column
    pid_m = tl.program_id(2)           # batch/hidden head

    # -----------------------------------------------------------------
    # Offsets for the i‑ and j‑tiles
    # -----------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the K dimension in tiles of size BLOCK_K
    # -----------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Pointers into the left and right matrices for this tile
        a_ptrs = (left_ptr + pid_m * stride_left_batch
                  + offs_i[:, None] * stride_left_i
                  + cur_k[None, :] * stride_left_k)
        b_ptrs = (right_ptr + pid_m * stride_right_batch
                  + cur_k[:, None] * stride_right_k
                  + offs_j[None, :] * stride_right_j)

        # Load with masking; inputs are fp16 → cast to fp32 for accumulation
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Matrix multiply the tile and accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # -----------------------------------------------------------------
    # Write the result
    # -----------------------------------------------------------------
    out_ptrs = (out_ptr + pid_m * stride_out_batch
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (binary or float)
        - weights      : dict of named weight tensors (float32)
        - config       : dict containing at least 'dim', 'hidden_dim',
                         'nomask' (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Load weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. Input layer‑norm over the last dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # -----------------------------------------------------------------
    # 2. Linear projections + gating (all in float32 for stability)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)          # [B, N, N, H]
    right = F.linear(x, right_proj_weight)        # [B, N, N, H]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # -----------------------------------------------------------------
    # 3. Apply optional mask (mask is broadcast over the hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask_unsq = mask.unsqueeze(-1).type_as(x)   # [B,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq

    # -----------------------------------------------------------------
    # 4. Fuse gates with projections
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core TriMul using a Triton kernel (fp16 inputs, fp32 output)
    # -----------------------------------------------------------------
    # Cast to half for the kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # right needs to be transposed on the two sequence axes: (j,k) -> (k,j)
    right_t = right.permute(0, 2, 1, 3).contiguous()   # [B, N, N, H] with axes (k,j)

    # Bring the hidden dimension to the second axis and flatten batch*head
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_perm = right_t.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N] (k,j)

    B, N, _, H = left.shape
    M = B * H   # total number of independent matmuls

    left_batch = left_perm.view(M, N, N)    # [M, N, N]   fp16
    right_batch = right_perm.view(M, N, N)  # [M, N, N]   fp16
    out_batch = torch.empty_like(left_batch, dtype=torch.float32)  # [M, N, N]

    # Strides (in elements) for the three tensors
    stride_l_batch, stride_l_i, stride_l_k = left_batch.stride()
    stride_r_batch, stride_r_k, stride_r_j = right_batch.stride()
    stride_o_batch, stride_o_i, stride_o_j = out_batch.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        M
    )

    _tri_mul_kernel[grid](
        left_batch,
        right_batch,
        out_batch,
        M, N, N,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_k, stride_r_j,
        stride_o_batch, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout [B, N, N, H]
    out = out_batch.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # float32

    # -----------------------------------------------------------------
    # 6. Post‑processing: layer‑norm, out‑gate, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate                       # broadcast over hidden dim
    out = F.linear(out, to_out_weight)         # back to original dim

    return out
