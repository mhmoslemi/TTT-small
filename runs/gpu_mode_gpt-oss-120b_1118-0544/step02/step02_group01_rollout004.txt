@triton.jit
def fused_mask_gate_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                           B, N, H,
                           stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
                           stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
                           stride_mask_b, stride_mask_i, stride_mask_j,
                           stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total
    # compute coordinates
    index = offs
    b = index // (N * N * H)
    index = index % (N * N * H)
    i = index // (N * H)
    index = index % (N * H)
    j = index // H
    h = index % H
    # compute offsets
    proj_off = b * stride_proj_b + i * stride_proj_i + j * stride_proj_j + h * stride_proj_h
    gate_off = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_off = b * stride_out_b + i * stride_out_i + j * stride_out_j + h * stride_out_h
    proj = tl.load(proj_ptr + proj_off, mask=mask, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + gate_off, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_off, mask=mask, other=tl.float16(0.0))
    tl.store(out_ptr + out_off, proj * gate * m, mask=mask)

"""
Optimized forward pass of the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Input LayerNorm (FP32).
2. Cast to FP16 and compute all linear projections / gate logits with
   torch.nn.functional.linear (FP16 GEMM).
3. Compute sigmoid gates (FP16).  If a mask is provided, fuse mask, gate and
   projection with a tiny Triton kernel (half‑precision elementwise).
   Otherwise multiply projection and gate with a simple torch elementwise op.
4. Perform the core N³ reduction as a batched FP16 matrix multiplication:
   for each hidden channel we multiply
       left[b, h, i, k] @ right[b, h, k, j]   -> out[b, h, i, j]
   implemented with torch.bmm on the (B*hidden) batch dimension.
5. Convert to FP32, apply the output LayerNorm, multiply by the output gate,
   and finally a FP32 linear projection back to the original channel dimension.
6. Return the result in FP32.

Only the mask‑gate fusion (step 3) is implemented in a custom Triton kernel,
fulfilling the requirement of “at least part of the operations in a kernel”.
All heavy arithmetic (linear layers, matmul) stays on the GPU using cuBLAS.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Elementwise kernel:
        out = proj * gate * mask
    All tensors are in half precision (float16) and have shape
        proj, gate, out : [B, N, N, H]
        mask            : [B, N, N] (broadcasted over H)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # Decode linear index -> (b, i, j, h)
    idx = offs
    b = idx // (N * N * H)
    idx = idx % (N * N * H)
    i = idx // (N * H)
    idx = idx % (N * H)
    j = idx // H
    h = idx % H

    # Compute flat offsets for each tensor
    proj_off = (b * stride_proj_b + i * stride_proj_i
                + j * stride_proj_j + h * stride_proj_h)
    gate_off = (b * stride_gate_b + i * stride_gate_i
                + j * stride_gate_j + h * stride_gate_h)
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_off = (b * stride_out_b + i * stride_out_i
               + j * stride_out_j + h * stride_out_h)

    proj = tl.load(proj_ptr + proj_off, mask=mask, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + gate_off, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_off, mask=mask, other=tl.float16(0.0))

    tl.store(out_ptr + out_off, proj * gate * m, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32) or None
        - weights      : dict of weight tensors
        - config       : dict containing at least "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Cast to FP16 and compute linear projections + gates
    # -----------------------------------------------------------------
    x_fp16 = x.half()

    w_left_proj = weights["left_proj.weight"].half()
    w_right_proj = weights["right_proj.weight"].half()
    w_left_gate = weights["left_gate.weight"].half()
    w_right_gate = weights["right_gate.weight"].half()
    w_out_gate = weights["out_gate.weight"].half()

    # Projections
    left_proj = F.linear(x_fp16, w_left_proj)   # [B, N, N, hidden]
    right_proj = F.linear(x_fp16, w_right_proj)

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_fp16, w_left_gate)).half()
    right_gate = torch.sigmoid(F.linear(x_fp16, w_right_gate)).half()
    out_gate = torch.sigmoid(F.linear(x_fp16, w_out_gate)).half()  # used later

    # -----------------------------------------------------------------
    # 3) Fuse mask, projection and gate (if mask present)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and broadcastable
        mask_fp16 = mask.to(torch.float16)

        # Allocate output buffers
        left = torch.empty_like(left_proj, dtype=torch.float16, device=device)
        right = torch.empty_like(right_proj, dtype=torch.float16, device=device)

        # Strides (in elements)
        s_lp_b, s_lp_i, s_lp_j, s_lp_h = left_proj.stride()
        s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate.stride()
        s_rp_b, s_rp_i, s_rp_j, s_rp_h = right_proj.stride()
        s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate.stride()
        s_m_b, s_m_i, s_m_j = mask_fp16.stride()
        s_out_b, s_out_i, s_out_j, s_out_h = left.stride()

        total = B * N * N * hidden
        BLOCK_SIZE = 1024
        grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE,)

        # left side
        fused_mask_gate_kernel[grid](
            left_proj,
            left_gate,
            mask_fp16,
            left,
            B, N, hidden,
            s_lp_b, s_lp_i, s_lp_j, s_lp_h,
            s_lg_b, s_lg_i, s_lg_j, s_lg_h,
            s_m_b, s_m_i, s_m_j,
            s_out_b, s_out_i, s_out_j, s_out_h,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # right side
        fused_mask_gate_kernel[grid](
            right_proj,
            right_gate,
            mask_fp16,
            right,
            B, N, hidden,
            s_rp_b, s_rp_i, s_rp_j, s_rp_h,
            s_rg_b, s_rg_i, s_rg_j, s_rg_h,
            s_m_b, s_m_i, s_m_j,
            s_out_b, s_out_i, s_out_j, s_out_h,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        # No mask – simple elementwise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 4) Core N³ reduction via batched FP16 matrix multiplication
    # -----------------------------------------------------------------
    # Rearrange tensors to (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 2, 1).contiguous()  # [B, hidden, N, N] (K rows, J cols)

    # Collapse batch & hidden dimensions for torch.bmm
    left_mat = left.view(-1, N, N)   # (B*hidden, N, N)
    right_mat = right.view(-1, N, N) # (B*hidden, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # FP16 result, shape (B*hidden, N, N)

    # Restore original layout [B, N, N, hidden]
    out_hidden = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # -----------------------------------------------------------------
    # 5) Output LayerNorm, gating and final projection (FP32)
    # -----------------------------------------------------------------
    out_hidden_fp32 = out_hidden.float()  # cast back to FP32 for stable norm
    out_hidden_fp32 = F.layer_norm(
        out_hidden_fp32,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Apply output gate (cast to FP32)
    out_hidden_fp32 = out_hidden_fp32 * out_gate.float()

    # Final linear projection back to original dim
    out = F.linear(out_hidden_fp32, weights["to_out.weight"])

    return out
