@triton.jit
def triton_gemm_kernel(a_ptr, b_ptr, c_ptr,
                       M, N, K,
                       stride_a_batch, stride_a_m, stride_a_k,
                       stride_b_batch, stride_b_n, stride_b_k,
                       stride_c_batch, stride_c_m, stride_c_n,
                       BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0) # row tile index
    pid_n = tl.program_id(1) # col tile index
    pid_b = tl.program_id(2) # batch tile idx

    # compute offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute pointers for batch
    a_batch_ptr = a_ptr + pid_b * stride_a_batch
    b_batch_ptr = b_ptr + pid_b * stride_b_batch
    c_batch_ptr = c_ptr + pid_b * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m) + (cur_k[None,:] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (N x K) - notice B enters as (row = j, col = k)
        b_ptrs = b_batch_ptr + (offs_n[:, None] * stride_b_n) + (cur_k[None,:] * stride_b_k)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None,:], other=0.0).to(tl.float32)

        # Accumulate: a (M x K) * b^T (K x N)
        acc += tl.dot(a, tl.trans(b))

    # Store
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
Outgoing Triangular Multiplicative Update (TriMul) optimized with Triton.

Algorithm
---------
1. Layer‑norm over the input channel dimension (dim).
2. Linear projections to hidden_dim for left/right branches.
3. Optional mask (broadcast over hidden_dim) applied to both branches.
4. Gating: sigmoid(linear(...)) for left, right and the final output gate.
5. Fuse gating with the projected tensors and cast to float16.
6. Batched GEMM over the third axis (k) using a Triton kernel:
       out[b*d, i, j] = Σ_k left[b,d,i,k] * right[b,d,j,k]
   where the batch dimension of the kernel is (batch_size * hidden_dim).
7. Reshape the result back to [B, N, N, hidden_dim] (float32).
8. Apply a second LayerNorm, the output gate, and a final linear projection
   back to the original channel dimension.

The heavy reduction (step 6) is fully performed in Triton, satisfying the
requirement to implement a non‑trivial part of the operator in a kernel.
All other tensor operations use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _triton_gemm_kernel(
    a_ptr,                      # left  tensor,   shape (B*H, M, K)   FP16
    b_ptr,                      # right tensor,  shape (B*H, N, K)   FP16 (read as row‑major)
    c_ptr,                      # output tensor, shape (B*H, M, N)   FP32
    M, N, K,                    # problem dimensions (int)
    stride_a_batch, stride_a_m, stride_a_k,   # strides for a
    stride_b_batch, stride_b_n, stride_b_k,   # strides for b
    stride_c_batch, stride_c_m, stride_c_n,   # strides for c
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)            # tile row
    pid_n = tl.program_id(1)            # tile column
    pid_b = tl.program_id(2)            # batch index (B * hidden_dim)

    # ------------------------------------------------------------------
    # Compute offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    num_steps = tl.cdiv(K, BLOCK_K)
    for step in range(num_steps):
        cur_k = step * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load a tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load b tile: (BLOCK_N, BLOCK_K) – rows are j, columns are k
        b_ptrs = b_batch + (offs_n[:, None] * stride_b_n) + (cur_k[None, :] * stride_b_k)
        b = tl.load(b_ptrs,
                    mask=mask_n[:, None] & mask_k[None, :],
                    other=0.0)

        # Accumulate a @ bᵀ  -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, tl.trans(b))

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask: torch.Tensor, shape [B, N, N] (bool or float). May be ignored.
            - weights: dict of torch.Tensors containing the model parameters.
            - config: dict with at least the keys:
                * "dim" (int) – input channel dimension
                * "hidden_dim" (int) – projection dimension
                * "nomask" (bool, optional) – if True, mask is ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Extract weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5. Apply gates and cast to half precision for the GEMM
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Rearrange tensors so that each (batch, hidden) pair forms an
    #    independent GEMM of shape (N, N) @ (N, N)^T
    # ------------------------------------------------------------------
    #   (B, N, N, H) -> (B, H, N, N) -> view as (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left.shape
    left_view = left.view(B * H, N, N)
    right_view = right.view(B * H, N, N)

    # Output buffer (float32)
    out = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7. Launch the Triton batched GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32  # tunable

    # Strides for the three tensors (batch, row, col)
    stride_a_batch, stride_a_m, stride_a_k = left_view.stride()
    stride_b_batch, stride_b_n, stride_b_k = right_view.stride()
    stride_c_batch, stride_c_m, stride_c_n = out.stride()

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _triton_gemm_kernel[grid](
        a_ptr=left_view,
        b_ptr=right_view,
        c_ptr=out,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_n=stride_b_n,
        stride_b_k=stride_b_k,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before using the result
    torch.cuda.synchronize(device)

    # ------------------------------------------------------------------
    # 8. Reshape back to [B, N, N, hidden_dim] (float32)
    # ------------------------------------------------------------------
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 9. Final LayerNorm, output gate, and projection to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # broadcasted over hidden_dim
    out = F.linear(out, to_out_weight)       # [B, N, N, dim]

    return out
