"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The forward pass consists of:
1. LayerNorm on the input tensor.
2. Linear projections (left/right) and three gate projections.
3. Optional binary mask applied to the projected tensors.
4. Gating (sigmoid) applied to left/right.
5. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication where the hidden dimension h is
   folded into the batch axis, i.e. a GEMM of shape (B*H, N, N) ×
   (B*H, N, N)^T.  The GEMM is implemented in Triton with fp16 data,
   fp32 accumulation and Tensor‑Core friendly tiling.
6. LayerNorm on the hidden dimension, multiplication by the output gate,
   and final linear projection back to the original channel dimension.
All heavy arithmetic is performed in fp16; the final result is cast back
to the input dtype (float32 by default).
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # B: (batch, N, K) – accessed as K×N
    stride_ch, stride_cm, stride_cn,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]

    A: (batch, M, K)   fp16
    B: (batch, N, K)   fp16 (will be read as K×N)
    C: (batch, M, N)   fp16, accumulation in fp32
    """
    pid_b = tl.program_id(2)          # batch * hidden
    pid_m = tl.program_id(0)          # tile over M (i)
    pid_n = tl.program_id(1)          # tile over N (j)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Batch offsets
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # Pointers to the current tile
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Triton‑based batched GEMM implementing:
        C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]

    Arguments:
        A, B: fp16 tensors of shape (batch, N, N)
    Returns:
        C: fp16 tensor of shape (batch, N, N)
    """
    assert A.shape == B.shape, "A and B must have identical shapes"
    batch, N, K = A.shape
    assert N == K, "TriMul requires square sequence dimension"

    # Choose block sizes based on N
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:  # up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    C = torch.empty_like(A)  # fp16 output

    _batched_matmul_kernel[grid](
        A, B, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),   # stride_ah, stride_am, stride_ak
        B.stride(0), B.stride(2), B.stride(1),   # stride_bh, stride_bk, stride_bn (note Bᵀ layout)
        C.stride(0), C.stride(1), C.stride(2),   # stride_ch, stride_cm, stride_cn
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (bool/int/float) or None
            - weights: dict of model parameters
            - config: dict with keys:
                "dim" (int) – input channel dimension C
                "hidden_dim" (int) – hidden dimension H
                "nomask" (bool, optional) – if True, ignore mask
    Returns:
        torch.Tensor [B, N, N, C] with the same dtype as the input.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for the heavy compute path
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Cast model weights to fp16
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 4. Optional masking (binary mask broadcasted on the hidden dim)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 5. Gate projections (sigmoid) – computed from the normalized input
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Core O(N³·H) reduction via Triton‑accelerated batched GEMM
    #    Reshape to (B*H, N, N) and compute out = left @ rightᵀ
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Bring hidden dimension next to batch for easy view
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left.view(B * H, N, N)   # (B*H, N, N)  A
    right_flat = right.view(B * H, N, N) # (B*H, N, N)  B

    out_flat = _batched_matmul_triton(left_flat, right_flat)  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7. Output LayerNorm on hidden dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 8. Apply output gate (element‑wise)
    # -----------------------------------------------------------------
    out = out * out_gate

    # -----------------------------------------------------------------
    # 9. Final linear projection back to original channel dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 10. Cast back to the original input dtype if needed
    # -----------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
