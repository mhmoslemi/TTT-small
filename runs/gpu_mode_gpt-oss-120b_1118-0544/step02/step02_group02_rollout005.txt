def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm over last dimension (dim) using weight/bias
    x = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=1e-5)

    # Projections
    # left = x @ left_proj_weight.T
    left = F.linear(x, weights['left_proj.weight'])
    right = F.linear(x, weights['right_proj.weight'])
    
    # Gating
    left_gate = torch.sigmoid(F.linear(x, weights['left_gate.weight']))  # shape B,N,N,hidden
    right_gate = torch.sigmoid(F.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(F.linear(x, weights['out_gate.weight']))
    
    # Multiply gates and mask (optional)
    # Use Triton kernel for elementwise multiplication and optional mask + cast to fp16
    # Prepare mask_f (float32) if needed
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype)  # float32
    else:
        mask_f = None

    # Compute left_h and right_h with gating, mask, cast to fp16
    total_elem = left.numel()
    left_h = torch.empty_like(left, dtype=torch.float16)
    right_h = torch.empty_like(right, dtype=torch.float16)

    # Determine grid and launch kernel
    BLOCK = 32768  # adjust
    grid = ( (total_elem + BLOCK - 1) // BLOCK, )
    gated_proj_kernel[grid](
        left_in_ptr = left,
        left_gate_ptr = left_gate,
        left_out_ptr = left_h,
        right_in_ptr = right,
        right_gate_ptr = right_gate,
        right_out_ptr = right_h,
        mask_ptr = mask_f if mask_f is not None else torch.tensor([], dtype=torch.float32, device=left.device),
        total_elem = total_elem,
        hidden_dim = hidden_dim,
        HAS_MASK = 0 if mask_f is None else 1,
        BLOCK = BLOCK,
        num_warps = 4,
    )
    # Release intermediate tensors to free memory
    del left, right, left_gate, right_gate
    torch.cuda.empty_cache()

    # Compute multiplication across k dimension using batched matrix multiplication
    # Permute to (B, hidden, N, N)
    left_h = left_h.permute(0, 3, 1, 2)  # B, hidden, N, N
    right_h = right_h.permute(0, 3, 1, 2)
    # Matrix multiply: (B, hidden, N, N) * (B, hidden, N, N)^T over last dim
    out_h = torch.matmul(left_h, right_h.transpose(-1, -2))  # B, hidden, N, N, fp16

    # Convert to float32
    out_fp32 = out_h.float()

    # Permute back to (B, N, N, hidden)
    out_fp32 = out_fp32.permute(0, 2, 3, 1).contiguous()

    # Final LayerNorm over hidden_dim
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights['to_out_norm.weight'],
        bias=weights['to_out_norm.bias'],
        eps=1e-5,
    )
    # Apply output gate
    out_gate = out_gate  # shape B,N,N,hidden
    out = out_norm * out_gate

    # Final linear projection to original dim
    out = F.linear(out, weights['to_out.weight'])

    return out

"""
Highly optimized outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the input tensor on the last dimension.
2. Linear projections to a hidden dimension (no bias) for left/right.
3. Compute left‑gate, right‑gate and out‑gate via linear + sigmoid.
4. Fuse mask (if present), gate multiplication and FP16 casting in a
   custom Triton kernel (produces left_h, right_h of dtype float16).
5. Batched matrix multiplication:
   * permute left_h/right_h to (B, hidden, N, N)
   * out_h = torch.matmul(left_h, right_h.transpose(-1, -2))
   (this is the einsum “… i k d, … j k d → … i j d”)
6. Convert the FP16 result to FP32, permute back to (B, N, N, hidden).
7. Layer‑norm over the hidden dimension, apply the out‑gate and final
   linear projection back to the original dim.

Only the element‑wise gating+mask+cast step is implemented in Triton;
all heavy matrix multiplications use PyTorch’s highly tuned cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def gated_proj_kernel(
    left_in_ptr,       # float32, [total]
    left_gate_ptr,     # float32, [total]
    left_out_ptr,      # float16, [total]
    right_in_ptr,      # float32, [total]
    right_gate_ptr,    # float32, [total]
    right_out_ptr,     # float16, [total]
    mask_ptr,          # float32, [mask_total] (broadcasted)
    total: tl.int32,   # total number of elements in left/right tensors
    hidden_dim: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # Load inputs
    l_val = tl.load(left_in_ptr + offs, mask=mask, other=0.0)
    lg_val = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r_val = tl.load(right_in_ptr + offs, mask=mask, other=0.0)
    rg_val = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # mask is stored without the hidden dimension, so broadcast by integer division
        mask_idx = offs // hidden_dim
        m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        l_val = l_val * lg_val * m_val
        r_val = r_val * rg_val * m_val
    else:
        l_val = l_val * lg_val
        r_val = r_val * rg_val

    # Cast to float16 and store
    tl.store(left_out_ptr + offs, l_val.to(tl.float16), mask=mask)
    tl.store(right_out_ptr + offs, r_val.to(tl.float16), mask=mask)


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – may be None
            - weights: dict of model weights (see skeleton)
            - config: dict with keys:
                  * "dim" (int)
                  * "hidden_dim" (int)
                  * "nomask" (bool, optional)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 0) LayerNorm on the input
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 1) Linear projections (bias‑free)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 2) Gating (sigmoid of linear layers)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 3) Fuse mask, gating, and cast to FP16 via Triton
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype)  # float32 broadcastable over hidden_dim
        has_mask = 1
    else:
        # dummy mask (won't be read because HAS_MASK=0)
        mask_f = torch.tensor([1.0], device=x.device, dtype=x.dtype)
        has_mask = 0

    total_elem = left.numel()  # B * N * N * hidden_dim
    left_h = torch.empty_like(left, dtype=torch.float16)
    right_h = torch.empty_like(right, dtype=torch.float16)

    BLOCK = 65536
    grid = ((total_elem + BLOCK - 1) // BLOCK,)

    gated_proj_kernel[grid](
        left_in_ptr=left,
        left_gate_ptr=left_gate,
        left_out_ptr=left_h,
        right_in_ptr=right,
        right_gate_ptr=right_gate,
        right_out_ptr=right_h,
        mask_ptr=mask_f,
        total=total_elem,
        hidden_dim=hidden_dim,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # Release memory that is no longer needed
    del left, right, left_gate, right_gate

    # --------------------------------------------------------------
    # 4) Batched matrix multiplication (the core TriMul reduction)
    #    Shape: (B, hidden_dim, N, N) @ (B, hidden_dim, N, N).T
    # --------------------------------------------------------------
    left_h = left_h.permute(0, 3, 1, 2)   # [B, hidden_dim, N, N]
    right_h = right_h.permute(0, 3, 1, 2) # [B, hidden_dim, N, N]

    # torch.matmul performs batched GEMM over the last two dims
    out_h = torch.matmul(left_h, right_h.transpose(-1, -2))  # [B, hidden_dim, N, N] (fp16)

    # Back to (B, N, N, hidden_dim) in FP32
    out_fp32 = out_h.permute(0, 2, 3, 1).contiguous().float()

    # --------------------------------------------------------------
    # 5) Final LayerNorm, gating and projection back to original dim
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out_norm * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
