def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for M,N within bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute base pointer for this batch
    a_batch_ptr = a_ptr + batch_idx * stride_a_batch
    b_batch_ptr = b_ptr + batch_idx * stride_b_batch
    c_batch_ptr = c_ptr + batch_idx * stride_c_batch

        mask_k = offs_k < K

    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,

    # offsets

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

The kernel fuses the expensive N³ reduction as a batched half‑precision
matrix multiplication using Triton. All linear projections, gating, and the
final LayerNorm/linear layers are performed in PyTorch.  The implementation
supports optional masking and runs entirely on the GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched half‑precision GEMM (A @ Bᵀ) over
# (batch, M=N, K=N, N=N).  Accumulates in fp32 and stores fp32.
# --------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    BATCH, M, N, K,                    # dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # base pointers for this batch
    a_batch = a_ptr + batch_idx * stride_a_batch
    b_batch = b_ptr + batch_idx * stride_b_batch
    c_batch = c_ptr + batch_idx * stride_c_batch

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        b_ptrs = b_batch + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # promote to fp32 for accumulation (TensorCore will still be used)
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)       # (BLOCK_M, BLOCK_N)

    # write out
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Entry point expected by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model parameters (float32)
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2) Cast to half for projection / gating work
    # --------------------------------------------------------------
    x_h = x.half()

    # cast projection / gate weights to half once
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    # --------------------------------------------------------------
    # 3) Linear projections (half)
    # --------------------------------------------------------------
    left_proj = F.linear(x_h, left_proj_w)      # [B, N, N, hidden_dim]
    right_proj = F.linear(x_h, right_proj_w)    # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 4) Gating (half)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 5) Apply mask (if any) and combine projection + gate
    # --------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask_h = mask.unsqueeze(-1).half()
        left = left_proj * left_gate * mask_h
        right = right_proj * right_gate * mask_h
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6) Prepare tensors for the batched matmul:
    #    left   : [B, H, N, N]  (i, k)   -> A
    #    right  : [B, H, N, N]  (j, k)   -> B, we need Bᵀ (k, j)
    # --------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)
    right_T = right_perm.permute(0, 1, 3, 2).contiguous()  # (B, H, N, N) where dims are (b,d,k,j)

    # flatten batch*hidden into a single batch dimension
    batch = B * hidden_dim
    left_flat = left_perm.view(batch, N, N).contiguous()   # (batch, M=N, K=N)
    right_flat = right_T.view(batch, N, N).contiguous()   # (batch, K=N, N)

    # --------------------------------------------------------------
    # 7) Batched half‑precision GEMM via Triton
    # --------------------------------------------------------------
    out_hidden = torch.empty((batch, N, N), dtype=torch.float32, device=x.device)

    # block sizes tuned for Tensor Cores (FP16)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_hidden,
        batch,
        N,
        N,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_hidden.stride(0),
        out_hidden.stride(1),
        out_hidden.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8) Restore original shape [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    out_hidden = (
        out_hidden.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden_dim]  (float32)

    # --------------------------------------------------------------
    # 9) Output LayerNorm, gating and final linear
    # --------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate was half – promote to float32 for the multiplication
    out_hidden = out_hidden * out_gate.float()

    # final projection back to dim
    out = F.linear(out_hidden, weights["to_out.weight"])  # [B, N, N, dim]

    return out
