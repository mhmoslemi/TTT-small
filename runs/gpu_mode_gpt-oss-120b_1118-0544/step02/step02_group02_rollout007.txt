"""
Outgoing Triangle Multiplicative Update (TriMul) – fast forward pass.

Algorithm
---------
1. Layer‑Norm the input pair representation (float32).
2. Cast to fp16 and compute the four linear projections (left, right, left‑gate,
   right‑gate, out‑gate) with fp16 weights.
3. Apply the optional mask (fp16) to the left/right projections.
4. Fuse the gating with a tiny Triton element‑wise kernel:
        left  = left  * left_gate
        right = right * right_gate
5. Perform the heavy reduction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   as a batched fp16 matrix‑multiply (torch.bmm) after reshaping.
6. Convert the reduction result to fp32, apply the output LayerNorm,
   multiply by the output gate, and finally linearly project back to the
   original feature dimension.

Only the element‑wise gating step is implemented in Triton; all other
operations use highly‑optimized PyTorch kernels (fp16 GEMM, torch.bmm,
torch.nn.functional.layer_norm, etc.) to achieve sub‑millisecond runtimes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused element‑wise multiplication (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def mul_kernel(
    a_ptr,            # input tensor A (fp16)
    b_ptr,            # input tensor B (fp16)
    out_ptr,          # output tensor (can be same as a_ptr)
    N_ELEM: tl.constexpr,   # total number of elements
    BLOCK: tl.constexpr,    # block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) – may be None
            - weights: dict with all model weight tensors (float32)
            - config: dict containing at least "dim" and "hidden_dim",
                      optionally "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()

    # ------------------------------------------------------------------
    # 3) Linear projections (bias‑free, fp16 weights)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, left_proj_weight.half())   # [B, N, N, hidden]
    right = F.linear(x_fp16, right_proj_weight.half())

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(x_fp16.dtype).unsqueeze(-1)           # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5) Gating values (sigmoid, fp16)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, left_gate_weight.half()).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_weight.half()).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_weight.half()).sigmoid()   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 6) Fuse gate multiplication with a Triton kernel
    # ------------------------------------------------------------------
    BLOCK = 65536  # ~64 Ki elements per program
    # left = left * left_gate (in‑place)
    total_elem = left.numel()
    grid = ((total_elem + BLOCK - 1) // BLOCK,)
    mul_kernel[grid](
        left, left_gate, left,
        N_ELEM=total_elem,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # right = right * right_gate (in‑place)
    total_elem = right.numel()
    grid = ((total_elem + BLOCK - 1) // BLOCK,)
    mul_kernel[grid](
        right, right_gate, right,
        N_ELEM=total_elem,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Batched matmul: out_h[b*h, i, j] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    (fp16 → fp32 accumulation inside cuBLAS)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # fp16 matmul, tensor cores accumulate in fp32 internally
    out_h = torch.bmm(left_flat, right_flat.transpose(1, 2))   # [B*H, N, N]

    # reshape back to [B, N, N, H] (still fp16)
    out_h = out_h.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Cast reduction result to float32 for the remaining ops
    # ------------------------------------------------------------------
    out = out_h.float()   # [B, N, N, hidden] float32

    # ------------------------------------------------------------------
    # 9) Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate (float32) and final projection
    # ------------------------------------------------------------------
    out = out * out_gate.float()
    out = F.linear(out, to_out_weight)   # [B, N, N, dim] float32

    return out
