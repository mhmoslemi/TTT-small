"""TriMul (outgoing) forward pass – Triton implementation.

The core operation is a batched matrix multiplication:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
where `left` and `right` have shape (B, N, N, hidden_dim).
We collapse the batch and hidden dimension into a single “head” dimension
BH = B * hidden_dim and compute a GEMM for each head:
    out_head = left_head @ right_headᵀ
using a custom Triton kernel with mixed‑precision (fp16 inputs,
fp32 accumulation). All surrounding linear layers, LayerNorm and gating
are performed with PyTorch. The kernel tiles the M (i) and N (j) axes
with 64×64 blocks and reduces over K (the sequence dimension) in 32‑wide
tiles. This balances register pressure and occupancy on modern GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                      # pointers
    M, N, K,                                  # matrix dimensions (int32)
    stride_ah, stride_am, stride_ak,          # strides for A: (head, i, k)
    stride_bh, stride_bk, stride_bn,          # strides for B: (head, k, j)
    stride_ch, stride_cm, stride_cn,          # strides for C: (head, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ with fp16 inputs and fp32 accumulation.

    A: (head, M, K)   fp16
    B: (head, K, N)   fp16 (already transposed)
    C: (head, M, N)   fp32
    """
    pid_head = tl.program_id(0)   # which (batch * hidden_dim) we process
    pid_m    = tl.program_id(1)   # block row index
    pid_n    = tl.program_id(2)   # block column index

    # Compute offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this head
    a_head_ptr = a_ptr + pid_head * stride_ah
    b_head_ptr = b_ptr + pid_head * stride_bh
    c_head_ptr = c_ptr + pid_head * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Mask out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Pointers to the current tiles
        a_ptrs = a_head_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_head_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (fp16) with zero padding for OOB elements
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        # Matrix‑multiply‑accumulate
        acc += tl.dot(a, b)   # dot returns fp16, accumulated into fp32

    # Write the result back (fp32)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_head_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with the following keys (all float32 tensors):
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config: dict with entries 'dim', 'hidden_dim', and optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1) Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden_dim (cast to fp16 for speed)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)   # half precision for the heavy GEMM path

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim] fp16
    right = F.linear(x_h, right_proj_w)  # same dtype

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcasted over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5) Gates (computed in FP32, then cast to fp16 for multiplication)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # FP32

    left_gate_h = left_gate.to(x_h.dtype)
    right_gate_h = right_gate.to(x_h.dtype)

    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 6) Core N³ contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape   # B, seq_len, seq_len, hidden_dim

    # Rearrange to (B, hidden_dim, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) – K axis transposed

    # Collapse batch and head dimensions for the kernel
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # (BH, M, K)
    right_mat = right_perm.view(BH, N, N)  # (BH, K, N)

    # Output buffer (float32 accumulator)
    out_mat = torch.empty((BH, N, N), dtype=torch.float32, device=x.device)

    # Tunable block sizes – chosen to stay within register budget
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,                                 # head dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # rows
        (N + BLOCK_N - 1) // BLOCK_N,       # cols
    )

    _tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # element‑wise gating (float32)
    out = F.linear(out, weights["to_out.weight"])   # back to dim (float32)

    return out
