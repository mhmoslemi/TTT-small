def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Cast to fp16
    x = x.to(torch.float16)

    # Cast weights to fp16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # Projections
    left = torch.nn.functional.linear(x, left_proj_w)  # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)

    # Mask
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Core multiplication via Triton kernel
    out = _tri_mul_kernel_triton(left, right)   # shape (B,N,N,H)

    # Output layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    A_ptr, B_ptr, C_ptr,
    batch, N, H,
    stride_batch, stride_row, stride_col, stride_hidden,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

"""
custom_kernel: Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update).

Algorithm
---------
1. Layer‑norm the input (B,N,N,C) using provided LN parameters.
2. Cast to fp16 and compute linear projections/gates (left, right, left_gate,
   right_gate, out_gate) with torch (highly optimized GEMM).
3. Apply the optional pairwise mask and pointwise sigmoid gates.
4. Core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM with a custom Triton kernel that works on the
   original (B,N,N,H) layout, avoiding any data copies.  The kernel loops over
   K‑tiles, loads FP16 tiles, accumulates in FP32 via `tl.dot` (Tensor‑Core
   friendly), and writes the result back in FP16.
5. Apply a second LayerNorm, multiply by the output gate and a final linear
   projection back to the original channel dimension.
6. Cast the result back to the input dtype (float32 in the reference tests).

The implementation keeps all heavy compute on the GPU, uses mixed‑precision for
speed, and respects the required Triton 3.3.1 API.
"""
import torch
import triton
import triton.language as tl

# ---------------------------------------------------------
# Triton kernel for the batched matrix multiplication used in TriMul.
# ---------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    batch, N, H,                             # sizes
    stride_batch, stride_row, stride_col, stride_h,   # strides (same for A,B,C)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C[b,i,j,h] = Σ_k A[b,i,k,h] * B[b,j,k,h] for all batches (b) and heads (h).

    A, B, C are fp16 tensors stored with strides:
        stride_batch : stride for the batch dimension (dim 0)
        stride_row   : stride for the first N dimension (dim 1)
        stride_col   : stride for the second N dimension (dim 2)
        stride_h     : stride for the hidden dimension (dim 3)
    The kernel treats the (batch, hidden) pair as a single logical batch of
    size batch*H.
    """
    pid_bh = tl.program_id(2)   # combined batch*hidden id
    pid_m  = tl.program_id(0)   # i‑tile
    pid_n  = tl.program_id(1)   # j‑tile

    # -----------------------------------------------------------------
    # Decompose the combined id into (batch, hidden) indices.
    # Both batch and H are run‑time values, so use integer division.
    # -----------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh - b * H          # pid_bh % H

    # Base pointer for this (b,h) slice
    a_base = A_ptr + b * stride_batch + h * stride_h
    b_base = B_ptr + b * stride_batch + h * stride_h
    c_base = C_ptr + b * stride_batch + h * stride_h

    # Tile offsets for the i and j dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to the current tile of A (i,k) and B (k,j)
    a_ptrs = a_base + offs_i[:, None] * stride_row + tl.arange(0, BLOCK_K)[None, :] * stride_col
    b_ptrs = b_base + tl.arange(0, BLOCK_K)[:, None] * stride_col + offs_j[None, :] * stride_row

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_tiles):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)                # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_col
        b_ptrs += BLOCK_K * stride_col

    # Write back the result in fp16
    c = acc.to(tl.float16)
    c_ptrs = c_base + offs_i[:, None] * stride_row + offs_j[None, :] * stride_col
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel.
    left, right: fp16 tensors of shape (B, N, N, H)
    Returns a fp16 tensor of the same shape containing the multiplication result.
    """
    assert left.shape == right.shape
    B, N, N2, H = left.shape
    assert N == N2, "TriMul requires square sequence dimension."

    # Output tensor (same layout as inputs, no extra copy)
    out = torch.empty_like(left)

    # Strides of the (B,N,N,H) layout
    stride_batch = left.stride(0)
    stride_row   = left.stride(1)   # stride for the first N dimension (i / j)
    stride_col   = left.stride(2)   # stride for the second N dimension (k)
    stride_h     = left.stride(3)   # hidden dimension stride (usually 1)

    # Heuristic block sizes (tuned for N up to 1024)
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_batch,
        stride_row,
        stride_col,
        stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ---------------------------------------------------------
# Public entry point expected by the evaluation harness.
# ---------------------------------------------------------
def custom_kernel(data):
    """
    Implements the forward pass of the outgoing TriMul module using Triton.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape (B, N, N, C), float32 or float16
            - mask_tensor : torch.Tensor of shape (B, N, N) or None
            - weights_dict: dict containing the model parameters (float32)
            - config_dict : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (float32 -> float16 for the heavy path)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2️⃣ Cast all model weights to fp16
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Linear projections (producing (B,N,N,H) tensors)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 4️⃣ Optional masking (binary mask broadcasted over the hidden dim)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 5️⃣ Gate projections (sigmoid) and gating of the projected tensors
    # -----------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Core O(N³·H) reduction via Triton
    # -----------------------------------------------------------------
    out = _tri_mul_triton(left, right)   # shape (B,N,N,H), fp16

    # -----------------------------------------------------------------
    # 7️⃣ Output LayerNorm, output gating and final projection back to C
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to the original dtype (float32 if needed)
    # -----------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
