Implementation of custom_kernel:

```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1) # shape B,N,N,1
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate

    # compute core out via Triton GEMM
    # cast to half
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape
    # Permute to (B, hidden_dim, N, N) and collapse batch * hidden_dim
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()
    # Now shape: (B, hidden_dim, N, N)
    batch_h = B * hidden_dim
    left_h = left_h.view(batch_h, N, N)
    right_h = right_h.view(batch_h, N, N)

    # allocate output buffer
    out_h = torch.empty((batch_h, N, N), dtype=torch.float32, device=device)

    # Determine block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # compute grid dims
    num_block_m = (N + BLOCK_M - 1) // BLOCK_M
    num_block_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_block_m, num_block_n, batch_h)

    # strides
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)
    stride_b_batch = right_h.stride(0)
    stride_b_n = right_h.stride(1)  # N dim stride
    stride_b_k = right_h.stride(2)  # K dim stride
    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    tri_mul_gemm_kernel[grid](
        left_h,
        right_h,
        out_h,
        batch_h,
        N,
        N,
        N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # out_h is currently shape (batch_h, N, N) float32
    # reshape back to [B, hidden_dim, N, N]
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous() # shape B,N,N,hidden_dim

    # to_out_norm (layernorm)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # gating
    out = out * out_gate
    # final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
```

"""
Outgoing Triangle Multiplicative Update (TriMul) kernel.

This implementation follows the PyTorch reference but uses a custom Triton
kernel for the core batched matrix multiplication:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

The heavy reduction over the sequence dimension is performed as a
batched GEMM (C = A @ Bᵀ) where each hidden dimension channel is treated as
a separate batch.  The kernel works on FP16 inputs with FP32 accumulation
and writes the result in FP32.  All other operations (LayerNorm, linear
projections, gating) are handled in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    a_ptr,  # left  tensor pointer  (batch, M, K)   fp16
    b_ptr,  # right tensor pointer  (batch, N, K)   fp16 (will be read transposed)
    c_ptr,  # output tensor pointer (batch, M, N)   fp32
    batch: tl.constexpr,  # total number of GEMMs = B * hidden_dim
    M: tl.constexpr,      # rows of A
    N: tl.constexpr,      # rows of B (cols of output)
    K: tl.constexpr,      # reduction dimension
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # row tile
    pid_n = tl.program_id(1)  # col tile
    pid_b = tl.program_id(2)  # batch tile

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the current batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k)
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # Load Bᵀ tile (K x N)
        b_ptrs = b_batch + (cur_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # Accumulate (dot returns fp16, cast to fp32)
        acc += tl.dot(a, b).to(tl.float32)

    # Store result
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float)
            - weights: dict of model parameters (see key list below)
            - config: dict with keys:
                * "dim" (int) – input feature dimension
                * "hidden_dim" (int) – internal hidden dimension
                * "nomask" (bool, optional) – True if mask should be ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)    # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul via Triton batched GEMM
    #    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Cast to half for Tensor‑core acceleration
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, N, N, hidden_dim

    # Rearrange to (batch, hidden_dim, N, N) and flatten batch*hidden_dim
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()
    batch_h = B * hidden_dim
    left_h = left_h.view(batch_h, N, N)   # [B*H, M, K]
    right_h = right_h.view(batch_h, N, N) # [B*H, N, K]

    # Output buffer (FP32)
    out_h = torch.empty((batch_h, N, N), dtype=torch.float32, device=device)

    # Tile sizes – chosen to fit typical B200 resources
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid dimensions
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_m, grid_n, batch_h)

    # Strides
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)
    stride_b_batch = right_h.stride(0)
    stride_b_n = right_h.stride(1)   # original N dimension
    stride_b_k = right_h.stride(2)   # K dimension (inner)
    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    # Launch the kernel
    tri_mul_gemm_kernel[grid](
        left_h,
        right_h,
        out_h,
        batch_h,
        N,          # M
        N,          # N
        N,          # K
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # back to [B, N, N, dim]

    return out
