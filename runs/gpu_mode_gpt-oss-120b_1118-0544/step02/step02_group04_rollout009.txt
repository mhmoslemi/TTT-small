"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections (left/right) are computed in float16 for speed.
3. Gating activations (left_gate, right_gate, out_gate) are computed in
   float32, then cast to half and fused with the projections via a tiny
   Triton element‑wise kernel.
4. Optional mask (if provided) is applied in float16 before the gating.
5. The hidden dimension is folded into the batch axis:
       (B, N, N, H) → (B*H, N, N)
6. Batched matrix multiplication `out = left @ rightᵀ` is performed with
   `torch.bmm` (half‑precision, cuBLAS) which is far faster than a custom
   reduction kernel.
7. The result is reshaped back to (B, N, N, H), converted to float32,
   normalized, multiplied by the output gate and finally projected back
   to the original channel dimension.
8. Only the element‑wise gating fusion is implemented in Triton, satisfying
   the requirement to use a custom kernel while keeping the heavy O(N³·H)
   work on the highly‑optimized cuBLAS path.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise fusion of projection and gate (and optional mask)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_kernel(
    proj_ptr,      # fp16 projected tensor
    gate_ptr,      # fp16 gate tensor
    mask_ptr,      # fp16 mask tensor (broadcasted), may be NULL when no mask
    out_ptr,       # fp16 output tensor
    total_elems,   # number of elements in the tensors
    HAS_MASK: tl.constexpr,   # 1 if mask is present, else 0
    BLOCK: tl.constexpr       # block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    proj = tl.load(proj_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)

    if HAS_MASK:
        m = tl.load(mask_ptr + offs, mask=mask)
        out = proj * gate * m
    else:
        out = proj * gate

    tl.store(out_ptr + offs, out, mask=mask)

def _apply_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor | None = None) -> torch.Tensor:
    """
    Fuse projection, gate and optional mask using a Triton kernel.
    All inputs must be contiguous and of dtype torch.float16.
    """
    assert proj.shape == gate.shape
    total = proj.numel()
    out = torch.empty_like(proj)
    BLOCK = 1024
    grid = ( (total + BLOCK - 1) // BLOCK, )

    has_mask = 1 if mask is not None else 0
    mask_ptr = mask if mask is not None else proj  # dummy, will be ignored when HAS_MASK=0

    _apply_gate_kernel[grid](
        proj, gate, mask_ptr, out,
        total,
        HAS_MASK=has_mask,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation using a hybrid Triton/PyTorch strategy.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32 or bool) or None
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # LayerNorm on the input
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, C), float32

    # --------------------------------------------------------------
    # Linear projections (half precision for speed)
    # --------------------------------------------------------------
    x_fp16 = x.half()
    left_proj = torch.nn.functional.linear(
        x_fp16, weights["left_proj.weight"].half()
    )  # (B, N, N, H), fp16
    right_proj = torch.nn.functional.linear(
        x_fp16, weights["right_proj.weight"].half()
    )  # (B, N, N, H), fp16

    # --------------------------------------------------------------
    # Optional masking (applied before gating)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] broadcast to hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # --------------------------------------------------------------
    # Gating (computed in float32, then cast)
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left_gate_fp16 = left_gate.half().contiguous()
    right_gate_fp16 = right_gate.half().contiguous()

    # --------------------------------------------------------------
    # Fuse projection + gate (Triton kernel)
    # --------------------------------------------------------------
    left = _apply_gate(left_proj, left_gate_fp16)   # (B, N, N, H), fp16
    right = _apply_gate(right_proj, right_gate_fp16)  # (B, N, N, H), fp16

    # --------------------------------------------------------------
    # Reshape for batched GEMM: fold hidden dimension into batch axis
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_flat = left.view(B * H, N, N)               # (B*H, N, N)
    # right needs to be transposed on the last two dims for left @ rightᵀ
    right_flat_T = right.view(B * H, N, N).transpose(1, 2).contiguous()  # (B*H, N, N)

    # --------------------------------------------------------------
    # Batched matrix multiplication (cuBLAS via torch.bmm) – fast half‑precision
    # --------------------------------------------------------------
    out_flat = torch.bmm(left_flat, right_flat_T)   # (B*H, N, N), fp16

    # --------------------------------------------------------------
    # Reshape back to (B, N, N, H) and convert to float32 for the rest
    # --------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H), fp16
    out = out.float()

    # --------------------------------------------------------------
    # Output LayerNorm, gate, and final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate  # broadcast gate (float32)

    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # (B, N, N, C), float32

    return out
