```
def triton_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_k, stride_right_j,
    stride_out_bh, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block col

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k_base = 0

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # compute pointers
        left_ptrs = left_ptr + pid_bh * stride_left_bh + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
```

```
@triton.jit
def triton_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_k, stride_right_j,
    stride_out_bh, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Compute pointers
        left_ptrs = left_ptr + pid_bh * stride_left_bh + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        right_ptrs = right_ptr + pid_bh * stride_right_bh + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j

        # Load tiles with appropriate masks
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Dot product
        acc += tl.dot(left, right)

    # Store
    out_ptrs = out_ptr + pid_bh * stride_out_bh + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul outgoing update: Fuses mask/gating, uses a custom Triton kernel for the
N³ batched matrix multiplication (left @ rightᵀ) and performs the remaining
operations (LayerNorms, linear projections) with PyTorch. The kernel works on
the flattened batch‑hidden dimension and computes a FP16 matmul with FP32
accumulation, returning a float32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton batched matrix multiplication kernel
# --------------------------------------------------------------
@triton.jit
def triton_matmul_kernel(
    left_ptr,      # const pointer to left matrix (BH,N,N) in fp16
    right_ptr,     # const pointer to right matrix (BH,N,N) in fp16 (k,j)
    out_ptr,       # const pointer to output matrix (BH,N,N) in fp16
    N,             # sequence length (int)
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_k, stride_right_j,
    stride_out_bh,   stride_out_i,   stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b, i, j] = sum_k left[b, i, k] * right[b, k, j]
    where b ∈ [0, BH) is the flattened batch‑hidden dimension.
    """
    pid_bh = tl.program_id(0)   # flattened batch*hidden index
    pid_m  = tl.program_id(1)   # block row (i)
    pid_n  = tl.program_id(2)   # block column (j)

    # offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows/cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers to the tiles
        left_tile_ptr  = left_ptr \
            + pid_bh * stride_left_bh \
            + offs_m[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k

        right_tile_ptr = right_ptr \
            + pid_bh * stride_right_bh \
            + offs_k[:, None] * stride_right_k \
            + offs_n[None, :] * stride_right_j

        # load tiles with proper masking
        left_tile  = tl.load(left_tile_ptr,
                             mask=(mask_m[:, None] & mask_k[None, :]),
                             other=0.0)
        right_tile = tl.load(right_tile_ptr,
                             mask=(mask_k[:, None] & mask_n[None, :]),
                             other=0.0)

        # inner product
        acc += tl.dot(left_tile, right_tile)   # FP16*FP16->FP32

    # write back the result
    out_tile_ptr = out_ptr \
        + pid_bh * stride_out_bh \
        + offs_m[:, None] * stride_out_i \
        + offs_n[None, :] * stride_out_j

    tl.store(out_tile_ptr,
             acc.to(tl.float16),
             mask=(mask_m[:, None] & mask_n[None, :]))

    hidden = cfg["hidden_dim"]

        s_right_bh, s_right_k, s_right_j,
        s_out_bh, s_out_i, s_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6) Restore shape to [B, N, N, hidden] and convert to FP32
    # -----------------------------------------------------------------

        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict containing model weights, keys:
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config       : dict with at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # --------------------------------------------------------------
    # 2) Linear projections and gates (float32)
    # --------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 3) Apply mask (if any) and combine projection with gating
    # --------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left_proj * left_gate * mask_f
        right = right_proj * right_gate * mask_f
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 4) Cast to FP16 and reshape for the batched matmul
    # --------------------------------------------------------------
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # bring hidden dimension to the second axis: (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # (i,k)
    right = right.permute(0, 3, 2, 1).contiguous() # (k,j)

    BH = B * hidden
    left = left.view(BH, N, N)     # (BH, i, k)
    right = right.view(BH, N, N)   # (BH, k, j)

    # allocate output buffer in FP16
    out_fp16 = torch.empty_like(left)

    # strides (in elements) for the contiguous tensors
    s_left_bh, s_left_i, s_left_k = left.stride()
    s_right_bh, s_right_k, s_right_j = right.stride()
    s_out_bh, s_out_i, s_out_j = out_fp16.stride()

    # --------------------------------------------------------------
    # 5) Triton batched GEMM: out = left @ rightᵀ
    # --------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    triton_matmul_kernel[grid](
        left,
        right,
        out_fp16,
        N,
        s_left_bh, s_left_i, s_left_k,
        s_right_bh, s_right_k, s_right_j,
        s_out_bh, s_out_i, s_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6) Restore shape to [B, N, N, hidden] and convert to FP32
    # --------------------------------------------------------------
    out_hidden = out_fp16.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # --------------------------------------------------------------
    # 7) Output LayerNorm, gate and final projection
    # --------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
