def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # projections
    left = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # mask
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # broadcast
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))  # shape same
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # cast to half for kernel
    left_h = left.half()
    right_h = right.half()

    B, N, _, _ = left_h.shape  # L shape [B,N,N,hidden]

    # permute to (B, hidden, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()
    # transpose right for K,N layout
    right_h_t = right_h.transpose(-2, -1).contiguous()

    batch = B * hidden
    # reshape to (batch, N, N)
    left_h = left_h.view(batch, N, N)
    right_h_t = right_h_t.view(batch, N, N)

    # allocate output
    out_h = torch.empty((batch, N, N), dtype=torch.float32, device=device)

    # strides
    stride_a_batch, stride_a_m, stride_a_k = left_h.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_h_t.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_h.stride()

    # block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # compute grid
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_m, grid_n, batch)

    batch_matmul_kernel[grid](
        left_h, right_h_t, out_h,
        batch=batch,
        M=N,
        N=N,
        K=N,
        stride_az=stride_a_batch,
        stride_am=stride_a_m,
        stride_ak=stride_a_k,
        stride_bz=stride_b_batch,
        stride_bk=stride_b_k,
        stride_bn=stride_b_n,
        stride_cz=stride_c_batch,
        stride_cm=stride_c_m,
        stride_cn=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape output
    out = out_h.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,hidden]

    # final layernorm
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)
    return out
```

def batch_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,

    pid_n = tl.program_id(1)  # col block

    # offset for rows and cols
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

    # base pointers for this batch

"""
Outgoing Triangle Multiplicative Update (TriMul) kernel.

The module performs:
1. LayerNorm over the input channel dimension.
2. Two linear projections with optional mask.
3. Gated element‑wise multiplication.
4. Batched matrix multiplication
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   implemented as a Triton kernel (batch * hidden GEMMs of size N×N).
5. LayerNorm + gating + final linear projection back to the original dimension.

All operations are performed in float32 except the large N×N GEMM which
uses fp16 for the operands and fp32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batch_matmul_kernel(
    A_ptr,               # left tensor (batch, M, K)    fp16
    B_ptr,               # right tensor (batch, K, N)   fp16
    C_ptr,               # output tensor (batch, M, N) fp32
    batch: tl.constexpr, # total number of independent GEMMs = B * hidden_dim
    M: tl.constexpr,     # rows of A (seq_len)
    N: tl.constexpr,     # cols of B (seq_len)
    K: tl.constexpr,     # reduction dim (seq_len)
    stride_az, stride_am, stride_ak,   # strides for A
    stride_bz, stride_bk, stride_bn,   # strides for B
    stride_cz, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)               # batch index
    pid_m = tl.program_id(0)               # tile row
    pid_n = tl.program_id(1)               # tile column

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = A_ptr + pid_z * stride_az
    b_batch = B_ptr + pid_z * stride_bz
    c_batch = C_ptr + pid_z * stride_cz

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Pointers to the first K‑block
    a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

    # Loop over K dimension
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = _k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # (BLOCK_M, BLOCK_K) fp16
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # (BLOCK_K, BLOCK_N) fp16

        acc += tl.dot(a, b).to(tl.float32)

        # Advance to next K‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back the result
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float). May be ignored.
            - weights: dict of model parameters (see code below)
            - config: dict containing at least:
                * "dim" (int) – input channel dimension
                * "hidden_dim" (int) – intermediate channel dimension
                * "nomask" (bool, optional) – if True, mask is ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # Unpack weights
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1) LayerNorm over the last dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4) Gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Batched GEMM: out = left @ rightᵀ  (over the K dimension)
    #    We implement the GEMM with a Triton kernel for speed.
    # --------------------------------------------------------------
    # Cast to half for the kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, N, N, hidden_dim

    # Rearrange to (batch, hidden, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_h = right_h.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]

    # Transpose right so that we can multiply A (i,k) with B (k,j)
    right_h_t = right_h.transpose(-2, -1).contiguous()  # now shape (B, hidden, N, N) where
                                                         # axis -2 is K, axis -1 is N

    # Merge batch and hidden dimensions
    batch = B * hidden_dim
    left_h = left_h.view(batch, N, N)      # [batch, M=N, K=N]
    right_h_t = right_h_t.view(batch, N, N)  # [batch, K=N, N]

    # Allocate output buffer (fp32 accumulation)
    out_h = torch.empty((batch, N, N), dtype=torch.float32, device=device)

    # Strides for the kernel
    stride_a_batch, stride_a_m, stride_a_k = left_h.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_h_t.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_h.stride()

    # Block sizes – chosen to give good occupancy on typical GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid dimensions
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_m, grid_n, batch)

    # Launch Triton kernel
    batch_matmul_kernel[grid](
        left_h,
        right_h_t,
        out_h,
        batch=batch,
        M=N,
        N=N,
        K=N,
        stride_az=stride_a_batch,
        stride_am=stride_a_m,
        stride_ak=stride_a_k,
        stride_bz=stride_b_batch,
        stride_bk=stride_b_k,
        stride_bn=stride_b_n,
        stride_cz=stride_c_batch,
        stride_cm=stride_c_m,
        stride_cn=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6) Reshape back to [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)      # [B, N, N, hidden_dim]
        .contiguous()
    )

    # --------------------------------------------------------------
    # 7) Final LayerNorm, gating and projection to original dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate               # apply outgoing gate
    out = F.linear(out, to_out_weight) # project back to dim

    return out
