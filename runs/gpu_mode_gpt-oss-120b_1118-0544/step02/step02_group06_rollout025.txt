import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, # left (batch, M, K) fp16
    b_ptr, # right (batch, N, K) fp16
    c_ptr, # output (batch, M, N) fp32
    batch: tl.constexpr,
    M: tl.constexpr,
    N: tl.constexpr,
    K: tl.constexpr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col
    pid_b = tl.program_id(2)  # batch

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # offsets for pointers
    a_ptrs = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_a_m + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_b * stride_b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # load tiles
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot product
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store result
    c_ptrs = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.
    Implements the computation described in the AlphaFold3 paper:

        out = LayerNorm(x)
        left = Linear(out) * mask * sigmoid(Linear(out))
        right = Linear(out) * mask * sigmoid(Linear(out))
        out_ij = Σ_k left_i k * right_j k   (batched GEMM per hidden dimension)
        out = LayerNorm(out_) * sigmoid(Linear(out_))
        out = Linear(out)

    The heavy matrix multiplication Σ_k left_i k * right_j k
    is performed by a custom Triton kernel for maximum throughput.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Unpack weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -------------------------------------------------
    # 1. Input LayerNorm over last dim (dim)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B, N, N, dim]

    # -------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -------------------------------------------------
    # 3. Apply mask if present
    # -------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------
    # 4. Gating (sigmoid)
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5. Batched GEMM (TriMul core) using Triton
    # -------------------------------------------------
    # Cast to half for compute efficiency
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, N, N, hidden_dim
    batch_dim = B * hidden_dim

    # reshape to (batch_dim, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(batch_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(batch_dim, N, N)

    # Allocate output buffer
    out_h = torch.empty((batch_dim, N, N), dtype=torch.float32, device=device)

    # Determine grid/block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # small enough to keep shared memory low

    # Compute number of program ids
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (num_pid_m, num_pid_n, batch_dim)

    # Compute strides
    stride_a_batch, stride_a_m, stride_a_k = left_h.stride()
    stride_b_batch, stride_b_n, stride_b_k = right_h.stride()
    # For right we need stride for k then n (because we want to traverse K dimension then column)
    # Our kernel reads right using (k, n) layout, so stride_b_k is right_h.stride(2), stride_b_n is right_h.stride(1)
    stride_b_batch = stride_b_batch
    stride_b_k = stride_b_k
    stride_b_n = stride_b_n

    stride_c_batch, stride_c_m, stride_c_n = out_h.stride()

    # Launch kernel
    tri_mul_outgoing_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        batch=batch_dim,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------
    # 6. Reshape back to [B, N, N, hidden_dim]
    # -------------------------------------------------
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # -------------------------------------------------
    # 7. Final LayerNorm, gating and projection
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,                     # left  tensor pointer (batch, M, K)  fp16
    b_ptr,                     # right tensor pointer (batch, N, K)  fp16 (will be read transposed)
    c_ptr,                     # output tensor pointer (batch, M, N) fp32
    batch: tl.constexpr,       # batch = B * hidden_dim
    M: tl.constexpr,           # rows of left (i dimension)
    N: tl.constexpr,           # rows of right (j dimension)
    K: tl.constexpr,           # reduction dimension (k)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out = left @ rightᵀ  where left ∈ ℝ^{batch×M×K},
    right ∈ ℝ^{batch×N×K}.  The reduction is over K.
    """
    pid_m = tl.program_id(0)          # tile row index
    pid_n = tl.program_id(1)          # tile col index
    pid_b = tl.program_id(2)          # batch (B * hidden_dim)

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + pid_b * stride_a_batch
    b_base = b_ptr + pid_b * stride_b_batch
    c_base = c_ptr + pid_b * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K in chunks of BLOCK_K
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load tiles
        a = tl.load(
            a_base + offs_m[:, None] * stride_a_m + cur_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_base + cur_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate dot product
        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    The computation follows the AlphaFold3 formulation:
      1. Layer‑norm the input over the last dimension.
      2. Linear projections → left/right (no bias).
      3. Optional mask applied to both projections.
      4. Gating with sigmoid(linear) and element‑wise multiplication.
      5. Core operation:   out_{i,j} = Σ_k left_{i,k} * right_{j,k}
         performed as a batched GEMM via a Triton kernel.
      6. Layer‑norm over the hidden dimension, apply outgoing gate,
         and a final linear projection back to the original dimension.

    The heavy GEMM (step 5) is executed on GPU with a custom Triton kernel
    for maximal throughput.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Unpack model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → left/right
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul (batched GEMM) using Triton
    # ------------------------------------------------------------------
    # Cast to half to leverage Tensor‑cores
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape               # B, N, N, hidden_dim
    batch_dim = B * hidden_dim               # number of independent GEMMs

    # reshape to (batch_dim, N, N) where each hidden slice is a separate GEMM
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(batch_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(batch_dim, N, N)

    # Output buffer (fp32 accumulator)
    out_h = torch.empty((batch_dim, N, N), dtype=torch.float32, device=device)

    # Tile sizes – chosen to fit typical GPU shared memory limits
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m, num_pid_n, batch_dim)

    # Strides for the three tensors (batch, row, col)
    stride_a_batch, stride_a_m, stride_a_k = left_h.stride()
    stride_b_batch, stride_b_n, stride_b_k = right_h.stride()
    # For the kernel we need stride_b_k (reduction dim) and stride_b_n (output dim)
    stride_c_batch, stride_c_m, stride_c_n = out_h.stride()

    # Launch Triton kernel
    tri_mul_outgoing_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        batch=batch_dim,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7. Final LayerNorm, gating, and projection to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # apply outgoing gate
    out = F.linear(out, to_out_weight)       # [B, N, N, dim]

    return out
