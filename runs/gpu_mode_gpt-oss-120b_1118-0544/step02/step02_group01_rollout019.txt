"""
TriMul (outgoing) forward pass.

The implementation follows the reference PyTorch module but fuses the
element‑wise mask‑and‑gate multiplication with a tiny Triton kernel.
The expensive N³ reduction is performed as a batched half‑precision
matrix multiplication using torch.bmm (cuBLAS).  The final output is
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Element‑wise kernel:
        out = proj * gate * mask
    proj, gate: [B, N, N, H] (float32)
    mask      : [B, N, N]      (float32, broadcast over H)
    out       : [B, N, N, H] (float32)
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # Decode linear index -> (b, i, j, h)
    tmp = offs
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # Compute flat offsets
    proj_off = (
        b * stride_proj_b
        + i * stride_proj_i
        + j * stride_proj_j
        + h * stride_proj_h
    )
    gate_off = (
        b * stride_gate_b
        + i * stride_gate_i
        + j * stride_gate_j
        + h * stride_gate_h
    )
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_off = (
        b * stride_out_b
        + i * stride_out_i
        + j * stride_out_j
        + h * stride_out_h
    )

    proj = tl.load(proj_ptr + proj_off, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=mask, other=0.0)

    tl.store(out_ptr + out_off, proj * gate * m, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor of shape [B, N, N] (or None)
        - weights_dict : mapping from weight names to tensors
        - config_dict  : must contain "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # Unpack arguments
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Gate computations (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 4) Fuse mask and gate (if mask present)
    # -----------------------------------------------------------------
    if (mask is not None) and (not nomask):
        # Ensure all tensors are contiguous for correct stride handling
        mask_f = mask.to(x.dtype).contiguous()
        left_proj_c = left_proj.contiguous()
        left_gate_c = left_gate.contiguous()
        right_proj_c = right_proj.contiguous()
        right_gate_c = right_gate.contiguous()

        # Allocate output buffers
        left = torch.empty_like(left_proj_c)
        right = torch.empty_like(right_proj_c)

        # Strides (in elements)
        s_lp_b, s_lp_i, s_lp_j, s_lp_h = left_proj_c.stride()
        s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate_c.stride()
        s_rp_b, s_rp_i, s_rp_j, s_rp_h = right_proj_c.stride()
        s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate_c.stride()
        s_m_b, s_m_i, s_m_j = mask_f.stride()
        s_out_b, s_out_i, s_out_j, s_out_h = left.stride()

        total = B * N * N * hidden_dim
        BLOCK_SIZE = 1024
        grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE,)

        # Left side fusion
        fused_mask_gate_kernel[grid](
            left_proj_c,
            left_gate_c,
            mask_f,
            left,
            B,
            N,
            hidden_dim,
            s_lp_b,
            s_lp_i,
            s_lp_j,
            s_lp_h,
            s_lg_b,
            s_lg_i,
            s_lg_j,
            s_lg_h,
            s_m_b,
            s_m_i,
            s_m_j,
            s_out_b,
            s_out_i,
            s_out_j,
            s_out_h,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # Right side fusion
        fused_mask_gate_kernel[grid](
            right_proj_c,
            right_gate_c,
            mask_f,
            right,
            B,
            N,
            hidden_dim,
            s_rp_b,
            s_rp_i,
            s_rp_j,
            s_rp_h,
            s_rg_b,
            s_rg_i,
            s_rg_j,
            s_rg_h,
            s_m_b,
            s_m_i,
            s_m_j,
            s_out_b,
            s_out_i,
            s_out_j,
            s_out_h,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        # Simple elementwise multiplication when no mask is used
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5) Core N³ reduction via batched half‑precision matrix multiplication
    # -----------------------------------------------------------------
    # Cast to fp16 for speed; the result will be cast back to fp32 later.
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange dimensions so that hidden_dim is the batch dimension for bmm
    # left: [B, hidden, N, N]  (i,k)
    # right: [B, hidden, N, N] (k,j) -> transpose last two dims
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    # Merge batch and hidden dimensions
    left_mat = left.view(B * hidden_dim, N, N)   # (B*H, N, K)
    right_mat = right.view(B * hidden_dim, N, N)  # (B*H, K, N)

    # Batched matrix multiplication (FP16)
    out_mat = torch.bmm(left_mat, right_mat)    # (B*H, N, N)

    # Restore original shape and dtype
    out_hidden = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden]

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, gating, and final linear projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
```
