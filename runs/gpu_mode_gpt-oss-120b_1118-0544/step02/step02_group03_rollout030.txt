"""
Optimized Triton implementation of the outgoing TriMul operation (AlphaFold3).

The kernel performs the expensive O(N³·H) tensor contraction:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
as a batched matrix multiplication using fp16 inputs and fp32 accumulation.
All other operations (layer‑norm, linear projections, mask, gating) are done
with PyTorch for simplicity, but the core contraction is fully Triton‑accelerated.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Batched matrix multiplication kernel (fp16 × fp16 → fp32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers to tensors
    M, N, K,                      # matrix dimensions (per batch)
    stride_az, stride_am, stride_ak,  # strides for A (batch, row, col)
    stride_bz, stride_bk, stride_bn,  # strides for B (batch, row, col)
    stride_cz, stride_cm, stride_cn,  # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr,
):
    pid_z = tl.program_id(2)               # batch * hidden dimension
    pid_m = tl.program_id(0)               # row tile
    pid_n = tl.program_id(1)               # col tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current block
        a_ptrs = (A_ptr
                  + pid_z * stride_az
                  + offs_m[:, None] * stride_am
                  + offs_k[None, :] * stride_ak)
        b_ptrs = (B_ptr
                  + pid_z * stride_bz
                  + offs_k[:, None] * stride_bk
                  + offs_n[None, :] * stride_bn)

        # load tiles (with zero padding)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # fused fp16·fp16 → fp32 matrix multiply
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # write back
    c_ptrs = (C_ptr
              + pid_z * stride_cz
              + offs_m[:, None] * stride_cm
              + offs_n[None, :] * stride_cn)
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)


def _batched_matmul_fp16(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A : (batch, M, K)   fp16
    B : (batch, K, N)   fp16
    Returns C : (batch, M, N) fp32
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # element strides
    stride_az, stride_am, stride_ak = A.stride()
    stride_bz, stride_bk, stride_bn = B.stride()
    stride_cz, stride_cm, stride_cn = C.stride()

    # tile sizes – tuned for RTX‑B200 Tensor Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_stages=2,
        num_warps=8,
    )
    return C


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]   (float32)
        - mask         : torch.Tensor [B, N, N]     (bool/float, optional)
        - weights      : dict of model tensors
        - config       : dict with keys "dim", "hidden_dim", "nomask" ...

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # still float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (left/right) and gates
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,H]
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # optional mask
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # gates (computed on fp16, sigmoid works in fp16)
    left_gate = torch.sigmoid(
        F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16))
    )

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core O(N³) update via batched matrix multiplication
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # reshape to (B, H, N, N) for easy batched GEMM
    left = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # transpose right on the last two dims
    right_T = right.permute(0, 1, 3, 2).contiguous()  # (B, H, N, N)

    # fuse batch and hidden dimension
    batch_hidden = B * hidden_dim
    left_flat = left.view(batch_hidden, N, N)       # (BH, M=N, K=N)
    right_flat = right_T.view(batch_hidden, N, N)   # (BH, K=N, N)

    # batched matmul: fp16×fp16 → fp32
    out_flat = _batched_matmul_fp16(left_flat, right_flat)  # (BH, N, N) fp32

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # fp32

    out = out * out_gate.to(out.dtype)   # apply out‑gate

    # final linear projection back to original dimension
    out_fp16 = out.to(torch.float16)
    output = F.linear(out_fp16, weights["to_out.weight"].to(torch.float16))
    output = output.to(torch.float32)

    return output
