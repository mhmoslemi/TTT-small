    b = tl.load(b_ptr + offs, mask=mask)

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Step 1: LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # Step 2: Linear projections
    left_raw = F.linear(x, weights["left_proj.weight"])
    right_raw = F.linear(x, weights["right_proj.weight"])

    # Step 3: Mask
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left_raw = left_raw * mask_f
        right_raw = right_raw * mask_f

    # Step 4: Gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left_raw * left_gate
    right = right_raw * right_gate

    # Convert for matmul (use fp16 to speed up)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape for batched matmul
    # left [B,N,N,h] -> (B, h, N, N)
    left_mat = left.permute(0,3,1,2).contiguous()
    # right transposed: (B, h, k, j) where k is third dim, j is second dim
    right_mat = right.permute(0,3,2,1).contiguous()

    # Batched matmul
    out_mat = torch.matmul(left_mat, right_mat)  # shape (B, h, N, N)

    # permute back to (B,N,N,h)
    out = out_mat.permute(0,2,3,1).contiguous().to(torch.float32)

    # Step 6: To-out layernorm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)

    # Step 7: Apply out_gate using Triton
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)

        gate_flat,
        out_flat,
        total_el,
        BLOCK=BLOCK,
    )

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton‑accelerated core.

The forward pass consists of:
1️⃣ Layer‑norm over the channel dimension.
2️⃣ Linear projections (left/right) and optional masking.
3️⃣ Gating (sigmoid of linear projections) applied to the projected tensors.
4️⃣ Core O(N³) reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is realized as a batched matrix multiplication
   left_mat @ right_matᵀ using torch.matmul on fp16 tensors.
5️⃣ Layer‑norm over the hidden dimension.
6️⃣ Element‑wise multiplication with the output‑gate (implemented as a tiny
   Triton kernel ‑ the “must‑use‑Triton” part).
7️⃣ Final linear projection back to the original channel dimension.

All tensors stay on the CUDA device; the heavy reduction is performed in
fp16 for speed while the surrounding normalisation/gating layers stay in
fp32 to preserve numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, n_elements, BLOCK: tl.constexpr):
    """c = a * b   (1‑D flattened tensors, dtype inferred from pointers)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) – may be ignored
            - weights: dict of model weight tensors
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm over the input channel dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # -----------------------------------------------------------------
    left_raw = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right_raw = F.linear(x, weights["right_proj.weight"]) # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3️⃣ Optional masking (broadcast on hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left_raw = left_raw * mask_f
        right_raw = right_raw * mask_f

    # -----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, hidden_dim]

    # Apply gates
    left = left_raw * left_gate
    right = right_raw * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Core O(N³) reduction via batched matrix multiplication (fp16)
    # -----------------------------------------------------------------
    # Cast to fp16 to accelerate the large GEMMs
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # (B, hidden, N, N) – rows i, columns k
    left_mat = left.permute(0, 3, 1, 2).contiguous()
    # (B, hidden, N, N) – rows k, columns j (swap the two N dimensions)
    right_mat = right.permute(0, 3, 2, 1).contiguous()

    # Batched matmul: out_mat[b, h] = left_mat[b, h] @ right_mat[b, h]
    out_mat = torch.matmul(left_mat, right_mat)  # [B, hidden_dim, N, N] (fp16)

    # Bring back to original layout and convert to fp32 for the following layers
    out = out_mat.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 6️⃣ LayerNorm over the hidden dimension
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Apply the output gate (Triton kernel)
    # -----------------------------------------------------------------
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)
    total_el = out_flat.numel()
    BLOCK = 1024
    grid = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        out_flat,
        total_el,
        BLOCK=BLOCK,
    )
    # out is now gated in‑place

    # -----------------------------------------------------------------
    # 8️⃣ Final linear projection back to the original dimension
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return out
