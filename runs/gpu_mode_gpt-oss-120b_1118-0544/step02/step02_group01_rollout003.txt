@triton.jit
def fused_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_SIZE: tl.constexpr,
):
    # linear index offset
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # compute indices
    tmp = offs
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # compute addresses
    proj_offset = b * stride_proj_b + i * stride_proj_i + j * stride_proj_j + h * stride_proj_h
    gate_offset = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_offset = b * stride_out_b + i * stride_out_i + j * stride_out_j + h * stride_out_h

    # load values
    proj = tl.load(proj_ptr + proj_offset, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + gate_offset, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_offset, mask=mask, other=0.0)

    # multiply and cast to half
    val = (proj * gate * m).to(tl.float16)

    # store
    tl.store(out_ptr + out_offset, val, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    B,N,_, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # layer norm input
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # linear projections
    left_proj = F.linear(x, weights["left_proj.weight"]).to(torch.float32)
    right_proj = F.linear(x, weights["right_proj.weight"]).to(torch.float32)

    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # shape [B,N,N,hidden_dim]

    # mask handling
    if mask is None:
        mask_f = torch.ones((B,N,N), dtype=x.dtype, device=x.device)
    else:
        mask_f = mask.to(x.dtype)  # float32

    # fuse mask and gate with projection to half using Triton
    # allocate output halves
    left = torch.empty_like(left_proj, dtype=torch.float16)
    right = torch.empty_like(right_proj, dtype=torch.float16)

    # compute strides
    s_lp = left_proj.stride()
    s_lg = left_gate.stride()
    s_rp = right_proj.stride()
    s_rg = right_gate.stride()
    s_m = mask_f.stride()

    # output strides (left, right)
    s_out = left.stride()

    total = B * N * N * hidden_dim
    BLOCK_SIZE = 1024  # choose compile-time constant, must be divisible by 2? we use 1024.

    grid = ( (total + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    # call kernel for left
    fused_mask_gate_kernel[grid](
        left_proj,
        left_gate,
        mask_f,
        left,
        B, N, hidden_dim,
        s_lp[0], s_lp[1], s_lp[2], s_lp[3],
        s_lg[0], s_lg[1], s_lg[2], s_lg[3],
        s_m[0], s_m[1], s_m[2],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # call for right
    fused_mask_gate_kernel[grid](
        right_proj,
        right_gate,
        mask_f,
        right,
        B, N, hidden_dim,
        s_rp[0], s_rp[1], s_rp[2], s_rp[3],
        s_rg[0], s_rg[1], s_rg[2], s_rg[3],
        s_m[0], s_m[1], s_m[2],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # At this point left and right are half.

    # Perform batched matmul
    # left shape: [B,N,N,H] -> [B*H,N,N]
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)

"""
Optimized forward implementation of the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Input LayerNorm.
2. Linear projections (left_proj, right_proj) and gate projections (left_gate,
   right_gate, out_gate).
3. Fuse projection, gate, and optional mask using a small Triton kernel.
   The kernel multiplies the projected values with the sigmoid gates and the
   mask, then writes the result directly in FP16 to avoid an extra cast.
4. Perform the N³ reduction as a batched half‑precision matrix multiplication:
   for each hidden channel h we compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using `torch.bmm` on tensors of shape (B*H, N, N).
5. Output LayerNorm, output gating, and final linear projection back to `dim`.

The Triton kernel is used for the element‑wise fusion (step 3) to reduce
memory traffic and to satisfy the requirement of having at least one custom
kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Elementwise kernel:
        out = (proj * gate * mask).cast(fp16)

    Shapes:
        proj, gate : [B, N, N, H]   (float32)
        mask       : [B, N, N]      (float32, broadcasted over H)
        out        : [B, N, N, H]   (float16)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # Decode linear offset to (b, i, j, h)
    tmp = offs
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # Compute flat offsets for each tensor
    proj_off = (
        b * stride_proj_b
        + i * stride_proj_i
        + j * stride_proj_j
        + h * stride_proj_h
    )
    gate_off = (
        b * stride_gate_b
        + i * stride_gate_i
        + j * stride_gate_j
        + h * stride_gate_h
    )
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_off = (
        b * stride_out_b
        + i * stride_out_i
        + j * stride_out_j
        + h * stride_out_h
    )

    # Load values (float32)
    proj = tl.load(proj_ptr + proj_off, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=mask, other=0.0)

    # Multiply in FP32, then cast to FP16
    val_fp16 = (proj * gate * m).to(tl.float16)

    # Store result
    tl.store(out_ptr + out_off, val_fp16, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of weight tensors
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections and gate computation (no bias)
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden]
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 3) Fuse mask, projection and gate (Triton kernel)
    # -----------------------------------------------------------------
    if mask is None:
        mask_tensor = torch.ones((B, N, N), dtype=x.dtype, device=x.device)
    else:
        mask_tensor = mask.to(dtype=x.dtype, device=x.device)

    # Ensure contiguity for stride calculation
    left_proj_c = left_proj.contiguous()
    left_gate_c = left_gate.contiguous()
    right_proj_c = right_proj.contiguous()
    right_gate_c = right_gate.contiguous()
    mask_c = mask_tensor.contiguous()

    # Allocate output buffers in FP16
    left = torch.empty_like(left_proj_c, dtype=torch.float16)
    right = torch.empty_like(right_proj_c, dtype=torch.float16)

    # Strides (in elements, not bytes)
    s_lp = left_proj_c.stride()
    s_lg = left_gate_c.stride()
    s_rp = right_proj_c.stride()
    s_rg = right_gate_c.stride()
    s_m = mask_c.stride()
    s_out = left.stride()  # same for left and right

    total_elements = B * N * N * hidden_dim
    BLOCK_SIZE = 1024
    grid = ((total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    # Left side
    fused_mask_gate_kernel[grid](
        left_proj_c,
        left_gate_c,
        mask_c,
        left,
        B,
        N,
        hidden_dim,
        s_lp[0],
        s_lp[1],
        s_lp[2],
        s_lp[3],
        s_lg[0],
        s_lg[1],
        s_lg[2],
        s_lg[3],
        s_m[0],
        s_m[1],
        s_m[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # Right side
    fused_mask_gate_kernel[grid](
        right_proj_c,
        right_gate_c,
        mask_c,
        right,
        B,
        N,
        hidden_dim,
        s_rp[0],
        s_rp[1],
        s_rp[2],
        s_rp[3],
        s_rg[0],
        s_rg[1],
        s_rg[2],
        s_rg[3],
        s_m[0],
        s_m[1],
        s_m[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # -----------------------------------------------------------------
    # 4) Batched matrix multiplication (the N³ reduction)
    # -----------------------------------------------------------------
    # left/right: [B,N,N,hidden] (FP16)
    # Reshape to (B*hidden, N, N) where each batch performs a matmul.
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Batched half‑precision GEMM
    out_mat = torch.bmm(left_mat, right_mat)   # (B*hidden, N, N), dtype=fp16

    # Reshape back to [B, N, N, hidden] and convert to FP32
    out_hidden = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )

    # -----------------------------------------------------------------
    # 5) Output LayerNorm, output gating and final projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate  # gating (float32)

    # Final linear projection back to original dimension
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
