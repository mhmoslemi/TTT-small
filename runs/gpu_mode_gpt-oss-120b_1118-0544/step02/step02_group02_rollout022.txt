def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1) layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # 2) linear proj and gates
    # no bias
    left_proj = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_weight)

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # 4) apply gate
    left = left_proj * left_gate     # float32

    left = left.to(torch.float16)

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0) 
    pid_m = tl.program_id(1) 
    pid_n = tl.program_id(2) 

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile
        a_ptrs = a_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)   # fp16

        # Load B tile (B is transposed to shape (batch, K, N))
        b_ptrs = b_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)   # fp16

        acc += tl.dot(a, b)   # accumulate in fp32

    # Store result
    c_ptrs = c_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton‑accelerated core.

The module performs:
  1. LayerNorm on the input tensor.
  2. Two bias‑free linear projections (left/right) and three gating projections.
  3. Optional masking of the projected tensors.
  4. Element‑wise gating, cast to FP16.
  5. Batched matrix multiplication:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
     implemented as a fused Triton GEMM with FP32 accumulation.
  6. LayerNorm, output gating and final linear projection back to `dim`.

Only the heavy reduction (step 5) is executed in Triton; all other operations
use PyTorch which is already highly optimized for the small‑dim linear layers.
The Triton kernel works on a flattened batch dimension (B × hidden_dim) and
supports arbitrary sequence lengths (N) up to 1024.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                     # matrix dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B   where A : (batch, M, K)  (FP16)
                        B : (batch, K, N)  (FP16, transposed view of right)
                        C : (batch, M, N)  (FP32)

    The kernel processes one tile (BLOCK_M × BLOCK_N) of each batch matrix.
    """
    pid_batch = tl.program_id(0)          # batch index (B * hidden_dim)
    pid_m = tl.program_id(1)              # block row
    pid_n = tl.program_id(2)              # block col

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M × K)
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)

        # Load B tile (K × N) – note B is already transposed (K,N)
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)

        # Accumulate (FP16×FP16 → FP32)
        acc += tl.dot(a, b)

    # Store the result (FP32)
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs,
             acc,
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim] (float32, on CUDA)
            mask:         torch.Tensor [B, N, N] (bool/float) – may be None
            weights:      dict of model parameters (see skeleton)
            config:       dict with at least:
                          - "dim" (int)
                          - "hidden_dim" (int)
                          - "nomask" (bool, optional)

    Returns:
        torch.Tensor [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right_proj = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 5) Apply gates and cast to FP16 for the heavy reduction
    # ------------------------------------------------------------------
    left = (left_proj * left_gate).to(torch.float16)   # [B, N, N, hidden_dim] FP16
    right = (right_proj * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 6) Reshape for batched GEMM: (B * hidden_dim, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()    # [B, hidden_dim, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    total_batch = B * hidden_dim

    left_flat = left.view(total_batch, N, N)        # (batch, M, K)
    # Right must be transposed on the last two dimensions to obtain (batch, K, N)
    right_t = right.transpose(1, 2)                 # (batch, K, N) – non‑contiguous view

    # ------------------------------------------------------------------
    # 7) Allocate output (FP32) and launch Triton GEMM kernel
    # ------------------------------------------------------------------
    out_fp32_flat = torch.empty((total_batch, N, N), dtype=torch.float32, device=device)

    # Strides for the three tensors (in elements)
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_t.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_fp32_flat.stride()

    # Tiling parameters (tuned for B200 / Ampere)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_t,
        c_ptr=out_fp32_flat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    # Ensure kernel completion before next operations
    torch.cuda.synchronize(device)

    # ------------------------------------------------------------------
    # 8) Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_fp32_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                      # apply output gate
    out = F.linear(out, to_out_weight)        # [B, N, N, dim]

    return out
