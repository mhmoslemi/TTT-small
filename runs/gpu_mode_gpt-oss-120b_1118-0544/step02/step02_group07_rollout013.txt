import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offset < N
    a = tl.load(a_ptr + offset, mask=mask)
    b = tl.load(b_ptr + offset, mask=mask)
    tl.store(c_ptr + offset, a * b, mask=mask)

def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.
    The implementation fuses the four linear projections (proj & gate for left and right)
    into a single matrix multiplication, performs the O(N³) reduction via batched
    half‑precision GEMM, and uses a minimal Triton kernel for the final gating.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Extract weights (all are float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    # torch.nn.functional.layer_norm works on the last dimension only
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    B, N, _, _ = x.shape

    # Cast to FP16 for the heavy math
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2) Fuse four linear layers (proj & gate for left/right)
    # -----------------------------------------------------------------
    # Build a combined weight matrix of shape (4*hidden, dim)
    # Weight tensors are stored as (out_features, in_features)
    # Ensure they are FP16
    w_left_proj = left_proj_weight.to(torch.float16)
    w_left_gate = left_gate_weight.to(torch.float16)
    w_right_proj = right_proj_weight.to(torch.float16)
    w_right_gate = right_gate_weight.to(torch.float16)

    combined_weight = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate], dim=0
    )  # (4*hidden, dim)

    # Flatten x to (B*N*N, dim) and apply the combined linear layer
    x_flat = x_fp16.view(-1, dim)  # (B*N*N, dim)
    proj = F.linear(x_flat, combined_weight)  # (B*N*N, 4*hidden)

    # Reshape back to (B, N, N, 4*hidden)
    proj = proj.view(B, N, N, 4 * hidden_dim)

    # Split the tensor
    left_proj, left_gate, right_proj, right_gate = torch.split(
        proj, hidden_dim, dim=-1
    )  # each: (B, N, N, hidden)

    # -----------------------------------------------------------------
    # 3) Apply mask to the *projections* (not to the gates)
    # -----------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # (B, N, N, 1)
        # mask_f is float32; cast to half to match left_proj/right_proj dtype
        mask_f = mask_f.to(torch.float16)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid) and combine with projected values
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate  # (B, N, N, hidden)  FP16
    right = right_proj * right_gate  # (B, N, N, hidden)  FP16

    # -----------------------------------------------------------------
    # 5) Batched GEMM: out = Σ_k left[...,k,d] * right[...,k,d]
    #    Operates on tensors of shape (B, hidden, N, N)
    # -----------------------------------------------------------------
    # Bring hidden dimension to axis 1
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    # Merge batch and hidden for torch.bmm
    left_mat = left.view(B * hidden_dim, N, N)   # (B*hidden, N, N)

    # Perform half‑precision batched matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    # Restore the original layout
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, hidden)  FP16

    # -----------------------------------------------------------------
    # 6) to_out_norm (LayerNorm over hidden)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden)  FP32 by default; returns same dtype as input (FP16 -> FP16)

    # -----------------------------------------------------------------
    # 7) Compute out_gate and apply it (Triton kernel)
    # -----------------------------------------------------------------
    # out_gate = sigmoid(Linear(x, out_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight.to(torch.float16)))
    # Ensure same shape (B, N, N, hidden)
    # Cast out to FP16 for element‑wise multiplication
    out_fp16 = out.to(torch.float16)

    # Triton element‑wise multiply (out_fp16 *= out_gate)
    total_el = out_fp16.numel()
    BLOCK = 1024
    grid = ((total_el + BLOCK - 1) // BLOCK,)

    elementwise_mul_kernel[grid](
        out_fp16,
        out_gate,
        out_fp16,
        total_el,
        BLOCK=BLOCK,
    )
    # Cast back to FP32 for final projection
    out_fp32 = out_fp16.to(torch.float32)

    # -----------------------------------------------------------------
    # 8) Final linear projection back to original dim
    # -----------------------------------------------------------------
    out_final = F.linear(out_fp32, to_out_weight)  # (B, N, N, dim)

    return out_final

"""
Outgoing Triangle Multiplicative Update (TriMul) – fast forward implementation.

The kernel follows the reference PyTorch module but aggressively fuses operations:
  * Four linear projections (left/right + their gates) are collapsed into a single
    matrix multiplication.
  * The O(N³) reduction is performed as a batched half‑precision GEMM:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    which is equivalent to batched matmul of shape (B·hidden, N, N).
  * A tiny Triton kernel applies the final out‑gate element‑wise (required
    usage of Triton).
All heavy tensor work is carried out in float16 to exploit tensor‑cores on an
Ada‑Lovelace B‑200 GPU, while the final output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """c = a * b  (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – may be ignored if config["nomask"] is True
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # --------------------------------------------------------------
    # Cast and cache all weights in the appropriate dtype
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the input channel dimension (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    B, N, _, _ = x.shape
    # --------------------------------------------------------------
    # 2️⃣ Fuse the four linear layers (proj & gate for left/right)
    # --------------------------------------------------------------
    # Build combined weight of shape (4*hidden, dim)
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    )  # (4*hidden, dim)

    # Flatten spatial dimensions and apply the combined linear mapping
    x_fp16 = x.to(torch.float16)
    x_flat = x_fp16.view(-1, dim)                     # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 4*hidden)

    # Restore 4‑D layout and split
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each: (B, N, N, hidden)

    # --------------------------------------------------------------
    # 3️⃣ Optional masking (applied only to the projections)
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # (B, N, N, 1)
        mask_f = mask_f.to(torch.float16)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and combine with projections
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)    # (B, N, N, hidden)  fp16
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate            # (B, N, N, hidden)  fp16
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Batched GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a half‑precision batched matmul.
    # --------------------------------------------------------------
    # Move hidden dimension to axis 1 and make tensors contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    # Merge batch and hidden into a single batch dimension
    left_mat = left.view(B * hidden_dim, N, N)   # (B*hidden, N, N)
    right_mat = right.view(B * hidden_dim, N, N)

    # Batched matmul: (B*hidden, N, N) @ (B*hidden, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    # Reshape back to (B, N, N, hidden)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, hidden)  fp16

    # --------------------------------------------------------------
    # 6️⃣ to_out_norm (LayerNorm over hidden dimension)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden)  fp16 (LayerNorm supports fp16)

    # --------------------------------------------------------------
    # 7️⃣ Compute out_gate and apply it (Triton kernel)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # (B, N, N, hidden) fp16

    # Triton element‑wise multiplication: out *= out_gate  (in‑place)
    total_el = out.numel()
    BLOCK = 1024
    grid = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        out,
        out_gate,
        out,
        total_el,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 8️⃣ Final projection back to original dimension
    # --------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # (B, N, N, dim)  fp16
    out = out.to(torch.float32)         # return float32 as required
    return out
