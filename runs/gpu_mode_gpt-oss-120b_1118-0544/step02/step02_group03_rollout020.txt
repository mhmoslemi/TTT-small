We can write the kernel signature:

@triton.jit
def tr_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    stride_b_right, stride_i_right, stride_k_right, stride_h_right,
    stride_b_out, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0) # i tile
    pid_n = tl.program_id(1) # j tile
    pid_bh = tl.program_id(2) # combined batch*hidden

    # compute batch and hidden indices:
    b = pid_bh // H
    h = pid_bh % H

    # compute offsets for i and j
    offs_i = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute base pointers:
    # base offsets for batch and hidden
    base_left = left_ptr + b * stride_b_left + h * stride_h_left
    base_right = right_ptr + b * stride_b_right + h * stride_h_right
    base_out = out_ptr + b * stride_b_out + h * stride_h_out

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K

"""
TriMul outgoing update (AlphaFold3) – Triton‑accelerated implementation.

The kernel computes, for each batch b and hidden channel h:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
where left and right are the gated projections of the input tensor.
All heavy linear work is performed in FP16 and the O(N³) reduction is
implemented as a batched matrix multiplication using a custom Triton
kernel that loads 64‑x‑64 tiles and accumulates in FP32 (Tensor‑Core
friendly).  The rest of the pipeline (LayerNorms, gates, final linear)
uses PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tr_mul_outgoing_kernel(
    left_ptr,               # fp16 *
    right_ptr,              # fp16 *
    out_ptr,                # fp32 *
    B, N, H,                # int: batch, seq_len, hidden
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    stride_b_right, stride_i_right, stride_k_right, stride_h_right,
    stride_b_out, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_M: tl.constexpr,  # tile size in i dimension
    BLOCK_N: tl.constexpr,  # tile size in j dimension
    BLOCK_K: tl.constexpr,  # tile size in k dimension
):
    pid_i = tl.program_id(0)               # i tile
    pid_j = tl.program_id(1)               # j tile
    pid_bh = tl.program_id(2)              # flattened (batch, hidden)

    # Recover batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # Compute offsets for the i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers that incorporate batch and hidden offsets
    left_base  = left_ptr  + b * stride_b_left  + h * stride_h_left
    right_base = right_ptr + b * stride_b_right + h * stride_h_right
    out_base   = out_ptr   + b * stride_b_out   + h * stride_h_out

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (i, k)
        a_ptrs = left_base + offs_i[:, None] * stride_i_left + offs_k[None, :] * stride_k_left
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile: (k, j)  – note the swap of i/k ordering
        b_ptrs = right_base + offs_k[:, None] * stride_k_right + offs_j[None, :] * stride_i_right
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # FP16·FP16 → FP32 accumulation
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Store the result to out[b, i, j, h]
    c_ptrs = out_base + offs_i[:, None] * stride_i_out + offs_j[None, :] * stride_j_out
    mask_store = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)


def launch_tr_mul_outgoing(left: torch.Tensor, right: torch.Tensor, out: torch.Tensor,
                          BLOCK_M=64, BLOCK_N=64, BLOCK_K=64):
    """Launch the Triton kernel for the core O(N³) update."""
    B, N, _N, H = left.shape
    assert N == _N, "Input dimensions must be square."
    # Strides in element units
    stride_b_left, stride_i_left, stride_k_left, stride_h_left = left.stride()
    stride_b_right, stride_i_right, stride_k_right, stride_h_right = right.stride()
    stride_b_out, stride_i_out, stride_j_out, stride_h_out = out.stride()

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    tr_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_b_left,
        stride_i_left,
        stride_k_left,
        stride_h_left,
        stride_b_right,
        stride_i_right,
        stride_k_right,
        stride_h_right,
        stride_b_out,
        stride_i_out,
        stride_j_out,
        stride_h_out,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # still float32

    # Cast to FP16 for the heavy linear work
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)    # [B, N, N, H]  fp16
    right = F.linear(x, right_proj_w)  # fp16

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gates (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))    # [B, N, N, H] fp16
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))   # fp16
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))      # fp16, used later

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core O(N³) update via Triton batched matmul
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)
    launch_tr_mul_outgoing(left, right, out)   # out in FP32

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # fp32

    # Apply out‑gate (cast to fp32)
    out = out * out_gate.to(out.dtype)

    # Final linear projection back to `dim`
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out_fp16 = out.to(torch.float16)
    output = F.linear(out_fp16, to_out_w)   # [B, N, N, dim] fp16
    output = output.to(torch.float32)

    return output
