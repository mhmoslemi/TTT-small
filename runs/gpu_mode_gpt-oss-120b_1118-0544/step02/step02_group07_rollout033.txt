def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch_head = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch and head index
    b = pid_batch_head // D
    d = pid_batch_head % D

    # compute offsets base
    offset_left_base = b * stride_left_b + d * stride_left_d
    offset_right_base = b * stride_right_b + d * stride_right_d
    offset_out_base = b * stride_out_b + d * stride_out_d

    # i, j offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out of bounds
    i_valid = i < N
    j_valid = j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k indices
        k_base = kk * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        k_valid = k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + offset_left_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device

    # Load weights:
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]; right_gate_weight = weights["right_gate.weight"]; out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Step 1: LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # Step 2: Linear projections
    left_proj = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden]
    right_proj = torch.nn.functional.linear(x, right_proj_weight)

    # Step 3: Mask if needed
    if not nomask:
        mask_unsqueezed = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left_proj = left_proj * mask_unsqueezed
        right_proj = right_proj * mask_unsqueezed

    # Step 4: Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Convert to fp16 for GEMM
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape

    # Output tensor allocate
    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Strides
    stride_left_b = left_fp16.stride(0)
    stride_left_i = left_fp16.stride(1)
    stride_left_k = left_fp16.stride(2)
    stride_left_d = left_fp16.stride(3)

    stride_right_b = right_fp16.stride(0)
    stride_right_i = right_fp16.stride(1)
    stride_right_k = right_fp16.stride(2)
    stride_right_d = right_fp16.stride(3)

    stride_out_b = out_fp16.stride(0)
    stride_out_i = out_fp16.stride(1)
    stride_out_j = out_fp16.stride(2)
    stride_out_d = out_fp16.stride(3)

    # Set block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid dimensions
    grid = (
        (B * hidden_dim),                # batch*head dimension
        ( (N + BLOCK_M - 1) // BLOCK_M ),  # i blocks
        ( (N + BLOCK_N - 1) // BLOCK_N ),  # j blocks
    )

        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_i, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,

    out = out_fp16.to(torch.float32)

    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # compute IDs
    pid_batch_head = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch and head indices
    b = pid_batch_head // D
    d = pid_batch_head % D

    # compute base offsets
    offset_left = b * stride_left_b + d * stride_left_d
    offset_right = b * stride_right_b + d * stride_right_d
    offset_out = b * stride_out_b + d * stride_out_d

    # compute i and j coordinates
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for bounds
    i_valid = i < N
    j_valid = j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_valid = k < N

        # Load left tile (M,K)
        left_ptrs = left_ptr + offset_left + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0)

        # Load right tile (K,N): note we need rows = k, cols = j
        right_ptrs = right_ptr + offset_right + k[:, None] * stride_right_k + j[None, :] * stride_right_i
        right = tl.load(right_ptrs, mask=k_valid[:, None] & j_valid[None, :], other=0.0)

        # dot (M,K) x (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + offset_out + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_valid[:, None] & j_valid[None, :])

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton implementation.

The heavy O(N³·hidden) contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is performed by a custom Triton kernel that directly accesses the
[batch, seq, seq, hidden] tensors with strided loads, avoiding the
expensive permute‑and‑contiguous copies required by a naïve batched
gemm.  All other operations (LayerNorms, Linear projections, gating,
masking) are executed with PyTorch in fp32, while the contraction uses
mixed‑precision (fp16 inputs, fp32 accumulation) to exploit tensor‑core
throughput.  This yields a compact, fast forward pass that satisfies the
required interface.

Author: OpenAI ChatGPT – Triton‑optimized TriMul (outgoing) kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: core O(N³) reduction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_core_kernel(
    left_ptr, right_ptr, out_ptr,               # tensor pointers
    B, N, D,                                    # batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = Σₖ left[...,k,:] * right[...,k,:] for a tile."""
    pid_batch_head = tl.program_id(0)   # combined batch * hidden index
    pid_i = tl.program_id(1)            # i‑block index
    pid_j = tl.program_id(2)            # j‑block index

    # ------------------------------------------------------------------
    # Decode batch and hidden‑channel index
    # ------------------------------------------------------------------
    b = pid_batch_head // D
    d = pid_batch_head % D

    # Base offsets for each tensor (no i/j/k yet)
    offset_left  = b * stride_left_b  + d * stride_left_d
    offset_right = b * stride_right_b + d * stride_right_d
    offset_out   = b * stride_out_b   + d * stride_out_d

    # ------------------------------------------------------------------
    # Tile coordinates
    # ------------------------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)   # rows of left
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)   # cols of right

    i_valid = i < N
    j_valid = j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_valid = k < N

        # left tile: shape (BLOCK_M, BLOCK_K)  -> (i, k)
        left_ptrs = left_ptr + offset_left \
                    + i[:, None] * stride_left_i \
                    + k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=i_valid[:, None] & k_valid[None, :],
                       other=0.0)

        # right tile: shape (BLOCK_K, BLOCK_N) -> (k, j)
        right_ptrs = right_ptr + offset_right \
                     + k[:, None] * stride_right_k \
                     + j[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=k_valid[:, None] & j_valid[None, :],
                        other=0.0)

        # Matrix‑multiply‑accumulate
        acc += tl.dot(left, right)   # result is fp32

    # ------------------------------------------------------------------
    # Store the result (convert to fp16 to keep memory low)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + offset_out \
               + i[:, None] * stride_out_i \
               + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=i_valid[:, None] & j_valid[None, :])

# ----------------------------------------------------------------------
# Entry point matching the required signature
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a fused Triton kernel.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim], dtype float32
            - mask: torch.Tensor of shape [B, N, N] (bool/float), may be ignored
            - weights: dict of model weight tensors (see code for keys)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim], dtype float32
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (over last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right_proj = F.linear(x, right_proj_weight) # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (broadcasted over hidden)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Convert to fp16 for the heavy contraction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape

    # Allocate output tensor (fp16) for the contraction result
    out_fp16 = torch.empty((B, N, N, hidden_dim),
                           dtype=torch.float16,
                           device=device)

    # ------------------------------------------------------------------
    # 6️⃣ Strides (in elements, not bytes) for each tensor
    # ------------------------------------------------------------------
    stride_left_b  = left_fp16.stride(0)
    stride_left_i  = left_fp16.stride(1)
    stride_left_k  = left_fp16.stride(2)
    stride_left_d  = left_fp16.stride(3)

    stride_right_b = right_fp16.stride(0)
    stride_right_i = right_fp16.stride(1)   # same as stride_left_i
    stride_right_k = right_fp16.stride(2)
    stride_right_d = right_fp16.stride(3)

    stride_out_b   = out_fp16.stride(0)
    stride_out_i   = out_fp16.stride(1)
    stride_out_j   = out_fp16.stride(2)
    stride_out_d   = out_fp16.stride(3)

    # ------------------------------------------------------------------
    # 7️⃣ Launch Triton kernel for the O(N³) reduction
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # batch * hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # i‑blocks
        (N + BLOCK_N - 1) // BLOCK_N,       # j‑blocks
    )

    tri_mul_core_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_d,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_d,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Convert back to fp32 and continue with the remaining steps
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)          # [B, N, N, hidden]

    # LayerNorm over hidden dimension
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to original dimension
    out = F.linear(out, to_out_weight)        # [B, N, N, dim]

    return out
