def custom_kernel(data):
    x, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)
    # layernorm
    x = F.layer_norm(x, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=1e-5)

    # linear projections (no bias)
    left = F.linear(x, weights['left_proj.weight'])   # shape [B,N,N,hidden_dim]
    right = F.linear(x, weights['right_proj.weight'])

    if not nomask:
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_f
        right = right * mask_f

    # gating
    left_gate = torch.sigmoid(F.linear(x, weights['left_gate.weight']))
    right_gate = torch.sigmoid(F.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(F.linear(x, weights['out_gate.weight']))

    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    B, N, _, H = left.shape

    # allocate output fp16
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=left.device)

    # Compute strides for Triton
    stride_b_l = left.stride(0)   # stride across batch
    stride_i_l = left.stride(1)   # stride across i
    stride_k_l = left.stride(2)   # stride across k
    stride_h_l = left.stride(3)   # stride across h

    # Right has same strides
    stride_b_r = right.stride(0)
    stride_i_r = right.stride(1)
    stride_k_r = right.stride(2)
    stride_h_r = right.stride(3)

    # Output strides
    stride_b_o = out.stride(0)
    stride_i_o = out.stride(1)

```
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bn = tl.program_id(2)  # batch * hidden

    b = pid_bn // H
    h = pid_bn % H

    # compute start offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # create range vectors
    offs_i = i_base + tl.arange(0, BLOCK_M)  # shape (BLOCK_M,)
    offs_j = j_base + tl.arange(0, BLOCK_N)  # shape (BLOCK_N,)
    offs_k = tl.arange(0, BLOCK_K)           # shape (BLOCK_K,)

    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointers
    left_base = left_ptr + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base = out_ptr + b * stride_bo + h * stride_ho

    # pointer offsets for A (left) and B (right). A is (i,k), B is (j,k) as we want B_T.
    # Compute A pointers: (i,k)
    a_ptrs = left_base + offs_i[:, None] * stride_il + offs_k[None, :] * stride_kl

    # Compute B pointers: need (k,j)
    b_ptrs = right_base + offs_k[:, None] * stride_kr + offs_j[None, :] * stride_ir

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for k in range(0, N, BLOCK_K):
        # compute actual offsets for this k tile
        cur_k = k + offs_k
        cur_mask_k = cur_k < N

        # load A tile
        a = tl.load(a_ptrs, mask=mask_i[:, None] & cur_mask_k[None, :], other=0.0).to(tl.float32)

        # load B tile
        b = tl.load(b_ptrs, mask=cur_mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(a, b)

        # increment pointers for next tile
        a_ptrs += BLOCK_K * stride_kl
        b_ptrs += BLOCK_K * stride_kr

    # store result
    # compute output offsets
    out_ptrs = out_base + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2) # batch*hidden

    # compute batch and hidden index
    b = pid_hb // H
    h = pid_hb % H

    # start offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # create offset vectors
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this batch/hidden
    left_base = left_ptr + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base = out_ptr + b * stride_bo + h * stride_ho

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        k_offset = k + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # compute pointers for current tile
        a_ptrs = left_base + offs_i[:, None] * stride_il + k_offset[None, :] * stride_kl
        b_ptrs = right_base + k_offset[:, None] * stride_kr + offs_j[None, :] * stride_ir

        # load tiles
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        a_f32 = a.to(tl.float32)
        b_f32 = b.to(tl.float32)

        # dot
        acc += tl.dot(a_f32, b_f32)

    out_ptrs = out_base + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo

"""
Outgoing Triangle Multiplicative Update (TriMul) – forward pass.

The costly part is the reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is a batched matrix multiplication of size N×N for every hidden channel.
The implementation below:
* Performs LayerNorms, linear projections, and gating with PyTorch (fast BLAS).
* Casts the gated tensors to fp16.
* Executes the N³ reduction with a custom Triton kernel that works on the
  original layout [B, N, N, H] (no costly permute‑and‑copy).
* Accumulates in fp32 and writes the result in fp16, later cast to fp32 for
  the final LayerNorm and output projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, H,                                     # sizes
    stride_bl, stride_il, stride_kl, stride_hl, # left strides
    stride_br, stride_ir, stride_kr, stride_hr, # right strides
    stride_bo, stride_io, stride_jo, stride_ho, # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul for the TriMul operation.

    Each program computes a BLOCK_M × BLOCK_N tile of the output matrix for a
    particular (batch, hidden) pair.  The innermost reduction over K is
    carried out in a loop, accumulating in fp32.
    """
    pid_i = tl.program_id(0)          # tile index over i dimension
    pid_j = tl.program_id(1)          # tile index over j dimension
    pid_hb = tl.program_id(2)         # combined batch*hidden id

    # recover batch and hidden ids
    b = pid_hb // H
    h = pid_hb % H

    # ------------------ offsets for the i / j tiles ------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base   = out_ptr   + b * stride_bo + h * stride_ho

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------ loop over the reduction dimension k ------------------
    for k in range(0, N, BLOCK_K):
        k_off = k + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left tile:   shape (BLOCK_M, BLOCK_K)  -> (i, k)
        a_ptrs = left_base + offs_i[:, None] * stride_il + k_off[None, :] * stride_kl
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right tile:  shape (BLOCK_K, BLOCK_N)  -> (k, j)
        b_ptrs = right_base + k_off[:, None] * stride_kr + offs_j[None, :] * stride_ir
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # matrix‑multiply the tiles
        acc += tl.dot(a, b)

    # ------------------ write the tile back ------------------
    out_ptrs = out_base + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim]  (float32)
            mask:         torch.Tensor [B, N, N]      (bool/float) – may be None
            weights:      dict of model parameters
            config:       dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 1) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 2) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4) Apply gates and cast to fp16 for the heavy reduction
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    B, N, _, H = left.shape  # B, N, N, hidden_dim

    # ------------------------------------------------------------------
    # 5) Triton kernel: out = Σ_k left[...,k,...] * right[...,k,...]
    # ------------------------------------------------------------------
    out_fp16 = torch.empty_like(left, dtype=torch.float16)  # [B,N,N,H]

    # strides for the three tensors (in elements, not bytes)
    stride_bl, stride_il, stride_kl, stride_hl = left.stride()
    stride_br, stride_ir, stride_kr, stride_hr = right.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out_fp16.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out_fp16,
        B,
        N,
        H,
        stride_bl,
        stride_il,
        stride_kl,
        stride_hl,
        stride_br,
        stride_ir,
        stride_kr,
        stride_hr,
        stride_bo,
        stride_io,
        stride_jo,
        stride_ho,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6) Cast back to fp32 and run the remaining (light) layers in PyTorch
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                     # apply output gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
