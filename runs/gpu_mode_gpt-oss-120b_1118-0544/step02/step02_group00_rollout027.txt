"""
TriMul (outgoing) forward pass accelerated with Triton.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the provided LN parameters.
2. Cast to fp16 and apply the four linear projections (left/right and their gates).
3. Optionally apply a binary mask before gating.
4. Apply sigmoid‑gates to the projected tensors.
5. Fuse the hidden dimension H into the batch axis:
      (B,N,N,H) → (B*H,N,N) for left and right.
   Transpose `right` so that the multiplication becomes a batched GEMM:
      out[b,i,j] = Σ_k left[b,i,k] * right_T[b,k,j]   (matrix‑multiply)
   This is performed by a custom Triton kernel that uses
   `tl.dot` (fp16×fp16 → fp32 accumulation) and tiled blocking.
6. Reshape back to (B,N,N,H), apply a second LayerNorm, multiply by the
   output gate, and linearly project back to the original channel dimension C.
7. Return the result in the same dtype as the input.

Only the O(N³·H) reduction (step 5) runs on the GPU via Triton; all other
operations use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers for the three tensors
    M, N, K,                           # matrix dimensions
    stride_ah, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # strides for B (batch, K, N)
    stride_ch, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b,i,j] = Σ_k A[b,i,k] * B[b,k,j]   (fp16·fp16 → fp32)."""
    pid_b = tl.program_id(2)   # batch * hidden dimension
    pid_m = tl.program_id(0)   # tile over M
    pid_n = tl.program_id(1)   # tile over N

    # Offsets for the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current batch slice
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # Pointers for the M×K and K×N tiles
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                                + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                                + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        # Load tiles with masking for boundary conditions
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)       # (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back the result (store as fp16)
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                                + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Triton‑based batched GEMM.
    A : (batch, M, K)   fp16
    B : (batch, K, N)   fp16  (already transposed)
    Returns C : (batch, M, N) fp16
    """
    assert A.shape == B.shape[:2] + (B.shape[2],)
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    # Choose block sizes. 128 is a good default for N≥512, else 64.
    if M <= 256:
        BLOCK = 64
    else:
        BLOCK = 128
    BLOCK_M = BLOCK_N = BLOCK
    BLOCK_K = 64 if K >= 64 else 32

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    _batched_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),   # batch, M, K strides
        B.stride(0), B.stride(1), B.stride(2),   # batch, K, N strides
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask
    Returns:
        torch.Tensor [B, N, N, C] with the same dtype as the input.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for the compute‑heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast all relevant weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4. Optional mask (binary) – broadcast on the hidden dim
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()    # (B,N,N,H)
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()    # (B,N,N,H)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Prepare tensors for the batched GEMM (fold hidden dim into batch)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape          # B, N, N, H
    # (B, H, N, N) → (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()          # (B*H, N, N)
    # Transpose right to shape (B*H, K, N) = (B*H, N, N) for GEMM
    right_T = right.permute(0, 2, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction via Triton batched GEMM
    # ------------------------------------------------------------------
    out_flat = _batched_gemm_triton(left, right_T)  # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8. Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)     # (B,N,N,dim)

    # Cast back to the original dtype if needed
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
