def custom_kernel(data):
    input_tensor, mask, weights, config = data
    ... // get dims
    # 1) LN
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)
    # 2) Proj
    left = F.linear(x, left_proj_weight)  # [B,N,N,H]
    right = F.linear(x, right_proj_weight)
    # 3) mask
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f
    # 4) Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)) # float32
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))
    # 5) Apply gates
    left = left * left_gate
    right = right * right_gate
    # Cast to half
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape

    # Permute to (B,H,N,N)
    left_permuted = left_fp16.permute(0,3,1,2).contiguous()
    right_permuted = right_fp16.permute(0,3,1,2).contiguous()
    # Right transpose in last two dims
    right_transposed = right_permuted.permute(0,1,3,2).contiguous()
    # reshape to [batch, M, K] where batch=B*H
    batch = B * H
    left_mat = left_permuted.view(batch, N, N)
    right_mat = right_transposed.view(batch, N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat, dtype=torch.float32)  # shape [batch,N,N]

    # Determine kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Strides
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    grid = (triton.cdiv(N, BLOCK_N), triton.cdiv(N, BLOCK_M), batch)

    @triton.jit
    def batched_gemm_kernel(
        a_ptr, b_ptr, c_ptr,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M: tl.constexpr,
        BLOCK_N: tl.constexpr,
        BLOCK_K: tl.constexpr,
    ):
        batch_id = tl.program_id(2)
        pid_m = tl.program_id(1)
        pid_n = tl.program_id(0)

        # Compute offsets for each block of output
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        # Initialize accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over K dimension
        for k in range(0, tl.cdiv(K, BLOCK_K)):
            # Offsets for a and b
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

            # Load a: shape (BLOCK_M, BLOCK_K)
            a_ptrs = a_ptr + batch_id * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
            a = tl.load(a_ptrs, mask=offs_m[:, None] < M and offs_k[None, :] < K, other=0.0).to(tl.float32)

            # Load b: shape (BLOCK_K, BLOCK_N)
            b_ptrs = b_ptr + batch_id * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
            b = tl.load(b_ptrs, mask=offs_k[:, None] < K and offs_n[None, :] < N, other=0.0).to(tl.float32)

            # Compute dot product
            acc += tl.dot(a, b)

        # Write back to C
        c_ptrs = c_ptr + batch_id * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
        mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
        tl.store(c_ptrs, acc, mask=mask)

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_aba, stride_ama, stride_aka,
    stride_bba, stride_bka, stride_bna,
    stride_cca, stride_cma, stride_cna,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # batch index
    batch_id = tl.program_id(2)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(0)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for M,N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

"""
Outgoing Triangle Multiplicative Update (TriMul) – forward only.

The core O(N³) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a custom batched GEMM kernel in Triton.
The kernel computes, for each hidden channel d (flattened into the batch dimension),
a matrix multiplication left[b,d] @ right[b,d].T using FP16 inputs and FP32
accumulation, eliminating an extra cast kernel used in the reference solution.
All layer‑norms, linear projections and gating are performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B where
       A: (batch, M, K)  FP16
       B: (batch, K, N)  FP16 (already transposed)
       C: (batch, M, N)  FP32
    """
    batch_id = tl.program_id(2)          # batch index (B*H)
    pid_m = tl.program_id(1)             # block row
    pid_n = tl.program_id(0)             # block col

    # -------------------- offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- reduction over K --------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = (a_ptr
                  + batch_id * stride_a_batch
                  + offs_m[:, None] * stride_a_m
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B tile: (BLOCK_K, BLOCK_N)
        b_ptrs = (b_ptr
                  + batch_id * stride_b_batch
                  + offs_k[:, None] * stride_b_k
                  + offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # -------------------- write back --------------------
    c_ptrs = (c_ptr
              + batch_id * stride_c_batch
              + offs_m[:, None] * stride_c_m
              + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Perform the outgoing TriMul forward pass.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict with the model parameters
            - config: dict with at least "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------
    # 0) extract weights
    # ------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------
    # 1) LayerNorm over the last dimension
    # ------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, N, N, dim]

    # ------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------
    left = F.linear(x, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, hidden_dim]

    # ------------------------------------------------------
    # 5) Apply gates and cast to FP16
    # ------------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    B, N, _, H = left.shape  # B, N, N, hidden_dim

    # ------------------------------------------------------
    # 6) Prepare tensors for batched GEMM
    # ------------------------------------------------------
    #   left [B, N, N, H] -> (B, H, N, N) -> (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(B * H, N, N)

    #   right needs to be transposed on the last two dims:
    #   right [B, N, N, H] -> (B, H, N, N) then transpose (K,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, H, N, N) with (k,i)
    right_mat = right_perm.view(B * H, N, N)

    # Output buffer (FP32)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=left_mat.device)

    # ------------------------------------------------------
    # 7) Launch Triton batched GEMM kernel
    # ------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32

    a_batch_stride, a_m_stride, a_k_stride = left_mat.stride()
    b_batch_stride, b_k_stride, b_n_stride = right_mat.stride()
    c_batch_stride, c_m_stride, c_n_stride = out_mat.stride()

    grid = (
        (N + BLOCK_N - 1) // BLOCK_N,   # program ids for columns
        (N + BLOCK_M - 1) // BLOCK_M,   # program ids for rows
        B * H,                          # batch dimension
    )

    batched_gemm_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=a_batch_stride,
        stride_a_m=a_m_stride,
        stride_a_k=a_k_stride,
        stride_b_batch=b_batch_stride,
        stride_b_k=b_k_stride,
        stride_b_n=b_n_stride,
        stride_c_batch=c_batch_stride,
        stride_c_m=c_m_stride,
        stride_c_n=c_n_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------
    # 8) Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------
    out = (
        out_mat.view(B, H, N, N)       # (B, H, N, N)
        .permute(0, 2, 3, 1)           # (B, N, N, H)
        .contiguous()
    )

    # ------------------------------------------------------
    # 9) Final LayerNorm, gating and projection
    # ------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # back to original dim

    return out
