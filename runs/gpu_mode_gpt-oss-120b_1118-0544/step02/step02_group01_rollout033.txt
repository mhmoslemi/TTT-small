"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the reference PyTorch module but fuses elementwise
mask‑gate multiplication with a tiny Triton kernel and replaces the O(N³)
reduction with a custom batched GEMM kernel written in Triton.  All layer‑norm
and linear operations are performed with PyTorch (float32), while the heavy
matrix multiplication is carried out in FP16 inside Triton to exploit the
GPU’s tensor cores.  The final output is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def fused_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK: tl.constexpr,
):
    """
    out = proj * gate * mask   (broadcasted mask over the hidden dimension)
    proj, gate : [B, N, N, H]  (float32)
    mask       : [B, N, N]     (float32)
    out        : [B, N, N, H]  (float32)
    """
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)

    # linear index -> (b,i,j,h)
    tmp = offs
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # bounds
    valid = (b < B) & (i < N) & (j < N) & (h < H)

    # compute flat offsets
    proj_off = (b * stride_proj_b + i * stride_proj_i +
                j * stride_proj_j + h * stride_proj_h)
    gate_off = (b * stride_gate_b + i * stride_gate_i +
                j * stride_gate_j + h * stride_gate_h)
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_off = (b * stride_out_b + i * stride_out_i +
               j * stride_out_j + h * stride_out_h)

    proj = tl.load(proj_ptr + proj_off, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=valid, other=0.0)

    tl.store(out_ptr + out_off, proj * gate * m, mask=valid)


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ B   where
        A : [batch, M, K]
        B : [batch, K, N]
        C : [batch, M, N]
    All tensors are in FP16, accumulation in FP32.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute tile start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds for M and N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k_block in range(0, num_k_blocks):
        k_start = k_block * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile (K x N)
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # accumulate
        acc += tl.dot(a, b)

    # convert to FP16 and store
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# TriMul forward entry point
# ----------------------------------------------------------------------


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool or float) or None
        - weights      : dict of weight tensors (float32)
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, C = x.shape
    hidden = cfg["hidden_dim"]
    device = x.device

    # ------------------------------------------------------------------
    # 1) input layer norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) linear projections & gating (float32)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 3) fuse mask and gate via Triton (if a mask is provided)
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure mask is float32 on same device
        mask_f = mask.to(dtype=x.dtype, device=device)

        # make inputs contiguous for safe stride handling
        left_proj_c = left_proj.contiguous()
        right_proj_c = right_proj.contiguous()
        left_gate_c = left_gate.contiguous()
        right_gate_c = right_gate.contiguous()
        mask_c = mask_f.contiguous()

        # allocate output buffers
        left = torch.empty_like(left_proj_c)
        right = torch.empty_like(right_proj_c)

        # strides (in elements)
        s_lp_b, s_lp_i, s_lp_j, s_lp_h = left_proj_c.stride()
        s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate_c.stride()
        s_rp_b, s_rp_i, s_rp_j, s_rp_h = right_proj_c.stride()
        s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate_c.stride()
        s_m_b, s_m_i, s_m_j = mask_c.stride()
        s_out_b, s_out_i, s_out_j, s_out_h = left.stride()

        total = B * N * N * hidden
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        # left side
        fused_mask_gate_kernel[grid](
            left_proj_c,
            left_gate_c,
            mask_c,
            left,
            B,
            N,
            hidden,
            s_lp_b,
            s_lp_i,
            s_lp_j,
            s_lp_h,
            s_lg_b,
            s_lg_i,
            s_lg_j,
            s_lg_h,
            s_m_b,
            s_m_i,
            s_m_j,
            s_out_b,
            s_out_i,
            s_out_j,
            s_out_h,
            BLOCK=BLOCK,
        )
        # right side
        fused_mask_gate_kernel[grid](
            right_proj_c,
            right_gate_c,
            mask_c,
            right,
            B,
            N,
            hidden,
            s_rp_b,
            s_rp_i,
            s_rp_j,
            s_rp_h,
            s_rg_b,
            s_rg_i,
            s_rg_j,
            s_rg_h,
            s_m_b,
            s_m_i,
            s_m_j,
            s_out_b,
            s_out_i,
            s_out_j,
            s_out_h,
            BLOCK=BLOCK,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4) core O(N³) update: batched GEMM in FP16 via Triton
    # ------------------------------------------------------------------
    # cast to half for the heavy compute
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reorder dimensions to (B, hidden, N, N) and prepare right transpose
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, K]
    right = right.permute(0, 3, 2, 1).contiguous()  # [B, hidden, K, N]

    # fuse batch and hidden dimensions
    batch_hidden = B * hidden
    left_mat = left.view(batch_hidden, N, N)   # [B*H, M, K]
    right_mat = right.view(batch_hidden, N, N)  # [B*H, K, N]
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # strides in elements
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_k = right_mat.stride(1)
    stride_b_n = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    # tile sizes – tuned for typical B200 memory layout
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden]
    out_hidden = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # ------------------------------------------------------------------
    # 5) output layer norm, gating and final linear projection
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])
    return out
