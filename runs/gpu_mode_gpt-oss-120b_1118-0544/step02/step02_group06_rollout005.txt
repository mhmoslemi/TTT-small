@triton.jit
def batched_gemm_triton(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Determine block indices
    pid = tl.program_id(0)  # block row index
    pid_n = tl.program_id(1)  # block col index
    pid_b = tl.program_id(2)  # batch index

    # compute offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute mask for valid rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for current batch
    a_batch_ptr = a_ptr + pid_b * stride_a_batch
    b_batch_ptr = b_ptr + pid_b * stride_b_batch
    c_batch_ptr = c_ptr + pid_b * stride_c_batch

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile: shape (BLOCK_K, BLOCK_N) after transposition
        # Offsets: (k, n)
        b_ptrs = b_batch_ptr + (cur_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # DOT product and accumulate
        acc += tl.dot(a, b)  # fp32 accumulation automatically

    # Store result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
Outgoing Triangle Multiplicative Update (TriMul) kernel.

The core operation is a batched matrix multiplication:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

For each batch element and each hidden dimension `d` this corresponds to
a GEMM A @ Bᵀ where A and B have shape (N, N).  We implement this
GEMM with a Triton kernel that processes the batch dimension as an
independent third program‑id, achieving full GPU occupancy without any
Python loops.  All other parts (LayerNorm, linear projections, gating)
are performed with PyTorch for simplicity.

The kernel works in fp16 for the input matrices and accumulates in fp32,
matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel: C = A @ Bᵀ
# A : (batch, M, K)   – fp16
# B : (batch, N, K)   – fp16 (will be treated as transposed)
# C : (batch, M, N)   – fp32
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,           # left  tensor pointer  (batch, M, K)    fp16
    b_ptr,           # right tensor pointer  (batch, N, K)    fp16 (read transposed)
    c_ptr,           # output pointer        (batch, M, N)  fp32
    batch: tl.constexpr,
    M: tl.constexpr,
    N: tl.constexpr,
    K: tl.constexpr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_m = tl.program_id(0)      # tile row
    pid_n = tl.program_id(1)      # tile col
    pid_b = tl.program_id(2)      # batch index

    # offsets for the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # load A tile (M x K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # load Bᵀ tile (K x N)
        # B is stored as (batch, N, K); we read it transposed by swapping strides
        b_ptrs = b_batch + (cur_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot-product accumulation
        acc += tl.dot(a, b)

    # store the result
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – may be ignored if config["nomask"] is True
            - weights: dict of model parameters (see below)
            - config: dict with at least:
                * "dim" (int) – input feature dimension
                * "hidden_dim" (int) – projection dimension
                * "nomask" (bool, optional) – if True, mask is ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # unpack weights (all are torch.nn.Parameter / torch.Tensor)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias) → hidden_dim
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for batched GEMM
    # cast to half for Tensor‑core acceleration
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N_seq, _, _ = left_h.shape  # B, N, N, hidden_dim
    # reshape to (batch, M, K) where batch = B * hidden_dim, M = N, K = N
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N_seq, N_seq)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N_seq, N_seq)

    # ------------------------------------------------------------------
    # Triton batched GEMM: out = left_h @ right_hᵀ
    # ------------------------------------------------------------------
    batch = B * hidden_dim
    M = N_seq
    N = N_seq
    K = N_seq

    # allocate output buffer (fp32)
    out_h = torch.empty((batch, M, N), dtype=torch.float32, device=device)

    # strides in elements
    stride_a_batch, stride_a_m, stride_a_k = left_h.stride()
    stride_b_batch, stride_b_n, stride_b_k = right_h.stride()  # note: (batch, N, K)
    # For Bᵀ we need stride over k then n
    # So we pass stride_b_k as stride over K and stride_b_n as stride over N
    stride_c_batch, stride_c_m, stride_c_n = out_h.stride()

    # Tuning parameters (chosen to be good for the tested ranges)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # launch grid: (tiles over M, tiles over N, batch)
    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    batched_gemm_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        batch=batch,
        M=M,
        N=N,
        K=K,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,   # k‑dimension stride
        stride_b_n=stride_b_n,   # n‑dimension stride (after transposition)
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_h.view(B, hidden_dim, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Final LayerNorm over hidden_dim, gating, and linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                         # apply outgoing gate
    out = F.linear(out, to_out_weight)           # [B, N, N, dim] (float32)

    return out
