- We have grid dimensions: number of i blocks = ceil(N/BLOCK_M), j blocks = ceil(N/BLOCK_N), hidden blocks = ceil(D/BLOCK_D), batch dimension maybe separate.

- For each block, we compute partial sums for all i and j within block and for hidden slice. We'll iterate over k in blocks of BLOCK_K.

- Use `tl.dot` to compute a small matrix multiplication for each hidden slice: within each k iteration, we load a tile of left of shape (BLOCK_M, BLOCK_K) for a given implicit hidden slice; we load a tile of right of shape (BLOCK_N, BLOCK_K). Multiply with `tl.dot` giving (BLOCK_M, BLOCK_N). Accumulate to accumulator for that hidden slice.

- For all hidden slices inside a BLOCK_D loop, we repeat.

Thus the kernel will have a triple loop: for d in hidden slices; for k in K blocks; compute dot.

Implementation details:

- Compute pointer offsets for left: shape [B, N, N, D]. We need to access left[b, i, k, d] where i = block_i + offset_i, k = block_k + offset_k, d = block_d + offset_d.

- Strides: we can compute strides from pointer info via python: left_ptr.stride(0), etc. In Triton we receive strides as arguments.

"""
fast_trimul.py

Implementation of the outgoing Triangle Multiplicative Update (TriMul) used in
AlphaFold3.  The core O(N³·C_hidden) reduction is performed by a custom Triton
kernel that computes a batched matrix multiplication across the "k" axis:

    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]

All other operations (LayerNorm, linear projections, gating, final
normalisation and projection) are done with PyTorch.  The kernel processes
each (batch, hidden) channel pair as a separate program, using FP16 loads and
Tensor‑Core `tl.dot` for the inner‑product while accumulating in FP32 for
numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, C,                               # batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_c,
    stride_right_b, stride_right_j, stride_right_k, stride_right_c,
    stride_out_b, stride_out_i, stride_out_j, stride_out_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]"""

    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_c = tl.program_id(2)          # combined (batch, hidden) channel

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # decode batch and hidden index from pid_c
    b = pid_c // C
    c = pid_c % C

    # create range indices
    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)

    i_mask = i_range < N
    j_mask = j_range < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k):
        k0 = k_tile * BLOCK_K
        k_range = k0 + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # ---------- left tile (M × K) ----------
        # left[b, i, k, c]
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            i_range[:, None] * stride_left_i + \
            k_range[None, :] * stride_left_k + \
            c * stride_left_c
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0
        )  # fp16

        # ---------- right tile (K × N) ----------
        # right[b, j, k, c] (note the k dimension is first for dot)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            k_range[:, None] * stride_right_k + \
            j_range[None, :] * stride_right_j + \
            c * stride_right_c
        right = tl.load(
            right_ptrs,
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0
        )  # fp16

        # dot product (M×K)·(K×N) -> (M×N)
        # tl.dot returns the dtype of its operands (fp16); cast to fp32.
        acc += tl.dot(left, right).to(tl.float32)

    # ---------- store ----------
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        i_range[:, None] * stride_out_i + \
        j_range[None, :] * stride_out_j + \
        c * stride_out_c
    tl.store(
        out_ptrs,
        acc,
        mask=i_mask[:, None] & j_mask[None, :]
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights      : dict of model weights (torch.Tensors)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Unpack weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    B, N, _, _ = input_tensor.shape

    # -------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (dim)
    # -------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # -------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # -------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # -------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # -------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # 5️⃣ Cast to fp16 for the heavy reduction
    # -------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # -------------------------------------------------------------
    # 6️⃣ Triton kernel: out_hidden[b,i,j,c] = Σ_k left*b * right*b
    # -------------------------------------------------------------
    out_hidden = torch.empty((B, N, N, hidden_dim), device=device, dtype=torch.float32)

    # strides (in elements)
    s_l_b, s_l_i, s_l_k, s_l_c = left_fp16.stride()
    s_r_b, s_r_j, s_r_k, s_r_c = right_fp16.stride()
    s_o_b, s_o_i, s_o_j, s_o_c = out_hidden.stride()

    # tile sizes – tuned for Ampere/B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_c = B * hidden_dim          # one program per (batch, hidden) pair

    tri_mul_kernel[(grid_i, grid_j, grid_c)](
        left_fp16,
        right_fp16,
        out_hidden,
        B,
        N,
        hidden_dim,
        s_l_b, s_l_i, s_l_k, s_l_c,
        s_r_b, s_r_j, s_r_k, s_r_c,
        s_o_b, s_o_i, s_o_j, s_o_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # -------------------------------------------------------------
    # 7️⃣ Final LayerNorm over hidden dimension
    # -------------------------------------------------------------
    out = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # -------------------------------------------------------------
    # 8️⃣ Apply out_gate element‑wise (still in fp32)
    # -------------------------------------------------------------
    out = out * out_gate

    # -------------------------------------------------------------
    # 9️⃣ Project back to original dimension
    # -------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
