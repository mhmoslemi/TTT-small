"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The core operation is a batched matrix multiplication:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be written as a batched GEMM:
    out_{b,d} = left_{b,d} @ right_{b,d}ᵀ
where left_{b,d} and right_{b,d} are (N,N) matrices.
The GEMM is performed in mixed‑precision (fp16 inputs, fp32 accumulation)
with a custom Triton kernel. All other operations (LayerNorms,
linear projections, gating, masking, final projection) are done with
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr,               # left matrix pointer (BH, M, K)   fp16
    b_ptr,               # right matrix pointer (BH, K, N) fp16 (already transposed)
    c_ptr,               # output pointer (BH, M, N)       fp32
    M, N, K,             # matrix dimensions
    stride_ah, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # strides for B (batch, K, N)
    stride_ch, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ  (A: (BH,M,K), B: (BH,K,N))"""
    pid_bh = tl.program_id(0)          # batch * head dimension
    pid_m = tl.program_id(1)           # block row
    pid_n = tl.program_id(2)           # block column

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch/head
    a_batch = a_ptr + pid_bh * stride_ah
    b_batch = b_ptr + pid_bh * stride_bh
    c_batch = c_ptr + pid_bh * stride_ch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Tile over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Pointers to the current tiles
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (fp16) with zero padding for OOB elements
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        # Multiply‑accumulate (accumulates in fp32)
        acc += tl.dot(a, b)

    # Store the result
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) or None
            - weights: dict of model parameters (all float32 tensors)
            - config: dict containing 'dim' and 'hidden_dim'
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden_dim (cast to fp16)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)                     # [B, N, N, dim]  fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim] fp16
    right = F.linear(x_h, right_proj_w) # same dtype

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gates (computed in FP32, then cast)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # FP32

    left_gate_h = left_gate.to(x_h.dtype)
    right_gate_h = right_gate.to(x_h.dtype)

    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 5) Core contraction using Triton (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape               # B, seq_len, seq_len, hidden_dim
    BH = B * hidden_dim                    # collapsed batch * head dimension

    # Permute to (B, hidden_dim, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) – K becomes first axis

    left_mat = left_perm.view(BH, N, N)                # (BH, M=N, K=N)   fp16
    right_mat = right_perm.view(BH, N, N)              # (BH, K=N, N)   fp16 (already transposed)

    # Output buffer (FP32 accumulator)
    out_mat = torch.empty((BH, N, N), dtype=torch.float32, device=x.device)

    # Tunable block sizes (good for up to N=1024)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim] (FP32)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # FP32 * FP32
    out = F.linear(out, weights["to_out.weight"])  # back to dim (FP32)

    return out
```
