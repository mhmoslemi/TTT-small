def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # 1. Input layer norm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)

    # 2. Projections
    x_h = x.to(torch.float16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # shape [B,N,N,H]
    right = F.linear(x_h, right_proj_w)

    # 3. Mask
    if mask is not None:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # 4. Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate.to(x_h.dtype)
    right = right * right_gate.to(x_h.dtype)

    # 5. Core contraction via Triton
    B, N, _, _ = left.shape

    # Permute to (B, H, N, N) / (B, H, K, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)
    out_mat = torch.empty((BH, N, N), dtype=torch.float32, device=x.device)

    # Launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _trmul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

                       bias=weights["to_out_norm.bias"],

    # 7. Final linear
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass – Triton implementation.

The module first applies a LayerNorm and two linear projections (left/right)
followed by optional masking and gating.  The core O(N³) contraction

    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]

is expressed as a batched matrix‑multiplication:
    (B·H, N, N) @ (B·H, N, N)ᵀ
and is executed by a custom Triton kernel with mixed‑precision (FP16 inputs,
FP32 accumulation).  The result is reshaped, normalized, gated and projected
back to the original feature dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ Bᵀ) with FP16 inputs & FP32 acc.
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({'BLOCK_M': 64,  'BLOCK_N': 64,  'BLOCK_K': 32},  num_stages=2, num_warps=4),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 64,  'BLOCK_K': 32},  num_stages=2, num_warps=8),
        triton.Config({'BLOCK_M': 64,  'BLOCK_N': 128, 'BLOCK_K': 32},  num_stages=2, num_warps=8),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 32},  num_stages=2, num_warps=8),
    ],
    key=['M', 'N', 'K']
)
@triton.jit
def _trmul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions (int32)
    stride_a_batch, stride_am, stride_ak,   # strides for A (BH, i, k)
    stride_b_batch, stride_bk, stride_bn,   # strides for B (BH, k, j)
    stride_c_batch, stride_cm, stride_cn,   # strides for C (BH, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ.

    A shape: (BH, M, K)  fp16
    B shape: (BH, K, N)  fp16   (right already transposed)
    C shape: (BH, M, N)  fp32 accumulator
    """
    pid_batch = tl.program_id(0)   # batch * head index
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile column

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch/head
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Pointers to the current tiles
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (FP16) with zero padding for OOB elements
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        # Matrix‑multiply‑accumulate (accumulate in FP32)
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Write the result (FP32)
    # ------------------------------------------------------------------
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with keys
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            All weight tensors are float32.
            - config: dict with entries 'dim' and 'hidden_dim'.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0) Unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2) Projections to hidden_dim (FP16 for speed)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)    # [B, N, N, hidden_dim]  fp16
    right = F.linear(x_h, right_proj_w) # same dtype

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gates (computed in FP32, then cast)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # fp32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # fp32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # fp32

    left = left * left_gate.to(x_h.dtype)
    right = right * right_gate.to(x_h.dtype)

    # ------------------------------------------------------------------
    # 5) Core N³ contraction via Triton (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Permute to (B, H, N, N) and (B, H, K, J) where K == J == N
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) – already transposed in K,J

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # (BH, M, K)
    right_mat = right_perm.view(BH, N, N)  # (BH, K, N)
    out_mat = torch.empty((BH, N, N), dtype=torch.float32, device=x.device)  # (BH, M, N)

    # Tunable block sizes (chosen for good occupancy on B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _trmul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # out_gate is FP32, out is FP32
    out = F.linear(out, weights["to_out.weight"])  # back to dim (FP32)

    return out
