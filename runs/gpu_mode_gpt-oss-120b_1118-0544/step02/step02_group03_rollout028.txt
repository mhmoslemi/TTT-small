TriMul outgoing update (AlphaFold3) – Triton‑accelerated core.

Algorithm

7. Layer‑norm the hidden result, multiply by the out‑gate and project back
   to the original dimension with a final linear layer.
8. Return the result in FP32.

All heavy computation (step 6) runs on the GPU via Triton; the remaining
operations use PyTorch but stay on the same device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Batched matmul kernel: C = A @ Bᵀ
# A : (batch, M, K)  fp16
# B : (batch, N, K)  fp16   (note B is stored transposed to match A@Bᵀ)
# C : (batch, M, N)  fp32
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,  # pointers
    M, N, K,              # problem size
    stride_az, stride_am, stride_ak,   # strides of A (z, m, k)
    stride_bz, stride_bn, stride_bk,   # strides of B (z, n, k) – B is stored as (batch, N, K)
    stride_cz, stride_cm, stride_cn,   # strides of C (z, m, n)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)   # batch index
    pid_m = tl.program_id(0)   # M tile
    pid_n = tl.program_id(1)   # N tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # compute pointers for this tile
        a_ptrs = (
            A_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        b_ptrs = (
            B_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk   # note B is stored as (batch, N, K)
            + offs_n[None, :] * stride_bn
        )

        # load with zero padding
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # fp16·fp16 -> fp32 accumulation
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # store result
    c_ptrs = (
        C_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)


def batched_matmul_fp16(A: torch.Tensor, B_T: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.
    A : (batch, M, K)  fp16
    B_T : (batch, N, K) fp16  (right matrix already transposed)
    Returns C : (batch, M, N) fp32
    """
    assert A.is_contiguous() and B_T.is_contiguous()
    batch, M, K = A.shape
    _, N, K2 = B_T.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # strides in elements
    stride_az, stride_am, stride_ak = A.stride()
    stride_bz, stride_bn, stride_bk = B_T.stride()
    stride_cz, stride_cm, stride_cn = C.stride()

    # tile sizes (tuned for A100/B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )

    batched_matmul_kernel[grid](
        A,
        B_T,
        C,
        M,
        N,
        K,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bn,
        stride_bk,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model tensors
        - config       : dict with keys "dim", "hidden_dim", "nomask" etc.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]          # input feature dimension C
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on input (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # still float32

    # ------------------------------------------------------------------
    # Cast to fp16 for heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections (left/right) to hidden_dim
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_weight)   # [B, N, N, H]  fp16
    right = F.linear(x_h, right_proj_weight)  # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # Optional masking (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear layers on the normalized input)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_weight))   # [B,N,N,H] fp16
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_weight))  # [B,N,N,H] fp16
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight))      # [B,N,N,H] fp16

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³) update using batched matmul: out = left @ rightᵀ
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden

    # Rearrange to (batch, hidden, N, N) then flatten batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    batch_hidden = B * H
    left_flat = left_perm.reshape(batch_hidden, N, N)   # (BH, N, N)
    right_flat = right_perm.reshape(batch_hidden, N, N) # (BH, N, N)

    # Transpose right for A @ Bᵀ
    right_T = right_flat.permute(0, 2, 1).contiguous()  # (BH, N, N)

    # Batched matrix multiplication (fp16 inputs → fp32 output)
    out_flat = batched_matmul_fp16(left_flat, right_T)  # (BH, N, N) fp32

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] fp32

    # ------------------------------------------------------------------
    # Output layer‑norm (hidden dimension)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # still fp32

    # ------------------------------------------------------------------
    # Apply out‑gate
    # ------------------------------------------------------------------
    out = out * out_gate.to(out.dtype)   # out_gate was fp16, cast to fp32 for multiplication

    # ------------------------------------------------------------------
    # Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    out_fp16 = out.to(torch.float16)
    output = F.linear(out_fp16, to_out_weight)   # [B,N,N,dim] fp16
    output = output.to(torch.float32)

    return output
