    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr

   pid_j = tl.program_id(1)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # derive batch and hidden index
    b = pid_bh // hidden
    h = pid_bh % hidden

    # compute offsets
    i_base = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_base = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i_base < N
    mask_j = j_base < N

    # base pointers for this batch and hidden
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # number of k tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k):
        k_base = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_base < N

        # Load left [i,k]
        left_offs = i_base[:, None] * stride_left_i + k_base[None, :] * stride_left_k
        left = tl.load(left_base + left_offs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # Load right_T [k,j] (transposed view)
        right_offs = k_base[:, None] * stride_right_k + j_base[None, :] * stride_right_j
        right_t = tl.load(right_base + right_offs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Multiply-accumulate using tensor core dot (FP16->FP32)
        acc += tl.dot(left, right_t)

    # Store result (fp16) to out
    out = acc.to(tl.float16)
    out_offs = i_base[:, None] * stride_out_i + j_base[None, :] * stride_out_j
    tl.store(out_base + out_offs, out, mask=mask_i[:, None] & mask_j[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # derive batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # tile offsets
    i_idx = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_idx = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i_idx < N
    mask_j = j_idx < N

    # base pointers for this batch and hidden channel
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator in float32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k):
        k_idx = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # Load left[i,k]
        left_off = i_idx[:, None] * stride_left_i + k_idx[None, :] * stride_left_k
        left = tl.load(left_base + left_off,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # Load right_T[k,j] (right transposed)
        right_off = k_idx[:, None] * stride_right_k + j_idx[None, :] * stride_right_j
        right_t = tl.load(right_base + right_off,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0)

        # Accumulate
        acc += tl.dot(left, right_t)   # fp16*fp16 -> fp32

    # Store result as fp16
    out_tile = acc.to(tl.float16)
    out_off = i_idx[:, None] * stride_out_i + j_idx[None, :] * stride_out_j
    tl.store(out_base + out_off,
             out_tile,
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul forward pass using Triton for the O(N^3) contraction.

    The heavy reduction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is performed by a custom Triton kernel that implements a tiled
    matrix multiplication for each hidden channel. All other ops
    (LayerNorm, linear projections, gating) are executed with PyTorch.
    The final output has shape [B, N, N, dim] and dtype float32.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # unpack weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1. LayerNorm on input
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # 2. Linear projections to hidden_dim (half precision)
    left = F.linear(x, left_proj_weight).to(torch.float16)
    right = F.linear(x, right_proj_weight).to(torch.float16)

    # 3. Optional mask (broadcast to hidden dim)
    if not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # 4. Gating (sigmoid) – computed in float32 then cast
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # stays fp32 for later

    left = left * left_gate
    right = right * right_gate

    # 5. Core contraction with Triton
    B, N, _, _ = left.shape

    # strides in elements
    lsb, lsi, lsk, lsh = left.stride()
    rsb, rsj, rsk, rsh = right.stride()
    # output buffer
    out = torch.empty_like(left)   # same shape & dtype (fp16)

    osb, osi, osj, osh = out.stride()

    # block sizes (tuned for H100)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    grid = (
        ( (N + BLOCK_I - 1) // BLOCK_I,
          (N + BLOCK_J - 1) // BLOCK_J,
          B * hidden_dim ),
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        lsb, lsi, lsk, lsh,
        rsb, rsj, rsk, rsh,
        osb, osi, osj, osh,
        BLOCK_I,
        BLOCK_J,
        BLOCK_K,
    )

    # 6. Convert to fp32 and apply output layernorm + gate
    out = out.to(torch.float32)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # 7. Apply out gate (elementwise) – we reuse a tiny Triton kernel for consistency
    # Define small kernel inline (or reuse earlier definition)
    # We'll implement a simple elementwise multiplication kernel
    BLOCK_E = 1024
    total = out.numel()
    grid_e = ( (total + BLOCK_E - 1) // BLOCK_E, )
    # Ensure the kernel is defined before usage

        out.view(-1),
        out_gate.view(-1),
        out.view(-1),
        total,
        BLOCK=BLOCK_E,
    )

    # 8. Final linear projection back to dim
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton accelerated core contraction.

The O(N³) reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a tiled matrix multiplication using a custom Triton kernel.
All other lightweight operations (LayerNorm, Linear projections, gating) are
performed with PyTorch. The kernel works in FP16 with FP32 accumulation,
leveraging Tensor Cores on H100. The final output is FP32 with shape
[batch, seq_len, seq_len, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Element‑wise multiplication kernel (used for the final out‑gate)
# -------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """c = a * b  (1‑D flattened tensors)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)

# -------------------------------------------------------------
# Core TriMul contraction kernel
# -------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)               # i‑tile
    pid_j = tl.program_id(1)               # j‑tile
    pid_bh = tl.program_id(2)              # combined batch‑hidden index

    # decode batch and hidden channel
    b = pid_bh // H
    h = pid_bh % H

    # tile start indices
    i_idx = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_idx = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i_idx < N
    mask_j = j_idx < N

    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k):
        k_idx = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # left[i, k]
        left_off = i_idx[:, None] * stride_left_i + k_idx[None, :] * stride_left_k
        left = tl.load(left_base + left_off,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right_T[k, j]  (transposed view of right[j, k])
        right_off = k_idx[:, None] * stride_right_k + j_idx[None, :] * stride_right_j
        right_t = tl.load(right_base + right_off,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0)

        acc += tl.dot(left, right_t)   # FP16·FP16 → FP32

    # write result as FP16
    out_tile = acc.to(tl.float16)
    out_off = i_idx[:, None] * stride_out_i + j_idx[None, :] * stride_out_j
    tl.store(out_base + out_off,
             out_tile,
             mask=mask_i[:, None] & mask_j[None, :])

# -------------------------------------------------------------
# Entrypoint – forward pass of the outgoing TriMul module
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored when config["nomask"] is True
            - weights: dict of model parameters
            - config: dict containing at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Unpack weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections to hidden_dim (use FP16 for the heavy tensors)
    # -----------------------------------------------------------------
    left  = F.linear(x, left_proj_weight).to(torch.float16)
    right = F.linear(x, right_proj_weight).to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)      # [B, N, N, 1]
        left  = left  * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – keep gates in FP32, cast to FP16 for multiplication
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # stays FP32

    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Core O(N³) contraction via Triton
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # Ensure contiguity (required for correct stride handling)
    left  = left.contiguous()
    right = right.contiguous()

    # strides in element units
    lsb, lsi, lsk, lsh = left.stride()
    rsb, rsj, rsk, rsh = right.stride()

    out = torch.empty_like(left)   # [B, N, N, hidden_dim] FP16
    osb, osi, osj, osh = out.stride()

    # block sizes (tuned for H100)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    grid = (
        ( (N + BLOCK_I - 1) // BLOCK_I,
          (N + BLOCK_J - 1) // BLOCK_J,
          B * hidden_dim ),
    )

    tri_mul_kernel[grid](
        left,                # left_ptr
        right,               # right_ptr
        out,                 # out_ptr
        B,
        N,
        hidden_dim,
        lsb, lsi, lsk, lsh,
        rsb, rsj, rsk, rsh,
        osb, osi, osj, osh,
        BLOCK_I,
        BLOCK_J,
        BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Convert back to FP32, apply final LayerNorm and out‑gate
    # -----------------------------------------------------------------
    out = out.to(torch.float32)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    # out_gate multiplication (element‑wise) via a tiny Triton kernel
    total_el = out.numel()
    BLOCK_E = 1024
    grid_e = ((total_el + BLOCK_E - 1) // BLOCK_E, )
    elementwise_mul_kernel[grid_e](
        out.view(-1),
        out_gate.view(-1),
        out.view(-1),
        total_el,
        BLOCK=BLOCK_E,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Project back to the original dimensionality
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
