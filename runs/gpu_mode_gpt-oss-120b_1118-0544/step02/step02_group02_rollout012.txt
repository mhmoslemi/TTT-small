```
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_batch = tl.program_id(2)  # batch*hidden
    
    # decode batch and hidden
    b = pid_batch // H
    h = pid_batch % H
    
    # compute offsets
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    
    # mask out-of-bounds
    mask_i = i_offset < N
    mask_j = j_offset < N
    
    # compute base pointers for this batch and hidden dim
    left_batch_ptr = left_ptr + b * stride_left_b + h * stride_left_d
    right_batch_ptr = right_ptr + b * stride_right_b + h * stride_right_d
    out_batch_ptr = out_ptr + b * stride_out_b + h * stride_out_d
    
    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N
        
        # left tile: (M, K)
        left_ptrs = left_batch_ptr + i_offset[:, None] * stride_left_i + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)
        
        # right tile: (K, N) -> we need it as (K,N) shape
        # Right dims: (b, j, k, d) -> we want rows K, cols N (j)
        right_ptrs = right_batch_ptr + k_offset[:, None] * stride_right_k + j_offset[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float16)
        
        # compute dot product
        acc += tl.dot(left, right)
    
    # store result
    out_ptrs = out_batch_ptr + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])
```

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm:
1. Layer‑norm the input tensor.
2. Linear projections for left/right and three sigmoid gates (left, right, out).
3. Optional mask is applied to left/right before gating.
4. Apply element‑wise gates and cast left/right to FP16.
5. Compute the heavy reduction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a custom Triton GEMM kernel where each (batch, hidden) pair is a
   separate matrix multiply of size N×N with accumulation in FP32.
6. Apply a second LayerNorm, multiply by the out‑gate and a final linear
   projection back to the original dimension.
The kernel tiles the multiplication (BLOCK_M × BLOCK_N output tiles,
BLOCK_K reduction tiles) and uses tensor‑core `tl.dot` for high throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,          # FP16 left tensor  [B, N, N, H]
    right_ptr,         # FP16 right tensor [B, N, N, H]
    out_ptr,           # FP32 output tensor [B, N, N, H]
    B, N, H,           # runtime sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes, for each (b, h), the matrix product:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    using tiled GEMM with TensorCore dot products.
    """
    pid_m = tl.program_id(0)          # tile row   (i dimension)
    pid_n = tl.program_id(1)          # tile col   (j dimension)
    pid_batch = tl.program_id(2)      # combined batch * hidden index

    # decode batch and hidden indices
    b = pid_batch // H
    h = pid_batch - b * H

    # starting coordinates of the tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # thread‑local offsets within the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, h) pair
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_d
    right_base = right_ptr + b * stride_right_b + h * stride_right_d
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_d

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of tiles over the reduction dimension K
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for k_tile in range(0, num_k_tiles):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (M x K) -----
        left_ptrs = left_base + offs_i[:, None] * stride_left_i \
                                 + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=tl.float16(0.0),
                       dtype=tl.float16)

        # ----- load right tile (K x N) -----
        # right is indexed as (k, j) for the dot product
        right_ptrs = right_base + offs_k[:, None] * stride_right_k \
                                    + offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=tl.float16(0.0),
                        dtype=tl.float16)

        # ----- accumulate dot product -----
        acc += tl.dot(left, right)   # (M, N) in FP32

    # ----- store result -----
    out_ptrs = out_base + offs_i[:, None] * stride_out_i \
                        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Outgoing TriMul forward pass with a Triton‑accelerated reduction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict containing the model's parameters
            - config: dict with at least "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) Load model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim], float32
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5) Apply gates and cast to FP16 for the reduction
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16).contiguous()
    right = (right * right_gate).to(torch.float16).contiguous()

    B, N, _, _ = left.shape  # B, N, N, hidden_dim

    # ------------------------------------------------------------------
    # 6) Triton kernel: batched GEMM over (i, j) for each hidden channel
    # ------------------------------------------------------------------
    out_fp32 = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # stride information (in elements)
    s_lb, s_li, s_lk, s_ld = left.stride()
    s_rb, s_rj, s_rk, s_rd = right.stride()
    s_ob, s_oi, s_oj, s_od = out_fp32.stride()

    # tile sizes – chosen to fit comfortably in shared memory and exploit Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    batch_dim = B * hidden_dim

    grid = (num_blocks_m, num_blocks_n, batch_dim)

    tri_mul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_fp32,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=s_lb,
        stride_left_i=s_li,
        stride_left_k=s_lk,
        stride_left_d=s_ld,
        stride_right_b=s_rb,
        stride_right_j=s_rj,
        stride_right_k=s_rk,
        stride_right_d=s_rd,
        stride_out_b=s_ob,
        stride_out_i=s_oi,
        stride_out_j=s_oj,
        stride_out_d=s_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Final LayerNorm, gating and projection back to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
