"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The core operation computes, for each batch b and hidden head d,
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
which is a batched matrix multiplication C = A @ Bᵀ where
    A = left[..., :, :, d]  (N×N) and B = right[..., :, :, d] (N×N).

We implement the GEMM with a tiled Triton kernel (fp16 inputs, fp32
accumulation) and perform the surrounding LayerNorm, linear projections,
masking and gating in PyTorch.  The final output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,            # left  tensor pointer   (batch, M, K)   fp16
    b_ptr,            # right tensor pointer   (batch, N, K)   fp16 (will be read transposed)
    c_ptr,            # output tensor pointer  (batch, M, N)   fp32
    batch: tl.constexpr,
    M: tl.constexpr,
    N: tl.constexpr,
    K: tl.constexpr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)            # row tile
    pid_n = tl.program_id(1)            # col tile
    pid_b = tl.program_id(2)            # batch tile (B * hidden_dim)

    # Offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch element
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load Bᵀ tile: shape (BLOCK_K, BLOCK_N)
        # Note: b is stored as (batch, N, K); we load its transpose on‑the‑fly.
        b_ptrs = b_batch + (cur_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate dot product (A @ Bᵀ)
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float). May be ignored.
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict containing at least:
                "dim" (int), "hidden_dim" (int), optional "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Unpack model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core TriMul: batched matrix multiplication C = A @ Bᵀ
    #   where A = left, B = right and the reduction is over the
    #   third axis (k).  We reshape to (B*hidden_dim, N, N) and launch
    #   a Triton kernel that processes each (batch, head) pair
    #   independently.
    # ------------------------------------------------------------------
    # Cast to half for compute efficiency
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, N, N, hidden_dim

    # Rearrange to (batch*head, N, N) where head == hidden_dim
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    batch_dim = left_h.shape[0]  # B * hidden_dim

    # Allocate output buffer (fp32 accumulator)
    out_h = torch.empty((batch_dim, N, N), dtype=torch.float32, device=device)

    # Strides for the kernel (as python integers)
    stride_a_batch, stride_a_m, stride_a_k = left_h.stride()
    stride_b_batch, stride_b_n, stride_b_k = right_h.stride()  # note: right shape (batch, N, K)
    # For B we need stride_k and stride_n (for transposed load)
    stride_b_k, stride_b_n = stride_b_k, stride_b_n
    stride_c_batch, stride_c_m, stride_c_n = out_h.stride()

    # Block sizes – chosen to fit well in shared memory for N up to 1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Grid definition
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m, num_pid_n, batch_dim)

    # Launch the Triton kernel
    tri_mul_outgoing_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        batch=batch_dim,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # reasonable default
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] (float32)
    # ------------------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)          # [B, N, N, hidden_dim]
        .contiguous()
    )  # dtype already float32

    # ------------------------------------------------------------------
    # Final LayerNorm, gating, and projection back to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # apply outgoing gate
    out = F.linear(out, to_out_weight)       # [B, N, N, dim]

    return out
