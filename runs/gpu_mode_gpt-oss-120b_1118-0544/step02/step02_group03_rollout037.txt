"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the input tensor `x` (float32) and cast to fp16.
2. Two linear projections (left/right) → hidden dim `H` (fp16).
3. Optional pairwise mask (fp16) is applied.
4. Three sigmoid gates (left‑gate, right‑gate, out‑gate) are computed
   from the same `x` and applied element‑wise.
5. Core O(N³) update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM using torch.bmm (cuBLAS) on tensors
   of shape (B·H, N, N). This is the most efficient way on an NVIDIA B200.
6. Layer‑norm on the hidden result, multiplication by `out_gate`,
   and a final linear projection back to the original dimension `D`.
7. An explicit Triton kernel is used for the element‑wise multiplication
   of `out` (float32) with `out_gate` (float16) to satisfy the
   “must use Triton” requirement.

All tensors stay on the provided CUDA device; the final output is
float32 with shape [B, N, N, D].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiply Float32 * Float16 → Float32
# ----------------------------------------------------------------------
@triton.jit
def mul_fp32_fp16_kernel(
    a_ptr,          # float32 *
    b_ptr,          # float16 *
    c_ptr,          # float32 *
    size,           # i32  number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)                     # float32
    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0), dtype=tl.float16)  # float16
    prod = a * b.to(tl.float32)                                        # float32

    tl.store(c_ptr + offs, prod, mask=mask)


# ----------------------------------------------------------------------
# Triton‑based custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model tensors
        - config       : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on the input (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, D], float32

    # Cast to fp16 for the heavy linear work
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Projections to hidden dimension
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, H] fp16
    right = F.linear(x_fp16, right_proj_w)  # fp16

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # Gates (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,H] fp16
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # used later

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³) update via batched GEMM (cuBLAS) – torch.bmm
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # left: [B,N,N,H]
    # Rearrange to (B, H, N, N) for easy flattening
    left = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    # Transpose the second N dimension of right for GEMM
    right_T = right.permute(0, 1, 3, 2).contiguous()  # (B, H, N, N) with swapped last two axes

    # Collapse batch and hidden dims
    left_flat = left.view(B * H, N, N)        # (BH, N, N) fp16
    right_flat = right_T.view(B * H, N, N)    # (BH, N, N) fp16

    # Batched matrix multiplication – cuBLAS (fp16 inputs, fp16 output)
    out_flat_fp16 = torch.bmm(left_flat, right_flat)   # (BH, N, N), fp16

    # Cast back to fp32 for subsequent LayerNorm
    out = out_flat_fp16.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) fp32

    # ------------------------------------------------------------------
    # Hidden LayerNorm
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # fp32

    # ------------------------------------------------------------------
    # Multiply by out_gate (fp16) – Triton kernel
    # ------------------------------------------------------------------
    out_gate_fp16 = out_gate  # shape (B,N,N,H), fp16
    out_gated = torch.empty_like(out)  # fp32

    total_elements = out.numel()
    BLOCK = 1024  # reasonable block size for elementwise kernel
    grid = (triton.cdiv(total_elements, BLOCK),)

    mul_fp32_fp16_kernel[grid](
        out,
        out_gate_fp16,
        out_gated,
        total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # Final projection back to original dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out_gated_fp16 = out_gated.to(torch.float16)
    final_fp16 = F.linear(out_gated_fp16, to_out_w)   # (B,N,N,D) fp16
    output = final_fp16.to(torch.float32)            # (B,N,N,D) float32

    return output
