"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The forward pass follows the reference PyTorch implementation:
1. Layer‑norm the input tensor (dim‑wise).
2. Linear projections to a hidden dimension (left/right) and three
   gating projections (left_gate, right_gate, out_gate).  All heavy
   tensors are cast to FP16.
3. Optional binary mask is applied before gating.
4. The core O(N³·H) reduction
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   is performed by a tiled batched GEMM written in Triton.  The hidden
   dimension is merged into the batch axis to obtain a single 3‑D
   batched matmul.
5. The result is reshaped back, Layer‑normed over the hidden dimension,
   multiplied by the output gate and projected back to the original
   channel dimension.
6. Output is returned in the original dtype (usually float32).

Only the reduction step is Triton‑accelerated; the remaining ops use
high‑performance PyTorch kernels.
"""

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched matrix multiplication (C = A @ Bᵀ)
#   A: (batch, M, K)   fp16
#   B: (batch, K, N)   fp16   (interpreted as the transpose of the original
#                                 right tensor)
#   C: (batch, M, N)   fp16   (accumulated in fp32)
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,   # A: batch, M, K strides
    stride_bh, stride_bk, stride_bn,   # B: batch, K, N strides (B is K×N)
    stride_ch, stride_cm, stride_cn,   # C: batch, M, N strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)  # batch * hidden
    pid_m = tl.program_id(0)  # tile over M (i)
    pid_n = tl.program_id(1)  # tile over N (j)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Starting pointers for this batch
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # Pointers to the beginning of the tile (k dimension will be stepped)
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        # k‑dim offsets for this tile
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

        # advance to next k‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A and B must have the same shape (batch, N, N) and be contiguous fp16.
    Returns C with the same shape and dtype as A/B.
    """
    assert A.shape == B.shape
    batch, N, K = A.shape
    assert N == K, "TriMul expects square sequence dimension."

    # Choose block sizes – multiples of 16 are friendly to Tensor Cores
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128
    BLOCK_K = 64  # 64 is a good trade‑off for most GPUs

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    C = torch.empty_like(A)

    _batched_matmul_kernel[grid](
        A_ptr=A,
        B_ptr=B,
        C_ptr=C,
        M=N,
        N=N,
        K=N,
        stride_ah=A.stride(0),
        stride_am=A.stride(1),
        stride_ak=A.stride(2),
        stride_bh=B.stride(0),
        stride_bk=B.stride(2),   # K dimension (since we treat B as K×N)
        stride_bn=B.stride(1),   # N dimension
        stride_ch=C.stride(0),
        stride_cm=C.stride(1),
        stride_cn=C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8 if N >= 512 else 4,
    )
    return C

        (dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
    )
    # Cast to FP16 for the compute‑heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 2. Linear projections (left/right) and gates – all in FP16
    # ------------------------------------------------------------------- #
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left  = torch.nn.functional.linear(x, left_proj_w)   # (B, N, N, H)
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------- #
    # 3. Optional masking (applied before gating)
    # ------------------------------------------------------------------- #
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------- #
    # 4. Gates (sigmoid) and apply
    # ------------------------------------------------------------------- #
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Core O(N³·H) reduction via Triton
    #    Collapse the hidden dimension into the batch axis.
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape  # H == hidden_dim
    # From (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Perform batched matmul: out_flat[b*h,i,j] = Σ_k left[i,k] * right[j,k]
    out_flat = _batched_matmul_triton(left, right)   # (B*H, N, N), fp16

    # Restore original layout → (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 6. Output LayerNorm, output gate and final projection
    # ------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float16)
    )   # (B, N, N, dim) in FP16

    # Cast back to the original dtype (typically float32)
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)
    return out
