    out = F.linear(out, to_out_weight)

"""
Outgoing Triangle Multiplicative Update (TriMul) – forward only.

The core of the operator is the reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where `left` and `right` are projected‑and‑gated versions of the input.
The reduction is implemented as a tiled batched matrix multiplication in Triton:
for each batch `b` and hidden channel `d` we compute a matrix product
    (left[b,:,:,d]) @ (right[b,:,:,d].transpose(1,2))

All LayerNorms, linear projections and gating are performed with PyTorch
(very cheap compared to the O(N³) reduction).  The reduction itself is
fused in a single Triton kernel that works on FP16 inputs and accumulates
in FP32, matching the precision of torch.bmm with tensor cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def outgoing_trimul_kernel(
    left_ptr,          # float16 *
    right_ptr,         # float16 *
    out_ptr,           # float32 *
    B, N, H,           # runtime sizes
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_o0, stride_o1, stride_o2, stride_o3,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Tiled batched matrix multiplication for the TriMul reduction.

    left  : (B, N, N, H)  – indexed as (b, i, k, h)
    right : (B, N, N, H)  – indexed as (b, j, k, h)
    out   : (B, N, N, H)  – indexed as (b, i, j, h)
    """
    pid_bh = tl.program_id(0)      # combined batch * hidden index
    pid_i  = tl.program_id(1)      # i‑tile
    pid_j  = tl.program_id(2)      # j‑tile

    # Decode batch and hidden channel
    b = pid_bh // H
    h = pid_bh %  H

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # Offsets for the i and j dimensions
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_offset in range(0, N, BLOCK_K):
        k_max = tl.minimum(N - k_offset, BLOCK_K)
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_I, BLOCK_K)  -> (i, k)
        left_ptrs = (
            left_ptr
            + b * stride_l0
            + offs_i[:, None] * stride_l1
            + offs_k[None, :] * stride_l2
            + h * stride_l3
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right block transposed: shape (BLOCK_K, BLOCK_J)  -> (k, j)
        right_ptrs = (
            right_ptr
            + b * stride_r0
            + offs_k[:, None] * stride_r2   # k as first dimension
            + offs_j[None, :] * stride_r1   # j as second dimension
            + h * stride_r3
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # Matrix multiplication of the two blocks
        acc += tl.dot(left, right)  # (BLOCK_I, BLOCK_J)

    # Store the result tile into the output tensor (FP32)
    out_ptrs = (
        out_ptr
        + b * stride_o0
        + offs_i[:, None] * stride_o1
        + offs_j[None, :] * stride_o2
        + h * stride_o3
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float), may be None
            - weights: dict of model weight tensors
            - config: dict containing at least "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) Load model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate      # [B, N, N, hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Cast to FP16 for the heavy reduction
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    B, N, _, H = left_fp16.shape     # B, N, N, hidden_dim

    # ------------------------------------------------------------------
    # 7) Triton kernel: outgoing triangle multiplication
    # ------------------------------------------------------------------
    out_fp32 = torch.empty((B, N, N, H), dtype=torch.float32, device=left_fp16.device)

    # Tile sizes – these are tuned for the B200 (Ampere) architecture.
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    # Strides for the three tensors (int64)
    stride_l0, stride_l1, stride_l2, stride_l3 = left_fp16.stride()
    stride_r0, stride_r1, stride_r2, stride_r3 = right_fp16.stride()
    stride_o0, stride_o1, stride_o2, stride_o3 = out_fp32.stride()

    grid = (
        B * H,                                  # programs over batch*hidden
        (N + BLOCK_I - 1) // BLOCK_I,           # i‑tiles
        (N + BLOCK_J - 1) // BLOCK_J,           # j‑tiles
    )

    outgoing_trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B, N, H,
        stride_l0, stride_l1, stride_l2, stride_l3,
        stride_r0, stride_r1, stride_r2, stride_r3,
        stride_o0, stride_o1, stride_o2, stride_o3,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate               # apply the output gate
    out = F.linear(out, to_out_weight) # [B, N, N, dim]

    return out
