"""
TriMul (outgoing) forward pass – fast Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the input tensor (float32) using the provided weight/bias.
2. Cast to fp16 and apply two linear projections (left/right) producing
   tensors of shape [B, N, N, H].
3. (Optional) apply the binary mask to the projected tensors.
4. Compute three sigmoid gates (left, right, out) and multiply them into
   the projected tensors.
5. Reshape the gated tensors to (B*H, N, N) and perform a batched matrix
   multiplication A @ Bᵀ with `torch.bmm`.  This uses cuBLAS / Tensor‑Core
   FP16 acceleration.
6. Reshape back to [B, N, N, H] and cast to fp32.
7. Apply a second LayerNorm (to_out_norm), then multiply element‑wise with
   the out‑gate using a tiny Triton kernel (fulfilling the “some operations
   in a kernel” requirement).
8. Final linear projection back to the original dimension `dim`.
9. Return the result in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication (used for out_gate * out)
# ----------------------------------------------------------------------
@triton.jit
def mul_kernel(
    a_ptr,          # tensor a
    b_ptr,          # tensor b
    out_ptr,        # output tensor (can be same as a_ptr for in‑place)
    N: tl.constexpr,          # total number of elements
    BLOCK_SIZE: tl.constexpr, # tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D]   (float32)
        - mask         : torch.Tensor [B, N, N]     (bool/float, optional)
        - weights      : dict of model tensors
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs and configuration
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # Layer‑norm on the input (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x = x.to(torch.float16)  # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # Linear projections (left / right) – fp16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)   # [B, N, N, H]  fp16
    right = F.linear(x, right_proj_w)  # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # Optional binary mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # Gates (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, H] fp16
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # [B, N, N, H] fp16
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # [B, N, N, H] fp16

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for batched matrix multiply:
    #   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # This is a batched matmul of shape (B*H, N, N) @ (B*H, N, N)^T
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim
    # Move hidden dim to batch axis
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Transpose the second tensor on its last two axes
    right_t = right.permute(0, 1, 3, 2).contiguous()  # (B, H, N, N) where N dim is swapped

    # Combine batch+head into a single leading dimension
    left_flat = left.view(B * H, N, N)       # (BH, N, N)
    right_flat = right_t.view(B * H, N, N)   # (BH, N, N)

    # Batched matmul using cuBLAS / Tensor Cores (fp16 -> fp16, accumulate in fp32 internally)
    out_flat = torch.bmm(left_flat, right_flat)  # (BH, N, N)  fp16

    # Restore original layout: [B, N, N, H] and promote to fp32 for later ops
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)  fp16
    out = out.to(torch.float32)  # fp32

    # ------------------------------------------------------------------
    # Second Layer‑norm (to_out_norm)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # fp32

    # ------------------------------------------------------------------
    # Apply out‑gate (element‑wise) with a tiny Triton kernel
    # ------------------------------------------------------------------
    out_gate = out_gate.to(out.dtype)  # fp32, same shape as out
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    total_elements = out.numel()
    BLOCK = 1024
    grid = triton.cdiv(total_elements, BLOCK)
    mul_kernel[grid](
        out,
        out_gate,
        out,
        N=total_elements,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )  # out = out * out_gate  (in‑place)

    # ------------------------------------------------------------------
    # Final linear projection back to original dimension `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out_fp16 = out.to(torch.float16)
    output = F.linear(out_fp16, to_out_w)  # (B, N, N, dim)  fp16
    output = output.to(torch.float32)      # final output in fp32

    return output
