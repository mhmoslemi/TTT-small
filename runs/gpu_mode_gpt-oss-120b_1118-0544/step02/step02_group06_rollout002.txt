"""
Outgoing Triangle Multiplicative Update (TriMul) implementation.

The core of the operation is a batched matrix multiplication:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
which can be expressed as a GEMM per hidden dimension slice:
    out_h = left_h @ right_hᵀ   (left_h/right_h : [B*H, N, N]).

We fuse the expensive GEMM into a custom Triton kernel that works on
float‑16 inputs and accumulates in float‑32.  The rest of the pipeline
(LayerNorm, linear projections, sigmoids, gating) is performed with
standard PyTorch ops.  The final result is returned as float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,      # left  tensor  (batch, M, K)  fp16
    b_ptr,      # right tensor  (batch, N, K)  fp16 (read as K×N)
    c_ptr,      # output tensor (batch, M, N)  fp32
    batch: tl.constexpr,
    M: tl.constexpr,
    N: tl.constexpr,
    K: tl.constexpr,
    stride_ah, stride_am, stride_ak,   # strides for a
    stride_bh, stride_bk, stride_bn,   # strides for b (note: bk, bn correspond to K and N after transpose)
    stride_ch, stride_cm, stride_cn,   # strides for c
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col
    pid_b = tl.program_id(2)  # batch index

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch
    a_batch = a_ptr + pid_b * stride_ah
    b_batch = b_ptr + pid_b * stride_bh
    c_batch = c_ptr + pid_b * stride_ch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for cur_k in range(0, K, BLOCK_K):
        cur_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M × K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile (K × N) – we read B as its transpose by swapping strides
        b_ptrs = b_batch + (cur_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate the partial product
        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (dim -> dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul via fused batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Convert to fp16 for Tensor‑core acceleration
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Reshape to (B * hidden_dim, N, N) where each slice is an independent GEMM
    # Shape after permute: [B, hidden_dim, N, N]
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_h.shape
    batch = B * H   # total number of GEMMs

    # Combine batch dimension for the kernel
    left_h = left_h.view(batch, N, N)
    right_h = right_h.view(batch, N, N)

    # Output buffer (fp32)
    out_h = torch.empty((batch, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64   # tuned for fp16 Tensor‑cores

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    # Strides for the three tensors
    stride_ah = left_h.stride(0)
    stride_am = left_h.stride(1)
    stride_ak = left_h.stride(2)

    stride_bh = right_h.stride(0)
    stride_bk = right_h.stride(2)  # K dimension stride (after transpose)
    stride_bn = right_h.stride(1)  # N dimension stride (after transpose)

    stride_ch = out_h.stride(0)
    stride_cm = out_h.stride(1)
    stride_cn = out_h.stride(2)

    # Launch the kernel
    tri_mul_outgoing_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        batch=batch,
        M=N,
        N=N,
        K=N,
        stride_ah=stride_ah,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bh=stride_bh,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_ch=stride_ch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Reshape back to [B, N, N, hidden_dim] (float32)
    # ------------------------------------------------------------------
    out = (
        out_h.view(B, H, N, N)   # [B, hidden_dim, N, N]
        .permute(0, 2, 3, 1)    # [B, N, N, hidden_dim]
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 7) Final LayerNorm, gating, and linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                       # apply outgoing gate
    out = F.linear(out, to_out_weight)         # [B, N, N, dim]

    return out
```
