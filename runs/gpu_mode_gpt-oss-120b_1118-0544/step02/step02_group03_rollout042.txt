def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # 1. LayerNorm on input (float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # 2. Fuse projection weights
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    # Concatenate
    combined_weight = torch.cat([left_proj_weight, right_proj_weight, left_gate_weight, right_gate_weight], dim=0)  # shape (4*hidden_dim, dim)

    # Cast to fp16 for speed
    x_fp16 = x_norm.to(torch.float16)
    combined_weight_fp16 = combined_weight.to(torch.float16)

    # Flatten input
    B, N, _, _ = input_tensor.shape
    x_flat = x_fp16.view(-1, dim)  # (B*N*N, dim)
    proj = torch.nn.functional.linear(x_flat, combined_weight_fp16)  # (B*N*N, 4*hidden_dim)

    proj = proj.view(B, N, N, 4*hidden_dim)

    # Split
    left = proj[..., :hidden_dim]  # (B,N,N,hidden)
    right = proj[..., hidden_dim:2*hidden_dim]
    left_gate = proj[..., 2*hidden_dim:3*hidden_dim]
    right_gate = proj[..., 3*hidden_dim:]

    # Gate activations
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # Apply mask if needed
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Core O(N^3) update using batched matmul (torch.bmm)
    # Permute to (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Transpose right for matmul
    right_t = right_perm.permute(0, 1, 3, 2).contiguous()

    # Reshape to batch dimension
    batch_hidden = B * hidden_dim
    left_flat = left_perm.reshape(batch_hidden, N, N)
    right_flat = right_t.reshape(batch_hidden, N, N)

    # Perform batched matmul (FP16)
    out_flat = torch.bmm(left_flat, right_flat)  # result fp16

    # Reshape back to (B,N,N,hidden) and cast to float32
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # 3. to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # 4. out_gate
    out_gate_weight = weights["out_gate.weight"]
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))  # shape (B,N,N,hidden)

    # Apply out_gate
    out = out * out_gate

    # 5. Final projection
    to_out_weight = weights["to_out.weight"]
    # Use fp16 for speed
    out_fp16 = out.to(torch.float16)
    to_out_weight_fp16 = to_out_weight.to(torch.float16)
    output = torch.nn.functional.linear(out_fp16, to_out_weight_fp16).to(torch.float32)

    return output

"""
TriMul outgoing update (AlphaFold3) – Triton‑fused core.

Algorithm
---------
1) Layer‑norm the 4‑D input.
2) Fuse the four linear projections (left, right, left_gate, right_gate) into a
   single dense multiply `x @ W_fused`.  This reads the input only once.
3) Apply sigmoid gates and (optionally) a binary mask using a tiny Triton
   element‑wise kernel (in‑place multiplication).
4) Core O(N³) update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (torch.bmm) on fp16 tensors.
5) Layer‑norm the hidden result, multiply by the out‑gate and project back to
   the original dimension.
All heavy arithmetic is performed in fp16 (tensor‑core friendly) and the
final output is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: in‑place elementwise multiplication (a = a * b)
# ----------------------------------------------------------------------
@triton.jit
def mul_inplace_kernel(
    a_ptr, b_ptr,               # tensor pointers (contiguous)
    N,                          # total number of elements
    BLOCK: tl.constexpr,        # compile‑time block size
):
    offs = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(a_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D]   (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model tensors (float32)
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, D]  float32

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 2. Fused projection: left, right, left_gate, right_gate
    # ------------------------------------------------------------------
    # gather and concatenate weight matrices (out_features, in_features)
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    fused_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w], dim=0
    )  # (4*H, D)

    # cast inputs & weights to fp16 for tensor‑core speed
    x_fp16 = x_norm.to(torch.float16)
    fused_w_fp16 = fused_w.to(torch.float16)

    # flatten the (B,N,N) dimensions and apply a single linear
    x_flat = x_fp16.view(-1, dim)                         # (B*N*N, D)
    proj_flat = F.linear(x_flat, fused_w_fp16)           # (B*N*N, 4*H)

    # reshape back to 4‑D
    proj = proj_flat.view(B, N, N, 4 * hidden_dim)       # (B, N, N, 4H)

    # split the concatenated projections
    left = proj[..., :hidden_dim]                        # (B,N,N,H)
    right = proj[..., hidden_dim:2 * hidden_dim]
    left_gate = proj[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate = proj[..., 3 * hidden_dim:]

    # sigmoid gates (still fp16)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Optional mask application (in‑place Triton kernel)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # broadcast mask over the hidden dimension and make it contiguous
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()
        # left = left * mask_fp16  (in‑place)
        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        mul_inplace_kernel[grid](
            left, mask_fp16, total, BLOCK=BLOCK, num_warps=4
        )
        # right = right * mask_fp16 (in‑place)
        total = right.numel()
        grid = (triton.cdiv(total, BLOCK),)
        mul_inplace_kernel[grid](
            right, mask_fp16, total, BLOCK=BLOCK, num_warps=4
        )

    # ------------------------------------------------------------------
    # 4. Core O(N³) update: batched GEMM (fp16 -> fp32)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) and compute left @ rightᵀ for each slice
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    right_t = right_perm.permute(0, 1, 3, 2).contiguous()  # (B, H, N, N) = rightᵀ

    batch_hidden = B * hidden_dim
    left_batch = left_perm.view(batch_hidden, N, N)      # (B*H, N, N) fp16
    right_batch = right_t.view(batch_hidden, N, N)       # (B*H, N, N) fp16

    # batched matrix multiply (uses cuBLAS/Tensor Cores)
    out_batch_fp16 = torch.bmm(left_batch, right_batch)  # (B*H, N, N) fp16

    # reshape back to (B, N, N, H) and cast to float32 for the rest
    out = (
        out_batch_fp16.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )                                                   # (B,N,N,H) fp32

    # ------------------------------------------------------------------
    # 5. Hidden‑dim layer‑norm
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )                                                   # fp32

    # ------------------------------------------------------------------
    # 6. Out‑gate (float32)
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"]
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))   # (B,N,N,H) fp32
    out = out * out_gate                                           # fp32

    # ------------------------------------------------------------------
    # 7. Final projection back to original dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]
    out_fp16 = out.to(torch.float16)
    to_out_w_fp16 = to_out_w.to(torch.float16)
    output = F.linear(out_fp16, to_out_w_fp16).to(torch.float32)   # (B,N,N,D)

    return output
