@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets
    offset_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offset_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for bounds
    mask_m = offset_m < M
    mask_n = offset_n < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offset_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for K
        mask_k = offset_k < K

        # compute pointer offsets for left tile (M x K)
        left_ptrs = left_ptr + batch_idx * stride_left_batch + offset_m[:, None] * stride_left_m + offset_k[None, :] * stride_left_k
        # compute pointer offsets for right tile (K x N) because we need transposed
        right_ptrs = right_ptr + batch_idx * stride_right_batch + offset_k[:, None] * stride_right_k + offset_n[None, :] * stride_right_n

        # load data
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + batch_idx * stride_out_batch + offset_m[:, None] * stride_out_m + offset_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

```
@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute coordinate offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile
        left_ptrs = left_ptr + batch_idx * stride_left_batch \
                    + offs_m[:, None] * stride_left_m \
                    + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right tile (transposed)
        right_ptrs = right_ptr + batch_idx * stride_right_batch \
                    + offs_k[:, None] * stride_right_k \
                    + offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_ptr + batch_idx * stride_out_batch \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the 4‑D input.
2. Linear projections → left / right (hidden dim) and three gating
   projections (left‑gate, right‑gate, out‑gate), all in fp32.
3. Apply mask (if present) and element‑wise gating.
4. Cast left/right to fp16, reshape to (B*H, N, N) and compute the
   O(N³) reduction
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   as a batched matrix multiplication using a custom Triton kernel.
5. Reshape back to (B,N,N,H), apply a second LayerNorm,
   multiply with out‑gate (element‑wise Triton kernel) and a final
   linear projection back to the original channel dimension.
All heavy arithmetic (the N³ reduction) runs on the GPU with
tensor‑core fused fp16→fp32 accumulation, while the rest of the
pipeline stays in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------
# Triton kernel: batched matmul for the TriMul reduction (left @ rightᵀ)
# --------------------------------------------------------------------
@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,                           # matrix sizes (all equal to seq_len)
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = left @ rightᵀ for many independent batches."""
    batch = tl.program_id(0)          # batch index (B*hidden)
    pid_m = tl.program_id(1)          # tile row index
    pid_n = tl.program_id(2)          # tile col index

    # Offsets in the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile: (M, K)
        left_ptrs = left_ptr + batch * stride_left_batch \
                    + offs_m[:, None] * stride_left_m \
                    + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile (transposed on‑the‑fly): (K, N)
        right_ptrs = right_ptr + batch * stride_right_batch \
                     + offs_k[:, None] * stride_right_k \
                     + offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        # Fused multiply‑accumulate (fp16×fp16 → fp32)
        acc += tl.dot(left, right)

    # Store the result
    out_ptrs = out_ptr + batch * stride_out_batch \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------------
# Tiny element‑wise kernel for out‑gate multiplication
# --------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, total_len: tl.constexpr, BLOCK: tl.constexpr):
    """c = a * b  (flattened 1‑D tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_len
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)

# --------------------------------------------------------------------
# Entry point required by the test harness
# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – optional
            - weights: dict of model weights (all tensors on the same CUDA device)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ----------------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------------
    # Unpack weights
    # ----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension
    # ----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ----------------------------------------------------------------
    # 3️⃣ Optional masking (broadcast on hidden)
    # ----------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # ----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # 5️⃣ Cast to fp16 and reshape for the batched GEMM
    # ----------------------------------------------------------------
    left_h = left.half()
    right_h = right.half()

    B, N, _, _ = left_h.shape  # batch, seq_len, seq_len, hidden
    # Make hidden dimension part of the batch for the kernel
    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    left_h = left_h.view(B * hidden_dim, N, N)   # [B*H, N, N]
    right_h = right_h.view(B * hidden_dim, N, N)

    # ----------------------------------------------------------------
    # 6️⃣ Triton batched matrix multiplication: out = left @ rightᵀ
    # ----------------------------------------------------------------
    out_h = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=left_h.device)

    # Strides (in elements)
    stride_l_batch = left_h.stride(0)
    stride_l_m = left_h.stride(1)
    stride_l_k = left_h.stride(2)

    stride_r_batch = right_h.stride(0)
    stride_r_n = right_h.stride(1)   # row stride (original i)
    stride_r_k = right_h.stride(2)   # column stride (original k)

    stride_o_batch = out_h.stride(0)
    stride_o_m = out_h.stride(1)
    stride_o_n = out_h.stride(2)

    # Block sizes – empirically good for B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (B * hidden_dim, num_blocks_m, num_blocks_n)

    trmul_kernel[grid](
        left_h, right_h, out_h,
        N, N, N,
        stride_l_batch, stride_l_m, stride_l_k,
        stride_r_batch, stride_r_n, stride_r_k,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, hidden_dim]
    # ----------------------------------------------------------------
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ----------------------------------------------------------------
    # 8️⃣ Second LayerNorm over hidden dimension
    # ----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B, N, N, H]

    # ----------------------------------------------------------------
    # 9️⃣ Apply out‑gate (element‑wise) using a tiny Triton kernel
    # ----------------------------------------------------------------
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)
    total = out_flat.numel()
    BLOCK = 1024
    grid_gate = ((total + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid_gate](
        out_flat, gate_flat, out_flat, total, BLOCK=BLOCK,
    )

    # ----------------------------------------------------------------
    # 🔟 Final linear projection back to original dimension
    # ----------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
