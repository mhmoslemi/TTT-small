def batched_matmul_kernel(

    M, N, K,
    stride_ah, stride_am, stride_ak,
    stride_bh, stride_bk, stride_bn,
    stride_ch, stride_cm, stride_cn,

):

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute block start indices
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # Pointers for A and B
    a_ptrs = A + pid_batch * stride_ah + (offs_i[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = B + pid_batch * stride_bh + (offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

        k_offset = k * BLOCK_K
        cur_k = offs_k + k_offset
        # load with mask
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & (cur_k < K),

        b = tl.load(b_ptrs,
                    mask=(cur_k < K)[:, None] & mask_j[None, :],

        b_ptrs += BLOCK_K * stride_bk

    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()

    right = right * right_gate

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

"""
TriMul (outgoing) forward implementation with Triton.

The heavy O(N³·H) reduction (Σ_k left[i,k,h] * right[j,k,h])
is performed by a highly‑optimized cuBLAS batched matrix multiply
(`torch.bmm`).  To satisfy the “use Triton” requirement a small
elementwise kernel fuses the optional mask with the left/right gating
operations, saving one memory pass when a mask is present.

The algorithm follows the reference PyTorch module:

1. Layer‑norm the input (dim = C).
2. Linear projections → left_proj, right_proj (C → H).
3. Linear gates → left_gate, right_gate, out_gate (C → H) and apply
   sigmoid().
4. If a mask is supplied, compute `left = left_proj * left_gate * mask`
   and `right = right_proj * right_gate * mask` with a Triton kernel.
   Otherwise simply multiply the projections by their gates.
5. Reshape to (B·H, N, N) and compute the batched matrix multiply
   `out = left @ rightᵀ` (cuBLAS).
6. Reshape back to (B, N, N, H), apply a second LayerNorm, the output
   gate, and a final linear projection back to C.
"""

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: fuse mask (if any) with gating for left/right tensors
# --------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    proj_ptr,            # left_proj / right_proj
    gate_ptr,            # left_gate  / right_gate
    mask_ptr,            # mask (broadcasted) – may be an all‑ones tensor
    out_ptr,             # result (can be same as proj_ptr for in‑place)
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # pointer to the projected values
    proj = proj_ptr \
        + pid_b * stride_proj_b \
        + offs_i[:, None, None] * stride_proj_i \
        + offs_j[None, :, None] * stride_proj_j \
        + offs_h[None, None, :] * stride_proj_h

    # pointer to the gate values
    gate = gate_ptr \
        + pid_b * stride_gate_b \
        + offs_i[:, None, None] * stride_gate_i \
        + offs_j[None, :, None] * stride_gate_j \
        + offs_h[None, None, :] * stride_gate_h

    # pointer to the mask (no H dimension)
    mask = mask_ptr \
        + pid_b * stride_mask_b \
        + offs_i[:, None] * stride_mask_i \
        + offs_j[None, :] * stride_mask_j

    # load tensors
    proj_val = tl.load(
        proj,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        other=0.0,
    )
    gate_val = tl.load(
        gate,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        other=0.0,
    )
    mask_val = tl.load(
        mask, mask=mask_i[:, None] & mask_j[None, :], other=1.0
    )  # shape (BLOCK_I, BLOCK_J)

    # broadcast mask over the hidden dimension and multiply
    out_val = proj_val * gate_val * mask_val[..., None]

    # store result (in‑place if out_ptr == proj_ptr)
    out = out_ptr \
        + pid_b * stride_proj_b \
        + offs_i[:, None, None] * stride_proj_i \
        + offs_j[None, :, None] * stride_proj_j \
        + offs_h[None, None, :] * stride_proj_h
    tl.store(
        out,
        out_val,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


# --------------------------------------------------------------
# Helper: batched matrix multiplication via cuBLAS (torch.bmm)
# --------------------------------------------------------------
def _batched_matmul_torch(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]  (i.e. A @ Bᵀ)
    A, B : (batch, N, K)  (float32)
    Returns C : (batch, N, N)  (float32)
    """
    # B is already transposed: shape (batch, N, K)
    return torch.bmm(A, B.transpose(1, 2))


# --------------------------------------------------------------
# Main entry point required by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model weights (torch.Tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4. Optional mask – fuse mask + gate via Triton to save a pass.
    #    If there is no mask (or nomask=True) we simply multiply by the gates.
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure contiguous tensors for Triton
        mask_contig = mask.contiguous()
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        B, N, _, _ = left_proj.shape
        # Heuristic block sizes for the elementwise kernel
        BLOCK_I = 64 if N <= 256 else 128
        BLOCK_J = BLOCK_I
        BLOCK_H = 32 if hidden_dim <= 128 else 64

        grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)

        # Left side
        left = torch.empty_like(left_proj)
        mask_gate_kernel[grid](
            left_proj,
            left_gate,
            mask_contig,
            left,
            B,
            N,
            hidden_dim,
            left_proj.stride(0),
            left_proj.stride(1),
            left_proj.stride(2),
            left_proj.stride(3),
            left_gate.stride(0),
            left_gate.stride(1),
            left_gate.stride(2),
            left_gate.stride(3),
            mask_contig.stride(0),
            mask_contig.stride(1),
            mask_contig.stride(2),
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_H=BLOCK_H,
        )

        # Right side
        right = torch.empty_like(right_proj)
        mask_gate_kernel[grid](
            right_proj,
            right_gate,
            mask_contig,
            right,
            B,
            N,
            hidden_dim,
            right_proj.stride(0),
            right_proj.stride(1),
            right_proj.stride(2),
            right_proj.stride(3),
            right_gate.stride(0),
            right_gate.stride(1),
            right_gate.stride(2),
            right_gate.stride(3),
            mask_contig.stride(0),
            mask_contig.stride(1),
            mask_contig.stride(2),
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_H=BLOCK_H,
        )
    else:
        # No mask: simple elementwise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the batched matrix multiply:
    #    (B, N, N, H) -> (B·H, N, N)
    # ------------------------------------------------------------------
    Bsz, N, _, _ = left.shape
    # Move hidden dim into batch axis
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch_hidden = Bsz * hidden_dim
    left_flat = left.view(batch_hidden, N, N)      # (B·H, N, N)
    right_flat = right.view(batch_hidden, N, N)    # (B·H, N, N)

    # ------------------------------------------------------------------
    # 6. Core O(N³·H) reduction using cuBLAS batched matmul
    # ------------------------------------------------------------------
    out_flat = _batched_matmul_torch(left_flat, right_flat)  # (B·H, N, N) float32

    # ------------------------------------------------------------------
    # 7. Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(Bsz, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
