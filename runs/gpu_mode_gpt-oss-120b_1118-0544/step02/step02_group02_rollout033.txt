```
@triton.jit
def triton_matmul_outgoing_kernel(
    a_ptr,    # *float16
    b_ptr,    # *float16 (transposed)
    c_ptr,    # *float32
    M, N, K,
    stride_aa, stride_ai, stride_ak, # A: batch, i, k
    stride_bb, stride_bk, stride_bn, # B: batch, k, n
    stride_cc, stride_ci, stride_cn, # C: batch, i, n
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)  # batch * block_m
    pid1 = tl.program_id(1)  # block_n

    # compute batch index and block_i index
    batch = pid0 // tl.cdiv(M, BLOCK_M)  # number of blocks in M dimension per batch
    block_i = pid0 % tl.cdiv(M, BLOCK_M)

    # compute block_j index
    block_j = pid1

    # compute offsets
    i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bound i,j
    mask_i = i < M
    mask_j = j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offset
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # pointers to A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch * stride_aa + i[:, None] * stride_ai + k_offset[None, :] * stride_ak
        # load
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # pointers to B block: B is shape (batch, K, N) i.e., rows = k, cols = j.
        b_ptrs = b_ptr + batch * stride_bb + k_offset[:, None] * stride_bk + j[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_ptr + batch * stride_cc + i[:, None] * stride_ci + j[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def triton_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_aa, stride_ai, stride_ak,
    stride_bb, stride_bk, stride_bn,
    stride_cc, stride_ci, stride_cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)           # over M_total
    pid_i = tl.program_id(1)           # block row index within each matrix
    pid_j = tl.program_id(2)           # block col index

    # row and col start
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks
    mask_i = i < N
    mask_j = j < N

    # base pointers for this batch
    a_base = a_ptr + batch * stride_aa
    b_base = b_ptr + batch * stride_bb
    c_base = c_ptr + batch * stride_cc

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < K

        # Load A[i, k] block
        a_ptrs = a_base + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B[k, j] block
        b_ptrs = b_base + k[:, None] * stride_bk + j[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_base + i[:, None] * stride_ci + j[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
Outgoing Triangle Multiplicative Update (TriMul) – fast forward pass.

Algorithm
---------
1. Layer‑Norm the input tensor.
2. Linear projections (left, right) and three gating projections (left_gate,
   right_gate, out_gate).  Apply an optional mask and sigmoid gating.
3. Cast the gated left/right tensors to FP16.
4. Reshape to (B*hidden, N, N) and transpose the right tensor.
5. Compute the batched matrix multiplication
        out[m,i,j] = Σ_k left[m,i,k] * right_T[m,k,j]
   with a custom Triton kernel (FP16 inputs, FP32 accumulation).
6. Reshape back to [B, N, N, hidden].
7. Final LayerNorm, output gating and linear projection back to the original
   feature dimension.

The heavy O(N³) reduction is performed entirely in the Triton kernel, which
operates on 64×64×64 tiles and uses 8 warps per program.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # dimensions (M = B*hidden, N = seq_len, K = seq_len)
    stride_aa, stride_ai, stride_ak,   # A strides: batch, i, k
    stride_bb, stride_bk, stride_bn,   # B strides: batch, k, n
    stride_cc, stride_ci, stride_cn,   # C strides: batch, i, n
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs: (batch, block_i, block_j)
    batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # coordinate ranges for this program
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # base pointers for the current batch (M‑dimension)
    a_batch = a_ptr + batch * stride_aa
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cc

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # tiled loop over the reduction dimension K
    for k in range(0, K, BLOCK_K):
        k_offset = k + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A block  [i, k]
        a_offs = a_batch + i[:, None] * stride_ai + k_offset[None, :] * stride_ak
        a = tl.load(a_offs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16 loads

        # Load B block  [k, j]  (B already transposed)
        b_offs = b_batch + k_offset[:, None] * stride_bk + j[None, :] * stride_bn
        b = tl.load(b_offs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)                     # fp16 loads

        # Accumulate
        acc += tl.dot(a, b)                     # fp16·fp16 → fp32

    # Write out C block
    c_offs = c_batch + i[:, None] * stride_ci + j[None, :] * stride_cn
    tl.store(c_offs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model weights
            - config: dict with at least "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) Model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)    # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # used later

    # ------------------------------------------------------------------
    # 5) Apply gates and cast to FP16 for the heavy reduction
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 6) Reshape to (B*hidden, N, N) and prepare transposed right tensor
    # ------------------------------------------------------------------
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # FP16
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N) # FP16
    right_T = right_flat.permute(0, 2, 1).contiguous()                             # FP16, (M, K, N)

    # ------------------------------------------------------------------
    # 7) Triton batched GEMM: out = left @ right_T
    # ------------------------------------------------------------------
    M_total = B * hidden_dim
    out_fp32 = torch.empty((M_total, N, N), dtype=torch.float32, device=left.device)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides (in elements)
    stride_aa = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bb = right_T.stride(0)
    stride_bk = right_T.stride(1)
    stride_bn = right_T.stride(2)

    stride_cc = out_fp32.stride(0)
    stride_ci = out_fp32.stride(1)
    stride_cn = out_fp32.stride(2)

    grid = (
        M_total,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_matmul_kernel[grid](
        left_flat.data_ptr(),
        right_T.data_ptr(),
        out_fp32.data_ptr(),
        M_total, N, N,
        stride_aa, stride_ai, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cc, stride_ci, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # FP32

    # ------------------------------------------------------------------
    # 9) Final LayerNorm, gating and projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # apply output gate
    out = F.linear(out, to_out_weight)       # [B, N, N, dim]

    return out
