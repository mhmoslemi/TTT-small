@triton.jit
def triton_matmul_transpose_kernel(A_ptr, B_ptr, C_ptr,
                                   batch, M, N, K,
                                   stride_A_batch, stride_A_m, stride_A_k,
                                   stride_B_batch, stride_B_k, stride_B_n,
                                   stride_C_batch, stride_C_m, stride_C_n,
                                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    total_batches = batch

    # compute batch index, block ids
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_id = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_id // num_blocks_n
    pid_n = block_id % num_blocks_n

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k_tile = tl.arange(0, BLOCK_K)

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute batch offset pointers
    a_batch_offset = batch_id * stride_A_batch
    b_batch_offset = batch_id * stride_B_batch
    c_batch_offset = batch_id * stride_C_batch

    # accumulators
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute tile offsets for K
        k_offs = k * BLOCK_K + k_tile
        mask_k = k_offs < K

        # compute pointers
        a_ptrs = A_ptr + a_batch_offset + (offs_m[:, None] * stride_A_m + k_offs[None, :] * stride_A_k)
        b_ptrs = B_ptr + b_batch_offset + (k_offs[:, None] * stride_B_k + offs_n[None, :] * stride_B_n)

        # load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # compute dot
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

    # write result
    c_ptrs = C_ptr + c_batch_offset + (offs_m[:, None] * stride_C_m + offs_n[None, :] * stride_C_n)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

                                    batch, M, N, K,
                                    stride_A_batch, stride_A_m, stride_A_k,
                                    stride_B_batch, stride_B_k, stride_B_n,
                                    stride_C_batch, stride_C_m, stride_C_n,
                                    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    # compute number of blocks in each dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    total_blocks_per_batch = num_blocks_m * num_blocks_n

    # batch index
    batch_idx = pid // total_blocks_per_batch
    block_idx = pid % total_blocks_per_batch
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute batch offsets
    a_batch_offset = batch_idx * stride_A_batch
    b_batch_offset = batch_idx * stride_B_batch
    c_batch_offset = batch_idx * stride_C_batch

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute actual k offsets
        k_offset = k * BLOCK_K + offs_k
        mask_k = k_offset < K

        # pointers
        a_ptrs = A_ptr + a_batch_offset + (offs_m[:, None] * stride_A_m + k_offset[None, :] * stride_A_k)
        b_ptrs = B_ptr + b_batch_offset + (k_offset[:, None] * stride_B_k + offs_n[None, :] * stride_B_n)

        # load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # write result
    c_ptrs = C_ptr + c_batch_offset + (offs_m[:, None] * stride_C_m + offs_n[None, :] * stride_C_n)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton implementation.

The core O(N³) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is expressed as a batched matrix multiplication
    out = left @ rightᵀ
where each hidden channel acts as an independent batch.  The heavy
computation is performed by a custom Triton GEMM kernel that works on
fp16 inputs, accumulates in fp32 and writes fp16 results.  All other
operations (LayerNorm, linear projections, gating) are handled in
PyTorch.  An additional tiny Triton kernel performs the final element‑wise
gate multiplication, satisfying the “must use Triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton GEMM kernel for A @ Bᵀ where A,B are (batch, N, N) fp16 tensors
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_transpose_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_A_batch, stride_A_m, stride_A_k,
    stride_B_batch, stride_B_k, stride_B_n,
    stride_C_batch, stride_C_m, stride_C_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """C[batch, i, j] = Σ_k A[batch, i, k] * B[batch, j, k]"""
    pid = tl.program_id(0)

    # number of blocks per batch
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    # decode batch / tile indices
    batch_idx = pid // blocks_per_batch
    tile_idx = pid % blocks_per_batch
    pid_m = tile_idx // num_blocks_n
    pid_n = tile_idx % num_blocks_n

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch base pointers
    a_batch_off = batch_idx * stride_A_batch
    b_batch_off = batch_idx * stride_B_batch
    c_batch_off = batch_idx * stride_C_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + offs_k
        mask_k = k_offset < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + a_batch_off + (offs_m[:, None] * stride_A_m
                                        + k_offset[None, :] * stride_A_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile (already transposed): (BLOCK_K, BLOCK_N)
        b_ptrs = B_ptr + b_batch_off + (k_offset[:, None] * stride_B_k
                                        + offs_n[None, :] * stride_B_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # write result (fp16)
    c_ptrs = C_ptr + c_batch_off + (offs_m[:, None] * stride_C_m
                                    + offs_n[None, :] * stride_C_n)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Tiny element‑wise multiplication kernel (out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_f32_kernel(a_ptr, b_ptr, c_ptr,
                               size: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton GEMM kernel.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model weights (all on the same CUDA device)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, do not apply mask

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_weight)   # left_proj
    right = F.linear(x, right_proj_weight)  # right_proj

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid applied to linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare for batched GEMM: fp16 + reshape to (batch*hidden, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Convert to fp16 for the heavy multiply
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # bring hidden dimension to batch axis
    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_h = right_h.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # collapse first two dimensions -> total_batch = B * hidden_dim
    total_batch = B * hidden_dim
    left_h = left_h.view(total_batch, N, N)
    right_h = right_h.view(total_batch, N, N)

    # allocate output buffer (fp16)
    out_h = torch.empty_like(left_h)  # shape (total_batch, N, N), dtype fp16

    # ------------------------------------------------------------------
    # 6. Launch Triton GEMM kernel (A @ Bᵀ)
    # ------------------------------------------------------------------
    # strides for contiguous (batch, M, K) layout
    stride_A_batch = left_h.stride(0)   # N * N
    stride_A_m = left_h.stride(1)       # N
    stride_A_k = left_h.stride(2)       # 1

    stride_B_batch = right_h.stride(0)  # N * N
    stride_B_k = right_h.stride(1)      # N   (row stride)
    stride_B_n = right_h.stride(2)      # 1   (col stride)

    stride_C_batch = out_h.stride(0)    # N * N
    stride_C_m = out_h.stride(1)        # N
    stride_C_n = out_h.stride(2)        # 1

    # block sizes (tuned for typical Ampere GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = total_batch * num_blocks_m * num_blocks_n

    batched_matmul_transpose_kernel[grid](
        left_h,
        right_h,
        out_h,
        total_batch,
        N,  # M
        N,  # N
        N,  # K
        stride_A_batch, stride_A_m, stride_A_k,
        stride_B_batch, stride_B_k, stride_B_n,
        stride_C_batch, stride_C_m, stride_C_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Restore original 4‑D layout and cast back to fp32
    # ------------------------------------------------------------------
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 8. Final LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # still fp32

    # ------------------------------------------------------------------
    # 9. Apply out_gate element‑wise via a tiny Triton kernel
    # ------------------------------------------------------------------
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)
    total_el = out_flat.numel()
    BLOCK = 1024
    grid = ( (total_el + BLOCK - 1) // BLOCK, )
    elementwise_mul_f32_kernel[grid](
        out_flat,
        gate_flat,
        out_flat,
        total_el,
        BLOCK=BLOCK,
    )
    # out is now gated in‑place

    # ------------------------------------------------------------------
    # 10. Project back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
```
