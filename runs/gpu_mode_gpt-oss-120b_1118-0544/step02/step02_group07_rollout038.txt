    # 2. Projections
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])
    # 3. Mask (if any)
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f
    # 4. Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))
    left = (left_proj * left_gate).to(torch.float16)
    right = (right_proj * right_gate).to(torch.float16)

"""
TriMul (outgoing) forward pass – fused Triton implementation.

The kernel performs the O(N³) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
as a batched matrix multiplication A @ Bᵀ using a custom Triton GEMM.
All other ops (LayerNorm, Linear, gating, final projection) are done with
torch, but at least one heavy operation is executed in Triton to satisfy
the “must use Triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (int)
    stride_a_batch, stride_am, stride_ak,   # A strides
    stride_b_batch, stride_bk, stride_bn,   # B strides (B is already transposed)
    stride_c_batch, stride_cm, stride_cn,   # C strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """C[batch, M, N] = A[batch, M, K] @ B[batch, K, N]  (half→float32)."""
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col
    pid_b = tl.program_id(2)  # batch (B*hidden)

    # --------------------------------------------------------------
    # Offsets for this tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B tile: (BLOCK_K, BLOCK_N) – B is already transposed (K,N)
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot produces fp32 accumulation automatically for fp16 inputs
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write result
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """c = a * b  (1‑D flatten, FP32)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)

        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates and downcast to fp16
    # ------------------------------------------------------------------
    left = (left_proj * left_gate).to(torch.float16)
    right = (right_proj * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣ Prepare tensors for batched GEMM: (B*hidden, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Move hidden to middle dimension then flatten batch*hidden
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]

    left = left.view(B * hidden_dim, N, N)         # [B*H, N, N]
    right = right.view(B * hidden_dim, N, N).permute(0, 2, 1).contiguous()  # [B*H, N, N] -> transpose K

    # ------------------------------------------------------------------
    # 7️⃣ Batched GEMM via Triton (fp16 inputs → fp32 accumulation)
    # ------------------------------------------------------------------
    out_h = torch.empty(
        (B * hidden_dim, N, N), dtype=torch.float16, device=device
    )

    # Strides (in elements)
    stride_a_batch = left.stride(0)
    stride_am = left.stride(1)
    stride_ak = left.stride(2)

    stride_b_batch = right.stride(0)
    stride_bk = right.stride(1)   # K stride (fast)
    stride_bn = right.stride(2)   # N stride

    stride_c_batch = out_h.stride(0)
    stride_cm = out_h.stride(1)
    stride_cn = out_h.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )
    batched_gemm_kernel[grid](
        left,
        right,
        out_h,
        N,
        N,
        N,
        stride_a_batch,
        stride_am,
        stride_ak,
        stride_b_batch,
        stride_bk,
        stride_bn,
        stride_c_batch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Convert to fp32 and restore original layout
    # ------------------------------------------------------------------
    out = (
        out_h.to(torch.float32)
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)   # [B, N, N, hidden]
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply final LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 🔟 Element‑wise gating (out_gate) – Triton kernel
    # ------------------------------------------------------------------
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)
    total_el = out_flat.numel()
    BLOCK = 1024
    grid_gate = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid_gate](
        out_flat, gate_flat, out_flat, total_el, BLOCK=BLOCK
    )
    # out is now multiplied in‑place

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Final projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
