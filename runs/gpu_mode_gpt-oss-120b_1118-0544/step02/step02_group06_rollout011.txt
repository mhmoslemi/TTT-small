@triton.jit
def tri_mul_outgoing_kernel(a_ptr, b_ptr, c_ptr,
    BATCH: tl.constexpr,
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

    pid = tl.program_id(axis=0) # M tile
    pid_n = tl.program_id(axis=1) # N tile
    pid_b = tl.program_id(axis=2) # batch

    # Compute offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute batch pointer offsets
    a_base = a_ptr + pid_b * stride_a_batch
    b_base = b_ptr + pid_b * stride_b_batch
    c_base = c_ptr + pid_b * stride_c_batch

    # load initial a block only once? We'll accumulate over K dimension as loops.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a_ptrs = a_base + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k)
        b_ptrs = b_base + (cur_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)

        # load with mask
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot
        acc += tl.dot(a, b)

    # Write result
    c_ptrs = c_base + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
Outgoing Triangle Multiplicative Update (TriMul) kernel.

This implementation follows the reference PyTorch TriMul but fuses many
operations:

* LayerNorm on the input.
* Combined Linear+Gate projections for the left and right tensors
  (two weight matrices are concatenated to reduce the number of GEMM calls).
* Optional masking.
* Half‑precision (fp16) matrix multiplication using a custom Triton kernel.
  The kernel computes, for each hidden slice h,
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
  which is a batched GEMM C = A @ Bᵀ with fp16 inputs and fp32 accumulation.
* Final LayerNorm, gating and projection back to the original dimension.

The kernel is deliberately lightweight and uses block sizes that give good
occupancy for the target B200 GPU. All heavy arithmetic is performed in
fp16/Tensor‑Core while the accumulation is in fp32, and the final output
remains fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,            # left  tensor (batch, M, K)   fp16
    b_ptr,            # right tensor (batch, K, N)   fp16 (read as transposed)
    c_ptr,            # output   (batch, M, N)   fp32
    BATCH: tl.constexpr,
    M: tl.constexpr,
    N: tl.constexpr,
    K: tl.constexpr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col
    pid_b = tl.program_id(2)  # batch element

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # load A tile (M x K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load Bᵀ tile (K x N) – note the swapped stride for the transposition
        b_ptrs = b_batch + (cur_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product and accumulation (fp16 -> fp32 via up‑cast on addition)
        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – may be ignored if config["nomask"]=True
            - weights: dict of model parameters (see reference implementation)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device).half()
    right_proj_weight = weights["right_proj.weight"].to(device).half()
    left_gate_weight = weights["left_gate.weight"].to(device).half()
    right_gate_weight = weights["right_gate.weight"].to(device).half()
    out_gate_weight = weights["out_gate.weight"].to(device).half()

    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)  # final projection stays fp32

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # Switch to fp16 for the heavy linear projections
    # ------------------------------------------------------------------
    x_h = x.half()                                 # [B, N, N, dim] fp16
    B, N, _, _ = x_h.shape

    # ------------------------------------------------------------------
    # Combined linear + gate for left side (proj + gate)
    # ------------------------------------------------------------------
    left_comb_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0)  # (2*H, dim)
    x_flat = x_h.view(-1, dim)                      # (B*N*N, dim)
    left_comb = F.linear(x_flat, left_comb_weight)  # (B*N*N, 2*H)
    left_comb = left_comb.view(B, N, N, 2 * hidden_dim)

    left = left_comb[..., :hidden_dim]              # (B, N, N, H)
    left_gate_raw = left_comb[..., hidden_dim:]     # (B, N, N, H)

    # ------------------------------------------------------------------
    # Combined linear + gate for right side (proj + gate)
    # ------------------------------------------------------------------
    right_comb_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0)  # (2*H, dim)
    right_comb = F.linear(x_flat, right_comb_weight)  # (B*N*N, 2*H)
    right_comb = right_comb.view(B, N, N, 2 * hidden_dim)

    right = right_comb[..., :hidden_dim]            # (B, N, N, H)
    right_gate_raw = right_comb[..., hidden_dim:]   # (B, N, N, H)

    # ------------------------------------------------------------------
    # Optional masking (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x_h.dtype).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gates (sigmoid) and apply them
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)        # (B, N, N, H) fp16
    right_gate = torch.sigmoid(right_gate_raw)      # (B, N, N, H) fp16
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Outgoing gate (separate linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight))  # (B, N, N, H) fp16
    out_gate = out_gate.float()                               # cast to fp32 for later

    # ------------------------------------------------------------------
    # Prepare tensors for batched GEMM: reshape to (B*H, N, N)
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()

    left_h = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_h = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch_dim = left_h.shape[0] * left_h.shape[1]    # B * H
    M = N
    K = N

    left_h = left_h.view(batch_dim, M, K)            # (B*H, N, N) fp16
    right_h = right_h.view(batch_dim, M, K)          # (B*H, N, N) fp16

    # ------------------------------------------------------------------
    # Allocate output buffer (fp32) and launch Triton kernel
    # ------------------------------------------------------------------
    out_h = torch.empty((batch_dim, M, M), dtype=torch.float32, device=device)

    # strides
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)

    stride_b_batch = right_h.stride(0)
    stride_b_k = right_h.stride(2)   # K dimension stride
    stride_b_n = right_h.stride(1)   # N dimension stride (will be used as column after transposition)

    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    # Tile sizes – tuned for the target GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (M + BLOCK_N - 1) // BLOCK_N,
        batch_dim,
    )

    tri_mul_outgoing_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        BATCH=batch_dim,
        M=M,
        N=M,
        K=K,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] (float32)
    # ------------------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)          # (B, N, N, H)
        .contiguous()
    )  # already fp32

    # ------------------------------------------------------------------
    # Final LayerNorm, gating and projection back to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate               # apply outgoing gate (fp32)
    out = F.linear(out, to_out_weight) # (B, N, N, dim) fp32

    return out
