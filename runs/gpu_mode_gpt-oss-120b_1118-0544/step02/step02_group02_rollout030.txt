                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # group id
    pid0 = tl.program_id(0) # group = batch*H
    pid1 = tl.program_id(1) # i block
    pid2 = tl.program_id(2) # j block

    batch = pid0 // H
    hidden = pid0 % H

    # offset indices
    i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    # read config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Grab weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5)

    # Projections (no bias)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gate linear
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply mask if present
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Apply gates
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # Prepare output tensor
    B, N, _, _ = left.shape  # B,N,N,hidden
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # Compute reduction via Triton kernel
    # get strides
    left_stride_b, left_stride_i, left_stride_k, left_stride_h = left.stride()
    right_stride_b, right_stride_i, right_stride_k, right_stride_h = right.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_h = out.stride()

    # Define block sizes
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # Compute grid
    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        left_stride_b, left_stride_i, left_stride_k, left_stride_h,
        right_stride_b, right_stride_i, right_stride_k, right_stride_h,
        out_stride_b, out_stride_i, out_stride_j, out_stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # At this point out is float32

    # Apply to_out_norm (LayerNorm over hidden_dim)
    out = torch.nn.functional.layer_norm(out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5)

    # Apply output gating (broadcast)
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The kernel fuses the O(N³) reduction:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
for each hidden channel `h`.  All LayerNorms, linear projections and
sigmoid gates are performed with PyTorch; the heavy reduction is executed
with a custom Triton kernel that works directly on the [B,N,N,H] layout
to avoid costly permutes/copies.  The kernel tiles the i‑j plane (32×32)
and the reduction dimension k (32) and uses `tl.dot` for each tile,
accumulating in float32 while the inputs are stored in float16.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid0 = tl.program_id(0)  # batch * hidden
    pid1 = tl.program_id(1)  # i‑tile
    pid2 = tl.program_id(2)  # j‑tile

    # decode batch and hidden indices
    b = pid0 // H
    h = pid0 % H

    # i and j coordinates of the tile
    i_off = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # pointer to the output tile (float32)
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i_off[:, None] * stride_out_i
        + j_off[None, :] * stride_out_j
        + h * stride_out_h
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k_off = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # left tile  (i, k)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i_off[:, None] * stride_left_i
            + k_off[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0,
        )  # fp16

        # right tile (j, k)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + j_off[:, None] * stride_right_i
            + k_off[None, :] * stride_right_k
            + h * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=j_mask[:, None] & k_mask[None, :],
            other=0,
        )  # fp16

        # matrix multiply of the tile (M×K)·(N×K)ᵀ → (M×N)
        # tl.dot returns fp16, cast to fp32 before accumulation
        acc += tl.dot(left, right, trans_b=True).to(tl.float32)

    # store the result
    store_mask = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    Compute the forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) – may be ignored
            - weights: dict of model weight tensors
            - config: dict containing at least:
                * "dim" (int) – input channel dimension
                * "hidden_dim" (int) – hidden dimension used in the TriMul update
                * "nomask" (bool, optional) – if True, mask is ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) Model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5) Apply gates and convert to half precision for the heavy reduction
    # ------------------------------------------------------------------
    left = (left * left_gate).half().contiguous()
    right = (right * right_gate).half().contiguous()

    B, N, _, _ = left.shape  # B, N, N, hidden_dim

    # ------------------------------------------------------------------
    # 6) Triton kernel for the O(N³) reduction
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Final LayerNorm, output gating and projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # back to [B, N, N, dim]

    return out
