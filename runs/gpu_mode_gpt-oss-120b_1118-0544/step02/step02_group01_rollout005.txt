"""
Optimized forward of the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input.
2. Linear projections → left_proj, right_proj.
3. Linear gates → left_gate, right_gate, out_gate.
4. Fuse projection, gate and (optional) mask with a tiny Triton kernel:
   out = proj * gate * mask   (shape [B, N, N, H]).
5. Reduce the N³ term with a batched half‑precision GEMM.
   left  : [B, H, N, N]  (L_i,k)
   right : [B, H, N, N]  (R_j,k)  (transposed on the last two axes)
   out_hidden = (left @ right) → [B, N, N, H].
6. Output layer‑norm, apply out_gate and final linear projection.
All heavy elementwise work (mask+gate) is done in Triton; the N³ reduction
uses cuBLAS via torch.bmm on FP16 tensors.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_H: tl.constexpr,
):
    """
    out = proj * gate * mask   (broadcast mask over H)
    All tensors are contiguous and stored in row‑major order.
    """
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode pid -> (b, i, j)
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base offsets for the three inputs and the output
    proj_base = b * stride_proj_b + i * stride_proj_i + j * stride_proj_j
    gate_base = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
    mask_off  = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_base  = b * stride_out_b + i * stride_out_i + j * stride_out_j

    # vector of hidden indices
    off_h = tl.arange(0, BLOCK_H)
    mask_h = off_h < H

    # load values (broadcast mask across H)
    proj = tl.load(proj_ptr + proj_base + off_h * stride_proj_h,
                   mask=mask_h, other=0.0)
    gate = tl.load(gate_ptr + gate_base + off_h * stride_gate_h,
                   mask=mask_h, other=0.0)
    m    = tl.load(mask_ptr + mask_off,
                   mask=mask_h, other=1.0)   # scalar broadcast

    # store result
    tl.store(out_ptr + out_base + off_h * stride_out_h,
             proj * gate * m, mask=mask_h)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict with the model tensors
        - config       : dict with at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, cfg = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = cfg["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Gates (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 4) Fuse mask and gate with a Triton kernel (if a mask is present)
    # -----------------------------------------------------------------
    if mask is not None:
        # ensure mask is float32 on the same device
        mask_f = mask.to(dtype=x.dtype)

        # make tensors contiguous for deterministic strides
        left_proj_c = left_proj.contiguous()
        right_proj_c = right_proj.contiguous()
        left_gate_c = left_gate.contiguous()
        right_gate_c = right_gate.contiguous()
        mask_c = mask_f.contiguous()

        # allocate output buffers
        left = torch.empty_like(left_proj_c)
        right = torch.empty_like(right_proj_c)

        # strides in element units (contiguous layout: B,N,N,H)
        s_lp_b, s_lp_i, s_lp_j, s_lp_h = left_proj_c.stride()
        s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_proj_c.stride()
        s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate_c.stride()
        s_rg_gate_b, s_rg_gate_i, s_rg_gate_j, s_rg_gate_h = right_gate_c.stride()
        s_m_b, s_m_i, s_m_j = mask_c.stride()
        s_out_b, s_out_i, s_out_j, s_out_h = left.stride()   # same for right

        total = B * N * N
        grid = (total,)

        # left side
        fused_mask_gate_kernel[grid](
            left_proj_c,
            left_gate_c,
            mask_c,
            left,
            B, N, hidden_dim,
            s_lp_b, s_lp_i, s_lp_j, s_lp_h,
            s_lg_b, s_lg_i, s_lg_j, s_lg_h,
            s_m_b, s_m_i, s_m_j,
            s_out_b, s_out_i, s_out_j, s_out_h,
            BLOCK_H=hidden_dim,
        )
        # right side
        fused_mask_gate_kernel[grid](
            right_proj_c,
            right_gate_c,
            mask_c,
            right,
            B, N, hidden_dim,
            s_rg_b, s_rg_i, s_rg_j, s_rg_h,
            s_rg_gate_b, s_rg_gate_i, s_rg_gate_j, s_rg_gate_h,
            s_m_b, s_m_i, s_m_j,
            s_out_b, s_out_i, s_out_j, s_out_h,
            BLOCK_H=hidden_dim,
        )
    else:
        # simple elementwise multiplication when no mask is provided
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5) Batched half‑precision matrix multiplication (the N³ term)
    # -----------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # shape -> [B, H, N, N] where H is hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (L_i,k)
    right = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (R_j,k)

    left_mat = left.view(B * hidden_dim, N, N)     # (B*H, N, N)
    right_mat = right.view(B * hidden_dim, N, N)   # (B*H, N, N)

    # cuBLAS batched GEMM in FP16
    out_mat = torch.bmm(left_mat, right_mat)      # (B*H, N, N), fp16

    # reshape back to [B, N, N, H]
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, output gate and final projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])   # [B, N, N, dim]

    return out
