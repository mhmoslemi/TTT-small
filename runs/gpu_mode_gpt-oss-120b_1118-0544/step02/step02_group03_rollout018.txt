def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', False)
    eps = 1e-5

    # fetch weights
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    # Cast to fp16 for heavy ops
    x_fp16 = x.to(torch.float16)

    # Projections
    left = torch.nn.functional.linear(x_fp16, left_proj_weight.to(torch.float16))
    right = torch.nn.functional.linear(x_fp16, right_proj_weight.to(torch.float16))

    # Apply mask if needed
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_weight.to(torch.float16)))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_weight.to(torch.float16)))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_weight.to(torch.float16)))

    # Apply gate
    left = left * left_gate
    right = right * right_gate

    # Core update: batched matmul
    B, N, _, _ = left.shape
    # reshape to (B*hidden_dim, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_T = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

"""
TriMul outgoing update (AlphaFold3) – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the input tensor `x` (float32).
2. Cast to fp16 and apply two linear projections (left / right) to the hidden
   dimension `H`.  All linear layers are bias‑free and executed with fp16
   weights for higher bandwidth.
3. Optionally apply a binary mask and three sigmoid gates (left, right,
   out).  The left/right gates are fused with the mask in a single elementwise
   Triton kernel.
4. Core O(N³) update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication  left @ rightᵀ  for each
   (batch, head) pair.  It is performed with a single torch.bmm call
   (fp16 inputs, fp32 accumulation) which is far faster than a naïve
   Triton implementation.
5. Apply a second LayerNorm over the hidden dimension, multiply by the
   out‑gate using a tiny Triton element‑wise kernel, and finally project back
   to the original dimension `dim` with a bias‑free linear layer (fp16 → fp32).
"""

def _elem_mul_kernel(a_ptr, b_ptr, c_ptr, size,
                     BLOCK_SIZE: tl.constexpr):

    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


def _elem_mul_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Element‑wise multiplication using Triton.
    a, b – same shape, contiguous, same dtype (float32 recommended).
    Returns a new tensor c = a * b.
    """
    assert a.shape == b.shape, "Shapes must match for elementwise multiplication"
    N = a.numel()
    c = torch.empty_like(a)
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(N, BLOCK_SIZE),)
    _elem_mul_kernel[grid](
        a,
        b,
        c,
        N,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    return c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D]   (float32)
        - mask         : torch.Tensor [B, N, N]     (bool/float, optional)
        - weights      : dict of model tensors
        - config       : dict with keys "dim", "hidden_dim", "nomask" etc.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # model weights (cast once to the precision we use)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"].to(torch.float16, non_blocking=True)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16, non_blocking=True)

    left_gate_weight = weights["left_gate.weight"].to(torch.float16, non_blocking=True)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16, non_blocking=True)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16, non_blocking=True)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"].to(torch.float16, non_blocking=True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, D] float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for heavy work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (left / right)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, left_proj_weight)   # [B, N, N, H] fp16
    right = F.linear(x_fp16, right_proj_weight)  # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 4) Optional binary mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))   # [B,N,N,H] fp16
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))  # [B,N,N,H] fp16
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))    # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 6) Apply left/right gates (mask already applied above)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Core O(N³) update – batched matrix multiplication
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N) where each slice is a matrix multiplication
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_T = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # torch.bmm uses cuBLAS (Tensor‑Core friendly); inputs fp16 → output fp16
    # Cast the result to fp32 for the subsequent LayerNorm.
    out_flat = torch.bmm(left_flat, right_T).to(torch.float32)  # (B*H, N, N) fp32
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 8) Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # still fp32

    # ------------------------------------------------------------------
    # 9) Apply out‑gate using a tiny Triton kernel (elementwise mul)
    # ------------------------------------------------------------------
    out_gate_fp32 = out_gate.to(torch.float32).contiguous()
    out = _elem_mul_triton(out, out_gate_fp32)  # fp32 result

    # ------------------------------------------------------------------
    # 10) Final projection back to original dimension
    # ------------------------------------------------------------------
    out_fp16 = out.to(torch.float16)
    output_fp16 = F.linear(out_fp16, to_out_weight)  # [B,N,N,D] fp16
    output = output_fp16.to(torch.float32)

    return output
