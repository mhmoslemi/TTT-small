"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‐normalize the input tensor.
2. Compute four linear maps (left/right projections and gates) with `F.linear`.
3. Fuse mask, projection and gate with a tiny Triton kernel:
   out = proj * gate * mask   (stored as float16 to halve memory traffic).
   This kernel processes every element of the 4‑D tensor `[B,N,N,H]`.
4. Perform the N³ contraction using a single `torch.einsum` on the
   half‑precision tensors:
       out = einsum('b i k d, b j k d -> b i j d', left, right)
   No transposes or extra copies are required.
5. Apply the output LayerNorm, the output gate, and the final linear
   projection.  The returned tensor is float32.

Only the mask‑gate‑fusion part is written in Triton; the rest uses
high‑performance PyTorch operators.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_mask_gate_fp16_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK: tl.constexpr,
):
    """
    out = proj * gate * mask   (output stored as float16)

    All tensors are in `[B, N, N, H]` except mask `[B, N, N]`.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)

    total = B * N * N * H
    mask_valid = offs < total

    # ----- decode linear index to (b,i,j,h) -----
    tmp = offs
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # ----- compute flat offsets -----
    proj_off = (b * stride_proj_b
                + i * stride_proj_i
                + j * stride_proj_j
                + h * stride_proj_h)
    gate_off = (b * stride_gate_b
                + i * stride_gate_i
                + j * stride_gate_j
                + h * stride_gate_h)
    mask_off = (b * stride_mask_b
                + i * stride_mask_i
                + j * stride_mask_j)
    out_off = (b * stride_out_b
               + i * stride_out_i
               + j * stride_out_j
               + h * stride_out_h)

    # ----- load and compute -----
    proj = tl.load(proj_ptr + proj_off, mask=mask_valid, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=mask_valid, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=mask_valid, other=0.0)
    out = proj * gate * m

    # store as half‑precision
    tl.store(out_ptr + out_off, out.to(tl.float16), mask=mask_valid)

    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model weights (float32 tensors)
        - config       : dict with at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)

    # unpack
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],

    left_proj = F.linear(x, weights["left_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    if mask is not None:

        left_proj_c = left_proj.contiguous()
        left_gate_c = left_gate.contiguous()
        right_proj_c = right_proj.contiguous()
        right_gate_c = right_gate.contiguous()
        mask_c = mask_f.contiguous()

        # allocate output buffers (float16 to reduce memory bandwidth)
        left = torch.empty_like(left_proj_c, dtype=torch.float16)
        right = torch.empty_like(right_proj_c, dtype=torch.float16)

        )
        # right side
        fused_mask_gate_fp16_kernel[grid](
            right_proj_c,
            right_gate_c,
            mask_c,
            right,
            B, N, hidden_dim,
            s_rp_b, s_rp_i, s_rp_j, s_rp_h,
            s_rg_b, s_rg_i, s_rg_j, s_rg_h,
            s_m_b, s_m_i, s_m_j,
            s_out_b, s_out_i, s_out_j, s_out_h,
            BLOCK=BLOCK,
        )
    else:

        left = (left_proj * left_gate).to(torch.float16)
        right = (right_proj * right_gate).to(torch.float16)

    # 4) N³ contraction (einsum) – operates on half precision
    # -----------------------------------------------------------------
    # shape: [B, N, N, hidden]
    out_hidden = torch.einsum("b i k d, b j k d -> b i j d", left, right)
    out_hidden = out_hidden.to(torch.float32)  # back to FP32 for LayerNorm

    # -----------------------------------------------------------------
    # 5) Output LayerNorm, gate, final projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
