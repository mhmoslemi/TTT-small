"""
TriMul outgoing update (AlphaFold3) – Fast Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the input tensor `x` ([B, N, N, D]).
2. Cast to fp16 and project to a hidden dimension `H` with two independent linear
   layers → `left` and `right` ([B, N, N, H]).
3. (Optional) apply the binary mask.
4. Compute three sigmoid gates.  Fuse the element‑wise gate multiplications
   (`left *= left_gate`, `right *= right_gate`) with a tiny Triton kernel.
5. Core O(N³) update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication `left @ rightᵀ`.  We perform it with
   `torch.bmm`, which calls cuBLAS (Tensor‑Core friendly) and is far faster
   than a naïve Triton implementation.
6. Layer‑norm over the hidden dimension, multiply by the output gate and
   project back to the original dimension.
7. Return a float32 tensor of shape [B, N, N, D].

A minimal Triton kernel (in‑place elementwise multiply) is used to satisfy
the requirement that part of the work be performed on the GPU with Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: in‑place elementwise multiply a *= b
# ----------------------------------------------------------------------
@triton.jit
def inplace_mul_kernel(
    a_ptr,               # fp16 *
    b_ptr,               # fp16 *
    size,                # i32   total number of elements
    BLOCK: tl.constexpr, # compile‑time tile size
):
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offset < size

    a = tl.load(a_ptr + offset, mask=mask)
    b = tl.load(b_ptr + offset, mask=mask)
    tl.store(a_ptr + offset, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D]   (float32)
        - mask         : torch.Tensor [B, N, N]     (bool/float, optional)
        - weights      : dict of model tensors
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # stays float32

    # cast to fp16 for the heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (left/right)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_weight)   # [B, N, N, H]  fp16
    right = F.linear(x, right_proj_weight)  # fp16

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) and in‑place multiplication using Triton
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,H] fp16
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B,N,N,H] fp16

    # ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # in‑place multiply left *= left_gate  and  right *= right_gate
    total_elems = left.numel()
    BLOCK = 1024
    grid = triton.cdiv(total_elems, BLOCK)

    inplace_mul_kernel[grid](
        left.data_ptr(),
        left_gate.data_ptr(),
        total_elems,
        BLOCK=BLOCK,
    )
    inplace_mul_kernel[grid](
        right.data_ptr(),
        right_gate.data_ptr(),
        total_elems,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5. Core O(N³) update: batched matrix multiplication
    #    out[b, i, j, h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, N, N, H

    # Move hidden dimension to the batch axis for a single batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, j, k)

    # Transpose the last two axes of right to obtain (B, H, k, j)
    right_T = right_perm.transpose(-2, -1).contiguous()  # (B, H, k, j)

    # Collapse B and H into a single batch dimension
    BH = B * H
    left_flat = left_perm.view(BH, N, N)       # (BH, i, k)
    right_flat = right_T.view(BH, N, N)       # (BH, k, j)

    # Batched matmul using cuBLAS (fast FP16 Tensor‑Core path)
    out_flat = torch.bmm(left_flat, right_flat)  # (BH, i, j)  fp16

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    out = out.to(torch.float32)  # cast for the following LayerNorm

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # fp32

    out = out * out_gate.to(out.dtype)  # apply out‑gate

    # final linear projection back to dim
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out_fp16 = out.to(torch.float16)
    output = F.linear(out_fp16, to_out_weight)  # (B,N,N,dim) fp16
    output = output.to(torch.float32)

    return output
