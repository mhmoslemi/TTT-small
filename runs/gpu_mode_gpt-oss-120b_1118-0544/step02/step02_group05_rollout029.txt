"""
Optimized outgoing TriMul forward pass.

The implementation follows the PyTorch reference but replaces the
N³ reduction with a highly‑optimized batched matrix multiplication
(`torch.matmul`) performed in fp16 (tensor‑core friendly) and uses a
tiny Triton kernel for the final element‑wise gating.  All other
operations (LayerNorm, linear projections, gates, masking) are carried
out with PyTorch and mixed‑precision where appropriate.  The result is
computed in fp32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _elemwise_mul_kernel(
    a_ptr,  # float32*
    b_ptr,  # float32*
    c_ptr,  # float32*
    size,   # int32
    BLOCK: tl.constexpr,
):
    """Element‑wise multiplication c = a * b (both float32)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict with the following keys (float32 tensors)
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config: dict with entries
                'dim' (int), 'hidden_dim' (int), optional 'nomask' (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    eps = 1e-5
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden_dim (cast to fp16)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))   # [B, N, N, hidden_dim]
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1] and cast to fp16
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gates (computed in FP32, then cast to fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim] fp32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B,N,N,hidden_dim] fp32

    left_gate_h = left_gate.to(x_h.dtype)
    right_gate_h = right_gate.to(x_h.dtype)

    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 5) Core N³ contraction via batched fp16 matmul
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) – transposed K axis

    B, H, N, _ = left_perm.shape
    # Collapse batch and head dimensions
    left_mat = left_perm.view(B * H, N, N)   # (BH, N, N) fp16
    right_mat = right_perm.view(B * H, N, N)  # (BH, N, N) fp16

    # Batched matrix multiplication (tensor‑core fp16)
    out_fp16 = torch.matmul(left_mat, right_mat)  # (BH, N, N) fp16

    # Restore original layout and cast to fp32 for the next steps
    out = out_fp16.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().float()  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 7) Apply out_gate with a tiny Triton kernel (element‑wise)
    # ------------------------------------------------------------------
    size = out.numel()
    BLOCK = 1024  # reasonable block size for element‑wise kernels
    out_gated = torch.empty_like(out)
    grid = ( (size + BLOCK - 1) // BLOCK, )
    _elemwise_mul_kernel[grid](
        out, out_gate, out_gated,
        size,
        BLOCK=BLOCK,
    )
    out = out_gated

    # ------------------------------------------------------------------
    # 8) Final projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])  # (B,N,N,dim) float32

    return out
