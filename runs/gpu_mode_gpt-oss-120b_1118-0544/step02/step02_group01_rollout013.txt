```
mask_n = offs_n < N
```

```
@triton.jit
def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH_H, N,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_k, stride_r_j,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # batch*hidden index
    pid_m = tl.program_id(1)  # row tile
    pid_n = tl.program_id(2)  # column tile

    # Compute batch and hidden from pid:
    b = pid // hidden_dim
    h = pid % hidden_dim

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left
        a_ptrs = left_ptr + b*stride_l_b + h*stride_l_h + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load right
        b_ptrs = right_ptr + b*stride_r_b + h*stride_r_h + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b).to(tl.float32)

    # store result
    c_ptrs = out_ptr + b*stride_o_b + h*stride_o_h + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

```
@triton.jit
def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_k, stride_r_j,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Determine batch index and hidden index
    b = pid // H
    h = pid % H

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile
        a_ptrs = left_ptr + b * stride_l_b + h * stride_l_h + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile
        b_ptrs = right_ptr + b * stride_r_b + h * stride_r_h + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b).to(tl.float32)

    # Store result
    c_ptrs = out_ptr + b * stride_o_b + h * stride_o_h + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_k, stride_r_j,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,

    # store result (cast back to FP16)

    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    The implementation follows the PyTorch reference but replaces the
    N³ reduction with a fused batched half‑precision matrix multiplication
    written in Triton. All other operations (LayerNorm, linear projections,
    gating, final projection) are performed with PyTorch for simplicity.
    """
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    device = x.device

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 4) Apply mask (if provided) and combine with gates
    # -----------------------------------------------------------------
    if mask is not None:
        # broadcast mask over the channel dimension
        mask = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Free unused tensors early to save memory
    del left_proj, right_proj, left_gate, right_gate, x

    # -----------------------------------------------------------------
    # 5) Cast to half‑precision and permute to (B, H, N, N)
    # -----------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k dimension first)

    B, H, N, _ = left.shape
    assert H == hidden_dim

    # -----------------------------------------------------------------
    # 6) Triton kernel for batched matmul: out = left @ right
    # -----------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float16)

    # strides (in elements)
    s_l_b, s_l_h, s_l_i, s_l_k = left.stride()
    s_r_b, s_r_h, s_r_k, s_r_j = right.stride()
    s_o_b, s_o_h, s_o_i, s_o_j = out.stride()

    # block size choices (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                                 # batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,          # tiles over rows (i)
        (N + BLOCK_N - 1) // BLOCK_N,          # tiles over cols (j)
    )

    batched_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_l_b,
        s_l_h,
        s_l_i,
        s_l_k,
        s_r_b,
        s_r_h,
        s_r_k,
        s_r_j,
        s_o_b,
        s_o_h,
        s_o_i,
        s_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7) Convert back to float32 and restore original layout
    # -----------------------------------------------------------------
    out_hidden = out.to(torch.float32).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 8) Output LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate

    out = F.linear(out_hidden, weights["to_out.weight"])

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_k, stride_r_j,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched half‑precision matrix multiplication used in the outgoing
    TriMul operation.

    Computes, for each (b, h) pair (where h indexes the hidden dimension):
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, k, j]

    The tensors `left` and `right` have shape [B, H, N, N] and are
    stored in row‑major order (contiguous after a .permute+contiguous).
    The output `out` has the same shape.
    """
    pid = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1) # tile row
    pid_n = tl.program_id(2) # tile column

    # recover batch and hidden indices
    b = pid // H
    h = pid % H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension K
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:  [M, K]
        a_ptrs = (
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + offs_m[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right tile: [K, N]
        b_ptrs = (
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + offs_k[:, None] * stride_r_k
            + offs_n[None, :] * stride_r_j
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product + accumulation in FP32
        acc += tl.dot(a, b).to(tl.float32)

    # store the result (back to FP16)
    c_ptrs = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + offs_m[:, None] * stride_o_i
        + offs_n[None, :] * stride_o_j
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
