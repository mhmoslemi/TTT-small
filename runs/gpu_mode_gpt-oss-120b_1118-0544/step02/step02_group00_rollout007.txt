"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy part of TriMul is the tensor contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is a batch of N×N matrix multiplications for each hidden channel h.
We fuse this reduction into a single Triton kernel that loads tiles of the
two operands, accumulates in FP32 and writes the result in FP16. All other
operations (LayerNorms, linear projections, gates, masking) are performed
with PyTorch (using FP16 weights) to keep the implementation simple yet
fast.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # B: (batch, K, N)   (note: K stride = dim 2 of original)
    stride_ch, stride_cm, stride_cn,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ B^T for each batch.
    A has shape (batch, M, K)   (FP16)
    B has shape (batch, N, K)   (FP16) but we treat it as (batch, K, N) via strides.
    C has shape (batch, M, N)   (FP16)
    """
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # batch * hidden

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the batch slice
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # Offsets for K dimension (will be advanced inside the loop)
    offs_k = tl.arange(0, BLOCK_K)

    # Initial pointers for the first K‑tile
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        # Compute mask for this K‑tile
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write result (FP16)
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[:, None])


def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑based batched GEMM for the TriMul contraction.
    left, right: tensors of shape (B*H, N, N), dtype=torch.float16, contiguous.
    Returns: tensor of shape (B*H, N, N), dtype=torch.float16.
    """
    assert left.is_contiguous() and right.is_contiguous()
    batch, N, K = left.shape
    assert N == K, "TriMul expects square sequence dimension"
    # Heuristic block sizes
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    out = torch.empty_like(left)

    _tri_mul_kernel[grid](
        left,                     # A_ptr
        right,                    # B_ptr
        out,                      # C_ptr
        N, N, N,                  # M, N, K
        left.stride(0), left.stride(1), left.stride(2),
        right.stride(0), right.stride(2), right.stride(1),  # note: B is accessed as (K,N)
        out.stride(0), out.stride(1), out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on input
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for the compute‑heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # Optional masking (applied before gating, as in reference)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gate projections
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for Triton GEMM (fold hidden dim into batch)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    # Collapse batch & hidden => (B*H, N, N)
    left = left.view(B * H, N, N)
    right = right.view(B * H, N, N)

    # ------------------------------------------------------------------
    # Core O(N³·H) reduction via Triton
    # ------------------------------------------------------------------
    out_hidden = _tri_mul_triton(left, right)        # (B*H, N, N)

    # ------------------------------------------------------------------
    # Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_hidden.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # Output LayerNorm, gate, and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,C)

    # Cast back to original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
