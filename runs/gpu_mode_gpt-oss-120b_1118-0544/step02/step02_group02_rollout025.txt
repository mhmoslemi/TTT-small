    BATCH: tl.constexpr,
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,

):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

@triton.jit
def triton_batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_offset,
    M, N, K,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_j,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out of bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # mask for K dimension
        mask_k = offs_k < K

        # load left tile (batch, offs_m, offs_k)
        a_ptrs = left_ptr + batch * stride_l_batch + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None,:], other=0.0)
        a = a.to(tl.float32)

        # load right tile (batch, offs_k, offs_n)
        b_ptrs = right_ptr + batch * stride_r_batch + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None,:], other=0.0)
        b = b.to(tl.float32)

        # dot and accumulate
        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_ptr + batch * stride_o_batch + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None,:])

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The core O(N³) computation is a batched matrix multiplication:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batch of (N×N) @ (N×N)ᵀ multiplications for every
hidden channel `d`.  This implementation keeps the heavyweight linear
projections, layer‑norms and gating in PyTorch (they are cheap) and
offloads the massive batched matmul to a single Triton kernel that works
directly on FP16 inputs and accumulates in FP32.  No extra cast kernel is
required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul (left @ rightᵀ) over many batches.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    # pointers
    left_ptr,          # *fp16  [BATCH, M, K]
    right_ptr,         # *fp16  [BATCH, K, N] (right is accessed transposed)
    out_ptr,           # *fp32  [BATCH, M, N]

    # matrix dimensions
    M, N, K,

    # strides (in elements, not bytes)
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,

    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    # tile start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ------- load A tile (left) -------
        a_ptrs = (left_ptr
                  + batch_id * stride_l_batch
                  + offs_m[:, None] * stride_l_m
                  + offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # ------- load B tile (rightᵀ) -------
        # right is stored as (batch, i, k).  We want (batch, k, j),
        # i.e. access with stride_r_k for the k‑dimension and stride_r_n for j.
        b_ptrs = (right_ptr
                  + batch_id * stride_r_batch
                  + offs_k[:, None] * stride_r_k
                  + offs_n[None, :] * stride_r_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        # accumulate the dot product of the two tiles
        acc += tl.dot(a, b)

    # ------- store the result -------
    out_ptrs = (out_ptr
                + batch_id * stride_o_batch
                + offs_m[:, None] * stride_o_m
                + offs_n[None, :] * stride_o_n)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model weights
            - config: dict with at least "dim" and "hidden_dim".
                      Optional "nomask" indicates masks should be ignored.

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 0) retrieve weight tensors
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1) Input layer‑norm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]  FP32
    right = F.linear(x, right_proj_weight) # same shape

    # --------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4) Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # FP32
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 5) Apply gates and cast to FP16 for the heavy reduction
    # --------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)   # [B, N, N, hidden_dim] FP16
    right = (right * right_gate).to(torch.float16)

    B, N, _, _ = left.shape  # B, N, N, hidden_dim

    # --------------------------------------------------------------
    # 6) Prepare tensors for the batched matmul:
    #    we need shape (batch_n, N, N) where batch_n = B * hidden_dim.
    # --------------------------------------------------------------
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # allocate output buffer (FP32)
    out_flat = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # 7) Launch Triton batched‑matmul kernel
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_flat,        # *fp16
        right_flat,       # *fp16 (will be accessed as transposed)
        out_flat,         # *fp32
        M=N, N=N, K=N,
        stride_l_batch=left_flat.stride(0),
        stride_l_m=left_flat.stride(1),
        stride_l_k=left_flat.stride(2),
        stride_r_batch=right_flat.stride(0),
        stride_r_k=right_flat.stride(2),   # original k‑stride (fastest)
        stride_r_n=right_flat.stride(1),   # original i‑stride, serves as column stride after transpose
        stride_o_batch=out_flat.stride(0),
        stride_o_m=out_flat.stride(1),
        stride_o_n=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Ensure kernel completion before next ops
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 8) Reshape back to [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 9) Output layer‑norm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # apply output gate (both FP32)
    out = F.linear(out, to_out_weight)       # [B, N, N, dim] (no bias)

    return out
