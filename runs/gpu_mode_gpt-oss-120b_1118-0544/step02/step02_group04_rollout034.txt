"""
TriMul (outgoing) forward implementation.

The heavy O(N³·H) reduction is performed with a fused batched matrix
multiplication written in Triton. After LayerNorm and linear projections
(in FP32) the left/right tensors are cast to FP16, reshaped to
(batch*hidden_dim, N, N) and multiplied as:
    out[b, i, j] = Σ_k left[b, i, k] * right[b, j, k]
which is equivalent to batched GEMM: out = left @ rightᵀ.
The kernel accumulates in FP32 for numerical stability and writes the
result back as FP32. The rest of the TriMul pipeline (gating, layer‑norm,
output projection) is expressed in PyTorch.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_fp16_kernel(
    A_ptr,                     # *float16
    B_ptr,                     # *float16 (already transposed: shape [..., K, N])
    C_ptr,                     # *float32
    M, N, K,
    stride_ah, stride_am, stride_ak,   # A: batch, row, col strides
    stride_bh, stride_bk, stride_bn,   # B: batch, row(K), col(N) strides
    stride_ch, stride_cm, stride_cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # batch*hidden slice

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # pointers to the start of the current tile (k = 0)
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am) + (tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K
        k_idx = k_offset + tl.arange(0, BLOCK_K)
        k_mask = k_idx < K

        # load a tile of A and B with proper masking
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & k_mask[None, :],
            other=tl.float16(0.0)
        )
        b = tl.load(
            b_ptrs,
            mask=k_mask[:, None] & mask_n[None, :],
            other=tl.float16(0.0)
        )
        acc += tl.dot(a, b)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store the result
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton_fp16(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matmul for FP16 inputs:
        C[b] = A[b] @ B[b].T
    A : (batch, M, K)  dtype=float16
    B : (batch, N, K)  dtype=float16
    Returns:
        C : (batch, M, N) dtype=float32
    """
    # transpose B to (batch, K, N)
    B_T = B.transpose(-2, -1).contiguous()
    batch, M, K = A.shape
    _, K2, N = B_T.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # heuristic block sizes
    BLOCK_M = 128 if M >= 128 else 64
    BLOCK_N = 128 if N >= 128 else 64
    BLOCK_K = 32

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    _batched_matmul_fp16_kernel[grid](
        A,
        B_T,
        C,
        M,
        N,
        K,
        A.stride(0), A.stride(1), A.stride(2),   # A strides
        B_T.stride(0), B_T.stride(1), B_T.stride(2),   # B_T strides
        C.stride(0), C.stride(1), C.stride(2),   # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) or None
            - weights: dict of model weight tensors
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    ########################################
    # 1. Input LayerNorm
    ########################################
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    ########################################
    # 2. Linear projections (left/right)
    ########################################
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    ########################################
    # 3. Optional mask (applied before gating)
    ########################################
    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    ########################################
    # 4. Gating (sigmoid)
    ########################################
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    ########################################
    # 5. Prepare for batched matmul
    #    shape -> (B*H, N, N) and cast to fp16
    ########################################
    B, N, _, _ = left.shape
    # move hidden dim to batch axis
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch_hidden = B * hidden_dim
    left_flat = left.view(batch_hidden, N, N).contiguous()
    right_flat = right.view(batch_hidden, N, N).contiguous()

    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    ########################################
    # 6. Core O(N³·H) reduction using Triton
    ########################################
    out_flat = _batched_matmul_triton_fp16(left_fp16, right_fp16)  # (B*H, N, N) fp32

    ########################################
    # 7. Reshape back to [B, N, N, H]
    ########################################
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    ########################################
    # 8. Output LayerNorm
    ########################################
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    ########################################
    # 9. Apply output gate
    ########################################
    out = out * out_gate

    ########################################
    # 10. Final linear projection back to original dim
    ########################################
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
