@triton.jit
def triangle_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # offsets for M and N dimensions
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for i and j within N
    mask_i = i < N
    mask_j = j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K tiles
    num_tiles_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_tiles_k):
        k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # calculate pointers for left
        # left[b, i, k, h]
        left_ptrs = left_ptr + b * stride_l_b + i[:, None] * stride_l_i + k[None, :] * stride_l_k + h * stride_l_h
        # right[b, j, k, h] (note: right's second dimension is j, third is k)
        right_ptrs = right_ptr + b * stride_r_b + j[:, None] * stride_r_j + k[None, :] * stride_r_k + h * stride_r_h

        # load
        left_tile = tl.load(left_ptrs, mask=(mask_i[:, None] & mask_k[None, :]), other=0.0).to(tl.float32)
        right_tile = tl.load(right_ptrs, mask=(mask_j[:, None] & mask_k[None, :]), other=0.0).to(tl.float32)

        # matrix multiply accumulate
        # we need (i,k) * (j,k) -> sum over k: left_tile (M,K) dot right_tile (N,K) transposed: use trans_b=True to treat right as (K,N)
        acc += tl.dot(left_tile, right_tile, trans_b=True)

    # write output
    out_ptrs = out_ptr + b * stride_o_b + i[:, None] * stride_o_i + j[None, :] * stride_o_j + h * stride_o_h
    tl.store(out_ptrs, acc, mask=(mask_i[:, None] & mask_j[None, :]))

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # fetch weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LNs, linear
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)
    # projection
    left = F.linear(x, left_proj_weight)  # shape B, N, N, hidden_dim
    right = F.linear(x, right_proj_weight)

    # mask if needed
    if not nomask and mask is not None:
        # mask shape B,N,N bool/float -> broadcast to hidden_dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # B,N,N,1
        left = left * mask_f
        right = right * mask_f

    # gating
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # apply gates and cast to fp16
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # compute triangle multiplication using Triton kernel
    B, N, _, H = left.shape  # hidden_dim = H

    # allocate out tensor in float32
    out_hidden = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # compute strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_hidden.stride()

    # configure block sizes (choose values based on N)
    # Use heuristics: 
    # block_m = block_n = min(64, 2**(int(math.log2(N)) // 2))? Simpler use 64.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # compute grid sizes
    total_bh = B * H
    # Ceil division
    n_tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    n_tiles_j = (N + BLOCK_N - 1) // BLOCK_N

    grid = (total_bh, n_tiles_i, n_tiles_j)

    # launch
    triangle_mul_kernel[grid](
        left,
        right,
        out_hidden,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        # perhaps also `preload` but not needed
    )

    # final ln and output gating
    out = F.layer_norm(out_hidden,
                       normalized_shape=(H,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)
    out = out * out_gate
    out = F.linear(out, to_out_weight)

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy O(N³) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM using a custom Triton kernel.
The kernel works directly on the (B, N, N, H) layout without any
permutes or copies, loading tiles of the two inputs (FP16) and
accumulating in FP32.  This eliminates the costly
torch.bmm + cast pipeline from the reference implementation.

All other operations (LayerNorm, Linear projections, gating) are
performed in PyTorch.  The final output is a float‑32 tensor of
shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
import math


@triton.jit
def triangle_mul_kernel(
    # Pointers
    left_ptr,          # FP16, shape [B, N, N, H]
    right_ptr,         # FP16, shape [B, N, N, H]
    out_ptr,           # FP32, shape [B, N, N, H]
    # Dimensions
    B, N, H,
    # Strides for left (b, i, k, h)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # Strides for right (b, j, k, h)
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    # Strides for output (b, i, j, h)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # Compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """ Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] """

    pid_bh = tl.program_id(0)          # combined batch * hidden index
    pid_i  = tl.program_id(1)          # tile over i dimension
    pid_j  = tl.program_id(2)          # tile over j dimension

    # Decode batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # Offsets within the i and j dimensions
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds mask for the output tile
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_tiles_k = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_tiles_k):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ------------------------------------------------------------------
        # Load left tile   shape (BLOCK_M, BLOCK_K)   ← left[b,i,k,h]
        # ------------------------------------------------------------------
        left_ptrs = (left_ptr
                     + b * stride_l_b
                     + i_offset[:, None] * stride_l_i
                     + k_offset[None, :] * stride_l_k
                     + h * stride_l_h)
        left_tile = tl.load(left_ptrs,
                            mask=(mask_i[:, None] & mask_k[None, :]),
                            other=0.0).to(tl.float32)

        # ------------------------------------------------------------------
        # Load right tile  shape (BLOCK_N, BLOCK_K)   ← right[b,j,k,h]
        # ------------------------------------------------------------------
        right_ptrs = (right_ptr
                      + b * stride_r_b
                      + j_offset[:, None] * stride_r_j
                      + k_offset[None, :] * stride_r_k
                      + h * stride_r_h)
        right_tile = tl.load(right_ptrs,
                             mask=(mask_j[:, None] & mask_k[None, :]),
                             other=0.0).to(tl.float32)

        # ------------------------------------------------------------------
        # GEMM: (M,K) * (N,K).T  →  (M,N)
        # ------------------------------------------------------------------
        acc += tl.dot(left_tile, right_tile, trans_b=True)

    # ------------------------------------------------------------------
    # Store the result (FP32) with proper masking.
    # ------------------------------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_o_b
                + i_offset[:, None] * stride_o_i
                + j_offset[None, :] * stride_o_j
                + h * stride_o_h)
    tl.store(out_ptrs, acc,
             mask=(mask_i[:, None] & mask_j[None, :]))


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model weights
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)    # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5. Apply gates and cast to FP16 for the heavy reduction
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Triangle multiplicative update via Triton GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    out_hidden = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # Strides (convert to Python ints for Triton)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_hidden.stride()

    # Tile sizes – chosen to fit in shared memory for all supported N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid dimensions
    total_bh = B * H
    n_tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    n_tiles_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (total_bh, n_tiles_i, n_tiles_j)

    # Launch kernel
    triangle_mul_kernel[grid](
        left,                     # fp16
        right,                    # fp16
        out_hidden,               # fp32
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7. Final LayerNorm, output gating and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_hidden,
        normalized_shape=(H,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate               # apply output gate (shape matches)
    out = F.linear(out, to_out_weight)  # project back to original dim

    return out
