```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_m = tl.program_id(0)  # row block index i
    pid_n = tl.program_id(1)  # col block index j
    pid_bh = tl.program_id(2)  # batch*hidden index

    # compute row and col offsets
    i_base = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bound rows/cols
    mask_i = i_base < N
    mask_j = j_base < N

    # Determine batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # compute base pointer offsets for this batch and hidden
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_base < N

        # Load A tile: left[i,k]
        a_ptrs = left_base + i_base[:, None] * stride_left_i + k_base[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile: right[j,k] need to transpose so shape (K,N)
```

```python
"""
Custom TriMul kernel:
- LayerNorm + Linear projections + gates (PyTorch)
- Core O(N^3) reduction performed by a Triton kernel that computes
  a batched matrix multiplication per hidden channel:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
  left/right are fp16, accumulation in fp32.
- Final LayerNorm, output gate, and linear projection (PyTorch).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def outgoing_tri_mul_gemm(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_bh = tl.program_id(2) # batch*hidden index

    # compute absolute indices
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # base pointers for this (b,h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left tile: shape (M, K)
        a_ptrs = left_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile (transposed): shape (K, N)
        b_ptrs = right_base + j[None, :] * stride_right_i + k[:, None] * stride_right_k
        b = tl.load(b_ptrs, mask=mask_j[None, :] & mask_k[:, None], other=0.0)

        acc += tl.dot(a, b)   # a: (M,K) fp16, b: (K,N) fp16, result fp32

    # write output tile
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Execute outgoing TriMul forward pass.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # extract weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim], float32
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 5) Apply gates and convert to fp16 for the reduction
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    B, N, _, _ = left.shape   # B, N, N, hidden_dim

    # ------------------------------------------------------------------
    # 6) Triton batched GEMM: out = Σ_k left[...,k,:] * right[...,k,:]
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # strides (in elements) for the tensors
    l_stride_b, l_stride_i, l_stride_k, l_stride_h = left.stride()
    r_stride_b, r_stride_i, r_stride_k, r_stride_h = right.stride()
    o_stride_b, o_stride_i, o_stride_j, o_stride_h = out.stride()

    # Tile sizes – chosen to give good occupancy on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M,
          (N + BLOCK_N - 1) // BLOCK_N,
          B * hidden_dim ),
    )

    outgoing_tri_mul_gemm[grid](

        o_stride_j,
        o_stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # 7) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

"""
Outgoing TriMul (Triangle Multiplicative Update) forward pass.

The operation consists of:
1. LayerNorm over the last dimension.
2. Two bias‑free linear projections (left/right) followed by optional mask.
3. Gating networks (sigmoid after linear).
4. Core O(N³) reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM in Triton (fp16 matmul, fp32 accumulation).
5. LayerNorm, output gate, and final linear projection back to the original dim.

Only the heavy reduction is performed in Triton; all other elementwise
operations stay in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def outgoing_tri_mul_gemm(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM for the outgoing triangle multiplicative update.

    Computes, for each (b, h):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    left/right are fp16, accumulation is fp32, result stored as fp32.
    """
    pid_m = tl.program_id(0)  # block row (i)
    pid_n = tl.program_id(1)  # block column (j)
    pid_bh = tl.program_id(2)  # batch * hidden index

    # absolute i/j coordinates of the tile
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # decode batch and hidden index from pid_bh
    b = pid_bh // H
    h = pid_bh % H

    # base pointers for this (b,h) slice
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left tile: shape (M, K)
        a_ptrs = left_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)           # fp16

        # right tile (transposed): shape (K, N)
        b_ptrs = right_base + j[None, :] * stride_right_i + k[:, None] * stride_right_k
        b = tl.load(b_ptrs,
                    mask=mask_j[None, :] & mask_k[:, None],
                    other=0.0)           # fp16

        # dot: fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # write result tile (fp32)
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Extract model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim], float32
    right = F.linear(x, right_proj_weight) # same dtype

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating networks (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 5) Apply gates and cast to fp16 for the triton GEMM
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    B, N, _, _ = left.shape  # B, N, N, hidden_dim

    # ------------------------------------------------------------------
    # 6) Batched GEMM via Triton
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # strides (in elements) for the tensors
    l_stride_b, l_stride_i, l_stride_k, l_stride_h = left.stride()
    r_stride_b, r_stride_i, r_stride_k, r_stride_h = right.stride()
    o_stride_b, o_stride_i, o_stride_j, o_stride_h = out.stride()

    # Tile sizes – empirically good for the B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M,
          (N + BLOCK_N - 1) // BLOCK_N,
          B * hidden_dim ),
    )

    outgoing_tri_mul_gemm[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        l_stride_b,
        l_stride_i,
        l_stride_k,
        l_stride_h,
        r_stride_b,
        r_stride_i,
        r_stride_k,
        r_stride_h,
        o_stride_b,
        o_stride_i,
        o_stride_j,
        o_stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, output gate, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # back to original dim

    return out
```
