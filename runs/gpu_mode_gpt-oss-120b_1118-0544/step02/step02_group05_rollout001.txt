"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

The module computes, for each batch and hidden channel `d`:

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are linear projections of the input tensor
followed by gating and optional masking.  The contraction is a batched
matrix multiplication `C = A @ Bᵀ` with `A` and `B` of shape (N, N) for
every hidden channel.  We implement this GEMM with a Triton kernel that
uses fp16 inputs, fp32 accumulation, and 2‑level tiling (BLOCK_M,
BLOCK_N, BLOCK_K).  All remaining operations (LayerNorm, linear
projections, gates, final LayerNorm and projection) are performed with
PyTorch.  The final output is float32 in shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                 # pointers to A, B, C
    M, N, K,                             # matrix dimensions (int32)
    stride_ah, stride_am, stride_ak,     # strides for A  (batch, row, col)
    stride_bh, stride_bk, stride_bn,     # strides for B  (batch, row, col)
    stride_ch, stride_cm, stride_cn,     # strides for C  (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ.

    A : (batch, M, K)   fp16
    B : (batch, K, N)   fp16   (already transposed)
    C : (batch, M, N)   fp32 accumulator
    """
    pid_batch = tl.program_id(0)  # batch * hidden dimension index
    pid_m = tl.program_id(1)      # block row
    pid_n = tl.program_id(2)      # block column

    # Compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch/head
    a_batch = a_ptr + pid_batch * stride_ah
    b_batch = b_ptr + pid_batch * stride_bh
    c_batch = c_ptr + pid_batch * stride_ch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # OOB masks
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Load A and B tiles (fp16)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        # Compute and accumulate
        acc += tl.dot(a, b)  # fp16 -> promoted to fp32 when added

    # Write C
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Triton‑backed forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32/bool) or None
            - weights: dict with the following keys (all float32):
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config: dict with entries 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # --------------------------------------------------------------
    # 2) Linear projections (to hidden_dim), cast to fp16
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)    # [B, N, N, hidden_dim] fp16
    right = F.linear(x_h, right_proj_w)  # same

    # --------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        # Ensure mask dtype matches fp16 for multiplication
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 4) Gates (computed in fp32, then cast to fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # fp32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # fp32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # fp32

    left_gate_h = left_gate.to(x_h.dtype)
    right_gate_h = right_gate.to(x_h.dtype)

    left = left * left_gate_h
    right = right * right_gate_h

    # --------------------------------------------------------------
    # 5) Core N³ contraction via Triton (batched GEMM)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Rearrange to (B, hidden, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) – k dimension first for GEMM

    # Collapse batch and head dimensions → (BH, N, N)
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # (BH, M=N, K=N)
    right_mat = right_perm.view(BH, N, N)  # (BH, K=N, N)

    # Allocate output buffer (float32 accumulator)
    out_mat = torch.empty((BH, N, N), dtype=torch.float32, device=x.device)

    # Tunable block sizes (work well for the tested ranges)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,                                 # batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,       # rows
        (N + BLOCK_N - 1) // BLOCK_N,       # cols
    )

    _tri_mul_matmul_kernel[grid](
        left_mat,                # a_ptr
        right_mat,               # b_ptr (already transposed)
        out_mat,                 # c_ptr
        N, N, N,                 # M, N, K
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6) Output LayerNorm, gating, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # out_gate is fp32; out is fp32 after norm
    out = F.linear(out, weights["to_out.weight"])  # back to dim, fp32

    return out
