def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm (dim)
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Projections (fp16)
    x_h = x_norm.to(torch.float16)
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Apply mask (if present)
    if mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # Gates (FP32)
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"])) # no bias
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate.to(torch.float16)

    out_{b,d} = left_{b,d} @ right_{b,d}ᵀ
with left/right of shape (B, H, N, N) where H = hidden_dim.
A custom Triton kernel computes the batched GEMM in mixed precision
(fp16 inputs, fp32 accumulation). All other operations (LayerNorm,
linear projections, gating, and final projection) are performed with
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dims (int32)
    stride_ah, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bh, stride_bk, stride_bn,   # strides for B (batch, row, col) – B is already transposed
    stride_ch, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ Bᵀ
    A : (batch, M, K)   fp16
    B : (batch, K, N)   fp16 (already transposed in the caller)
    C : (batch, M, N)   fp32 accumulator
    """
    pid_batch = tl.program_id(0)   # batch * head dimension
    pid_m = tl.program_id(1)       # block row (i)
    pid_n = tl.program_id(2)       # block col (j)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for the current batch/head
    a_batch = a_ptr + pid_batch * stride_ah
    b_batch = b_ptr + pid_batch * stride_bh
    c_batch = c_ptr + pid_batch * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Load A tile (fp16)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)

        # Load B tile (fp16) – B is already transposed (K×N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        # Multiply‑accumulate (dot returns fp16, promotion to fp32 on add)
        acc += tl.dot(a, b)

    # Write the result (fp32)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask: torch.Tensor or None, shape [B, N, N]
            - weights: dict with the following keys (float32 tensors):
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config: dict with entries 'dim' (int) and 'hidden_dim' (int)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                     # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden_dim (cast to fp16)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim] fp16
    right = F.linear(x_h, right_proj_w) # same dtype

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # Ensure mask is float16 and broadcast over the hidden dimension
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gating (computed in FP32, then cast to fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))      # [B,N,N,hidden_dim] FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))    # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))        # FP32

    left_gate_h = left_gate.to(x_h.dtype)
    right_gate_h = right_gate.to(x_h.dtype)

    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 5) Core cubic contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Rearrange tensors to (B, hidden_dim, N, N) and ensure contiguity
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, k, j) – already transposed

    # Collapse batch and head dimensions for the kernel
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # (BH, M, K)
    right_mat = right_perm.view(BH, N, N)  # (BH, K, N)

    # Output buffer (fp32 accumulator)
    out_mat = torch.empty((BH, N, N), dtype=torch.float32, device=x.device)

    # Tunable block sizes – chosen for good occupancy on modern GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        left_mat.stride(0),
        left_mat.stride(1),
        left_mat.stride(2),
        right_mat.stride(0),
        right_mat.stride(1),
        right_mat.stride(2),
        out_mat.stride(0),
        out_mat.stride(1),
        out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    out = out * out_gate  # broadcasted multiplication (FP32)

    # Final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"])  # (B, N, N, dim) – FP32

    return out
```
