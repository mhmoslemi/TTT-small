@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    BHD, M, N,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Number of blocks
    num_pb = tl.cdiv(M, BLOCK_M)
    num_pb_n = tl.cdiv(N, BLOCK_N)
    # total blocks per batch = num_pb * num_pb_n
    total_blocks_per_batch = num_pb * num_pb_n
    batch_idx = pid // total_blocks_per_batch
    block_idx = pid % total_blocks_per_batch
    block_i = block_idx // num_pb_n
    block_j = block_idx % num_pb_n

    # Compute offsets
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # offsets for K
    offs_k = tl.arange(0, BLOCK_K)

    # Compute masks
    mask_i = offs_i < M
    mask_j = offs_j < N

    # base offsets for batch
    batch_offset = batch_idx * stride_l_batch

    # pointers
    a_ptrs = left_ptr + batch_offset + offs_i[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
    b_ptrs = right_ptr + batch_offset + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_n

    # accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        # compute K offset base for each tile
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (offs_k < K - kk*BLOCK_K), other=0.0)  # shape [BM, BK]
        b = tl.load(b_ptrs, mask=(offs_k < K - kk*BLOCK_K)[:, None] & mask_j[None, :], other=0.0)  # shape [BK, BN]
        # cast to fp32 for dot
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_l_k
        b_ptrs += BLOCK_K * stride_r_k

    # apply masks for i and j out-of-bounds
    acc = tl.where(mask_i[:, None] & mask_j[None, :], acc, 0.0)

    # store
    out_offset = batch_idx * stride_o_batch + offs_i[:, None] * stride_o_m + offs_j[None, :] * stride_o_n
    tl.store(out_offset, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
   input_tensor, mask, weights, config = data
   dim = config["dim"]
   hidden_dim = config["hidden_dim"]
   nomask = config.get("nomask", False)

   # get weight tensors
   norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
   left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]
   left_gate_weight = weights["left_gate.weight"]; right_gate_weight = weights["right_gate.weight"]
   out_gate_weight = weights["out_gate.weight"]
   to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]
   to_out_weight = weights["to_out.weight"]

   # LayerNorm
   x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

   # Linear projections
   left = F.linear(x, left_proj_weight)   # float32
   right = F.linear(x, right_proj_weight)

   if not nomask and mask is not None:
       mask_f = mask.unsqueeze(-1).to(x.dtype)
       left = left * mask_f
       right = right * mask_f

   # Gates
   left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
   right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
   out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

   # Apply gates and cast to half
   left = (left * left_gate).to(torch.float16)
   right = (right * right_gate).to(torch.float16)

   B, N, _, _ = left.shape # shape B,N,N,hidden_dim
   # Rearrange to (B*hidden_dim, N, N)
   left = left.permute(0, 3, 1, 2).contiguous()
   right = right.permute(0, 3, 1, 2).contiguous()
   batch_hidden = B * hidden_dim
   left = left.view(batch_hidden, N, N)
   right = right.view(batch_hidden, N, N)

   # Output buffer
   out = torch.empty((batch_hidden, N, N), device=left.device, dtype=torch.float32)

   # Block sizes
   BLOCK_M = 64
   BLOCK_N = 64
   BLOCK_K = 32

   num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
   num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
   total_blocks = batch_hidden * num_blocks_m * num_blocks_n

   grid = (total_blocks,)

   trmul_outgoing_kernel[grid](
       left,
       right,
       out,
       batch_hidden,
       N, N, N,
       left.stride(0), left.stride(1), left.stride(2),
       right.stride(0), right.stride(2), right.stride(1),  # note swapped K,N strides
       out.stride(0), out.stride(1), out.stride(2),
       BLOCK_M=BLOCK_M,
       BLOCK_N=BLOCK_N,
       BLOCK_K=BLOCK_K,
       num_warps=4,

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.
The core O(N³) reduction  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a custom Triton batched matrix‑multiply kernel that works on
half‑precision data and accumulates in float32.  All layer‑norms, linear
projections and gating are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing triangle update
# ----------------------------------------------------------------------
@triton.jit
def trmul_outgoing_kernel(
    left_ptr,      # half *   (batch_hidden, M, K)
    right_ptr,     # half *   (batch_hidden, K, N) – note swapped strides
    out_ptr,       # float *  (batch_hidden, M, N)
    batch_hidden,  # total batch * hidden_dim
    M, N, K,       # sequence length (M = N = K)
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # Number of tiles per dimension
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_tiles_m * num_tiles_n

    # Map program id -> (batch, tile_i, tile_j)
    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    tile_i = tile_idx // num_tiles_n
    tile_j = tile_idx % num_tiles_n

    # Offsets for the M (i) and N (j) dimensions
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptr = (
            left_ptr
            + batch_idx * stride_l_batch
            + offs_i[:, None] * stride_l_m
            + cur_k[None, :] * stride_l_k
        )
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (transposed view): shape (BLOCK_K, BLOCK_N)
        b_ptr = (
            right_ptr
            + batch_idx * stride_r_batch
            + cur_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_n
        )
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Cast to FP32 and accumulate
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # Store the result
    out_ptr_base = (
        out_ptr
        + batch_idx * stride_o_batch
        + offs_i[:, None] * stride_o_m
        + offs_j[None, :] * stride_o_n
    )
    tl.store(out_ptr_base, acc, mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict with model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # used later

    # ------------------------------------------------------------------
    # 5) Apply gates and cast to half precision
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    B, N, _, _ = left.shape  # B, N, N, hidden_dim

    # ------------------------------------------------------------------
    # 6) Rearrange for batched GEMM: (B*hidden_dim, N, N)
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    batch_hidden = B * hidden_dim
    left = left.view(batch_hidden, N, N)
    right = right.view(batch_hidden, N, N)

    # ------------------------------------------------------------------
    # 7) Triton GEMM kernel (left @ rightᵀ)
    # ------------------------------------------------------------------
    out = torch.empty((batch_hidden, N, N), device=left.device, dtype=torch.float32)

    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = batch_hidden * num_tiles_m * num_tiles_n
    grid = (total_tiles,)

    trmul_outgoing_kernel[grid](
        left,
        right,
        out,
        batch_hidden,
        N, N, N,
        left.stride(0), left.stride(1), left.stride(2),
        right.stride(0), right.stride(2), right.stride(1),   # swapped K/N strides
        out.stride(0), out.stride(1), out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # back to original dim

    return out
