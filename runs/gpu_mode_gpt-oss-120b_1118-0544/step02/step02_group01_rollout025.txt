def fused_mask_gate_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr):
   pid = tl.program_id(0)
   offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
   # compute mask for offset < total
   total = B * N * N * H
   mask = offset < total
   # decode offset to indices:
   tmp = offset
   b = tmp // (N * N * H)
   tmp = tmp % (N * N * H)
   i = tmp // (N * H)
   tmp = tmp % (N * H)
   j = tmp // H
   h = tmp % H

   # compute addresses:
   proj_offset = b * stride_proj_b + i * stride_proj_i + j * stride_proj_j + h * stride_proj_h
   gate_offset = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
   mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
   # load:
   proj = tl.load(proj_ptr + proj_offset, mask=mask, other=0.0)
   gate = tl.load(gate_ptr + gate_offset, mask=mask, other=0.0)
   m = tl.load(mask_ptr + mask_offset, mask=mask, other=0.0)
   # multiply
   out = proj * gate * m
   tl.store(out_ptr + proj_offset, out, mask=mask)  # same offset as proj for out

def custom_kernel(data):
    x, mask, weights, cfg = data
    bs, seq_len, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # Normalization
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Linear projections
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    if mask is not None:
        mask = mask.to(x.dtype) # float32
        # Ensure contiguity
        left_proj_c = left_proj
        left_gate_c = left_gate
        right_proj_c = right_proj
        right_gate_c = right_gate
        mask_c = mask

        left = torch.empty_like(left_proj_c)
        right = torch.empty_like(right_proj_c)

        # Strides
        s_lp_b, s_lp_i, s_lp_j, s_lp_h = left_proj_c.stride()
        s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate_c.stride()

        s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate_c.stride()
        s_m_b, s_m_i, s_m_j = mask_c.stride()
        # out stride same as proj
        s_out_b, s_out_i, s_out_j, s_out_h = left_proj_c.stride()

        total = bs * seq_len * seq_len * hidden_dim
        BLOCK_SIZE = 16384 # maybe bigger for performance depending on memory; but keep 1024 or 2048

        fused_mask_gate_kernel[grid](

            left_gate_c,
            mask_c,
            left,
            bs, seq_len, hidden_dim,
            s_lp_b, s_lp_i, s_lp_j, s_lp_h,
            s_lg_b, s_lg_i, s_lg_j, s_lg_h,
            s_m_b, s_m_i, s_m_j,
            s_out_b, s_out_i, s_out_j, s_out_h,

"""
Highly‑optimized forward implementation of the outgoing Triangle Multiplicative
Update (TriMul) used in AlphaFold‑3 style models.

The kernel fuses mask, left/right gating and projection multiplication into a
single Triton kernel, while the cubic reduction over the sequence dimension is
performed as a batched half‑precision matrix multiplication (torch.bmm), which
leverages cuBLAS.  LayerNorms, linear projections and the final output projection
are kept in PyTorch for simplicity and to avoid unnecessary data movement.

The function `custom_kernel` follows the required signature:
    (input_tensor, mask, weights, config) -> output_tensor
where `weights` is a dict containing all model parameters.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Element‑wise kernel: out = proj * gate * mask

    proj, gate : [B, N, N, H]    (float32)
    mask       : [B, N, N]       (float32, broadcasted over H)
    out        : [B, N, N, H]    (float32)
    """
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offset < total

    # ----- decode linear offset -> (b, i, j, h) -----
    tmp = offset
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # ----- compute flat offsets for each tensor -----
    proj_off = (
        b * stride_proj_b
        + i * stride_proj_i
        + j * stride_proj_j
        + h * stride_proj_h
    )
    gate_off = (
        b * stride_gate_b
        + i * stride_gate_i
        + j * stride_gate_j
        + h * stride_gate_h
    )
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_off = (
        b * stride_out_b
        + i * stride_out_i
        + j * stride_out_j
        + h * stride_out_h
    )

    # ----- load, compute and store -----
    proj = tl.load(proj_ptr + proj_off, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=mask, other=0.0)

    tl.store(out_ptr + out_off, proj * gate * m, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict with model parameters
        - config       : dict with at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Gates (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 4) Fuse mask and gate (if mask present) via Triton kernel
    # -----------------------------------------------------------------
    if mask is not None:
        # ensure mask is float32 and on same device
        mask = mask.to(x.dtype)

        # output buffers for fused results
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # strides in element units
        s_lp_b, s_lp_i, s_lp_j, s_lp_h = left_proj.stride()
        s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate.stride()
        s_rp_b, s_rp_i, s_rp_j, s_rp_h = right_proj.stride()
        s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate.stride()
        s_m_b, s_m_i, s_m_j = mask.stride()
        # output uses same layout as projection tensors
        s_out_b, s_out_i, s_out_j, s_out_h = left_proj.stride()

        total = B * N * N * hidden_dim
        BLOCK_SIZE = 1024
        grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE,)

        # left side
        fused_mask_gate_kernel[grid](
            left_proj,
            left_gate,
            mask,
            left,
            B,
            N,
            hidden_dim,
            s_lp_b,
            s_lp_i,
            s_lp_j,
            s_lp_h,
            s_lg_b,
            s_lg_i,
            s_lg_j,
            s_lg_h,
            s_m_b,
            s_m_i,
            s_m_j,
            s_out_b,
            s_out_i,
            s_out_j,
            s_out_h,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # right side
        fused_mask_gate_kernel[grid](
            right_proj,
            right_gate,
            mask,
            right,
            B,
            N,
            hidden_dim,
            s_rp_b,
            s_rp_i,
            s_rp_j,
            s_rp_h,
            s_rg_b,
            s_rg_i,
            s_rg_j,
            s_rg_h,
            s_m_b,
            s_m_i,
            s_m_j,
            s_out_b,
            s_out_i,
            s_out_j,
            s_out_h,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5) Batched half‑precision matrix multiplication (the N³ core)
    # -----------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape to (B, hidden, N, N) – left and right are transposed appropriately
    left = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right = right.permute(0, 3, 2, 1).contiguous()        # [B, H, N, N] (transpose inner k)

    B, H, N, _ = left.shape
    left_mat = left.view(B * H, N, N)    # (B*H, N, N)
    right_mat = right.view(B * H, N, N)  # (B*H, N, N)

    # cuBLAS batched GEMM (FP16)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), dtype=fp16

    # reshape back to [B, N, N, H] and convert to float32
    out_hidden = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
