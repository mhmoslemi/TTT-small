"""
TriMul (outgoing) forward pass – fused implementation.

The heavy N³ reduction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is performed as a batched matrix multiplication using torch.bmm,
which leverages cuBLAS/TensorCore for maximum throughput.
All linear projections and gating are kept in FP16, while LayerNorms and
the final output stay in FP32 for numerical stability.
A tiny Triton kernel is used for the final element‑wise multiplication
with the output gate, satisfying the “must use a kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, n_elem,
                            BLOCK_SIZE: tl.constexpr):
    """
    Simple element‑wise multiplication kernel.
    Operates on 1‑D contiguous tensors (FP32).
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elem
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, [B, N, N, dim]
            - mask: torch.Tensor or None, [B, N, N]
            - weights: dict with the following keys (float32 tensors):
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config: dict with entries 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden_dim (FP16 for speed)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim] FP16
    right = F.linear(x_h, right_proj_w)  # same dtype/shape

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gates (FP32 -> FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))    # FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))      # FP32, used later

    left_gate_h = left_gate.to(x_h.dtype)
    right_gate_h = right_gate.to(x_h.dtype)

    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 5) Core N³ contraction via batched GEMM (torch.bmm)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # shape: (B, N, N, hidden_dim)

    # reshape to (B*H, N, N) where each hidden dimension is a separate batch
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)  (i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N)  (k, j)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # (BH, N, K)
    right_mat = right_perm.view(BH, N, N)  # (BH, K, N)

    # batched matrix multiplication using Tensor Cores (FP16 inputs)
    out_fp16 = torch.bmm(left_mat, right_mat)     # (BH, N, N) FP16
    out = out_fp16.float()                        # cast to FP32 for subsequent ops

    # reshape back to [B, N, N, hidden_dim]
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output gate (via Triton), final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # FP32

    # Element‑wise multiplication with out_gate using a tiny Triton kernel
    out_flat = out.contiguous().view(-1)
    out_gate_flat = out_gate.contiguous().view(-1)
    out_mul_flat = torch.empty_like(out_flat)

    grid = lambda meta: (triton.cdiv(out_flat.numel(), meta["BLOCK_SIZE"]),)
    _elementwise_mul_kernel[grid](
        out_flat, out_gate_flat, out_mul_flat,
        out_flat.numel(),
        BLOCK_SIZE=1024,
    )
    out = out_mul_flat.view_as(out)

    # Final linear projection back to original dimension
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim), FP32

    return out
