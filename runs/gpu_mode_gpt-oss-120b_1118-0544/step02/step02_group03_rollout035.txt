def apply_mask_gate_kernel(
    left_in_ptr, right_in_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    N_elements,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_elements
    left = tl.load(left_in_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_in_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    left_out = left * m * lg
    right_out = right * m * rg
    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)
```

"""
TriMul outgoing update (AlphaFold3) – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the 4‑D input x ∈ [B, N, N, C] (fp32) → cast to fp16.
2. Linear projections (fp16) produce left/right tensors of shape [B, N, N, H].
3. (Optional) binary mask is applied to left/right (broadcast over H) in PyTorch.
4. Compute three sigmoid gates (fp16) from x: left_gate, right_gate, out_gate.
5. Fuse element‑wise multiplication of left/right with their respective gates via a
   tiny Triton kernel (in‑place, no extra memory).
6. Core O(N³) update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication:
        left_perm  = left.permute(0,3,1,2)   # (B, H, N, N)
        right_perm = right.permute(0,3,1,2).permute(0,1,3,2)  # (B, H, N, N)
        left_flat  = left_perm.view(B*H, N, N)
        right_flat = right_perm.view(B*H, N, N)
        out_flat   = torch.bmm(left_flat, right_flat)   # fp16, Tensor‑Core
   Convert result to fp32.
7. Reshape to [B, N, N, H] and apply a second LayerNorm.
8. Multiply by out_gate (cast to fp32) and project back to the original dimension C
   with a final linear layer (fp16 → fp32).

The Triton kernel is only used for the element‑wise gating (step 5), satisfying
the requirement that part of the computation runs on Triton while the heavy
batched matmul leverages cuBLAS for optimal throughput on the B200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_apply_kernel(
    left_ptr,                # fp16 *
    right_ptr,               # fp16 *
    left_gate_ptr,           # fp16 *
    right_gate_ptr,          # fp16 *
    BLOCK_SIZE: tl.constexpr,
    numel: tl.int64,
):
    """In‑place multiplication:
    left  = left  * left_gate
    right = right * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel

    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # store in‑place
    tl.store(left_ptr + offs, left * lgate, mask=mask)
    tl.store(right_ptr + offs, right * rgate, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, C]   (float32)
        - mask         : torch.Tensor[B, N, N] (bool/float) or None
        - weights      : dict of model weight tensors (float32)
        - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (fp32) → fp16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                # [B, N, N, C]  fp32
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (fp16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)    # [B, N, N, H]
    right = F.linear(x, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Compute gates (fp16) – sigmoid
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, H] fp16
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # used later

    # ------------------------------------------------------------------
    # 5. Fuse element‑wise gate multiplication with Triton
    # ------------------------------------------------------------------
    numel = left.numel()
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(numel, BLOCK_SIZE),)

    _gate_apply_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        BLOCK_SIZE=BLOCK_SIZE,
        numel=numel,
        num_warps=4,
    )
    # now `left` and `right` are already multiplied by their gates

    # ------------------------------------------------------------------
    # 6. Core O(N³) update via batched matrix multiplication (cuBLAS)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, H

    # reshape to (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # transpose right on the last two dims -> (B, H, N, N) where dims are (k, j)
    right_T = right.permute(0, 1, 3, 2).contiguous()

    # flatten batch and hidden dimensions
    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)      # (BH, N, N) fp16
    right_flat = right_T.view(BH, N, N)  # (BH, N, N) fp16

    # batched matmul, result is fp16; we cast to fp32 afterwards
    out_flat = torch.bmm(left_flat, right_flat)   # (BH, N, N) fp16
    out_flat = out_flat.to(torch.float32)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) fp32

    # ------------------------------------------------------------------
    # 7. Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # fp32

    # ------------------------------------------------------------------
    # 8. Apply out_gate (fp32) and final projection back to C
    # ------------------------------------------------------------------
    out = out * out_gate.to(out.dtype)   # out_gate was fp16 → cast to fp32

    to_out_w = weights["to_out.weight"].to(torch.float16)  # (C, H)
    out_fp16 = out.to(torch.float16)
    output = F.linear(out_fp16, to_out_w)   # (B,N,N,C) fp16
    output = output.to(torch.float32)

    return output
