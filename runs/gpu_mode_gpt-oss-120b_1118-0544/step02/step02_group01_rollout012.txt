def fused_gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                           B, N, H,
                           stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
                           stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
                           stride_mask_b, stride_mask_i, stride_mask_j,
                           stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                           BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offset < total
    # decode index similar to previous code
    tmp = offset
    b = tmp // (N*N*H)
    tmp = tmp % (N*N*H)
    i = tmp // (N*H)
    tmp = tmp % (N*H)
    j = tmp // H
    h = tmp % H

    proj = tl.load(proj_ptr + b*stride_proj_b + i*stride_proj_i + j*stride_prog_j + h*stride_proj_h, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + b*stride_gate_b + i*stride_gate_i + j*stride_gate_j + h*stride_gate_h, mask=mask, other=0.0)

"""
TriMul (outgoing) forward pass implemented with Triton for elementwise
gating/masking and final out‑gate multiplication.  The heavy N³ reduction is
performed with a batched half‑precision GEMM (torch.bmm).  The pipeline is:

1. Input LayerNorm (torch.nn.functional.layer_norm)
2. Linear projections (left/right) and gate pre‑activations (torch.nn.functional.linear)
3. Triton kernel that fuses:
      left = left_proj * left_gate * mask
      right = right_proj * right_gate * mask
4. Half‑precision batched matrix multiplication:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
5. Output LayerNorm (torch.nn.functional.layer_norm)
6. Triton kernel that multiplies the normalized tensor by the out‑gate.
7. Final linear projection to the original dimension.

All tensor shapes are preserved; the final output is float32
[batch, seq_len, seq_len, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_lr_gate_mask_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, H,
    # strides for left_proj
    lp_b, lp_i, lp_j, lp_h,
    # strides for left_gate
    lg_b, lg_i, lg_j, lg_h,
    # strides for right_proj
    rp_b, rp_i, rp_j, rp_h,
    # strides for right_gate
    rg_b, rg_i, rg_j, rg_h,
    # strides for mask (no channel dim)
    m_b, m_i, m_j,
    # strides for outputs
    lo_b, lo_i, lo_j, lo_h,
    ro_b, ro_i, ro_j, ro_h,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # decode linear index -> (b,i,j,h)
    tmp = offs
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # compute flat offsets
    off_lp = b * lp_b + i * lp_i + j * lp_j + h * lp_h
    off_lg = b * lg_b + i * lg_i + j * lg_j + h * lg_h
    off_rp = b * rp_b + i * rp_i + j * rp_j + h * rp_h
    off_rg = b * rg_b + i * rg_i + j * rg_j + h * rg_h
    off_m  = b * m_b + i * m_i + j * m_j
    off_lo = b * lo_b + i * lo_i + j * lo_j + h * lo_h
    off_ro = b * ro_b + i * ro_i + j * ro_j + h * ro_h

    lp = tl.load(left_proj_ptr + off_lp, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + off_lg, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + off_rp, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + off_rg, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + off_m, mask=mask, other=1.0)

    tl.store(left_out_ptr + off_lo, lp * lg * m, mask=mask)
    tl.store(right_out_ptr + off_ro, rp * rg * m, mask=mask)


@triton.jit
def mul_outgate_kernel(
    out_ptr, gate_ptr,
    B, N, H,
    out_b, out_i, out_j, out_h,
    gate_b, gate_i, gate_j, gate_h,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # decode index
    tmp = offs
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    off_out = b * out_b + i * out_i + j * out_j + h * out_h
    off_gate = b * gate_b + i * gate_i + j * gate_j + h * gate_h

    val = tl.load(out_ptr + off_out, mask=mask, other=0.0)
    g   = tl.load(gate_ptr + off_gate, mask=mask, other=0.0)

    tl.store(out_ptr + off_out, val * g, mask=mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing Triangle Multiplicative Update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model weights
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, cfg = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = cfg["hidden_dim"]

    # ---------------------------------------------------------------
    # 1) Input LayerNorm
    # ---------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------------------------------------------------------------
    # 2) Linear projections and gate pre‑activations (no bias)
    # ---------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden]
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate_pre = F.linear(x, weights["left_gate.weight"])
    right_gate_pre = F.linear(x, weights["right_gate.weight"])
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ---------------------------------------------------------------
    # 3) Fuse mask and gating via a Triton kernel
    # ---------------------------------------------------------------
    if mask is None:
        mask_tensor = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask_tensor = mask.to(dtype)

    # ensure contiguous tensors for stride simplicity
    left_proj = left_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_proj = right_proj.contiguous()
    right_gate = right_gate.contiguous()
    mask_tensor = mask_tensor.contiguous()

    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # strides in elements
    lp_s = left_proj.stride()
    lg_s = left_gate.stride()
    rp_s = right_proj.stride()
    rg_s = right_gate.stride()
    m_s = mask_tensor.stride()
    lo_s = left.stride()
    ro_s = right.stride()

    total = B * N * N * hidden_dim
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    fused_lr_gate_mask_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask_tensor,
        left,
        right,
        B, N, hidden_dim,
        lp_s[0], lp_s[1], lp_s[2], lp_s[3],
        lg_s[0], lg_s[1], lg_s[2], lg_s[3],
        rp_s[0], rp_s[1], rp_s[2], rp_s[3],
        rg_s[0], rg_s[1], rg_s[2], rg_s[3],
        m_s[0], m_s[1], m_s[2],
        lo_s[0], lo_s[1], lo_s[2], lo_s[3],
        ro_s[0], ro_s[1], ro_s[2], ro_s[3],
        BLOCK=BLOCK,
    )

    # ---------------------------------------------------------------
    # 4) Batched half‑precision matrix multiplication (the N³ part)
    # ---------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # right needs to be transposed on the pairwise axes (j <-> k)
    right_t = right.permute(0, 2, 1, 3).contiguous()  # [B, N, N, H] where dims are (b, k, j, h)

    # reshape to (B*H, N, N) for batched GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (b, h, i, k)
    right_mat = right_t.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)  # (b, h, k, j)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), fp16

    out_hidden = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, H]

    # ---------------------------------------------------------------
    # 5) Output LayerNorm
    # ---------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ---------------------------------------------------------------
    # 6) Multiply by out_gate using a Triton kernel
    # ---------------------------------------------------------------
    # out_gate already has shape [B, N, N, H] and is float32
    total2 = B * N * N * hidden_dim
    grid2 = ((total2 + BLOCK - 1) // BLOCK,)

    mul_outgate_kernel[grid2](
        out_hidden,
        out_gate,
        B, N, hidden_dim,
        out_hidden.stride(0), out_hidden.stride(1), out_hidden.stride(2), out_hidden.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        BLOCK=BLOCK,
    )

    # ---------------------------------------------------------------
    # 7) Final linear projection back to original dimension
    # ---------------------------------------------------------------
    out = F.linear(out_hidden, weights["to_out.weight"])  # [B, N, N, dim]
    return out
