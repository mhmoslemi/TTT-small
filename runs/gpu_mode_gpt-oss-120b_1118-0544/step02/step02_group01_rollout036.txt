import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# fused kernel
@triton.jit
def fuse_scale_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                           B, N, H,
                           stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
                           stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
                           stride_mask_b, stride_mask_i, stride_mask_j,
                           stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # decode linear idx
    idx = offs
    b = idx // (N * N * H)
    idx = idx % (N * N * H)
    i = idx // (N * H)
    idx = idx % (N * H)
    j = idx // H
    h = idx % H

    # compute offsets
    proj_off = b * stride_proj_b + i * stride_proj_i + j * stride_proj_j + h * stride_proj_h
    gate_off = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_off = b * stride_out_b + i * stride_out_i + j * stride_out_j + h * stride_out_h

    proj = tl.load(proj_ptr + proj_off, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)
    out = (proj * gate * m).to(tl.float16)
    tl.store(out_ptr + out_off, out, mask=mask)

# batched GEMM kernel
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute 3D grid indices
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    total_blocks = batch * num_m * num_n
    # pid ranges over total_blocks
    b = pid // (num_m * num_n)
    tmp = pid % (num_m * num_n)
    pid_m = tmp // num_n
    pid_n = tmp % num_n

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid m,n
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this batch
    a_ptrs = a_ptr + b * stride_a_batch
    b_ptrs = b_ptr + b * stride_b_batch
    c_ptrs = c_ptr + b * stride_c_batch

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(a_ptrs + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k),
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n),
                    mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

        # update pointers for next k block
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store result
    c = acc.to(tl.float16)
    tl.store(c_ptrs + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n),
             c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing Triangle Multiplicative Update
    (TriMul) operator.  The implementation:
      * applies LayerNorm in PyTorch,
      * projects inputs to a hidden dimension,
      * fuses mask, gating and casting to half‑precision with a tiny Triton kernel,
      * executes the N³ reduction via a custom batched GEMM kernel (half‑precision),
      * finishes with LayerNorm, output gating and a final linear projection.
    All tensors are expected to reside on the CUDA device.
    """
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # -----------------------------------------------------------------
    # 2) Linear projections (float32) and gate pre‑computations
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 3) Fuse mask + gate + cast to fp16 (Triton)
    # -----------------------------------------------------------------
    # Ensure tensors are contiguous for stride‑based addressing
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()    # stays float32 for later

    if mask is None:
        mask_f = torch.ones((B, N, N), dtype=x.dtype, device=x.device)
    else:
        mask_f = mask.to(dtype=x.dtype)

    mask_f = mask_f.contiguous()

    # allocate output buffers (fp16)
    left = torch.empty_like(left_proj, dtype=torch.float16, device=x.device)
    right = torch.empty_like(right_proj, dtype=torch.float16, device=x.device)

    # strides in element units
    s_lp_b, s_lp_i, s_lp_j, s_lp_h = left_proj.stride()
    s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate.stride()
    s_rp_b, s_rp_i, s_rp_j, s_rp_h = right_proj.stride()
    s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate.stride()
    s_m_b, s_m_i, s_m_j = mask_f.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = left.stride()

    total = B * N * N * hidden_dim
    BLOCK_SIZE = 1024
    grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    # left side fused kernel
    fuse_scale_mask_kernel[grid](
        left_proj,
        left_gate,
        mask_f,
        left,
        B, N, hidden_dim,
        s_lp_b, s_lp_i, s_lp_j, s_lp_h,
        s_lg_b, s_lg_i, s_lg_j, s_lg_h,
        s_m_b, s_m_i, s_m_j,
        s_out_b, s_out_i, s_out_j, s_out_h,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # right side fused kernel
    fuse_scale_mask_kernel[grid](
        right_proj,
        right_gate,
        mask_f,
        right,
        B, N, hidden_dim,
        s_rp_b, s_rp_i, s_rp_j, s_rp_h,
        s_rg_b, s_rg_i, s_rg_j, s_rg_h,
        s_m_b, s_m_i, s_m_j,
        s_out_b, s_out_i, s_out_j, s_out_h,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # -----------------------------------------------------------------
    # 4) Batched GEMM (N³ reduction) in fp16 via Triton
    # -----------------------------------------------------------------
    # permute to (B, hidden, N, N) where left = (i,k) and right = (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (k,j)

    batch = B * hidden_dim
    # view as (batch, N, N)
    a = left_perm.view(batch, N, N)
    b = right_perm.view(batch, N, N)
    c = torch.empty_like(a, dtype=torch.float16)

    # strides in elements
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    grid_gemm = (batch * num_m * num_n,)

    batched_gemm_kernel[grid_gemm](
        a, b, c,
        batch, N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden)
    out_hidden = c.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # -----------------------------------------------------------------
    # 5) Output LayerNorm, output gate, final linear
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(out_hidden,
                              (hidden_dim,),
                              weight=weights["to_out_norm.weight"],
                              bias=weights["to_out_norm.bias"])
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])
    return out

"""
Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update
(TriMul) as used in AlphaFold‑3.  The implementation:
  1. Layer‑norms and linear projections are done in PyTorch (float32).
  2. Masking, gating and casting to fp16 are fused in a tiny Triton kernel.
  3. The N³ reduction (∑ₖ left[b,i,k,d]·right[b,j,k,d]) is performed by a
     custom batched GEMM kernel written in Triton (fp16 × fp16 → fp32).
  4. The result is layer‑normed, gated and finally projected back to the
     original channel dimension.
All tensors are expected to reside on the CUDA device.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Element‑wise kernel: proj * gate * mask  →  fp16
# ----------------------------------------------------------------------
@triton.jit
def fuse_scale_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    valid = offs < total

    # ---- decode linear index to (b,i,j,h) ----
    idx = offs
    b = idx // (N * N * H)
    idx = idx % (N * N * H)
    i = idx // (N * H)
    idx = idx % (N * H)
    j = idx // H
    h = idx % H

    # ---- compute byte offsets (in elements) ----
    proj_off = b * stride_proj_b + i * stride_proj_i + j * stride_proj_j + h * stride_proj_h
    gate_off = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_off  = b * stride_out_b  + i * stride_out_i  + j * stride_out_j  + h * stride_out_h

    proj = tl.load(proj_ptr + proj_off, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_off, mask=valid, other=1.0)   # 1.0 → no‑mask case
    out = (proj * gate * m).to(tl.float16)
    tl.store(out_ptr + out_off, out, mask=valid)


# ----------------------------------------------------------------------
# 2) Batched GEMM kernel: (B*H, N, N) × (B*H, N, N)ᵀ  →  (B*H, N, N)
#    a is (M=N, K=N)   b is (K=N, N=N)   c is (M=N, N=N)
#    accumulation is performed in fp32, result stored as fp32.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ---- decode grid indices ----
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    b = pid // (num_m * num_n)                     # which batch element
    pid_rem = pid % (num_m * num_n)
    pid_m = pid_rem // num_n
    pid_n = pid_rem % num_n

    # ---- block offsets ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch_ptr = a_ptr + b * stride_a_batch
    b_batch_ptr = b_ptr + b * stride_b_batch
    c_batch_ptr = c_ptr + b * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K dimension ----
    for kp in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kp * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M×K)
        a = tl.load(
            a_batch_ptr
            + (offs_m[:, None] * stride_a_m)
            + (offs_k[None, :] * stride_a_k),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile (K×N)
        b = tl.load(
            b_batch_ptr
            + (offs_k[:, None] * stride_b_k)
            + (offs_n[None, :] * stride_b_n),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot‑product
        acc += tl.dot(a, b)

        # advance pointers for next k‑block
        a_batch_ptr += BLOCK_K * stride_a_k
        b_batch_ptr += BLOCK_K * stride_b_k

    # store result (fp32)
    tl.store(
        c_batch_ptr
        + (offs_m[:, None] * stride_c_m)
        + (offs_n[None, :] * stride_c_n),
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# 3) Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of outgoing TriMul.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict with the module's parameters
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections + gate pre‑computations (float32)
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 3) Fuse mask, gate and cast to fp16 (Triton)
    # -----------------------------------------------------------------
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()          # stays float32 for later use

    if mask is None:
        mask_f = torch.ones((B, N, N), dtype=x.dtype, device=x.device)
    else:
        mask_f = mask.to(dtype=x.dtype, device=x.device)

    mask_f = mask_f.contiguous()

    left = torch.empty_like(left_proj, dtype=torch.float16, device=x.device)
    right = torch.empty_like(right_proj, dtype=torch.float16, device=x.device)

    # strides (in elements)
    s_lp_b, s_lp_i, s_lp_j, s_lp_h = left_proj.stride()
    s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate.stride()
    s_rp_b, s_rp_i, s_rp_j, s_rp_h = right_proj.stride()
    s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate.stride()
    s_m_b, s_m_i, s_m_j = mask_f.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = left.stride()

    total = B * N * N * hidden_dim
    BLOCK_ELEM = 1024
    grid = ((total + BLOCK_ELEM - 1) // BLOCK_ELEM,)

    # left side
    fuse_scale_mask_kernel[grid](
        left_proj,
        left_gate,
        mask_f,
        left,
        B,
        N,
        hidden_dim,
        s_lp_b,
        s_lp_i,
        s_lp_j,
        s_lp_h,
        s_lg_b,
        s_lg_i,
        s_lg_j,
        s_lg_h,
        s_m_b,
        s_m_i,
        s_m_j,
        s_out_b,
        s_out_i,
        s_out_j,
        s_out_h,
        BLOCK_SIZE=BLOCK_ELEM,
    )
    # right side
    fuse_scale_mask_kernel[grid](
        right_proj,
        right_gate,
        mask_f,
        right,
        B,
        N,
        hidden_dim,
        s_rp_b,
        s_rp_i,
        s_rp_j,
        s_rp_h,
        s_rg_b,
        s_rg_i,
        s_rg_j,
        s_rg_h,
        s_m_b,
        s_m_i,
        s_m_j,
        s_out_b,
        s_out_i,
        s_out_j,
        s_out_h,
        BLOCK_SIZE=BLOCK_ELEM,
    )

    # -----------------------------------------------------------------
    # 4) Batched GEMM (N³ reduction) in fp16 → fp32
    # -----------------------------------------------------------------
    # left: (i,k), right: (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (k,j)

    batch = B * hidden_dim
    a = left_perm.view(batch, N, N)   # (batch, M=N, K=N)  fp16
    b = right_perm.view(batch, N, N)  # (batch, K=N, N=N)  fp16
    c = torch.empty((batch, N, N), dtype=torch.float32, device=x.device)

    # strides (elements)
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch * grid_m * grid_n,)

    batched_gemm_kernel[grid](
        a,
        b,
        c,
        batch,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden]
    out_hidden = c.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # -----------------------------------------------------------------
    # 5) Output LayerNorm, output gate and final projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])
    return out
