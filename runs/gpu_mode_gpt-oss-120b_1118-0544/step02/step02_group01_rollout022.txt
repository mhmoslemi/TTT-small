"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the reference PyTorch module but:
* LayerNorm and Linear projections are performed with torch.nn.functional (highly
  optimized GEMM kernels).
* Masking and gating are fused in a single Triton kernel that writes both the
  left and right tensors, reducing kernel launch overhead.
* The core N³ reduction is realized as a batched half‑precision matrix
  multiplication (left @ rightᵀ) via torch.bmm, which maps to cuBLAS.
All tensors are kept on the GPU; the final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_mask_gate_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, H,
    # strides for left_proj
    lp_stride_b, lp_stride_i, lp_stride_j, lp_stride_h,
    # strides for left_gate
    lg_stride_b, lg_stride_i, lg_stride_j, lg_stride_h,
    # strides for right_proj
    rp_stride_b, rp_stride_i, rp_stride_j, rp_stride_h,
    # strides for right_gate
    rg_stride_b, rg_stride_i, rg_stride_j, rg_stride_h,
    # strides for mask (no H dim)
    m_stride_b, m_stride_i, m_stride_j,
    # strides for left_out
    lo_stride_b, lo_stride_i, lo_stride_j, lo_stride_h,
    # strides for right_out
    ro_stride_b, ro_stride_i, ro_stride_j, ro_stride_h,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Elementwise fusion:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
    Operates on the flattened view of [B, N, N, H].
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    total = B * N * N * H
    mask = offs < total

    # Decode linear index -> (b, i, j, h)
    tmp = offs
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # Compute flat offsets for each tensor
    lp_off = b * lp_stride_b + i * lp_stride_i + j * lp_stride_j + h * lp_stride_h
    lg_off = b * lg_stride_b + i * lg_stride_i + j * lg_stride_j + h * lg_stride_h
    rp_off = b * rp_stride_b + i * rp_stride_i + j * rp_stride_j + h * rp_stride_h
    rg_off = b * rg_stride_b + i * rg_stride_i + j * rg_stride_j + h * rg_stride_h
    m_off  = b * m_stride_b + i * m_stride_i + j * m_stride_j
    lo_off = b * lo_stride_b + i * lo_stride_i + j * lo_stride_j + h * lo_stride_h
    ro_off = b * ro_stride_b + i * ro_stride_i + j * ro_stride_j + h * ro_stride_h

    # Load values (mask broadcasted over H)
    left_proj = tl.load(left_proj_ptr + lp_off, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + lg_off, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + rp_off, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + rg_off, mask=mask, other=0.0)
    m = tl.load(mask_ptr + m_off, mask=mask, other=0.0)

    # Apply gating and mask
    left_val = left_proj * left_gate * m
    right_val = right_proj * right_gate * m

    tl.store(left_out_ptr + lo_off, left_val, mask=mask)
    tl.store(right_out_ptr + ro_off, right_val, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict mapping weight names to tensors
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # -----------------------------------------------------------------
    # 2) Linear projections
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 3) Fuse mask and gating (if mask is provided)
    # -----------------------------------------------------------------
    if mask is not None:
        # Ensure mask dtype matches the computation dtype (float32) and is contiguous
        mask_f = mask.to(x.dtype).contiguous()

        # Contiguify all tensors to guarantee simple stride patterns
        left_proj_c = left_proj.contiguous()
        left_gate_c = left_gate.contiguous()
        right_proj_c = right_proj.contiguous()
        right_gate_c = right_gate.contiguous()
        mask_c = mask_f.contiguous()

        # Allocate output buffers
        left = torch.empty_like(left_proj_c)
        right = torch.empty_like(right_proj_c)

        # Gather strides (in elements)
        s_lp_b, s_lp_i, s_lp_j, s_lp_h = left_proj_c.stride()
        s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate_c.stride()
        s_rp_b, s_rp_i, s_rp_j, s_rp_h = right_proj_c.stride()
        s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate_c.stride()
        s_m_b, s_m_i, s_m_j = mask_c.stride()
        s_lo_b, s_lo_i, s_lo_j, s_lo_h = left.stride()
        s_ro_b, s_ro_i, s_ro_j, s_ro_h = right.stride()

        total = B * N * N * hidden_dim
        # A block size of 32768 yields ~2 KB per thread block and maps well to
        # the typical GPU occupancy on B200.
        BLOCK_SIZE = 32768
        grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE,)

        fused_mask_gate_kernel[grid](
            left_proj_c, left_gate_c,
            right_proj_c, right_gate_c,
            mask_c,
            left, right,
            B, N, hidden_dim,
            s_lp_b, s_lp_i, s_lp_j, s_lp_h,
            s_lg_b, s_lg_i, s_lg_j, s_lg_h,
            s_rp_b, s_rp_i, s_rp_j, s_rp_h,
            s_rg_b, s_rg_i, s_rg_j, s_rg_h,
            s_m_b, s_m_i, s_m_j,
            s_lo_b, s_lo_i, s_lo_j, s_lo_h,
            s_ro_b, s_ro_i, s_ro_j, s_ro_h,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 4) Core TriMul: batched matrix multiplication (FP16)
    # -----------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape to (B*hidden, N, N) and prepare right as transposed
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 2, 1).contiguous()  # [B, hidden, N, N] -> transposed on last two dims

    left_mat = left.view(-1, N, N)    # (B*hidden, N, N)
    right_mat = right.view(-1, N, N)  # (B*hidden, N, N)

    out_mat = torch.bmm(left_mat, right_mat)   # (B*hidden, N, N), FP16

    # reshape back to [B, N, N, hidden]
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # -----------------------------------------------------------------
    # 5) Output LayerNorm, gate, and final linear projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
```
