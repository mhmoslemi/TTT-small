    device = input_tensor.device
    dtype = input_tensor.dtype

    hidden_dim = config["hidden_dim"]

    # LayerNorm input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Cast to fp16
    x_fp16 = x.to(torch.float16)

    # Linear projections
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # Gates
    left_gate = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)))

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()

    # Apply mask and gates
    if not nomask and mask is not None:

        mask_fp16 = mask.to(torch.float16).contiguous()
        # Ensure left and right contiguous
        apply_gate_mask_inplace(left, left_gate, mask_fp16, hidden_dim)
        apply_gate_mask_inplace(right, right_gate, mask_fp16, hidden_dim)
    else:

        left = left * left_gate
        right = right * right_gate

    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_T = right.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N) (transpose)

@triton.jit
def gated_mul_mask_kernel(
    out_ptr, in_ptr, gate_ptr, mask_ptr,
    total_elements,
    H: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements
    x = tl.load(in_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    out = x * g * m
    tl.store(out_ptr + offs, out, mask=mask)

"""
TriMul (outgoing) forward pass optimized with Triton.

Algorithm
---------
1. Layer‑norm the input tensor `x` ∈ ℝ^{B×N×N×C}.
2. Cast to fp16 and apply two linear projections (left / right) to hidden size H.
3. Compute three sigmoid gates (left, right, out) from the same normalized input.
4. Fuse mask (if present) with the left/right gates via a lightweight Triton kernel:
      left  = left  * left_gate  * mask
      right = right * right_gate * mask
   (when `nomask=True` the mask step is skipped and only left/right *= gate).
5. Core update: for each hidden slice h,
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication
      out = left_perm @ right_permᵀ
   where left_perm  ∈ ℝ^{BH×N×N} and right_permᵀ ∈ ℝ^{BH×N×N}.
   The multiplication uses torch.bmm (fp16 Tensor‑Core accelerated).
6. Layer‑norm the hidden result, multiply by the out‑gate, and project back to
   the original dimension C.
7. Return the final tensor of shape [B, N, N, C] (float32).

Only the element‑wise gating+mask fusion is performed with a custom Triton
kernel; the heavy O(N³) core is delegated to cuBLAS via torch.bmm for
maximum performance on the B200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def gated_mul_mask_kernel(
    out_ptr,               # fp16 *
    in_ptr,                # fp16 *
    gate_ptr,              # fp16 *
    mask_ptr,              # fp16 *
    total_elements: tl.int64,
    H: tl.constexpr,       # hidden dimension (compile‑time constant)
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fused multiplication: out = in * gate * mask (mask broadcast over H)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # load tensors
    x = tl.load(in_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # mask is stored without the hidden dimension; broadcast by integer division
    mask_offs = offs // H
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    out = x * g * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_mask_inplace(tensor, gate, mask, hidden_dim):
    """Apply `tensor = tensor * gate * mask` in‑place using the Triton kernel."""
    total = tensor.numel()
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total, BLOCK_SIZE),)
    gated_mul_mask_kernel[grid](
        tensor,            # out_ptr = in‑place overwrite
        tensor,            # in_ptr
        gate,              # gate_ptr
        mask,              # mask_ptr (flattened B×N×N)
        total,
        H=hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=8,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]   (float32)
        - mask         : torch.Tensor [B, N, N]     (bool/float, optional)
        - weights      : dict of model tensors
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C], float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)   # [B, N, N, H], fp16
    right = F.linear(x, right_proj_w)  # [B, N, N, H], fp16

    # ------------------------------------------------------------------
    # 4. Gates (fp16)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Make everything contiguous before the Triton kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()

    # ------------------------------------------------------------------
    # 5. Optional mask + gate fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> fp16, flatten
        mask_fp16 = mask.to(torch.float16).contiguous()
        # Apply mask & gates in‑place for left and right
        _apply_gate_mask_inplace(left, left_gate, mask_fp16, hidden_dim)
        _apply_gate_mask_inplace(right, right_gate, mask_fp16, hidden_dim)
    else:
        # No mask → simple element‑wise multiplication (fast PyTorch)
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core O(N³) update via batched matrix multiplication (fp16)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # (B, N, N, H)

    # reshape to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # right matrix needs to be transposed on the last two dims (k ↔ j)
    right_T = right_perm.permute(0, 1, 3, 2).contiguous()  # (B, H, N, N)

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)        # (BH, N, N)
    right_flat = right_T.view(BH, N, N)        # (BH, N, N)

    # Batched fp16 GEMM (Tensor‑Core accelerated)
    out_flat = torch.bmm(left_flat, right_flat)   # (BH, N, N), fp16

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 7. Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.float()
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # float32

    # ------------------------------------------------------------------
    # 8. Apply out‑gate (float32) and final projection (fp16)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.float()

    to_out_w = weights["to_out.weight"].to(torch.float16)   # (C, H)
    out_fp16 = out_norm.to(torch.float16)
    output_fp16 = F.linear(out_fp16, to_out_w)            # (B, N, N, C), fp16

    # Return float32 tensor as required
    return output_fp16.to(torch.float32)
