"""
Outgoing Triangle Multiplicative Update (TriMul) – optimized forward pass.

Algorithm
---------
1. Layer‑norm the input tensor `x` on the last channel dimension.
2. Linear projections `left_proj`, `right_proj` (bias‑free) produce tensors of
   shape [B, N, N, H].
3. Optional binary mask is applied (broadcast on H).
4. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) from `x`.
5. Apply the left/right gates and cast the gated tensors to FP16.
6. Prepare `right` in transposed (k, j) layout, then flatten the batch
   and hidden dimensions:  (B·H, N, N) for both `left` and `right_T`.
7. Run a custom Triton kernel that performs a batched GEMM:
       out[b] = left[b] @ right_T[b]          (FP16 × FP16 → FP32)
   The kernel tiles the matrices (64×64) and accumulates in FP32.
8. Reshape the result back to [B, N, N, H] (float32).
9. Apply a second LayerNorm over the hidden dimension, multiply by
   `out_gate`, and finally linearly project back to the original `dim`.

The Triton kernel eliminates the intermediate FP16‑>FP32 cast used in the
reference implementation and performs the heavy N³ reduction directly on
the GPU, yielding a substantial speed‑up.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_fp16_fp32_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_A_batch, stride_A_row, stride_A_col,
    stride_B_batch, stride_B_row, stride_B_col,
    stride_C_batch, stride_C_row, stride_C_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_WARPS: tl.constexpr,
):
    """
    Batched GEMM kernel:
        C[b, i, j] = Σ_k A[b, i, k] * B[b, k, j]
    where A, B are FP16 and C is FP32.
    """
    pid = tl.program_id(0)          # combines batch * tile_m
    pid_n = tl.program_id(1)        # tile_n

    # Number of tiles along the M dimension
    tiles_m = tl.cdiv(M, BLOCK_M)

    batch_id = pid // tiles_m
    tile_m = pid % tiles_m
    tile_n = pid_n

    # Offsets for the current block
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- Load A block (M × K) ----
        a_ptrs = (
            A_ptr
            + batch_id * stride_A_batch
            + offs_m[:, None] * stride_A_row
            + offs_k[None, :] * stride_A_col
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ---- Load B block (K × N) ----
        b_ptrs = (
            B_ptr
            + batch_id * stride_B_batch
            + offs_k[:, None] * stride_B_row
            + offs_n[None, :] * stride_B_col
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # ---- Matrix multiply‑accumulate ----
        acc += tl.dot(a, b)

    # ---- Write back C block (M × N) ----
    c_ptrs = (
        C_ptr
        + batch_id * stride_C_batch
        + offs_m[:, None] * stride_C_row
        + offs_n[None, :] * stride_C_col
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float) or None
            - weights      : dict with all linear / layer‑norm parameters
            - config       : dict with at least:
                                * "dim" (int)
                                * "hidden_dim" (int)
                                * "nomask" (bool, optional)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) Extract model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]  (FP32)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # used later

    # ------------------------------------------------------------------
    # 5) Apply gates and cast to FP16 for the heavy reduction
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 6) Prepare tensors for the batched GEMM
    #    left:  (B, H, i, k)
    #    right_T: (B, H, k, j)  <-- transposed on the two N dimensions
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()       # (B, H, k, j)

    # Flatten batch and hidden dimension → (B·H, N, N)
    left_flat = left_perm.view(-1, N, N)      # dtype=fp16
    right_flat = right_perm.view(-1, N, N)    # dtype=fp16

    batch_h = left_flat.shape[0]              # B * hidden_dim

    # Output buffer (FP32)
    out_fp32_flat = torch.empty(
        (batch_h, N, N), dtype=torch.float32, device=device
    )

    # ------------------------------------------------------------------
    # 7) Triton batched GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    num_warps = 8

    grid_x = batch_h * ((N + BLOCK_M - 1) // BLOCK_M)
    grid_y = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_x, grid_y)

    _batched_gemm_fp16_fp32_kernel[
        grid,
        num_warps=num_warps,
    ](
        A_ptr=left_flat,
        B_ptr=right_flat,
        C_ptr=out_fp32_flat,
        M=N,
        N=N,
        K=N,
        stride_A_batch=left_flat.stride(0),
        stride_A_row=left_flat.stride(1),
        stride_A_col=left_flat.stride(2),
        stride_B_batch=right_flat.stride(0),
        stride_B_row=right_flat.stride(1),
        stride_B_col=right_flat.stride(2),
        stride_C_batch=out_fp32_flat.stride(0),
        stride_C_row=out_fp32_flat.stride(1),
        stride_C_col=out_fp32_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_WARPS=num_warps,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to [B, N, N, hidden_dim] (float32)
    # ------------------------------------------------------------------
    out = out_fp32_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Second LayerNorm, output gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # apply the output gate
    out = F.linear(out, to_out_weight)       # [B, N, N, dim]

    return out
