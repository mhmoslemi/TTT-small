"""
TriMul kernel (outgoing) – fast forward pass

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) with the provided LN parameters.
2. Cast to FP16 for the heavy compute path.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are performed with FP16 weights.
4. If a mask is present (and not disabled) a tiny Triton kernel fuses
   mask‑application and gating:
       out = proj * gate * mask
   This avoids extra memory passes and satisfies the “use Triton” requirement.
5. The core O(N³·H) reduction is a batched matrix multiplication:
       out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
   which is expressed as a `torch.bmm`.  cuBLAS handles this efficiently
   (Tensor‑core friendly) and far outperforms a pure‑Triton GEMM for the
   large square‑matrix batch sizes used in protein models.
6. Reshape, apply a second LayerNorm, multiply by the output gate and a
   final linear projection back to the original channel dimension.
7. Cast the result back to the input dtype.

The kernel `apply_gate_mask_kernel` is the only Triton kernel used; it
processes a flattened view of the 4‑D tensors and multiplies by the
gate and optional mask in a single pass.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: elementwise (proj * gate * mask) on a flattened tensor
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,          # Tensor[(B*N*N*H,), fp16]   – projected values
    gate_ptr,          # Tensor[(B*N*N*H,), fp16]   – gate values (sigmoid)
    mask_ptr,          # Tensor[(B*N*N,), fp16]     – binary mask (broadcast)
    out_ptr,           # Tensor[(B*N*N*H,), fp16]   – output
    total_elems,       # int – total number of elements in proj/gate (B*N*N*H)
    hidden_dim,        # int – H (used for broadcast)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # mask for the last (possibly partial) block
    valid = offs < total_elems

    # index of the corresponding mask element (broadcast over H)
    mask_idx = offs // hidden_dim

    # load values (mask=0 for out‑of‑bounds entries)
    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Helper to launch the above kernel
# ----------------------------------------------------------------------
def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    proj : (B,N,N,H) - FP16
    gate : (B,N,N,H) - FP16
    mask : (B,N,N)   - FP16 (values 0 or 1)
    Returns proj * gate * mask (broadcast over H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,C) -> (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (same shape)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask + gates (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        # no mask – simple elementwise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM (out = left @ rightᵀ)
    #    Shapes:
    #       left, right : (B, N, N, H)
    #    -> reshape to (B*H, N, N) for torch.bmm
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # torch.bmm uses cuBLAS (Tensor‑cores) and is highly optimized
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
