@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # each program is block of output (i_tile, j_tile) for a specific batch*hidden idx
    # we handle combined batch*hidden index
    pid = tl.program_id(0) # combined batch*hidden * (grid_m * grid_n)?? Actually we can use 2D grid: program_id(0) for block index over M dimension, program_id(1) for block over N dimension, program_id(2) for batch*hidden.
    # Use three axes: (block_m, block_n, batch_hidden)
    block_m = tl.program_id(0)
    block_n = tl.program_id(1)
    batch_hidden = tl.program_id(2)

    # decode batch and hidden from combined index
    bh = batch_hidden
    bh_div = bh // H   # batch idx
    d = bh % H

    batch = bh_div # if B <= 2

    # offsets
    i_base = block_m * BLOCK_M
    j_base = block_n * BLOCK_N

    # initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute offsets for left tile
        k_base = k * BLOCK_K
        # compute actual indexes
        i_idx = i_base + tl.arange(0, BLOCK_M)
        k_idx = k_base + tl.arange(0, BLOCK_K)
        j_idx = j_base + tl.arange(0, BLOCK_N)

        # mask for out of bounds
        mask_i = i_idx < N
        mask_j = j_idx < N
        mask_k = k_idx < N

        # compute linear offsets for loads
        # left offset: batch * stride_left_b + i_idx * stride_left_i + k_idx[:,None]*stride_left_k + d*stride_left_h
        left_ptrs = left_ptr + batch * stride_left_b + i_idx[:, None] * stride_left_i + k_idx[None, :] * stride_left_k + d * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right_T offset: batch * stride_right_b + k_idx[:, None]*stride_right_k + j_idx[None, :] * stride_right_j + d * stride_right_h
        right_ptrs = right_ptr + batch * stride_right_b + k_idx[:, None] * stride_right_k + j_idx[None, :] * stride_right_j + d * stride_right_h
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate
        acc += tl.dot(left, right)
    # apply scaling? no scaling

    # store result
    out_ptrs = out_ptr + batch * stride_out_b + i_idx[:, None] * stride_out_i + j_idx[None, :] * stride_out_j + d * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass implemented with a fused Triton matmul kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` over its channel dimension.
2. Compute left/right projections (linear, no bias) and their sigmoid gates.
3. Optionally apply the pairwise mask to the projections.
4. Multiply projections by their gates.
5. Perform the O(N³) reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication of shape (B*D, N, N) where
   D = hidden_dim.  This is realized with a custom Triton kernel that
   tiles the matrices and uses `tl.dot` (Tensor‑Core friendly) on FP16 data.
6. Apply a second LayerNorm on the hidden dimension.
7. Compute the output gate (sigmoid) and multiply element‑wise.
8. Project back to the original channel dimension.

The heavy O(N³) work is performed entirely in the Triton kernel, while the
lighter linear/LN/gate ops use highly optimised torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -----------------------------------------------------------
# Triton kernel for the batched matrix multiplication
# -----------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    left_ptr,           # *half, shape (B, N, N, H)
    right_ptr,          # *half, shape (B, N, N, H)
    out_ptr,            # *float, shape (B, N, N, H)

    B, N, H,            # dimensions

    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    pid_m = tl.program_id(0)      # block row (i)
    pid_n = tl.program_id(1)      # block col (j)
    pid_bh = tl.program_id(2)     # combined batch*hidden index

    batch = pid_bh // H          # actual batch index
    hidden = pid_bh % H           # hidden dimension index

    # Tile offsets
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the shared dimension k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # -------- left tile (i,k) --------
        left_offset = (
            batch * stride_left_b
            + i[:, None] * stride_left_i
            + k[None, :] * stride_left_k
            + hidden * stride_left_h
        )
        left = tl.load(left_ptr + left_offset,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # -------- right tile (k,j) --------   (right transposed)
        right_offset = (
            batch * stride_right_b
            + k[:, None] * stride_right_k
            + j[None, :] * stride_right_j
            + hidden * stride_right_h
        )
        right = tl.load(right_ptr + right_offset,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate dot product
        acc += tl.dot(left, right)

    # -------- store result --------
    out_offset = (
        batch * stride_out_b
        + i[:, None] * stride_out_i
        + j[None, :] * stride_out_j
        + hidden * stride_out_h
    )
    tl.store(out_ptr + out_offset,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

# -----------------------------------------------------------
# Entry point
# -----------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model weights
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # -------------------------------------------------------
    # Unpack weights
    # -------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -------------------------------------------------------
    # 1. LayerNorm over input channels
    # -------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # -------------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------------
    left = F.linear(x, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # -------------------------------------------------------
    # 3. Optional masking
    # -------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # -------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B,N,N,hidden]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))     # [B,N,N,hidden]

    # -------------------------------------------------------
    # 5. Apply gates
    # -------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------
    # 6. Cast projections to FP16 for the O(N³) reduction
    # -------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, _ = left_h.shape  # B, N, N, H

    # -------------------------------------------------------
    # 7. Triton batched matmul: out = left @ right^T (per hidden dim)
    # -------------------------------------------------------
    out_h = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides in element units (not bytes)
    l_str = left_h.stride()
    r_str = right_h.stride()
    o_str = out_h.stride()

    # Tile sizes (tuned for B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    tri_mul_matmul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        hidden_dim,
        # left strides
        l_str[0],
        l_str[1],
        l_str[2],
        l_str[3],
        # right strides (j, k correspond to dimensions 1 and 2)
        r_str[0],
        r_str[1],
        r_str[2],
        r_str[3],
        # out strides
        o_str[0],
        o_str[1],
        o_str[2],
        o_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------------
    # 8. Second LayerNorm over the hidden dimension
    # -------------------------------------------------------
    out = F.layer_norm(
        out_h,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B, N, N, hidden_dim]

    # -------------------------------------------------------
    # 9. Apply output gate (element‑wise)
    # -------------------------------------------------------
    out = out * out_gate

    # -------------------------------------------------------
    # 10. Final linear projection back to original dim
    # -------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
