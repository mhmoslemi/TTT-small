def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)
    dtype = input_tensor.dtype

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    x = x.to(torch.float16)

    # linear projections
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left = torch.nn.functional.linear(x, left_proj_w)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # mask
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # fuse multiplication left * left_gate, right * right_gate using Triton
    left = elementwise_mul(left, left_gate)
    right = elementwise_mul(right, right_gate)

    # core matmul
    B, N, _, _ = left.shape

    # reshape to (B*hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_perm = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)  # transpose inner

    # bmm
    out_flat = torch.bmm(left_perm, right_perm)  # fp16? Let it be fp16
    # ensure fp32 for later ops
    out = out_flat.to(torch.float32).reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # apply to_out_norm (layernorm over hidden_dim)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # apply out_gate
    out = out * out_gate  # out_gate currently fp16? Actually out_gate from earlier is fp16 due to x fp16, with sigmoid resulting fp16. So we need to cast out_gate to same dtype as out (float32). Cast: out_gate = out_gate.to(out.dtype)

    # final projection
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out_fp16 = out.to(torch.float16)
    output = torch.nn.functional.linear(out_fp16, to_out_w)  # shape [B,N,N,dim] fp16
    output = output.to(torch.float32)
    return output

def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE
    range = tl.arange(0, BLOCK_SIZE)
    idx = offset + range
    mask = idx < size
    a = tl.load(a_ptr + idx, mask=mask)
    b = tl.load(b_ptr + idx, mask=mask)
    out = a * b
    tl.store(out_ptr + idx, out, mask=mask)

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last channel dimension.
2. Cast to fp16 and compute left/right projections together with their
   gating linear layers in a fused matmul (weights are concatenated).
3. Apply the optional binary mask to the projections.
4. Fuse the elementwise gating (left * left_gate, right * right_gate)
   using a small Triton kernel.
5. Core O(N³) update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each (b,h) pair:
       out_{bh} = left_{bh} @ right_{bh}ᵀ
   The multiplication is performed with torch.bmm in fp32.
6. Layer‑norm the hidden result, multiply by the out‑gate and project back
   to the original channel dimension.
The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, size,
                           BLOCK_SIZE: tl.constexpr):
    """Element‑wise multiplication a * b -> out (1‑D kernel)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    out = a * b
    tl.store(out_ptr + offs, out, mask=mask)


def elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Fused element‑wise multiplication using Triton."""
    assert a.shape == b.shape, "Shapes must match for elementwise_mul"
    size = a.numel()
    out = torch.empty_like(a)

    BLOCK_SIZE = 65536  # 2^16, good trade‑off for 1‑D kernels
    grid = (triton.cdiv(size, BLOCK_SIZE),)

    elementwise_mul_kernel[grid](
        a, b, out, size,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict with model parameters
        - config       : dict, must contain "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # Unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------ #
    # 2. Cast to fp16 and compute projections + gate pre‑activations
    #    (fused by concatenating weights)
    # ------------------------------------------------------------------ #
    x_fp16 = x.to(torch.float16)

    # concatenate proj & gate weights for left & right streams
    left_weight_comb = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)
    right_weight_comb = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)

    # fused linear: first H entries are projection, next H are gate pre‑act
    left_comb = F.linear(x_fp16, left_weight_comb)   # [B, N, N, 2H] fp16
    right_comb = F.linear(x_fp16, right_weight_comb)  # [B, N, N, 2H] fp16

    left = left_comb[..., :hidden_dim]                     # projection
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])  # gate

    right = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])

    # ------------------------------------------------------------------ #
    # 3. Optional mask (applied before gating)
    # ------------------------------------------------------------------ #
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------ #
    # 4. Fuse gating (left * left_gate, right * right_gate) via Triton
    # ------------------------------------------------------------------ #
    left = elementwise_mul(left, left_gate)
    right = elementwise_mul(right, right_gate)

    # out‑gate (computed from the normalized input, no mask)
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)))

    # ------------------------------------------------------------------ #
    # 5. Core O(N³) update: batched matmul across (batch * hidden_dim)
    #    out_{bh} = left_{bh} @ right_{bh}ᵀ
    # ------------------------------------------------------------------ #
    B, N, _, _ = left.shape

    # reshape to (B*H, N, N) and transpose right on the last two axes
    left_flat = (
        left.permute(0, 3, 1, 2)          # (B, H, N, N)
        .contiguous()
        .view(B * hidden_dim, N, N)
        .to(torch.float32)               # fp32 accumulation
    )
    right_flat = (
        right.permute(0, 3, 2, 1)         # (B, H, N, N) with k <-> j swapped
        .contiguous()
        .view(B * hidden_dim, N, N)
        .to(torch.float32)
    )

    # batched matrix multiplication (fp32)
    out_flat = torch.bmm(left_flat, right_flat)      # (B*H, N, N) fp32

    # reshape back to (B, N, N, H)
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)          # (B, N, N, H)
        .contiguous()
    )

    # ------------------------------------------------------------------ #
    # 6. Hidden‑dim LayerNorm, out‑gate multiplication and final projection
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # fp32

    out = out * out_gate.to(out.dtype)  # apply out‑gate

    # final linear projection back to original dimension (fp32)
    output = F.linear(out, weights["to_out.weight"])

    return output
