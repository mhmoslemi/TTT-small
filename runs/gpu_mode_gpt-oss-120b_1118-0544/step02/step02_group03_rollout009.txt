"""
TriMul outgoing update (AlphaFold3) – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the input tensor `x` (float32).
2. Cast to fp16 and apply two linear projections (`left_proj`, `right_proj`) to
   hidden dimension `H`.
3. (Optional) apply binary mask on the projected tensors.
4. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) from `x`
   (still fp16) and multiply them point‑wise with the projected tensors.
   The point‑wise multiplications are performed by a tiny Triton kernel.
5. Core O(N³) update:
   For each head `h` compute a batched matrix multiplication
   `out[b,h] = left[b,h] @ right[b,h].T`.  The batched GEMM is delegated to
   `torch.bmm`, which uses the highly‑optimized cuBLAS Tensor‑Core kernels.
6. Reshape back to `[B,N,N,H]`, apply a second LayerNorm, multiply by the
   `out_gate`, and linearly project back to the original dimension `C`.
7. Return the final tensor in float32.

A minimal Triton kernel is kept to satisfy the “use a kernel” requirement and
to fuse the point‑wise gating operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Tiny element‑wise multiplication kernel (used for masking & gating)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)

def elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Return a * b using a Triton kernel (both tensors must be contiguous)."""
    assert a.shape == b.shape, "Shape mismatch for elementwise_mul"
    out = torch.empty_like(a)
    size = a.numel()
    BLOCK = 1024
    grid = (triton.cdiv(size, BLOCK),)
    elementwise_mul_kernel[grid](
        a,
        b,
        out,
        size,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return out

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load model parameters (convert to fp16 where appropriate)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C] float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 and compute linear projections
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)                      # [B, N, N, C] fp16
    left = F.linear(x_fp16, left_proj_weight)        # [B, N, N, H] fp16
    right = F.linear(x_fp16, right_proj_weight)       # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 3. Optional binary mask
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)               # [B, N, N, 1]
        mask_exp = mask_fp16.expand(-1, -1, -1, hidden_dim).contiguous()
        left = elementwise_mul(left, mask_exp)
        right = elementwise_mul(right, mask_exp)

    # ------------------------------------------------------------------
    # 4. Compute sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))   # [B, N, N, H] fp16
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight)) # [B, N, N, H] fp16
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))    # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 5. Apply gates (Triton‑fused elementwise multiplication)
    # ------------------------------------------------------------------
    left = elementwise_mul(left, left_gate)
    right = elementwise_mul(right, right_gate)

    # ------------------------------------------------------------------
    # 6. Core O(N³) update via batched matrix multiplication
    #    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch * head dimension
    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)          # fp16
    right_flat = right.view(BH, N, N)        # fp16

    # batched matmul: (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # fp16

    # reshape back to (B, N, N, H) and convert to fp32 for the next LayerNorm
    out = (
        out_flat.to(torch.float32)
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H] float32

    # ------------------------------------------------------------------
    # 7. Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # float32

    # ------------------------------------------------------------------
    # 8. Apply out‑gate (converted to float32) and final projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(out.dtype)                 # float32
    out_fp16 = out.to(torch.float16)                    # fp16 for final linear
    output = F.linear(out_fp16, to_out_weight)          # [B, N, N, C] fp16
    output = output.to(torch.float32)                   # back to fp32

    return output
