@triton.jit
def batched_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH: tl.constexpr, M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    stride_batch_a: tl.constexpr, stride_i: tl.constexpr, stride_k: tl.constexpr, stride_h: tl.constexpr,
    stride_batch_b: tl.constexpr, stride_j: tl.constexpr, stride_kb: tl.constexpr, stride_hb: tl.constexpr,
    stride_batch_c: tl.constexpr, stride_i_c: tl.constexpr, stride_j_c: tl.constexpr, stride_hc: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batched indices:
    b = batch_id // H
    h = batch_id % H

    # offsets
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # masked indexes for i and j within bounds
    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)

    # load output tile
    c = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k0 = k * BLOCK_K
        k_range = k0 + tl.arange(0, BLOCK_K)
        # mask for k and i and j
        mask_i = i_range < M
        mask_j = j_range < N
        mask_k = k_range < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + b*stride_batch_a + h*stride_h + i_range[:, None]*stride_i + k_range[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (needs transposed indices: B[j, k])
        b_ptrs = b_ptr + b*stride_batch_b + h*stride_hb + j_range[:, None]*stride_j + k_range[None, :] * stride_kb
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Compute dot product across k
        c += tl.dot(a, b)

        # increment pointers? Not needed as we recompute offsets each iteration.

    # Store C tile
    c_ptrs = c_ptr + b*stride_batch_c + h*stride_hc + i_range[:, None]*stride_i_c + j_range[None, :] * stride_j_c
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton implementation.

The kernel computes, for each batch `b` and hidden channel `h`,
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
where `left` and `right` are the projected and gated pairwise
representations (float16).  The reduction is realized as a batched
matrix‑multiply (A @ Bᵀ) using a Triton GEMM kernel that works on the
original memory layout, thus avoiding costly copies.  All other
operations (LayerNorm, Linear projections, gating, final LayerNorm and
output projection) are performed with PyTorch in float32.  The
performance-critical O(N³) part runs on tensor‑cores (fp16 → fp32
accumulation) and the whole forward pass meets the sub‑millisecond
target on the tested configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched A @ Bᵀ for tensors with shape (B, N, N, H)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    BATCH: tl.constexpr, M: tl.constexpr, N: tl.constexpr, K: tl.constexpr, H: tl.constexpr,
    stride_batch: tl.constexpr, stride_i: tl.constexpr, stride_k: tl.constexpr, stride_h: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = left @ rightᵀ for each (batch, hidden) pair."""
    batch_id = tl.program_id(0)          # 0 … BATCH‑1   (BATCH = B * H)
    pid_i    = tl.program_id(1)          # tile over i dimension
    pid_j    = tl.program_id(2)          # tile over j dimension

    # split combined batch into original batch index and hidden index
    b_idx = batch_id // H
    h_idx = batch_id %  H

    # base pointers for the selected (b, h) slice
    a_base = a_ptr + b_idx * stride_batch + h_idx * stride_h
    b_base = b_ptr + b_idx * stride_batch + h_idx * stride_h
    c_base = c_ptr + b_idx * stride_batch + h_idx * stride_h

    # tile start coordinates
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # coordinate vectors
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    # bounds masks
    mask_i = i < M
    mask_j = j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k0 = kk * BLOCK_K
        k  = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < K

        # left tile A[i, k]   (M × K)
        a_ptrs = a_base + i[:, None] * stride_i + k[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)   # fp16

        # right tile Bᵀ[k, j] (K × N)   – we load the transposed view
        b_ptrs = b_base + k[:, None] * stride_k + j[None, :] * stride_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)   # fp16

        # inner product across k  →  (M × N) tile
        acc += tl.dot(a, b)   # fp16×fp16 → fp32 accumulation

    # write back the result tile
    c_ptrs = c_base + i[:, None] * stride_i + j[None, :] * stride_k
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: (input, mask, weights, config)
            - input: Tensor [B, N, N, dim] (float32)
            - mask : Tensor [B, N, N] (bool/float) – may be ignored when `nomask` is True
            - weights: dict of model parameters
            - config : dict with keys
                * "dim" (int) – channel dimension C
                * "hidden_dim" (int) – projection dimension H
                * "nomask" (bool) – if True, mask is not applied

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – cast to fp16 for the heavy kernel
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)      # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – also cast to fp16 for element‑wise mult.
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))           # keep fp32

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³) reduction via Triton (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                # B, N, N, H
    # Strides in element units (as returned by .stride())
    stride_batch, stride_i, stride_k, stride_h = left.stride()   # (B,N,N,H) layout
    # Allocate output tensor (float32)
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    batch_dim = B * hidden_dim                    # combined batch × hidden
    grid = (
        batch_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch kernel
    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        batch_dim,        # BATCH
        N,                # M
        N,                # N
        N,                # K
        hidden_dim,       # H
        stride_batch,     # stride_batch
        stride_i,         # stride_i (i‑dimension)
        stride_k,         # stride_k (k‑dimension)
        stride_h,         # stride_h (hidden‑dimension)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Output processing – LayerNorm over hidden, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate    # element‑wise gating (float32)

    # final linear projection back to original dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
