"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton implementation.

Algorithm
---------
1.  Layer‐norm the input tensor (FP32).
2.  Linear projections (dim → hidden_dim) → left/right (FP16).
3.  Optional mask (broadcasted on the hidden dimension) and gate
    sigmoid(linear) (FP32 → FP16) are applied to left/right.
4.  Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    which is a batched GEMM per hidden channel:
        out_{b,d} = left_{b,d} @ right_{b,d}ᵀ
    Implemented as a single Triton kernel that processes
    (batch*head) as the first program dimension and tiles
    the (i,j) axes (BLOCK_M×BLOCK_N) with a K‑loop (BLOCK_K).
    Inputs are FP16, accumulation is FP32.
5.  Output layer‑norm + output gate (FP32) and final
    linear projection back to `dim`.

The kernel fuses the N³ reduction while the surrounding
operations remain in PyTorch for simplicity.  All tensors
are assumed to reside on the same CUDA device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix dimensions (int32)
    stride_ah, stride_am, stride_ak,    # strides for A (batch/head, i, k)
    stride_bh, stride_bk, stride_bn,    # strides for B (batch/head, k, j)
    stride_ch, stride_cm, stride_cn,    # strides for C (batch/head, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ  (A: (BH,M,K), B: (BH,K,N), C: (BH,M,N))

    All tensors are stored in row‑major order.
    A and B are fp16, C is fp32 accumulation.
    """
    pid_bh = tl.program_id(0)   # batch*head index
    pid_m = tl.program_id(1)    # row tile
    pid_n = tl.program_id(2)    # column tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch/head
    a_batch = a_ptr + pid_bh * stride_ah
    b_batch = b_ptr + pid_bh * stride_bh
    c_batch = c_ptr + pid_bh * stride_ch

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # load masks for out-of-bounds handling
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # compute memory addresses
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load tiles (FP16, zero‑pad OOB)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        # matrix multiply‑accumulate
        acc += tl.dot(a, b)   # dot returns FP16, accumulated into FP32

    # write back result
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict with keys:
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config: dict with entries 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    device = x.device
    dtype = x.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2) Linear projections (FP16 for speed)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim] (fp16)
    right = F.linear(x_h, right_proj_w)  # same shape

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid(linear)) – computed in FP32 then cast to fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim] FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # FP32, used later

    # Cast gates to fp16 for element‑wise multiplication
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 5) Core N³ contraction via Triton GEMM
    # ------------------------------------------------------------------
    B, N_seq, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Re‑order to (B, H, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right needs to be transposed on the last two axes (k ↔ j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) where dim2 = K, dim3 = J

    # Collapse batch and head dimensions for the kernel
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N_seq, N_seq)   # (BH, M=N, K=N)
    right_mat = right_perm.view(BH, N_seq, N_seq) # (BH, K=N, N=N)

    # Output buffer (FP32 accumulator)
    out_mat = torch.empty((BH, N_seq, N_seq), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Kernel launch parameters (tuned for A100 / B200)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,                                     # program_id(0): batch*head
        (N_seq + BLOCK_M - 1) // BLOCK_M,        # program_id(1): row tiles
        (N_seq + BLOCK_N - 1) // BLOCK_N,        # program_id(2): column tiles
    )

    _tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N_seq, N_seq, N_seq,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # (B,N,N,hidden_dim) – FP32

    out = out * out_gate  # broadcasting, still FP32

    out = F.linear(out, weights["to_out.weight"])  # (B,N,N,dim) – FP32

    return out
