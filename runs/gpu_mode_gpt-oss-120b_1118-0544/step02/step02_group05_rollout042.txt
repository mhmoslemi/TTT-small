```
@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_batch_a, stride_i_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_j_b,
    stride_batch_c, stride_i_c, stride_j_c,
    hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)  # index over batch*head
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    batch = pid // hidden_dim
    head = pid % hidden_dim

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    a_base = a_ptr + batch * stride_batch_a + head
    b_base = b_ptr + batch * stride_batch_b + head
    c_base = c_ptr + batch * stride_batch_c + head

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        a_mask = (offs_i[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_j[None, :] < N)

        a = tl.load(a_base + offs_i[:, None] * stride_i_a + offs_k[None, :] * stride_k_a,
                    mask=a_mask, other=0.0)
        # for b we need tile (K,N) where rows are k, cols are j
        b = tl.load(b_base + offs_k[:, None] * stride_k_b + offs_j[None, :] * stride_j_b,
                    mask=b_mask, other=0.0)
        acc += tl.dot(a, b)

"""
TriMul (outgoing) – forward pass using Triton.

The core operation is a batched matrix multiplication:
    out[b,head,i,j] = Σₖ left[b,i,k,head] * right[b,j,k,head]

* Input is LayerNorm'ed then projected (fp16) to a hidden dimension.
* Optional mask and per‑channel sigmoid gates are applied in fp16.
* The N³ contraction is performed by a custom Triton kernel that
  processes each (batch, head) pair as an independent GEMM.
* The kernel works directly on the original [B,N,N,head] layout,
  avoiding any permutes or copies.  Accumulation is in fp32, the
  result is stored in fp32 and later LayerNorm'ed, gated and projected
  back to the input dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dims (int32)
    stride_batch_a, stride_i_a, stride_k_a,   # strides for A: (batch, i, k)
    stride_batch_b, stride_k_b, stride_j_b,   # strides for B: (batch, k, j)
    stride_batch_c, stride_i_c, stride_j_c,   # strides for C: (batch, i, j)
    hidden_dim: tl.constexpr,          # hidden dimension (runtime const)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for TriMul:
    A: (batch, M, K)    fp16
    B: (batch, K, N)    fp16   (original right tensor, not transposed)
    C: (batch, M, N)    fp32 accumulator
    """
    pid = tl.program_id(0)   # linear index over (batch * hidden_dim)
    pid_i = tl.program_id(1) # block row
    pid_j = tl.program_id(2) # block column

    # decompose pid into batch and head
    batch = pid // hidden_dim
    head = pid % hidden_dim

    # offsets for this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this (batch,head)
    a_base = a_ptr + batch * stride_batch_a + head
    b_base = b_ptr + batch * stride_batch_b + head
    c_base = c_ptr + batch * stride_batch_c + head

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # masks for out‑of‑bounds loads
        a_mask = (offs_i[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_j[None, :] < N)

        # load tiles: A[i,k] and B[k,j] (note B is not transposed)
        a = tl.load(
            a_base
            + offs_i[:, None] * stride_i_a
            + offs_k[None, :] * stride_k_a,
            mask=a_mask,
            other=0.0,
        )
        b = tl.load(
            b_base
            + offs_k[:, None] * stride_k_b
            + offs_j[None, :] * stride_j_b,
            mask=b_mask,
            other=0.0,
        )

        # matrix‑multiply‑accumulate
        acc += tl.dot(a, b)          # a,b are fp16 → acc stays fp32
    # write back
    c_mask = (offs_i[:, None] < M) & (offs_j[None, :] < N)
    tl.store(
        c_base
        + offs_i[:, None] * stride_i_c
        + offs_j[None, :] * stride_j_c,
        acc,
        mask=c_mask,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32/bool) or None
            - weights: dict with the following keys (all float32 tensors)
                * norm.weight, norm.bias
                * left_proj.weight, right_proj.weight
                * left_gate.weight, right_gate.weight, out_gate.weight
                * to_out_norm.weight, to_out_norm.bias
                * to_out.weight
            - config: dict containing at least:
                * dim (int) – input channel dimension
                * hidden_dim (int) – hidden dimension used inside TriMul
                * nomask (bool, optional) – if True the mask can be ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2) Projections to hidden_dim (cast to fp16 for speed)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)    # [B, N, N, hidden_dim]  fp16
    right = F.linear(x_h, right_proj_w) # same shape

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) – computed in fp32 then cast to fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # fp32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # fp32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # fp32

    left = left * left_gate.to(x_h.dtype)
    right = right * right_gate.to(x_h.dtype)

    # ------------------------------------------------------------------
    # 5) Core N³ contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    # No permute/copy – we work on the native layout [B,N,N,head]
    # Kernel expects:
    #   A : (B, N, N, head)   -> strides (batch,i,k,head)
    #   B : (B, N, N, head)   -> we read it as (batch,k,j,head)
    #   C : (B, head, N, N)   -> we allocate in that layout directly
    out_mat = torch.empty((B, hidden_dim, N, N), dtype=torch.float32, device=device)

    # Strides (in elements) for the three tensors
    stride_batch_a = left.stride(0)    # B stride = N*N*hidden_dim
    stride_i_a = left.stride(1)        # i stride = N*hidden_dim
    stride_k_a = left.stride(2)        # k stride = hidden_dim

    stride_batch_b = right.stride(0)   # same as left
    stride_k_b = right.stride(2)       # k stride = hidden_dim
    stride_j_b = right.stride(1)       # j stride = N*hidden_dim

    stride_batch_c = out_mat.stride(0) # B stride = hidden_dim*N*N
    stride_i_c = out_mat.stride(2)     # i stride = N
    stride_j_c = out_mat.stride(3)     # j stride = 1

    # Tunable block sizes (chosen for good occupancy on B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                # batch * head dimension
        (N + BLOCK_M - 1) // BLOCK_M,  # rows
        (N + BLOCK_N - 1) // BLOCK_N,  # cols
    )

    _tri_mul_gemm_kernel[grid](
        left,
        right,
        out_mat,
        N, N, N,
        stride_batch_a, stride_i_a, stride_k_a,
        stride_batch_b, stride_k_b, stride_j_b,
        stride_batch_c, stride_i_c, stride_j_c,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Rearrange back to [B, N, N, hidden_dim]
    out = out_mat.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]  FP32

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                # out_gate is FP32
    out = F.linear(out, weights["to_out.weight"])  # back to dim (FP32)

    return out
