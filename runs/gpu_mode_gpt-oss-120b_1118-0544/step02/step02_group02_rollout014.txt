def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    B, N, H,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    i_tile = tl.program_id(0)
    j_tile = tl.program_id(1)
    bh = tl.program_id(2)  # batch*hidden

    b = bh // H
    d = bh % H

    # compute global offsets
    i_offset = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i_offset < N
    mask_j = j_offset < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left tile: shape (BLOCK_I, BLOCK_K)
        left_ptrs = left_ptr + b * stride_left_b + i_offset[:, None] * stride_left_i + k_offset[None, :] * stride_left_k + d * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right tile transposed (k, j)
        right_ptrs = right_ptr + b * stride_right_b + k_offset[:, None] * stride_right_k + j_offset[None, :] * stride_right_j + d * stride_right_h

        # Actually base index: b * stride_right_b + j_offset * stride_right_j + k_offset * stride_right_k + d * stride_right_h

        # So pointer formula: b* stride_right_b + k_offset[:, None] * stride_right_k + j_offset[None, :] * stride_right_j + d * stride_right_h

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The costly O(N³·H) reduction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is implemented as a tiled matrix‑multiply kernel in Triton.
Each program computes a (BLOCK_I, BLOCK_J) tile of the output for a
specific (batch, hidden) pair, loading tiles of size (BLOCK_I, BLOCK_K)
and (BLOCK_K, BLOCK_J) from the two operand tensors, accumulating in
float32.  All LayerNorms, linear projections and gating are performed
in PyTorch (half‑precision where possible) and only the reduction is
off‑loaded to Triton, eliminating the large permute‑and‑view copies of
the previous implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel for the O(N³) reduction
# --------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,            # fp16 input [B,N,N,H]
    right_ptr,           # fp16 input [B,N,N,H]
    out_ptr,             # fp32 output [B,N,N,H]

    # strides for left
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for out
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    B, N, H,                         # runtime sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids: tile over i, j and (batch*hidden)
    i_tile = tl.program_id(0)
    j_tile = tl.program_id(1)
    bh_id  = tl.program_id(2)          # combined batch*hidden index

    # decode batch and hidden index
    b = bh_id // H
    d = bh_id % H

    # global offsets for the i/j tile
    i_off = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i_off < N
    mask_j = j_off < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ----- loop over the reduction dimension k -----
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # ---- load left tile (i, k) ----
        # address: b, i_off, k_off, d
        left_ptrs = (left_ptr
                     + b * stride_left_b
                     + i_off[:, None] * stride_left_i
                     + k_off[None, :] * stride_left_k
                     + d * stride_left_h)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ---- load right tile transposed (k, j) ----
        # address: b, j_off, k_off, d  -> we swap axes to get (k, j)
        right_ptrs = (right_ptr
                      + b * stride_right_b
                      + k_off[:, None] * stride_right_k
                      + j_off[None, :] * stride_right_j
                      + d * stride_right_h)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # ---- matrix multiply and accumulate ----
        acc += tl.dot(left, right)   # (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J)

    # ---- write back the tile ----
    out_ptrs = (out_ptr
                + b * stride_out_b
                + i_off[:, None] * stride_out_i
                + j_off[None, :] * stride_out_j
                + d * stride_out_h)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Performs the outgoing TriMul forward pass.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model weights
        - config       : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns
    -------
    torch.Tensor [B, N, N, D] (float32)
    """
    # ------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------
    # fetch model parameters (float32) and create fp16 copies
    # ------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (H, D)
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # fp16 versions for the heavy path
    left_proj_weight_h = left_proj_weight.half()
    right_proj_weight_h = right_proj_weight.half()
    left_gate_weight_h = left_gate_weight.half()
    right_gate_weight_h = right_gate_weight.half()
    out_gate_weight_h = out_gate_weight.half()

    # ------------------------------------------------------
    # 0) Input LayerNorm (float32)
    # ------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, D] float32

    # ------------------------------------------------------
    # 1) Cast to FP16 for the rest of the forward
    # ------------------------------------------------------
    x_fp16 = x.half()   # [B, N, N, D] fp16

    # ------------------------------------------------------
    # 2) Linear projections (bias‑free, fp16)
    # ------------------------------------------------------
    left = F.linear(x_fp16, left_proj_weight_h)   # [B, N, N, H] fp16
    right = F.linear(x_fp16, right_proj_weight_h)

    # ------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------
    # 4) Gating (sigmoid, fp16)
    # ------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight_h))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight_h))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight_h))   # [B, N, N, H] fp16

    # ------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------
    # 6) Heavy reduction via Triton
    # ------------------------------------------------------
    B, N, _, _ = left.shape
    H = hidden_dim

    # output tensor (float32) = result of reduction
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # strides (int64) for the three tensors
    stride_left = left.stride()   # (b,i,k,h)
    stride_right = right.stride()
    stride_out = out.stride()

    # Tile sizes – chosen to fit comfortably in shared memory
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    i_tiles = (N + BLOCK_I - 1) // BLOCK_I
    j_tiles = (N + BLOCK_J - 1) // BLOCK_J

    grid = (i_tiles, j_tiles, B * H)

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        B, N, H,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------
    # 7) Final LayerNorm, gating and projection (float32)
    # ------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )                                          # [B,N,N,H] float32

    # out_gate was computed in fp16 – cast to float32 for the multiplication
    out_gate_fp32 = out_gate.float()
    out = out * out_gate_fp32

    # final linear projection back to original dimensionality
    out = F.linear(out, to_out_weight)          # [B,N,N,D] float32

    return out
