"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton‑fused O(N³) reduction.

Algorithm
---------
1. LayerNorm over the channel dimension of the input.
2. Linear projections to a hidden dimension (left/right) and three
   gating projections (left_gate, right_gate, out_gate).  The left/right
   tensors are optionally masked and multiplied by their respective gates.
3. The core O(N³) operation:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix multiplication A @ Bᵀ for each hidden channel.
   We fuse this reduction in a custom Triton kernel that:
   * works on fp16 inputs (left/right) and accumulates in fp32,
   * processes many (batch × hidden_dim) matrices in parallel,
   * avoids extra transposes by loading the right matrix with swapped strides.
4. LayerNorm over the hidden dimension of the product.
5. Element‑wise multiplication with the out‑gate (already computed in step 2).
6. Final linear projection back to the original channel dimension.

The implementation uses Triton for the heavy O(N³) reduction (and satisfies the
“must use Triton” requirement) while the remaining cheap ops are performed
with PyTorch.  It supports optional masking, batch sizes 1‑2, sequence lengths
up to 1024, and hidden dimensions up to 384.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton kernel – batched matrix multiplication (A @ Bᵀ)
# ---------------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                              # matrix dimensions (all = seq_len)
    stride_am, stride_ak, stride_ab,      # A strides: row, col, batch
    stride_bk, stride_bn, stride_bb,      # B strides (with K‑N transposed view)
    stride_cm, stride_cn, stride_cb,      # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)               # batch × hidden index
    pid_m = tl.program_id(0)               # output‑row tile
    pid_n = tl.program_id(1)               # output‑col tile

    # Offsets within the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Pointers to the current tile of A and B
        a_ptrs = a_ptr + pid_b * stride_ab \
                 + offs_m[:, None] * stride_am \
                 + k_offset[None, :] * stride_ak
        b_ptrs = b_ptr + pid_b * stride_bb \
                 + k_offset[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)    # fp16 × fp16 → fp32

    # Write the result
    c_ptrs = c_ptr + pid_b * stride_cb \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ---------------------------------------------------------------------------
# Entry point expected by the evaluator
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – optional
            - weights: dict of model weights (all on the same CUDA device)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, the mask is ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype  # expected float32

    # -----------------------------------------------------------------------
    # Unpack model weights
    # -----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    B, N, _, _ = input_tensor.shape  # (B, seq_len, seq_len, dim)

    # -----------------------------------------------------------------------
    # 1️⃣ LayerNorm over dim
    # -----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # -----------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim] (fp32)
    right = F.linear(x, right_proj_weight)  # same shape

    # -----------------------------------------------------------------------
    # 3️⃣ Optional masking
    # -----------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)           # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # -----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # kept fp32 for later

    # Apply gates and switch to fp16 for the heavy reduction
    left = left.to(torch.float16) * left_gate
    right = right.to(torch.float16) * right_gate

    # -----------------------------------------------------------------------
    # 5️⃣ Prepare tensors for the batched matmul
    #    left, right → shape (B*hidden_dim, N, N)
    # -----------------------------------------------------------------------
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)   # fp16
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)  # fp16

    # We need rightᵀ for the matmul; a simple view (no copy) suffices
    right_T = right_flat.transpose(1, 2)   # shape (B*H, N, N) with swapped strides

    total_batches = left_flat.shape[0]     # B * hidden_dim

    # -----------------------------------------------------------------------
    # 6️⃣ Allocate output buffer (fp32 accumulator)
    # -----------------------------------------------------------------------
    out_flat = torch.empty((total_batches, N, N), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------------
    # 7️⃣ Compute strides for the Triton kernel
    # -----------------------------------------------------------------------
    stride_am = left_flat.stride(1)   # row stride (N)
    stride_ak = left_flat.stride(2)   # col stride (1)
    stride_ab = left_flat.stride(0)   # batch stride (N*N)

    stride_bk = right_T.stride(1)    # will be 1 (acts as K‑row stride)
    stride_bn = right_T.stride(2)    # will be N (acts as N‑col stride)
    stride_bb = right_T.stride(0)    # batch stride (N*N)

    stride_cm = out_flat.stride(1)   # row stride (N)
    stride_cn = out_flat.stride(2)   # col stride (1)
    stride_cb = out_flat.stride(0)   # batch stride (N*N)

    # -----------------------------------------------------------------------
    # 8️⃣ Triton launch configuration
    # -----------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m, num_pid_n, total_batches)

    batched_matmul_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        N, N, N,                     # M, N, K
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # -----------------------------------------------------------------------
    # 9️⃣ Reshape back to [B, N, N, hidden_dim]
    # -----------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)        # [B, N, N, hidden_dim]
        .contiguous()
    )  # fp32

    # -----------------------------------------------------------------------
    # 10️⃣ Final LayerNorm over hidden dimension
    # -----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # -----------------------------------------------------------------------
    # 11️⃣ Apply the out‑gate (element‑wise)
    # -----------------------------------------------------------------------
    out = out * out_gate

    # -----------------------------------------------------------------------
    # 12️⃣ Project back to the original dimension
    # -----------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
