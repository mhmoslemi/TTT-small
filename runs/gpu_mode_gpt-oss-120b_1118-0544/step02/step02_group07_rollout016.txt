    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    
    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight) # [B,N,N,hidden]

    # Mask
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Convert to half for matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    right_h = right_h.permute(0,3,1,2).contiguous().view(B * hidden_dim, N, N)

    # Allocate output
    out_h = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=x.device)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_A_batch, stride_Am, stride_Ak,
                          stride_B_batch, stride_Bk, stride_Bn,
                          stride_C_batch, stride_Cm, stride_Cn,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    a_ptrs = A_ptr + batch * stride_A_batch + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + batch * stride_B_batch + offs_n[:, None] * stride_Bn + offs_k[None, :] * stride_Bk

"""Outgoing Triangle Multiplicative Update (TriMul) – fused Triton kernel.

The heavy O(N³) reduction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is realized as a batched matrix multiplication:
    left[b,d] @ right[b,d]ᵀ   with   left, right ∈ ℝ^{B·H × N × N}
where H = hidden_dim.  A single Triton kernel processes the full batch,
tiling the M (i) and N (j) axes and streaming over K (the reduction axis).
All other operations (LayerNorm, Linear projections, gating) are performed
with PyTorch in FP32, while the large tensors fed to the kernel are stored
in FP16 to reduce memory traffic.  The final output is cast back to FP32.

The implementation supports optional masking, arbitrary batch size (1‑2),
sequence lengths up to 1024, and hidden dimensions of 128.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_A_batch, stride_Am, stride_Ak,
    stride_B_batch, stride_Bk, stride_Bn,
    stride_C_batch, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of matrices.

    A : (batch, M, K)      (FP16)
    B : (batch, K, N)      (FP16) – accessed transposed, i.e. as (batch, N, K)
    C : (batch, M, N)      (FP32 accumulator)
    """
    pid = tl.program_id(0)

    # compute batch index and 2‑D tile indices
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_blocks_m * num_blocks_n)
    block_flat = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_flat // num_blocks_n
    pid_n = block_flat % num_blocks_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds handling
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # base pointers for this batch
    a_base = A_ptr + batch * stride_A_batch
    b_base = B_ptr + batch * stride_B_batch
    c_base = C_ptr + batch * stride_C_batch

    # pointer arithmetic for tiles
    a_ptrs = a_base + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    # B is accessed transposed: we want B_T[n, k] = B[k, n]
    b_ptrs = b_base + offs_n[:, None] * stride_Bn + offs_k[None, :] * stride_Bk

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)
        # up‑cast to FP32 for higher‑precision accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_N)
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # store result
    c_ptrs = c_base + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict with the following keys
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # unpack model weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm over channel dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3️⃣ Optional masking (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Cast to FP16 and reshape for batched GEMM
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, N, N, hidden_dim
    # (batch, hidden, N, N) --> (B*hidden, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # -----------------------------------------------------------------
    # 6️⃣ Batched matrix multiplication using Triton
    # -----------------------------------------------------------------
    out_h = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=x.device)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_batch = B * hidden_dim
    grid = (total_batch * num_blocks_m * num_blocks_n,)

    batched_matmul_kernel[grid](
        left_h,
        right_h,
        out_h,
        N, N, N,
        left_h.stride(0), left_h.stride(1), left_h.stride(2),   # A: batch, row, col
        right_h.stride(0), right_h.stride(1), right_h.stride(2),  # B: batch, row(K), col(N)
        out_h.stride(0), out_h.stride(1), out_h.stride(2),     # C: batch, row, col
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Restore original 4‑D layout
    # -----------------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)          # [B, N, N, hidden_dim]
        .contiguous()
    )

    # -----------------------------------------------------------------
    # 8️⃣ Final LayerNorm over hidden dimension
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Apply output gate (element‑wise)
    # -----------------------------------------------------------------
    out = out * out_gate

    # -----------------------------------------------------------------
    # 🔟 Project back to the original dimension
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
```
