@triton.jit
def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs: batch hidden (pid0), block_i (pid1), block_j (pid2)
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # derive batch and hidden index, but only need batch of A and B
    # The batch dimension includes hidden dimension already; we treat each hidden slice as its own batch via pid_batch.
    # But we also need to compute offset for batch and hidden: we treat hidden dimension already flattened in batch dimension.
    # So we just need stride_a_batch, stride_b_batch, stride_c_batch = N*N*H, etc.

    # compute offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    k_offset = tl.arange(0, BLOCK_K)

    mask_i = i_offset < M
    mask_j = j_offset < N
    # we will handle k mask dynamically in loop

    # Compute base pointers for A and B
    a_ptrs = A_ptr + pid_batch * stride_a_batch + i_offset[:, None] * stride_a_i + k_offset[None, :] * stride_a_k
    b_ptrs = B_ptr + pid_batch * stride_b_batch + k_offset[:, None] * stride_b_k + j_offset[None, :] * stride_b_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension blocks
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for kb in range(0, num_k_blocks):
        # mask for k out-of-bounds
        k_valid = (kb * BLOCK_K + k_offset) < K
        # load A and B
        a = tl.load(a_ptrs, mask=mask_i[:, None] & k_valid[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=k_valid[:, None] & mask_j[None, :], other=0.0)
        acc += tl.dot(a, b)
        # increment pointers to next k block
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store result
    c_ptrs = C_ptr + pid_batch * stride_c_batch + i_offset[:, None] * stride_c_i + j_offset[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

def tri_mul_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Ai, stride_Ak, stride_Ad,
    stride_Bb, stride_Bk, stride_Bj, stride_Bd,
    stride_Cb, stride_Ci, stride_Cj, stride_Cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_i = tl.program_id(2)
    pid_j = tl.program_id(3)

    # define offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # compute base pointers for this batch & hidden
    a_ptr = A + pid_b * stride_Ab + pid_h * stride_Ad + offs_i[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
    b_ptr = B + pid_b * stride_Bb + pid_h * stride_Bd + offs_k[:, None] * stride_Bk + offs_j[None, :] * stride_Bj

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        cur_k = k * BLOCK_K
        k_valid = (cur_k + offs_k) < K
        a = tl.load(a_ptr, mask=mask_i[:, None] & k_valid[None, :], other=0.0, dtype=tl.float16)
        b = tl.load(b_ptr, mask=k_valid[:, None] & mask_j[None, :], other=0.0, dtype=tl.float16)
        acc += tl.dot(a, b)
        # increment pointers to next k block
        a_ptr += BLOCK_K * stride_Ak
        b_ptr += BLOCK_K * stride_Bk

    # store results
    c_ptr = C + pid_b * stride_Cb + pid_h * stride_Cd + offs_i[:, None] * stride_Ci + offs_j[None, :] * stride_Cj
    tl.store(c_ptr, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps=1e-5

    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # convert to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=device)

    # get strides
    stride_a_batch = left_fp16.stride(0)
    stride_a_i = left_fp16.stride(1)
    stride_a_k = left_fp16.stride(2)
    stride_a_d = left_fp16.stride(3)

    stride_b_batch = right_fp16.stride(0)
    stride_b_k = right_fp16.stride(2)  # row (k)
    stride_b_j = right_fp16.stride(1)  # col (j)
    stride_b_d = right_fp16.stride(3)

    stride_c_batch = out.stride(0)
    stride_c_i = out.stride(1)
    stride_c_j = out.stride(2)
    stride_c_d = out.stride(3)

    # block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (bs, hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left_fp16, right_fp16, out,
        N, N, N,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
        stride_b_batch, stride_b_k, stride_b_j, stride_b_d,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # out now fp32

    # final layernorm over hidden_dim
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # apply out_gate
    out = out * out_gate

    # final linear projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM using a custom Triton kernel.
The kernel works directly on the original [B, N, N, H] layout,
avoiding costly transposes.  All other operations (LayerNorm,
linear projections, gating) are performed with PyTorch in fp32,
while the GEMM itself runs on fp16 inputs with fp32 accumulation.
A Triton kernel thus handles the core reduction, satisfying the
requirement to use Triton and dramatically reducing runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel for the core O(N³) reduction
# -------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    A,               # left  [B, N, N, H]  (fp16)
    B,               # right [B, N, N, H]  (fp16)
    C,               # out   [B, N, N, H]  (fp32)
    M, N, K,         # M=N rows (i), N=cols (j), K=sum dimension (k)
    stride_Ab, stride_Ai, stride_Ak, stride_Ad,
    stride_Bb, stride_Bk, stride_Bj, stride_Bd,
    stride_Cb, stride_Ci, stride_Cj, stride_Cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids: batch, hidden, i‑tile, j‑tile
    pid_b = tl.program_id(0)   # batch index
    pid_h = tl.program_id(1)   # hidden‑channel index
    pid_i = tl.program_id(2)   # i‑tile index
    pid_j = tl.program_id(3)   # j‑tile index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Base pointers for this (batch, hidden) slice
    # ------------------------------------------------------------------
    a_ptr = A + pid_b * stride_Ab + pid_h * stride_Ad \
            + offs_i[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
    b_ptr = B + pid_b * stride_Bb + pid_h * stride_Bd \
            + offs_k[:, None] * stride_Bk + offs_j[None, :] * stride_Bj

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k in blocks of size BLOCK_K
    # ------------------------------------------------------------------
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        cur_k = k * BLOCK_K
        k_valid = (cur_k + offs_k) < K

        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & k_valid[None, :],
                    other=0.0)               # fp16 load
        b = tl.load(b_ptr,
                    mask=k_valid[:, None] & mask_j[None, :],
                    other=0.0)               # fp16 load

        acc += tl.dot(a, b)                # fp16*fp16 -> fp32 accumulate

        # advance pointers to the next k‑block
        a_ptr += BLOCK_K * stride_Ak
        b_ptr += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    c_ptr = C + pid_b * stride_Cb + pid_h * stride_Cd \
            + offs_i[:, None] * stride_Ci + offs_j[None, :] * stride_Cj
    tl.store(c_ptr,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

# -------------------------------------------------------------
# Entry point used by the evaluator
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – may be ignored
            - weights: dict of model weights
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – projection dimension H
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # Unpack config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Convert to fp16 for the Triton GEMM
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N_seq, _, _ = left_fp16.shape
    assert B == input_tensor.shape[0], "batch size mismatch"
    assert N_seq == input_tensor.shape[1], "sequence length mismatch"

    # ------------------------------------------------------------------
    # 6️⃣ Allocate output buffer (fp32)
    # ------------------------------------------------------------------
    out = torch.empty((B, N_seq, N_seq, hidden_dim), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7️⃣ Gather strides (needed by the kernel)
    # ------------------------------------------------------------------
    stride_Ab = left_fp16.stride(0)
    stride_Ai = left_fp16.stride(1)
    stride_Ak = left_fp16.stride(2)
    stride_Ad = left_fp16.stride(3)

    stride_Bb = right_fp16.stride(0)
    stride_Bk = right_fp16.stride(2)   # row (k) of the transposed view
    stride_Bj = right_fp16.stride(1)   # col (j) of the transposed view
    stride_Bd = right_fp16.stride(3)

    stride_Cb = out.stride(0)
    stride_Ci = out.stride(1)
    stride_Cj = out.stride(2)
    stride_Cd = out.stride(3)

    # ------------------------------------------------------------------
    # 8️⃣ Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B,                                 # batch dimension
        hidden_dim,                        # hidden channel dimension
        (N_seq + BLOCK_M - 1) // BLOCK_M,  # i‑tiles
        (N_seq + BLOCK_N - 1) // BLOCK_N   # j‑tiles
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        N_seq, N_seq, N_seq,          # M, N, K
        stride_Ab, stride_Ai, stride_Ak, stride_Ad,
        stride_Bb, stride_Bk, stride_Bj, stride_Bd,
        stride_Cb, stride_Ci, stride_Cj, stride_Cd,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 🔟 Apply out gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return out
