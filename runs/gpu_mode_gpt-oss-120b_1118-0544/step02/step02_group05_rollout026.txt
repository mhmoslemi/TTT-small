"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton implementation.

The core operation is:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right are obtained from linear projections and gated element‑wise.
We treat each (batch, head) pair as an independent matrix multiplication:
    left  → A of shape (M=N, K=N)
    right → B of shape (K=N, N=N)
and compute C = A @ B (batched). The heavy N³ work is performed by a
custom Triton kernel that tiles the (i, j) output axes and reduces over k.
All other layers (LayerNorm, Linear, Sigmoid, mask handling) are done with
PyTorch. Mixed‑precision (fp16 for inputs, fp32 accumulation) is used for
speed, while the final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,            # Tensor pointers
    M, N, K,                        # Matrix dimensions (int32)
    stride_ah, stride_am, stride_ak,   # Strides for A (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Strides for B (batch, k, j)
    stride_ch, stride_cm, stride_cn,   # Strides for C (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ B
    A: (batch, M, K)   fp16
    B: (batch, K, N)   fp16
    C: (batch, M, N)   fp32 accumulator
    """
    pid_batch = tl.program_id(0)   # batch * head index
    pid_m = tl.program_id(1)       # block row (i)
    pid_n = tl.program_id(2)       # block column (j)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the current batch/head
    a_batch = a_ptr + pid_batch * stride_ah
    b_batch = b_ptr + pid_batch * stride_bh
    c_batch = c_ptr + pid_batch * stride_ch

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds elements
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Compute element pointers
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (fp16) with zero padding for OOB
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)  # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)  # (BLOCK_K, BLOCK_N)

        # Accumulate (fp16 * fp16 -> fp16, then promote to fp32)
        acc += tl.dot(a, b)

    # Write back results (fp32)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with the following keys (all float32 tensors)
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config: dict with entries 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden_dim (use fp16 for speed)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)  # cast once

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim] fp16
    right = F.linear(x_h, right_proj_w)  # same dtype

    # ------------------------------------------------------------------
    # 3) Apply optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Compute gates (in FP32) and apply to left/right
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # FP32

    # Cast gates to fp16 for multiplication with left/right
    left_gate_h = left_gate.to(x_h.dtype)
    right_gate_h = right_gate.to(x_h.dtype)

    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 5) Core contraction: out = Σ_k left[..., i, k, d] * right[..., j, k, d]
    #    Implemented as batched matrix multiplication via Triton.
    # ------------------------------------------------------------------
    B, N_seq, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim
    # Reorder tensors to (B, hidden_dim, i, k) and (B, hidden_dim, k, j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hd, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hd, k, j)

    # Collapse batch and head dimensions for the kernel
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N_seq, N_seq)   # (BH, M, K)
    right_mat = right_perm.view(BH, N_seq, N_seq)  # (BH, K, N)

    # Output buffer (float32 accumulator)
    out_mat = torch.empty((BH, N_seq, N_seq), dtype=torch.float32, device=x.device)

    # Tile sizes – chosen for good occupancy on modern GPUs
    BLOCK_M = min(128, N_seq)
    BLOCK_N = min(128, N_seq)
    BLOCK_K = min(32, N_seq)

    grid = (
        BH,
        (N_seq + BLOCK_M - 1) // BLOCK_M,
        (N_seq + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N_seq, N_seq, N_seq,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # both FP32
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim] FP32

    return out
