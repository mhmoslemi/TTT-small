    right_perm = x_layer_norm.permute(0, 2, 3, 1)
    
    # Step 3: Gate computations
    left_gate = tl.sigmoid(left_perm)
    right_gate = tl.sigmoid(right_perm)
    out_gate = tl.sigmoid(compute_out_gate(input_tensor, weights))
    
    # Step 4: Mask handling
    if mask is not None and not config["nomask"] and mask.any():
        mask = mask.to(left_perm.dtype)
        left_perm *= mask[: , None, None, :]
        right_perm *= mask[: , None, None, :]
    else:

    out_left = triton_tiled_gemm(left_perm, right_perm, hidden_dim, BLOCK_X, BLOCK_Y)
    out_left_mask = (out_left % {}}...)
    out_right = out_left * out_gate
    
    # Step 7: Final projection and norm
    return output_projection(out_right, ...)
