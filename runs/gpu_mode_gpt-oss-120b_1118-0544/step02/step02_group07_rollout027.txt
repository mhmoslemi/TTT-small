@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k, stride_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program id mapping
    pid_i = tl.program_id(0)  # i block
    pid_j = tl.program_id(1)  # j block
    pid_bd = tl.program_id(2)  # batch*hidden

    # compute batch and hidden indices
    b = pid_bd // H
    d = pid_bd % H

    # compute base offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute mask for within bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers
        a_ptrs = left_ptr + b*stride_b + d*stride_d + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        b_ptrs = right_ptr + b*stride_b + d*stride_d + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i

        # load
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)   # a, b shapes: (M,K), (K,N) both float32

    # Compute final offset for output
    c_ptrs = out_ptr + b*stride_b + d*stride_d + offs_i[:, None] * stride_i + offs_j[None, :] * stride_i

    # store result
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton implementation.

The core O(N³) reduction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is performed by a custom Triton batched GEMM kernel that computes,
for each (batch, hidden) pair, a matrix multiplication of shape
(N, N) @ (N, N)ᵀ.  All other operations (layer‑norm, linear projections,
gating, final projection) are done with PyTorch for simplicity.
This satisfies the “must use Triton” requirement while providing a
substantial speed‑up over a naïve PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # const * fp16
    right_ptr,         # const * fp16
    out_ptr,           # * fp16   (output)
    B, N, H,           # runtime ints: batch, seq_len, hidden_dim
    stride_b, stride_i, stride_k, stride_d,  # strides in elements
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Program IDs: (i block, j block, batch*hidden block)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)

    # Resolve batch and hidden indices from the fused dimension
    b = pid_bd // H
    d = pid_bd % H

    # Offsets for the output tile (i, j)
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k in tiles
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers to the current tile of A (left) and B (right)
        a_ptrs = left_ptr + \
                 b * stride_b + d * stride_d + \
                 offs_i[:, None] * stride_i + \
                 offs_k[None, :] * stride_k

        b_ptrs = right_ptr + \
                 b * stride_b + d * stride_d + \
                 offs_k[:, None] * stride_k + \
                 offs_j[None, :] * stride_i   # note: stride_i == stride_j

        # Load tiles, apply masks, cast to FP32
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Matrix‑multiply‑accumulate
        acc += tl.dot(a, b)

    # Store the result back in FP16
    c_ptrs = out_ptr + \
             b * stride_b + d * stride_d + \
             offs_i[:, None] * stride_i + \
             offs_j[None, :] * stride_i
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to FP16 for the O(N³) reduction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output buffer in FP16
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6️⃣ Triton batched GEMM kernel (the heavy O(N³) work)
    # ------------------------------------------------------------------
    stride_b, stride_i, stride_k, stride_d = left_fp16.stride()  # (B, i, k, d) strides

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M ),
        ( (N + BLOCK_N - 1) // BLOCK_N ),
        B * hidden,
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden,
        stride_b, stride_i, stride_k, stride_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Convert the result back to FP32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣ Final LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Apply the output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Project back to the original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
```
