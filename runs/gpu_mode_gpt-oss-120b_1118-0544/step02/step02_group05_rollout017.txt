@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_ah, stride_ai, stride_ak, stride_ah_old,
    stride_bh, stride_bk, stride_bn, # note: b shape (B,N,N,H); for kernel we need strides: batch (B), k (?), n (j). b_stride_k = H, b_stride_n = N*H? Actually for loading tile b [k,j], we need stride_k = H, stride_j = N*H.
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0) # combined batch/head
    pid_m = tl.program_id(1)  # block row (i)
    pid_n = tl.program_id(2)  # block col (j)

    # compute batch b and head h indices
    bh = pid
    b_idx = bh // H
    h_idx = bh % H

    # compute offsets for i and j
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute base pointers for this batch/head
    a_batch_ptr = a_ptr + b_idx * stride_ah + h_idx * stride_ah_old
    b_batch_ptr = b_ptr + b_idx * stride_bh + h_idx * stride_bh_old
    c_batch_ptr = c_ptr + b_idx * stride_ch + h_idx * stride_ch_old

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # masks for out-of-bound
        a_mask = (offs_i[:, None] < N) & (offs_k[None, :] < N)
        b_mask = (offs_k[:, None] < N) & (offs_j[None, :] < N)

        # pointers for tiles
        a_ptrs = a_batch_ptr + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        # right tile: (k, j) because we want to multiply left(i,k) * right(k,j)
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=a_mask, other=0.0) # shape (BM, BK)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0) # shape (BK, BN)

        acc += tl.dot(a, b) # accum in float32

    c_mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    c_ptrs = c_batch_ptr + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=c_mask)
```

"""
TriMul (outgoing) forward pass – Triton implementation.

The core operation is:
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which can be interpreted as a batched matrix multiplication:
    for each (b,d)   C = A @ Bᵀ
    A : (N, N)  = left[b, d, i, k]
    B : (N, N)  = right[b, d, j, k]   (note the transpose on the second axis)
We fuse the whole N³ contraction into a single Triton kernel.
All other operations (LayerNorm, Linear projections, sigmoid gates, masking)
are performed in PyTorch, while the heavy reduction is executed in mixed‑precision
(fp16 inputs, fp32 accumulation). No copies of the large tensors are performed;
the kernel works directly on the original memory layout using arbitrary strides.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: Batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,                 # tensor pointers (fp16 for a,b; fp32 for c)
    B, N, H,                             # batch, sequence length, hidden_dim
    stride_a_batch, stride_a_i, stride_a_k,   # strides for left tensor (B,N,N,H)
    stride_b_batch, stride_b_k, stride_b_j,   # strides for right tensor (B,N,N,H) – note k and j order
    stride_c_batch, stride_c_i, stride_c_j,   # strides for output tensor (B,N,N,H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes C[b,i,j,h] = Σₖ A[b,i,k,h] * B[b,j,k,h]   (h is hidden dimension)
    The hidden dimension `h` is fused into the batch dimension (BH = B*H)
    via program_id(0). The kernel works on arbitrary strides, so no
    data rearrangement is needed.
    """
    pid = tl.program_id(0)        # combined batch * hidden (BH)
    pid_m = tl.program_id(1)      # block row (i)
    pid_n = tl.program_id(2)      # block column (j)

    # decode batch (b) and head (h) from the fused dimension
    bh = pid
    b = bh // H
    h = bh % H

    # offsets within the i and j axes
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this (b,h) pair
    a_batch_ptr = a_ptr + b * stride_a_batch + h  # hidden stride == 1
    b_batch_ptr = b_ptr + b * stride_b_batch + h
    c_batch_ptr = c_ptr + b * stride_c_batch + h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # mask for out‑of‑bounds loads
        a_mask = (offs_i[:, None] < N) & (offs_k[None, :] < N)
        b_mask = (offs_k[:, None] < N) & (offs_j[None, :] < N)

        # pointers to the current tiles
        a_ptrs = a_batch_ptr \
                 + offs_i[:, None] * stride_a_i \
                 + offs_k[None, :] * stride_a_k
        b_ptrs = b_batch_ptr \
                 + offs_k[:, None] * stride_b_k \
                 + offs_j[None, :] * stride_b_j

        # load tiles (fp16) with zero padding for OOB
        a = tl.load(a_ptrs, mask=a_mask, other=0.0, dtype=tl.float16)   # (BM, BK)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0, dtype=tl.float16)   # (BK, BN)

        # matrix multiply‑accumulate (fp32 accumulation)
        acc += tl.dot(a, b)

    # write back the result (fp32) with masking
    c_mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    c_ptrs = c_batch_ptr \
             + offs_i[:, None] * stride_c_i \
             + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=c_mask)


# ----------------------------------------------------------------------
# Forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict with the following keys (all float32 tensors)
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config: dict with entries 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2) Linear projections (cast to fp16)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim] fp16
    right = F.linear(x_h, right_proj_w) # same dtype

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gates (computed in FP32, cast to fp16 for element‑wise multiplication)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # FP32

    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 5) Core N³ contraction – fused batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim
    # Allocate output buffer (float32 accumulator)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=x.device)

    # Block size selection (tuned for modern GPUs)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Compute strides (int64)
    stride_a_batch = left.stride(0)
    stride_a_i = left.stride(1)
    stride_a_k = left.stride(2)   # hidden dim stride (== H)
    stride_b_batch = right.stride(0)
    stride_b_k = right.stride(2)   # same as left.stride(2)
    stride_b_j = right.stride(1)   # N * H
    stride_c_batch = out.stride(0)
    stride_c_i = out.stride(1)
    stride_c_j = out.stride(2)

    # Grid: (BH, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    BH = B * H
    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_j,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden_dim] – FP32

    out = out * out_gate  # element‑wise gating (FP32)

    out = F.linear(out, weights["to_out.weight"])  # back to (B,N,N,dim) – FP32

    return out
