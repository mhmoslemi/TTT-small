"""
TriMul (outgoing) forward pass – Triton implementation.

The core operation is a batched matrix multiplication:
    out_{b,d} = left_{b,d} @ right_{b,d}ᵀ
where left/right are obtained from a linear projection of the input
tensor and are gated and masked.  The kernel computes this GEMM in
mixed precision (FP16 inputs, FP32 accumulation) with tiling over the
M (i), N (j) and K (summation) axes.  All other
operations (LayerNorm, linear projections, sigmoids, final LayerNorm
and projection) are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions
    stride_ah, stride_am, stride_ak,   # strides for A (batch_head, M, K)
    stride_bh, stride_bk, stride_bn,   # strides for B (batch_head, K, N)
    stride_ch, stride_cm, stride_cn,   # strides for C (batch_head, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM C = A @ B   where
      A: (BH, M, K)   fp16
      B: (BH, K, N)   fp16
      C: (BH, M, N)   fp32 accumulator
    The kernel tiles over M, N and reduces over K.
    """
    pid_bh = tl.program_id(0)   # which (batch * hidden_dim) we are processing
    pid_m = tl.program_id(1)    # block row index
    pid_n = tl.program_id(2)    # block column index

    # Compute the start indices for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch/head
    a_batch_ptr = a_ptr + pid_bh * stride_ah
    b_batch_ptr = b_ptr + pid_bh * stride_bh
    c_batch_ptr = c_ptr + pid_bh * stride_ch

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for loading out-of-bounds elements
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Compute pointers for tiles
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (fp16) with zero padding for OOB
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        # Multiply‑accumulate
        acc += tl.dot(a, b)   # tl.dot returns fp16; accumulation is fp32

    # Write output (fp32) with boundary checks
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict containing required weight tensors:
                * 'norm.weight', 'norm.bias'
                * 'left_proj.weight', 'right_proj.weight'
                * 'left_gate.weight', 'right_gate.weight', 'out_gate.weight'
                * 'to_out_norm.weight', 'to_out_norm.bias'
                * 'to_out.weight'
            - config: dict with at least 'dim' and 'hidden_dim' keys.
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden_dim (FP16)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim] FP16
    right = F.linear(x_h, right_proj_w)  # same dtype

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # Cast mask to half and broadcast
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gates (computed in FP32, then cast to FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # FP32

    left_gate_h = left_gate.to(x_h.dtype)
    right_gate_h = right_gate.to(x_h.dtype)

    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 5) Core N^3 contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N_seq, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim
    # Re‑order to (B, H, N, N) where H = hidden_dim
    # left: (B, H, i, k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right needs to be transposed on the last two axes: (B, H, k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # Collapse batch and head dimensions for a single GEMM batch dimension
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N_seq, N_seq)          # (BH, M, K)
    right_mat = right_perm.view(BH, N_seq, N_seq)        # (BH, K, N)

    # Allocate output buffer (FP32 accumulator)
    out_mat = torch.empty((BH, N_seq, N_seq), dtype=torch.float32, device=x.device)

    # Tunable block sizes – good balance of occupancy and register pressure
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,
        (N_seq + BLOCK_M - 1) // BLOCK_M,
        (N_seq + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N_seq, N_seq, N_seq,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output gate, final projection (all FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate               # out_gate is FP32, shape broadcasted
    out = F.linear(out, weights["to_out.weight"])   # final projection to dim

    return out
