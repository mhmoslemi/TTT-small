Pseudo:

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_size, N, BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n
):
    pid_batch = tl.program_id(0) # batch index (0..batch_size-1)
    pid_m = tl.program_id(1) # block row
    pid_n = tl.program_id(2) # block col

    # Compute offsets
    batch_offset_l = pid_batch * stride_l_batch
    batch_offset_r = pid_batch * stride_r_batch
    batch_offset_o = pid_batch * stride_o_batch

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers
    a_ptrs = left_ptr + batch_offset_l + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
    # For right, we need to get elements at (k, n) i.e., rows over k, columns over n
    b_ptrs = right_ptr + batch_offset_r + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k*BLOCK_K + offs_k < N), other=0.0)
        b = tl.load(b_ptrs, mask=(k*BLOCK_K + offs_k[:, None] < N) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        # Increment pointers
        a_ptrs += BLOCK_K * stride_l_k
        b_ptrs += BLOCK_K * stride_r_k

    # Store result
    c_ptrs = out_ptr + batch_offset_o + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    left_ptr, right_ptr, out_ptr,
    B_H: tl.constexpr, N: tl.constexpr,
    stride_l_batch: tl.constexpr,
    stride_l_i: tl.constexpr,
    stride_l_k: tl.constexpr,
    stride_r_batch: tl.constexpr,
    stride_r_j: tl.constexpr,
    stride_r_k: tl.constexpr,
    stride_o_batch: tl.constexpr,
    stride_o_i: tl.constexpr,
    stride_o_j: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    block_i = pid_m * BLOCK_M
    block_j = pid_n * BLOCK_N

    # create range
    offs_i = block_i + tl.arange(0, BLOCK_M)
    offs_j = block_j + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers base
    left_base = left_ptr + pid_batch * stride_l_batch
    right_base = right_ptr + pid_batch * stride_r_batch
    out_base = out_ptr + pid_batch * stride_o_batch

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # offsets for k
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Compute pointer for left: (i,k)
        a_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k

        # Compute pointer for right: (k,j)

    right_h_view = right_h.permute(0, 3, 1, 2).contiguous().view(batch_hidden, N, N)

    # Compute strides
    stride_l_batch = left_h_view.stride(0)
    stride_l_i = left_h_view.stride(1)
    stride_l_k = left_h_view.stride(2)
    stride_r_batch = right_h_view.stride(0)
    stride_r_j = right_h_view.stride(1)
    stride_r_k = right_h_view.stride(2)
    stride_o_batch = out_h.stride(0)
    stride_o_i = out_h.stride(1)
    stride_o_j = out_h.stride(2)

    # Triton grid
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (batch_hidden,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left_h_view,
        right_h_view,
        out_h,
        batch_hidden,
        N,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_j, stride_r_k,
        stride_o_batch, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape output back to (B,N,N,hidden)
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # Cast back

    # Apply to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # Apply out_gate gating
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_hidden, N,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute global row and col offsets
    block_i = pid_m * BLOCK_M
    block_j = pid_n * BLOCK_N

    # Offsets within block
    offs_i = block_i + tl.arange(0, BLOCK_M)
    offs_j = block_j + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this batch
    left_batch_ptr = left_ptr + pid_batch * stride_l_batch
    right_batch_ptr = right_ptr + pid_batch * stride_r_batch
    out_batch_ptr = out_ptr + pid_batch * stride_o_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block (M x K)
        a_ptrs = left_batch_ptr + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right block (K x N) – note we need (k, j)
        b_ptrs = right_batch_ptr + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Store result
    out_ptrs = out_batch_ptr + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
Outgoing Triangle Multiplicative Update (TriMul) – fast Triton implementation.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Project the normalized tensor to three groups of linear maps:
   * left_proj & right_proj  (2 × hidden_dim)
   * left_gate, right_gate, out_gate (3 × hidden_dim)
   All five maps are performed with two batched GEMMs (projectors and gates),
   then split into the required tensors.
3. Apply optional binary mask to left/right.
4. Multiply left/right by their respective sigmoid gates.
5. Cast left/right to fp16 and compute the core reduction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   with a custom Triton kernel that treats each (batch,hidden) pair as an
   independent matrix multiplication: left @ rightᵀ.
6. Convert the result back to fp32, apply a second LayerNorm,
   multiply by the out‑gate, and linearly project back to the original
   channel dimension.

The Triton kernel fuses the O(N³) reduction, works on fp16 inputs,
accumulates in fp32, and processes the (batch × hidden) matrices in a
single launch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # half*   [B*H, N, N]
    right_ptr,         # half*   [B*H, N, N]
    out_ptr,           # half*   [B*H, N, N]
    batch_hidden: tl.int32,
    N: tl.int32,
    stride_l_batch: tl.int64,
    stride_l_i: tl.int64,
    stride_l_k: tl.int64,
    stride_r_batch: tl.int64,
    stride_r_j: tl.int64,
    stride_r_k: tl.int64,
    stride_o_batch: tl.int64,
    stride_o_i: tl.int64,
    stride_o_j: tl.int64,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # Program IDs (batch, row‑tile, col‑tile)
    # --------------------------------------------------------------
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Global offsets of the current tile
    off_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = off_i < N
    mask_j = off_j < N

    # Base pointers for this batch (batch × hidden)
    left_base = left_ptr + pid_batch * stride_l_batch
    right_base = right_ptr + pid_batch * stride_r_batch
    out_base = out_ptr + pid_batch * stride_o_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = kk * BLOCK_K
        off_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = off_k < N

        # left block: shape (BLOCK_M, BLOCK_K)  – rows i, cols k
        a_ptrs = left_base + off_i[:, None] * stride_l_i + off_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right block (transposed): shape (BLOCK_K, BLOCK_N) – rows k, cols j
        b_ptrs = right_base + off_k[:, None] * stride_r_k + off_j[None, :] * stride_r_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write the result
    # --------------------------------------------------------------
    out_ptrs = out_base + off_i[:, None] * stride_o_i + off_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask:        torch.Tensor of shape [B, N, N] (bool or float)
            - weights:     dict of model weights (see keys in the problem statement)
            - config:      dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # Extract model parameters
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (projectors) – two GEMMs
    # --------------------------------------------------------------
    proj_weight = torch.cat([left_proj_weight, right_proj_weight], dim=0)   # (2*H, dim)
    proj = F.linear(x, proj_weight)                                        # [B,N,N,2*H]

    left = proj[..., :hidden_dim]                                          # [B,N,N,H]
    right = proj[..., hidden_dim:]                                         # [B,N,N,H]

    # --------------------------------------------------------------
    # 3️⃣ Gates – three GEMMs + sigmoid
    # --------------------------------------------------------------
    gate_weight = torch.cat([left_gate_weight, right_gate_weight, out_gate_weight], dim=0)  # (3*H, dim)
    gates = F.linear(x, gate_weight)                                                    # [B,N,N,3*H]

    left_gate = torch.sigmoid(gates[..., :hidden_dim])
    right_gate = torch.sigmoid(gates[..., hidden_dim:2 * hidden_dim])
    out_gate = torch.sigmoid(gates[..., 2 * hidden_dim:])  # [B,N,N,H]

    # --------------------------------------------------------------
    # 4️⃣ Optional mask
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 5️⃣ Apply gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Cast to fp16 for the core O(N³) reduction
    # --------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Free intermediate fp32 tensors to save memory
    del left, right, left_gate, right_gate, proj, gates, x

    # --------------------------------------------------------------
    # 7️⃣ Triton kernel: out_h = left_h @ right_hᵀ   (batched GEMM)
    # --------------------------------------------------------------
    B, N, _, _ = left_h.shape
    batch_hidden = B * hidden_dim

    # Reshape to (B*H, N, N) – contiguous layout
    left_h_view = left_h.permute(0, 3, 1, 2).contiguous().view(batch_hidden, N, N)
    right_h_view = right_h.permute(0, 3, 1, 2).contiguous().view(batch_hidden, N, N)

    # Allocate output buffer (fp16)
    out_h = torch.empty((batch_hidden, N, N), dtype=torch.float16, device=device)

    # Strides (in elements)
    stride_l_batch = left_h_view.stride(0)
    stride_l_i = left_h_view.stride(1)
    stride_l_k = left_h_view.stride(2)

    stride_r_batch = right_h_view.stride(0)
    stride_r_j = right_h_view.stride(1)
    stride_r_k = right_h_view.stride(2)

    stride_o_batch = out_h.stride(0)
    stride_o_i = out_h.stride(1)
    stride_o_j = out_h.stride(2)

    # Tiling parameters – tuned for typical B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_h_view,
        right_h_view,
        out_h,
        batch_hidden,
        N,
        stride_l_batch,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_j,
        stride_r_k,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, hidden] and convert to fp32
    # --------------------------------------------------------------
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 9️⃣ Final LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 🔟 Apply out‑gate and final linear projection
    # --------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
