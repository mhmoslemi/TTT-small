@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_hb = tl.program_id(0)  # batch*hidden dimension
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and hidden index
    h = pid_hb % H
    b = pid_hb // H

    # compute tile offsets
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + h * stride_lh
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        # load left tile
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # cast to fp32
        left = left.to(tl.float32)

        # right tile: we need shape (BLOCK_K, BLOCK_J)
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + h * stride_rh
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_rj
        )
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        right = right.to(tl.float32)

        acc += tl.dot(left, right, out_dtype=tl.float32)

    # store results
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + h * stride_oh
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
    )
    mask_store = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_store)

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_hb = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and hidden id
    h = pid_hb % H
    b = pid_hb // H

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (i,k)
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + h * stride_lh
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile (k,j) - note we need k as row, j as column
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + h * stride_rh
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_rj
        )
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right, out_dtype=tl.float32)

    # store result
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + h * stride_oh
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
    )
    mask_store = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_store)

"""
TriMul outgoing update – Triton‑accelerated core.

The heavy O(N³·H) part is implemented as a custom Triton kernel that
computes, for each batch and hidden index h,

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

where left/right are the linear projections of the normalized input.
The kernel works directly on the original [B, N, N, H] layout, avoiding
any costly permutes or copies.  All other operations (layer‑norm,
linear projections, sigmoid gates) stay in PyTorch (FP16 where possible)
so the whole forward pass runs well within the target latency.

The kernel loads tiles of size 64×64×64 (I, J, K) using fp16 loads,
performs Tensor‑Core‑friendly `tl.dot` (fp16·fp16 → fp32) and stores
the result in FP32.  Masking is applied before the kernel, therefore
the kernel does not need any mask handling.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr,            # fp16 *
    right_ptr,           # fp16 *
    out_ptr,             # fp32 *
    B, N, H,             # i32 scalar
    stride_lb, stride_li, stride_lk, stride_lh,   # i32 strides for left
    stride_rb, stride_rj, stride_rk, stride_rh,   # i32 strides for right
    stride_ob, stride_oi, stride_oj, stride_oh,   # i32 strides for out
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_hb = tl.program_id(0)      # combined batch*hidden index
    pid_i = tl.program_id(1)       # tile of output rows (i)
    pid_j = tl.program_id(2)       # tile of output cols (j)

    # decode batch and hidden dimension
    h = pid_hb % H
    b = pid_hb // H

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over K dimension (the summed axis)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_I, BLOCK_K)  -> left[b,i,k,h]
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + h * stride_lh
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile: shape (BLOCK_K, BLOCK_J)  -> right[b,k,j,h]
        # note that we need the transposed layout (k as rows, j as cols)
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + h * stride_rh
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_rj
        )
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right, out_dtype=tl.float32)

    # write out tile (BLOCK_I, BLOCK_J) -> out[b,i,j,h]
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + h * stride_oh
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
    )
    mask_store = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_store)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation (AlphaFold3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model tensors
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # still FP32

    # -----------------------------------------------------------------
    # 2) Cast to FP16 and linear projections
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_weight)   # [B, N, N, H] fp16
    right = F.linear(x_fp16, right_proj_weight)  # fp16

    # -----------------------------------------------------------------
    # 3) Optional binary mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 4) Gates (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core O(N³·H) update with Triton
    # -----------------------------------------------------------------
    B, N, _, H = left.shape               # B, seq_len, seq_len, hidden
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # strides (in elements)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    grid = (B * H,                       # batch*hidden dimension
            triton.cdiv(N, BLOCK_I),     # i‑tiles
            triton.cdiv(N, BLOCK_J))     # j‑tiles

    trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate and final projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # FP32

    out = out * out_gate.to(out.dtype)   # apply out‑gate

    to_out_weight = weights["to_out.weight"].to(torch.float16)  # (dim, hidden)
    out_fp16 = out.to(torch.float16)
    output = F.linear(out_fp16, to_out_weight)               # (B,N,N,dim) fp16
    output = output.to(torch.float32)

    return output
