  linear projection, keeping the bulk of the work in FP16.

The kernel operates on tensors of shape [B, N, N, H] (H = hidden_dim) and
applies: out = Σ_k left[..., i, k, h] * right[..., j, k, h].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_proj_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Element‑wise fusion: out = proj * gate * mask
    All tensors are half‑precision. `mask` is broadcasted over the H dimension.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    total = B * N * N * H
    active = offs < total

    # decode linear index -> (b,i,j,h)
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem = rem % (N * H)
    j = rem // H
    h = rem % H

    # compute flat offsets for each tensor
    proj_off = b * stride_proj_b + i * stride_proj_i + j * stride_proj_j + h * stride_proj_h
    gate_off = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_off  = b * stride_out_b  + i * stride_out_i  + j * stride_out_j  + h * stride_out_h

    proj = tl.load(proj_ptr + proj_off, mask=active, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + gate_off, mask=active, other=tl.float16(0.0))
    m    = tl.load(mask_ptr + mask_off, mask=active, other=tl.float16(0.0))

    tl.store(out_ptr + out_off, proj * gate * m, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model weight tensors
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------- unpack inputs ------------------- #
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # ------------------- model weights ------------------- #
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ---------------------------------------------------- #
    # 1) Input LayerNorm (FP16 via autocast)
    # ---------------------------------------------------- #
    with torch.cuda.amp.autocast(enabled=True):
        x_norm = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

        # linear projections (FP16)
        left_proj = F.linear(x_norm, left_proj_w)          # [B,N,N,H] fp16
        right_proj = F.linear(x_norm, right_proj_w)

        # gates (FP16, sigmoid)
        left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))
        right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
        out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    # ---------------------------------------------------- #
    # 2) Fuse mask & gate with projection (Triton)
    # ---------------------------------------------------- #
    if mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1] and cast to fp16
        mask_h = mask.unsqueeze(-1).to(left_proj.dtype).contiguous()

        # make tensors contiguous for safe stride handling
        left_proj = left_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_proj = right_proj.contiguous()
        right_gate = right_gate.contiguous()
        mask_h = mask_h.contiguous()

        # allocate output buffers (half precision)
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # strides (in elements)
        s_lp = left_proj.stride()
        s_lg = left_gate.stride()
        s_rp = right_proj.stride()
        s_rg = right_gate.stride()
        s_m = mask_h.stride()
        s_out = left.stride()   # same for right

        total = B * N * N * hidden_dim
        BLOCK_SIZE = 1024
        grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE,)

        # left side fusion
        fused_proj_gate_mask_kernel[grid](
            left_proj,
            left_gate,
            mask_h,
            left,
            B, N, hidden_dim,
            s_lp[0], s_lp[1], s_lp[2], s_lp[3],
            s_lg[0], s_lg[1], s_lg[2], s_lg[3],
            s_m[0], s_m[1], s_m[2],
            s_out[0], s_out[1], s_out[2], s_out[3],
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # right side fusion
        fused_proj_gate_mask_kernel[grid](
            right_proj,
            right_gate,
            mask_h,
            right,
            B, N, hidden_dim,
            s_rp[0], s_rp[1], s_rp[2], s_rp[3],
            s_rg[0], s_rg[1], s_rg[2], s_rg[3],
            s_m[0], s_m[1], s_m[2],
            s_out[0], s_out[1], s_out[2], s_out[3],
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        # no mask – simple element‑wise multiplication (still FP16)
        left = (left_proj * left_gate).contiguous()
        right = (right_proj * right_gate).contiguous()

    # ---------------------------------------------------- #
    # 3) Batched N³ reduction via half‑precision GEMM
    # ---------------------------------------------------- #
    # reshape to ([B, H, N, N]) and prepare for bmm
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (k dimension transposed)

    B_, H_, N_, _ = left.shape
    # view as (B*H, N, N) matrices
    left_mat = left.view(B_ * H_, N_, N_)
    right_mat = right.view(B_ * H_, N_, N_)

    # half‑precision batched matrix multiplication (uses cuBLAS)
    out_mat = torch.bmm(left_mat, right_mat)       # (B*H, N, N), fp16

    # reshape back to [B, N, N, H] (still fp16)
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.float()                # cast to fp32 for LayerNorm

    # ---------------------------------------------------- #
    # 4) Output LayerNorm, output gate and final projection
    # ---------------------------------------------------- #
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # out_gate is fp16; promote to fp32 before multiplication
    out_hidden = out_hidden * out_gate.float()
    # final linear projection back to `dim` (fp32)
    out = F.linear(out_hidden, to_out_w)            # [B, N, N, dim]

    return out
```
