def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # compute projections and gates
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"])) # keep float32 for later

    left = left * left_gate
    right = right * right_gate

    # allocate output
    out = torch.empty_like(left, dtype=torch.float32)  # shape [B,N,N,H]

    # launch triton kernel
    B, N, _, H = left.shape
    # kernel parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H
    )

    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        H=H,
        N=N,
        stride_left_batch=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_k=left.stride(2),
        stride_left_d=left.stride(3),
        stride_right_batch=right.stride(0),
        stride_right_j=right.stride(1),
        stride_right_k=right.stride(2),
        stride_right_d=right.stride(3),
        stride_out_batch=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_d=out.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # out now contains [B,N,N,H] float32

    # final LayerNorm over hidden_dim
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)

    # Apply out_gate (need same dtype)
    out = out * out_gate

    # Final linear projection to dim
    out = F.linear(out, weights["to_out.weight"])

    return out

    mask_j = offs_j < N

"""
Outgoing Triangle Multiplicative Update (TriMul) kernel.

The heavy core of TriMul is a batched matrix multiplication:
   out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]
where `left` and `right` are the gated, (optionally masked) projections of the
input tensor.  We implement this reduction with a tiled Triton GEMM that
treats the hidden dimension `d` as a batch index, thereby avoiding the costly
reshape / permute used in naive implementations.  All other ops (LayerNorm,
linear projections, gating) are performed with plain PyTorch which is cheap
relative to the O(N³·hidden) work of the GEMM.

The kernel works on FP16 inputs (left/right) and accumulates in FP32,
achieving high throughput on modern GPUs (B200).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel for the TriMul core.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,          # fp16 tensor,   shape (B, N, N, H)
    right_ptr,         # fp16 tensor,   shape (B, N, N, H)
    out_ptr,           # fp32 tensor,   shape (B, N, N, H)
    B: tl.int32,       # batch size
    H: tl.int32,       # hidden dimension
    N: tl.int32,       # sequence length
    stride_left_batch: tl.int32,
    stride_left_i:     tl.int32,
    stride_left_k:     tl.int32,
    stride_left_d:     tl.int32,
    stride_right_batch: tl.int32,
    stride_right_j:    tl.int32,
    stride_right_k:    tl.int32,
    stride_right_d:    tl.int32,
    stride_out_batch: tl.int32,
    stride_out_i:     tl.int32,
    stride_out_j:     tl.int32,
    stride_out_d:     tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)                     # tile row
    pid_j = tl.program_id(1)                     # tile col
    pid   = tl.program_id(2)                     # combined (batch, hidden)

    # decode batch and hidden indices
    b = pid // H
    d = pid % H

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, d) slice
    left_base  = left_ptr  + b * stride_left_batch  + d * stride_left_d
    right_base = right_ptr + b * stride_right_batch + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_batch   + d * stride_out_d

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left[i, k, d]   -> (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + (offs_i[:, None] * stride_left_i) + (cur_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right[j, k, d] (transposed) -> (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + (cur_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_j)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # accumulate dot product (into FP32)
        acc += tl.dot(a, b).to(tl.float32)

    # write result
    c_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model parameters (see code)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 2. Linear projections (float32) → cast to FP16 for the core GEMM
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)           # [B, N, N, 1] float32
        mask_f = mask_f.to(torch.float16)                 # now FP16
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of a linear projection)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # keep FP32 for later

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core TriMul: batched GEMM left @ rightᵀ over the K dimension
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)  # [B, N, N, H] FP32 accumulator

    # Tiled launch configuration
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        H=H,
        N=N,
        stride_left_batch=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_k=left.stride(2),
        stride_left_d=left.stride(3),
        stride_right_batch=right.stride(0),
        stride_right_j=right.stride(1),
        stride_right_k=right.stride(2),
        stride_right_d=right.stride(3),
        stride_out_batch=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_d=out.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6. Final LayerNorm over hidden dimension, gating and output projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    out = out * out_gate  # apply outgoing gate (both FP32)

    out = F.linear(out, weights["to_out.weight"])  # back to original dim

    return out
