import triton
import triton.language as tl

@triton.jit
def tri_mul_forward_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_b_l, stride_i_l, stride_k_l, stride_d_l,
    stride_b_r, stride_i_r, stride_k_r, stride_d_r,
    stride_b_o, stride_i_o, stride_j_o, stride_d_o,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # block row i
    pid_n = tl.program_id(1)  # block col j
    pid_z = tl.program_id(2)  # batch * hidden
    
    # compute batch, hidden
    batch = pid_z // H
    hidden = pid_z % H

    # base offsets for this batch and hidden
    offset_batch = batch * stride_b_l
    offset_hidden = hidden * stride_d_l  # same for all

    # i and j offsets
    i_offset = pid_m * BLOCK_M
    j_offset = pid_n * BLOCK_N

    # create range vectors
    i_range = i_offset + tl.arange(0, BLOCK_M)
    j_range = j_offset + tl.arange(0, BLOCK_N)

    # masks for boundaries
    mask_i = i_range < N
    mask_j = j_range < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_tile in range(0, N, BLOCK_K):
        k_offset = k_tile
        k_range = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # Load left tile (i,k)
        left_ptrs = left_ptr + offset_batch + offset_hidden + i_range[:, None] * stride_i_l + k_range[None, :] * stride_k_l
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile (k,j)
        # Need to load as (k,j) to match dot.
        right_ptrs = right_ptr + offset_batch + offset_hidden + k_range[:, None] * stride_k_r + j_range[None, :] * stride_i_r
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(left, right)  # left: MxK, right: KxN

    # Store result
    out_ptrs = out_ptr + offset_batch + offset_hidden + i_range[:, None] * stride_i_o + j_range[None, :] * stride_j_o
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The bottleneck is the O(N³) reduction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
This is a batched matrix multiplication of many (N×N) matrices.
The kernel below computes all B*hidden_dim matmuls in a single Triton
kernel, loading FP16 tiles, accumulating in FP32 (tensor cores) and
storing the result directly as float32.  All other ops (LayerNorm,
linear projections, gating) are performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_forward_kernel(
    left_ptr, right_ptr, out_ptr,           # pointers
    B, H,                                   # batch size, hidden dim (runtime)
    stride_bl, stride_il, stride_kl, stride_dl,   # left strides
    stride_br, stride_ir, stride_kr, stride_dr,   # right strides
    stride_bo, stride_io, stride_jo, stride_do,   # out strides
    N: tl.constexpr,                        # sequence length (compile‑time)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)   # block row (i)
    pid_n = tl.program_id(1)   # block col (j)
    pid_z = tl.program_id(2)   # batch*hidden

    # ------------------------------------------------------------------
    # Decode batch and hidden indices
    # ------------------------------------------------------------------
    batch = pid_z // H
    hidden = pid_z % H

    # Base offsets for this (batch, hidden) pair
    offset_batch = batch * stride_bl
    offset_hidden = hidden * stride_dl          # stride_dl is 1 for contiguous layout

    # ------------------------------------------------------------------
    # Compute i/j tile offsets
    # ------------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    i_idx = i0 + tl.arange(0, BLOCK_M)
    j_idx = j0 + tl.arange(0, BLOCK_N)

    mask_i = i_idx < N
    mask_j = j_idx < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # ----- Load left tile (i, k) -----
        # Shape: (BLOCK_M, BLOCK_K)
        left_ptrs = (left_ptr
                     + offset_batch + offset_hidden
                     + i_idx[:, None] * stride_il
                     + k_idx[None, :] * stride_kl)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ----- Load right tile (k, j)  -----
        # We load it transposed so that we can use tl.dot directly.
        # Shape: (BLOCK_K, BLOCK_N)
        right_ptrs = (right_ptr
                      + offset_batch + offset_hidden
                      + k_idx[:, None] * stride_kr
                      + j_idx[None, :] * stride_ir)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # ----- Dot product accumulation -----
        # left: (M, K), right: (K, N) -> (M, N)
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # Write the result (FP32) back to global memory
    # ------------------------------------------------------------------
    out_ptrs = (out_ptr
                + offset_batch + offset_hidden
                + i_idx[:, None] * stride_io
                + j_idx[None, :] * stride_jo)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            - mask         : torch.Tensor [B, N, N]        (bool/float) or None
            - weights      : dict of model parameters (see skeleton)
            - config       : dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) Load model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim], float32
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # used later

    # ------------------------------------------------------------------
    # 5) Apply gates and cast to FP16 for the heavy reduction
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16).contiguous()
    right = (right * right_gate).to(torch.float16).contiguous()

    B, N, _, H = left.shape  # H == hidden_dim

    # ------------------------------------------------------------------
    # 6) Triton kernel: batched matmul over the k dimension
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # Strides (in elements) for the tensors
    stride_bl, stride_il, stride_kl, stride_dl = left.stride()
    stride_br, stride_ir, stride_kr, stride_dr = right.stride()
    stride_bo, stride_io, stride_jo, stride_do = out.stride()

    # Tile sizes – chosen to fit in registers and shared memory
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # Grid dimensions
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch‑hidden tiles
    )

    tri_mul_forward_kernel[grid](
        left,
        right,
        out,
        B,
        H,
        stride_bl, stride_il, stride_kl, stride_dl,
        stride_br, stride_ir, stride_kr, stride_dr,
        stride_bo, stride_io, stride_jo, stride_do,
        N,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Final LayerNorm, output gating and projection back to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
