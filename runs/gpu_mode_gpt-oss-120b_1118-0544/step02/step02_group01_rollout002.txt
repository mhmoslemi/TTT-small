@triton.jit
def fused_mask_gate_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                           B, N, H,
                           stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
                           stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
                           stride_mask_b, stride_mask_i, stride_mask_j,
                           stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                           BLOCK_SIZE: tl.constexpr):
   pid = tl.program_id(0)
   block_start = pid * BLOCK_SIZE
   offs = block_start + tl.arange(0, BLOCK_SIZE)
   within = offs < B * N * N * H

   # decode linear index to indices
   idx = offs
   b = idx // (N * N * H)
   idx = idx % (N * N * H)
   i = idx // (N * H)
   idx = idx % (N * H)
   j = idx // H
   h = idx % H

   # compute offsets
   proj_offset = b*stride_proj_b + i*stride_proj_i + j*stride_proj_j + h*stride_proj_h
   gate_offset = b*stride_gate_b + i*stride_gate_i + j*stride_gate_j + h*stride_gate_h
   mask_offset = b*stride_mask_b + i*stride_mask_i + j*stride_mask_j
   out_offset = b*stride_out_b + i*stride_out_i + j*stride_out_j + h*stride_out_h

   proj = tl.load(proj_ptr + proj_offset, mask=within, other=0.0)
   gate = tl.load(gate_ptr + gate_offset, mask=within, other=0.0)
   m = tl.load(mask_ptr + mask_offset, mask=within, other=0.0)

   tl.store(out_ptr + out_offset, proj * gate * m, mask=within)
```

       out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   We reshape the tensors so that the hidden dimension is merged into the batch
   dimension and use `torch.bmm` which maps to cuBLAS.
5. Apply a second LayerNorm, an output gate, and a final linear projection
   back to the original channel dimension.

All heavy element‑wise work (mask + gate fusion) is executed in the Triton kernel,
while the large matrix multiplication leverages cuBLAS for speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Element‑wise fusion kernel:
        out = proj * gate * mask   (mask broadcast over H)
    Arguments are raw pointers and strides in *elements* (not bytes).
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Determine which threads are within bounds.
    valid = offs < B * N * N * H

    # Decode a linear offset to (b, i, j, h)
    idx = offs
    b = idx // (N * N * H)
    idx = idx % (N * N * H)
    i = idx // (N * H)
    idx = idx % (N * H)
    j = idx // H
    h = idx % H

    # Compute flat memory offsets for each tensor.
    proj_off = (b * stride_proj_b + i * stride_proj_i +
                j * stride_proj_j + h * stride_proj_h)
    gate_off = (b * stride_gate_b + i * stride_gate_i +
                j * stride_gate_j + h * stride_gate_h)
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_off  = (b * stride_out_b + i * stride_out_i +
                j * stride_out_j + h * stride_out_h)

    # Load values (masked loads return 0.0 for out‑of‑bounds threads).
    proj = tl.load(proj_ptr + proj_off, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_off, mask=valid, other=0.0)

    # Store the fused result.
    tl.store(out_ptr + out_off, proj * gate * m, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32) or ``None``
        - weights      : dict mapping layer names to their parameters
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections and gating (bias‑free)
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 3) Fuse mask and gate (if mask is provided) via Triton
    # -----------------------------------------------------------------
    if mask is not None:
        # Ensure mask is float and on same device/dtype as other tensors.
        mask_f = mask.to(dtype=x.dtype, device=x.device)

        # Prepare output buffers
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # Strides in *elements*
        s_lp_b, s_lp_i, s_lp_j, s_lp_h = left_proj.stride()
        s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate.stride()
        s_rp_b, s_rp_i, s_rp_j, s_rp_h = right_proj.stride()
        s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate.stride()
        s_m_b, s_m_i, s_m_j = mask_f.stride()
        s_out_b, s_out_i, s_out_j, s_out_h = left.stride()

        total_elements = B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # Left side fusion
        fused_mask_gate_kernel[grid](
            left_proj, left_gate, mask_f, left,
            B, N, hidden_dim,
            s_lp_b, s_lp_i, s_lp_j, s_lp_h,
            s_lg_b, s_lg_i, s_lg_j, s_lg_h,
            s_m_b, s_m_i, s_m_j,
            s_out_b, s_out_i, s_out_j, s_out_h,
            BLOCK_SIZE=BLOCK,
        )
        # Right side fusion
        fused_mask_gate_kernel[grid](
            right_proj, right_gate, mask_f, right,
            B, N, hidden_dim,
            s_rp_b, s_rp_i, s_rp_j, s_rp_h,
            s_rg_b, s_rg_i, s_rg_j, s_rg_h,
            s_m_b, s_m_i, s_m_j,
            s_out_b, s_out_i, s_out_j, s_out_h,
            BLOCK_SIZE=BLOCK,
        )
    else:
        # No mask → simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 4) Core O(N³) multiplication via batched FP16 GEMM
    # -----------------------------------------------------------------
    # Cast to FP16 for the heavy matmul
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Reshape to (B*H, N, K) where K == N
    left_mat = left.permute(0, 3, 1, 2).reshape(-1, N, N)   # [B*H, N, N]
    right_mat = right.permute(0, 3, 2, 1).reshape(-1, N, N)  # [B*H, N, N]

    # Batched matrix multiply: (B*H, N, N) x (B*H, N, N) -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # FP16 result

    # Restore original layout and dtype
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # -----------------------------------------------------------------
    # 5) Output LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
