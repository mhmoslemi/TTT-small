@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_ak, stride_am, stride_an,   # for A: batch, m, k
    stride_bk, stride_bn, stride_bm,   # for B: batch, k, n (but we will store B as k, n)
    stride_ck, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # decode (batch, block_i, block_j)
    # There are TOTAL blocks = num_blocks_M * num_blocks_N * batch
    # compute as:
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_index = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_i = block_index // tl.cdiv(N, BLOCK_N)
    block_j = block_index % tl.cdiv(N, BLOCK_N)

    # Offsets
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # Clamps
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute pointers for A and B at the start of this block
    # A: shape (M,K)
    # B: shape (K,N)
    a_ptrs = A_ptr + batch * batch_stride_a + offs_m[:,None] * stride_am + tl.arange(0, BLOCK_K)[None,:] * stride_ak
    b_ptrs = B_ptr + batch * batch_stride_b + tl.arange(0, BLOCK_K)[:,None] * stride_bk + offs_n[None,:] * stride_bn

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles
        a = tl.load(a_ptrs, mask=mask_m[:,None] & (tl.arange(0,BLOCK_K) + k*BLOCK_K < K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0,BLOCK_K) + k*BLOCK_K < K)[:,None] & mask_n[None,:], other=0.0)
        # cast to FP32
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        # dot product
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store result
    c_ptrs = C_ptr + batch * batch_stride_c + offs_m[:,None] * stride_cm + offs_n[None,:] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:,None] & mask_n[None,:])

def batched_matmul_kernel(
    A, B, C,
    BATCH, M, N, K,
    stride_aa, stride_ab, stride_ac,
    stride_ba, stride_bb, stride_bc,
    stride_ca, stride_cb, stride_cc,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

def custom_kernel(data):
    input_tensor, mask, weights, cfg = data
    dim = cfg["dim"]; hidden_dim = cfg["hidden_dim"]
    B,N,_,_ = input_tensor.shape

    # 1. Input layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # 2. Projections
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # 3. Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # 4. Apply mask + gates, fusing via Triton (if mask else CPU)
    if mask is not None:
        # mask shape [B,N,N], need broadcast along hidden dim
        mask_f = mask.to(x.dtype)
        # allocate half tensors for left/right
        left = torch.empty_like(left_proj, dtype=torch.float16)
        right = torch.empty_like(right_proj, dtype=torch.float16)

        left_proj_c = left_proj.contiguous()
        left_gate_c = left_gate.contiguous()
        right_proj_c = right_proj.contiguous()
        right_gate_c = right_gate.contiguous()
        mask_c = mask_f.contiguous()

        # get strides
        s_lp_b, s_lp_i, s_lp_j, s_lp_h = left_proj_c.stride()
        s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate_c.stride()
        s_rp_b, s_rp_i, s_rp_j, s_rp_h = right_proj_c.stride()
        s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate_c.stride()
        s_m_b, s_m_i, s_m_j = mask_c.stride()
        s_out_b, s_out_i, s_out_j, s_out_h = left.stride()

        total = B * N * N * hidden_dim

            left_proj_c,
            left_gate_c,
            mask_c,
            left,
            B, N, hidden_dim,
            s_lp_b, s_lp_i, s_lp_j, s_lp_h,
            s_lg_b, s_lg_i, s_lg_j, s_lg_h,
            s_m_b, s_m_i, s_m_j,
            s_out_b, s_out_i, s_out_j, s_out_h,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        fused_mask_gate_kernel[grid](
            right_proj_c,
            right_gate_c,
            mask_c,
            right,
            B, N, hidden_dim,
            s_rp_b, s_rp_i, s_rp_j, s_rp_h,
            s_rg_b, s_rg_i, s_rg_j, s_rg_h,
            s_m_b, s_m_i, s_m_j,
            s_out_b, s_out_i, s_out_j, s_out_h,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        left = (left_proj * left_gate).to(torch.float16)
        right = (right_proj * right_gate).to(torch.float16)

    # 5. Permute and reshape for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # flatten batch*hidden into one dimension
    batch_hidden = B * hidden_dim
    left_mat = left_perm.view(batch_hidden, N, N)
    right_mat = right_perm.view(batch_hidden, N, N)

    # allocate output

    # Triton GEMM
    # compute strides
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    # GPU launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_hidden * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_hidden,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,

    # reshape to original shape
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # convert to float32
    out_hidden = out_hidden.to(torch.float32)

    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])
    return out

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute batch, block i, j
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    block_i = block_id // num_blocks_n
    block_j = block_id % num_blocks_n

    # Offsets in M,N dimensions
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to start of A block (M,K)
    a_ptrs = A_ptr + batch * stride_am + offs_m[:, None] * stride_ak + tl.arange(0, BLOCK_K)[None, :] * stride_ab

    # Pointer to start of B block (K,N)
    b_ptrs = B_ptr + batch * stride_bk + tl.arange(0, BLOCK_K)[:, None] * stride_bb + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute K range start
        k_offset = kk * BLOCK_K
        # load A tile
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) + k_offset < K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) + k_offset < K)[:, None] & mask_n[None, :], other=0.0)
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_ab
        b_ptrs += BLOCK_K * stride_bb

    c_ptrs = C_ptr + batch * stride_cm + offs_m[:, None] * stride_cn + offs_n[None, :] * stride_cb
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.
The implementation fuses mask & gate elementwise multiplication in a small
Triton kernel and performs the heavy N³ reduction with a custom batched GEMM
kernel (B·hidden_dim independent matrix multiplications of size N×N).  All
other cheap ops (LayerNorm, linear projections, final projection) run in
PyTorch.  The output dtype is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Tiny kernel: elementwise (proj * gate * mask) → store as float16
# ----------------------------------------------------------------------
@triton.jit
def fused_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    valid = offs < total

    # Decode linear index → (b, i, j, h)
    tmp = offs
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # Compute flat offsets
    proj_off = b * stride_proj_b + i * stride_proj_i + j * stride_proj_j + h * stride_proj_h
    gate_off = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_off = b * stride_out_b + i * stride_out_i + j * stride_out_j + h * stride_out_h

    proj = tl.load(proj_ptr + proj_off, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=valid, other=0.0)

    tl.store(out_ptr + out_off, (proj * gate * m).to(tl.float16), mask=valid)


# ----------------------------------------------------------------------
# Batched GEMM kernel: (B*H) independent matmuls of (N,N,N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # compute block indices
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    block_i = block_id // num_blocks_n
    block_j = block_id % num_blocks_n

    # offsets within the block
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the current block in A and B
    a_ptrs = (
        A_ptr
        + batch_idx * stride_a_batch
        + offs_m[:, None] * stride_a_m
        + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    )
    b_ptrs = (
        B_ptr
        + batch_idx * stride_b_batch
        + tl.arange(0, BLOCK_K)[:, None] * stride_b_k
        + offs_n[None, :] * stride_b_n
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) + k_offset < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(tl.arange(0, BLOCK_K) + k_offset < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store result as float16
    c_ptrs = (
        C_ptr
        + batch_idx * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    Arguments:
        data = (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right_proj = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Gates (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # (B,N,N,hidden)

    # -----------------------------------------------------------------
    # 4) Fuse mask & gate (Triton) → produce half‑precision tensors
    # -----------------------------------------------------------------
    if mask is not None:
        # ensure mask is float32 on same device
        mask_f = mask.to(dtype=x.dtype, device=device)

        # make everything contiguous
        left_proj_c = left_proj.contiguous()
        left_gate_c = left_gate.contiguous()
        right_proj_c = right_proj.contiguous()
        right_gate_c = right_gate.contiguous()
        mask_c = mask_f.contiguous()

        # allocate half‑precision output buffers
        left = torch.empty_like(left_proj_c, dtype=torch.float16)
        right = torch.empty_like(right_proj_c, dtype=torch.float16)

        # strides (in elements)
        s_lp_b, s_lp_i, s_lp_j, s_lp_h = left_proj_c.stride()
        s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate_c.stride()
        s_rp_b, s_rp_i, s_rp_j, s_rp_h = right_proj_c.stride()
        s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate_c.stride()
        s_m_b, s_m_i, s_m_j = mask_c.stride()
        s_out_b, s_out_i, s_out_j, s_out_h = left.stride()

        total = B * N * N * hidden_dim
        BLOCK_SIZE = 1024
        grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE,)

        # left side
        fused_mask_gate_kernel[grid](
            left_proj_c,
            left_gate_c,
            mask_c,
            left,
            B,
            N,
            hidden_dim,
            s_lp_b,
            s_lp_i,
            s_lp_j,
            s_lp_h,
            s_lg_b,
            s_lg_i,
            s_lg_j,
            s_lg_h,
            s_m_b,
            s_m_i,
            s_m_j,
            s_out_b,
            s_out_i,
            s_out_j,
            s_out_h,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # right side
        fused_mask_gate_kernel[grid](
            right_proj_c,
            right_gate_c,
            mask_c,
            right,
            B,
            N,
            hidden_dim,
            s_rp_b,
            s_rp_i,
            s_rp_j,
            s_rp_h,
            s_rg_b,
            s_rg_i,
            s_rg_j,
            s_rg_h,
            s_m_b,
            s_m_i,
            s_m_j,
            s_out_b,
            s_out_i,
            s_out_j,
            s_out_h,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        # No mask: simple elementwise product followed by cast to FP16
        left = (left_proj * left_gate).to(torch.float16)
        right = (right_proj * right_gate).to(torch.float16)

    # -----------------------------------------------------------------
    # 5) Batched GEMM (N³ reduction) using Triton
    # -----------------------------------------------------------------
    # Permute to bring hidden dim upfront and reshape to (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) with (k,j) order

    batch_hidden = B * hidden_dim
    left_mat = left_perm.view(batch_hidden, N, N)
    right_mat = right_perm.view(batch_hidden, N, N)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_hidden * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_hidden,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden)
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, gate, final projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])
    return out
