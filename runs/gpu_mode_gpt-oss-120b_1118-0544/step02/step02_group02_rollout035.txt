        left  = (left  * left_gate ) * mask
        right = (right * right_gate) * mask
   The tensors are cast to float16 before the heavy reduction.
4. Perform the core reduction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication L @ Rᵀ for each (batch, head).
   The reduction is executed by a custom Triton kernel that works on a
   (BLOCK_M × BLOCK_N) tile of the output, loads tiles of L and Rᵀ with
   arbitrary strides, accumulates in float32 and writes the result directly
   in float32 (no extra cast kernel required).
5. Permute the result back to (B, N, N, hidden_dim), apply a second
   LayerNorm, the output gate, and the final Linear projection to the original
   dimensionality.

The Triton kernel fuses the whole N³ reduction for every head, avoiding the
expensive contiguous copies required by the naïve torch.bmm approach.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    # Pointers
    left_ptr, right_ptr, out_ptr,
    # Sizes
    B, H, N,
    # Strides (in elements)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    # Compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: out = left @ rightᵀ  (left,right: [B,H,N,N])."""
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_z = tl.program_id(2)   # batch*head index

    # ------------------------------------------------------------------
    # Compute batch and head indices from linear pid_z
    # ------------------------------------------------------------------
    batch_idx = pid_z // H
    head_idx  = pid_z % H

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (batch,head) matrix pair
    left_base  = left_ptr  + batch_idx * stride_l_b + head_idx * stride_l_h
    right_base = right_ptr + batch_idx * stride_r_b + head_idx * stride_r_h
    out_base   = out_ptr   + batch_idx * stride_o_b + head_idx * stride_o_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension (the summed axis)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # --------------------------------------------------------------
        # Load a tile of left: shape (BLOCK_M, BLOCK_K)
        # --------------------------------------------------------------
        ptrs_l = left_base + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        mask_l = mask_m[:, None] & mask_k[None, :]
        a = tl.load(ptrs_l, mask=mask_l, other=0.0).to(tl.float32)

        # --------------------------------------------------------------
        # Load a transposed tile of right (Rᵀ) : shape (BLOCK_K, BLOCK_N)
        #   Rᵀ[k, j] = right[j, k]
        # --------------------------------------------------------------
        ptrs_r = right_base + offs_n[None, :] * stride_r_i + offs_k[:, None] * stride_r_k
        mask_r = mask_k[:, None] & mask_n[None, :]
        b = tl.load(ptrs_r, mask=mask_r, other=0.0).to(tl.float32)

        # --------------------------------------------------------------
        # Rank‑K update
        # --------------------------------------------------------------
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the block back to global memory
    # ------------------------------------------------------------------
    out_ptrs = out_base + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    mask_out = mask_m[:, None] & mask_n[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a fused Triton kernel.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict with the model parameters (see skeleton)
            - config: dict with at least "dim", "hidden_dim" and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Gating vectors
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Apply per‑element gates
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Cast to float16 for the heavy reduction (Tensor‑Core friendly)
    # ------------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 7) Prepare tensors for the batched GEMM: (B, H, N, N)
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2)   # [B, H, N, N]  (non‑contiguous)
    right_perm = right.permute(0, 3, 1, 2)

    B, N, _, _ = input_tensor.shape
    H = hidden_dim

    # Output buffer (float32, because we accumulate in FP32)
    out_fp32 = torch.empty((B, H, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 8) Gather strides (in elements) for the kernel
    # ------------------------------------------------------------------
    l_s_b, l_s_h, l_s_i, l_s_k = left_perm.stride()
    r_s_b, r_s_h, r_s_i, r_s_k = right_perm.stride()
    o_s_b, o_s_h, o_s_i, o_s_j = out_fp32.stride()

    # ------------------------------------------------------------------
    # 9) Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    tri_mul_kernel[grid](
        left_ptr=left_perm,
        right_ptr=right_perm,
        out_ptr=out_fp32,
        B=B,
        H=H,
        N=N,
        stride_l_b=l_s_b,
        stride_l_h=l_s_h,
        stride_l_i=l_s_i,
        stride_l_k=l_s_k,
        stride_r_b=r_s_b,
        stride_r_h=r_s_h,
        stride_r_i=r_s_i,
        stride_r_k=r_s_k,
        stride_o_b=o_s_b,
        stride_o_h=o_s_h,
        stride_o_i=o_s_i,
        stride_o_j=o_s_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 10) Rearrange back to (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    out = out_fp32.permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 11) Second LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                      # apply output gate
    out = F.linear(out, to_out_weight)        # [B, N, N, dim]

    return out
```
