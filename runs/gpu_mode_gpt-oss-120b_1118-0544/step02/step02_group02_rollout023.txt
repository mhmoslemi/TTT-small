def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']; hidden_dim = config['hidden_dim']; nomask = config.get('nomask', False)

    # LayerNorm x
    x = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=1e-5)

    # Projections
    left = F.linear(x, weights['left_proj.weight'])
    right = F.linear(x, weights['right_proj.weight'])

    # Mask
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights['left_gate.weight']))
    right_gate = torch.sigmoid(F.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(F.linear(x, weights['out_gate.weight']))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to fp16
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # Permute and reshape
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B*hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B*hidden_dim, N, N)

    # Allocate output
    out_fp32 = torch.empty_like(left_flat, dtype=torch.float32) # shape (B*hidden_dim, N, N)

    # Get strides
    a_ptr = left_flat.data_ptr()
    b_ptr = right_flat.data_ptr()
    c_ptr = out_fp32.data_ptr()

    M = N
    K = N
    # Strides for A
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    # For B transposed:
    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)  # stride along k as row (original last dim)
    stride_b_n = right_flat.stride(1)  # stride along n (original j)
    # For C
    stride_c_batch, stride_c_m, stride_c_n = out_fp32.stride()

    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute grid
    grid = (B*hidden_dim, (M + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    batched_matmul_kernel[grid](
        a_ptr, b_ptr, c_ptr,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Now out_fp32 shape (B*hidden_dim,N,N). Reshape back
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layer norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'], eps=1e-5)

    # Apply out_gate (shape B,N,N,hidden_dim) after gating: we have out_gate shape B,N,N,hidden_dim (since we compute earlier)
    out = out * out_gate

    # Final linear projection
    out = F.linear(out, weights['to_out.weight'])
    return out

@triton.jit
def batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Compute pointer offsets
        a_ptrs = A + pid_batch * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        b_ptrs = B + pid_batch * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

        # Load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Store
    c_ptrs = C + pid_batch * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    A, B, C,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,

    stride_Cb, stride_Cm, stride_Cn,

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy computation is the reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication of shape
    (B*D, N, N) @ (B*D, N, N).T
for D = hidden_dim.  This kernel implements the batched GEMM in Triton
with FP16 inputs and FP32 accumulation, eliminating the extra cast kernel
used previously.  All cheap operations (LayerNorm, linear projections,
gate sigmoids) are kept in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A,               # fp16 tensor [BHD, M, K]
    B,               # fp16 tensor [BHD, K, N]  (here K=N, N=N)
    C,               # fp32 tensor [BHD, M, N]  (output)
    M, N, K,         # int scalar sizes
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (B is stored as K,N after transposition)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch * hidden_dim index
    pid_m = tl.program_id(1)   # block row of C
    pid_n = tl.program_id(2)   # block column of C

    # ------------------ tile offsets ------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------ accumulate ------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers with proper strides
        a_ptrs = A + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        b_ptrs = B + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

        # load (fp16 -> fp32)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # matrix‑multiply‑accumulate
        acc += tl.dot(a, b)

    # ------------------ store result ------------------
    c_ptrs = C + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model weights (see skeleton)
            - config: dict containing at least:
                * "dim" (int)
                * "hidden_dim" (int)
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Cast to FP16 for the heavy matmul
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 7) Rearrange to [batch*hidden_dim, N, N] (contiguous)
    # ------------------------------------------------------------------
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # 8) Allocate output buffer (FP32)
    # ------------------------------------------------------------------
    out_fp32 = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 9) Launch Triton batched GEMM kernel
    # ------------------------------------------------------------------
    M = N
    K = N

    # strides for the flattened tensors (contiguous layout)
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # row stride after transpose (k dimension)
    stride_b_n = right_flat.stride(1)   # column stride (original j)
    stride_c_batch, stride_c_m, stride_c_n = out_fp32.stride()

    # Tunable tile sizes (experimentally good for B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_fp32,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 10) Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 11) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
