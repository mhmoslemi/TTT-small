"""
TriMul outgoing update (AlphaFold3) – Triton‑accelerated forward pass.

The implementation follows the reference PyTorch module:
1. Layer‑norm the input tensor `x` (shape [B, N, N, D]).
2. Linear projections `left` and `right` to a hidden dimension H.
3. (Optional) binary mask multiplication.
4. Three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) applied element‑wise.
5. Core O(N³) update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matmul `left @ rightᵀ` for each (b, h) pair.
   This heavy step is performed with a custom Triton kernel that
   uses fp16·fp16 → fp32 accumulation.
6. Layer‑norm on the hidden result, gated by `out_gate`,
   and a final linear projection back to the original dimension D.
All heavy arithmetic is done in fp16, the final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel: C = A @ B   (A: (B, M, K), B: (B, K, N))
# Uses fp16 inputs, fp32 accumulation.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                      # matrix sizes (int32)
    stride_az, stride_am, stride_ak,  # A strides
    stride_bz, stride_bk, stride_bn,  # B strides
    stride_cz, stride_cm, stride_cn,  # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)   # batch dimension (B*H)
    pid_m = tl.program_id(0)       # block row
    pid_n = tl.program_id(1)       # block column

    # -------------------- compute block offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- loop over K dimension --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current tile
        a_ptrs = (
            A_ptr
            + pid_batch * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        b_ptrs = (
            B_ptr
            + pid_batch * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # load with zero padding where out‑of‑bounds
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # fused fp16·fp16 → fp32 dot product
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # -------------------- write result --------------------
    c_ptrs = (
        C_ptr
        + pid_batch * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)


def batched_gemm_fp16(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.

    A: (batch, M, K)  fp16, contiguous
    B: (batch, K, N)  fp16, contiguous
    Returns C: (batch, M, N)  fp32
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # strides in elements
    stride_az, stride_am, stride_ak = A.stride()
    stride_bz, stride_bk, stride_bn = B.stride()
    stride_cz, stride_cm, stride_cn = C.stride()

    # block sizes (heuristic for RTX‑A6000/B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )

    batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    return C


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D]   (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim], fp32

    # Cast to fp16 for the heavy linear work
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)   # [B, N, N, H] fp16
    right = F.linear(x, right_proj_w)  # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 3. Optional binary mask
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask shape [B, N, N] -> broadcast to hidden dim
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,H] fp16
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # [B,N,N,H] fp16
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))     # [B,N,N,H] fp16

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N^3) update via batched GEMM
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Rearrange to (B, H, N, N) and make contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Transpose right on the last two dimensions for standard GEMM
    right_T = right.permute(0, 1, 3, 2).contiguous()  # (B, H, N, N)

    # Merge batch and hidden dimensions
    batch_hidden = B * hidden_dim
    left_flat = left.view(batch_hidden, N, N)          # (BH, N, N)
    right_flat = right_T.view(batch_hidden, N, N)      # (BH, N, N)

    # Perform batched GEMM (fp16 inputs → fp32 output)
    out_flat = batched_gemm_fp16(left_flat, right_flat)  # (BH, N, N) fp32

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + out gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # fp32

    # Apply out_gate (cast to fp32)
    out = out * out_gate.to(out.dtype)

    # ------------------------------------------------------------------
    # 7. Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    out_fp16 = out.to(torch.float16)
    output_fp16 = F.linear(out_fp16, to_out_w)   # (B,N,N,dim) fp16
    output = output_fp16.to(torch.float32)

    return output
