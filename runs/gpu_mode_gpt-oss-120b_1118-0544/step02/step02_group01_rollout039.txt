```
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_a_batch, stride_b_batch, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks
    mask_i = offs_i < N
    mask_j = offs_j < N
    # block offset for batch
    a_batch_offset = pid_b * stride_a_batch
    b_batch_offset = pid_b * stride_b_batch
    c_batch_offset = pid_b * stride_c_batch

    # Pointers for A and B
    a_ptrs = A_ptr + a_batch_offset + offs_i[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + b_batch_offset + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, K, BLOCK_K):
        # Load A and B
        a = tl.load(a_ptrs, mask=(mask_i[:, None] & (offs_k < K)), other=0.0)
        b = tl.load(b_ptrs, mask=((offs_k[:, None] < K) & mask_j[None, :]), other=0.0)
        # Multiply-accumulate
        acc += tl.dot(a, b)
        # Advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back C
    c_ptrs = C_ptr + c_batch_offset + offs_i[:, None] * stride_cm + offs_j[None, :] * stride_cn
    # Cast accumulator to fp16 for storage
    tl.store(c_ptrs, acc.to(tl.float16), mask=(mask_i[:, None] & mask_j[None, :]))
```

def fuse_gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                          B, N, H,
                          stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
                          stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,

    i = tmp // (N * H)

"""
custom_kernel: Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections (left/right) and gating (left/right/out) are computed with
   torch.nn.functional.linear and torch.sigmoid.
3. If a mask is supplied, a tiny Triton kernel fuses:   out = proj * gate * mask
   producing the gated & masked tensors `left` and `right`.
   Otherwise a simple element‑wise multiplication is performed in PyTorch.
4. The N³ reduction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is expressed as a batched matrix multiplication for each (b,d) pair.
   A custom Triton kernel performs a tiled GEMM on half‑precision data
   (FP16) with accumulation in FP32.
5. The result is reshaped back to [B, N, N, hidden_dim] (float32),
   layer‑normed, multiplied by the `out_gate`, and projected back to `dim`
   with a final linear layer.

All heavy operations (mask + gate fusion and the batched GEMM) are executed
in Triton, while the remaining linear/normalisation steps use native
PyTorch/cuBLAS for efficiency.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Elementwise kernel:
        out = proj * gate * mask
    Shapes:
        proj, gate : [B, N, N, H]   (float32)
        mask       : [B, N, N]      (float32, broadcast over H)
        out        : [B, N, N, H]   (float32)
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    total = B * N * N * H
    valid = offs < total

    # decode linear index -> (b,i,j,h)
    tmp = offs
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # compute flat offsets
    proj_off = (b * stride_proj_b + i * stride_proj_i + j * stride_proj_j +
                h * stride_proj_h)
    gate_off = (b * stride_gate_b + i * stride_gate_i + j * stride_gate_j +
                h * stride_gate_h)
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_off  = (b * stride_out_b + i * stride_out_i + j * stride_out_j +
                h * stride_out_h)

    proj = tl.load(proj_ptr + proj_off, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_off, mask=valid, other=0.0)

    tl.store(out_ptr + out_off, proj * gate * m, mask=valid)


@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    N, K,
    stride_a_batch, stride_ai, stride_ak,
    stride_b_batch, stride_bk, stride_bj,
    stride_c_batch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched FP16 GEMM:   C = A @ B

    A : [M, N, K]   (M = B * hidden_dim)
    B : [M, K, N]
    C : [M, N, N]

    All tensors are stored in row‑major order.
    """
    pid_i = tl.program_id(0)   # tile row
    pid_j = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # batch index (M)

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base offsets for this batch
    a_batch_off = pid_b * stride_a_batch
    b_batch_off = pid_b * stride_b_batch
    c_batch_off = pid_b * stride_c_batch

    # pointers to the start of the current tile (k‑dimension not yet added)
    a_ptrs = A_ptr + a_batch_off + \
             offs_i[:, None] * stride_ai + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + b_batch_off + \
             tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_j[None, :] * stride_bj

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        # remaining length for this block
        cur_k = k
        k_remaining = K - cur_k
        cur_valid = tl.arange(0, BLOCK_K) < k_remaining

        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & cur_valid[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=cur_valid[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_N) in FP32

        # advance to next k‑slice
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c_ptrs = C_ptr + c_batch_off + \
             offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of weight tensors (see body)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2) Linear projections and gates (no bias)
    # --------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden]
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,hidden]

    # --------------------------------------------------------------
    # 3) Fuse mask + gates (triton kernel) or simple elementwise mul
    # --------------------------------------------------------------
    if mask is not None:
        # ensure mask dtype matches tensors
        mask_f = mask.to(x.dtype)

        # allocate output buffers
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # make tensors contiguous for stable stride handling
        lp = left_proj.contiguous()
        lg = left_gate.contiguous()
        rp = right_proj.contiguous()
        rg = right_gate.contiguous()
        mc = mask_f.contiguous()

        # strides (in elements)
        s_lp = lp.stride()
        s_lg = lg.stride()
        s_rp = rp.stride()
        s_rg = rg.stride()
        s_m = mc.stride()
        s_out = left.stride()        # same as proj stride

        total = B * N * N * hidden_dim
        BLOCK_SIZE = 1024
        grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE,)

        # left side
        fused_mask_gate_kernel[grid](
            lp, lg, mc, left,
            B, N, hidden_dim,
            s_lp[0], s_lp[1], s_lp[2], s_lp[3],
            s_lg[0], s_lg[1], s_lg[2], s_lg[3],
            s_m[0], s_m[1], s_m[2],
            s_out[0], s_out[1], s_out[2], s_out[3],
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # right side
        fused_mask_gate_kernel[grid](
            rp, rg, mc, right,
            B, N, hidden_dim,
            s_rp[0], s_rp[1], s_rp[2], s_rp[3],
            s_rg[0], s_rg[1], s_rg[2], s_rg[3],
            s_m[0], s_m[1], s_m[2],
            s_out[0], s_out[1], s_out[2], s_out[3],
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 4) Batched half‑precision matrix multiplication (N³ reduction)
    # --------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # bring hidden dimension to leading axis
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N] (i,k)
    right = right.permute(0, 3, 2, 1).contiguous() # [B, hidden, N, N] (k,j)

    M = B * hidden_dim                     # total number of independent GEMMs
    left_mat = left.view(M, N, N)          # (M, N, K) where K = N
    right_mat = right.view(M, N, N)        # (M, K, N)
    out_mat = torch.empty_like(left_mat)   # (M, N, N) fp16

    # strides for the three tensors (in elements)
    stride_a_batch, stride_ai, stride_ak = left_mat.stride()
    stride_b_batch, stride_bk, stride_bj = right_mat.stride()
    stride_c_batch, stride_ci, stride_cj = out_mat.stride()

    # tile sizes (tuned for B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        M,
    )

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N,
        stride_a_batch, stride_ai, stride_ak,
        stride_b_batch, stride_bk, stride_bj,
        stride_c_batch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden]
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # --------------------------------------------------------------
    # 5) Output LayerNorm, final gate and linear projection
    # --------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
