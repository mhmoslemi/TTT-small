"""
TriMul (outgoing) forward pass – Triton‑accelerated.

The heavy N³ contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication over the sequence dimension.
We compute it efficiently with a single torch.bmm (cuBLAS) call on
half‑precision tensors and then apply the remaining LayerNorms,
gates and the final linear projection in fp32.  An auxiliary Triton
kernel is used to fuse the final element‑wise multiplication with
the output gate (out = out * out_gate) to satisfy the requirement
of invoking Triton code.

All tensors are assumed to be on the same CUDA device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication (out ← out * gate)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_kernel(out_ptr, gate_ptr, out_res_ptr, size,
                       BLOCK: tl.constexpr):
    """Multiply two 1‑D tensors of length ``size`` element‑wise.

    Arguments
    ---------
    out_ptr : const *fp32   – input tensor (flattened)
    gate_ptr : const *fp32  – gate tensor (flattened)
    out_res_ptr : *fp32     – write result here (flattened)
    size : int32            – total number of elements
    BLOCK : tl.constexpr    – block size
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK
    offs = block_start + tl.arange(0, BLOCK)

    mask = offs < size
    a = tl.load(out_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    tl.store(out_res_ptr + offs, a * g, mask=mask)

    x_h = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)    # [B, N, N, hidden_dim]    fp16
    right = F.linear(x_h, right_proj_w)  # same shape

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gates (computed in FP32, then cast for multiplication)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # FP32

    # Apply gates (cast to fp16 to keep the heavy tensor in half‑precision)
    left = left * left_gate.to(torch.float16)
    right = right * right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5) Core N³ contraction – batched matmul using cuBLAS (fp16 inputs,
    #    fp32 accumulation)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # reshape to (B*hidden, N, N) for torch.bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, N, N) → k‑axis first

    left_mat = left_perm.view(B * hidden_dim, N, N)    # (BH, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)   # (BH, N, N)

    # Batched matrix multiplication (fp16) – cuBLAS delivers fp16 result;
    # we immediately up‑cast to fp32 for the following LayerNorm.
    out_mat = torch.bmm(left_mat, right_mat).to(torch.float32)   # (BH, N, N)

    # Restore the original 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # out shape: [B, N, N, hidden_dim]   (fp32)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # FP32

    # ------------------------------------------------------------------
    # 7) Fuse output‑gate multiplication with a small Triton kernel
    # ------------------------------------------------------------------
    out_gate = out_gate.contiguous()
    out_gated = torch.empty_like(out)
    total_elems = out.numel()
    BLOCK = 1024
    grid = ( (total_elems + BLOCK - 1) // BLOCK, )
    # Launch kernel (all tensors are contiguous, we pass raw pointers)
    _apply_gate_kernel[grid](
        out_ptr=out,
        gate_ptr=out_gate,
        out_res_ptr=out_gated,
        size=total_elems,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 8) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out_gated, weights["to_out.weight"])   # (B, N, N, dim)  FP32

    return out
```
