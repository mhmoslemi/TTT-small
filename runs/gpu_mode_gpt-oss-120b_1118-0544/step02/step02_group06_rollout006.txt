"""
TriMul outgoing kernel (AlphaFold3 style) – fast batched GEMM with Triton.

The core operation is:
    out[b, i, j, d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right are affine projections of the input pair representation,
optionally masked and gated.  This can be expressed as a batch of
independent matrix multiplications:
    (B * hidden_dim) GEMMs of size (N, N) @ (N, N)^T.

We implement the GEMM entirely in Triton (no torch.bmm) to obtain
high throughput.  All surrounding linear layers, layer‑norms and gates
are performed with PyTorch (float32) and the tensors are cast to
float16 before the Triton GEMM.  The kernel works for any batch size B,
sequence length N and hidden dimension C_h (here called hidden_dim).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (A @ Bᵀ) for the TriMul core.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  tensor pointer  (batch, M, K)   fp16
    b_ptr,               # right tensor pointer  (batch, N, K)   fp16 (read as transposed)
    c_ptr,               # output tensor pointer (batch, M, N)   fp32
    batch: tl.constexpr, # total GEMM count = B * hidden_dim
    M: tl.constexpr,     # rows of A ( = N )
    N: tl.constexpr,     # cols of output ( = N )
    K: tl.constexpr,     # reduction dim ( = N )
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_m = tl.program_id(0)  # block row index
    pid_n = tl.program_id(1)  # block col index
    pid_b = tl.program_id(2)  # batch (gemm) index

    # compute offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this gemm
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # load A tile (M x K) – fp16
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # load Bᵀ tile (K x N) – fp16
        # B is stored as (batch, N, K); we read it as transposed
        b_ptrs = b_batch + (cur_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # accumulate; a,b are fp16, cast to fp32 for reduction
        acc += tl.dot(a, b).to(tl.float32)

    # store the result (fp32)
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point – performs the full forward pass of the outgoing
# Triangle Multiplicative Update.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update.

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – may be None
            - weights: dict containing model parameters:
                * norm.weight, norm.bias
                * left_proj.weight, right_proj.weight
                * left_gate.weight, right_gate.weight, out_gate.weight
                * to_out_norm.weight, to_out_norm.bias
                * to_out.weight
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden channel dimension C_h
                * "nomask" (bool, optional) – if True, mask is ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape: [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias) → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)          # fp32
    right = F.linear(x, right_proj_weight)        # fp32

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → broadcast to [B, N, N, 1]
        mask_f = mask.to(dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gate to projections
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to half for the GEMM
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, N, N, hidden_dim

    # Reshape to (B * hidden_dim, N, N) – each hidden slice becomes a separate GEMM
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # Triton batched GEMM: out = left_h @ right_hᵀ
    # ------------------------------------------------------------------
    batch_total = left_h.shape[0]      # B * hidden_dim
    M = N = K = N                       # square matrices

    # Choose block sizes – good defaults for most GPUs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Compute grid dimensions
    num_pid_m = (M + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m, num_pid_n, batch_total)

    # Strides in number of elements (not bytes)
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)

    stride_b_batch = right_h.stride(0)
    stride_b_k = right_h.stride(1)   # row stride of right_h (acts as K dimension after transposition)
    stride_b_n = right_h.stride(2)   # column stride (N dimension)

    # Allocate output buffer (fp32)
    out_h = torch.empty((batch_total, N, N), dtype=torch.float32, device=device)

    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    # Launch Triton kernel
    tri_mul_outgoing_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        batch=batch_total,
        M=M,
        N=N,
        K=K,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] (float32)
    # ------------------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)          # [B, N, N, hidden_dim]
        .contiguous()
    )  # dtype already float32

    # ------------------------------------------------------------------
    # Final LayerNorm over hidden_dim, gating, and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # apply outgoing gate
    out = F.linear(out, to_out_weight)       # project back to original dim

    return out
