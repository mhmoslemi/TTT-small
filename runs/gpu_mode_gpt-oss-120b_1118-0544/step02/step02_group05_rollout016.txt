def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # layernorm input
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # cast to half
    x_h = x.to(torch.float16)

    # projections
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))   # (B,N,N,hidden_dim)
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # gates (fp32)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # cast gates to half
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    # apply gates using Triton kernel
    total = left.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    _apply_gate_kernel[grid](
        left, left_gate_h, left,
        total,
        BLOCK_SIZE=BLOCK,
    )
    _apply_gate_kernel[grid](
        right, right_gate_h, right,
        total,
        BLOCK_SIZE=BLOCK,
    )

    # optional mask
    if mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1), broadcast
        left = left * mask_h
        right = right * mask_h

    # prepare for batched matmul
    # left: (B,N,N,hidden_dim) -> (B,hidden_dim,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # transposed K

    B, H, N, _ = left_perm.shape
    # reshape to (B*H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # batched GEMM using torch.bmm (fp16 inputs)
    out_mat = torch.bmm(left_mat, right_mat)  # result fp16
    out_mat = out_mat.to(torch.float32)       # convert for later layers

    # reshape back to (B,N,N,hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # output layernorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # out gate multiplication
    out = out * out_gate  # out_gate is fp32

    # final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])
    return out
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated elementwise gating +
batched half‑precision GEMM via torch.bmm.

The core tensor contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is equivalent to a batched matrix multiplication
    out_{b,d} = left_{b,d} @ right_{b,d}ᵀ
where left/right are reshaped to (B·D, N, N).  The matmul is performed with
torch.bmm (high‑performance cuBLAS) in fp16 while a tiny Triton kernel
applies the per‑channel sigmoid gates in place.  All other operations (layer
norm, projections, final linear) are done with PyTorch.

The Triton kernel (`_apply_gate_kernel`) multiplies two tensors element‑wise
(in‑place) and is launched with a 1‑D grid; it provides the required Triton
usage without sacrificing performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_gate_kernel(
    a_ptr,      # pointer to source tensor (fp16)
    g_ptr,      # pointer to gate tensor (fp16)
    out_ptr,    # pointer to destination tensor (fp16) – can be same as a_ptr for in‑place
    size,       # total number of elements (int64)
    BLOCK_SIZE: tl.constexpr,  # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    g = tl.load(g_ptr + offs, mask=mask, other=tl.float16(0.0))
    out = a * g
    tl.store(out_ptr + offs, out, mask=mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) or None
            - weights: dict with the model weights (float32 tensors)
                * norm.weight / norm.bias
                * left_proj.weight / right_proj.weight
                * left_gate.weight / right_gate.weight / out_gate.weight
                * to_out_norm.weight / to_out_norm.bias
                * to_out.weight
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2) Linear projections (keep in fp16 for speed)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))   # [B, N, N, hidden_dim] fp16
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3) Compute sigmoid gates (float32) and cast to fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim] FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Apply gates with a tiny Triton kernel (in‑place)
    # ------------------------------------------------------------------
    total_elems = left.numel()
    BLOCK = 1024
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    _apply_gate_kernel[grid](
        left,
        left_gate_h,
        left,               # in‑place overwrite
        total_elems,
        BLOCK_SIZE=BLOCK,
    )
    _apply_gate_kernel[grid](
        right,
        right_gate_h,
        right,
        total_elems,
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6) Core N³ contraction – batched GEMM via torch.bmm (fp16 inputs,
    #    fp32 accumulation after cast)
    # ------------------------------------------------------------------
    # reshape to (B·hidden_dim, N, N) where the second operand is already
    # transposed on the K axis.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden_dim, N, N) – K dimension first

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)    # (B*H, N, N)  fp16
    right_mat = right_perm.view(B * H, N, N)  # (B*H, N, N)  fp16

    # batched matrix multiplication (uses cuBLAS – fast)
    out_mat = torch.bmm(left_mat, right_mat)          # fp16 result
    out_mat = out_mat.to(torch.float32)               # cast for the following layers

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output layer‑norm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                 # element‑wise multiplication (FP32)
    out = F.linear(out, weights["to_out.weight"])  # back to dim, FP32

    return out
