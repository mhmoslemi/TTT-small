@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ah, stride_ak,
    stride_bk, stride_bn,
    stride_ch, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # compute program id for batch * blocks
    pid = tl.program_id(0)
    # total blocks per batch
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for current batch
    a_batch_ptr = a_ptr + batch_idx * batch_stride_a
    b_batch_ptr = b_ptr + batch_idx * batch_stride_b
    c_batch_ptr = c_ptr + batch_idx * batch_stride_c

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # A block load
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_ah + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k_offset + tl.arange(0, BLOCK_K) < K), other=0.0).to(tl.float32)
        # B block load in transposed layout
        b_ptrs = b_batch_ptr + ((k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=(k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float32)
        # dot product
        acc += tl.dot(a, b)  # a shape (BLOCK_M, BLOCK_K), b shape (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
    # write to C
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_ch + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16) if tl.dtype(c_ptrs) == tl.float16 else acc, mask=mask_m[:, None] & mask_n[None, :])

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) kernel.

The heavy O(N³) reduction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is performed with a custom Triton batched GEMM kernel that fuses the
outer batch (batch * hidden_dim) and the two spatial dimensions.  The rest
of the operations (LayerNorm, linear projections, gating) are kept in
PyTorch for simplicity.  The kernel uses fp16 for the large tensors and
accumulates in fp32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel: computes C = A @ Bᵀ for each batch.
# A : (batch, M, K)   – fp16
# B : (batch, N, K)   – fp16 (loaded as K×N transposed)
# C : (batch, M, N)   – fp32
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (runtime)
    stride_ah, stride_ak,              # A: stride for M and K
    stride_bk, stride_bn,              # B: stride for K and N
    stride_ch, stride_cn,              # C: stride for M and N
    batch_stride_a, batch_stride_b, batch_stride_c,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # Number of blocks per batch
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    # Decode batch index and block indices within the batch
    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the current batch
    a_batch = a_ptr + batch_idx * batch_stride_a
    b_batch = b_ptr + batch_idx * batch_stride_b
    c_batch = c_ptr + batch_idx * batch_stride_c

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block (M × K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_ah
                            + cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B block transposed (K × N) -> we need Bᵀ for the dot
        b_ptrs = b_batch + (cur_k[:, None] * stride_bk
                            + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        # Dot product over K
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # Write back results
    c_ptrs = c_batch + (offs_m[:, None] * stride_ch
                        + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored when "nomask" is True
            - weights: dict of model weights (all on the same CUDA device)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – projection hidden dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Prepare tensors for batched GEMM (fp16)
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, N, N, hidden_dim
    # (B, hidden_dim, N, N) -> (B*hidden_dim, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Allocate output tensor for the reduction (fp32)
    out_h = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 6️⃣ Launch the Triton batched GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    a_stride = left_h.stride()   # (batch, M, K)
    b_stride = right_h.stride()
    c_stride = out_h.stride()

    # Grid: one program per (batch * block_M * block_N)
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = (B * hidden_dim) * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    batched_gemm_kernel[grid](
        left_h.data_ptr(),
        right_h.data_ptr(),
        out_h.data_ptr(),
        N, N, N,
        a_stride[1], a_stride[2],
        b_stride[2], b_stride[1],
        c_stride[1], c_stride[2],
        a_stride[0], b_stride[0], c_stride[0],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)          # [B, N, N, hidden_dim]
        .contiguous()
    )  # float32

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply out_gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
