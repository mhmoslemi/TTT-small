@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH_M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bi, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)   # row block index for output
    pid_n = tl.program_id(1)   # col block index
    pid_b = tl.program_id(2)   # batch slice index

    # Starting indices
    start_i = pid_m * BLOCK_M
    start_j = pid_n * BLOCK_N

    # Offsets along i & j axes
    offs_i = start_i + tl.arange(0, BLOCK_M)
    offs_j = start_j + tl.arange(0, BLOCK_N)

    # Mask for out-of-range
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Compute base pointers for this batch slice
    a_base = a_ptr + pid_b * stride_am
    b_base = b_ptr + pid_b * stride_bm
    c_base = c_ptr + pid_b * stride_cm

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for kk in range(num_k):
        start_k = kk * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)

        mask_k = offs_k < K

        # Load A block (shape BLOCK_M x BLOCK_K)
        a_ptrs = a_base + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak

        # Load B_T block (shape BLOCK_K x BLOCK_N)

"""
TriMul (outgoing) forward pass with a custom Triton kernel for the
core N³ reduction. The algorithm follows the PyTorch reference:

1) LayerNorm over the last dimension.
2) Linear projections (left/right) and gating projections (sigmoid).
3) Optional mask is fused with projection and gating (elementwise).
4) Batched matrix multiplication:
   out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a Triton GEMM over (B*hidden) batches, using FP16
   for inputs and accumulator.
5) Output LayerNorm, gating, and final linear projection back to dim.
The kernel processes all batches in one launch (grid_z = B*hidden_dim)
and uses block‑wise tiles (128×128×32) for high occupancy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_cnt, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bi, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Arguments:
        a_ptr, b_ptr, c_ptr: pointers to [batch_cnt, N, N] tensors (FP16)
        batch_cnt          : total number of independent GEMMs (= B*hidden_dim)
        N, K               : both equal to the sequence length
        stride_*           : strides (in elements) for each dimension
        BLOCK_*            : compile‑time tile sizes
    Computes c = a @ bᵀ for each batch slice.
    """
    pid_i = tl.program_id(0)          # block row index (i)
    pid_j = tl.program_id(1)          # block col index (j)
    pid_b = tl.program_id(2)          # batch slice index (0 .. batch_cnt-1)

    # Offsets for i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this batch slice
    a_base = a_ptr + pid_b * stride_am
    b_base = b_ptr + pid_b * stride_bm
    c_base = c_ptr + pid_b * stride_cm

    # Accumulator in FP16 (sufficient for N≤1024, hidden_dim≤128)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # Loop over the reduction dimension k
    num_k = tl.cdiv(K, BLOCK_K)
    for kk in range(num_k):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a block of A (M×K)
        a = tl.load(
            a_base + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a block of Bᵀ (K×N).  Bᵀ row = k, col = j  -> original B[j,k]
        b = tl.load(
            b_base + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bi,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # Accumulate the dot product
        acc += tl.dot(a, b)

    # Write the result block to C
    c = c_base + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model weights (as in the reference code)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------------------------------------------------
    # unpack inputs
    # -------------------------------------------------------------
    x, mask, weights, cfg = data
    dim, hidden_dim = cfg["dim"], cfg["hidden_dim"]
    B, N, _, _ = x.shape

    # -------------------------------------------------------------
    # 1) Input LayerNorm
    # -------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])       # [B, N, N, H]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------
    # 3) Gating (sigmoid) and optional mask fusion
    # -------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    if mask is not None and mask.numel() > 0:
        mask = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Release intermediate tensors not needed further
    del left_proj, right_proj, left_gate, right_gate

    # -------------------------------------------------------------
    # 4) Batched GEMM for the N³ reduction (implemented in Triton)
    # -------------------------------------------------------------
    # Cast to FP16 for the kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange to [B, H, N, N] and flatten batch×hidden into one dimension
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    batch_cnt = B * hidden_dim
    left_mat = left.view(batch_cnt, N, N)          # [BH, N, N]  FP16
    right_mat = right.view(batch_cnt, N, N)

    # Allocate output buffer (FP16, will be cast to FP32 later)
    out_hidden = torch.empty_like(left_mat, dtype=torch.float16)

    # Strides in element units
    a_stride_m, a_stride_i, a_stride_k = left_mat.stride()
    b_stride_m, b_stride_i, b_stride_k = right_mat.stride()
    c_stride_m, c_stride_i, c_stride_j = out_hidden.stride()

    # Tile sizes – tuned for Ampere GPUs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: (row blocks, col blocks, batch slices)
    grid_x = (N + BLOCK_M - 1) // BLOCK_M
    grid_y = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = batch_cnt
    grid = (grid_x, grid_y, grid_z)

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_hidden,
        batch_cnt,
        N,
        N,
        a_stride_m,
        a_stride_i,
        a_stride_k,
        b_stride_m,
        b_stride_i,
        b_stride_k,
        c_stride_m,
        c_stride_i,
        c_stride_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert result back to FP32 and reshape to [B, N, N, H]
    out_hidden = out_hidden.to(torch.float32)
    out_hidden = out_hidden.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------
    # 5) Output LayerNorm, gating, and final projection
    # -------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
```
