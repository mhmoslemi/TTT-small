"""
TriMul (outgoing) forward pass – Triton implementation.

The heavy part of TriMul is the N³ reduction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which can be expressed as a batched matrix multiplication
    out_{b,d} = left_{b,d} @ right_{b,d}ᵀ .
We perform the layer‑norms, linear projections and sigmoid gates with
PyTorch (highly‑optimized cuBLAS kernels) and launch a custom Triton
kernel for the batched GEMM (fp16 inputs, fp32 accumulation).  The kernel
tiles the i‑ (M) and j‑ (N) axes (BLOCK_M × BLOCK_N) and reduces over the
k‑axis in blocks (BLOCK_K).  Autotuning selects a good tiling configuration
for the given shape.

The final steps (output LayerNorm, output‑gate scaling and linear projection)
are performed again with PyTorch.  The complete function follows the required
signature `custom_kernel(data)`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
# A : (BH, M, K)   fp16
# B : (BH, K, N)   fp16   (the "right" tensor is already transposed)
# C : (BH, M, N)   fp32 accumulator
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 32}, num_warps=4),
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 32}, num_warps=8),
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 64, "BLOCK_K": 32}, num_warps=8),
        triton.Config({"BLOCK_M": 64, "BLOCK_N": 128, "BLOCK_K": 32}, num_warps=8),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dims (int32)
    stride_ah, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bh, stride_bk, stride_bn,   # strides for B (batch, row, col)
    stride_ch, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # which (batch * head) we are processing
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block column

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers for this batch/head
    a_batch_ptr = a_ptr + pid_batch * stride_ah
    b_batch_ptr = b_ptr + pid_batch * stride_bh
    c_batch_ptr = c_ptr + pid_batch * stride_ch

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + offs_k

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        b_mask = (cur_k[:, None] < K) & (offs_n[None, :] < N)

        # Pointers to the current tiles
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        b_ptrs = b_batch_ptr + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (fp16) with zero‑padding for oob elements
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        # Multiply‑accumulate
        acc += tl.dot(a, b)   # fp16 dot, accumulated into fp32

    # Write the result back (fp32)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict containing model weights (float32 tensors)
            - config: dict with keys 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden_dim (convert to fp16 for speed)
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)    # [B, N, N, hidden_dim], fp16
    right = F.linear(x_fp16, right_proj_w)  # same shape

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(x_fp16.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4) Gates (computed in FP32, cast back to fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim] FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # FP32

    left_gate_h = left_gate.to(x_fp16.dtype)
    right_gate_h = right_gate.to(x_fp16.dtype)

    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 5) Core N³ contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Rearrange to (B, hidden_dim, N, N) and make contiguous
    # left: (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: need to transpose the K axis (the middle N) -> (B, H, N, N) where K becomes last dim
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # Collapse batch and head dimensions for the kernel
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)          # (BH, M, K)
    right_mat = right_perm.view(BH, N, N)        # (BH, K, N)

    # Output buffer (float32 accumulator)
    out_mat = torch.empty((BH, N, N), dtype=torch.float32, device=x.device)

    # Choose tiling parameters (they will be tuned automatically)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                     # M, N, K (all equal to seq_len)
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # out_gate is FP32, out is FP32
    out = F.linear(out, weights["to_out.weight"])  # back to dim (FP32)

    return out
