def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(axis=2)
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute base pointer for batch
    a_base = A_ptr + pid_b * stride_ab
    b_base = B_ptr + pid_b * stride_bb
    c_base = C_ptr + pid_b * stride_cb

    # pointers for tile
    a_ptrs = a_base + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_base + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

def _batched_gemm_triton(A, B):
    assert A.shape[0] == B.shape[0] and A.shape[1] == B.shape[2] and A.shape[2] == B.shape[1] # maybe
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)
    
    # select block sizes
    BLOCK_M = 128 if M >= 128 else 64
    BLOCK_N = 128 if N >= 128 else 64
    BLOCK_K = 32
    # grid
    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    batched_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C

@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_aba, stride_ama, stride_aka,
    stride_bba, stride_bkb, stride_bnb,
    stride_cca, stride_cma, stride_cna,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(axis=2)
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    # Compute offsets for tiles
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Mask for valid rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for each batch
    a_base = A_ptr + pid_b * stride_aba
    b_base = B_ptr + pid_b * stride_bba
    c_base = C_ptr + pid_b * stride_cca

    # pointers to current tile
    a_ptrs = a_base + (offs_m[:, None] * stride_ama + offs_k[None, :] * stride_aka)
    b_ptrs = b_base + (offs_k[:, None] * stride_bkb + offs_n[None, :] * stride_bnb)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        k_mask = (k_offset + offs_k) < K

        a = tl.load(a_ptrs + k_offset * stride_aka,
                    mask=mask_m[:, None] & k_mask[None, :],
                    other=0.0)
        b = tl.load(b_ptrs + k_offset * stride_bkb,
                    mask=k_mask[:, None] & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)

    # write result
    c_ptrs = c_base + (offs_m[:, None] * stride_cma + offs_n[None, :] * stride_cna)
    tl.store(c_ptrs, acc.to(A_ptr.dtype) if tl.dtype(A_ptr) == tl.float16 else acc,
             mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) forward pass using Triton for the O(N³·H) reduction.

Algorithm
---------
1. Layer‑norm the input tensor (`x`).
2. Cast `x` to float16 and cast all linear weights to float16.
3. Compute left/right projections and their gates (all in float16).
   Optional mask is applied before the gates.
4. Reshape tensors so that the hidden dimension is merged into the batch:
       (B, N, N, H) → (B*H, N, N).
   The right tensor is viewed as transposed in the reduction dimension so
   that the operation becomes a batched matrix multiplication:
       out[b,i,j] = Σ_k left[b,i,k] * right[b,k,j].
5. Perform the batched matmul with a custom Triton kernel (FP16 inputs,
   FP32 accumulator, FP16 output).  This kernel tiles the M×N output
   matrix and reduces over K with `tl.dot`.
6. Cast the result back to float32, reshape to (B,N,N,H) and apply a
   second LayerNorm, the output gate, and the final linear projection
   to the original feature dimension.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # B: (batch, K, N)
    stride_ch, stride_cm, stride_cn,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_b = tl.program_id(axis=2)   # batch*hidden
    pid_m = tl.program_id(axis=0)   # i tile
    pid_n = tl.program_id(axis=1)   # j tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch slice
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # pointers to the beginning of the tile (k = 0)
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        # mask for the current k‑tile
        k_mask = (k_offset + offs_k) < K

        a = tl.load(
            a_ptrs + k_offset * stride_ak,
            mask=mask_m[:, None] & k_mask[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs + k_offset * stride_bk,
            mask=k_mask[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # write back result (stored as FP16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.
    A : (batch, M, K)  FP16
    B : (batch, K, N)  FP16 (may be non‑contiguous)
    Returns C : (batch, M, N)  FP16
    """
    assert A.is_cuda and B.is_cuda
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # tune block sizes (simple heuristic)
    BLOCK_M = 128 if M >= 128 else 64
    BLOCK_N = 128 if N >= 128 else 64
    BLOCK_K = 32

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        A.stride(0),
        A.stride(1),
        A.stride(2),
        B.stride(0),
        B.stride(1),
        B.stride(2),
        C.stride(0),
        C.stride(1),
        C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model weight tensors
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input layer‑norm (float32)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Cast to float16 for the heavy linear work
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # Cast weights to float16 once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (float16)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)          # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)        # (B,N,N,H)

    # -----------------------------------------------------------------
    # 4. Optional masking (float16)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 5. Gating (float16)
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Prepare for batched GEMM
    # -----------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim
    # (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # merge batch and hidden dimensions
    left_flat = left.view(B * H, N, N)               # (B*H, M=N, K=N)
    # right must be transposed in the reduction dimension
    right_t = right.view(B * H, N, N).transpose(1, 2)   # (B*H, K=N, N)

    # -----------------------------------------------------------------
    # 7. Core O(N³·H) reduction via Triton
    # -----------------------------------------------------------------
    out_flat = _batched_gemm_triton(left_flat, right_t)   # (B*H, N, N), FP16

    # -----------------------------------------------------------------
    # 8. Reshape back to (B,N,N,H) and cast to float32 for the rest
    # -----------------------------------------------------------------
    out = out_flat.to(torch.float32)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # -----------------------------------------------------------------
    # 9. Output LayerNorm (float32)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -----------------------------------------------------------------
    # 10. Apply output gate and final projection
    # -----------------------------------------------------------------
    out = out * out_gate.to(torch.float32)   # out_gate was FP16, cast to FP32
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
