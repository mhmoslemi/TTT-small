"""
custom_kernel: Outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑normalize the input pair representation.
2. Fuse the projection (dim → hidden) and gate (dim → hidden) linear layers
   for the “left” and “right” tensors by a single matmul with concatenated
   weights, then split and apply a sigmoid gate.
3. Optional mask (broadcast over hidden_dim) is applied to the projected
   tensors before gating.
4. Cast the gated tensors to fp16, reshape to (B*H, N, N) and run a
   batched GEMM (C = A @ Bᵀ) with a Triton kernel that accumulates in fp32.
5. Reshape the result back to (B, N, N, hidden_dim).
6. Apply a second LayerNorm, an output gate, and a final linear projection
   back to the original feature dimension.

The heavy O(N³·H) portion (step 4) is performed entirely in Triton,
while the lighter linear / norm / gate operations use PyTorch (cuBLAS‑
accelerated) kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr,            # left  (batch, M, K)   fp16
    b_ptr,            # right (batch, N, K)   fp16 (read transposed)
    c_ptr,            # output (batch, M, N) fp32
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # col tile
    pid_b = tl.program_id(2)   # batch tile (B*hidden_dim)

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k)
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # Bᵀ tile: (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (cur_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # dot product, accumulate in fp32
        acc += tl.dot(a, b).to(tl.float32)

    # write back
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – optional
            - weights: dict of PyTorch parameters (see docstring)
            - config: dict with keys:
                * "dim" (int)
                * "hidden_dim" (int)
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]        # (hidden_dim, dim)
    left_gate_weight = weights["left_gate.weight"]          # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]        # (hidden_dim, dim)
    out_gate_weight = weights["out_gate.weight"]            # (hidden_dim, dim)
    to_out_norm_weight = weights["to_out_norm.weight"]      # (hidden_dim,)
    to_out_norm_bias = weights["to_out_norm.bias"]          # (hidden_dim,)
    to_out_weight = weights["to_out.weight"]                # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Project + gate (fused via concatenated weights)
    # ------------------------------------------------------------------
    # concatenate projection and gate weights for left side
    left_concat_w = torch.cat([left_proj_weight, left_gate_weight], dim=0)  # (2*hidden_dim, dim)
    right_concat_w = torch.cat([right_proj_weight, right_gate_weight], dim=0)  # (2*hidden_dim, dim)

    # one matmul each side
    left_comb = F.linear(x, left_concat_w)   # [B, N, N, 2*hidden_dim]
    right_comb = F.linear(x, right_concat_w)  # [B, N, N, 2*hidden_dim]

    left_proj = left_comb[..., :hidden_dim]          # (B,N,N,hidden_dim)
    left_gate = left_comb[..., hidden_dim:].sigmoid()

    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # out gate (separate linear)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> broadcast over hidden_dim
        mask_f = mask.to(dtype).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4. Apply gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Batched GEMM via Triton (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, hidden_dim

    # fp16 for tensor cores
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (batch*hidden_dim, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # allocate fp32 output
    out_h = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # strides for Triton
    stride_a_batch, stride_a_m, stride_a_k = left_h.stride()
    stride_b_batch, stride_b_n, stride_b_k = right_h.stride()  # note: (batch, N, K)
    stride_c_batch, stride_c_m, stride_c_n = out_h.stride()

    # kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m, num_pid_n, B * hidden_dim)

    _tri_mul_gemm_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Reshape back to (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)  # (B, N, N, hidden_dim)
        .contiguous()
    )  # dtype fp32

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # (B, N, N, dim)

    return out
