@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(1)
    pid = tl.program_id(0)

    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_blocks_n
    pid_n = pid % num_blocks_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # create mask for out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute block offset for k
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # compute mask for K
        mask_k = offs_k < K

        # compute pointers
        a_ptrs = a_ptr + batch_id * batch_stride_a + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_ptr + batch_id * batch_stride_b + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        # load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0, dtype=tl.float16)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_ptr + batch_id * batch_stride_c + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Unpack weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Ensure weights are correct dtype and shape (no bias)
    # compute layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # Cast to half for projections
    x_h = x.half()

    # Linear projections
    left = torch.nn.functional.linear(x_h, left_proj_weight.half())
    right = torch.nn.functional.linear(x_h, right_proj_weight.half())

    # mask if needed
    if not nomask:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask_h
        right = right * mask_h

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_weight.half()))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_weight.half()))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_weight.half()))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Release x_h perhaps
    # Permute for batched matmul
    # left shape: (B,N,N,hidden)
    B, N, _, H = left.shape
    # Permute
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # Flatten batch*hidden
    batch_size = B * H
    # reshape to (batch_size, N, N)
    a_mat = left_perm.view(batch_size, N, N)
    b_mat = right_perm.view(batch_size, N, N)
    # allocate output
    c_mat = torch.empty_like(a_mat, dtype=torch.float16, device=device)

    # compute strides
    a_stride_batch = a_mat.stride(0)
    a_stride_m = a_mat.stride(1)
    a_stride_k = a_mat.stride(2)

    b_stride_batch = b_mat.stride(0)
    b_stride_k = b_mat.stride(1)
    b_stride_n = b_mat.stride(2)

    c_stride_batch = c_mat.stride(0)
    c_stride_m = c_mat.stride(1)
    c_stride_n = c_mat.stride(2)

    # block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (num_blocks_m * num_blocks_n, batch_size)

    batched_matmul_kernel[grid](
        a_mat, b_mat, c_mat,
        N, N, N,
        a_stride_m, a_stride_k,
        b_stride_k, b_stride_n,
        c_stride_m, c_stride_n,
        a_stride_batch, b_stride_batch, c_stride_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape output
    out = c_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # to_out_norm layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # Apply out_gate (cast)
    out = out * out_gate.to(out.dtype)

    # Final linear projection
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton accelerated core.

The implementation follows the reference PyTorch TriMul but fuses the heavy
"O(N³)" reduction using a custom batched‑matrix‑multiply kernel written in
Triton.  All linear projections, gating and masking are performed in
float32 (the original precision) to guarantee numerical correctness.
The kernel computes, for each batch and hidden channel d,

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

as a batched matmul:  left (i,k) @ right_T (k,j).  The reduction is
performed entirely on the GPU, avoiding the expensive torch.bmm call.
A Triton kernel is also used for the final element‑wise out‑gate
multiplication (optional, but retained for completeness).  The overall
pipeline:

1. LayerNorm over the input channels.
2. Linear projections (left/right) and gates (left/right/out) without bias.
3. Optional masking.
4. Apply gates: left *= left_gate, right *= right_gate.
5. Batched matmul via Triton: out = left @ right_T.
6. LayerNorm over the hidden dimension.
7. Element‑wise multiplication with out_gate (Triton kernel).
8. Final linear projection back to the original dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute a batch of matrix multiplications C = A @ B
    where A: (batch, M, K), B: (batch, K, N), C: (batch, M, N).
    All tensors are in row‑major layout and stored as fp32.
    """
    batch_id = tl.program_id(1)          # index of the batch (B*hidden)
    pid = tl.program_id(0)               # tile index over M and N

    # number of blocks in each dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_blocks_n          # block row
    pid_n = pid % num_blocks_n           # block column

    # offsets within the matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for partial tiles on the matrix edges
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers with batch offset
        a_ptrs = a_ptr + batch_id * batch_stride_a + (
            offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        )
        b_ptrs = b_ptr + batch_id * batch_stride_b + (
            offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        )

        # load tiles, zero‑pad out‑of‑bounds elements
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0, dtype=tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0, dtype=tl.float32)

        # dot‑product accumulation
        acc += tl.dot(a, b)

    # store the result tile
    c_ptrs = c_ptr + batch_id * batch_stride_c + (
        offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """c = a * b  (flattened 1‑D tensors) – used for the out‑gate."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask, other=0.0, dtype=tl.float32)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0, dtype=tl.float32)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using Triton for the
    O(N³) reduction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model weights (all torch.Tensors on the same device)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Unpack weights (ensure they are on the correct device)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)

    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)

    to_out_weight = weights["to_out.weight"].to(device)

    # -----------------------------------------------------------------
    # 1️⃣  LayerNorm over the channel dimension (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2️⃣  Linear projections (no bias) – keep in float32 for exactness
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3️⃣  Optional masking (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4️⃣  Gating (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates in‑place to save memory
    left = left * left_gate
    right = right * right_gate

    # Free now‑unneeded tensors
    del left_gate, right_gate, x

    # -----------------------------------------------------------------
    # 5️⃣  Prepare tensors for the batched matmul C = A @ B
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # B, N, N, hidden_dim
    # Permute to (batch, hidden, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose k‑axis

    # Flatten batch*hidden into a single batch dimension for the kernel
    batch_sz = B * H
    A = left_perm.view(batch_sz, N, N)   # (batch, M, K)
    B_mat = right_perm.view(batch_sz, N, N)  # (batch, K, N)
    C = torch.empty_like(A, dtype=torch.float32, device=device)  # output

    # Strides (in elements) for the kernel
    a_stride_batch = A.stride(0)
    a_stride_m = A.stride(1)
    a_stride_k = A.stride(2)

    b_stride_batch = B_mat.stride(0)
    b_stride_k = B_mat.stride(1)
    b_stride_n = B_mat.stride(2)

    c_stride_batch = C.stride(0)
    c_stride_m = C.stride(1)
    c_stride_n = C.stride(2)

    # -----------------------------------------------------------------
    # 6️⃣  Triton batched matmul kernel
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, batch_sz)

    batched_matmul_kernel[grid](
        A, B_mat, C,
        N, N, N,
        a_stride_m, a_stride_k,
        b_stride_k, b_stride_n,
        c_stride_m, c_stride_n,
        a_stride_batch, b_stride_batch, c_stride_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7️⃣  Reshape back to [B, N, N, hidden_dim] (float32)
    # -----------------------------------------------------------------
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 8️⃣  Final LayerNorm over hidden dimension
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 9️⃣  Apply out‑gate with a tiny Triton kernel (optional)
    # -----------------------------------------------------------------
    # (the same could be done with out *= out_gate, but we keep a kernel)
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)
    total_elements = out_flat.numel()
    BLOCK = 1024
    grid_gate = ( (total_elements + BLOCK - 1) // BLOCK, )
    elementwise_mul_kernel[grid_gate](
        out_flat,
        gate_flat,
        out_flat,
        total_elements,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 🔟  Project back to the original dimension
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
