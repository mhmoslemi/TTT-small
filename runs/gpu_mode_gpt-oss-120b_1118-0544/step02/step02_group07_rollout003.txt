def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # weight dict
    ...

    # layernorm: we need to apply to last dim (dim)
    x = F.layer_norm(...)

    # Projections:
    left = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # mask
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(x.dtype)  # produce broadcast mask
        left = left * mask_f
        right = right * mask_f

    # Gates:
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Convert to fp16 and ensure contiguous
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, D = left_h.shape  # D = hidden_dim

    # Prepare output tensor
    out_h = torch.empty((B, N, N, D), dtype=torch.float16, device=left_h.device)

    # Compute strides
    stride_bl, stride_il, stride_kl, stride_dl = left_h.stride()
    stride_br, stride_ir, stride_kr, stride_dr = right_h.stride()
    stride_bo, stride_io, stride_ko, stride_do = out_h.stride()

    # Choose block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # grid
    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * D)

    # run kernel
    tri_mul_kernel[grid](
        left_h, right_h, out_h,
        B, N, D,
        stride_bl, stride_il, stride_kl, stride_dl,
        stride_br, stride_ir, stride_kr, stride_dr,
        stride_bo, stride_io, stride_ko, stride_do,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Now out_h is result in fp16; cast to float32
    out = out_h.to(torch.float32)

    # to_out_norm
    out = F.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # apply out_gate (elementwise multiply)
    out = out * out_gate

    # final projection
    out = F.linear(out, to_out_weight)

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton implementation.

The kernel performs the core O(N³) reduction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
as a batched matrix multiplication per (batch, hidden) channel.
The reduction is fused in a Triton kernel that loads tiles of the
left and right tensors, accumulates in FP32 and writes the result in FP16.
All other operations (LayerNorm, Linear projections, gating) are carried out
with PyTorch (FP32) for simplicity. The final output is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: fused reduction for the outgoing TriMul
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,        # tl.pointer(tl.float16)
    right_ptr,       # tl.pointer(tl.float16)
    out_ptr,         # tl.pointer(tl.float16)
    B, N, D,         # runtime sizes: batch, seq_len, hidden_dim
    stride_bl, stride_il, stride_kl, stride_dl,   # left strides
    stride_br, stride_ir, stride_kr, stride_dr,   # right strides
    stride_bo, stride_io, stride_ko, stride_do,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
       for a single (b,d) pair.
    """
    pid_i = tl.program_id(0)   # tile row (i)
    pid_j = tl.program_id(1)   # tile col (j)
    pid_c = tl.program_id(2)   # combined batch * hidden index

    # decode combined index
    d = pid_c % D
    b = pid_c // D

    # tile offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows/cols
    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        k_offset = k + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ---- left tile (M x K) ----
        left_off = (
            b * stride_bl +
            d * stride_dl +
            i_offset[:, None] * stride_il +
            k_offset[None, :] * stride_kl
        )
        left_tile = tl.load(left_ptr + left_off,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # (M, K)

        # ---- right tile (K x N) ----  (note: we load transposed)
        right_off = (
            b * stride_br +
            d * stride_dr +
            k_offset[:, None] * stride_kr +
            j_offset[None, :] * stride_ir
        )
        right_tile = tl.load(right_ptr + right_off,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)   # (K, N)

        # accumulate
        acc += tl.dot(left_tile, right_tile)   # (M, N)

    # write result (cast back to FP16)
    out_off = (
        b * stride_bo +
        d * stride_do +
        i_offset[:, None] * stride_io +
        j_offset[None, :] * stride_ko
    )
    tl.store(out_ptr + out_off,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# --------------------------------------------------------------
# Entry point used by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config["nomask"]=True
            - weights: dict of model weights
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Unpack weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3️⃣ Optional masking
    # -----------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Cast to FP16 and ensure contiguity before Triton kernel
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, D = left_h.shape  # D == hidden_dim

    # output buffer for the reduction (FP16)
    out_h = torch.empty((B, N, N, D), dtype=torch.float16, device=left_h.device)

    # -----------------------------------------------------------------
    # 6️⃣ Prepare strides for the kernel (in elements, not bytes)
    # -----------------------------------------------------------------
    stride_bl, stride_il, stride_kl, stride_dl = left_h.stride()
    stride_br, stride_ir, stride_kr, stride_dr = right_h.stride()
    stride_bo, stride_io, stride_ko, stride_do = out_h.stride()

    # -----------------------------------------------------------------
    # 7️⃣ Triton kernel launch (batched matmul across sequence dimension)
    # -----------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * D,
    )
    tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        D,
        stride_bl,
        stride_il,
        stride_kl,
        stride_dl,
        stride_br,
        stride_ir,
        stride_kr,
        stride_dr,
        stride_bo,
        stride_io,
        stride_ko,
        stride_do,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Cast back to FP32 for the remaining layers
    out = out_h.to(torch.float32)

    # -----------------------------------------------------------------
    # 8️⃣ Output LayerNorm over hidden dimension
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Apply the output gate (element‑wise)
    # -----------------------------------------------------------------
    out = out * out_gate

    # -----------------------------------------------------------------
    # 🔟 Final linear projection back to original channel dimension
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
