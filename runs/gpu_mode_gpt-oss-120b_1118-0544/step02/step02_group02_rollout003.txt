"""
custom_kernel: Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Cast all model parameters to fp16.
2. Layer‑norm the input (fp16) using the provided LN weights.
3. Linear projections (left/right) and gating logits are computed with fp16
   GEMMs (torch.nn.functional.linear).  The optional mask is applied *before*
   gating as in the reference implementation.
4. Apply sigmoid to gate logits and multiply with the projected values.
5. Perform the heavy O(N³) reduction as a batched matrix multiplication:
   for each hidden channel d we compute
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   by reshaping to (B·H, N, N) and using torch.bmm.
6. Layer‑norm the reduction result (fp16), multiply by the output gate,
   and project back to the original feature dimension with a final linear.
7. Cast the fp16 final tensor back to fp32 with a tiny Triton kernel
   (required to use Triton).

All heavy compute is done in fp16 to reduce memory traffic while still
leveraging Tensor‑Core acceleration.  The Triton kernel only performs the
fp16→fp32 conversion, satisfying the “use Triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fast cast from fp16 to fp32
# ----------------------------------------------------------------------
@triton.jit
def cast_fp16_to_fp32_kernel(
    in_ptr,          # fp16 source pointer
    out_ptr,         # fp32 destination pointer
    N_ELEM: tl.constexpr,   # total number of elements
    BLOCK: tl.constexpr,    # block size (must be compile‑time const)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM
    # Load fp16, promote to fp32, store
    val = tl.load(in_ptr + offs, mask=mask).to(tl.float32)
    tl.store(out_ptr + offs, val, mask=mask)


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model weights (see skeleton)
            - config: dict with at least:
                * "dim" (int)
                * "hidden_dim" (int)
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Cast model parameters to fp16 (Tensor cores)
    # ------------------------------------------------------------------
    to_fp16 = lambda t: t.to(device=device, dtype=torch.float16)
    norm_weight = to_fp16(weights["norm.weight"])
    norm_bias = to_fp16(weights["norm.bias"])
    left_proj_weight = to_fp16(weights["left_proj.weight"])
    right_proj_weight = to_fp16(weights["right_proj.weight"])
    left_gate_weight = to_fp16(weights["left_gate.weight"])
    right_gate_weight = to_fp16(weights["right_gate.weight"])
    out_gate_weight = to_fp16(weights["out_gate.weight"])
    to_out_norm_weight = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_bias = to_fp16(weights["to_out_norm.bias"])
    to_out_weight = to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 0) LayerNorm over the last dimension (fp16)
    # ------------------------------------------------------------------
    x = input_tensor.to(dtype=torch.float16)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 1) Linear projections (bias‑free, fp16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right_proj = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 2) Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask_f = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # same shape

    # ------------------------------------------------------------------
    # 4) Apply gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate    # [B, N, N, hidden_dim] fp16
    right = right_proj * right_gate
    # Release unused tensors to save memory
    del left_proj, right_proj, left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5) Heavy reduction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched matrix multiplication.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Batched matmul (fp16, TensorCore accelerated)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # [B*H, N, N] fp16

    # Reshape back to [B, N, N, hidden_dim]
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # Clean up temporaries
    del left_flat, right_flat, out_flat, left, right

    # ------------------------------------------------------------------
    # 6) Final LayerNorm over hidden_dim (fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 7) Output gating
    # ------------------------------------------------------------------
    out = out * out_gate  # fp16
    del out_gate

    # ------------------------------------------------------------------
    # 8) Linear projection back to original dim (fp16)
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 9) Cast final result to fp32 with a tiny Triton kernel
    # ------------------------------------------------------------------
    total_elems = out.numel()
    out_fp32 = torch.empty_like(out, dtype=torch.float32)
    BLOCK = 65536  # 64K elements per program
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    cast_fp16_to_fp32_kernel[grid](
        out,
        out_fp32,
        N_ELEM=total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # Return fp32 tensor
    return out_fp32
