"""
outgoing TriMul (TriMul) forward pass – Triton implementation.

The operation is:
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
where
    left = (x ‑‑> LayerNorm ‑‑> Linear(dim→hidden_dim)) * left_gate * mask
    right = (x ‑‑> LayerNorm ‑‑> Linear(dim→hidden_dim)) * right_gate * mask

This is a batched matrix multiplication per hidden channel:
    out_{b,d} = left_{b,d} @ right_{b,d}ᵀ
with left_{b,d}, right_{b,d} ∈ ℝ^{N×N}.

We implement the heavy N³ reduction as a fused Triton GEMM kernel,
using fp16 inputs and fp32 accumulation. All other operations
(LayerNorm, Linear projections, gates, final LayerNorm and Linear)
are performed in PyTorch for simplicity. The kernel supports arbitrary
sequence lengths and optional masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (int32)
    stride_ah, stride_am, stride_ak,   # strides for A (batch*head, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for B (batch*head, k, j)
    stride_ch, stride_cm, stride_cn,   # strides for C (batch*head, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B  (where B is already transposed for the einsum)

    A shape: (BH, M, K)  fp16
    B shape: (BH, K, N)  fp16
    C shape: (BH, M, N)  fp32 accumulator
    """
    pid_bh = tl.program_id(0)   # collapsed batch * hidden_dim
    pid_m = tl.program_id(1)    # block row
    pid_n = tl.program_id(2)    # block col

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch/head
    a_batch = a_ptr + pid_bh * stride_ah
    b_batch = b_ptr + pid_bh * stride_bh
    c_batch = c_ptr + pid_bh * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for OOB accesses
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Compute pointers
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (fp16) with zero padding for OOB
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        # Cast to fp32 for accumulation
        a_f32 = a.to(tl.float32)
        b_f32 = b.to(tl.float32)

        # Accumulate
        acc += tl.dot(a_f32, b_f32)

    # Write back result
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (optional, float32/bool) or None
            - weights: dict with weight tensors (float32) containing:
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config: dict with keys 'dim' and 'hidden_dim' (int)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim], FP32

    # ------------------------------------------------------------------
    # 2) Projections to hidden_dim (cast to fp16 for speed)
    # ------------------------------------------------------------------
    # Cast input and projection weights to fp16
    x_fp16 = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, hidden_dim] fp16
    right = F.linear(x_fp16, right_proj_w)  # fp16

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # Ensure mask is float16 and broadcastable
        mask_h = mask.unsqueeze(-1).to(x_fp16.dtype)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gates (computed in FP32, then cast to fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # FP32

    # Cast gates to fp16 before multiplication
    left_gate_h = left_gate.to(x_fp16.dtype)
    right_gate_h = right_gate.to(x_fp16.dtype)

    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 5) Core N³ contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N_seq, _, _ = left.shape  # B, N, N, hidden_dim
    # Rearrange to (B, hidden_dim, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j)

    # Collapse batch and head dimensions
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N_seq, N_seq)   # (BH, M, K) where M=N, K=N
    right_mat = right_perm.view(BH, N_seq, N_seq)  # (BH, K, N)

    # Output buffer (float32 accumulator)
    out_mat = torch.empty((BH, N_seq, N_seq), dtype=torch.float32, device=x.device)

    # Tiling parameters (tuned for typical N and hidden_dim)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,
        (N_seq + BLOCK_M - 1) // BLOCK_M,
        (N_seq + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N_seq,              # M
        N_seq,              # N
        N_seq,              # K
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()  # FP32

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # FP32

    # Apply output gate (broadcasted)
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])  # FP32, no bias

    return out
