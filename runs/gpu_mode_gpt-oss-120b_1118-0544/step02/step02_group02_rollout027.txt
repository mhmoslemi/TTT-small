"""
Outgoing Triangle Multiplicative Update (TriMul) – forward pass.

The heavy O(N³·C) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication C = A @ Bᵀ where
A and B are the gated projections of the input.  A single Triton kernel
performs the whole batch of GEMMs in FP16 with accumulation in FP32,
eliminating the extra cast step used in the reference implementation.
All other lightweight operations (LayerNorm, linear projections, gating)
are kept in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes (all == seq_len)
    stride_ab, stride_am, stride_ak,   # A: batch, row(i), col(k)
    stride_bbatch, stride_bk, stride_bn,  # B: batch, col(k), row(j)  (B is used transposed)
    stride_cbatch, stride_cm, stride_cn,  # C: batch, row(i), col(j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where each batch element computes
    C = A @ B.T   (A: M×K, B: N×K)  -> C: M×N.
    All tensors are stored in row‑major order.  A and B are FP16,
    C is accumulated and written out as FP32.
    """
    pid_batch = tl.program_id(2)       # batch * hidden dimension
    pid_m = tl.program_id(0)           # block row of C
    pid_n = tl.program_id(1)           # block column of C

    # -------------------- offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # batch stride offsets
    a_batch_offset = pid_batch * stride_ab
    b_batch_offset = pid_batch * stride_bbatch
    c_batch_offset = pid_batch * stride_cbatch

    # accumulate in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- main loop over K --------------------
    num_k = tl.cdiv(K, BLOCK_K)
    for k_iter in range(0, num_k):
        k_base = k_iter * BLOCK_K
        # current K indices
        cur_k = k_base + offs_k

        # ------- load A block (M x K) -------
        a_ptrs = (
            A_ptr
            + a_batch_offset
            + (offs_m[:, None] * stride_am)
            + (cur_k[None, :] * stride_ak)
        )
        a_mask = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float16)

        # ------- load B block (K x N) – note B is transposed -------
        b_ptrs = (
            B_ptr
            + b_batch_offset
            + (cur_k[:, None] * stride_bk)   # rows = k
            + (offs_n[None, :] * stride_bn)  # cols = n
        )
        b_mask = (cur_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float16)

        # ------- multiply‑accumulate -------
        acc += tl.dot(a, b)   # FP16×FP16 → FP32 accumulation

    # -------------------- write result --------------------
    c_ptrs = (
        C_ptr
        + c_batch_offset
        + (offs_m[:, None] * stride_cm)
        + (offs_n[None, :] * stride_cn)
    )
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask: torch.Tensor or None, shape [B, N, N]
            - weights: dict with all linear and LayerNorm parameters
            - config: dict containing at least "dim" and "hidden_dim", may also contain "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim], dtype float32
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------- 0) unpack weights --------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -------------------- 1) input LayerNorm --------------------
    # shape: [B, N, N, dim]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # -------------------- 2) projections --------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]  (float32)
    right = F.linear(x, right_proj_weight)

    # -------------------- 3) optional mask --------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -------------------- 4) gating --------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # -------------------- 5) apply gates and cast to FP16 --------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # -------------------- 6) reshape for batched GEMM --------------------
    # bring hidden_dim to the batch dimension: [B, hidden, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left.shape
    batch_hidden = B * H

    # flatten batch and hidden into a single batch dimension
    left_flat = left.view(batch_hidden, N, N).contiguous()
    right_flat = right.view(batch_hidden, N, N).contiguous()

    # -------------------- 7) Triton batched GEMM: C = A @ Bᵀ --------------------
    out_fp32 = torch.empty((batch_hidden, N, N), dtype=torch.float32, device=device)

    # strides (int64)
    stride_ab = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bbatch = right_flat.stride(0)
    stride_bn = right_flat.stride(1)   # original row (j) stride
    stride_bk = right_flat.stride(2)   # original col (k) stride

    stride_cbatch = out_fp32.stride(0)
    stride_cm = out_fp32.stride(1)
    stride_cn = out_fp32.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch_hidden,
    )

    batched_gemm_kernel[grid](
        A_ptr=left_flat,
        B_ptr=right_flat,
        C_ptr=out_fp32,
        M=N,
        N=N,
        K=N,
        stride_ab=stride_ab,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bbatch=stride_bbatch,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cbatch=stride_cbatch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # -------------------- 8) reshape back to [B,N,N,hidden] --------------------
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------- 9) final LayerNorm, gating, projection --------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # apply output gate
    out = F.linear(out, to_out_weight)       # [B, N, N, dim]

    return out
```
