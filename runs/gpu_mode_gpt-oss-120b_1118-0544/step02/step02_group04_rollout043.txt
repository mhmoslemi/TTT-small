```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor,
                                       normalized_shape=(dim,),
                                       weight=weights["norm.weight"],
                                       bias=weights["norm.bias"])

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask if needed
    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # broadcast
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Convert to half for faster kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # rearrange for batched matmul
    B, N, _, _ = left.shape
    # Permute to (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    batch_hidden = B * hidden_dim
    left_flat = left.view(batch_hidden, N, N)
    right_flat = right.view(batch_hidden, N, N)

    # Kernel compute
    out_flat = _batched_outgoing_trimul_kernel(left_flat, right_flat)

    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # Output layernorm
    out = torch.nn.functional.layer_norm(out,
                                         normalized_shape=(hidden_dim,),
                                         weight=weights["to_out_norm.weight"],
                                         bias=weights["to_out_norm.bias"])
    # Apply out gate (out_gate was computed in float32)
    # Ensure out_gate dtype matches out dtype (float32)
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections (left/right) and gating (left/right/out) with optional mask.
3. Cast the gated left/right tensors to FP16 and rearrange so the hidden dimension
   becomes the batch axis: (B, N, N, H) → (B*H, N, N).
4. Compute the batched matrix multiplication
        out[b, i, j] = Σ_k left[b, i, k] * right[b, j, k]
   using a custom Triton kernel that processes many (i,j) tiles in parallel.
   The kernel loads tiles in FP16, performs a tensor‑core dot, and accumulates
   in FP32 for full precision.
5. Reshape the result back to (B, N, N, H) and apply a second LayerNorm,
   the output gate, and the final linear projection to the original dimension.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batched_outgoing_trimul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # B: (batch, N, K) (note: K is the reduction axis)
    stride_ch, stride_cm, stride_cn,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ  (A,B are FP16, C is FP32)."""
    pid_batch = tl.program_id(axis=2)  # batch * hidden
    pid_m = tl.program_id(axis=0)      # tile over M (i)
    pid_n = tl.program_id(axis=1)      # tile over N (j)

    # coordinates within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base offsets for this batch
    a_batch = pid_batch * stride_ah
    b_batch = pid_batch * stride_bh
    c_batch = pid_batch * stride_ch

    # offsets for the K‑dimension tile (static)
    offs_k = tl.arange(0, BLOCK_K)

    # pointers to the current (i,k) and (k,j) tiles
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K
        k_idx = k_offset + offs_k
        mask_k = k_idx < K

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
            dtype=tl.float16,
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0.0),
            dtype=tl.float16,
        )
        # Tensor‑core dot (FP16 inputs, FP32 accumulation)
        acc += tl.dot(a, b).to(tl.float32)

        # advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write out the result
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _batched_outgoing_trimul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    A: (batch, M, K)   FP16
    B: (batch, N, K)   FP16
    Returns C: (batch, M, N)   FP32
    """
    assert A.ndim == 3 and B.ndim == 3
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2

    # Heuristic block sizes (tunable)
    BLOCK_M = 64 if M >= 64 else 32
    BLOCK_N = 64 if N >= 64 else 32
    BLOCK_K = 64 if K > 256 else 32

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    _batched_outgoing_trimul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        A.stride(0),
        A.stride(1),
        A.stride(2),
        B.stride(0),
        B.stride(2),  # stride over K in B (last dim)
        B.stride(1),  # stride over N in B
        C.stride(0),
        C.stride(1),
        C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model weight tensors
            - config: dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (left / right) – no bias
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3️⃣ Optional masking (applied before gating)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)            # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise gating
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Cast to FP16 and rearrange for the batched matmul kernel
    # -----------------------------------------------------------------
    # shape after cast: (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left = left.to(torch.float16).permute(0, 3, 1, 2).contiguous()
    right = right.to(torch.float16).permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left.shape
    batch_hidden = B * H
    left_flat = left.view(batch_hidden, N, N)
    right_flat = right.view(batch_hidden, N, N)

    # -----------------------------------------------------------------
    # 6️⃣ Core O(N³·H) reduction via Triton
    # -----------------------------------------------------------------
    out_flat = _batched_outgoing_trimul(left_flat, right_flat)  # (B*H, N, N), FP32

    # -----------------------------------------------------------------
    # 7️⃣ Reshape back to (B, N, N, H)
    # -----------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # -----------------------------------------------------------------
    # 8️⃣ Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # gating (float32)
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```
