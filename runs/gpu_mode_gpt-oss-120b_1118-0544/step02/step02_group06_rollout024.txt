"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton‑accelerated version.

The core operation is a batched matrix multiplication:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
which is equivalent to a batch of GEMMs C = A @ Bᵀ where each
batch = B * hidden_dim.  The heavy reduction over the sequence dimension
(N) is performed by a custom Triton kernel (tri_mul_outgoing_kernel).
All other steps (LayerNorm, linear projections, gating) use PyTorch
functional APIs.

Input:
    - input_tensor:   torch.Tensor, shape [B, N, N, dim], float32
    - mask:           torch.Tensor, shape [B, N, N] (bool/float) or None
    - weights:        dict of model parameters (see keys below)
    - config:         dict with keys "dim", "hidden_dim", optional "nomask"

Output:
    - torch.Tensor, shape [B, N, N, dim], float32
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,                # left  tensor pointer,   (batch, M, K)  fp16
    b_ptr,                # right tensor pointer,   (batch, N, K)  fp16 (read transposed)
    c_ptr,                # output tensor pointer, (batch, M, N)  fp32
    batch: tl.constexpr,  # total number of independent GEMMs = B * hidden_dim
    M: tl.constexpr,      # rows of A
    N: tl.constexpr,      # rows of B (cols of output)
    K: tl.constexpr,      # reduction dimension
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ Bᵀ  (fp16 inputs, fp32 accumulation)."""
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # batch index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Load Bᵀ tile (K x N) – we treat B as (batch, K, N) by swapping strides
        b_ptrs = b_batch + (cur_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Dot product accumulation (fp16·fp16 → fp32)
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

        data: tuple (input_tensor, mask, weights, config)

            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (see below)
            - config: dict with at least
                * "dim" (int)
                * "hidden_dim" (int)
                * "nomask" (bool, optional, default True)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for GEMM (cast to fp16 for Tensor‑cores)
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N_seq, _, H = left_h.shape  # B, N, N, hidden_dim

    # Collapse batch and hidden dimension → (B*H, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * H, N_seq, N_seq)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * H, N_seq, N_seq)

    # ------------------------------------------------------------------
    # Triton batched GEMM: out = left_h @ right_hᵀ
    # ------------------------------------------------------------------
    # Output buffer (float32)
    out_h = torch.empty((B * H, N_seq, N_seq), dtype=torch.float32, device=device)

    # Tunable block sizes (empirically good for Hopper/B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Grid dimensions
    num_pid_m = (N_seq + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N_seq + BLOCK_N - 1) // BLOCK_N
    batch_dim = left_h.shape[0]
    grid = (num_pid_m, num_pid_n, batch_dim)

    # Strides (in elements, not bytes)
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)

    stride_b_batch = right_h.stride(0)
    # To read Bᵀ we swap the last two strides
    stride_b_k = right_h.stride(2)   # original K‑dimension stride
    stride_b_n = right_h.stride(1)   # original N‑dimension stride

    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    # Launch kernel
    tri_mul_outgoing_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        batch=batch_dim,
        M=N_seq,
        N=N_seq,
        K=N_seq,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] (still float32)
    # ------------------------------------------------------------------
    out = out_h.view(B, H, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Final LayerNorm, gating, and projection back to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
```
