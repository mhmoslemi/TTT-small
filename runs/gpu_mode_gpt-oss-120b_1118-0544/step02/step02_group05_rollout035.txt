def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps = 1e-5

    # LayerNorm over dim
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Convert to fp16 for projection
    x_h = x_norm.to(torch.float16)

    # Linear projections (no bias)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left = F.linear(x_h, left_proj_w)  # shape [B,N,N,hidden_dim] (fp16)
    right = F.linear(x_h, right_proj_w)  # same shape

    # Mask
    if mask is not None:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))  # x_norm is fp32
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Cast gates to fp16 and apply
    left_gate_h = left_gate.to(x_h.dtype)
    right_gate_h = right_gate.to(x_h.dtype)
    left = left * left_gate_h
    right = right * right_gate_h

    # Call Triton kernel
    B, N, _, _ = left.shape
    H = hidden_dim

    # Get strides for left/right/out in terms of elements (since we will pass pointer addresses)
    # left stride: (B,N,N,H) shape
    # Use left.stride() returns tuple of strides in elements (not bytes) - needed for pointer arithmetic
    stride_left_b, stride_left_i, stride_left_j, stride_left_h = left.stride()
    # For right pointer stride: same stride as left
    stride_right_b, stride_right_i, stride_right_j, stride_right_h = right.stride()
    # Output stride after kernel: out will be (B,N,N,H) same layout as left/right (but we need float32). We'll allocate out tensor.

    out = torch.empty((B, N, N, H), dtype=torch.float32, device=x.device)

    # Output stride
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Kernel block sizes (tunable)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid Launch
    # We use (BH, tiles_i, tiles_j)
    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _tri_mul_kernel[grid](
        left,
        right,

def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Program IDs
    pid = tl.program_id(0)   # batch*head index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Derive batch index b and head index h
    b = pid // H
    h = pid % H

    # Offsets for M and N
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers base for this batch/head
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out-of-bounds
        mask_a = (offs_i[:, None] < N) & (offs_k[None, :] < N)
        mask_b = (offs_k[:, None] < N) & (offs_j[None, :] < N)

        # Compute pointer offsets
        # left: index (i, k)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_j
        # right: index (k, j) after transpose
        b_ptrs = right_base + offs_k[:, None] * stride_right_j + offs_j[None, :] * stride_right_i

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)  # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)  # fp16

        # If tl.dot returns fp16, we need to cast:

"""
TriMul (outgoing) forward pass – Triton implementation.

The core operation is a batched matrix multiplication over the sequence axis:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
for every batch b and hidden channel h.  This is expressed as a GEMM
C = A @ Bᵀ where A and B have shape (B*H, N, N).  The kernel tiles the
output (i, j) dimensions (BLOCK_M × BLOCK_N) and reduces over the
sequence dimension K in BLOCK_K steps, using fp16 inputs and fp32
accumulation.  All surrounding LayerNorms, linear projections and gates
are performed with PyTorch; only the N³ contraction is off‑loaded to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr,               # fp16 *, shape (B, N, N, H)
    right_ptr,              # fp16 *, shape (B, N, N, H)
    out_ptr,                # fp32 *, shape (B, N, N, H)
    B, N, H,                # int32
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

    The batch dimension is flattened as pid = b * H + h.
    """
    pid = tl.program_id(0)          # flattened (b, h)
    pid_m = tl.program_id(1)        # tile row (i)
    pid_n = tl.program_id(2)        # tile column (j)

    # recover batch and hidden indices
    b = pid // H
    h = pid % H

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)

        # masks for loads
        mask_a = (offs_i[:, None] < N) & (offs_k[None, :] < N)   # left[i,k]
        mask_b = (offs_k[:, None] < N) & (offs_j[None, :] < N)   # right[k,j]

        # pointers
        # left[b, i, k, h]
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_j
        # right[b, j, k, h] -> we need (k, j) order for the dot
        b_ptrs = right_base + offs_k[:, None] * stride_right_j + offs_j[None, :] * stride_right_i

        # load tiles (fp16)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # dot‑product accumulate (fp32)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # write out with mask
    mask_out = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=mask_out)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with the following keys (all float32 tensors)
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config: dict with entries 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden_dim (cast to fp16)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)    # [B, N, N, hidden_dim]  fp16
    right = F.linear(x_h, right_proj_w)  # same shape

    # ------------------------------------------------------------------
    # 3) Optional masking (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gates (computed in FP32, then cast to fp16 for multiplication)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # FP32

    left_gate_h = left_gate.to(x_h.dtype)
    right_gate_h = right_gate.to(x_h.dtype)

    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 5) Core N³ contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape = (B, N, N, hidden_dim)

    # Ensure contiguous layout for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()

    # Allocate output tensor (float32 accumulator)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=x.device)

    # Strides (in elements) for the tensors
    stride_left_b, stride_left_i, stride_left_j, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_j, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tunable block sizes – good trade‑off for typical N (256‑1024)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                     # flattened (batch, head) dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,       # tiles over j
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_j, stride_left_h,
        stride_right_b, stride_right_i, stride_right_j, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # apply output gate (out_gate shape matches out)
    out = out * out_gate

    # final projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out
