"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy N³ reduction is implemented as a custom Triton kernel that performs
a batched matrix multiplication across the hidden dimension:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
where `left` and `right` are the projected, gated (and optionally masked)
tensors of shape [B, N, N, H].  The kernel treats (B*H) as a combined batch
dimension and tiles the output (i,j) axes with BLOCK_M×BLOCK_N tiles while
looping over K‑tiles of size BLOCK_K.  Accumulation is performed in FP32 for
numeric stability and the result is stored as FP16 (later cast back to FP32).

All other cheap operations (LayerNorms, linear projections, gating,
masking, final projection) are performed with PyTorch.  At least the core
N³ computation is executed on the GPU via Triton, satisfying the problem
requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,               # const float16 *
    right_ptr,              # const float16 *
    out_ptr,                #      float16 *
    B, H, N,                # int32 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM for TriMul.

    Computes:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    where `left` and `right` have layout [B, N, N, H] with indices
        left  -> (b, i, k, h)
        right -> (b, j, k, h)
    """
    pid = tl.program_id(0)

    # -----------------------------------------------------------------
    # Compute 3‑D block indices (combined batch, i‑tile, j‑tile)
    # -----------------------------------------------------------------
    num_m_blocks = tl.cdiv(N, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)

    bh = pid // (num_m_blocks * num_n_blocks)                 # combined (b, h) index
    rem = pid % (num_m_blocks * num_n_blocks)
    m_block = rem // num_n_blocks
    n_block = rem % num_n_blocks

    b = bh // H               # batch index
    h = bh % H                # hidden‑dim index

    # -----------------------------------------------------------------
    # Starting coordinates of the tile
    # -----------------------------------------------------------------
    i0 = m_block * BLOCK_M
    j0 = n_block * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the K dimension
    # -----------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right block needs to be transposed to (BLOCK_K, BLOCK_N)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
        )
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # accumulate dot product
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # -----------------------------------------------------------------
    # Write results back (store as FP16)
    # -----------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32) or None
        - weights      : dict of model weights (tensors on the same device)
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections and gates (no bias)
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 3) Apply gates and optional mask (elementwise)
    # -----------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Cast to half‑precision for the heavy GEMM
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # -----------------------------------------------------------------
    # 4) Triton batched GEMM (the N³ reduction)
    # -----------------------------------------------------------------
    # Output buffer (FP16, later cast back)
    out_hidden = torch.empty((B, N, N, hidden_dim),
                             dtype=torch.float16,
                             device=x.device)

    # Tile sizes – tuned for the B200 GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid size: one program per (batch*head) tile pair
    num_m_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_n_blocks = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * hidden_dim * num_m_blocks * num_n_blocks
    grid = (total_blocks,)

    tri_mul_gemm_kernel[grid](
        left, right, out_hidden,
        B, hidden_dim, N,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        out_hidden.stride(0), out_hidden.stride(1), out_hidden.stride(2), out_hidden.stride(3),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 for the remaining layers
    out_hidden = out_hidden.to(torch.float32)

    # -----------------------------------------------------------------
    # 5) Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
