"""
TriMul (outgoing) forward pass – Triton + PyTorch implementation.

The computational core `out = left @ right` is performed with a batched
matrix‑multiplication (`torch.matmul`) that relies on cuBLAS/wmma for
maximum throughput.  A small Triton kernel fuses the elementwise mask
and gate multiplications, reducing the number of memory passes.
All remaining linear layers and LayerNorms are carried out with
torch.nn.functional for simplicity.  The result is a float‑32 tensor
with shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _apply_gates_mask_kernel(
    x_ptr,    # input value pointer (half)
    gate_ptr, # gate pointer (half)
    mask_ptr, # mask pointer (half)
    out_ptr,  # output pointer (half)
    numel,    # total number of elements
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel

    x = tl.load(x_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    m = tl.load(mask_ptr + offs, mask=mask, other=0.0).to(tl.float32)

    out = x * g * m
    tl.store(out_ptr + offs, out.to(tl.float16), mask=mask)

        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Project to hidden dimension (half precision)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,hidden]
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3) Compute gates (float32) and cast to half
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden], fp32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # fp32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # fp32

    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Prepare mask in half‑precision, broadcast to hidden dim
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    if mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)                # [B,N,N,1]
    else:
        mask_h = torch.ones((B, N, N, 1), dtype=torch.float16, device=x.device)

    # Expand to full hidden dimension (contiguous)
    mask_h = mask_h.expand(-1, -1, -1, hidden_dim).contiguous()

    # ------------------------------------------------------------------
    # 5) Fuse mask and gate multiplication using a tiny Triton kernel
    # ------------------------------------------------------------------
    numel = left.numel()
    BLOCK = 1024
    grid = lambda meta: ((numel + meta['BLOCK'] - 1) // meta['BLOCK'],)

    # In‑place update: left = left * left_gate * mask
    _apply_gates_mask_kernel[grid](
        left,
        left_gate_h,
        mask_h,
        left,
        numel,
        BLOCK=BLOCK,
    )
    # In‑place update for right
    _apply_gates_mask_kernel[grid](
        right,
        right_gate_h,
        mask_h,
        right,
        numel,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 6) Batched matrix multiplication (outgoing TriMul)
    #    left: (B, N, N, hidden) -> (B, hidden, N, N)
    #    right: (B, N, N, hidden) -> (B, hidden, N, N) with K dimension first
    # ------------------------------------------------------------------
    left_mat = left.permute(0, 3, 1, 2)   # (B, hidden, N, N)   half
    right_mat = right.permute(0, 3, 2, 1) # (B, hidden, N, N)   half, K as first inner dim

    # torch.matmul uses cuBLAS and exploits tensor cores for half inputs
    out = torch.matmul(left_mat, right_mat)   # (B, hidden, N, N) half

    # Cast to float32 for the remaining layers
    out = out.to(torch.float32)

    # Restore original layout: (B, N, N, hidden)
    out = out.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass – Triton + PyTorch implementation.

The heavy N³ contraction is performed as a batched matrix multiplication
with torch.matmul (cuBLAS/wmma), while a tiny Triton kernel fuses the
mask and gate element‑wise multiplications into a single pass.  The
remaining linear layers and LayerNorms are executed with torch.nn.functional.
The final output is float32 with shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _apply_gates_mask_kernel(
    x_ptr,       # input tensor (half)
    gate_ptr,    # gate tensor (half)
    mask_ptr,    # mask tensor (half, same shape as x)
    out_ptr,     # output tensor (half, may be same as x_ptr)
    numel,       # total number of elements (int)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    m = offs < numel

    # Load as fp16, promote to fp32 for the multiplication
    x   = tl.load(x_ptr   + offs, mask=m, other=0.0).to(tl.float32)
    g   = tl.load(gate_ptr+ offs, mask=m, other=0.0).to(tl.float32)
    mk  = tl.load(mask_ptr+ offs, mask=m, other=0.0).to(tl.float32)

    out = x * g * mk
    tl.store(out_ptr + offs, out.to(tl.float16), mask=m)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of tensors (all float32)
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Project to hidden dimension (half precision)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left  = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,hidden]
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3) Compute gates (float32) and cast to half
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden] fp32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # fp32
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # fp32

    left_gate_h  = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Prepare mask (half, broadcasted to hidden dim)
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    if mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)           # [B,N,N,1]
    else:
        mask_h = torch.ones((B, N, N, 1), dtype=torch.float16, device=x.device)

    # Expand to full hidden dimension and make contiguous (adds ~size of left)
    mask_h = mask_h.expand(-1, -1, -1, hidden_dim).contiguous()

    # ------------------------------------------------------------------
    # 5) Fuse mask and gate multiplication with a tiny Triton kernel
    # ------------------------------------------------------------------
    numel = left.numel()                                 # B*N*N*hidden
    BLOCK = 1024
    grid = lambda meta: ((numel + meta['BLOCK'] - 1) // meta['BLOCK'],)

    # In‑place: left  = left  * left_gate_h  * mask_h
    _apply_gates_mask_kernel[grid](
        left,
        left_gate_h,
        mask_h,
        left,
        numel,
        BLOCK=BLOCK,
    )
    # In‑place: right = right * right_gate_h * mask_h
    _apply_gates_mask_kernel[grid](
        right,
        right_gate_h,
        mask_h,
        right,
        numel,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 6) Batched matrix multiplication (outgoing TriMul)
    #    left  : (B, hidden, N, N)  – rows i, cols k
    #    right : (B, hidden, N, N)  – rows k, cols j (k index first)
    # ------------------------------------------------------------------
    left_mat  = left.permute(0, 3, 1, 2)   # (B, hidden, N, N) half
    right_mat = right.permute(0, 3, 2, 1)  # (B, hidden, N, N) half, k as inner dim

    # cuBLAS‑accelerated batched GEMM; FP16 inputs → FP16 output (accumulate in FP32 internally)
    out = torch.matmul(left_mat, right_mat)   # (B, hidden, N, N) half

    # Convert to float32 for the remaining operations
    out = out.to(torch.float32)

    # Restore the original layout: (B, N, N, hidden)
    out = out.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])   # back to dim (float32)

    return out
