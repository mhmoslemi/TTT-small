import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def cast_fp16_to_fp32_kernel(in_ptr, out_ptr, N_ELEM: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM
    val = tl.load(in_ptr + offs, mask=mask) # loads fp16
    val_f32 = val.to(tl.float32)
    tl.store(out_ptr + offs, val_f32, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1. LayerNorm
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # 2. Linear projections
    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden)
    right = F.linear(x, right_proj_weight)

    # 3. Mask
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # 4. Gating
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # 5. Apply gates and cast to fp16 for the heavy reduction
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    B, N, _, _ = left.shape  # B,N,N,hidden

    # 6. Reshape for batched GEMM without extra copy
    left = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*hidden,N,N)
    right = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # 7. Batched matmul (FP16)
    out_fp16 = torch.bmm(left, right.transpose(1, 2))  # (B*hidden,N,N)

    # 8. Reshape back to (B,N,N,hidden) and cast to fp32 via Triton
    out_fp16 = out_fp16.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,hidden)
    total_elem = out_fp16.numel()
    out_fp32 = torch.empty_like(out_fp16, dtype=torch.float32)

    BLOCK = 65536
    grid = ( (total_elem + BLOCK - 1) // BLOCK, )
    cast_fp16_to_fp32_kernel[grid](
        in_ptr=out_fp16,
        out_ptr=out_fp32,
        N_ELEM=total_elem,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # 9. Output LayerNorm, gating, projection
    out = F.layer_norm(out_fp32,
                       normalized_shape=(hidden,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # (B,N,N,dim)
    return out

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy N³ reduction is performed as a batched half‑precision GEMM
(`torch.bmm`) without any unnecessary contiguous copies.  The only
Triton kernel is a tiny FP16→FP32 cast used after the GEMM to keep the
interface requirements while avoiding the large copy + cast of the
previous implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: cast a flat FP16 buffer to FP32
# -------------------------------------------------------------
@triton.jit
def cast_fp16_to_fp32_kernel(
    in_ptr,          # FP16 source
    out_ptr,         # FP32 destination
    N_ELEM: tl.constexpr,   # total number of elements
    BLOCK: tl.constexpr,    # block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM
    val = tl.load(in_ptr + offs, mask=mask).to(tl.float32)
    tl.store(out_ptr + offs, val, mask=mask)


# -------------------------------------------------------------
# Entry point used by the evaluation harness
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict with the model weight tensors
            - config: dict with at least "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Weight shortcuts
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the `dim` channel
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5) Apply gates and cast to FP16 for the O(N³) reduction
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    B, N, _, _ = left.shape  # B, N, N, hidden

    # ------------------------------------------------------------------
    # 6) Reshape to batched GEMM layout without extra copies
    #    left/right: (B, hidden, N, N) -> (B*hidden, N, N)
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # ------------------------------------------------------------------
    # 7) Batched matrix multiplication (FP16, tensor cores)
    # ------------------------------------------------------------------
    out_fp16 = torch.bmm(left, right.transpose(1, 2))  # [B*hidden, N, N]

    # ------------------------------------------------------------------
    # 8) Reshape back to (B, N, N, hidden) and cast to FP32 via Triton
    # ------------------------------------------------------------------
    out_fp16 = out_fp16.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    total_elems = out_fp16.numel()
    out_fp32 = torch.empty_like(out_fp16, dtype=torch.float32)

    BLOCK = 65536
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    cast_fp16_to_fp32_kernel[grid](
        in_ptr=out_fp16,
        out_ptr=out_fp32,
        N_ELEM=total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 9) Final LayerNorm, output gating and projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # apply the output gate
    out = F.linear(out, to_out_weight)       # [B, N, N, dim]

    return out
