"""
Optimized forward pass for the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor.
2. Compute four linear projections: left_proj, right_proj, left_gate, right_gate,
   and the output gate (all without bias).  Apply sigmoid to the gates.
3. Fuse element‑wise mask, projection and gate multiplication in a small
   Triton kernel (if a mask is provided).  If no mask, perform the element‑wise
   product with PyTorch.
4. Compute the core N³ reduction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication `L @ Rᵀ` for each (batch, hidden)
   channel.  We reshape to (B·H, N, N) and use `torch.bmm` with FP16
   arithmetic for speed.
5. Apply a second Layer‑norm, multiply by the output gate, and project back to
   the original feature dimension.

The kernel `fused_mask_gate_kernel` performs the element‑wise
`proj * gate * mask` operation in a single pass, reducing memory traffic.
All other heavy operations rely on highly‑optimized PyTorch (cuBLAS) kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Element‑wise fusion: out = proj * gate * mask

    Arguments
    ----------
    proj_ptr, gate_ptr : pointers to [B, N, N, H] tensors (float32)
    mask_ptr           : pointer to [B, N, N] tensor (float32, broadcast over H)
    out_ptr            : pointer to output tensor [B, N, N, H] (float32)
    B, N, H            : dimensions
    stride_*           : strides in elements for each tensor
    BLOCK_SIZE         : number of elements processed per program instance
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # mask for out‑of‑bounds linear indices
    valid = offs < B * N * N * H

    # decode flat index -> (b,i,j,h)
    tmp = offs
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # compute memory offsets
    proj_off = (b * stride_proj_b + i * stride_proj_i +
                j * stride_proj_j + h * stride_proj_h)
    gate_off = (b * stride_gate_b + i * stride_gate_i +
                j * stride_gate_j + h * stride_gate_h)
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_off  = (b * stride_out_b + i * stride_out_i +
                j * stride_out_j + h * stride_out_h)

    # load values (zero for OOB)
    proj = tl.load(proj_ptr + proj_off, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_off, mask=valid, other=0.0)

    # store fused result
    tl.store(out_ptr + out_off, proj * gate * m, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model parameters
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # -----------------------------------------------------------------
    # 2) Linear projections and gate computation (no bias)
    # -----------------------------------------------------------------
    left_proj  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,H]

    # -----------------------------------------------------------------
    # 3) Fuse mask and gate (if a mask is provided)
    # -----------------------------------------------------------------
    if mask is not None:
        # Ensure mask is float32 and on the same device
        mask = mask.to(x.dtype).contiguous()

        # Allocate output tensors
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # Strides in elements
        s_lp_b, s_lp_i, s_lp_j, s_lp_h = left_proj.stride()
        s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate.stride()
        s_rp_b, s_rp_i, s_rp_j, s_rp_h = right_proj.stride()
        s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate.stride()
        s_m_b, s_m_i, s_m_j           = mask.stride()
        s_out_b, s_out_i, s_out_j, s_out_h = left.stride()

        total = B * N * N * hidden_dim
        BLOCK_SIZE = 1024
        grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE,)

        # left side fusion
        fused_mask_gate_kernel[grid](
            left_proj, left_gate, mask, left,
            B, N, hidden_dim,
            s_lp_b, s_lp_i, s_lp_j, s_lp_h,
            s_lg_b, s_lg_i, s_lg_j, s_lg_h,
            s_m_b, s_m_i, s_m_j,
            s_out_b, s_out_i, s_out_j, s_out_h,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # right side fusion
        fused_mask_gate_kernel[grid](
            right_proj, right_gate, mask, right,
            B, N, hidden_dim,
            s_rp_b, s_rp_i, s_rp_j, s_rp_h,
            s_rg_b, s_rg_i, s_rg_j, s_rg_h,
            s_m_b, s_m_i, s_m_j,
            s_out_b, s_out_i, s_out_j, s_out_h,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 4) Core N³ reduction via batched matrix multiplication (FP16)
    # -----------------------------------------------------------------
    # Cast to half precision for speed
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange to (B*H, N, N) and transpose right for matmul
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (transposed)

    left_mat  = left.view(-1, N, N)   # (B*H, N, N)
    right_mat = right.view(-1, N, N)  # (B*H, N, N)

    # Batched matrix multiplication: out = left @ right
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N), dtype=fp16

    # Restore original shape [B, N, N, H]
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # -----------------------------------------------------------------
    # 5) Output LayerNorm, gate multiplication and final projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
```
