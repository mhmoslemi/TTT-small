"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B, N, N, dim]).
2. Compute four linear projections of `x`:
   - left_proj, right_proj          (no bias)
   - left_gate, right_gate, out_gate (sigmoid‑gated)
3. Fuse mask (if provided), gate, and projection with a small Triton kernel:
   out = proj * gate * mask   →   left, right tensors of shape [B,N,N,hidden].
4. Cast `left` and `right` to float16 and reshape to (B, hidden, N, N).
5. Perform the N³ reduction as a batched half‑precision matrix multiplication:
      out = left @ rightᵀ   (per hidden channel)
   This is a single call to `torch.matmul`, which uses cuBLAS.
6. Cast the result back to float32, permute to [B, N, N, hidden].
7. Apply the output layer‑norm, multiply by `out_gate`, and finally project
   back to the original channel dimension with a linear layer.
All heavy element‑wise work (mask × gate × proj) is done in Triton, while the
dominant N³ reduction leverages the highly tuned cuBLAS GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_SIZE: tl.constexpr,
):
    """
    elementwise kernel: out = proj * gate * mask

    Arguments
    ---------
    proj_ptr, gate_ptr : pointers to tensors of shape [B, N, N, H] (float32)
    mask_ptr           : pointer to tensor of shape [B, N, N] (float32, broadcasted)
    out_ptr            : pointer to output tensor of shape [B, N, N, H] (float32)
    B, N, H            : problem dimensions
    stride_*           : strides (in elements) for each tensor dimension
    BLOCK_SIZE         : number of elements processed per program instance
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # guard against out‑of‑bounds
    valid = offs < B * N * N * H

    # decode linear index → (b, i, j, h)
    tmp = offs
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # compute flat offsets for each tensor
    proj_off = (b * stride_proj_b + i * stride_proj_i +
                j * stride_proj_j + h * stride_proj_h)
    gate_off = (b * stride_gate_b + i * stride_gate_i +
                j * stride_gate_j + h * stride_gate_h)
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_off  = (b * stride_out_b + i * stride_out_i +
                j * stride_out_j + h * stride_out_h)

    proj = tl.load(proj_ptr + proj_off, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=valid, other=0.0)
    m    = tl.load(mask_ptr + mask_off, mask=valid, other=0.0)

    tl.store(out_ptr + out_off, proj * gate * m, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict mapping weight names to tensors
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack arguments
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections and gate computations (no bias)
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 3) Fuse mask, projection and gate (Triton)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype)  # ensure same dtype as projections

        # allocate output buffers
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # obtain strides (in elements) for all tensors
        s_lp_b, s_lp_i, s_lp_j, s_lp_h = left_proj.stride()
        s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate.stride()
        s_rp_b, s_rp_i, s_rp_j, s_rp_h = right_proj.stride()
        s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate.stride()
        s_m_b, s_m_i, s_m_j = mask_f.stride()
        s_out_b, s_out_i, s_out_j, s_out_h = left.stride()

        total = B * N * N * hidden_dim
        BLOCK_SIZE = 4096
        grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE,)

        # left side
        fused_mask_gate_kernel[grid](
            left_proj,
            left_gate,
            mask_f,
            left,
            B,
            N,
            hidden_dim,
            s_lp_b,
            s_lp_i,
            s_lp_j,
            s_lp_h,
            s_lg_b,
            s_lg_i,
            s_lg_j,
            s_lg_h,
            s_m_b,
            s_m_i,
            s_m_j,
            s_out_b,
            s_out_i,
            s_out_j,
            s_out_h,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # right side
        fused_mask_gate_kernel[grid](
            right_proj,
            right_gate,
            mask_f,
            right,
            B,
            N,
            hidden_dim,
            s_rp_b,
            s_rp_i,
            s_rp_j,
            s_rp_h,
            s_rg_b,
            s_rg_i,
            s_rg_j,
            s_rg_h,
            s_m_b,
            s_m_i,
            s_m_j,
            s_out_b,
            s_out_i,
            s_out_j,
            s_out_h,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 4) Batched N³ reduction via half‑precision matrix multiplication
    # -----------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape to (B, H, N, N) where the last two dims are (i,k) and (k,i)
    left = left.permute(0, 3, 1, 2)          # (B, H, N, N)  ← (i,k)
    right_t = right.permute(0, 3, 2, 1)      # (B, H, N, N)  ← (k,i) = transpose of (i,k)

    # batched matrix multiplication: for each (b, h) compute L @ Rᵀ
    out = torch.matmul(left, right_t)       # (B, H, N, N)

    # back to [B, N, N, H] and convert to fp32 for the remaining ops
    out = out.permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # -----------------------------------------------------------------
    # 5) Output LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
