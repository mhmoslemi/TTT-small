def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Projections in half
    x_h = x_norm.to(torch.float16)

    left = torch.nn.functional.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    B, N, _, _ = x.shape

    if mask is not None:
        # ensure mask dtype float16 and shape
        mask_h = mask.to(torch.float16)  # shape [B,N,N]
        # Apply mask and gates using Triton kernel
        BLOCK_SIZE = 1024
        total_elements = B * N * N * hidden_dim
        grid = ( (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE, )
        _apply_gate_mask_kernel[grid](
            left, right,
            left_gate_h, right_gate_h,
            mask_h,
            B, N, hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # After kernel left and right modified in-place
    else:
        # elementwise multiply gates
        left = left * left_gate_h
        right = right * right_gate_h

    # Now compute contraction via batched matmul
    # Permute shapes
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # uses fp16 -> fp16 result
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()  # convert to float32 for layernorm

    # Output layernorm and gating and final linear
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated gating/mask step +
batched GEMM via torch.bmm.

The core contraction is
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication left @ rightᵀ for each
(batch, hidden) pair.  We compute the projections and layer‑norms in
PyTorch, fuse the mask‑ and gate‑application with a tiny Triton kernel,
then perform the O(N³) reduction with the highly‑optimized cuBLAS
batched GEMM (torch.bmm).  The final layer‑norm, gating and output
projection complete the forward pass.  The implementation works for
both masked and un‑masked inputs and supports the dimension ranges used
by the benchmark suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,          # [B, N, N, H]   fp16 (in‑place)
    left_gate_ptr, right_gate_ptr,  # same shape as left/right (fp16)
    mask_ptr,                     # [B, N, N]      fp16 (0/1 mask)
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    total_elems: tl.int32,
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place: left = left * left_gate * mask,
                right = right * right_gate * mask.
    All tensors are contiguous in memory with layout (B,N,N,H)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # compute multi‑dim indices
    N2H = N * N * H
    NH = N * H

    b = offs // N2H
    rem = offs % N2H
    i = rem // NH
    rem = rem % NH
    k = rem // H
    d = rem % H

    # flat index for the 4‑D tensors
    idx = ((b * N + i) * N + k) * H + d
    # flat index for mask (no H dimension)
    mask_idx = ((b * N + i) * N + k)

    # Load values (zero‑padding for OOB)
    l = tl.load(left_ptr + idx, mask=mask, other=tl.float16(0.0))
    r = tl.load(right_ptr + idx, mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + idx, mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + idx, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))

    # Apply gate and mask
    tl.store(left_ptr + idx,  l * lg * m, mask=mask)
    tl.store(right_ptr + idx, r * rg * m, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32) or None
            - weights: dict with the following keys (float32 tensors)
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config: dict with at least 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden_dim (use fp16 for speed)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))   # [B, N, N, hidden_dim] fp16
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))  # same shape

    # ------------------------------------------------------------------
    # 3) Gates (computed in FP32 then cast)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # FP32

    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 4) Apply mask + gates (fused Triton kernel if mask present)
    # ------------------------------------------------------------------
    if mask is not None:
        # Convert mask to fp16 (values 0.0 / 1.0)
        mask_h = mask.to(torch.float16)
        # Launch kernel
        BLOCK_SIZE = 1024
        total_elems = B * N * N * hidden_dim
        grid = ( (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE, )
        _gate_mask_kernel[grid](
            left, right,
            left_gate_h, right_gate_h,
            mask_h,
            B, N, hidden_dim,
            total_elems,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # The kernel updates `left` and `right` in‑place.
    else:
        # No mask – simple element‑wise multiplication
        left = left * left_gate_h
        right = right * right_gate_h

    # ------------------------------------------------------------------
    # 5) Core N³ contraction via batched GEMM (torch.bmm)
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  – k‑axis first

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # (BH, M, K)
    right_mat = right_perm.view(BH, N, N)  # (BH, K, N)

    # Batched matrix multiplication (fp16 inputs, fp16 output)
    out_mat = torch.bmm(left_mat, right_mat)  # (BH, N, N) fp16

    # Restore original layout: [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16
    out = out.float()  # convert to FP32 for the following layers

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # FP32

    out = out * out_gate  # element‑wise gating (FP32)

    out = F.linear(out, weights["to_out.weight"])  # back to dim, FP32

    return out
