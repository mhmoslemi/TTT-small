"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The core operation is:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right are linear projections of the input followed by
element‑wise gating.  This is a batched GEMM (A @ Bᵀ) for each
hidden dimension.  The kernel below implements the batched GEMM in
Triton (fp16×fp16 → fp32 accumulation) and the surrounding
operations are performed with PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A, B : (batch, M, K)   fp16
#   C    : (batch, M, N)   fp32
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,                 # ptr to left tensor  (batch, M, K)
    b_ptr,                 # ptr to right tensor (batch, N, K)  (will be loaded transposed)
    c_ptr,                 # ptr to output tensor (batch, M, N)
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)        # row tile
    pid_n = tl.program_id(1)        # col tile
    pid_b = tl.program_id(2)        # batch tile

    # Compute offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for cur_k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = cur_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_a_m) + (k_offset[None, :] * stride_a_k),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load Bᵀ tile (original B is (N, K); we need (K, N))
        b = tl.load(
            b_batch + (k_offset[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate dot product (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # Store result (fp32)
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for outgoing TriMul.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool or float)
            - weights: dict with the following keys:
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict containing at least:
                * "dim" (int)
                * "hidden_dim" (int)
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm over the last dim
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Batched GEMM: out = left @ rightᵀ  (over the K dimension)
    # ------------------------------------------------------------------
    # Convert to fp16 for the GEMM
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, N, N, hidden_dim

    # Rearrange to (B * hidden_dim, N, N) where each slice is an independent GEMM
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    batch = B * hidden_dim
    M = N
    K = N
    # Allocate output buffer in fp32
    out_h = torch.empty((batch, N, N), dtype=torch.float32, device=device)

    # Strides (in elements)
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)

    stride_b_batch = right_h.stride(0)
    # For Bᵀ we treat dimension 2 as K and dimension 1 as N after transposition
    stride_b_k = right_h.stride(2)   # original column stride
    stride_b_n = right_h.stride(1)   # original row stride

    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    # Tunable block sizes; chosen to map well to tensor cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_pid_m = (M + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m, num_pid_n, batch)

    # Launch Triton kernel
    tri_mul_outgoing_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        M=M,
        N=N,
        K=K,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] and convert to fp32
    # ------------------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)          # [B, N, N, hidden_dim]
        .contiguous()
    )  # already fp32

    # ------------------------------------------------------------------
    # Final LayerNorm, output gating, and projection back to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
