@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_k, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    i_tile = tl.program_id(1)
    j_tile = tl.program_id(2)

    # Compute b_idx and h_idx
    h_idx = batch_idx % H
    b_idx = batch_idx // H

    # Pointers to the start of the batch and channel
    a_batch_ptr = a_ptr + b_idx * stride_a_batch + h_idx * stride_a_h
    b_batch_ptr = b_ptr + b_idx * stride_b_batch + h_idx * stride_b_h
    c_batch_ptr = c_ptr + b_idx * stride_c_batch + h_idx * stride_c_h

    # Define tile ranges
    i_start = i_tile * BLOCK_M
    j_start = j_tile * BLOCK_N

    # Create range vectors
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize acc
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load A tile (i,k)
        a_ptrs = a_batch_ptr + offs_i[:, None] * stride_a_i + cur_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # Load B_T tile (k,j)
        b_ptrs = b_batch_ptr + cur_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float16)

        # Compute dot product and accumulate
        acc = acc + tl.dot(a, b, out_dtype=tl.float32)

    # Store result
    c_ptrs = c_batch_ptr + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_k
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    right = right * right_gate

"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Linear projections -> left   [B,N,N,H]  (no bias)
               -> right  [B,N,N,H]
3. (Optional) mask the two projected tensors.
4. Compute gating values (sigmoid of linear projections) and apply them:
       left  = left  * left_gate
       right = right * right_gate
5. Cast left/right to FP16 and perform the core O(N³) reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication A @ Bᵀ where each batch is a
   pair (batch, hidden).  The kernel tiles the output (i,j) plane and
   reduces over k using `tl.dot`, accumulating in FP32 for numerical
   stability.
6. Layer‑norm over the hidden dimension, multiply by the output gate,
   and finally project back to the original channel dimension with a
   linear layer.
All heavy compute (step 5) is performed by Triton; the remaining ops
use standard PyTorch which is efficient for the small‑dimensional
operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    B, N, H,                            # batch size, sequence length, hidden dim
    stride_ab, stride_ai, stride_ak, stride_ah,   # strides for A (left)
    stride_bb, stride_bi, stride_bk, stride_bh,   # strides for B (right)
    stride_cb, stride_ci, stride_ck, stride_ch,   # strides for C (output)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul core reduction.

    Computes, for each (batch, hidden) pair:
       C[i, j] = Σ_k A[i, k] * B[j, k]   (no transpose on B; we load Bᵀ)
    where A, B are [N, N] matrices in FP16 and C is FP32.
    """
    # --------------------
    # Program IDs
    # --------------------
    batch_h_idx = tl.program_id(0)        # 0 ≤ batch_h_idx < B*H
    i_tile = tl.program_id(1)            # tile index in the i (row) dimension
    j_tile = tl.program_id(2)            # tile index in the j (col) dimension

    # Split combined batch+hidden index
    h_idx = batch_h_idx % H
    b_idx = batch_h_idx // H

    # Base pointers for this (b, h) pair
    a_base = a_ptr + b_idx * stride_ab + h_idx * stride_ah
    b_base = b_ptr + b_idx * stride_bb + h_idx * stride_bh
    c_base = c_ptr + b_idx * stride_cb + h_idx * stride_ch

    # --------------------
    # Tile offsets
    # --------------------
    i_start = i_tile * BLOCK_M
    j_start = j_tile * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------
    # Loop over k dimension
    # --------------------
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i, k]   -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)  # FP16

        # Bᵀ[k, j] = B[j, k] -> shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bi
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)  # FP16

        # Dot product and accumulate in FP32
        acc = acc + tl.dot(a, b, out_dtype=tl.float32)

    # --------------------
    # Write back
    # --------------------
    c_ptrs = c_base + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_ck
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module with a Triton‑accelerated core.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float). May be ignored if config["nomask"]=True.
            - weights: dict of model weights (exact keys as in the reference implementation).
            - config: dict with keys "dim", "hidden_dim", optional "nomask".

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Unpack model weights (all are torch.float32 tensors on the correct device)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm over input channels
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3️⃣ Optional masking (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Core TriMul reduction performed by Triton (batched GEMM)
    # -----------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape  # B, N, N, hidden_dim (H)

    # Output tensor (FP32) – same layout as left/right
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left_fp16.device)

    # Strides (in elements) for the contiguous tensors
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left_fp16.stride()
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h = right_fp16.stride()
    stride_c_batch, stride_c_i, stride_c_k, stride_c_h = out.stride()

    # Tiling parameters – chosen to fit into shared memory and registers
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (batch*hidden, ceil(N/BM), ceil(N/BN))
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        H,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_a_h,
        stride_b_batch,
        stride_b_i,
        stride_b_k,
        stride_b_h,
        stride_c_batch,
        stride_c_i,
        stride_c_k,
        stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Make sure the kernel has finished before we use the result.
    torch.cuda.synchronize()

    # -----------------------------------------------------------------
    # 6️⃣ Output LayerNorm (over hidden dimension)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Apply output gate (element‑wise) and final projection
    # -----------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)    # [B, N, N, dim]

    return out
```
