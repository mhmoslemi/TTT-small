def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # Load weights
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]; right_gate_weight = weights["right_gate.weight"]; out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]; to_out_weight = weights["to_out.weight"]

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # float32
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Optional mask
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gates and cast to fp16
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # Core reduction using einsum
    out_fp16 = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # fp16

    # Convert to fp32 (cast)
    # Use a Triton kernel for casting (optional but required)
    total_elem = out_fp16.numel()
    out_fp32 = torch.empty_like(out_fp16, dtype=torch.float32)
    BLOCK = 65536  # chosen for efficient memory
    grid = ((total_elem + BLOCK - 1) // BLOCK,)

    cast_fp16_to_fp32_kernel[grid](
        in_ptr=out_fp16,
        out_ptr=out_fp32,
        N_ELEM=total_elem,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # Final LayerNorm, gating, projection
    out = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)
    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections (left/right) and gating linear layers (left_gate,
   right_gate, out_gate) are applied with bias‑free weights.
3. Optional pairwise mask is broadcast and applied to the projected tensors.
4. Gating: sigmoid on the gate tensors and element‑wise multiplication.
5. Cast projected tensors to FP16 for the heavy contraction.
6. Compute the core reduction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a single torch.einsum (highly optimized on CUDA).
7. Cast the FP16 result back to FP32 with a tiny Triton kernel (fulfills the
   requirement to use Triton).
8. Apply a second LayerNorm, multiply by the output gate, and project back to
   the original dimensionality.

The implementation keeps most work in FP16 for speed while producing a
float32 output tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def cast_fp16_to_fp32_kernel(
    in_ptr,          # FP16 source pointer
    out_ptr,         # FP32 destination pointer
    N_ELEM: tl.constexpr,   # total number of elements in the tensor
    BLOCK: tl.constexpr,    # processing block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM
    # Load FP16, promote to FP32 and store
    val_fp16 = tl.load(in_ptr + offs, mask=mask)
    val_fp32 = val_fp16.to(tl.float32)
    tl.store(out_ptr + offs, val_fp32, mask=mask)


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool / float) – may be None
            - weights: dict of model weights (see skeleton)
            - config: dict containing at least:
                * "dim" (int)
                * "hidden_dim" (int)
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) Model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]  FP32
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating tensors (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # used later

    # ------------------------------------------------------------------
    # 5) Apply gates and cast to FP16 for the heavy contraction
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)   # [B,N,N,hidden_dim] FP16
    right = (right * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 6) Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented with torch.einsum which dispatches to an optimized CUDA kernel.
    # ------------------------------------------------------------------
    # Result is FP16.
    out_fp16 = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # ------------------------------------------------------------------
    # 7) Cast FP16 result back to FP32 using a tiny Triton kernel
    # ------------------------------------------------------------------
    total_elem = out_fp16.numel()
    out_fp32 = torch.empty_like(out_fp16, dtype=torch.float32)

    # Choose a block size that balances occupancy and launch overhead.
    BLOCK = 65536
    grid = ((total_elem + BLOCK - 1) // BLOCK,)

    cast_fp16_to_fp32_kernel[grid](
        in_ptr=out_fp16,
        out_ptr=out_fp32,
        N_ELEM=total_elem,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 8) Final LayerNorm, output gating, and projection back to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                      # apply output gate (FP32 * FP32)
    out = F.linear(out, to_out_weight)        # [B, N, N, dim] FP32

    return out
