@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_aha,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # batch/head index
    pid_m = tl.program_id(1)  # M block
    pid_n = tl.program_id(2)  # N block

    # calculate offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointer for this batch/head.
    a_ptr_base = a_ptr + pid * stride_aha
    b_ptr_base = b_ptr + pid * stride_bh
    c_ptr_base = c_ptr + pid * stride_ch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offsets
        k_offs = k * BLOCK_K + offs_k
        # masks for out-of-bounds
        a_mask = (offs_m[:, None] < M) & (k_offs[None, :] < K)
        b_mask = (k_offs[:, None] < K) & (offs_n[None, :] < N)

        # Load A tile
        a_ptrs = a_ptr_base + offs_m[:, None] * stride_am + k_offs[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=a_mask, other=0.0) # shape (BLOCK_M, BLOCK_K)
        # Load B tile
        b_ptrs = b_ptr_base + k_offs[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=b_mask, other=0.0) # (BLOCK_K, BLOCK_N)

        # accumulate
        acc += tl.dot(a, b)  # result (BLOCK_M, BLOCK_N), fp32

    # Store result
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_ptr_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)

"""
TriMul (outgoing) forward pass optimized with a Triton batched GEMM kernel.

The core of the operator is the contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
for each batch `b`, sequence positions `i,j` and hidden channel `d`.
We rewrite this as a batched matrix multiplication per hidden channel:
    out_{b,d} = left_{b,d} @ right_{b,d}ᵀ,
where left_{b,d} and right_{b,d} are (N,N) matrices.
The batched GEMM for all (b,d) pairs is performed by a single
Triton kernel that works on half‑precision inputs and accumulates in
float32. All surrounding operations (layer‑norm, linear projections,
gates, mask handling and final projection) are executed with PyTorch
for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# A : (BH, M, K)   FP16
# B : (BH, K, N)   FP16   (already transposed on the original 'right')
# C : (BH, M, N)   FP32 accumulator
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix dimensions
    stride_ah, stride_am, stride_ak,         # strides for A (batch, row, col)
    stride_bh, stride_bk, stride_bn,         # strides for B (batch, row, col)
    stride_ch, stride_cm, stride_cn,         # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication: C = A @ Bᵀ."""
    pid_batch = tl.program_id(0)   # which (batch * hidden) we process
    pid_m = tl.program_id(1)       # block row index
    pid_n = tl.program_id(2)       # block column index

    # Compute absolute offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers for the current batch/head
    a_batch_ptr = a_ptr + pid_batch * stride_ah
    b_batch_ptr = b_ptr + pid_batch * stride_bh
    c_batch_ptr = c_ptr + pid_batch * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        b_mask = (cur_k[:, None] < K) & (offs_n[None, :] < N)

        # Load A tile (M x K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)      # (BLOCK_M, BLOCK_K)

        # Load B tile (K x N)
        b_ptrs = b_batch_ptr + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)      # (BLOCK_K, BLOCK_N)

        # Matrix‑multiply‑accumulate
        acc += tl.dot(a, b)   # dot returns FP16 → accumulates into FP32

    # Write the result back (FP32)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with the following keys (float32 tensors):
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config: dict with entries 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden_dim (use fp16 for speed)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim] fp16
    right = F.linear(x_h, right_proj_w)  # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gates (computed in FP32, then cast to fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # FP32

    left_gate_h = left_gate.to(x_h.dtype)
    right_gate_h = right_gate.to(x_h.dtype)

    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 5) Core N³ contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N_seq, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim
    # Rearrange to (B, hidden_dim, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) – K axis already transposed

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N_seq, N_seq)   # (BH, M, K)
    right_mat = right_perm.view(BH, N_seq, N_seq) # (BH, K, N)

    # Output buffer (float32 accumulator)
    out_mat = torch.empty((BH, N_seq, N_seq), dtype=torch.float32, device=x.device)

    # --------------------------------------------------------------
    # Kernel launch parameters
    # --------------------------------------------------------------
    # Tune block sizes for the B200 (balanced register use & occupancy)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,
        (N_seq + BLOCK_M - 1) // BLOCK_M,
        (N_seq + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N_seq, N_seq, N_seq,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before proceeding
    torch.cuda.synchronize()

    # Reshape back to [B, N, N, hidden_dim] (float32)
    out = out_mat.view(B, hidden_dim, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # still float32

    out = out * out_gate  # element‑wise gating (float32)

    out = F.linear(out, weights["to_out.weight"])  # back to dim (float32)

    return out
