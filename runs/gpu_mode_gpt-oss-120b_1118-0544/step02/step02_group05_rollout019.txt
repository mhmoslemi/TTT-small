@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_ab, stride_ai, stride_ak,
    stride_bb, stride_bk, stride_bj,
    stride_cb, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch and head indices
    b = pid // H
    h = pid % H

    # offsets for this tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointer offset for batch b
    a_batch_ptr = a_ptr + b * stride_ab
    b_batch_ptr = b_ptr + b * stride_bb
    c_batch_ptr = c_ptr + b * stride_cb

    # start of slice for head h
    a_ptr_h = a_batch_ptr + h
    b_ptr_h = b_batch_ptr + h
    c_ptr_h = c_batch_ptr + h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        # compute offsets for k tile
        cur_k = kk * BLOCK_K + offs_k
        # create masks for bounds
        a_mask = (offs_i[:, None] < N) & (cur_k[None, :] < N)
        b_mask = (cur_k[:, None] < N) & (offs_j[None, :] < N)

        # calculate pointers
        a_ptrs = a_ptr_h + offs_i[:, None] * stride_ai + cur_k[None, :] * stride_ak
        b_ptrs = b_ptr_h + cur_k[:, None] * stride_bk + offs_j[None, :] * stride_bj

        # load
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # dot
        acc += tl.dot(a, b)

    # store
    c_mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    c_ptrs = c_ptr_h + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=c_mask)

    # linear projection: we will compute using torch.nn.functional.linear for fp16.

    x_h = x.to(torch.float16)

    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))

    right_gate_h = right_gate.to(torch.float16)

    left = left * left_gate_h
    right = right * right_gate_h

    B, N, _, _ = left.shape

    # Prepare output
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32  # tunable

"""
TriMul (outgoing) forward pass – Triton implementation.

The core operation:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is a batched matrix multiplication (C = A @ Bᵀ) where A and B have
shape (B, hidden_dim, N, N).  We avoid costly permutes/copies by
keeping the tensors in their native layout (B, N, N, hidden_dim) and
providing the correct strides to a Triton kernel.  The kernel works on
tiles (BLOCK_M x BLOCK_N) and reduces over K (the sequence dimension)
in blocks (BLOCK_K).  Masking and gating are performed in PyTorch
before the kernel to keep the kernel simple.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to left, right, output
    B, N, H,                           # batch, seq_len, hidden_dim
    stride_ab, stride_ai, stride_ak,   # strides for A (batch, i, k)
    stride_bb, stride_bk, stride_bj,   # strides for B (batch, k, j)
    stride_cb, stride_ci, stride_cj,   # strides for C (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the outgoing TriMul contraction.

    A: (B, N, N, H)   -> rows i, cols k   (M=N, K=N)
    B: (B, N, N, H)   -> rows j, cols k   (N=N, K=N)  (will be accessed as Bᵀ)
    C: (B, N, N, H)   -> rows i, cols j   (M=N, N=N)
    All tensors are stored in the layout (B, N, N, H) with H as the
    innermost dimension (stride_H = 1).
    """
    pid = tl.program_id(0)   # combined (batch, head) index
    pid_m = tl.program_id(1) # tile row (i)
    pid_n = tl.program_id(2) # tile column (j)

    # ---------------------------------------------------------------
    # Decompose pid into batch and head indices
    # ---------------------------------------------------------------
    b = pid // H
    h = pid % H

    # Offsets for the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers for this batch
    a_batch = a_ptr + b * stride_ab
    b_batch = b_ptr + b * stride_bb
    c_batch = c_ptr + b * stride_cb

    # Pointers for the current head (H is the innermost dimension -> stride 1)
    a_head = a_batch + h
    b_head = b_batch + h
    c_head = c_batch + h

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over the reduction dimension K
    # ---------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = kk * BLOCK_K + offs_k

        # Masks for out‑of‑bounds loads
        a_mask = (offs_i[:, None] < N) & (cur_k[None, :] < N)
        b_mask = (cur_k[:, None] < N) & (offs_j[None, :] < N)

        # Pointers for the current K‑tile
        a_ptrs = a_head + offs_i[:, None] * stride_ai + cur_k[None, :] * stride_ak
        b_ptrs = b_head + cur_k[:, None] * stride_bk + offs_j[None, :] * stride_bj

        # Load tiles (FP16) – OOB elements are zero‑filled
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Matrix‑multiply‑accumulate
        acc += tl.dot(a, b)          # result is up‑cast to FP32 by the +=

    # ---------------------------------------------------------------
    # Write the output tile
    # ---------------------------------------------------------------
    c_mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    c_ptrs = c_head + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (all tensors are float32)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ---------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim], FP32

    # ---------------------------------------------------------------
    # 2) Linear projections to hidden_dim (cast to fp16 for speed)
    # ---------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim], fp16
    right = F.linear(x_h, right_proj_w)  # same dtype/shape

    # ---------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden_dim)
    # ---------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ---------------------------------------------------------------
    # 4) Gates (computed in FP32, then cast to fp16)
    # ---------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # same shape
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # same shape

    left_gate_h = left_gate.to(x_h.dtype)
    right_gate_h = right_gate.to(x_h.dtype)

    left = (left * left_gate_h).contiguous()
    right = (right * right_gate_h).contiguous()

    B, N, _, _ = left.shape  # batch, seq_len, seq_len, hidden_dim

    # ---------------------------------------------------------------
    # 5) Core N³ contraction – batched GEMM via Triton
    # ---------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Tunable block sizes (empirically good for a wide range of N)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (batch*head, tiles over i, tiles over j)
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Strides of the tensors in their native layout (B, N, N, H)
    stride_ab = left.stride(0)   # batch stride
    stride_ai = left.stride(1)   # i (row) stride = N * H
    stride_ak = left.stride(2)   # k (col) stride = H

    stride_bb = right.stride(0)  # batch stride
    stride_bk = right.stride(2)  # k stride = H
    stride_bj = right.stride(1)  # j stride = N * H

    stride_cb = out.stride(0)    # batch stride
    stride_ci = out.stride(1)    # i stride = N * H
    stride_cj = out.stride(2)    # j stride = H

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_ab, stride_ai, stride_ak,
        stride_bb, stride_bk, stride_bj,
        stride_cb, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final projection
    # ---------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # out_gate is FP32, out is FP32
    out = F.linear(out, weights["to_out.weight"])  # back to `dim` (FP32)

    return out
