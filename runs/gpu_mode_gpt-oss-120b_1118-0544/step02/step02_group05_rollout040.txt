(fp16 inputs, fp32 accumulation) using a 2‑level tiling
(BLOCK_M × BLOCK_N tiles with a BLOCK_K reduction step).
Block sizes are chosen to keep the per‑program accumulator small
(64 × 64 × 4 B ≈ 16 KB) so the kernel stays register‑bound and
exploits Tensor‑Cores on the B200. All other operations (LayerNorm,
linear projections, gating, mask handling) are performed with
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dims
    stride_ah, stride_am, stride_ak,   # strides for A (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for B (batch, k, j)
    stride_ch, stride_cm, stride_cn,   # strides for C (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  (A: M×K, B: N×K → Bᵀ: K×N)"""
    pid_batch = tl.program_id(0)   # which (batch * head) we are processing
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block column

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch/head
    a_batch = a_ptr + pid_batch * stride_ah
    b_batch = b_ptr + pid_batch * stride_bh
    c_batch = c_ptr + pid_batch * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Pointers for the current tiles
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (fp16) with zero padding for OOB elements
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        # Matrix‑multiply‑accumulate (tensor‑core MMA)
        acc += tl.dot(a, b)   # result casted to fp32 before accumulation

    # Store the result
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.BoolTensor or torch.FloatTensor of shape [B, N, N] (or None)
            - weights: dict with the following keys (all float32 tensors)
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with entries "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = x.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden_dim (FP16 for speed)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    # shape: [B, N, N, hidden_dim] (fp16)
    left = F.linear(x_h, left_proj_w)
    right = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.dtype != torch.bool:
        # mask already float – just broadcast
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h
    elif mask is not None and mask.dtype == torch.bool:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h
    # If mask is None (or nomask), nothing to do

    # ------------------------------------------------------------------
    # 4) Gates (computed in FP32, then cast to FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim] FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # FP32

    # Cast gates to fp16 for element‑wise multiplication
    left_gate_h = left_gate.to(x_h.dtype)
    right_gate_h = right_gate.to(x_h.dtype)

    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 5) Core N³ contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N_seq, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim
    # Rearrange to (B, hidden_dim, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  (k as leading dim)

    # Collapse batch & head dimensions
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N_seq, N_seq)   # (BH, M, K)
    right_mat = right_perm.view(BH, N_seq, N_seq) # (BH, K, N)

    # Output buffer (FP32 accumulator)
    out_mat = torch.empty((BH, N_seq, N_seq), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 6) Launch Triton kernel
    # ------------------------------------------------------------------
    # Tile sizes – tuned for the B200 (keep accumulator < 16 KB)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        BH,
        (N_seq + BLOCK_M - 1) // BLOCK_M,
        (N_seq + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N_seq, N_seq, N_seq,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # out_gate is FP32, broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])  # → [B, N, N, dim] (FP32)

    return out
