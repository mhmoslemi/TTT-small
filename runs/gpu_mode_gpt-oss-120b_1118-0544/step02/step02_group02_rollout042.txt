    stride_bh, stride_bj, stride_bk,
    stride_ch, stride_ci, stride_cj,

@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    B, H, N,
    stride_a_batch, stride_a_hidden, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_hidden, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_hidden, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)       # batch*hidden
    pid_m = tl.program_id(1)     # i block
    pid_n = tl.program_id(2)     # j block

    # Compute batch and hidden indices
    b = pid // H
    h = pid - b * H

    # Base pointers for this (b, h)
    a_base = a_ptr + b * stride_a_batch + h * stride_a_hidden
    b_base = b_ptr + b * stride_b_batch + h * stride_b_hidden
    c_base = c_ptr + b * stride_c_batch + h * stride_c_hidden

    # Offsets for i and j dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load A tile (i, k)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + cur_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (k, j) from right (transposed view)
        b_ptrs = b_base + cur_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

This implementation follows the reference PyTorch code but moves the
computationally heavy reduction:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
into a custom Triton GEMM kernel.  The kernel treats the hidden dimension
as an extra batch axis, performing a batched matrix multiplication
with FP16 inputs and FP32 accumulation.  All other operations
(LayerNorm, linear projections, gating, final LayerNorm and projection)
remain in PyTorch.  This avoids the extra copy‑and‑cast overhead of the
previous solution and keeps memory traffic low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # problem sizes
    B, H, N,
    # strides for A (left)
    stride_a_batch, stride_a_hidden, stride_a_i, stride_a_k,
    # strides for B (right)
    stride_b_batch, stride_b_hidden, stride_b_i, stride_b_k,
    # strides for C (output)
    stride_c_batch, stride_c_hidden, stride_c_i, stride_c_j,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM for TriMul:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    """
    pid = tl.program_id(0)      # combined batch*hidden id
    pid_m = tl.program_id(1)    # i‑tile
    pid_n = tl.program_id(2)    # j‑tile

    # Decode batch and hidden indices
    batch_idx = pid // H
    hidden_idx = pid - batch_idx * H

    # Base pointers for this (batch, hidden) pair
    a_base = a_ptr + batch_idx * stride_a_batch + hidden_idx * stride_a_hidden
    b_base = b_ptr + batch_idx * stride_b_batch + hidden_idx * stride_b_hidden
    c_base = c_ptr + batch_idx * stride_c_batch + hidden_idx * stride_c_hidden

    # Tile offsets for the i and j dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + cur_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (transposed view): shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + cur_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # FP16 dot → FP32 accumulate
        acc += tl.dot(a, b)

    # Store the result tile
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model weights
            - config: dict containing at least:
                * "dim" (int)          – input channel dimension
                * "hidden_dim" (int)   – projection dimension
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) Grab model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to FP16 for the heavy reduction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Ensure contiguous layout (original layout is already contiguous)
    left_fp16 = left_fp16.contiguous()
    right_fp16 = right_fp16.contiguous()

    B, N, _, _ = left_fp16.shape

    # ------------------------------------------------------------------
    # 6) Triton batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)  # [B, N, N, hidden]

    # Strides for the original layout [B, N, N, H] (contiguous)
    stride_a_batch = left_fp16.stride(0)   # N*N*H
    stride_a_i = left_fp16.stride(1)       # N*H
    stride_a_k = left_fp16.stride(2)       # H
    stride_a_hidden = left_fp16.stride(3)  # 1

    stride_b_batch = right_fp16.stride(0)
    stride_b_i = right_fp16.stride(1)
    stride_b_k = right_fp16.stride(2)
    stride_b_hidden = right_fp16.stride(3)

    stride_c_batch = out_fp32.stride(0)
    stride_c_i = out_fp32.stride(1)        # N*H
    stride_c_j = out_fp32.stride(2)        # H
    stride_c_hidden = out_fp32.stride(3)   # 1

    # Kernel launch configuration
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    batch_hidden = B * hidden_dim
    grid = (
        batch_hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_gemm_kernel[grid](
        a_ptr=left_fp16,
        b_ptr=right_fp16,
        c_ptr=out_fp32,
        B=B,
        H=hidden_dim,
        N=N,
        stride_a_batch=stride_a_batch,
        stride_a_hidden=stride_a_hidden,
        stride_a_i=stride_a_i,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_hidden=stride_b_hidden,
        stride_b_i=stride_b_i,
        stride_b_k=stride_b_k,
        stride_c_batch=stride_c_batch,
        stride_c_hidden=stride_c_hidden,
        stride_c_i=stride_c_i,
        stride_c_j=stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7) Final LayerNorm, gating and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
