"""
Outgoing Triangle Multiplicative Update (TriMul) kernel.

Implements the forward pass of the “outgoing” TriMul operation from
AlphaFold3. The core computation is:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We first apply LayerNorm, linear projections and gating (all in PyTorch),
then perform the batched matrix multiplication left @ rightᵀ with a
custom Triton kernel. The kernel computes many independent GEMMs,
one per (batch, hidden_dim) pair, using fp16 arithmetic and fp32
accumulation. Finally a second LayerNorm, gating and projection return
a tensor of shape [B, N, N, dim] (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  tensor pointer  (batch, M, K)   fp16
    b_ptr,               # right tensor pointer  (batch, N, K)   fp16 (will be read transposed)
    c_ptr,               # output tensor pointer (batch, M, N)   fp32
    batch: tl.constexpr, # number of independent GEMMs = B * hidden_dim
    M: tl.constexpr,     # rows of A
    N: tl.constexpr,     # rows of B (cols of output)
    K: tl.constexpr,     # reduction dimension
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = left @ rightᵀ for many independent GEMMs."""
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # batch (gemm) index

    # Compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load tile of A: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load tile of Bᵀ: we need B transposed, i.e. B[k, n]
        b_ptrs = b_batch + (cur_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Dot product (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b).to(tl.float32)

    # Store result (fp32)
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    """

    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm over channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Apply mask if required (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Convert to half precision for the GEMM
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, N, N, hidden_dim

    # Reshape to (B * hidden_dim, N, N) where each hidden slice is a separate GEMM
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # Triton batched GEMM: out = left @ rightᵀ  (over K dimension)
    # ------------------------------------------------------------------
    batch_dim = left_h.shape[0]   # = B * hidden_dim

    # Output buffer (float32)
    out_h = torch.empty((batch_dim, N, N), dtype=torch.float32, device=device)

    # Strides
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)

    stride_b_batch = right_h.stride(0)
    stride_b_k = right_h.stride(2)   # k dimension stride
    stride_b_n = right_h.stride(1)   # n dimension stride (will be used as transposed)

    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    # Tunable block sizes (chosen to fit typical B200 L2 and registers)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Grid dimensions
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m, num_pid_n, batch_dim)

    # Launch kernel
    tri_mul_outgoing_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        batch=batch_dim,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)          # [B, N, N, hidden_dim]
        .contiguous()
    )  # already float32

    # ------------------------------------------------------------------
    # Final LayerNorm, output gating, and projection back to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                # apply outgoing gate
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
```
