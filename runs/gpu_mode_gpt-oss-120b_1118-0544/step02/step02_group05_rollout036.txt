def apply_mask_gate_kernel(a_ptr, b_ptr, mask_ptr, gate_a_ptr, gate_b_ptr, total, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

   carried out with torch.bmm which maps to cuBLAS/Tensor‑Core (fast).
7. Layer‑norm, output‑gate, final linear projection back to the original
   dimension (all float32).

Only the gate‑fusion step is written in Triton; the expensive N³ contraction
uses cuBLAS for peak performance on the B200 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_gate_kernel(
    a_ptr,                # left  tensor (fp16) flattened
    b_ptr,                # right tensor (fp16) flattened
    gate_a_ptr,           # left gate  (fp16) flattened
    gate_b_ptr,           # right gate (fp16) flattened
    total_elements: tl.int64,
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place element‑wise multiplication:
       a = a * gate_a
       b = b * gate_b
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    ga = tl.load(gate_a_ptr + offs, mask=mask, other=1.0)
    gb = tl.load(gate_b_ptr + offs, mask=mask, other=1.0)

    a = a * ga
    b = b * gb

    tl.store(a_ptr + offs, a, mask=mask)
    tl.store(b_ptr + offs, b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None (bool/float)
            - weights: dict with the following keys (all float32 tensors)
                * 'norm.weight', 'norm.bias'
                * 'left_proj.weight', 'right_proj.weight'
                * 'left_gate.weight', 'right_gate.weight', 'out_gate.weight'
                * 'to_out_norm.weight', 'to_out_norm.bias'
                * 'to_out.weight'
            - config: dict with entries 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Projections to hidden_dim (cast to fp16)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]  fp16
    right = F.linear(x_h, right_proj_w)  # fp16

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gates (float32 -> cast to fp16 for the cheap element‑wise step)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # float32

    # Cast left/right gates to half‑precision
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5) Fuse gating with a tiny Triton kernel (in‑place)
    # ------------------------------------------------------------------
    total = left.numel()
    BLOCK_SIZE = 8192  # chosen to give good occupancy
    grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    # flatten tensors for the kernel (they are contiguous after the ops above)
    left_flat = left.view(-1)
    right_flat = right.view(-1)
    left_gate_flat = left_gate_h.view(-1)
    right_gate_flat = right_gate_h.view(-1)

    _apply_gate_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # Release gate tensors we no longer need
    del left_gate, right_gate, left_gate_h, right_gate_h

    # ------------------------------------------------------------------
    # 6) Core N³ contraction – batched GEMM via cuBLAS (torch.bmm)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # (B, H, N, N)   left:   i-k
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # (B, H, N, N)   right:  j-k  -> we need (k, j) for matmul, so transpose last two dims
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)          # (BH, M, K)
    right_mat = right_perm.view(BH, N, N)        # (BH, K, N)

    # Batched matrix multiplication: out = left @ right
    out_mat_fp16 = torch.bmm(left_mat, right_mat)  # fp16 output, fp32 accumulation internally
    out_mat = out_mat_fp16.to(torch.float32)       # cast back to fp32 for the rest

    # Restore original 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # out_gate is float32

    # Final linear projection back to original dimension (float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
