import triton
import triton.language as tl

@triton.jit
def tri_mul_gemm_kernel(left_ptr, right_ptr, out_ptr,
                        B, H, N,
                        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
                        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
                        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # Program ID
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    # offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # ranges
    range_i = tl.arange(0, BLOCK_M)
    range_j = tl.arange(0, BLOCK_N)
    range_k = tl.arange(0, BLOCK_K)

    # Compute masks for i and j
    i_mask = i_start + range_i < N
    j_mask = j_start + range_j < N

    # Base pointers for this batch and head
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K

        k_mask = k_start + range_k < N

        # left tile ptr: (i, k)
        left_ptrs = left_base + (i_start + range_i)[:, None] * stride_l_i + (k_start + range_k)[None, :] * stride_l_k
        # right tile ptr: (k, j) need to load with k as first dimension
        right_ptrs = right_base + (k_start + range_k)[:, None] * stride_r_k + (j_start + range_j)[None, :] * stride_r_i

        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # Accumulate dot product
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_base + (i_start + range_i)[:, None] * stride_o_i + (j_start + range_j)[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])

def custom_kernel(data):
    """
    Triton-accelerated outgoing TriMul (AlphaFold3) forward pass.
    ...
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections (float32)
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply mask if present
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * left_gate * mask_f
        right = right_proj * right_gate * mask_f
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Cast to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Allocate output
    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=input_tensor.device)

    # Triton kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), B * hidden_dim)

    # strides for left, right, out
    stride_l_b, stride_l_i, stride_l_j, stride_l_h = left.stride()
    # left shape B,N,N,H so stride order: (N*N*H, N*H, H, 1)
    # three dims (b,i,j,h) but we treat i as second dim, j as third dim.
    # However for kernel we need stride for i and k, where k corresponds to third dimension originally (j). 
    # We will treat the third dimension as K (the column in left) and second dimension as i.
    # So stride_k = stride_l_j
    stride_l_k = stride_l_j

    stride_r_b, stride_r_i, stride_r_j, stride_r_h = right.stride()
    # For right we need stride_i and stride_k similarly:
    stride_r_k = stride_r_j

    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_hidden.stride()
    # When storing output, we treat i and j as first two dims.

    # Launch kernel
    tri_mul_gemm_kernel[grid](
        left, right, out_hidden,
        B, hidden_dim, N,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Cast output to float32
    out_hidden = out_hidden.to(torch.float32)

    # Final LayerNorm, gate, linear
    out_hidden = F.layer_norm(out_hidden, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) forward pass with a Triton GEMM kernel.
The kernel fuses the N³ reduction (left @ rightᵀ) across all
batch and hidden dimensions.  All layer‑norms, linear projections,
and gating are performed with PyTorch; only the large
matrix‑multiplication is executed on GPU via Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_i = tl.program_id(0)                # tile index in i (rows)
    pid_j = tl.program_id(1)                # tile index in j (cols)
    pid_bh = tl.program_id(2)               # flattened batch*head index

    b = pid_bh // H                         # batch index
    h = pid_bh % H                          # hidden (head) index

    # start coordinates of this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # thread‑local coordinates within the tile
    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)

    # boundary masks
    mask_i = i < N
    mask_j = j < N

    # base pointers for this (b,h)
    left_base  = left_ptr  + b * stride_l_b  + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b  + h * stride_o_h

    # accumulator in fp32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kv in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kv * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left tile: shape (BLOCK_M, BLOCK_K)  -> indices (i,k)
        left = tl.load(
            left_base + i[:, None] * stride_l_i + k[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right tile: shape (BLOCK_K, BLOCK_N) -> indices (k,j)
        right = tl.load(
            right_base + k[:, None] * stride_r_k + j[None, :] * stride_r_i,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # multiply‑accumulate (FP16 inputs, FP32 accumulation)
        acc += tl.float32(tl.dot(left, right))

    # write result (cast back to fp16)
    out = acc.to(tl.float16)
    tl.store(
        out_base + i[:, None] * stride_o_i + j[None, :] * stride_o_j,
        out,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of model weight tensors
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections and gates (all in float32)
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 3) Apply mask and gates (element‑wise)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * left_gate * mask_f
        right = right_proj * right_gate * mask_f
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 4) Cast to half‑precision and make contiguous for the kernel
    # -----------------------------------------------------------------
    left = left.contiguous().to(torch.float16)
    right = right.contiguous().to(torch.float16)

    # -----------------------------------------------------------------
    # 5) Triton GEMM: out_hidden[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # -----------------------------------------------------------------
    out_hidden = torch.empty((B, N, N, hidden), dtype=torch.float16, device=device)

    # Choose tile sizes (ensure BLOCK_M*BLOCK_N <= 1024)
    BLOCK_M, BLOCK_N, BLOCK_K = 32, 32, 32

    # Strides in element units
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out_hidden.stride()

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * hidden,                     # batch * hidden
    )

    tri_mul_gemm_kernel[grid](
        left,
        right,
        out_hidden,
        B,
        hidden,
        N,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_h,
        s_r_b,
        s_r_i,
        s_r_k,
        s_r_h,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to float32 for the remaining layers
    out_hidden = out_hidden.to(torch.float32)

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, out_gate and final linear projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])
    return out
