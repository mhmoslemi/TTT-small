    # Compute k range for this part of the tensor
    k_start = 0
    k_end = seq_len
    
    # Iterate k index
    for k in range(k_start, k_end):
        # ComputeBlock 1
        input_block = input_ptr + offs_batch_m*(seq_len*orig_channels) + pid_i*(orig_channels) + k*(orig_channels)
        mask_block = 1.0
        if mask_ptr is not None:
            mask_block = mask_ptr + offs_batch_m*(seq_len*seq_len) + k*(seq_len) + pid_j
        
        # Load inputs with masking
        input_vals = tl.load(input_block, mask=(k < seq_len))
        
        # Project for left update
        input_left = input_vals[None, :] * left_proj_weight_ptr
        input_right = input_vals[:, None] * right_proj_weight_ptr
        
        # Gate operations for left and right updates
        if left_gate_weight_ptr is not None:
            input_left_g = tl.load(left_gate_weight_ptr)[:, None]
            input_left = input_left.sigmoid() * input_left
        
        if right_gate_weight_ptr is not None:
            input_right_g = tl.load(right_gate_weight_ptr)[None, :]
            input_right = input_right.sigmoid() * input_right
        
        # Dot product and reduce for this specific k
        dot_vals = tl.dot(input_left, input_right)
        
        # Add to accumulation
        acc[pid_j, pid_i] += dot_vals
    
    # Final output after reduction
    out_flat = acc.to(tl.float16)
    
    # Apply output gate
    out_gate = tl.load(out_gate_weight_ptr).sigmoid()
    out_flat = out_flat * out_gate
    
    # Apply output LayerNorm and linear projection
    mean = torch.mean(out_flat, dim=[2])
    variance = torch.mean((out_flat - mean[:, None])**2, dim=[2])
    norm_inv_std = hidden_dim * to_out_norm_weight_ptr
    norm_inv_std = norm_inv_std * torch.rsqrt(variance + 1e-5)
    out_flat = (out_flat - mean[:, None]) * norm_inv_std + to_out_norm_bias_ptr
    
    # Final linear projection with weight scaling
    out_flat = out_flat[None, None, :] * to_out_weight_ptr
    out_flat = out_flat.sum(-1)
    
    # Write out results
    out_flat_ptr = out_ptr + coeffs_batch_m*(BLOCK) + coeffs_batch_n*(seq_len) + pid_i*(seq_len) + pid_j
    tl.store(out_flat_ptr, out_flat)
