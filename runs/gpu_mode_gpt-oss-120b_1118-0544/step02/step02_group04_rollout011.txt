"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM kernel.
The kernel computes, for each batch and hidden channel,
    out[b, i, j] = Σ_k left[b, i, k] * right[b, j, k]
where left/right are the projected and gated tensors.
All heavy O(N³·H) work is performed in half‑precision using a custom
Triton GEMM that accumulates in float32.  Surrounding LayerNorms,
gates and final linear projection are done in PyTorch (float32).
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_fp16_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # B: (batch, N, K)  (K dimension is the fast axis)
    stride_ch, stride_cm, stride_cn,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM:
    C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]   (A,B are fp16, C stored as fp16)
    Accumulation is performed in fp32 for numerical stability.
    """
    pid_b = tl.program_id(axis=2)   # batch * hidden
    pid_m = tl.program_id(axis=0)   # i‑tile
    pid_n = tl.program_id(axis=1)   # j‑tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base offsets for the current batch (includes hidden dimension)
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # pointer to the start of the K‑tile (k = 0)
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                                + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                                + offs_n[None, :] * stride_bn)

    # accumulation in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        # mask for the current k‑tile
        k_offset = k * BLOCK_K
        k_valid = (k_offset + tl.arange(0, BLOCK_K)) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_n[None, :],
                    other=0.0)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result as fp16
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                                + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_fp16(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the above Triton kernel.
    A, B: (batch, N, N) fp16  (K dimension is the last one)
    Returns C: (batch, N, N) fp16
    """
    assert A.ndim == 3 and B.ndim == 3
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Heuristic block sizes (tuned for typical N up to 1024)
    if M <= 64:
        BLOCK_M = 32
        BLOCK_N = 32
    elif M <= 128:
        BLOCK_M = 64
        BLOCK_N = 64
    else:
        BLOCK_M = 128
        BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    _batched_gemm_fp16_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(2), B.stride(1),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- First LayerNorm (float32) ----------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B,N,N,dim] fp32

    # ---------- Projections (half‑precision) ----------
    x_half = x.half()
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x_half, left_proj_w)   # [B,N,N,hidden] fp16
    right = torch.nn.functional.linear(x_half, right_proj_w)  # fp16

    # ---------- Optional masking ----------
    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1] fp32
        mask_h = mask_f.half()                           # fp16
        left = left * mask_h
        right = right * mask_h

    # ---------- Gating (float32) ----------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # cast gates to half for elementwise multiplication
    left_gate_h = left_gate.half()
    right_gate_h = right_gate.half()
    left = left * left_gate_h
    right = right * right_gate_h

    # ---------- Reshape for batched GEMM ----------
    B, N, _, H = left.shape  # H == hidden_dim
    # (B, N, N, H) -> (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    batch_hidden = B * H
    left_flat = left.view(batch_hidden, N, N)   # (B*H, N, N) fp16
    right_flat = right.view(batch_hidden, N, N) # fp16

    # ---------- Core O(N³·H) reduction via Triton ----------
    out_flat = _batched_gemm_fp16(left_flat, right_flat)  # (B*H, N, N) fp16

    # ---------- Reshape back to [B,N,N,hidden] ----------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16
    out = out.to(torch.float32)  # convert back to fp32 for the remaining ops

    # ---------- Second LayerNorm ----------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```
