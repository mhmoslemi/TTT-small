def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_aba, stride_ak, stride_am,  # Actually we need strides: a: (batch, M, K)
    stride_bbk, stride_bjn, stride_bn,  # ranges? Actually for B: (batch, K, N)
    stride_cbc, stride_ccj, stride_cn, 
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute mask for boundary
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) # accumulate in fp32

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        # Load A and B tiles
        a = tl.load(
            a_ptr + batch * stride_aba + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        b = tl.load(
            b_ptr + batch * stride_bbk + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # convert to fp32
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)  # (M,N)
    # write back
    c = acc.to(tl.float16)
    tl.store(
        c_ptr + batch * stride_cbc + offs_m[:, None] * stride_cc + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :]
    )

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy N³ reduction is expressed as a batched half‑precision GEMM:
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
We flatten the (batch, hidden) dimensions and compute all GEMMs with a
single Triton kernel, eliminating the per‑head torch.bmm overhead.
LayerNorms, linear projections and gating are kept in PyTorch.
The final output is in float32.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent batches.
# A: [BATCH, M, K]   (float16)
# B: [BATCH, K, N]   (float16)   (already transposed, i.e. original right matrix)
# C: [BATCH, M, N]   (float16)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    BATCH, M, N, K,                     # problem size
    stride_a_batch, stride_a_row, stride_a_col,   # strides for A
    stride_b_batch, stride_b_row, stride_b_col,   # strides for B
    stride_c_batch, stride_c_row, stride_c_col,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: [BLOCK_M, BLOCK_K]
        a_ptrs = (
            a_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_row
            + offs_k[None, :] * stride_a_col
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B tile: [BLOCK_K, BLOCK_N] (B is already transposed)
        b_ptrs = (
            b_ptr
            + batch_id * stride_b_batch
            + offs_k[:, None] * stride_b_row
            + offs_n[None, :] * stride_b_col
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Cast to FP32, compute dot, accumulate
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # Write back to C
    c_ptrs = (
        c_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    c = acc.to(tl.float16)
    tl.store(
        c_ptrs,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]      (float32)
        - mask         : torch.Tensor of shape [B, N, N] or None (float32/bool)
        - weights      : dict of weight tensors
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections + gates (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,hidden]

    # Apply mask if present (broadcast over hidden dim)
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)  # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Apply gating
    left = left_proj * left_gate   # [B,N,N,hidden]
    right = right_proj * right_gate

    # Cast to half-precision for the heavy reduction
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Batched GEMM via Triton: out = left @ rightᵀ over the k dimension
    #    Shapes after permutation:
    #      left  -> (B, hidden, N, N)
    #      right -> (B, hidden, N, N) then transposed to (B, hidden, N, N) with swapped last axes
    #    Both are flattened to (B*hidden, N, N)
    # ------------------------------------------------------------------
    # Permute to bring hidden before the two sequence dimensions
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, hidden, N, N)

    # Transpose the last two axes of right so that we multiply A @ Bᵀ
    right_T = right_perm.transpose(-1, -2).contiguous()  # (B, hidden, N, N)

    # Flatten batch and hidden dimension
    left_flat = left_perm.view(B * hidden_dim, N, N)      # (B*H, N, N)
    right_flat = right_T.view(B * hidden_dim, N, N)      # (B*H, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements)
    stride_a_batch, stride_a_row, stride_a_col = left_flat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_flat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_flat.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    num_m_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_n_blocks = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * hidden_dim, num_m_blocks, num_n_blocks)

    batched_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        BATCH=B * hidden_dim,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 4) Reshape the result back to [B, N, N, hidden]
    # ------------------------------------------------------------------
    out_hidden = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
```
