    right = F.linear(x, weights["right_proj.weight"])

    # mask
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # half precision for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, N, N, hidden_dim

    # reformat to (B*hidden_dim, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # allocate output
    out_h = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # strides
    stride_a_batch, stride_a_m, stride_a_k = left_h.stride()
    stride_b_batch, stride_b_n, stride_b_k = right_h.stride()  # note we need stride for K and N
    # Actually right_h.stride() returns (stride0, stride1, stride2) which correspond to (batch, N, K). So stride_b_batch = stride0, stride_b_n = stride1, stride_b_k = stride2.
    # We'll use stride_b_k for K and stride_b_n for N.

    stride_c_batch, stride_c_m, stride_c_n = out_h.stride()

    # block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Grid
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )
    # launch
    tri_mul_gemm_kernel[grid](
        left_h,
        right_h,
        out_h,
        B * hidden_dim,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B,N,N,hidden_dim]
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # final layernorm, gating, projection
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) optimized with Triton.

The core operation is:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We treat it as a batched GEMM where each hidden dimension `d` (and batch `b`)
forms an independent matrix multiply (N×N)·(N×N)ᵀ.  The heavy
computation is performed by a Triton kernel that loads tiles in fp16
and accumulates in fp32.  All other lightweight operations (LayerNorm,
linear projections, gating) are done with PyTorch which incurs negligible
overhead compared to the O(N³·hidden_dim) GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch: tl.constexpr,  # B * hidden_dim
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM kernel: C = A @ Bᵀ
    A : (batch, M, K)   fp16
    B : (batch, N, K)   fp16 (interpreted as transposed inside the kernel)
    C : (batch, M, N)   fp32
    """
    pid_b = tl.program_id(2)          # which GEMM in the batch
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile column

    # Compute row/column offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile  (M x K)
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + cur_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load Bᵀ tile (K x N) – we load B in transposed order
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + cur_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # Accumulate the dot product
        acc += tl.dot(a, b)

    # Write back the result
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float); may be ignored if config["nomask"] is True
            - weights: dict containing model parameters:
                * norm.weight / norm.bias
                * left_proj.weight / right_proj.weight
                * left_gate.weight / right_gate.weight / out_gate.weight
                * to_out_norm.weight / to_out_norm.bias
                * to_out.weight
            - config: dict with at least:
                * "dim" (int)
                * "hidden_dim" (int)
                * "nomask" (bool, optional, defaults to False)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul via batched GEMM in Triton
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (Tensor Core friendly)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # note shape is [B, N, N, hidden_dim]

    # Rearrange to (B * hidden_dim, N, N) – each (b,d) pair is an independent GEMM
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Allocate output buffer (fp32 accumulation)
    out_h = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # Strides for the kernel (batch, M, K) and (batch, N, K)
    stride_a_batch, stride_a_m, stride_a_k = left_h.stride()
    stride_b_batch, stride_b_n, stride_b_k = right_h.stride()  # note: (batch, N, K)
    stride_c_batch, stride_c_m, stride_c_n = out_h.stride()

    # Tile sizes – tuned for typical B200 GPUs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    _tri_mul_gemm_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        batch=B * hidden_dim,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6) Final LayerNorm, gating, and projection back to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
