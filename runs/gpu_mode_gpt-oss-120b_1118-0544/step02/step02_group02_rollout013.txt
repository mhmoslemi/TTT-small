The heavy part of the TriMul module is the reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
It can be written as a batched matrix multiplication:
    out_{b,d} = left_{b,d} @ right_{b,d}ᵀ
where left/right_{b,d} ∈ ℝ^{N×N} and the batch dimension is B⋅hidden_dim.

This implementation:
* Performs the cheap operations (LayerNorm, linear projections, gating) in PyTorch.
* Casts the gated tensors to FP16.
* Uses a custom Triton kernel that computes the batched GEMM with
  FP16 inputs and FP32 accumulation, writing the result directly as FP32.
  The kernel avoids the extra copy/convert step of the previous solution.
* Finally applies the output LayerNorm, gating and projection in PyTorch.

The kernel works for any square sequence length N, batch size B∈{1,2},
dim ∈ {128,384} and hidden_dim (c_z) ∈ {128,384,768}.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM where B is treated as transposed (rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_outgoing_kernel(
    A_ptr,                 # left  FP16  [batch, M, K]
    B_ptr,                 # right FP16  [batch, K, N] (actually transposed view of [batch, N, K])
    C_ptr,                 # out   FP32  [batch, M, N]
    M, N, K,
    stride_Abatch, stride_Ai, stride_Ak,
    stride_Bbatch, stride_Bk, stride_Bn,
    stride_Cbatch, stride_Ci, stride_Cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)    # which (b, hidden) pair
    pid_m     = tl.program_id(1)    # tile row
    pid_n     = tl.program_id(2)    # tile col

    # ------------------------------------------------------------------
    # Pointers to the current batch matrix
    # ------------------------------------------------------------------
    a_batch_ptr = A_ptr + batch_idx * stride_Abatch
    b_batch_ptr = B_ptr + batch_idx * stride_Bbatch
    c_batch_ptr = C_ptr + batch_idx * stride_Cbatch

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A : [M, K] – row‑major
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_Ai) + (offs_k[None, :] * stride_Ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B_T : [K, N] – we treat the original right matrix as transposed,
        #                so row stride = N (stride_Bk), col stride = 1 (stride_Bn)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_Bk) + (offs_n[None, :] * stride_Bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_Ci) + (offs_n[None, :] * stride_Cj)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights      : dict of model weights (see skeleton in the prompt)
        - config       : dict containing at least:
            * "dim"         – input channel dimension
            * "hidden_dim" – hidden channel dimension
            * "nomask"      – bool, optional (default False)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype torch.float32
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) Model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)                              # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)       # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5) Apply gates and cast to FP16 – this is the data that the
    #    Triton kernel will consume.
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)   # [B, N, N, hidden_dim]
    right = (right * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 6) Prepare tensors for the batched GEMM:
    #    We want shape (B*hidden, N, N) for both operands.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total_batches = B * hidden_dim

    # bring hidden_dim to the second axis and make the tensors contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat  = left_perm.view(total_batches, N, N)   # [B*H, N, N]  FP16
    right_flat = right_perm.view(total_batches, N, N)  # [B*H, N, N]  FP16

    # ------------------------------------------------------------------
    # 7) Allocate output buffer (FP32)
    # ------------------------------------------------------------------
    out_flat = torch.empty((total_batches, N, N), dtype=torch.float32,
                           device=input_tensor.device)

    # ------------------------------------------------------------------
    # 8) Strides for the kernel (row‑major layout)
    # ------------------------------------------------------------------
    stride_Abatch = N * N
    stride_Ai = N          # row stride in A
    stride_Ak = 1          # col stride in A

    # B is accessed as transposed, hence:
    stride_Bbatch = N * N
    stride_Bk = N          # row stride in B_T (original col stride)
    stride_Bn = 1          # col stride in B_T (original row stride)

    stride_Cbatch = N * N
    stride_Ci = N          # row stride in C
    stride_Cj = 1          # col stride in C

    # ------------------------------------------------------------------
    # 9) Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_gemm_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_Abatch, stride_Ai, stride_Ak,
        stride_Bbatch, stride_Bk, stride_Bn,
        stride_Cbatch, stride_Ci, stride_Cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 10) Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out_hidden = (
        out_flat.view(B, hidden_dim, N, N)          # [B, H, N, N]
        .permute(0, 2, 3, 1)                       # [B, N, N, H]
        .contiguous()
    )                                                # FP32

    # ------------------------------------------------------------------
    # 11) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out_hidden,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)
    out = out * out_gate               # apply output gate
    out = F.linear(out, to_out_weight) # [B, N, N, dim]

    return out
```
