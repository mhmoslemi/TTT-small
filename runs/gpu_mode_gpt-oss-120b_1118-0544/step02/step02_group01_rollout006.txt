Pseudo:

```
@triton.jit
def fused_proj_gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                                B, N, H,
                                stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
                                stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
                                stride_mask_b, stride_mask_i, stride_mask_j,
                                stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                                BLOCK_SIZE: tl.constexpr):
   pid = tl.program_id(0)
   offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
   # compute valid
   total = B * N * N * H
   mask = offset < total
   # decode index
   b = offset // (N * N * H)
   idx = offset % (N * N * H)
   i = idx // (N * H)
   idx = idx % (N * H)
   j = idx // H
   h = idx % H
   # compute linear pointers
   proj_ptrs = proj_ptr + b*stride_proj_b + i*stride_proj_i + j*stride_proj_j + h*stride_proj_h
   gate_ptrs = gate_ptr + b*stride_gate_b + i*stride_gate_i + j*stride_gate_j + h*stride_gate_h
   mask_ptrs = mask_ptr + b*stride_mask_b + i*stride_mask_i + j*stride_mask_j
   out_ptrs = out_ptr + b*stride_out_b + i*stride_out_i + j*stride_out_j + h*stride_out_h
   proj = tl.load(proj_ptrs, mask=mask, other=tl.float16(0.0))
   gate = tl.load(gate_ptrs, mask=mask, other=tl.float16(0.0))
   m = tl.load(mask_ptrs, mask=mask, other=tl.float16(0.0))
   tl.store(out_ptrs, proj * gate * m, mask=mask)
```

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Input LayerNorm (float32).
2. Cast to fp16 and perform all linear projections/gates in half‑precision.
3. Fuse mask, gate and projection with a tiny Triton kernel (only when a mask is
   present).  The kernel computes: out = proj * gate * mask.
4. Perform the core N³ reduction as a batched matrix multiplication:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   This is expressed as a single torch.bmm over (B*H) matrices in fp16,
   fully exploiting Tensor Cores.
5. Convert back to fp32, apply output LayerNorm, out‑gate, and final projection.

The implementation keeps the heavy computation on the GPU, uses fp16 where
possible to reduce memory traffic, and includes at least one Triton kernel as
required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_proj_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Compute out = proj * gate * mask element‑wise.
    Arguments:
        proj_ptr, gate_ptr, mask_ptr, out_ptr: pointers to tensors.
        B, N, H                              : tensor dimensions.
        stride_*                            : strides (in elements) for each tensor.
        BLOCK_SIZE                           : number of elements processed per program.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)  # linear offsets
    total = B * N * N * H
    mask = offs < total

    # decode (b,i,j,h) from linear offset
    tmp = offs
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # compute flat memory offsets
    proj_off = (
        b * stride_proj_b
        + i * stride_proj_i
        + j * stride_proj_j
        + h * stride_proj_h
    )
    gate_off = (
        b * stride_gate_b
        + i * stride_gate_i
        + j * stride_gate_j
        + h * stride_gate_h
    )
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_off = (
        b * stride_out_b
        + i * stride_out_i
        + j * stride_out_j
        + h * stride_out_h
    )

    proj = tl.load(proj_ptr + proj_off, mask=mask, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + gate_off, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_off, mask=mask, other=tl.float16(0.0))

    tl.store(out_ptr + out_off, proj * gate * m, mask=mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32) or None
        - weights      : dict of weight tensors
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Cast to half‑precision and prepare half‑precision weights
    # -----------------------------------------------------------------
    x_h = x.to(torch.float16)

    # Linear weights (half)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Projections and gates in fp16
    # -----------------------------------------------------------------
    left_proj = F.linear(x_h, left_proj_w)          # [B,N,N,hidden]
    right_proj = F.linear(x_h, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))

    # out_gate stays in float32 (no precision loss needed here)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 4) Fuse mask, projection and gate (if mask is present)
    # -----------------------------------------------------------------
    if mask is not None:
        # Convert mask to half and add the trailing hidden dimension for broadcasting
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]

        # Allocate destination tensors
        left = torch.empty_like(left_proj)   # fp16
        right = torch.empty_like(right_proj)

        # Strides (in elements)
        lp_s = left_proj.stride()
        lg_s = left_gate.stride()
        rp_s = right_proj.stride()
        rg_s = right_gate.stride()
        m_s = mask_h.stride()
        lo_s = left.stride()
        ro_s = right.stride()

            left,
            B,

            lp_s[2],
            lp_s[3],

            lo_s[0],

            lo_s[2],
            lo_s[3],
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # right side
        fused_proj_gate_mask_kernel[grid](
            right_proj,
            right_gate,
            mask_h,
            right,
            B,
            N,
            hidden_dim,
            rp_s[0],
            rp_s[1],
            rp_s[2],

            rg_s[2],

            m_s[0],

            m_s[2],
            ro_s[0],
            ro_s[1],
            ro_s[2],
            ro_s[3],
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:

        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # Rearrange tensors to (B, H, N, N) for batched gemm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k ↔ j)

    # Collapse batch and head dimensions
    left_mat = left_perm.view(B * hidden_dim, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N) # (B*H, N, N)

    # Batched half‑precision GEMM (Tensor Core)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N), fp16

    # Reshape back to [B, N, N, H]   (still fp16)
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final projection (float32)
    # -----------------------------------------------------------------
    out_hidden_f32 = out_hidden.to(torch.float32)

    )

    # Apply out‑gate (float32)
    out_hidden_gated = out_hidden_norm * out_gate

    # Final linear projection back to original dimension
    out = F.linear(out_hidden_gated, weights["to_out.weight"])

    return out
