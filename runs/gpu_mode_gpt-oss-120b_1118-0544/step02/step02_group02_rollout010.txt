    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Get weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # Convert to appropriate dtype
    x_norm = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)
    # Cast to fp16 for speed
    x = x_norm.to(torch.float16)

    # Prepare combined weights (fp16)
    combined_left_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0).to(torch.float16)
    combined_right_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0).to(torch.float16)
    out_gate_weight_fp16 = out_gate_weight.to(torch.float16)

    # Linear combined
    left_combined = F.linear(x, combined_left_weight)   # shape [B,N,N,2*hidden_dim]
    right_combined = F.linear(x, combined_right_weight)

    # Split
    left_proj = left_combined[..., :hidden_dim]   # fp16
    left_gate_logits = left_combined[..., hidden_dim:]   # fp16
    right_proj = right_combined[..., :hidden_dim]
    right_gate_logits = right_combined[..., hidden_dim:]

    # Apply sigmoid gating
    left_gate = torch.sigmoid(left_gate_logits)
    right_gate = torch.sigmoid(right_gate_logits)

    left = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Reduction
    # Use torch.einsum (optimized)
    out_h = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # Cast to float32 with Triton cast kernel
    total_elems = out_h.numel()

"""
Outgoing Triangle Multiplicative Update (TriMul) fused implementation.

The kernel performs:
  1. Input LayerNorm.
  2. Fused linear + gating for left/right branches (two fused GEMMs).
  3. Optional pair‑masking.
  4. Core O(N³) reduction   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
     implemented with a single torch.einsum (equivalent to many batched GEMMs).
  5. Cast the FP16 reduction result to FP32 with a tiny Triton kernel.
  6. Output LayerNorm, output‑gate, and final linear projection back to dim.
All heavy arithmetic is done in FP16; only the tiny kernel and the
final LayerNorm/projection use FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def cast_fp16_to_fp32_kernel(
    in_ptr,          # FP16 source pointer
    out_ptr,         # FP32 destination pointer
    N_ELEM: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM
    val = tl.load(in_ptr + offs, mask=mask).to(tl.float32)
    tl.store(out_ptr + offs, val, mask=mask)


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float) – may be None
            - weights: dict with the model parameters
            - config: dict containing at least:
                * "dim" (int)
                * "hidden_dim" (int)
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) Grab model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32) → cast to fp16 for the heavy work
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim], float32
    x_fp16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Fuse projection + gating for left and right branches
    #    (proj and gate share the same input so we can stack the weight matrices)
    # ------------------------------------------------------------------
    # Stack weights for left branch: [proj ; gate]
    left_combined_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0).to(torch.float16)
    right_combined_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0).to(torch.float16)

    # Linear transform (fp16)
    left_combined = F.linear(x_fp16, left_combined_weight)   # [B, N, N, 2*hidden_dim]
    right_combined = F.linear(x_fp16, right_combined_weight)

    # Split projection and gate logits
    left_proj = left_combined[..., :hidden_dim]                 # fp16
    left_gate_logits = left_combined[..., hidden_dim:]          # fp16
    right_proj = right_combined[..., :hidden_dim]
    right_gate_logits = right_combined[..., hidden_dim:]

    # Apply sigmoid gating
    left_gate = torch.sigmoid(left_gate_logits)
    right_gate = torch.sigmoid(right_gate_logits)

    left = left_proj * left_gate      # [B, N, N, hidden_dim], fp16
    right = right_proj * right_gate   # [B, N, N, hidden_dim], fp16

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(x_fp16.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4) Core O(N³) reduction (einsum ≈ batched GEMM)
    # ------------------------------------------------------------------
    # out_h : [B, N, N, hidden_dim]  (still fp16)
    out_h = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # ------------------------------------------------------------------
    # 5) Cast fp16 reduction result to fp32 via a tiny Triton kernel
    # ------------------------------------------------------------------
    total_elems = out_h.numel()
    out_fp32 = torch.empty_like(out_h, dtype=torch.float32)
    BLOCK = 65536  # size of a Triton block; tweakable
    grid = ((total_elems + BLOCK - 1) // BLOCK,)
    cast_fp16_to_fp32_kernel[grid](
        in_ptr=out_h,
        out_ptr=out_fp32,
        N_ELEM=total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6) Output gating
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight.to(torch.float16)))   # [B, N, N, hidden_dim], fp16
    out_gate = out_gate.to(torch.float32)   # bring to same dtype as out_fp32 for broadcasting

    # ------------------------------------------------------------------
    # 7) Final LayerNorm, gating and projection back to original dim
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )                               # [B, N, N, hidden_dim], fp32
    out_norm = out_norm * out_gate  # apply output gate (elementwise)

    # Linear projection back to dim (float32)
    out = F.linear(out_norm, to_out_weight)   # [B, N, N, dim], float32

    return out
