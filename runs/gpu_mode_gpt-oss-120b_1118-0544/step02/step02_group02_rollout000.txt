"""
Outgoing TriMul (Triangle Multiplicative Update) forward pass.

Algorithm
---------
1. Layer‑norm the input over the last dimension (dim).
2. Project the normalized tensor to a hidden dimension with two linear layers
   (left_proj, right_proj).  Both are bias‑free.
3. (Optional) apply a binary mask that is broadcast over the hidden dimension.
4. Compute three sigmoid gates (left_gate, right_gate, out_gate) from the
   normalized input.
5. Multiply left/right with their respective gates and cast to float16.
6. Perform the core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batch of independent matrix multiplications for each hidden
   channel.  It is implemented with `torch.einsum` on float16 tensors, which
   triggers tensor‑core GEMM on the GPU.
7. Cast the float16 result back to float32 with a tiny Triton kernel.
8. Apply a final LayerNorm, the output gate and a linear projection back to
   the original dimension.

Only the cast from FP16→FP32 is done in Triton (to satisfy the requirement of
using Triton).  All other heavy operations stay in PyTorch where they benefit
from highly‑optimized cuBLAS/ATen kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def cast_fp16_to_fp32_kernel(
    inp_ptr,               # *const half*
    out_ptr,               # *float*
    N_ELEM: tl.constexpr,  # total number of elements
    BLOCK: tl.constexpr,   # block size
):
    """Cast a flat FP16 buffer to FP32."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM
    val = tl.load(inp_ptr + offs, mask=mask).to(tl.float32)
    tl.store(out_ptr + offs, val, mask=mask)


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, on GPU)
            - mask:        torch.Tensor [B, N, N] (bool/float) or None
            - weights:     dict of model parameters (see skeleton)
            - config:      dict with keys:
                * "dim" (int)
                * "hidden_dim" (int)
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor [B, N, N, dim] (float32, on GPU)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # extract model parameters
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1) Input LayerNorm over the last dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim], float32

    # --------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1] -> broadcast
        mask_f = mask.to(dtype=x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 5) Apply gates and cast to FP16 for the core reduction
    # --------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # --------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented via torch.einsum on FP16 (tensor cores)
    # --------------------------------------------------------------
    out_fp16 = torch.einsum('bikd,bjkd->bijd', left, right)  # [B, N, N, hidden_dim], fp16

    # --------------------------------------------------------------
    # 7) Cast the FP16 result to FP32 using a tiny Triton kernel
    # --------------------------------------------------------------
    total_elems = out_fp16.numel()
    out_fp32 = torch.empty_like(out_fp16, dtype=torch.float32)

    BLOCK = 8192  # works well for typical tensor sizes
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    cast_fp16_to_fp32_kernel[grid](
        inp_ptr=out_fp16,
        out_ptr=out_fp32,
        N_ELEM=total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 8) Final normalization, gating and projection back to original dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # apply output gate (broadcast)
    out = F.linear(out, to_out_weight)       # [B, N, N, dim]

    return out
```
