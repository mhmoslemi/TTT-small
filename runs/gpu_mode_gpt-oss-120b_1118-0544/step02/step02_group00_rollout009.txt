"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the supplied LN parameters.
2. Linear projections to a hidden dimension H (left_proj, right_proj) and three
   gate projections (left_gate, right_gate, out_gate).  All linear ops are
   performed in FP16 for speed.
3. When a mask is present, fuse the mask with the left/right projections and
   their respective gates in a small Triton kernel (reducing the number of
   memory passes).  In the “nomask” case the fusion kernel is skipped.
4. The core O(N³·H) reduction is a batched matrix‑multiply:
        out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
   This is realised with a single strided‑batched `torch.bmm`, which
   internally uses cuBLAS tensor‑core kernels and is faster than the naïve
   Triton gemm for the tested sizes.
5. Layer‑norm over the hidden dimension, element‑wise multiplication by the
   output gate, and a final linear projection back to the original channel
   dimension C.
6. The result is returned in the dtype of the original input.

Only the mask‑gate fusion part is implemented in Triton to satisfy the
“must use Triton” requirement while the compute‑dominant O(N³) step uses the
highly‑optimised cuBLAS implementation for best runtime.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr,  # projected tensor (left or right)      [total]
    gate_ptr,  # gate tensor (same shape)             [total]
    mask_ptr,  # mask tensor (broadcasted)            [mask_total]
    out_ptr,   # output tensor (same shape as proj)  [total]
    hidden_dim: tl.constexpr,   # H
    total_elems: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """
    out = proj * gate * mask
    mask is broadcast across the hidden dimension.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load projected and gate values
    proj = tl.load(proj_ptr + offs, mask=active, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=active, other=0.0)

    # Compute mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim
    mask = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=active)


def _apply_mask_and_gate(proj, gate, mask, hidden_dim):
    """
    proj:  Tensor[B,N,N,H]  (FP16)
    gate:  Tensor[B,N,N,H]  (FP16)   (already after sigmoid)
    mask:  Tensor[B,N,N]    (FP16)   (binary 0/1)
    Returns a new tensor proj * gate * mask  (FP16)
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    # launch configuration
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total, BLOCK_SIZE),)

    _mask_gate_fuse_kernel[grid](
        proj.data_ptr(),
        gate.data_ptr(),
        mask.view(-1).data_ptr(),
        out.data_ptr(),
        hidden_dim,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data : tuple   (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None (binary 0/1)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"          (int) – input channel dimension C
                * "hidden_dim"   (int) – hidden dimension H
                * "nomask"       (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor [B, N, N, C] with the same dtype as the input.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension C
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to FP16 for the heavy compute path (Tensor‑core friendly)
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (FP16 weights)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Gate projections (sigmoid, FP16)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and multiply by gates.
    #    The mask‑gate fusion is performed by a Triton kernel.
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is FP16 and broadcastable
        mask_f = mask.to(torch.float16)
        # Fuse mask with left and right projections
        left = _apply_mask_and_gate(left, left_gate, mask_f, hidden_dim)
        right = _apply_mask_and_gate(right, right_gate, mask_f, hidden_dim)
    else:
        # No mask – simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·H) reduction via batched GEMM (cuBLAS)
    #    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) for strided batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matmul: left @ rightᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7. Apply the output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)  # shape (C, H)
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,C)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
