@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ai, # pointer strides for A: batch, M, K
    stride_bk, stride_bn, stride_bi, # B: batch, K, N
    stride_cm, stride_cn, stride_ci, # C: batch, M, N
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(axis=2)
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    # Offsets for M,N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers base for this batch
    a_batch_ptr = A_ptr + pid_batch * stride_ai
    b_batch_ptr = B_ptr + pid_batch * stride_bi
    c_batch_ptr = C_ptr + pid_batch * stride_ci

    # Offsets for K within loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: rows M batch, columns K
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile: rows K, cols N
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # accumulate
        # Use dot product: a (BLOCK_M, BLOCK_K); b (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

    # Write back to C
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass.

The heavy reduction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is implemented with a fused Triton kernel that performs a batched
matrix multiplication for each hidden slice (d) in parallel:

    left  : (B, H, N, K)  with K = N
    right : (B, H, N, K)  → transposed to (B, H, K, N)
    out   : (B, H, N, N)

All other operations (LayerNorm, Linear projections, gating,
final LayerNorm, output gate and projection) are carried out
with PyTorch.  Half‑precision is used inside the kernel to
reduce memory traffic while accumulating in float32 for
numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for batched matrix multiplication:
#   C[b, i, j] = Σ_k A[b, i, k] * B[b, k, j]
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ai,   # A: (batch, M, K)
    stride_bk, stride_bn, stride_bi,   # B: (batch, K, N)
    stride_cm, stride_cn, stride_ci,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(axis=2)   # batch * hidden dimension
    pid_m = tl.program_id(axis=0)       # i‑tile
    pid_n = tl.program_id(axis=1)       # j‑tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch slice
    a_batch = pid_batch * stride_ai
    b_batch = pid_batch * stride_bi
    c_batch = pid_batch * stride_ci

    # pointer to the start of the tile (k = 0)
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                                + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                                + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)            # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                                + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matmul using Triton.
    A : (batch, M, K)   float16/float32
    B : (batch, K, N)   float16/float32
    Returns C : (batch, M, N) with the same dtype as A.
    """
    assert A.dim() == 3 and B.dim() == 3
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # simple block‑size heuristics
    BLOCK_M = 64 if M <= 256 else 128
    BLOCK_N = 64 if N <= 256 else 128
    BLOCK_K = 32

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(1), A.stride(2), A.stride(0),   # am, ak, ai
        B.stride(1), B.stride(2), B.stride(0),   # bk, bn, bi
        C.stride(1), C.stride(2), C.stride(0),   # cm, cn, ci
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameter tensors
            - config       : dict containing "dim" and "hidden_dim" (and optional "nomask")

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. First LayerNorm (over channel dimension C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if (mask is not None) and (not nomask):
        mask_f = mask.to(x.dtype).unsqueeze(-1)           # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid on linear transforms of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the O(N³·H) reduction
    #    Convert to (B, H, N, N) → flatten batch*hidden → (B*H, N, N)
    #    Right side is transposed to (K, N) for the matmul.
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    left_flat = left.view(B * H, N, N)          # (B*H, i, k)
    right_flat = right.view(B * H, N, N)        # (B*H, j, k)

    # transpose right so that K (the reduction axis) is the middle dimension
    right_t = right_flat.transpose(1, 2).contiguous()   # (B*H, k, j)

    # ------------------------------------------------------------------
    # 6. Batched matrix multiplication in half precision
    # ------------------------------------------------------------------
    left_half = left_flat.to(torch.float16)
    right_half = right_t.to(torch.float16)

    out_half = _batched_matmul_triton(left_half, right_half)   # (B*H, i, j)
    out = out_half.to(torch.float32)                           # back to fp32

    # ------------------------------------------------------------------
    # 7. Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8. Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 9. Output gate and final linear projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
