"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

The core contraction is
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

which can be seen as a batch of H independent matrix multiplications:
    out_{b,h} = left_{b,h} @ right_{b,h}ᵀ
where left_{b,h}, right_{b,h} ∈ ℝ^{N×N}.

We fuse the N³ reduction into a custom Triton GEMM kernel that
* operates in fp16 for the two inputs,
* accumulates in fp32,
* tiles the M (i) and N (j) axes (BLOCK_M×BLOCK_N) and reduces over K (the
  sequence dimension) in BLOCK_K‑sized steps,
* handles arbitrary N via masks for out‑of‑bounds loads.

All surrounding operations (LayerNorms, linear projections, sigmoids,
masking, final projection) are performed with PyTorch.
The final tensor is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # input / output pointers
    M, N, K,                            # matrix sizes (int32)
    stride_ah, stride_am, stride_ak,   # strides for A (batch/head, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for B (batch/head, k, j)
    stride_ch, stride_cm, stride_cn,   # strides for C (batch/head, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM C = A @ Bᵀ
    A : (BH, M, K)   fp16
    B : (BH, K, N)   fp16   (already transposed on the k‑axis)
    C : (BH, M, N)   fp32 accumulator
    """
    pid_batch = tl.program_id(0)  # which (batch * head)
    pid_m = tl.program_id(1)      # block row
    pid_n = tl.program_id(2)      # block column

    # Compute start offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the start of the current batch/head
    a_batch = a_ptr + pid_batch * stride_ah
    b_batch = b_ptr + pid_batch * stride_bh
    c_batch = c_ptr + pid_batch * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Pointers to the current tiles
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (fp16) with zero padding for OOB elements
        a = tl.load(a_ptrs, mask=a_mask, other=tl.float16(0.0))
        b = tl.load(b_ptrs, mask=b_mask, other=tl.float16(0.0))

        # Matrix‑multiply‑accumulate (fp16·fp16 → fp32)
        acc += tl.dot(a, b)

    # Write back the result (fp32) with proper masking
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float) or None
            - weights: dict of tensors (all float32) with keys:
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config: dict with at least 'dim' and 'hidden_dim' entries

    Returns:
        torch.Tensor: shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2) Projections to hidden_dim (cast to fp16 for the GEMM)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)                     # fp16 version of x
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim] fp16
    right = F.linear(x_h, right_proj_w) # same dtype

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # Convert mask to the same dtype and broadcast over the last dim
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gates (computed in FP32, then cast to fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # FP32

    # Cast gates to fp16 to fuse with left/right before the heavy GEMM
    left_gate_h = left_gate.to(x_h.dtype)
    right_gate_h = right_gate.to(x_h.dtype)

    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 5) Core N³ contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape   # B, seq_len, seq_len, hidden_dim

    # Rearrange to (B, H, i, k) and (B, H, k, j) respectively
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, k, j)

    # Collapse batch and head dimensions → (BH, M, K) and (BH, K, N)
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # (BH, M, K)
    right_mat = right_perm.view(BH, N, N)  # (BH, K, N)

    # Output buffer (float32 accumulator)
    out_mat = torch.empty((BH, N, N), dtype=torch.float32, device=x.device)

    # Tiling parameters – chosen for good occupancy on modern GPUs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                              # M, N, K
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),   # A strides
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),# B strides
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),     # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # elementwise gate (FP32)
    out = F.linear(out, weights["to_out.weight"])  # back to dim, FP32

    return out
```
