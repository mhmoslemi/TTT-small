def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Step 1: layernorm (float32)
    x = torch.nn.functional.layer_norm(input_tensor,
                normalized_shape=(dim,),
                weight=weights["norm.weight"],
                bias=weights["norm.bias"],
                eps=1e-5)

    # Convert to half
    x_h = x.to(torch.float16)

    # Combine weights for left and right
    left_combined_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_combined_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)

    # Compute combined projections
    left_combined = torch.nn.functional.linear(x_h, left_combined_w)   # shape (B,N,N,2*hidden_dim)
    right_combined = torch.nn.functional.linear(x_h, right_combined_w)

    # Split
    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    if not nomask and mask is not None:

        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Compute out_gate
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate_pre = torch.nn.functional.linear(x_h, out_gate_weight)  # shape (B,N,N,hidden_dim)
    out_gate = torch.sigmoid(out_gate_pre)

    # Batched matrix mult
    B, N, _, _ = left.shape
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def cast_fp16_to_fp32_kernel(in_ptr, out_ptr, N_ELEM, BLOCK):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM
    val = tl.load(in_ptr + offs, mask=mask).to(tl.float32)
    tl.store(out_ptr + offs, val, mask=mask)

def custom_kernel(data):
    # same logic...
    # after binary operations produce out_h (half)

    # final out_half shape (B,N,N,dim)
    total = out_half.numel()
    out_fp32 = torch.empty_like(out_half, dtype=torch.float32)
    BLOCK = 65536  # ensure enough threads
    grid = ( (total + BLOCK - 1)//BLOCK, )
    cast_fp16_to_fp32_kernel[grid](
        in_ptr=out_half,
        out_ptr=out_fp32,
        N_ELEM=total,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out_fp32

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def cast_fp16_to_fp32_kernel(in_ptr, out_ptr, N_ELEM, BLOCK):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM
    val = tl.load(in_ptr + offs, mask=mask).to(tl.float32)
    tl.store(out_ptr + offs, val, mask=mask)


def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.
    All heavy linear projections and the N³ reduction are performed in
    half precision (fp16) to exploit Tensor Cores.  A tiny Triton kernel
    is used to cast the final fp16 result back to fp32 before the
    output LayerNorm and projection.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ----------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ----------------------------------------------------
    # 2) Cast to fp16 for the heavy part
    # ----------------------------------------------------
    x_h = x.to(torch.float16)

    device = x_h.device

    # ----------------------------------------------------
    # 3) Fuse projection + gate linear layers (fp16)
    # ----------------------------------------------------
    left_combined_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16).to(device)
    right_combined_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16).to(device)

    left_combined = F.linear(x_h, left_combined_weight)   # [B, N, N, 2*H]
    right_combined = F.linear(x_h, right_combined_weight)

    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    # ----------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # ----------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ----------------------------------------------------
    # 5) Gating (sigmoid, fp16)
    # ----------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate   # [B, N, N, H] fp16
    right = right_proj * right_gate

    # ----------------------------------------------------
    # 6) Output gate (fp16)
    # ----------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(torch.float16).to(device)
    out_gate_pre = F.linear(x_h, out_gate_weight)   # [B, N, N, H] fp16
    out_gate = torch.sigmoid(out_gate_pre)          # fp16

    # ----------------------------------------------------
    # 7) Batched matrix multiplication: out = left @ rightᵀ
    #    (B*H) independent matmuls of size N×N in fp16)
    # ----------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_h = torch.bmm(left_flat, right_flat.transpose(1, 2))  # fp16, shape (B*H, N, N)

    # reshape back to [B, N, N, H]
    out_h = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ----------------------------------------------------
    # 8) Cast to fp32 (tiny Triton kernel)
    # ----------------------------------------------------
    total_elem = out_h.numel()
    out_fp32 = torch.empty_like(out_h, dtype=torch.float32)
    BLOCK = 65536
    grid = ((total_elem + BLOCK - 1) // BLOCK,)
    cast_fp16_to_fp32_kernel[grid](
        in_ptr=out_h,
        out_ptr=out_fp32,
        N_ELEM=total_elem,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # ----------------------------------------------------
    # 9) Final LayerNorm, gating and projection back to dim
    # ----------------------------------------------------
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # cast out_gate to fp32 for multiplication
    out_gate_f32 = out_gate.to(torch.float32)
    out = out * out_gate_f32
    out = F.linear(out, weights["to_out.weight"])   # final fp32 output

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

All large linear projections and the N³ reduction are performed in half‑precision
(fp16) so that Tensor Cores on the B200 can be used.  A tiny Triton kernel
converts the final fp16 tensor back to fp32 before the last LayerNorm and
output projection, satisfying the requirement to contain a custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def cast_fp16_to_fp32_kernel(
    in_ptr,          # fp16 source tensor
    out_ptr,         # fp32 destination tensor
    N_ELEM: tl.constexpr,   # total number of elements
    BLOCK: tl.constexpr,    # block size processed per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM
    val = tl.load(in_ptr + offs, mask=mask).to(tl.float32)
    tl.store(out_ptr + offs, val, mask=mask)


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass using fp16 for the heavy work.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model weights (see skeleton)
            - config: dict with at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ----------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ----------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    x_h = x.to(torch.float16)

    # ----------------------------------------------------

    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
