"""
Optimized forward pass for the outgoing Triangle Multiplicative Update (TriMul).

The kernel fuses the element‑wise mask, projection and gate multiplication:
    out = proj * gate * mask
for both the left and right branches.  The heavy N³ reduction is carried out
as a batched half‑precision matrix multiplication (torch.bmm) after the
gate‑masked tensors have been prepared.  All other operations (LayerNorms,
linear projections, final projection) are performed with PyTorch.
The final output is a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Element‑wise kernel: out = proj * gate * mask
    proj, gate : [B, N, N, H]    (float32)
    mask       : [B, N, N]       (float32, broadcasted over H)
    out        : [B, N, N, H]    (float32)
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    total = B * N * N * H
    valid = offs < total

    # Decode linear index -> (b, i, j, h)
    b = offs // (N * N * H)
    tmp = offs % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # Compute flat offsets for each tensor
    proj_off = (b * stride_proj_b
                + i * stride_proj_i
                + j * stride_proj_j
                + h * stride_proj_h)
    gate_off = (b * stride_gate_b
                + i * stride_gate_i
                + j * stride_gate_j
                + h * stride_gate_h)
    mask_off = (b * stride_mask_b
                + i * stride_mask_i
                + j * stride_mask_j)  # broadcast over H
    out_off = (b * stride_out_b
               + i * stride_out_i
               + j * stride_out_j
               + h * stride_out_h)

    proj = tl.load(proj_ptr + proj_off, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=valid, other=0.0)

    tl.store(out_ptr + out_off, proj * gate * m, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (cuda)
        - mask         : torch.Tensor of shape [B, N, N] (cuda) or None
        - weights      : dict mapping weight names to tensors (cuda)
        - config       : dict with at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, cuda)
    """
    # -----------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # -----------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------
    # 3) Gates (sigmoid)
    # -----------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,hidden]

    # -----------------------------------------------------------
    # 4) Fuse mask and gate (if a mask is supplied) using Triton
    # -----------------------------------------------------------
    if mask is not None:
        # Ensure mask is float32 and on same device
        mask_f = mask.to(x.dtype)

        # Make all tensors contiguous for safe stride handling
        left_proj_c = left_proj.contiguous()
        left_gate_c = left_gate.contiguous()
        right_proj_c = right_proj.contiguous()
        right_gate_c = right_gate.contiguous()
        mask_c = mask_f.contiguous()

        # Allocate output buffers
        left = torch.empty_like(left_proj_c)
        right = torch.empty_like(right_proj_c)

        # Strides (in elements)
        s_lp_b, s_lp_i, s_lp_j, s_lp_h = left_proj_c.stride()
        s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate_c.stride()
        s_rp_b, s_rp_i, s_rp_j, s_rp_h = right_proj_c.stride()
        s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate_c.stride()
        s_m_b, s_m_i, s_m_j = mask_c.stride()
        s_out_b, s_out_i, s_out_j, s_out_h = left.stride()  # same for right

        total_elems = B * N * N * hidden_dim
        BLOCK_SIZE = 1024
        grid = ((total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE,)

        # Left side
        fused_mask_gate_kernel[grid](
            left_proj_c,
            left_gate_c,
            mask_c,
            left,
            B, N, hidden_dim,
            s_lp_b, s_lp_i, s_lp_j, s_lp_h,
            s_lg_b, s_lg_i, s_lg_j, s_lg_h,
            s_m_b, s_m_i, s_m_j,
            s_out_b, s_out_i, s_out_j, s_out_h,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # Right side
        fused_mask_gate_kernel[grid](
            right_proj_c,
            right_gate_c,
            mask_c,
            right,
            B, N, hidden_dim,
            s_rp_b, s_rp_i, s_rp_j, s_rp_h,
            s_rg_b, s_rg_i, s_rg_j, s_rg_h,
            s_m_b, s_m_i, s_m_j,
            s_out_b, s_out_i, s_out_j, s_out_h,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        # No mask – just element‑wise multiplication with gates
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------
    # 5) Core N³ reduction via batched half‑precision matmul
    # -----------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Rearrange to (B, H, N, N) where the last two dims are (i,k) and (j,k)
    left_mat = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_mat = right_fp16.permute(0, 3, 1, 2).contiguous() # [B, H, N, N] (j,k)

    # Collapse batch and head dimensions for torch.bmm
    left_flat = left_mat.view(-1, N, N)                     # (B*H, N, N)
    right_flat = right_mat.view(-1, N, N)                   # (B*H, N, N)

    # Multiply left (i,k) with right transposed over k -> (i,j)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # Restore shape [B, N, N, H]
    out_hidden = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # -----------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # -----------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
```
