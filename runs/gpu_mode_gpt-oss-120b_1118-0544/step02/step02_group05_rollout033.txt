"""
TriMul (outgoing) forward pass – Triton implementation.

The core operation is a batched GEMM:
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which can be expressed as a matrix multiplication per channel:
    left_d (N×K) @ right_dᵀ (K×N) → (N×N)
We fuse this contraction into a single Triton kernel that processes
(B * hidden_dim) independent GEMMs in one launch, using half‑precision
inputs (fp16) and fp32 accumulation. All other elementwise work
(LayerNorm, Linear projections, gating, mask) is performed with PyTorch
for simplicity.

The kernel tiles the M (i) and N (j) axes (BLOCK_M, BLOCK_N) and
reduces over K in BLOCK_K tiles, handling arbitrary sequence lengths
with bounds masks.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dims (int32)
    stride_ah, stride_am, stride_ak,   # strides for A (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for B (batch, k, j)
    stride_ch, stride_cm, stride_cn,   # strides for C (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ
    A : (batch, M, K)   fp16
    B : (batch, K, N)   fp16   (right matrix already transposed)
    C : (batch, M, N)   fp32 accumulator
    """
    pid_batch = tl.program_id(0)   # which batch*head we are processing
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block column

    # Offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for the current batch/head
    a_batch_ptr = a_ptr + pid_batch * stride_ah
    b_batch_ptr = b_ptr + pid_batch * stride_bh
    c_batch_ptr = c_ptr + pid_batch * stride_ch

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Pointers to current tiles
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (fp16) with zero padding for OOB elements
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        # Matrix‑multiply‑accumulate (fp32 accumulation)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Store result
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict with required weight tensors (float32)
            - config: dict with keys 'dim', 'hidden_dim', optionally 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    torch.cuda.empty_cache()
    with torch.no_grad():
        # Unpack
        x, mask, weights, config = data
        dim = config["dim"]
        hidden_dim = config["hidden_dim"]
        nomask = config.get("nomask", True)

        eps = 1e-5

        # ------------------------------------------------------------------
        # 1) Input LayerNorm (FP32)
        # ------------------------------------------------------------------
        x = F.layer_norm(
            x,
            (dim,),
            weight=weights["norm.weight"],
            bias=weights["norm.bias"],
            eps=eps,
        )  # [B, N, N, dim] – FP32

        # ------------------------------------------------------------------
        # 2) Linear projections to hidden_dim (cast to fp16)
        # ------------------------------------------------------------------
        # Cast input to half for the heavy GEMM path
        x_h = x.to(torch.float16)

        left_proj_w = weights["left_proj.weight"].to(torch.float16)
        right_proj_w = weights["right_proj.weight"].to(torch.float16)

        # torch.nn.functional.linear expects weight shape (out_features, in_features)
        left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim] fp16
        right = F.linear(x_h, right_proj_w)  # same dtype

        # ------------------------------------------------------------------
        # 3) Optional mask (broadcast over hidden_dim)
        # ------------------------------------------------------------------
        if (not nomask) and mask is not None:
            # mask shape: [B, N, N] -> [B, N, N, 1]
            mask_h = mask.to(x_h.dtype).unsqueeze(-1)
            left = left * mask_h
            right = right * mask_h

        # ------------------------------------------------------------------
        # 4) Gating (sigmoid) – computed in FP32, then cast to fp16
        # ------------------------------------------------------------------
        left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # FP32
        right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # FP32
        out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # FP32

        # Cast gating to fp16 for element‑wise multiplication
        left_gate_h = left_gate.to(x_h.dtype)
        right_gate_h = right_gate.to(x_h.dtype)

        left = left * left_gate_h
        right = right * right_gate_h

        # ------------------------------------------------------------------
        # 5) Core N³ contraction – batched GEMM via Triton
        # ------------------------------------------------------------------
        B, N_seq, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

        # Rearrange to (B, H, i, k) and (B, H, k, j) and make contiguous
        left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
        right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, k, j)

        # Collapse batch and head dimensions for the kernel
        BH = B * hidden_dim
        left_mat = left_perm.view(BH, N_seq, N_seq)   # (BH, M, K)
        right_mat = right_perm.view(BH, N_seq, N_seq) # (BH, K, N)

        # Allocate output buffer (FP32 accumulator)
        out_mat = torch.empty((BH, N_seq, N_seq), dtype=torch.float32, device=x.device)

        # Choose tile sizes. 128 is a good default for A100/H100‑class GPUs.
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32

        grid = (
            BH,
            (N_seq + BLOCK_M - 1) // BLOCK_M,
            (N_seq + BLOCK_N - 1) // BLOCK_N,
        )

        _tri_mul_matmul_kernel[grid](
            left_mat,
            right_mat,
            out_mat,
            N_seq,               # M
            N_seq,               # N
            N_seq,               # K
            left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
            right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
            out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,   # enough warps for the chosen block size
        )

        # Reshape back to [B, N, N, hidden_dim]
        out = out_mat.view(B, hidden_dim, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

        # ------------------------------------------------------------------
        # 6) Output LayerNorm, output gate, final projection back to dim
        # ------------------------------------------------------------------
        out = F.layer_norm(
            out,
            (hidden_dim,),
            weight=weights["to_out_norm.weight"],
            bias=weights["to_out_norm.bias"],
            eps=eps,
        )  # FP32

        # Apply output gate (element‑wise)
        out = out * out_gate  # both FP32

        # Final linear projection to original dimension
        out = F.linear(out, weights["to_out.weight"])  # (dim, hidden_dim) weight

        return out
