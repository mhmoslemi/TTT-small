@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_abl, stride_ai, stride_ak,  # a batch, row, col strides
    stride_bbl, stride_bk, stride_bn,  # b batch, row(K), col(N) strides
    stride_cbl, stride_ci, stride_cj,  # c batch, row, col strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # Compute offsets for row and col within each block:
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute offset for loading A and B pointers
    # A: shape (batch, M, K)
    a_ptrs = a_ptr + pid_batch * stride_abl + offs_i[:, None] * stride_ai + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    # B: shape (batch, K, N) but we treat as transposed view; we set strides accordingly: stride_bk = stride for K dimension (col index of original) , stride_bn = stride for N dimension (row index)
    # So B element B[batch, k, n]
    b_ptrs = b_ptr + pid_batch * stride_bbl + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_j[None, :] * stride_bn

    # Initialize accumulator  (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_i < M)[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) < K)[:, None] & (offs_j < N)[None, :], other=0.0)
        # we need to cast to float32 for accumulation
        acc += tl.dot(a, b)  # both a,b shape (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        # Increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store result to C (output)
    c_ptrs = c_ptr + pid_batch * stride_cbl + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    # Store as float16 maybe use to_float16 conversion
    tl.store(c_ptrs, acc.to(tl.float16))

"""
custom_kernel: Outgoing Triangular Multiplicative Update (TriMul).

The kernel computes
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
for all batches b and hidden channels d in a single fused Triton kernel.
The heavy O(N³) reduction is performed as a batched half‑precision matrix
multiplication (A @ Bᵀ) with float‑32 accumulation, eliminating the many
tiny kernel launches of torch.bmm and reducing memory traffic.
All other cheap operations (LayerNorm, Linear, gating) are done with
PyTorch.  The final element‑wise out‑gate multiplication is performed in
the forward pass before the last linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                         # matrix shapes (M,N,K are equal to seq_len)
    stride_abl, stride_ai, stride_ak,  # strides for A (batch, row, col)
    stride_bbl, stride_bk, stride_bn,  # strides for B (batch, row(K), col(N)) – B is transposed view
    stride_cbl, stride_ci, stride_cj,  # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision matmul: C = A @ Bᵀ
    A: (batch, M, K)    B: (batch, N, K) (interpreted as (batch, K, N))
    C: (batch, M, N)
    """
    pid_batch = tl.program_id(2)          # batch dimension
    pid_m = tl.program_id(0)              # block row index
    pid_n = tl.program_id(1)              # block col index

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of the current block (before the k‑loop)
    a_ptrs = (
        a_ptr
        + pid_batch * stride_abl
        + offs_m[:, None] * stride_ai
        + offs_k[None, :] * stride_ak
    )
    b_ptrs = (
        b_ptr
        + pid_batch * stride_bbl
        + offs_k[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # load A and B tiles, handling out‑of‑bounds with masks
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        # half‑precision dot, promoted to float32 for accumulation
        acc += tl.dot(a, b).to(tl.float32)

        # advance K‑axis pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back the result
    c_ptrs = (
        c_ptr
        + pid_batch * stride_cbl
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs,
              acc.to(tl.float16),
              mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model weights (tensors on the same device)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Unpack weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm on the channel dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3️⃣ Optional masking (broadcast on hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Prepare half‑precision tensors for the heavy O(N³) reduction
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, _ = left_h.shape  # batch, seq_len, seq_len, hidden_dim

    # Collapse (B, hidden_dim) into a single batch dimension
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (half‑precision)
    out_h = torch.empty_like(left_h)  # shape (B*hidden_dim, N, N), dtype float16

    # -----------------------------------------------------------------
    # 6️⃣ Batched matmul kernel launch (A @ Bᵀ)
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    batch_h = B * hidden_dim
    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M,
          (N + BLOCK_N - 1) // BLOCK_N,
          batch_h ),
    )

    batched_matmul_kernel[grid](
        left_h,
        right_h,
        out_h,
        M=N,
        N=N,
        K=N,
        stride_abl=left_h.stride(0),
        stride_ai=left_h.stride(1),
        stride_ak=left_h.stride(2),
        stride_bbl=right_h.stride(0),
        stride_bk=right_h.stride(2),   # transposed view: k‑stride = original column stride
        stride_bn=right_h.stride(1),   # transposed view: n‑stride = original row stride
        stride_cbl=out_h.stride(0),
        stride_ci=out_h.stride(1),
        stride_cj=out_h.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, hidden_dim] (float32 for the rest)
    # -----------------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)            # [B, N, N, hidden_dim]
        .contiguous()
        .to(torch.float32)
    )

    # -----------------------------------------------------------------
    # 8️⃣ Final LayerNorm (hidden dimension) and out‑gate
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate

    # -----------------------------------------------------------------
    # 9️⃣ Project back to original channel dimension
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]
    return out
