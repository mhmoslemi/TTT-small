@triton.jit
def batch_matmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_left_m, stride_left_i, stride_left_k,
    stride_right_m, stride_right_j, stride_right_k,
    stride_out_m, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_idx = tl.program_id(0)  # m dimension
    pid_i = tl.program_id(1)      # block row
    pid_j = tl.program_id(2)      # block col

    # Compute offsets
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    # Each block will compute C[i0:i0+BLOCK_M, j0:j0+BLOCK_N] for current batch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        
        # Compute offets for left block (A)
        offs_i = i0 + tl.arange(0, BLOCK_M)[:, None]  # shape (BLOCK_M,1)
        offs_k = k0 + tl.arange(0, BLOCK_K)[None, :]  # shape (1, BLOCK_K)
        a_ptrs = left_ptr + batch_idx * stride_left_m + offs_i * stride_left_i + offs_k * stride_left_k
        a = tl.load(a_ptrs, mask=(offs_i < N) & (offs_k < N), other=0.0)  # shape (BLOCK_M,BLOCK_K)
        a = a.to(tl.float32)

        # Right block (need transposed)
        # Load B_T (k,j)
        offs_k2 = k0 + tl.arange(0, BLOCK_K)[:, None]  # (BLOCK_K,1)
        offs_j = j0 + tl.arange(0, BLOCK_N)[None, :]   # (1,BLOCK_N)
        b_ptrs = right_ptr + batch_idx * stride_right_m + offs_k2 * stride_right_k + offs_j * stride_right_j
        b = tl.load(b_ptrs, mask=(offs_k2 < N) & (offs_j < N), other=0.0)  # shape (BLOCK_K,BLOCK_N)
        b = b.to(tl.float32)
        
        acc += tl.dot(a, b)  # shape (BLOCK_M,BLOCK_N)
    
    # Store result
    # Compute out pointer
    offs_i_out = i0 + tl.arange(0, BLOCK_M)[:, None]
    offs_j_out = j0 + tl.arange(0, BLOCK_N)[None, :]
    c_ptrs = out_ptr + batch_idx * stride_out_m + offs_i_out * stride_out_i + offs_j_out * stride_out_j
    tl.store(c_ptrs, acc, mask=(offs_i_out < N) & (offs_j_out < N))

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute start row/col for this block
    i0 = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j0 = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i0 < N
    mask_j = j0 < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_batch * stride_am + i0[:, None] * stride_ai + k_idx[None, :] * stride_ak

        # Load B tile transposed: we want B_T[k, j]
        b_ptrs = b_ptr + pid_batch * stride_bm + k_idx[:, None] * stride_bk + j0[None, :] * stride_bj

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm
    # weight and bias may be on device already
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # Projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # shape [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # Convert to half and reshape for batched matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape
    M = B * hidden_dim
    # permute to (B, H, N, N)
    left_flat = left_h.permute(0, 3, 1, 2).contiguous().view(M, N, N)
    right_flat = right_h.permute(0, 3, 1, 2).contiguous().view(M, N, N)

    # allocate output
    out_flat = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # stride values
    stride_am, stride_ai, stride_ak = left_flat.stride()
    stride_bm, stride_bj, stride_bk = right_flat.stride()
    stride_cm, stride_ci, stride_cj = out_flat.stride()

    # kernel launch
    BLOCK_M = 64
    BLOCK_N = 64

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton‐accelerated version.

The core of the module is the reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
for each batch b and hidden channel d.  This is a batched matrix multiplication
A @ Bᵀ where A = left and B = right.  We fuse this O(N³) operation in a
custom Triton kernel that works on the fused batch‑hidden dimension
M = B * hidden_dim.  The kernel loads tiles of A and B (in fp16), computes the
dot product with fp32 accumulation and writes the result in fp32.
All other operations (layer‑norms, linear projections, gating) are performed
with PyTorch and remain in fp32, while the heavy reduction runs fully on the
GPU with Tensor‑Core acceleration.

The implementation supports optional masking, arbitrary batch size (1‑2) and
sequence lengths up to 1024.  The final output is a float32 tensor of shape
[bs, seq_len, seq_len, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N,                              # M = B*hidden_dim, N = seq_len
    stride_am, stride_ai, stride_ak,   # strides for A (batch, i, k)
    stride_bm, stride_bj, stride_bk,   # strides for B (batch, j, k)
    stride_cm, stride_ci, stride_cj,   # strides for C (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of matrices.
    A, B: [M, N, N] (fp16), C: [M, N, N] (fp32).
    """
    pid_batch = tl.program_id(0)  # which matrix in the batch
    pid_i = tl.program_id(1)      # output block row
    pid_j = tl.program_id(2)      # output block col

    # start indices of the tile
    i0 = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j0 = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i0 < N
    mask_j = j0 < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # ---- load A tile (i, k) ----
        a_ptrs = a_ptr + pid_batch * stride_am \
                 + i0[:, None] * stride_ai \
                 + k_idx[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)   # fp16 → fp32

        # ---- load B tile transposed: we need B_T[k, j] (k, j) ----
        b_ptrs = b_ptr + pid_batch * stride_bm \
                 + k_idx[:, None] * stride_bk \
                 + j0[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # matrix multiplication of the tile
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ---- write the result ----
    c_ptrs = c_ptr + pid_batch * stride_cm \
             + i0[:, None] * stride_ci \
             + j0[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    N³ reduction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) – may be ignored
            - weights: dict containing all model parameters as torch.Tensors
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # -----------------------------------------------------------------
    # 1️⃣  LayerNorm over the channel dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3️⃣  Optional masking (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4️⃣  Gating (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣  Prepare tensors for the Triton batched matmul
    # -----------------------------------------------------------------
    # cast to half to reduce memory bandwidth
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape          # B, N, N, H
    M = B * hidden_dim                  # fused batch‑hidden dimension

    # reshape to (M, N, N) – contiguous layout required by Triton
    left_flat = left_h.permute(0, 3, 1, 2).contiguous().view(M, N, N)
    right_flat = right_h.permute(0, 3, 1, 2).contiguous().view(M, N, N)

    # allocate output buffer (fp32, as required by the rest of the model)
    out_flat = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # strides for the three tensors (batch, first N dim, second N dim)
    stride_am, stride_ai, stride_ak = left_flat.stride()
    stride_bm, stride_bj, stride_bk = right_flat.stride()
    stride_cm, stride_ci, stride_cj = out_flat.stride()

    # -----------------------------------------------------------------
    # 6️⃣  Triton batched matmul: out_flat = left_flat @ right_flatᵀ
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        M,                                 # batch‑hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,      # rows
        (N + BLOCK_N - 1) // BLOCK_N,      # cols
    )
    batched_matmul_outgoing_kernel[grid](
        left_flat, right_flat, out_flat,
        M, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7️⃣  Reshape back to [B, N, N, hidden_dim]
    # -----------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8️⃣  Final LayerNorm over the hidden dimension
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 9️⃣  Apply out‑gate (element‑wise)
    # -----------------------------------------------------------------
    out = out * out_gate

    # -----------------------------------------------------------------
    # 🔟  Project back to the original dimension
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
