"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton‑accelerated version.

The heavy O(N³·C_z) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication with a custom Triton
kernel that processes every hidden slice (batch × hidden_dim) in a single
launch.  All other operations (LayerNorm, Linear projections, gating,
final LayerNorm & projection) are performed with PyTorch.  The kernel
accumulates in FP32 and stores the result in FP16 before converting to
FP32 for the remaining FP32 pipeline.  An additional tiny Triton kernel
applies the final out‑gate element‑wise.

Input:
    data = (input_tensor, mask, weights, config)
Output:
    torch.Tensor of shape [B, N, N, dim] (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: batched matmul for left @ rightᵀ over all hidden slices
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions (M=N, K=N)
    stride_az, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bz, stride_bk, stride_bn,   # strides for B (batch, row, col) – B is transposed view
    stride_cz, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)      # 0‑th program id -> batch × hidden slice
    pid_m    = tl.program_id(1)      # tile over output rows (i)
    pid_n    = tl.program_id(2)      # tile over output cols (j)

    # -------------------- tile offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- K‑loop --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch_id * stride_az \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0).to(tl.float32)

        # B tile (already transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + batch_id * stride_bz \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # -------------------- write result --------------------
    c_ptrs = c_ptr + batch_id * stride_cz \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(mask_m[:, None] & mask_n[None, :]))


# ----------------------------------------------------------------------
# Kernel 2: elementwise multiplication (out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, size: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offset < size
    a = tl.load(a_ptr + offset, mask=mask)
    b = tl.load(b_ptr + offset, mask=mask)
    tl.store(c_ptr + offset, a * b, mask=mask)


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – may be ignored
            - weights: dict of model parameters (torch.Tensors)
            - config: dict with keys:
                * "dim": input channel dimension (C)
                * "hidden_dim": hidden projection dimension (C_z)
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to FP16 for the heavy reduction
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, N, N, hidden_dim

    # ------------------------------------------------------------------
    # 6️⃣ Prepare tensors for batched GEMM:
    #   view as (B*hidden_dim, N, N)
    # ------------------------------------------------------------------
    batch_hidden = B * hidden_dim
    a = left_h.view(batch_hidden, N, N)
    b = right_h.transpose(2, 1).contiguous().view(batch_hidden, N, N)  # transposed view (K,N)

    # output placeholder (FP16)
    c = torch.empty_like(a)

    # ------------------------------------------------------------------
    # 7️⃣ Launch Triton GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batch_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=N,
        N=N,
        K=N,
        stride_az=a.stride(0), stride_am=a.stride(1), stride_ak=a.stride(2),
        stride_bz=b.stride(0), stride_bk=b.stride(1), stride_bn=b.stride(2),
        stride_cz=c.stride(0), stride_cm=c.stride(1), stride_cn=c.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # typical for FP16 GEMM
    )
    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, hidden_dim] and promote to FP32
    # ------------------------------------------------------------------
    out = (
        c.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 9️⃣ Final LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 🔟 Apply out_gate element‑wise using a tiny Triton kernel
    # ------------------------------------------------------------------
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)
    total_el = out_flat.numel()
    BLOCK = 1024
    grid_gate = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid_gate](
        a_ptr=out_flat,
        b_ptr=gate_flat,
        c_ptr=out_flat,
        size=total_el,
        BLOCK=BLOCK,
    )
    # out is now gated in‑place

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Project back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
