def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    P, N, K: tl.constexpr,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    batch = pid // (ceil(N/BLOCK_M)*ceil(N/BLOCK_N))
    block_idx = pid % (ceil(N/BLOCK_M)*ceil(N/BLOCK_N))
    pid_m = block_idx // ceil(N/BLOCK_N)
    pid_n = block_idx % ceil(N/BLOCK_N)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_ptrs = left_ptr + batch*stride_left_batch + offs_m[:, None]*stride_left_m + tl.arange(0, BLOCK_K)[None, :]*stride_left_k
    b_ptrs = right_ptr + batch*stride_right_batch + tl.arange(0, BLOCK_K)[:, None]*stride_right_k + offs_n[None, :] * stride_right_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        a = tl.load(a_ptrs, mask=offs_m < N and k + tl.arange(0, BLOCK_K) < K, other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=k + tl.arange(0, BLOCK_K) < K and offs_n < N, other=0.0).to(tl.float16)
        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_left_k

    # store
    out_ptrs = out_ptr + batch*stride_out_batch + offs_m[:, None]*stride_out_m + offs_n[None, :]*stride_out_n

def batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    P, N,
    stride_left_batch, stride_left_m, stride_left_n,
    stride_right_batch, stride_right_m, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets per tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bound rows/cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, N,
    stride_left_batch, stride_left_m, stride_left_n,
    stride_right_batch, stride_right_m, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # pointers base for this batch
    left_batch_ptr = left_ptr + pid_batch * stride_left_batch
    right_batch_ptr = right_ptr + pid_batch * stride_right_batch
    out_batch_ptr = out_ptr + pid_batch * stride_out_batch

    # iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (M x K)
        a_ptrs = left_batch_ptr + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_n
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (K x N)
        b_ptrs = right_batch_ptr + offs_k[:, None] * stride_right_m + offs_n[None, :] * stride_right_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product: accumulate in fp32
        acc += tl.dot(a, b)

    # store result
    c_ptrs = out_batch_ptr + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm (float32)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=1e-5)

    # Cast to half for projection/gating
    x_h = x.half()

    # Cast weights to half
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    # Linear projections (no bias)
    left = F.linear(x_h, left_proj_w)   # (B,N,N,H) half
    right = F.linear(x_h, right_proj_w)

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    # Apply mask if present
    if not nomask and mask is not None:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_h
        right = right * mask_h

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape
    # reshape for batched GEMM
    # bring hidden dim to batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    P = B * hidden_dim
    left_flat = left_perm.view(P, N, N)
    right_flat = right_perm.view(P, N, N)

    # allocate output in fp32
    out_flat = torch.empty((P, N, N), dtype=torch.float32, device=left.device)

    # strides as 64-bit ints
    stride_left_batch = left_flat.stride(0)
    stride_left_m = left_flat.stride(1)
    stride_left_n = left_flat.stride(2)

    stride_right_batch = right_flat.stride(0)
    stride_right_m = right_flat.stride(1)
    stride_right_n = right_flat.stride(2)

    stride_out_batch = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    # Define block sizes (tuneable)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (P,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_gemm_kernel[grid](
        left_ptr=left_flat,
        right_ptr=right_flat,
        out_ptr=out_flat,
        batch=P,
        N=N,
        stride_left_batch=stride_left_batch,
        stride_left_m=stride_left_m,
        stride_left_n=stride_left_n,
        stride_right_batch=stride_right_batch,
        stride_right_m=stride_right_m,
        stride_right_n=stride_right_n,
        stride_out_batch=stride_out_batch,
        stride_out_m=stride_out_m,
        stride_out_n=stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B,N,N,hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # final LayerNorm over hidden dim
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # apply output gate (convert to float32)
    out_gate_f = out_gate.float()
    out = out * out_gate_f

    # final linear back to dim
    out = F.linear(out, weights["to_out.weight"])
    return out

"""
Outgoing TriMul forward pass (AlphaFold3 style) with a fused Triton GEMM.
The kernel computes the heavy O(N³·H) reduction:
    out[b,i,j,d] = Σ_k (left[b,i,k,d] * right[b,j,k,d])
where left/right are the projected and gated tensors.
All linear projections and gating are done in FP16, the GEMM accumulates in FP32
and writes the result directly in FP32, removing the separate cast kernel from
the previous implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, N,
    stride_left_batch, stride_left_m, stride_left_n,
    stride_right_batch, stride_right_m, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched half‑precision GEMM with fp32 accumulation.

    Computes, for each batch index p:
        out[p, i, j] = Σ_k left[p, i, k] * right[p, k, j]
    where left/right are stored in row‑major [P, N, N] layout.
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this batch
    left_batch_ptr = left_ptr + pid_batch * stride_left_batch
    right_batch_ptr = right_ptr + pid_batch * stride_right_batch
    out_batch_ptr = out_ptr + pid_batch * stride_out_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (M x K)
        a_ptrs = (
            left_batch_ptr
            + offs_m[:, None] * stride_left_m
            + offs_k[None, :] * stride_left_n
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # Load B tile (K x N)
        b_ptrs = (
            right_batch_ptr
            + offs_k[:, None] * stride_right_m
            + offs_n[None, :] * stride_right_n
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # Tensor‑core dot product (fp16 * fp16 -> fp32)
        acc += tl.dot(a, b)

    # Store the result (fp32)
    c_ptrs = (
        out_batch_ptr
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Args:
        data : (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            mask         : torch.Tensor [B, N, N]       (bool/float) – optional
            weights      : dict of model parameters
            config       : dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) LayerNorm on the input (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 1) Cast to half and fetch half‑precision weights
    # ------------------------------------------------------------------
    x_h = x.half()

    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free) in FP16
    # ------------------------------------------------------------------
    left = F.linear(x_h, left_proj_w)   # [B, N, N, H]  fp16
    right = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Gating (sigmoid) in FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B, N, N, H] fp16
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))    # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1] fp16
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape  # B, N, N, H
    # ------------------------------------------------------------------
    # 6) Batched GEMM via Triton (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Bring hidden dim into the batch axis: [B, H, N, N] -> (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    P = B * hidden_dim
    left_flat = left_perm.view(P, N, N)   # fp16
    right_flat = right_perm.view(P, N, N)  # fp16

    # Output buffer (FP32)
    out_flat = torch.empty((P, N, N), dtype=torch.float32, device=left.device)

    # Strides (row‑major)
    stride_left_batch = left_flat.stride(0)
    stride_left_m = left_flat.stride(1)
    stride_left_n = left_flat.stride(2)

    stride_right_batch = right_flat.stride(0)
    stride_right_m = right_flat.stride(1)
    stride_right_n = right_flat.stride(2)

    stride_out_batch = out_flat.stride(0)
    stride_out_m = out_flat.stride(1)
    stride_out_n = out_flat.stride(2)

    # Tile sizes (tuned for A200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        P,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        left_ptr=left_flat,
        right_ptr=right_flat,
        out_ptr=out_flat,
        batch=P,
        N=N,
        stride_left_batch=stride_left_batch,
        stride_left_m=stride_left_m,
        stride_left_n=stride_left_n,
        stride_right_batch=stride_right_batch,
        stride_right_m=stride_right_m,
        stride_right_n=stride_right_n,
        stride_out_batch=stride_out_batch,
        stride_out_m=stride_out_m,
        stride_out_n=stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # ------------------------------------------------------------------
    # 7) Reshape back to [B, N, N, H] (float32)
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)  # [B, H, N, N]
        .permute(0, 2, 3, 1)                # [B, N, N, H]
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 8) Final LayerNorm, output gate and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # out_gate was FP16, cast to FP32 for the element‑wise multiplication
    out = out * out_gate.float()

    # Final linear projection (FP32 weight)
    out = F.linear(out, weights["to_out.weight"])

    return out
