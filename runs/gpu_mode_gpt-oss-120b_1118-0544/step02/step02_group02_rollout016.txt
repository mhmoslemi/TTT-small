```
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak, stride_bk, stride_bn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute batch index
    batch_idx = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    tile_idx = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = tile_idx // tl.cdiv(N, BLOCK_N)
    pid_n = tile_idx % tl.cdiv(N, BLOCK_N)

    # compute offsets for the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute M and N masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator to zero (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # load A block (M,K)
        a = tl.load(a_ptr + batch_idx * batch_stride_a + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        # load B_T block (K,N) from original B (shape [batch, N, K]) using transposed indexing
        b = tl.load(b_ptr + batch_idx * batch_stride_b + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        # compute dot product
        acc += tl.dot(a, b)

    # store the result (fp32)
    tl.store(c_ptr + batch_idx * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
             acc, mask=mask_m[:, None] & mask_n[None, :])
```

"""
Outgoing Triangle Multiplicative Update (TriMul) – forward pass.

The bottleneck of TriMul is the O(N³) reduction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We implement this reduction with a custom Triton batched GEMM kernel that
computes, for every hidden channel d, the matrix product
    out[b,…] = left[b,…] @ right[b,…]^T
using FP16 inputs and FP32 accumulation (tensor‑core friendly).  All
LayerNorms, linear projections and gates are performed in PyTorch; the
kernel only handles the heavy matrix multiplication and writes the result
directly in FP32, eliminating the extra cast kernel used previously.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    batch_stride_a, batch_stride_b, batch_stride_c,   # batch strides
    stride_am, stride_ak,                    # A: (M,K) strides
    stride_bk, stride_bn,                    # B: (N,K) strides (B stored as [N,K])
    stride_cm, stride_cn,                    # C: (M,N) strides
    M, N, K,                                 # matrix sizes (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # Number of blocks in each dimension
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_m * num_n

    # Decode program ID into batch index and tile indices
    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    pid_m = tile_idx // num_n
    pid_n = tile_idx % num_n

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + batch_idx * batch_stride_a
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile transposed (we treat B as [batch, N, K], load B_T[k, j])
        b = tl.load(
            b_ptr
            + batch_idx * batch_stride_b
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate (FP16 inputs, FP32 accumulation)
        acc += tl.dot(a, b)

    # Store the result (FP32)
    c = acc
    c_ptrs = (
        c_ptr
        + batch_idx * batch_stride_c
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass using a fused Triton kernel
    for the heavy matrix multiplication.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – may be None
            - weights: dict of model weights (no bias terms)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) Load model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_raw = F.linear(x, left_proj_weight)      # [B, N, N, hidden_dim]
    right_raw = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left_raw * left_gate * mask_f
        right = right_raw * right_gate * mask_f
    else:
        left = left_raw * left_gate
        right = right_raw * right_gate

    # Release unused tensors early
    del left_raw, right_raw, left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 5) Cast to FP16 for the reduction
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6) Prepare tensors for batched GEMM: shape (B*hidden_dim, N, N)
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, hidden_dim, N, N]

    left_flat = left.view(B * hidden_dim, N, N)
    right_flat = right.view(B * hidden_dim, N, N)

    # Output buffer (FP32)
    out_fp32 = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7) Launch Triton batched GEMM kernel (left @ right.T)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides for the flattened tensors (they are contiguous)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)   # = K
    stride_a_k = left_flat.stride(2)   # = 1

    stride_b_batch = right_flat.stride(0)
    stride_b_n = right_flat.stride(1)  # = K (original N dimension)
    stride_b_k = right_flat.stride(2)  # = 1

    stride_c_batch = out_fp32.stride(0)
    stride_c_m = out_fp32.stride(1)    # = N
    stride_c_n = out_fp32.stride(2)    # = 1

    # Compute launch grid
    num_m_tiles = (N + BLOCK_M - 1) // BLOCK_M
    num_n_tiles = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_batch = num_m_tiles * num_n_tiles
    total_tiles = (B * hidden_dim) * tiles_per_batch
    grid = (total_tiles,)

    batched_matmul_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_fp32,
        batch_stride_a=stride_a_batch,
        batch_stride_b=stride_b_batch,
        batch_stride_c=stride_c_batch,
        stride_am=stride_a_m,
        stride_ak=stride_a_k,
        stride_bk=stride_b_k,
        stride_bn=stride_b_n,
        stride_cm=stride_c_m,
        stride_cn=stride_c_n,
        M=N,
        N=N,
        K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Ensure kernel has finished
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 8) Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Final LayerNorm, output gating and projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                       # apply the output gate
    out = F.linear(out, to_out_weight)         # [B, N, N, dim]

    return out
