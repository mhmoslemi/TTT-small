    a_ptr, b_ptr, c_ptr,

    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)

"""
Outgoing Triangle Multiplicative Update (TriMul) kernel.

The heavy part of TriMul is a batched matrix multiplication:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
for each batch b and hidden dimension d.
We reshape the tensors so each (b,d) slice becomes a separate GEMM:
    (N,N) @ (N,N)^T
which we implement with a custom Triton kernel (fp16 input, fp32 accumulation).
All surrounding operations (LayerNorm, Linear projections, gating, masking)
are performed with PyTorch for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ B^T  (A,B are fp16, accumulate in fp32)."""
    pid_m = tl.program_id(0)  # row tile
    pid_n = tl.program_id(1)  # col tile
    pid_b = tl.program_id(2)  # batch tile (B * hidden_dim)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch pointers
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B^T tile: (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (cur_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)  # fp16 * fp16 -> fp32 accumulation

    # store result
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor,
                       BLOCK_M=128, BLOCK_N=128, BLOCK_K=64) -> torch.Tensor:
    """
    Perform batched GEMM on half‑precision tensors using Triton.
    Input tensors must have shape (B*H, N, N) and be contiguous.
    Returns a tensor of shape (B*H, N, N) in FP16.
    """
    batch, N, _ = left.shape
    assert right.shape == (batch, N, N)

    # Output allocation (FP16, later cast to FP32)
    out = torch.empty((batch, N, N), dtype=torch.float16, device=left.device)

    # Strides in elements
    stride_a_batch, stride_a_m, stride_a_k = left.stride()
    stride_b_batch, stride_b_n, stride_b_k = right.stride()  # note order: (batch, row, col)
    stride_c_batch, stride_c_m, stride_c_n = out.stride()

    # Grid dimensions
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_m, grid_n, batch)

    batched_gemm_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float) – may be None
            - weights: dict with all linear/ln weights (no bias except LayerNorm)
            - config: dict with at least keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- unpack weights -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----- LayerNorm over last dim -----
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ----- Linear projections (no bias) -----
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----- Optional mask (broadcast over hidden dim) -----
    if not nomask:
        if mask.dtype is torch.bool:
            mask_f = mask.to(x.dtype)
        else:
            mask_f = mask
        mask_f = mask_f.unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----- Gates (sigmoid of linear projections) -----
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----- Core TriMul: batched GEMM via Triton -----
    # Cast to half for compute, keep FP32 for output later
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape

    # reshape to (B*hidden_dim, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Triton batched GEMM
    out_h = _batched_gemm_fp16(left_h, right_h,
                               BLOCK_M=128, BLOCK_N=128, BLOCK_K=64)

    # Convert back to FP32 and reshape to [B, N, N, hidden_dim]
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )

    # ----- Final LayerNorm, output gating and projection -----
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
