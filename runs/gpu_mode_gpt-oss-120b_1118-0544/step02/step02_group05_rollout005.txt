"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

The heavy part of the TriMul operator is the N³ contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which can be expressed as a batched matrix multiplication:
    out_{b,d} = left_{b,d} @ right_{b,d}ᵀ
with left,right ∈ ℝ^{N×N} for each (batch,head) pair.
The kernel below performs this batched GEMM in mixed precision
(fp16 inputs, fp32 accumulation) using Tensor‑Core friendly tiles.
All other operations (LayerNorms, linear projections, gates, mask handling)
are done with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers to A, B, C
    M, N, K,                            # matrix dimensions (int32)
    stride_ah, stride_am, stride_ak,    # strides for A (batch/head, M, K)
    stride_bh, stride_bk, stride_bn,    # strides for B (batch/head, K, N)
    stride_ch, stride_cm, stride_cn,    # strides for C (batch/head, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ
    
    A : (BH, M, K)  fp16
    B : (BH, K, N)  fp16   (already transposed along the original right matrix)
    C : (BH, M, N)  fp32 accumulator
    """
    pid_batch = tl.program_id(0)   # which (batch * head) we are processing
    pid_m = tl.program_id(1)       # block-row index
    pid_n = tl.program_id(2)       # block-col index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch/head
    a_batch = a_ptr + pid_batch * stride_ah
    b_batch = b_ptr + pid_batch * stride_bh
    c_batch = c_ptr + pid_batch * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Pointers to the current tiles
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (fp16) with zero for OOB elements
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N)

        # Matrix‑multiply‑accumulate (fp16->fp32)
        acc += tl.dot(a, b)

    # Write the result back (fp32)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = x.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden_dim (cast to fp16)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim], fp16
    right = F.linear(x_h, right_proj_w)  # same dtype

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gates (computed in FP32, then cast to fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # FP32

    left_gate_h = left_gate.to(x_h.dtype)
    right_gate_h = right_gate.to(x_h.dtype)

    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 5) Core N³ contraction – batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Rearrange to (B, hidden_dim, N, N) – convenient for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # Right needs to be transposed on the last two axes (k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) == (B, H, K, N)

    # Collapse batch and head dimensions
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # (BH, M=N, K=N)  fp16
    right_mat = right_perm.view(BH, N, N)  # (BH, K=N, N=N) fp16

    # Output buffer (FP32 accumulator)
    out_mat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # Tunable block sizes – adapt to sequence length
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
    else:
        BLOCK_M = 128
        BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,                                 # batch*head dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # tiles over M (i)
        (N + BLOCK_N - 1) // BLOCK_N,       # tiles over N (j)
    )

    _tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,  # M, N, K (all equal to seq_len)
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply the output gate (both are FP32)
    out = out * out_gate

    # Final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out
