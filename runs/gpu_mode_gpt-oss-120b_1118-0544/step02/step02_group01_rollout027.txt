@triton.jit
def mask_gate_fuse_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                          B, N, H,
                          stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
                          stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
                          stride_mask_b, stride_mask_i, stride_mask_j,
                          stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                          BLOCK_H: tl.constexpr):
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # compute base offsets
    # each thread processes BLOCK_H hidden values
    offs_h = tl.arange(0, BLOCK_H)

    # compute offsets for proj, gate, out
    proj_base = b*stride_proj_b + i*stride_proj_i + j*stride_proj_j
    gate_base = b*stride_gate_b + i*stride_gate_i + j*stride_gate_j
    out_base = b*stride_out_b + i*stride_out_i + j*stride_out_j
    mask_off = b*stride_mask_b + i*stride_mask_i + j*stride_mask_j

    # load mask scalar
    mask = tl.load(mask_ptr + mask_off)  # broadcast
    # load proj and gate vectors
    proj = tl.load(proj_ptr + proj_base + offs_h*stride_proj_h, mask=offs_h < H, other=0.0)
    gate = tl.load(gate_ptr + gate_base + offs_h*stride_gate_h, mask=offs_h < H, other=0.0)

    out = proj * gate * mask

    tl.store(out_ptr + out_base + offs_h*stride_out_h, out, mask=offs_h < H)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # compute input layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # linear projs
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    if mask is not None:
        # Ensure mask dtype matches
        mask_f = mask.to(x.dtype)
        # Ensure contiguous for strides
        left_proj_c = left_proj.contiguous()
        right_proj_c = right_proj.contiguous()
        left_gate_c = left_gate.contiguous()
        right_gate_c = right_gate.contiguous()
        mask_c = mask_f.contiguous()
        # allocate left, right
        left = torch.empty_like(left_proj_c)
        right = torch.empty_like(right_proj_c)

        # get strides in elements (not bytes)
        s_lp_b, s_lp_i, s_lp_j, s_lp_h = left_proj_c.stride()
        s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate_c.stride()
        s_rp_b, s_rp_i, s_rp_j, s_rp_h = right_proj_c.stride()
        s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate_c.stride()
        s_m_b, s_m_i, s_m_j = mask_c.stride()
        s_out_b, s_out_i, s_out_j, s_out_h = left.stride()

        grid = (B, N, N)  # program_id(0) = b, (1)=i, (2)=j

        # left
        mask_gate_fused_kernel[grid](
            left_proj_c,
            left_gate_c,
            mask_c,
            left,
            B, N, hidden_dim,
            s_lp_b, s_lp_i, s_lp_j, s_lp_h,
            s_lg_b, s_lg_i, s_lg_j, s_lg_h,
            s_m_b, s_m_i, s_m_j,
            s_out_b, s_out_i, s_out_j, s_out_h,
            BLOCK_H=32,
        )
        # right
        mask_gate_fused_kernel[grid](
            right_proj_c,
            right_gate_c,
            mask_c,
            right,
            B, N, hidden_dim,
            s_rp_b, s_rp_i, s_rp_j, s_rp_h,
            s_rg_b, s_rg_i, s_rg_j, s_rg_h,
            s_m_b, s_m_i, s_m_j,
            s_out_b, s_out_i, s_out_j, s_out_h,
            BLOCK_H=32,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # convert to half for matmul
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # permute dimensions for matmul
    # left: (B,N,N,hidden) -> (B, hidden, N, N) with i index first
    left = left.permute(0, 3, 1, 2).contiguous()
    # right: (B,N,N,hidden) -> (B, hidden, N, N) with kernel swapped (j,k)
    right = right.permute(0, 3, 2, 1).contiguous()

    # reshape to batch = B*hidden
    left_mat = left.view(-1, N, N)
    right_mat = right.view(-1, N, N)

    # batched matmul
    out_mat = torch.bmm(left_mat, right_mat)   # retains dtype half

    # reshape back
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # output layernorm and final gating/proj
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out_hidden = out_hidden * out_gate
    out = torch.nn.functional.linear(out_hidden, weights["to_out.weight"])
    return out

@triton.jit
def mask_gate_fused_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_H: tl.constexpr
):
    # get coordinates
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # compute base offsets
    proj_offset = b*stride_proj_b + i*stride_proj_i + j*stride_proj_j
    gate_offset = b*stride_gate_b + i*stride_gate_i + j*stride_gate_j
    out_offset = b*stride_out_b + i*stride_out_i + j*stride_out_j
    mask_offset = b*stride_mask_b + i*stride_mask_i + j*stride_mask_j

    # load mask scalar (broadcast)
    mask = tl.load(mask_ptr + mask_offset)

    # hidden offsets
    offs_h = tl.arange(0, BLOCK_H)
    # mask out-of-range
    mask_h = offs_h < H

    proj = tl.load(proj_ptr + proj_offset + offs_h * stride_proj_h,
                   mask=mask_h, other=0.0)
    gate = tl.load(gate_ptr + gate_offset + offs_h * stride_gate_h,
                   mask=mask_h, other=0.0)

    out = proj * gate * mask

    tl.store(out_ptr + out_offset + offs_h * stride_out_h,
             out, mask=mask_h)
```

    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_H: tl.constexpr
):
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    proj_off0 = b * stride_proj_b + i * stride_proj_i + j * stride_proj_j
    gate_off0 = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
    out_off0 = b * stride_out_b + i * stride_out_i + j * stride_out_j
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    mask = tl.load(mask_ptr + mask_off)

    # loop over hidden dim
    for off in range(0, H, BLOCK_H):
        cur_off = off
        # compute actual block size (may be less for last)

                       mask=active, other=0.0)

                       mask=active, other=0.0)

        tl.store(out_ptr + out_off0 + (cur_off + offs) * stride_out_h,
                 out, mask=active)

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

The kernel fuses element‑wise masking and gating:
    out = proj * gate * mask
where `proj` and `gate` are the linear projection and sigmoid gate tensors.
The main N³ reduction is performed with a batched half‑precision GEMM
(`torch.bmm`), which maps each hidden channel to an independent matrix
multiply (size N×N). All other operations (LayerNorms, linear layers,
final projection) are executed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_H: tl.constexpr,
):
    """Fuse projection, gate and optional mask.

    Arguments are raw pointers and strides (in elements). Each program instance
    handles a single (b, i, j) position and loops over the hidden dimension
    H in chunks of BLOCK_H.
    """
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # base offsets for the current (b,i,j) element
    proj_base = b * stride_proj_b + i * stride_proj_i + j * stride_proj_j
    gate_base = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
    out_base = b * stride_out_b + i * stride_out_i + j * stride_out_j
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # broadcasted mask scalar (float32)
    mask_val = tl.load(mask_ptr + mask_off)

    # iterate over hidden dimension
    off = 0
    while off < H:
        cur_off = off
        offs = tl.arange(0, BLOCK_H)
        valid = (cur_off + offs) < H

        # load projection and gate vectors
        proj = tl.load(
            proj_ptr + proj_base + (cur_off + offs) * stride_proj_h,
            mask=valid,
            other=0.0,
        )
        gate = tl.load(
            gate_ptr + gate_base + (cur_off + offs) * stride_gate_h,
            mask=valid,
            other=0.0,
        )

        # fused elementwise multiplication
        out = proj * gate * mask_val

        # write result
        tl.store(
            out_ptr + out_base + (cur_off + offs) * stride_out_h,
            out,
            mask=valid,
        )
        off += BLOCK_H


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict mapping weight names to tensors (float32, CUDA)
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]

    B, N, _, _ = x.shape

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections
    # -----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Gates (sigmoid)
    # -----------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # -----------------------------------------------------------------
    # 4) Fuse mask (if any) and gating via Triton kernel
    # -----------------------------------------------------------------
    if mask is not None:
        # Ensure mask is float32 and contiguous
        mask_f = mask.to(x.dtype).contiguous()
        # Make all tensors contiguous for reliable stride handling
        left_proj_c = left_proj.contiguous()
        right_proj_c = right_proj.contiguous()
        left_gate_c = left_gate.contiguous()
        right_gate_c = right_gate.contiguous()

        # Allocate output buffers
        left = torch.empty_like(left_proj_c)
        right = torch.empty_like(right_proj_c)

        # Strides (in elements)
        s_lp_b, s_lp_i, s_lp_j, s_lp_h = left_proj_c.stride()
        s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate_c.stride()
        s_rp_b, s_rp_i, s_rp_j, s_rp_h = right_proj_c.stride()
        s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate_c.stride()
        s_m_b, s_m_i, s_m_j = mask_f.stride()
        s_out_b, s_out_i, s_out_j, s_out_h = left.stride()

        # Grid: one program per (b,i,j)
        grid = (B, N, N)

        # Left side
        _mask_gate_fuse_kernel[grid](
            left_proj_c,
            left_gate_c,
            mask_f,
            left,
            B,
            N,
            hidden_dim,
            s_lp_b,
            s_lp_i,
            s_lp_j,
            s_lp_h,
            s_lg_b,
            s_lg_i,
            s_lg_j,
            s_lg_h,
            s_m_b,
            s_m_i,
            s_m_j,
            s_out_b,
            s_out_i,
            s_out_j,
            s_out_h,
            BLOCK_H=32,
        )
        # Right side
        _mask_gate_fuse_kernel[grid](
            right_proj_c,
            right_gate_c,
            mask_f,
            right,
            B,
            N,
            hidden_dim,
            s_rp_b,
            s_rp_i,
            s_rp_j,
            s_rp_h,
            s_rg_b,
            s_rg_i,
            s_rg_j,
            s_rg_h,
            s_m_b,
            s_m_i,
            s_m_j,
            s_out_b,
            s_out_i,
            s_out_j,
            s_out_h,
            BLOCK_H=32,
        )
    else:
        # No mask: simple elementwise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5) Batched half‑precision matrix multiplication (core N³ reduction)
    # -----------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape for batched GEMM: (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) -> (B, H, N, N) with k as first axis

    # collapse batch*hidden into a single batch dimension
    left_mat = left.view(-1, N, N)   # (B*H, N, N)
    right_mat = right.view(-1, N, N)  # (B*H, N, N)

    # batched matrix multiply (FP16)
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)

    # restore original layout
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
