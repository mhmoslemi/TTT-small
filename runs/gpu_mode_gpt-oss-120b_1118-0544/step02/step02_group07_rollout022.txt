@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # i tile
    pid_j = tl.program_id(1)  # j tile
    pid_bd = tl.program_id(2)  # combined batch and hidden dim

    # compute offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # derive batch and hidden
    b = pid_bd // D
    d = pid_bd % D

    # masks for i and j
    mask_i = i_offset < N
    mask_j = j_offset < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k_offset = k * BLOCK_K
        k_offset = cur_k_offset + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left tile (M x K)
        # compute pointer offsets
        # left_ptr base + b*stride_l_b + d*stride_l_d
        # plus i_offset * stride_l_i + k_offset * stride_l_k
        # need to broadcast i_offset over K and k_offset over M
        offs_li = i_offset[:, None] * stride_l_i + k_offset[None, :] * stride_l_k
        # add batch and hidden offset
        ptr_l = left_ptr + b * stride_l_b + d * stride_l_d + offs_li
        # load with mask
        left = tl.load(ptr_l, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile (K x N): we need right indexed by (k, j)
        # pointer offset = b*stride_r_b + d*stride_r_d + k_offset[:, None] * stride_r_k + j_offset[None, :] * stride_r_i
        offs_rk = k_offset[:, None] * stride_r_k + j_offset[None, :] * stride_r_i
        ptr_r = right_ptr + b * stride_r_b + d * stride_r_d + offs_rk
        right = tl.load(ptr_r, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product accumulate
        acc += tl.dot(left, right)

    # store result
    # compute output pointers: out has shape (B,N,N,D) with strides (stride_o_b, stride_o_i, stride_o_j, stride_o_d)
    # pointer for each (i,j) and fixed b,d
    offs_o = i_offset[:, None] * stride_o_i + j_offset[None, :] * stride_o_j
    ptr_o = out_ptr + b * stride_o_b + d * stride_o_d + offs_o
    tl.store(ptr_o, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton‑accelerated core.

The heavy O(N³·D) reduction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is implemented as a batched matrix multiplication over the
(sequence) dimensions (i, j) for each hidden channel d using a
custom Triton kernel.  The kernel loads tiles of `left` (M×K) and
`right` (K×N) in fp16, accumulates in fp32, and writes the result
to `out`.  All remaining linear layers, LayerNorms and gating are
performed with PyTorch (mixed‑precision where beneficial).  At
least one Triton kernel is used, satisfying the requirement while
keeping the implementation simple and correct.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute one block of the outgoing triangle multiplicative update:

        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    The block is indexed by (pid_i, pid_j, pid_bd) where
        pid_i → i‑tile   (size BLOCK_M)
        pid_j → j‑tile   (size BLOCK_N)
        pid_bd → combined batch‑hidden index (B·D)

    left/right are fp16, out is fp32.
    """
    pid_i = tl.program_id(0)          # tile over i (rows)
    pid_j = tl.program_id(1)          # tile over j (cols)
    pid_bd = tl.program_id(2)         # batch * hidden index

    # -----------------------------------------------------------------
    # Decode batch and hidden‑channel from pid_bd
    # -----------------------------------------------------------------
    b = pid_bd // D
    d = pid_bd % D

    # -----------------------------------------------------------------
    # Offsets for the i‑ and j‑tiles
    # -----------------------------------------------------------------
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # -----------------------------------------------------------------
    # Accumulator (fp32)
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the reduction dimension k
    # -----------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_tiles):
        k_base = k * BLOCK_K
        k_off = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # ------------------------- left tile (M×K) -------------------
        # left[b, i, k, d]   → stride: (b,i,k,d) = (stride_l_b, stride_l_i,
        #                                              stride_l_k, stride_l_d)
        left_ptrs = left_ptr + \
            b * stride_l_b + d * stride_l_d + \
            i_off[:, None] * stride_l_i + \
            k_off[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # -------------------------- right tile (K×N) -----------------
        # right[b, j, k, d]  → we need (k, j) ordering for a matmul.
        # stride order is the same as left, just swapping i ↔ j.
        right_ptrs = right_ptr + \
            b * stride_r_b + d * stride_r_d + \
            k_off[:, None] * stride_r_k + \
            j_off[None, :] * stride_r_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # ----------------------- accumulate -------------------------
        acc += tl.dot(left, right)   # (M×K)·(K×N) → (M×N)

    # -----------------------------------------------------------------
    # Store the result block
    # -----------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_o_b + d * stride_o_d + \
        i_off[:, None] * stride_o_i + \
        j_off[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, on CUDA)
            - mask: torch.Tensor [B, N, N] (bool or float, on CUDA)
            - weights: dict of model parameters (all float32 CUDA tensors)
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – projection dimension
                * "nomask" (bool, optional) – if True, the mask is ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Load model weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm on the channel dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to half for the large linear projections + gating
    # -----------------------------------------------------------------
    x_h = x.half()  # fp16

    # -----------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) to hidden dimension
    # -----------------------------------------------------------------
    left = F.linear(x_h, left_proj_weight.half())    # [B, N, N, hidden_dim] fp16
    right = F.linear(x_h, right_proj_weight.half())  # [B, N, N, hidden_dim] fp16

    # -----------------------------------------------------------------
    # 4️⃣ Optional masking (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 5️⃣ Gating (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_weight.half()))   # fp16
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_weight.half())) # fp16
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))     # fp16

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Prepare tensors for Triton kernel
    # -----------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, D = left.shape  # D == hidden_dim
    out = torch.empty((B, N, N, D), dtype=torch.float32, device=left.device)

    # Strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # -----------------------------------------------------------------
    # 7️⃣ Triton kernel launch (batched matmul over k)
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j tiles
        B * D,                          # batch * hidden
    )

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, D,
        stride_l_b, stride_l_i, stride_l_k, stride_l_d,
        stride_r_b, stride_r_i, stride_r_k, stride_r_d,
        stride_o_b, stride_o_i, stride_o_j, stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Post‑processing: LayerNorm, out‑gate, final linear
    # -----------------------------------------------------------------
    # Normalize over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # Apply out‑gate (convert to float32 for safety)
    out = out * out_gate.to(out.dtype)

    # Final projection back to original channel dimension
    out = F.linear(out, to_out_weight)   # [B, N, N, dim] float32

    return out
