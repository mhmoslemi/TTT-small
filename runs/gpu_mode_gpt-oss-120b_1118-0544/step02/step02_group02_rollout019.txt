def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_dl,
    stride_br, stride_jr, stride_kr, stride_dr,
    stride_bo, stride_io, stride_jo, stride_do,
    MASK: tl.constexpr,  # not needed maybe
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Compute batch*hid
    pid = tl.program_id(0)  # let axis along total_batch = B*H
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    total_ph = B*H
    batch = pid // H
    hid = pid % H

    # Compute block start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets for left and right for this batch and hidden
    # base pointers for left and right for given batch and hidden
    left_batch_ptr = left_ptr + batch * stride_bl + hid * stride_dl
    right_batch_ptr = right_ptr + batch * stride_br + hid * stride_dr
    out_batch_ptr = out_ptr + batch * stride_bo + hid * stride_do

    # Offsets rhythm
    # Prepare row/col indices
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute mask for out-of-bounds in i, j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k tiles
    for k in range(tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_batch_ptr + offs_i[:, None] * stride_il + offs_k[None, :] * stride_kl
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right_T: shape (BLOCK_K, BLOCK_N)
        # original right indexing: (b, j, k, d). For row=k, col=j
        right_ptrs = right_batch_ptr + offs_k[:, None] * stride_kr + offs_j[None, :] * stride_jr
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(left, right)

    # Store out
    out_ptrs = out_batch_ptr + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_il, stride_kl, stride_dl,
    stride_rb, stride_jr, stride_kr, stride_dr,
    stride_ob, stride_io, stride_jo, stride_do,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch_hidden = tl.program_id(0)  # combined batch and hidden
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    total_ph = B*H
    batch = pid_batch_hidden // H
    hid = pid_batch_hidden % H

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this batch and hidden
    left_base = left_ptr + batch * stride_lb + hid * stride_dl
    right_base = right_ptr + batch * stride_rb + hid * stride_dr
    out_base = out_ptr + batch * stride_ob + hid * stride_do

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile: (M,K)
        left_ptrs = left_base + offs_i[:, None] * stride_il + offs_k[None, :] * stride_kl
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile: (K,N)
        # we treat right as transposed: row = k, col = j
        right_ptrs = right_base + offs_k[:, None] * stride_kr + offs_j[None, :] * stride_jr
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # store results
    out_ptrs = out_base + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_il, stride_kl, stride_dl,
    stride_rb, stride_jr, stride_kr, stride_dr,
    stride_ob, stride_io, stride_jo, stride_do,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_ph = tl.program_id(0)  # batch * hidden
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    total_ph = B * H

    # but schedule grid ensures not.

    hid = pid_ph % H

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # compute offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers
    left_base = left_ptr + batch * stride_lb + hid * stride_dl
    right_base = right_ptr + batch * stride_rb + hid * stride_dr
    out_base = out_ptr + batch * stride_ob + hid * stride_do

        offs_k = k_start + tl.arange(0, BLOCK_K)

"""
TriMul (outgoing) forward pass fused with a custom Triton kernel.

The kernel computes the heavy reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
using a tiled FP16 matrix multiplication (tensor‑cores) that avoids the
large contiguous copies required by a naïve `torch.bmm` implementation.
All other operations (LayerNorm, linear projections, gating, final
LayerNorm and output projection) are performed with PyTorch, keeping the
overall memory traffic low.

Key points:
- left/right are projected, masked, gated and cast to FP16.
- The Triton kernel processes (batch, hidden) as a combined dimension
  and tiles over the i‑j output plane while looping over k tiles.
- Results are accumulated in FP32 and written directly to the output
  tensor, which is then passed through the remaining LayerNorm,
  output gating and linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched triangular multiplication (outgoing)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # shapes
    stride_lb, stride_il, stride_kl, stride_dl,  # left strides
    stride_rb, stride_jr, stride_kr, stride_dr,  # right strides
    stride_ob, stride_io, stride_jo, stride_do,  # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    ph = tl.program_id(0)      # combined batch * hidden
    pid_i = tl.program_id(1)   # tile over i (row)
    pid_j = tl.program_id(2)   # tile over j (col)

    # decompose combined dimension
    batch = ph // H
    hid   = ph % H

    # block start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # index vectors
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + batch * stride_lb + hid * stride_dl
    right_base = right_ptr + batch * stride_rb + hid * stride_dr
    out_base   = out_ptr   + batch * stride_ob + hid * stride_do

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension in tiles
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + offs_i[:, None] * stride_il + offs_k[None, :] * stride_kl
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)          # dtype = fp16 (inherited)

        # right tile (transposed): shape (BLOCK_K, BLOCK_N)
        # we need right[k, j] = right[b, j, k, d]
        right_ptrs = right_base + offs_k[:, None] * stride_kr + offs_j[None, :] * stride_jr
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)          # dtype = fp16

        # FP16 * FP16 -> FP32 accumulation via tensor cores
        acc += tl.dot(left, right)

    # store the FP32 result
    out_ptrs = out_base + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model parameters
            - config: dict containing at least:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    B, N, _, C = input_tensor.shape
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (over C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    # weight shapes: (out_features, in_features) = (H, C)
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]  (float32)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Apply gates and cast to FP16 for the heavy reduction
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 6) Triton kernel: compute out[b,i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    # ------------------------------------------------------------------
    out_fp32 = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # strides (in element counts, not bytes)
    stride_lb, stride_il, stride_kl, stride_dl = left.stride()
    stride_rb, stride_jr, stride_kr, stride_dr = right.stride()
    stride_ob, stride_io, stride_jo, stride_do = out_fp32.stride()

    # tiling sizes – chosen to fit in shared memory & registers
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_fp32,
        B=B,
        N=N,
        H=H,
        stride_lb=int(stride_lb),
        stride_il=int(stride_il),
        stride_kl=int(stride_kl),
        stride_dl=int(stride_dl),
        stride_rb=int(stride_rb),
        stride_jr=int(stride_jr),
        stride_kr=int(stride_kr),
        stride_dr=int(stride_dr),
        stride_ob=int(stride_ob),
        stride_io=int(stride_io),
        stride_jo=int(stride_jo),
        stride_do=int(stride_do),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # 8 warps give good occupancy on B200
    )

    # ------------------------------------------------------------------
    # 7) Final LayerNorm over hidden dimension, gating, and output projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(H,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                       # apply output gate
    out = F.linear(out, to_out_weight)         # [B, N, N, dim]

    return out
