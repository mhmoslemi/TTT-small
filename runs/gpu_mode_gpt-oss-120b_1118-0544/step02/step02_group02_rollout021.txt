@triton.jit
def trilmul_outgoing_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    B,
    N,
    H,
    stride_left_batch,
    stride_left_i,
    stride_left_k,
    stride_left_h,
    stride_right_batch,
    stride_right_k,
    stride_right_i,
    stride_right_h,
    stride_out_batch,
    stride_out_i,
    stride_out_j,
    stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_bh = tl.program_id(2) # batch*hidden index

    # compute b and h
    b = pid_bh // H
    h = pid_bh - b * H

    # compute absolute offsets for rows and cols
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # boundaries
    mask_m = offs_m < N
    mask_n = offs_n < N

    # split M dim into two halves for registers
    offs_m_1 = pid_m * BLOCK_M + tl.arange(0, BLOCK_M // 2)
    offs_m_2 = pid_m * BLOCK_M + tl.arange(BLOCK_M // 2, BLOCK_M)

    # base pointer offsets for this batch+hidden
    left_base = left_ptr + b * stride_left_batch + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base = out_ptr + b * stride_out_batch + h * stride_out_h

    # pointer expressions (initial k=0)
    a_ptrs_1 = left_base + offs_m_1[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
    a_ptrs_2 = left_base + offs_m_2[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
    b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i

    acc_1 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)
    acc_2 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)

    # loop over K blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < N

        # load A blocks with masking
        a_1 = tl.load(a_ptrs_1, mask=(offs_m_1 < N)[:, None] & mask_k[None, :], other=0.0)
        a_2 = tl.load(a_ptrs_2, mask=(offs_m_2 < N)[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & (offs_n < N)[None, :], other=0.0)

        # cast to fp32
        a_1 = a_1.to(tl.float32)
        a_2 = a_2.to(tl.float32)
        b = b.to(tl.float32)

        # accumulate
        acc_1 += tl.dot(a_1, b)
        acc_2 += tl.dot(a_2, b)

        # increment pointers for next k block
        a_ptrs_1 += BLOCK_K * stride_left_k
        a_ptrs_2 += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # combine
    acc = tl.concatenate([acc_1, acc_2], axis=0)  # shape (BLOCK_M, BLOCK_N)

    # apply mask for out-of-bounds
    mask = mask_m[:, None] & mask_n[None, :]
    # store
    c_ptrs = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=mask)

"""
Outgoing TriMul forward pass.

This implementation follows the PyTorch reference but replaces the costly
batched matrix multiplication (the N³ reduction) with a custom Triton kernel.
The kernel computes, for each batch element *b* and hidden channel *h*:

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

All tensors are stored in FP16 for the reduction while accumulation is done in
FP32, avoiding the extra copy and dtype conversion required by the previous
implementation.  LayerNorms, linear projections and gating remain in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trilmul_outgoing_kernel(
    left_ptr,            # FP16 left tensor   [B, N, N, H]
    right_ptr,           # FP16 right tensor  [B, N, N, H]
    out_ptr,             # FP32 output tensor [B, N, N, H]
    B, N, H,                                 # dimensions
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_k, stride_right_i, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Program IDs: (block row, block col, batch*hidden)
    pid_m = tl.program_id(0)  # block row (i)
    pid_n = tl.program_id(1)  # block col (j)
    pid_bh = tl.program_id(2) # flattened index over (batch, hidden)

    # Resolve batch and hidden indices
    b = pid_bh // H
    h = pid_bh - b * H

    # Offsets for the current tile
    offs_m_1 = pid_m * BLOCK_M + tl.arange(0, BLOCK_M // 2)          # first half of rows
    offs_m_2 = pid_m * BLOCK_M + tl.arange(BLOCK_M // 2, BLOCK_M)   # second half of rows
    offs_n   = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)              # columns
    offs_k   = tl.arange(0, BLOCK_K)                                # reduction axis

    # Base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_left_batch  + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_batch   + h * stride_out_h

    # Pointer templates for the K‑loop (k = 0)
    a_ptrs_1 = left_base + offs_m_1[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
    a_ptrs_2 = left_base + offs_m_2[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
    b_ptrs   = right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i

    # Accumulators in FP32
    acc_1 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)
    acc_2 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < N

        # Load A tiles (with masking)
        a_1 = tl.load(a_ptrs_1,
                      mask=(offs_m_1 < N)[:, None] & mask_k[None, :],
                      other=tl.float16(0.0))
        a_2 = tl.load(a_ptrs_2,
                      mask=(offs_m_2 < N)[:, None] & mask_k[None, :],
                      other=tl.float16(0.0))
        # Load B tile
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & (offs_n < N)[None, :],
                    other=tl.float16(0.0))

        # Promote to FP32 for the dot product
        a_1 = a_1.to(tl.float32)
        a_2 = a_2.to(tl.float32)
        b   = b.to(tl.float32)

        # Outer‑product accumulation via matrix‑dot
        acc_1 += tl.dot(a_1, b)   # (BLOCK_M/2, BLOCK_N)
        acc_2 += tl.dot(a_2, b)

        # Advance pointers to the next K‑tile
        a_ptrs_1 += BLOCK_K * stride_left_k
        a_ptrs_2 += BLOCK_K * stride_left_k
        b_ptrs   += BLOCK_K * stride_right_k

    # Store results with bounds checking
    store_mask_1 = (offs_m_1 < N)[:, None] & (offs_n < N)[None, :]
    store_mask_2 = (offs_m_2 < N)[:, None] & (offs_n < N)[None, :]

    c_ptrs_1 = out_base + offs_m_1[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    c_ptrs_2 = out_base + offs_m_2[:, None] * stride_out_i + offs_n[None, :] * stride_out_j

    tl.store(c_ptrs_1, acc_1, mask=store_mask_1)
    tl.store(c_ptrs_2, acc_2, mask=store_mask_2)


def custom_kernel(data):
    """
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C]  (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict with model parameters
            - config: dict containing "dim", "hidden_dim", optionally "nomask"
    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) Model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (over the channel dimension C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]  (float32)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Cast to FP16 for the heavy reduction
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape  # B, N, N, hidden_dim

    # ------------------------------------------------------------------
    # 7) Triton batched matrix multiplication:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float32)  # [B, N, N, H] in FP32

    # Strides (as integer values)
    stride_ld_batch = left.stride(0)
    stride_ld_i = left.stride(1)
    stride_ld_k = left.stride(2)
    stride_ld_h = left.stride(3)

    stride_rd_batch = right.stride(0)
    stride_rd_i = right.stride(1)   # column (j)
    stride_rd_k = right.stride(2)   # row (k)
    stride_rd_h = right.stride(3)

    stride_out_batch = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # Tile sizes – chosen to fit in registers and shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,   # one program per (batch, hidden) pair
    )

    trilmul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_batch=stride_ld_batch,
        stride_left_i=stride_ld_i,
        stride_left_k=stride_ld_k,
        stride_left_h=stride_ld_h,
        stride_right_batch=stride_rd_batch,
        stride_right_k=stride_rd_k,
        stride_right_i=stride_rd_i,
        stride_right_h=stride_rd_h,
        stride_out_batch=stride_out_batch,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 8) Final LayerNorm, output gating and projection back to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # apply output gate
    out = F.linear(out, to_out_weight)       # [B, N, N, dim]

    return out
