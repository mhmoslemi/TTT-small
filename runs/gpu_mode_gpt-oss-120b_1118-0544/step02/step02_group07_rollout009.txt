@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # X dimension for M,N blocks
    batch_pid = tl.program_id(1)  # batch dimension (B*hidden)
    # Determine block coordinates
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    # Compute offsets for each block
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # For k dimension loop:
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        # Load a block
        a_ptrs = a_ptr + batch_pid * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_pid * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        # Mask for out-of-bounds
        a_mask = (offs_m < M) & (offs_k < K)
        b_mask = (offs_k < K) & (offs_n < N)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)
        # Compute dot product
        acc += tl.dot(a, b)
    # Store result
    c_ptrs = c_ptr + batch_pid * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m < M) & (offs_n < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)
```

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    batch_id = tl.program_id(1)

    # Compute block indices
    grid_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // grid_n
    block_n = pid % grid_n

    # Offsets within block
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block
        a_ptrs = a_ptr + batch_id * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B block
        b_ptrs = b_ptr + batch_id * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Store result

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton‐accelerated core reduction.

The kernel computes the batched matrix multiplication
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is the expensive O(N³) step of the TriMul module.
All other operations (LayerNorms, linear projections, gating) are performed
with PyTorch (fp32) while the large tensors fed to the reduction are cast to
fp16.  The Triton kernel processes the batched GEMM in FP16 (with FP32
accumulation) and writes the result back as fp16, after which the surrounding
PyTorch code restores the original ordering, applies the output LayerNorm,
output gate and final projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    batch_stride_a, stride_am, stride_ak,    # A strides
    batch_stride_b, stride_bk, stride_bn,    # B strides (B is already transposed)
    batch_stride_c, stride_cm, stride_cn,    # C strides
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,  # matrix sizes (M = N = K)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B  for a batch of matrices.

    A: (batch, M, K)   – row‑major (stride_am, stride_ak)
    B: (batch, K, N)   – row‑major after transposition (stride_bk, stride_bn)
    C: (batch, M, N)   – row‑major (stride_cm, stride_cn)
    """
    pid = tl.program_id(0)          # combines block M and block N
    batch_id = tl.program_id(1)     # batch axis (B * hidden_dim)

    # 2‑D grid of output blocks
    grid_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // grid_n
    block_n = pid % grid_n

    # Offsets for the current tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A (M x K) block
        a_ptrs = (
            a_ptr
            + batch_id * batch_stride_a
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B (K x N) block (B has been transposed beforehand)
        b_ptrs = (
            b_ptr
            + batch_id * batch_stride_b
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate the product
        acc += tl.dot(a, b)

    # Write the block to C
    c_ptrs = (
        c_ptr
        + batch_id * batch_stride_c
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]   (float32)
            - mask: torch.Tensor [B, N, N]               (bool/float, optional)
            - weights: dict of model weights
            - config: dict with keys "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------- unpack weights --------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -------------------- LayerNorm over channel dim --------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # -------------------- Linear projections (no bias) --------------------
    left = F.linear(x, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # -------------------- Optional masking --------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)       # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -------------------- Gating (sigmoid) --------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -------------------- Prepare for batched GEMM --------------------
    # Cast to fp16 to reduce memory traffic
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # left_h shape: [B, N, N, hidden_dim]

    # Move hidden_dim to batch dimension: (B * hidden_dim, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Transpose right for the multiplication (now shape: batch, K, N)
    right_h_t = right_h.transpose(1, 2).contiguous()

    # Output buffer
    out_h = torch.empty((B * hidden_dim, N, N), dtype=torch.float16, device=input_tensor.device)

    # -------------------- Strides (element based) --------------------
    batch_stride_a = left_h.stride(0)
    stride_am = left_h.stride(1)
    stride_ak = left_h.stride(2)

    batch_stride_b = right_h_t.stride(0)
    stride_bk = right_h_t.stride(1)
    stride_bn = right_h_t.stride(2)

    batch_stride_c = out_h.stride(0)
    stride_cm = out_h.stride(1)
    stride_cn = out_h.stride(2)

    # -------------------- Launch Triton kernel --------------------
    M = N
    K = N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = (M + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_m * grid_n, B * hidden_dim)   # (blocks over M×N, batch)

    batched_matmul_kernel[grid](
        left_h,
        right_h_t,
        out_h,
        batch_stride_a,
        stride_am,
        stride_ak,
        batch_stride_b,
        stride_bk,
        stride_bn,
        batch_stride_c,
        stride_cm,
        stride_cn,
        M,
        N,
        K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # -------------------- Restore original layout --------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden_dim]

    # -------------------- Final LayerNorm over hidden_dim --------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # -------------------- Apply output gate --------------------
    out = out * out_gate

    # -------------------- Linear projection back to original dim --------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
