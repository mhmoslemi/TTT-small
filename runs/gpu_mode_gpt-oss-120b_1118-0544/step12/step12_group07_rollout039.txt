"""
TriMul outgoing (AlphaFold3) forward pass with Triton fusion.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for compute‑heavy parts.
3. Fuse each pair of projection + gate linear layers into a single matmul
   (weight matrices are concatenated along the output dimension).
4. Fuse element‑wise multiplication with optional mask in a Triton kernel:
       left  = left_proj  * left_gate  * mask
       right = right_proj * right_gate * mask
5. Core reduction (TriMul) performed as a batched GEMM:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and apply the final linear projection back to `dim`.
7. Cast the result back to the original dtype.

The kernel in step 4 is required by the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _mask_gate_fusion_kernel(
    left_proj_ptr,   # fp16 *
    left_gate_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse projection, gating and optional mask:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
    `mask` has shape (B,N,N) and is broadcasted over the hidden dim.
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard against OOB
    mask_valid = offs < total_elems

    # index in the mask (one element per (b,i,k))
    mask_idx = offs // hidden_dim

    # load
    lp = tl.load(left_proj_ptr + offs, mask=mask_valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask_valid, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask_valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask_valid, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)   # default 1.0 = no mask

    left_out  = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs,  left_out,  mask=mask_valid)
    tl.store(right_out_ptr + offs, right_out, mask=mask_valid)


def _apply_mask_gate(
    left_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_proj: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 4096,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launches the Triton kernel that fuses projection, gating and mask.
    All tensors must be fp16 and contiguous.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _mask_gate_fusion_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) – optional; may be all‑ones.
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Prepare concatenated weights for projection+gate
    # ------------------------------------------------------------------
    # left: [proj, gate] -> (2*H, dim)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    # right: [proj, gate] -> (2*H, dim)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)
    # out_gate weight (H, dim)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Single matmul for left side (proj + gate)
    # ------------------------------------------------------------------
    left_cat = F.linear(x, left_concat_w)  # (B, N, N, 2*H)
    left_proj = left_cat[..., :hidden_dim]               # (B, N, N, H)
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Single matmul for right side (proj + gate)
    # ------------------------------------------------------------------
    right_cat = F.linear(x, right_concat_w)  # (B, N, N, 2*H)
    right_proj = right_cat[..., :hidden_dim]               # (B, N, N, H)
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Compute out_gate (sigmoid)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Fuse mask, gating and projection
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_mask_gate(
            left_proj, left_gate, right_proj, right_gate, mask_fp16
        )
    else:
        # No mask (or ignored): simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 8) Core reduction (TriMul) – batched GEMM over the hidden dim
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) x (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype:
        out = out.to(dtype)

    return out
