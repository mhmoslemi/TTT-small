"""
TriMul (outgoing) forward pass – Triton‑accelerated core reduction.

Algorithm
---------
1. LayerNorm over the channel dimension.
2. Cast to fp16 for compute‑intensive parts.
3. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) – all bias‑free.
4. Fuse mask, projection and gating with a small Triton kernel
   (optional, depending on the presence of a mask).
5. Core reduction (TriMul) implemented as a batch GEMM:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is equivalent to a batched matrix multiplication
   `C = A @ B^T` where A and B have shape (B*H, N, N).  The GEMM
   is performed by a custom Triton kernel that tiles over the
   sequence dimension.
6. Output LayerNorm, multiply by `out_gate`, and apply the final
   linear projection back to `dim`.
7. Cast the result back to the original dtype.

The implementation mixes PyTorch for the lightweight operations
(LayerNorm, linear layers, sigmoid) and Triton for the two
performance‑critical parts: (i) projection‑gate‑mask fusion and
(ii) the batched matrix multiplication.
"""

    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,

    Fuse projection, gating and mask:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
    Mask has shape [B, N, N] and is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # each hidden entry repeats the same mask element
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(left_out_ptr + offs,  left * lgate * m, mask=mask)
    tl.store(right_out_ptr + offs, right * rgate * m, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch Triton kernel that fuses projection, gating and mask.
    All tensors must be fp16, contiguous and on the same device.
    Returns:
        left, right – tensors of shape (B, N, N, H) in fp16.
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication A @ B^T
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr,          # fp16 *
    b_ptr,          # fp16 *
    c_ptr,          # fp16 *
    total_batches,  # i64 scalar = B*H
    N,              # i64 scalar = seq_len
    stride_ah, stride_ak, stride_ab,   # strides for A
    stride_bh, stride_bk, stride_bn,   # strides for B (original, not transposed)
    stride_ch, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B^T for a batch of matrices.
    A, B, C have shape (total_batches, N, N).
    B is accessed as transposed without materializing the transpose.
    """
    pid = tl.program_id(0)

    # Compute tile indices
    num_tile_m = tl.cdiv(N, BLOCK_M)
    num_tile_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_tile_m * num_tile_n

    batch_id = pid // tiles_per_batch
    tile_id = pid % tiles_per_batch
    tile_m = tile_id // num_tile_n
    tile_n = tile_id % num_tile_n

    # Base pointers for this batch
    a_batch_ptr = a_ptr + batch_id * stride_ah
    b_batch_ptr = b_ptr + batch_id * stride_bh
    c_batch_ptr = c_ptr + batch_id * stride_ch

    # Offsets within the tile
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows/cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        k_offsets = tl.arange(0, BLOCK_K)
        mask_k = (k_base + k_offsets) < N

        # Load A tile (M x K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_ak + (k_base + k_offsets)[None, :] * stride_ab
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile transposed (K x N) -> original B accessed as (col,row)
        # B^T[k, n] = B[n, k]
        b_ptrs = b_batch_ptr + (k_base + k_offsets)[:, None] * stride_bn + offs_n[None, :] * stride_bk
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Load linear weights (bias‑free) and cast to fp16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, dtype=torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (fp16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6) Apply mask and fuse with projection & gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape [B, N, N] -> broadcast over hidden dim
        mask_fp16 = mask.to(device, dtype=torch.float16).contiguous()
        left_proj  = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate  = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7) Core reduction (TriMul) via Triton batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    # strides for the flat tensors (they are contiguous)
    stride_ah = left_flat.stride(0)
    stride_ak = left_flat.stride(1)
    stride_ab = left_flat.stride(2)

    stride_bh = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_ch = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_batches = B * H
    grid = (total_batches * triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),)

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        total_batches,
        N,
        stride_ah, stride_ak, stride_ab,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
