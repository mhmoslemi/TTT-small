"""
TriMul outgoing kernel (AlphaFold3) – forward pass only.

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) – all without bias.
4. Apply mask (if present) by simple element‑wise multiplication.
5. Core reduction (TriMul) computed with a custom Triton batched
   matrix‑multiply kernel:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is performed as a batched GEMM per hidden slice:
       out_h = left_h @ right_hᵀ.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and apply a final linear projection back to the original dimension.
7. Cast the result back to the input dtype.

The heavy reduction (step 5) is executed in Triton; all other
operations use PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_transpose_kernel(
    a_ptr,        # fp16 *
    b_ptr,        # fp16 *
    c_ptr,        # fp16 *
    M, N, K,      # matrix dimensions (int64)
    stride_am, stride_ak,               # A strides
    stride_bk, stride_bn,               # B strides (treated as transposed)
    stride_cm, stride_cn,               # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch of matrices.
    A : (M, K)
    B : (N, K)   (original B is (K, N); we treat it as transposed)
    C : (M, N)
    """
    pid = tl.program_id(0)
    num_m_blocks = tl.cdiv(M, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_n_blocks
    pid_n = pid % num_n_blocks

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        b_mask = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        # Load tiles
        a = tl.load(a_ptr + (offs_m[:, None] * stride_am
                             + offs_k[None, :] * stride_ak),
                    mask=a_mask, other=0.0).to(tl.float16)
        b = tl.load(b_ptr + (offs_k[:, None] * stride_bk
                             + offs_n[None, :] * stride_bn),
                    mask=b_mask, other=0.0).to(tl.float16)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c = acc.to(tl.float16)
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptr + (offs_m[:, None] * stride_cm
                      + offs_n[None, :] * stride_cn),
             c, mask=c_mask)


def _batched_matmul_transpose(
    left: torch.Tensor,
    right: torch.Tensor,
    block_m: int = 128,
    block_n: int = 128,
    block_k: int = 32,
) -> torch.Tensor:
    """
    Perform batched matrix multiplication with transposition on the second
    operand using the Triton kernel defined above.

    Args:
        left:  Tensor of shape (B, N, N, H)  (fp16)
        right: Tensor of shape (B, N, N, H)  (fp16)
        block_*: tiling sizes.

    Returns:
        out: Tensor of shape (B, N, N, H) (fp16)
    """
    B, N, _, H = left.shape
    total = B * H

    # Rearrange to (total, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Extract strides (in elements)
    stride_am = left_flat.stride(0)
    stride_ak = left_flat.stride(2)          # K dimension stride
    # For Bᵀ we swap the last two strides
    stride_bk = right_flat.stride(2)         # original column stride (1)
    stride_bn = right_flat.stride(1)         # original row stride (N)
    stride_cm = out_flat.stride(0)
    stride_cn = out_flat.stride(2)

    M = total
    K = N
    N_dim = N

    grid = (triton.cdiv(M, block_m) * triton.cdiv(N_dim, block_n),)

    _batched_matmul_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M, N_dim, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict with the following keys (all float32):
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Apply mask (if any) and fuse with projection & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape: (B, N, N) -> (B, N, N, 1) for broadcasting
        mask = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul) using Triton batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose(left, right)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
