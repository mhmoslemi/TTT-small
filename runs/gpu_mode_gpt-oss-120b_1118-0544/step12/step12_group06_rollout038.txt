"""
Outgoing Triangle Multiplicative Update (TriMul) – forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to fp16 for compute‑intensive parts.
3. Linear projections (left / right) and gating (left / right / out) using the
   provided weights (bias‑free). Gates use a sigmoid.
4. Apply the optional mask together with the gates (simple PyTorch fusion).
5. Core triangular reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a tiled batched matrix multiplication in a Triton kernel
   (left @ rightᵀ) on tensors of shape (B*H, N, N).
6. Output LayerNorm, element‑wise multiplication by the out‑gate, final
   linear projection back to the original channel dimension.
7. Cast the result to float32 (required output dtype).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 * (B*H, N, N)
    right_ptr,         # fp16 * (B*H, N, N)
    out_ptr,           # fp16 * (B*H, N, N)
    B_H, N,            # int
    stride_l0, stride_l1, stride_l2,
    stride_r0, stride_r1, stride_r2,
    stride_o0, stride_o1, stride_o2,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out = left @ rightᵀ for each batch‑hidden slice.
    left, right, out have shape (B*H, N, N) in fp16.
    The kernel tiles the i‑, j‑ and k‑dimensions.
    """
    pid_bh = tl.program_id(2)          # index over B*H
    pid_i = tl.program_id(0)           # tile over i (row)
    pid_j = tl.program_id(1)           # tile over j (col)

    # start indices of the current tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left[b_h, i, k]
        left = tl.load(
            left_ptr + pid_bh * stride_l0 + i[:, None] * stride_l1 + k[None, :] * stride_l2,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # rightᵀ -> need right[b_h, k, j]
        right = tl.load(
            right_ptr + pid_bh * stride_r0 + k[:, None] * stride_r1 + j[None, :] * stride_r2,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N) fp32

    # write back
    tl.store(
        out_ptr + pid_bh * stride_o0 + i[:, None] * stride_o1 + j[None, :] * stride_o2,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

        - config       : dict with keys:
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask" (optional) – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = inp.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Prepare FP16 weights (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gates (FP16, no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply optional mask together with gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) -> (B, N, N, 1)
        mask_fp16 = mask.to(device).to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core triangular reduction via Triton (batched GEMM)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) for the kernel
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # block sizes – chosen to give good occupancy on typical GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # i dimension
        triton.cdiv(N, BLOCK_N),   # j dimension
        B * H,                     # batch * hidden
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate multiplication and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to float32 (required output dtype)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
