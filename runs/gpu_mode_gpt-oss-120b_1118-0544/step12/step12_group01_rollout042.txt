    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_n = tl.program_id(1)
    pid_z = tl.program_id(2)
    b = pid_z // H
    h = pid_z % H

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        mask_k = offs_k < N

        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + offs_m[:, None]*stride_left_i + offs_k[None, :]*stride_left_k

        right = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

    device = input_tensor.device
    dtype = input_tensor.dtype

        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # 2. Cast to fp16
    x = x.to(torch.float16)

    # 3. Load weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # 4. Projections
    left_proj = F.linear(x, left_proj_w)  # (B,N,N,hidden_dim)

    if mask is not None and not nomask:

        mask = mask.to(torch.float16).unsqueeze(-1)

    assert right.is_contiguous()
    B, N, _, H = left.shape
    # output shape same
    out = torch.empty_like(left)

    # get strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # define block sizes
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    return out

So we can just pass the same strides for right (b, j, k, h) as (stride_b, stride_i, stride_k, stride_h). So we can reuse same stride tuple.

Thus define stride_right_b = stride_left_b, stride_right_j = stride_left_i, stride_right_k = stride_left_k, stride_right_h = stride_left_h.

"""
TriMul outgoing (AlphaFold3) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for compute‑heavy parts.
3. Linear projections (`left_proj`, `right_proj`) and gating projections
   (`left_gate`, `right_gate`, `out_gate`) – all bias‑free.
4. Optional mask is applied to the projected tensors (before gating).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   performed by a custom Triton kernel that executes a blocked
   matrix‑multiply for each batch‑hidden pair.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and apply the final linear projection back to `dim`.
7. Cast the result back to the original dtype.

At least one operation (the N³ reduction) is implemented in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    out_ptr,           # fp16 *
    B, N, H,           # i32 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Layout:
        left  : (B, N, N, H)  indexed as (b, i, k, h)
        right : (B, N, N, H)  indexed as (b, j, k, h)
        out   : (B, N, N, H)  indexed as (b, i, j, h)
    """
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_z = tl.program_id(2)  # combined batch * hidden

    # decode batch and hidden index
    b = pid_z // H
    h = pid_z % H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointer to output tile
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in tiles
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, h]  -> (M, K)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right[b, j, k, h] -> (N, K)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_n[:, None] * stride_right_j
            + offs_k[None, :] * stride_right_k
        )
        right = tl.load(
            right_ptrs,
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(left, right)  # (M, K) • (N, K) -> (M, N)

    # store result as fp16
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _tri_mul_launcher(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Launches the Triton kernel for the core TriMul reduction.

    Args:
        left:  Tensor of shape (B, N, N, H), fp16, contiguous.
        right: Tensor of shape (B, N, N, H), fp16, contiguous.

    Returns:
        out: Tensor of shape (B, N, N, H), fp16.
    """
    assert left.is_contiguous()
    assert right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides for each tensor (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    # right uses the same layout (b, j, k, h)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = (
        right.stride()
    )
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    _tri_mul_kernel[
        grid
    ](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) – optional; may be None.
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast linear weights to fp16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (fp16)
    # --------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5) Optional mask (applied before gating)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # --------------------------------------------------------------
    # 6) Gating projections (sigmoid, fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 7) Apply gating
    # --------------------------------------------------------------
    left = left_proj * left_gate    # (B, N, N, hidden_dim)
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 8) Core reduction (TriMul) – implemented in Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = _tri_mul_launcher(left, right)  # (B, N, N, hidden_dim), fp16

    # --------------------------------------------------------------
    # 9) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 10) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype:
        out = out.to(dtype)

    return out
