"""
TriMul outgoing kernel (AlphaFold3) – forward pass only.
Implemented with a fused Triton GEMM kernel for the core
`einsum('... i k d, ... j k d -> ... i j d')` operation.
The rest of the pipeline (LayerNorms, linear projections, gating,
and optional mask) is performed with PyTorch for simplicity.

Algorithm
---------
1. Layer‑norm over the channel dimension (`dim`).
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections (left/right) and gating projections
   (left_gate, right_gate, out_gate) – all bias‑free.
4. Apply mask (if present) and fuse with gating (element‑wise).
5. Core reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched matrix multiplication per hidden slice
   using a custom Triton kernel that computes
       C = A @ Bᵀ   with A,B ∈ ℝ^{(B·D)×N×N}.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
7. Cast back to the original input dtype (usually float32).

The kernel uses 64×64×64 tiling and accumulates in FP32 for
numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _batched_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions (all = seq_len)
    stride_ab, stride_am, stride_ak,   # strides for A: (batch, row, col)
    stride_bb, stride_bk, stride_bn,   # strides for B (will be accessed as Bᵀ)
    stride_cb, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for each batch element.
    A, B, C have shape (batch, M, K), (batch, N, K), (batch, M, N) respectively.
    All tensors are in row‑major order and stored as FP16; accumulation is in FP32.
    """
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start offsets of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of this batch
    a_batch_ptr = a_ptr + batch * stride_ab
    b_batch_ptr = b_ptr + batch * stride_bb
    c_batch_ptr = c_ptr + batch * stride_cb

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Masks for the current K‑block
        mask_k = offs_k < K

        # Load A block [M, K]
        a_ptrs = a_batch_ptr + \
                 (offs_m[:, None] * stride_am) + \
                 (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load Bᵀ block [K, N]  (B is accessed as transposed)
        b_ptrs = b_batch_ptr + \
                 (offs_k[:, None] * stride_bk) + \
                 (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = c_batch_ptr + \
             (offs_m[:, None] * stride_cm) + \
             (offs_n[None, :] * stride_cn)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict with the following keys (all float32):
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gating (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Apply mask and fuse with projection & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Convert mask to FP16 (0.0 or 1.0) and broadcast
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction (TriMul) via Triton batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    = (left_d) @ (right_d)ᵀ   for each hidden slice d
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Rearrange to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output placeholder
    out_flat = torch.empty_like(left_flat)

    # Block sizes (tuned for typical B200/Turing GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_batches = left_flat.shape[0]
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    # Strides (in elements)
    stride_ab, stride_am, stride_ak = left_flat.stride()
    stride_bb, stride_bk, stride_bn = right_flat.stride()
    stride_cb, stride_cm, stride_cn = out_flat.stride()

    grid = (num_batches, num_blocks_m, num_blocks_n)
    _batched_matmul_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
