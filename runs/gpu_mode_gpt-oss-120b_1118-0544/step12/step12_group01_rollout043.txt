def trmul_kernel(a_ptr, b_ptr, c_ptr,
                 BATCH, M, N, K,
                 stride_a_batch, stride_a_row, stride_a_col,
                 stride_b_batch, stride_b_row, stride_b_col,
                 stride_c_batch, stride_c_row, stride_c_col,
                 BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    start_m = pid_m * BLOCK_M
    start_n = pid_n * BLOCK_N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # offsets for A block
        a_offset_row = start_m + tl.arange(0, BLOCK_M)
        a_offset_col = k * BLOCK_K + tl.arange(0, BLOCK_K)

        a_ptrs = a_ptr + pid_batch * stride_a_batch + a_offset_row[:, None] * stride_a_row + a_offset_col[None, :] * stride_a_col

        # offsets for B_T block (load B as transposed)
        b_offset_row = k * BLOCK_K + tl.arange(0, BLOCK_K)
        b_offset_col = start_n + tl.arange(0, BLOCK_N)

        # For B^T, row stride = stride_b_col (since we want k dimension as row), col stride = stride_b_row
        b_ptrs = b_ptr + pid_batch * stride_b_batch + b_offset_row[:, None] * stride_b_row + b_offset_col[None, :] * stride_b_col

@triton.jit
def trmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_aa, stride_ab, stride_ac,
    stride_bb, stride_bc, stride_bd,
    stride_cc, stride_cd, stride_ce,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A
        a_ptrs = a_ptr + pid_batch * stride_aa + offs_m[:, None] * stride_ab + offs_k[None, :] * stride_ac
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)

        # Load B^T (transpose)
        # B is stored as (row, col). We need B_T block (k, n)
        b_ptrs = b_ptr + pid_batch * stride_bb + offs_k[:, None] * stride_bc + offs_n[None, :] * stride_bd
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        # Dot product
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + pid_batch * stride_cc + offs_m[:, None] * stride_cd + offs_n[None, :] * stride_ce
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

"""
TriMul outgoing (AlphaFold3) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for compute‑heavy parts.
3. Linear projections (`left_proj`, `right_proj`) and gating projections
   (`left_gate`, `right_gate`, `out_gate`) – all bias‑free.
4. Optional mask is applied (broadcast over the hidden dimension).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched matrix‑multiply (A @ Bᵀ) using a custom
   Triton kernel.  The hidden dimension is folded into the batch
   dimension (B·H) giving a large number of independent GEMM’s that
   fully exploit Tensor Cores.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and apply the final linear projection back to `dim`.
7. Cast the result back to the original dtype.

Only the reduction (step 5) is performed in Triton; all other
operations use highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM computing C = A @ Bᵀ
# A, B, C are (batch, M, K) where M = N, K = N.
# ----------------------------------------------------------------------
@triton.jit
def _trmul_kernel(
    a_ptr,                     # fp16 *
    b_ptr,                     # fp16 *
    c_ptr,                     # fp16 *
    BATCH: tl.constexpr,       # total batches = B * hidden_dim
    M: tl.constexpr,           # sequence length
    N: tl.constexpr,           # sequence length (same as M)
    K: tl.constexpr,           # sequence length (same as M)
    stride_aa, stride_ab, stride_ac,   # strides for A (batch, row, col)
    stride_bb, stride_bc, stride_bd,   # strides for B (batch, row, col)
    stride_cc, stride_cd, stride_ce,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # -------------------- offset computation --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # -------------------- accumulator --------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- main loop over K --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # A block: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + pid_batch * stride_aa
            + offs_m[:, None] * stride_ab
            + offs_k[None, :] * stride_ac
        )
        a = tl.load(
            a_ptrs,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )

        # Bᵀ block: we need B transposed, i.e. (k, n)
        # B is stored row‑major (row, col). For the transposed view,
        # the row stride becomes the original column stride.
        b_ptrs = (
            b_ptr
            + pid_batch * stride_bb
            + offs_k[:, None] * stride_bc   # k as row
            + offs_n[None, :] * stride_bd   # n as col
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )

        # Dot product
        acc += tl.dot(a, b)

    # -------------------- write back --------------------
    c_ptrs = (
        c_ptr
        + pid_batch * stride_cc
        + offs_m[:, None] * stride_cd
        + offs_n[None, :] * stride_ce
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


def _batched_trmul(
    left: torch.Tensor, right: torch.Tensor, block_m: int = 64, block_n: int = 64, block_k: int = 32
) -> torch.Tensor:
    """
    Compute the TriMul reduction using the Triton kernel.
    `left` and `right` have shape (B, N, N, H) and dtype fp16.
    Returns a tensor of the same shape and dtype.
    """
    B, N, _, H = left.shape

    # Fold hidden dim into the batch dimension: (B*H, N, N)
    left_f = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_f = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_f = torch.empty_like(left_f)

    # Strides (int64)
    stride_a_batch, stride_a_row, stride_a_col = left_f.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_f.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_f.stride()

    grid = (B * H, tl.cdiv(N, block_m), tl.cdiv(N, block_n))
    _trmul_kernel[grid](
        a_ptr=left_f,
        b_ptr=right_f,
        c_ptr=out_f,
        BATCH=B * H,
        M=N,
        N=N,
        K=N,
        stride_aa=stride_a_batch,
        stride_ab=stride_a_row,
        stride_ac=stride_a_col,
        stride_bb=stride_b_batch,
        stride_bc=stride_b_row,
        stride_bd=stride_b_col,
        stride_cc=stride_c_batch,
        stride_cd=stride_c_row,
        stride_ce=stride_c_col,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )

    # Restore original shape (B, N, N, H)
    out = out_f.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) – optional; may be None.
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast linear weights to fp16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (fp16)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5) Gating projections (sigmoid, fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6) Apply mask (if any) and combine projections with gates
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * left_gate * mask_fp16
        right = right * right_gate * mask_fp16
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 7) Core reduction (TriMul) – batched GEMM via Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    out = _batched_trmul(left, right)  # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype:
        out = out.to(dtype)

    return out
