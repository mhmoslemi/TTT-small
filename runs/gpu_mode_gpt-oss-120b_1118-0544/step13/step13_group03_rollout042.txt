@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)  # batch * head
    pid_x = tl.program_id(0)  # block i
    pid_y = tl.program_id(1)  # block j

    # compute batch and head indices
    b = pid_z // H
    h = pid_z % H

    # compute offsets
    offs_i = pid_x * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_y * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for i,j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers to start of batch and head
    left_base = left_ptr + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base = out_ptr + b * stride_ob + h * stride_oh

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (M, K)
        left_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile (K, N) (since we need right transposed)
        right_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # convert to fp32
        left = left.to(tl.float32)
        right = right.to(tl.float32)

        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul outgoing (AlphaFold3) forward pass – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑normalize the input over the channel dimension `dim`.
2. Linear projections (left/right) and gating projections (left_gate, right_gate,
   out_gate) are performed with PyTorch (bias‑free).
3. Apply the optional mask and fuse it with the gating by element‑wise
   multiplication.
4. Compute the O(N³) “tri‑mul” reduction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a custom Triton kernel that performs a batched matrix multiplication
   (left @ rightᵀ) for each (batch, head) pair.
5. Layer‑norm the result, multiply by `out_gate`, and apply the final linear
   projection back to the original dimension.
6. Return the output in the original dtype (usually float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    left_ptr,     # fp16 *
    right_ptr,    # fp16 *
    out_ptr,      # fp16 *
    B, N, H,      # i32 scalars
    # strides for left
    stride_lb, stride_li, stride_lk, stride_lh,
    # strides for right
    stride_rb, stride_rj, stride_rk, stride_rh,
    # strides for out
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ rightᵀ for each (batch, head) pair.
    left, right, out have shape [B, N, N, H] (contiguous, last dim = H).
    """
    pid_z = tl.program_id(2)   # batch * head
    pid_x = tl.program_id(0)   # block row
    pid_y = tl.program_id(1)   # block col

    # decode batch and head indices
    b = pid_z // H
    h = pid_z % H

    # offsets within the N dimension
    offs_i = pid_x * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_y * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, h)
    left_base = left_ptr + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base = out_ptr + b * stride_ob + h * stride_oh

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + \
                    offs_i[:, None] * stride_li + \
                    offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile (transposed): shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + \
                     offs_k[:, None] * stride_rk + \
                     offs_j[None, :] * stride_rj
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # promote to fp32 for accumulation
        left = left.to(tl.float32)
        right = right.to(tl.float32)

        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # write back
    out_ptrs = out_base + \
               offs_i[:, None] * stride_oi + \
               offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict with the following keys (all float32):
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Linear projections and gating (bias‑free)
    # ------------------------------------------------------------------
    # Cast weights to same dtype as x for matmul
    left_proj_w = weights["left_proj.weight"].to(x.dtype)
    right_proj_w = weights["right_proj.weight"].to(x.dtype)
    left_gate_w = weights["left_gate.weight"].to(x.dtype)
    right_gate_w = weights["right_gate.weight"].to(x.dtype)
    out_gate_w = weights["out_gate.weight"].to(x.dtype)

    left_proj = F.linear(x, left_proj_w)          # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 3) Apply mask (if any) and fuse with gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is float and broadcastable
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left_proj * left_gate * mask_f
        right = right_proj * right_gate * mask_f
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Cast to fp16 for the heavy reduction
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Core reduction (TriMul) using Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, N, N, hidden_dim

    # Prepare output tensor
    out = torch.empty_like(left, dtype=torch.float16)

    # Compute strides (in elements) for each tensor
    # Tensor layout is contiguous: (B, N, N, H)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Block sizes – tune for the target hardware (B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions
    num_blocks_i = triton.cdiv(N, BLOCK_M)
    num_blocks_j = triton.cdiv(N, BLOCK_N)
    num_blocks_z = B * H

    grid = (num_blocks_i, num_blocks_j, num_blocks_z)

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    # Cast back to fp32 for the remaining ops (more stable)
    out = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)
    to_out_w = weights["to_out.weight"].to(dtype)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 6) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype:
        out = out.to(dtype)

    return out
