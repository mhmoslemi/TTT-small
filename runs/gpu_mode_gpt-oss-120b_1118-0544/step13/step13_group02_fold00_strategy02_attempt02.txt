    """TriangleMultiplicativeUpdate module for protein models."""

        self.seqlen = config["seqlen"]
        self.dim = config["dim"]
        self.hidden_dim = config["hidden_dim"]

        if not config.get("no_layernorm", False):
            self.norm = nn.LayerNorm(self.dim)
        else:
            self.norm = None

        self.left_proj = nn.Linear(self.dim, self.hidden_dim, bias=False)
        self.right_proj = nn.Linear(self.dim, self.hidden_dim, bias=False)
        self.left_gate = nn.Linear(self.dim, self.hidden_dim, bias=False)
        self.right_gate = nn.Linear(self.dim, self.hidden_dim, bias=False)
        self.out_gate = nn.Linear(self.dim, self.hidden_dim, bias=False)
        
        # Output transformation
        self.to_out_norm = nn.LayerNorm(self.hidden_dim)
        self.to_out = nn.Linear(self.hidden_dim, self.dim, bias=False)

        # Flatten for batch processing
        input_flat = input.view(input.size(0) * input.size(1) * input.size(2), input.size(3))

            pass

        right_proj = input_flat @ self.right_proj.weight
