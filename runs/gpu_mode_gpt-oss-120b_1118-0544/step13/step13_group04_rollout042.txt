"""
TriMul outgoing (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm over the input channel dimension.
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections (left/right) and gating projections
   (left_gate, right_gate, out_gate) – bias‑free.
4. Fuse mask, projection and gating in a small Triton kernel
   (executed only when a mask is provided and not ignored).
5. Core reduction (the “tri‑mul”) implemented as a batched matrix
   multiplication in Triton:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is equivalent to a batched GEMM per hidden slice.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and apply a final linear projection back to `dim`.
7. Cast the result back to the original dtype.

The implementation mixes PyTorch for the cheap element‑wise ops and a
custom Triton kernel for the expensive reduction, achieving a large
speed‑up on large sequence lengths.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gating and mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64  = B*N*N*H
    hidden_dim,      # i64  = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and optional mask:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    All tensors are flattened in row‑major order.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # load values
    left  = tl.load(left_proj_ptr + offs,  mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # each mask element repeats `hidden_dim` times
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # 1.0 ⇒ no masking

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr  + offs, left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that fuses projection, gating and mask.
    All inputs must be contiguous FP16 tensors.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (left @ right^T)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch of matrices.
    A, B, C: (BATCH, M, K), (BATCH, N, K), (BATCH, M, N)
    B is accessed as transposed inside the kernel (using swapped strides).
    """
    pid = tl.program_id(0)

    # number of blocks per matrix
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch = pid // blocks_per_batch
    block_id = pid % blocks_per_batch
    pid_m = block_id // num_blocks_n
    pid_n = block_id % num_blocks_n

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B tile (transposed): we treat B as (K, N) inside the kernel
        b_ptrs = (
            B_ptr
            + batch * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = (
        C_ptr
        + batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with keys:
                "dim" (C), "hidden_dim" (H), optional "nomask" (bool)
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Apply mask and fuse projection & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous FP16 on same device
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask (or ignored): simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul) via Triton batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) for the batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # strides (all tensors are contiguous, row‑major)
    a_batch_stride = left_flat.stride(0)
    a_m_stride     = left_flat.stride(1)
    a_k_stride     = left_flat.stride(2)

    b_batch_stride = right_flat.stride(0)
    # we need Bᵀ, so swap the inner strides
    b_k_stride = right_flat.stride(2)   # becomes stride over K after transpose
    b_n_stride = right_flat.stride(1)   # becomes stride over N after transpose

    c_batch_stride = out_flat.stride(0)
    c_m_stride     = out_flat.stride(1)
    c_n_stride     = out_flat.stride(2)

    # launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * H * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        a_batch_stride,
        a_m_stride,
        a_k_stride,
        b_batch_stride,
        b_k_stride,
        b_n_stride,
        c_batch_stride,
        c_m_stride,
        c_n_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
