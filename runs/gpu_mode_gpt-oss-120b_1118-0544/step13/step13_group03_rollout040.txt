import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64
    hidden_dim,      # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask_valid = offs < total_elems

    # Compute mask index: each hidden_dim entries correspond to same mask element
    mask_idx = offs // hidden_dim

    left = tl.load(left_proj_ptr + offs, mask=mask_valid, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask_valid, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask_valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask_valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    left_out = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask_valid)
    tl.store(right_out_ptr + offs, right_out, mask=mask_valid)

def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:

    All tensors must be contiguous FP16 and have shape (B,N,N,hidden_dim).
    Returns (left, right) tensors of same shape.

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Apply LayerNorm over the channel dimension `dim`.
2. Cast to FP16 for compute‑intensive parts.
3. Linear projections (`left_proj`, `right_proj`) and gating projections
   (`left_gate`, `right_gate`, `out_gate`) – all bias‑free.
4. Fuse projection, gating and optional mask in a single Triton kernel
   (`_proj_gate_mask_kernel`).  When no mask is used we fall back to a simple
   element‑wise fusion.
5. Core reduction (the “tri‑mul”):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication via `torch.bmm`.
6. Apply a second LayerNorm, multiply by `out_gate`, and project back to the
   original dimension with `to_out`.
7. Cast the result back to the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64
    hidden_dim,      # i64
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    All tensors are flattened in row‑major order.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # Guard for out‑of‑bounds
    mask_valid = offs < total_elems

    # Index of the corresponding mask element (each mask value is repeated
    # `hidden_dim` times across the hidden axis)
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs, mask=mask_valid, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask_valid, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask_valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask_valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)   # 1.0 ⇒ no masking

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr  + offs, left_out,  mask=mask_valid)
    tl.store(right_out_ptr + offs, right_out, mask=mask_valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,

    right_gate: torch.Tensor,
    mask: torch.Tensor,

    Launch the Triton kernel that fuses projection, gating and mask.
    All inputs must be contiguous FP16 tensors.
    Returns (left, right) each of shape (B, N, N, H).
    """

    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out
