"""
TriMul outgoing (AlphaFold3) forward pass – Triton‑accelerated version.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for compute‑intensive parts.
3. Linear projections (left/right) and gating projections
   (left_gate, right_gate, out_gate) – all bias‑free.
4. Fuse projection, gating and optional mask in a single Triton kernel
   (`_proj_gate_mask_kernel`).  When no mask is used we fall back to a
   simple element‑wise PyTorch fusion.
5. Core reduction (the “tri‑mul”):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication via a custom Triton
   kernel (`_tri_mul_kernel`).
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and apply the final linear projection back to `dim`.
7. Cast the result back to the original dtype.

The kernel `_proj_gate_mask_kernel` handles the element‑wise fusion of
projections, gates and mask, while `_tri_mul_kernel` performs the
batched GEMM needed for the O(N³) reduction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Kernel 1: fuse projection, gating and mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask:
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    All tensors are flattened in row‑major order.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # each mask element is repeated `hidden_dim` times along the hidden axis
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # 1.0 ⇒ no masking

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr  + offs, left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that fuses projection, gating and mask.
    All inputs must be contiguous FP16 tensors.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Kernel 2: batched tri‑mul (matrix multiplication)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,   # fp16 *
    right_ptr,  # fp16 *
    out_ptr,    # fp16 *
    M,          # i64 scalar = batch*heads
    N,          # i64 scalar = seq_len
    K,          # i64 scalar = seq_len (same as N)
    stride_left_m, stride_left_i, stride_left_k,
    stride_right_m, stride_right_i, stride_right_k,
    stride_out_m, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication:
       out[m,i,j] = Σ_k left[m,i,k] * right[m,j,k]
       (right is used without transposition, because we index it with j as
        the row dimension and k as the column dimension).
    """
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block column
    pid_b = tl.program_id(2)  # batch*head index

    # Compute offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Guard against out‑of‑bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers for this batch
    left_tile_ptr = left_ptr + pid_b * stride_left_m \
                    + offs_i[:, None] * stride_left_i \
                    + offs_k[None, :] * stride_left_k
    right_tile_ptr = right_ptr + pid_b * stride_right_m \
                     + offs_j[:, None] * stride_right_i \
                     + offs_k[None, :] * stride_right_k

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles
        left_tile = tl.load(left_tile_ptr, mask=mask_i[:, None] & (offs_k < K), other=0.0)
        right_tile = tl.load(right_tile_ptr, mask=mask_j[:, None] & (offs_k < K), other=0.0)

        # Accumulate
        acc += tl.dot(left_tile, right_tile)

        # Advance pointers
        left_tile_ptr += BLOCK_K * stride_left_k
        right_tile_ptr += BLOCK_K * stride_right_k

    # Write back
    out_tile_ptr = out_ptr + pid_b * stride_out_m \
                   + offs_i[:, None] * stride_out_i \
                   + offs_j[None, :] * stride_out_j
    tl.store(out_tile_ptr, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_batched(
    left: torch.Tensor,
    right: torch.Tensor,
    block_m: int = 64,
    block_n: int = 64,
    block_k: int = 32,
) -> torch.Tensor:
    """
    Perform the tri‑mul reduction using a Triton kernel.
    `left` and `right` are expected to have shape (B, N, N, H) and dtype fp16.
    Returns a tensor of the same shape and dtype.
    """
    B, N, _, H = left.shape
    M = B * H

    # reshape to (M, N, N)
    left_ = left.permute(0, 3, 1, 2).contiguous().view(M, N, N)
    right_ = right.permute(0, 3, 1, 2).contiguous().view(M, N, N)

    out = torch.empty_like(left_)

    # strides (in elements)
    stride_l_m, stride_l_i, stride_l_k = left_.stride()
    stride_r_m, stride_r_i, stride_r_k = right_.stride()
    stride_o_m, stride_o_i, stride_o_j = out.stride()

    grid = (
        triton.cdiv(N, block_m),
        triton.cdiv(N, block_n),
        M,
    )
    _tri_mul_kernel[grid](
        left_,
        right_,
        out,
        M,
        N,
        N,
        stride_l_m,
        stride_l_i,
        stride_l_k,
        stride_r_m,
        stride_r_i,
        stride_r_k,
        stride_o_m,
        stride_o_i,
        stride_o_j,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )

    # reshape back to (B, N, N, H)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict with keys:
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Apply mask and fuse with projection & gating (Triton if needed)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous FP16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask (or ignored): simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul) via Triton batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out = _tri_mul_batched(left, right)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
