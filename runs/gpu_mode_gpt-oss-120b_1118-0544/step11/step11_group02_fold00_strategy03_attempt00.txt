    x: [B, N, N, C]

  Operation steps:

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(x_ptr, y_ptr, mask_ptr, out_ptr, dim_block_size: tl.constexpr, hidden_dim_block_size: tl.constexpr, 
                   BLOCK_SIZE: tl.constexpr, num_batches: tl.constexpr, num_seqlens: tl.constexpr, 
                   num_hidden_dim: tl.constexpr, num_blocks: tl.constexpr, mask_size: tl.constexpr, nomask: tl.constexpr, 
                   has_mask: tl.constexpr, total_elements: tl.constexpr):
    # Get program ID
    b_idx = tl.program_id(0)
    # Get local block indices
    n_inner = tl.program_id(1)
    k = tl.program_id(2)

    # Get offsets and start index
    n_outer_offset = b_idx * num_blocks * BLOCK_SIZE + n_inner * BLOCK_SIZE * hidden_dim_block_size
    seq_start = n_outer_offset * num_hidden_dim + k

    # Compute offsets
    off_d = (seq_start // num_hidden_dim) % dim_block_size
    off_k = (seq_start // dim_block_size) % dim_block_size
    off_h = (seq_start // (dim_block_size * dim_block_size)) % hidden_dim_block_size
    off_b = seq_start // (dim_block_size * dim_block_size * hidden_dim_block_size) - 1  # program_id from -1 to B-1

    # Check if block is within bounds
    if off_b < 0 or off_b >= num_batches:
        return
    
    # Load x and y
    x = tl.load(x_ptr, (has_mask or mask_ptr not None), default=0.0)
    y = tl.load(y_ptr, (has_mask or mask_ptr not None), default=0.0)
