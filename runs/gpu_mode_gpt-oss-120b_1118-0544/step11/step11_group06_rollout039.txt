def custom_kernel(data):
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Input layer norm
    x = torch.nn.functional.layer_norm(inp, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Cast to fp16
    x = x.to(torch.float16)

    # Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)  # (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Apply mask and gating
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16)
        # Ensure contiguous
        mask_fp16 = mask_fp16.contiguous()
        # Kernel
        left, right = _apply_proj_gate_mask(left_proj, right_proj, left_gate, right_gate, mask_fp16, block_size=4096)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # Should not happen as mask conditions already considered, but for completeness
            left = left * mask[..., None].to(torch.float16)
            right = right * mask[..., None].to(torch.float16)

    # Core reduction
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layer norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    out = torch.nn.functional.linear(out, to_out_w)

    # Cast to float32
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass implemented with Triton fusion.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Cast to fp16 for compute‑intensive steps.
3. Linear projections (left / right) and corresponding gates (left / right / out)
   using the provided weight tensors (bias‑free).  Gates are passed through sigmoid.
4. Fuse projection, gating and optional mask in a single Triton kernel
   (`_proj_gate_mask_kernel`).  The kernel produces the tensors `left` and `right`
   of shape (B, N, N, H).
5. Core triangular reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM by reshaping to (B*H, N, N) and using
   `torch.bmm`, which calls cuBLAS.
6. Apply output LayerNorm, multiply by the out‑gate and project back to the
   original channel dimension.
7. Cast the result back to float32 (the required output dtype).

At least one operation (the projection‑gate‑mask fusion) is performed inside a
Triton kernel, satisfying the requirement to use custom kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)
    right_gate_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    left_out_ptr,    # fp16 * (B, N, N, H)
    right_out_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # i64 = B * N * N * H
    hidden_dim,      # i64 = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask for the left and right branches."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # each (b,i,j) entry owns `hidden_dim` consecutive elements
    mask_idx = offs // hidden_dim

    left_proj = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 4096,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gating and mask for the left/right tensors via a Triton kernel.

    All tensors must be fp16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)

    Returns
    -------
    left_out, right_out : torch.Tensor
        Shape (B, N, N, H)
    """
    total = left_proj.numel()
    B, N, _, H = left_proj.shape
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32/float16)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict with keys:
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask" (optional) – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = inp.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to fp16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Apply mask + gating (fused via Triton when mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device).to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16, block_size=4096
        )
    else:
        # No mask or mask ignored: simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # fallback (should rarely happen)
            mask_fp16 = mask.to(device).to(torch.float16).unsqueeze(-1)
            left = left * mask_fp16
            right = right * mask_fp16

    # ------------------------------------------------------------------
    # 7) Core triangular reduction:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) without extra copies
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Batched matrix multiplication (fp16)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate multiplication and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to float32 (required output dtype)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
