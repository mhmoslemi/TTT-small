def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bm, stride_bn,
    stride_cz, stride_cm, stride_cn,

    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)  # batch
    pid_m = tl.program_id(1)  # tile row
    pid_n = tl.program_id(2)  # tile col

    # Compute offsets

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

def _batched_matmul(left, right, block_m=64, block_n=64, block_k=32):
    # left shape (B, N, N, H); but we will permute to (B, H, N, N)
    B, N, _, H = left.shape
    # permute to (B, H, N, N) to combine B*H.
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch = B * H
    left_flat = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)
    # allocate output
    out = torch.empty_like(left_flat)  # fp16

    # compute strides
    a_stride_z, a_stride_m, a_stride_k = left_flat.stride()
    b_stride_z, b_stride_m, b_stride_k = right_flat.stride()
    c_stride_z, c_stride_m, c_stride_n = out.stride()
    # Launch kernel
    grid = (batch, tl.cdiv(N, block_m), tl.cdiv(N, block_n))
    # Note that program_id order: pid_z = program_id(0), pid_m = program_id(1), pid_n = program_id(2)
    _batched_gemm_kernel[grid](
        left_flat, right_flat, out,
        N, N, N,
        a_stride_z, a_stride_m, a_stride_k,
        b_stride_z, b_stride_m, b_stride_k,
        c_stride_z, c_stride_m, c_stride_n,
        BLOCK_M=block_m, BLOCK_N=block_n, BLOCK_K=block_k
    )
    # reshape back
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated.

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension.
2. Cast to fp16 for compute‑intensive operations.
3. Linear projections (left, right) and gating projections (left_gate,
   right_gate, out_gate) – all bias‑free.
4. Fuse projection, gating and optional mask in a small Triton kernel.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM (A @ Bᵀ) via Triton.  The batch dimension
   is B·H (hidden dim), each GEMM has shape N×N.
6. Layer‑norm over the hidden dimension, multiply by out_gate and apply the
   final linear projection back to the original channel dimension.
7. Cast the result to the original dtype (float32).

Only the element‑wise fusion (step 4) and the batched GEMM (step 5) are
implemented in Triton; the remaining operations use highly‑optimized
PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse projection, gating and optional mask:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
    The mask (B,N,N) is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask is indexed per (b,i,k), i.e. every hidden_dim elements share the same mask entry
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=valid, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)  # default 1.0 = no mask

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that fuses projection, gating and mask.
    All tensors must be fp16 and contiguous.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all int64)
    stride_az, stride_am, stride_ak,   # strides for A: (batch, row, col)
    stride_bz, stride_bm, stride_bn,   # strides for B (original layout)
    stride_cz, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for out = A @ Bᵀ
    A: (batch, M, K)
    B: (batch, N, K)   (we treat Bᵀ on‑the‑fly)
    C: (batch, M, N)
    Accumulation is in fp32, result stored as fp16.
    """
    pid_z = tl.program_id(0)   # batch index
    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile col

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile
    a_ptrs = a_ptr + pid_z * stride_az + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_z * stride_bz + tl.arange(0, BLOCK_K)[:, None] * stride_bn + offs_n[None, :] * stride_bm

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        # Compute how many valid k elements remain in this block
        cur_k = tl.arange(0, BLOCK_K)
        valid_k = cur_k < (K - k * BLOCK_K)

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & valid_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=valid_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

        # Advance pointers to the next k‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bn

    # Write back the result
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_z * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul(
    left: torch.Tensor,
    right: torch.Tensor,
    block_m: int = 64,
    block_n: int = 64,
    block_k: int = 32,
) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using the Triton
    batched GEMM kernel.
    Input tensors are fp16 of shape (B, N, N, H).
    Returns a tensor of the same shape (fp16).
    """
    B, N, _, H = left.shape
    # bring hidden dimension next to batch for easy flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch = B * H
    left_flat = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)
    out_flat = torch.empty_like(left_flat)

    # strides (in element count)
    a_sz = left_flat.stride()
    b_sz = right_flat.stride()
    c_sz = out_flat.stride()

    grid = (
        batch,
        triton.cdiv(N, block_m),
        triton.cdiv(N, block_n),
    )
    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        a_sz[0],
        a_sz[1],
        a_sz[2],
        b_sz[0],
        b_sz[1],
        b_sz[2],
        c_sz[0],
        c_sz[1],
        c_sz[2],
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) – optional
            - weights: dictionary of model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1) Input LayerNorm over `dim`
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # 2) Cast to fp16 for compute‑intensive parts
    x = x.to(torch.float16)

    # 3) Cast linear weights to fp16 (biases are None)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # 4) Linear projections (fp16)
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # 5) Gating projections (sigmoid, fp16)
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # 6) Fuse mask, projection and gating
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask – simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 7) Core reduction (TriMul) – batched GEMM over the hidden dimension
    out = _batched_matmul(left, right)

    # 8) Output LayerNorm, out‑gate, final linear projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # 9) Cast back to original dtype
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
