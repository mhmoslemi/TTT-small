@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # compute batch, block indices
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_id = pid // (num_blocks_m * num_blocks_n)
    block_index = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_index // num_blocks_n
    pid_n = block_index % num_blocks_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers
    a_ptrs = a_ptr + batch_id * stride_batch_a + offs_m[:, None] * stride_m_a + offs_k[None, :] * stride_k_a
    b_ptrs = b_ptr + batch_id * stride_batch_b + offs_k[:, None] * stride_k_b + offs_n[None, :] * stride_n_b

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K - k*BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K - k*BLOCK_K) & mask_n[None, :], other=0.0)
        # compute dot
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_k_a
        b_ptrs += BLOCK_K * stride_k_b

    # store result
    c_ptrs = c_ptr + batch_id * stride_batch_c + offs_m[:, None] * stride_m_c + offs_n[None, :] * stride_n_c
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul outgoing (AlphaFold3) forward pass.

The implementation fuses the projection, gating and optional mask in a small
Triton kernel, then performs the heavy N³ reduction with a custom batched
matrix‑multiply kernel (also in Triton).  The rest (LayerNorms, sigmoids,
final linear) uses PyTorch functional ops.  All compute‑intensive parts run
in float16; the final output is cast back to the input dtype (float32).
"""

import torch
import torch.nn.functional as F
from typing import Tuple, Dict

import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Fusion of projection, gating and mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    The mask (B,N,N) is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # index of the mask element (b,i,k) for each hidden element
    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=valid, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lgate = tl.load(left_gate_ptr + offs,  mask=valid, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default no‑mask

    left_out  = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs,  left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that fuses projection, gating and optional mask.
    All tensors must be fp16 and contiguous.
    Returns (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# 2. Batched matrix multiplication (B*H) x (N,N) * (N,N)^T
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    batch, M, N, K,                    # dimensions
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of tiles
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # decode batch and tile indices
    batch_id = pid // (num_blocks_m * num_blocks_n)
    block_index = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_index // num_blocks_n
    pid_n = block_index % num_blocks_n

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this tile
    a_ptrs = a_ptr + batch_id * stride_batch_a \
                    + offs_m[:, None] * stride_m_a \
                    + offs_k[None, :] * stride_k_a
    b_ptrs = b_ptr + batch_id * stride_batch_b \
                    + offs_k[:, None] * stride_k_b \
                    + offs_n[None, :] * stride_n_b

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # mask for the current K slice
        mask_k = (offs_k + k_offset) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_N) in fp32

        # advance pointers to next K‑slice
        a_ptrs += BLOCK_K * stride_k_a
        b_ptrs += BLOCK_K * stride_k_b

    # store result
    c_ptrs = c_ptr + batch_id * stride_batch_c \
                    + offs_m[:, None] * stride_m_c \
                    + offs_n[None, :] * stride_n_c
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul(a: torch.Tensor, b: torch.Tensor,
                    BLOCK_M: int = 128, BLOCK_N: int = 128, BLOCK_K: int = 32) -> torch.Tensor:
    """
    Batched matrix multiplication:
        out[b,i,j] = sum_k a[b,i,k] * b[b,j,k]
    a: (B, M, K)   fp16
    b: (B, N, K)   fp16
    returns (B, M, N) fp16
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, N, _ = b.shape

    out = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    stride_batch_a, stride_m_a, stride_k_a = a.stride()
    stride_batch_b, stride_k_b, stride_n_b = b.stride()
    stride_batch_c, stride_m_c, stride_n_c = out.stride()

    grid = (batch * ((M + BLOCK_M - 1) // BLOCK_M) *
            ((N + BLOCK_N - 1) // BLOCK_N),)

    _batched_matmul_kernel[grid](
        a, b, out,
        batch, M, N, K,
        stride_batch_a, stride_m_a, stride_k_a,
        stride_batch_b, stride_k_b, stride_n_b,
        stride_batch_c, stride_m_c, stride_n_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# 3. Full custom kernel
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) – may be None or all‑ones.
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast linear weights to fp16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (fp16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5) Gating projections (sigmoid, fp16)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6) Fuse mask, projection and gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask (or ignored): simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7) Core reduction (TriMul) – batched GEMM over the hidden dim
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # (B*H, N, N) x (B*H, N, N)^T -> (B*H, N, N)
    out_flat = _batched_matmul(left_flat, right_flat,
                               BLOCK_M=128, BLOCK_N=128, BLOCK_K=32)

    # reshape back to (B, N, N, H)
    out = out_flat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
