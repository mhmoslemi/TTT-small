@triton.jit
def batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute batch id and tile indices
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_in_batch = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = pid_in_batch // tl.cdiv(N, BLOCK_N)
    pid_n = pid_in_batch % tl.cdiv(N, BLOCK_N)

    # Block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers offset for this batch
    a_ptrs = A + batch * batch_stride_a
    b_ptrs = B + batch * batch_stride_b
    c_ptrs = C + batch * batch_stride_c

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute indexes
        k_offset = k * BLOCK_K
        # Load A block
        a = tl.load(
            a_ptrs + (offs_m[:, None] * stride_am + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak),
            mask=(offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K),
            other=0.0,
        )
        # Load B block (B is transposed: shape (batch, K, N))
        b = tl.load(
            b_ptrs + ((k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=(k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        # Compute dot product
        acc += tl.dot(a, b)

    # Write back
    c = acc.to(tl.float16)
    tl.store(
        c_ptrs + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )
```

"""
TriMul outgoing (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for the compute‑intensive part.
3. Linear projections (left/right) and gating projections (left/right/out).
4. Apply mask (if present) and element‑wise gating.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication using a custom Triton kernel.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
7. Cast the result back to the original dtype (float32).

The kernel in step 5 computes many independent matrix products
(batch * hidden_dim) with tiling and fp16 arithmetic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matmul (A @ B^T) where A and B are (B, M, K) and (B, N, K)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A, B, C,               # pointers
    M, N, K,               # matrix sizes
    stride_am, stride_ak,  # A strides (row, col)
    stride_bk, stride_bn,  # B strides (row=K, col=N) because B is KxN
    stride_cm, stride_cn,  # C strides (row, col)
    batch_stride_a,        # stride to the next batch in A
    batch_stride_b,        # stride to the next batch in B
    batch_stride_c,        # stride to the next batch in C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Compute batch index and tile indices
    # --------------------------------------------------------------
    tiles_m = tl.cdiv(M, BLOCK_M)
    tiles_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (tiles_m * tiles_n)
    pid_in_batch = pid % (tiles_m * tiles_n)
    pid_m = pid_in_batch // tiles_n
    pid_n = pid_in_batch % tiles_n

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_batch = A + batch * batch_stride_a
    b_batch = B + batch * batch_stride_b
    c_batch = C + batch * batch_stride_c

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K)

        # Load A tile (BLOCK_M x BLOCK_K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am
                     + (k_offset + k_range)[None, :] * stride_ak),
            mask=(offs_m[:, None] < M) &
                 (k_offset + k_range)[None, :] < K,
            other=0.0,
        )

        # Load B tile (BLOCK_K x BLOCK_N) – B is stored as (K, N)
        b = tl.load(
            b_batch + ((k_offset + k_range)[:, None] * stride_bk
                     + offs_n[None, :] * stride_bn),
            mask=(k_offset + k_range)[:, None] < K &
                 (offs_n[None, :] < N),
            other=0.0,
        )

        # Dot product accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back result tile
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm
                 + offs_n[None, :] * stride_cn),
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor,
                           BLOCK_M=128, BLOCK_N=128, BLOCK_K=32) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B^T using Triton.
    A: (batch, M, K)  fp16
    B: (batch, N, K)  fp16  (will be interpreted as KxN inside the kernel)
    Returns C: (batch, M, N) fp16
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, _ = B.shape
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Strides
    stride_am, stride_ak = A.stride(1), A.stride(2)
    # B is (batch, N, K) but we treat it as (batch, K, N) for the kernel
    stride_bk, stride_bn = B.stride(2), B.stride(1)
    stride_cm, stride_cn = C.stride(1), C.stride(2)

    batch_stride_a = A.stride(0)
    batch_stride_b = B.stride(0)
    batch_stride_c = C.stride(0)

    # Grid: one program per output tile (over all batches)
    grid_m = (M + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch * grid_m * grid_n
    grid = (total_blocks,)

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )
    return C


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) – optional; may be all‑ones.
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim), float32

    # --------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy parts
    # --------------------------------------------------------------
    x = x.half()

    # --------------------------------------------------------------
    # 3) Cast linear weights to fp16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # --------------------------------------------------------------
    # 4) Linear projections (fp16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5) Gating projections (sigmoid, fp16)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6) Apply mask and gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left  = left_proj * left_gate * mask_fp
        right = right_proj * right_gate * mask_fp
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched matmul (left @ right^T)
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Transpose right for the matmul (need KxN)
    right_t = right_flat.transpose(1, 2).contiguous()

    # Batched matmul using Triton
    out_flat = _batched_matmul_triton(left_flat, right_t,
                                      BLOCK_M=128, BLOCK_N=128, BLOCK_K=32)  # (B*H, N, N)

    # Restore original shape (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9) Cast back to original dtype (float32)
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
