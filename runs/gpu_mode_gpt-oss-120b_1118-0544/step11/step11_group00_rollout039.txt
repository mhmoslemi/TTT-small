import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64
    hidden_dim,      # i64
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and mask.
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask
    mask is broadcast over hidden_dim.
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard
    mask = offs < total_elems

    # compute mask index: each element repeats hidden_dim times
    mask_idx = offs // hidden_dim

    left = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = left * lgate * m
    right_out = right * rgate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """Launch kernel that computes
    left = left_proj * left_gate * mask
    right = right_proj * right_gate * mask
    All tensors must be fp16 and contiguous.
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Implements:
      1. LayerNorm on input.
      2. Linear projections + gating.
      3. Fusion of projection, gate and optional mask via a Triton kernel.
      4. Core reduction out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
      5. Output LayerNorm, out‑gate and final linear projection.
    Returns tensor of shape [B, N, N, dim] in the original dtype.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1) Input LayerNorm over channel dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # 2) Cast to fp16 for compute‑heavy parts
    x_fp16 = x.to(torch.float16)

    # 3) Convert weights to fp16 (they are stored as float32)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # 4) Linear projections (no bias)
    left_proj = F.linear(x_fp16, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x_fp16, right_proj_w) # (B, N, N, H)

    # 5) Gating projections with sigmoid
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    # 6) Fuse mask, projection and gating
    if mask is not None and not nomask:
        # mask shape [B, N, N]; broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 7) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) to use batched GEMM
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Perform batch matrix multiplication (left * right^T)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 8) Output layer norm, out‑gate and final linear projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # 9) Cast back to original dtype
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

import triton
import triton.language as tl
from typing import Tuple, Dict

def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):

    Computes
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask

    `mask` has shape [B, N, N] and is broadcast over the hidden
    dimension `hidden_dim`.
    """

    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    active = offs < total_elems

    mask_idx = offs // hidden_dim

    left  = tl.load(left_proj_ptr + offs,  mask=active, other=0.0)
    right = tl.load(right_proj_ptr + offs, mask=active, other=0.0)

    3. Linear projections (left/right) and gating projections (left/right/out).
    4. Fuse projection, gating and optional mask via a Triton kernel.
    5. Core reduction:

    6. Output layer‑norm, element‑wise out‑gate, and final linear projection.
    7. Cast back to the original input dtype.

    # 4) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x_fp16, left_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # ------------------------------------------------------------------
