@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr
):
    pid = tl.program_id(0)
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    total_tiles = num_tiles_m * num_tiles_n
    batch = pid // total_tiles
    tile_index = pid % total_tiles
    tile_m = tile_index // num_tiles_n
    tile_n = tile_index % num_tiles_n

    # Compute offset coordinates
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A block
        a_ptrs = a_ptr + batch * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)

        # Load B block (transposed)
        b_ptrs = b_ptr + batch * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        # dot
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,

    num_warps: tl.constexpr,
):

    tile_index = pid % tiles_per_batch
    tile_m = tile_index // num_tiles_n
    tile_n = tile_index % num_tiles_n

    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M

    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Implements the operations described in the AlphaFold3 paper:
      1. LayerNorm over the input channel dimension.
      2. Linear projections (left/right) and gating projections.
      3. Optional masking.
      4. Core reduction:
            out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
         performed with a fused Triton kernel.
      5. Output LayerNorm, gating and final linear projection.
    Returns a tensor of shape [B, N, N, dim] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------------------
    # 1) Input LayerNorm
    # -------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # -------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy parts
    # -------------------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------------------
    # 3) Cast weights to fp16
    # -------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------------------
    # 4) Linear projections
    # -------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # -------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # -------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # -------------------------------------------------------------
    # 6) Apply mask and gating
    # -------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape (B, N, N) -> (B, N, N, 1)
        mask_exp = mask.unsqueeze(-1).to(torch.float16)
        left = left_proj * left_gate * mask_exp
        right = right_proj * right_gate * mask_exp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape  # H == hidden_dim

    # -------------------------------------------------------------
    # 7) Core TriMul reduction using Triton
    # -------------------------------------------------------------
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)  # fp16 output

    # Strides
    batch_stride_a = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    batch_stride_b = right_flat.stride(0)
    # For transposed access we swap the last two strides
    stride_bk = right_flat.stride(2)   # original col stride
    stride_bn = right_flat.stride(1)   # original row stride

    batch_stride_c = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    total_batches = B * H
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (total_batches * tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N),)

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------
    # 8) Output LayerNorm, gating and final linear projection
    # -------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # -------------------------------------------------------------
    # 9) Cast back to original dtype
    # -------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    """
    Batched matrix multiplication for the TriMul core:
        out[b,i,j] = Σ_k a[b,i,k] * b[b,j,k]
    The `b` matrix is accessed in transposed order, i.e.
    we load b[k,j] = right[b,j,k].
    """
    pid = tl.program_id(0)

    # number of tiles per batch
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_tiles_m * num_tiles_n

    batch = pid // tiles_per_batch
    tile_index = pid % tiles_per_batch
    tile_m = tile_index // num_tiles_n
    tile_n = tile_index % num_tiles_n

    # offsets inside the output tile
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A block (shape BLOCK_M x BLOCK_K)
        a_ptrs = a_ptr + batch * batch_stride_a \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & (offs_k[None, :] < K)),
                    other=0.0)

        # Load B block transposed (shape BLOCK_K x BLOCK_N)
        b_ptrs = b_ptr + batch * batch_stride_b \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=((offs_k[:, None] < K) & mask_n[None, :]),
                    other=0.0)

        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = c_ptr + batch * batch_stride_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Steps:
      1. LayerNorm over the input channel dimension.
      2. Linear projections (left/right) and gating projections.
      3. Optional masking.
      4. Core reduction:
            out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
         performed with a fused Triton kernel.
      5. Output LayerNorm, gating and final linear projection.
    Returns a tensor of shape [B, N, N, dim] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Apply mask and gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_exp = mask.unsqueeze(-1).to(torch.float16)          # (B, N, N, 1)
        left = left_proj * left_gate * mask_exp
        right = right_proj * right_gate * mask_exp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape  # H == hidden_dim

    # ------------------------------------------------------------------
    # 7) Core TriMul reduction using Triton
    # ------------------------------------------------------------------
    # Reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)  # fp16 output buffer

    # Strides for the kernel
    batch_stride_a = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    batch_stride_b = right_flat.stride(0)
    # Transposed access: swap last two strides
    stride_bk = right_flat.stride(2)   # original column stride
    stride_bn = right_flat.stride(1)   # original row stride

    batch_stride_c = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    total_batches = B * H
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (total_batches *
            ((N + BLOCK_M - 1) // BLOCK_M) *
            ((N + BLOCK_N - 1) // BLOCK_N),)

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        N,          # M
        N,          # N
        N,          # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
