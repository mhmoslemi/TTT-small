"""
TriMul (outgoing) forward pass – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for compute‑heavy parts.
3. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) – bias‑free.
4. Apply optional mask and fuse it with projection + gating (elementwise;
   a separate Triton kernel is used for the heavy reduction only).
5. Core reduction (TriMul):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (B*H matrices of size N×N)
   where the second operand is transposed. Implemented with a custom
   Triton GEMM kernel.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and apply the final linear projection back to `dim`.
7. Cast the result back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _bmm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,          # a: (batch, M, K)
    stride_bb, stride_bk, stride_bn,          # b: (batch, N, K) → we treat it as (batch, K, N)
    stride_cb, stride_cm, stride_cn,          # c: (batch, M, N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ   (A,B,C are fp16, accumulation in fp32)."""
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers with batch offset
    a_ptrs = a_ptr + batch_idx * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_idx * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        # Load tiles
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)
        # Compute dot product
        acc += tl.dot(a, b)
        # Advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + batch_idx * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             c,
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])


def _batched_gemm_triton(a: torch.Tensor, b: torch.Tensor,
                         block_m: int = 64, block_n: int = 64, block_k: int = 64) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for batched 3‑D tensors A (B, M, K) and B (B, N, K)
    using a Triton kernel.
    Returns C with shape (B, M, N) and dtype matching the inputs.
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, N, Kb = b.shape
    assert K == Kb, "Inner dimensions must match"

    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    # Strides
    stride_ab = a.stride(0)
    stride_am = a.stride(1)
    stride_ak = a.stride(2)

    stride_bb = b.stride(0)
    # We treat B as transposed: we need stride for K (original last dim) and N (original middle dim)
    stride_bk = b.stride(2)   # K dimension
    stride_bn = b.stride(1)   # N dimension

    stride_cb = c.stride(0)
    stride_cm = c.stride(1)
    stride_cn = c.stride(2)

    grid = (batch,
            triton.cdiv(M, block_m),
            triton.cdiv(N, block_n))
    _bmm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=M,
        N=N,
        K=K,
        stride_ab=stride_ab, stride_am=stride_am, stride_ak=stride_ak,
        stride_bb=stride_bb, stride_bk=stride_bk, stride_bn=stride_bn,
        stride_cb=stride_cb, stride_cm=stride_cm, stride_cn=stride_cn,
        BLOCK_M=block_m, BLOCK_N=block_n, BLOCK_K=block_k,
    )
    return c


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to fp16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Apply mask and fuse with projection & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape: [B, N, N] → broadcast over hidden dim
        mask_fp16 = mask.to(device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * mask_fp16 * left_gate
        right = right_proj * mask_fp16 * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul) via batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    → for each (b,h) compute left[b,:, :,h] @ right[b,:, :,h]ᵀ
    # ------------------------------------------------------------------
    # Rearrange to (B*H, N, N) for the batched GEMM
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)
    B, H, N, _ = left.shape
    left_flat = left.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right.view(B * H, N, N) # (B*H, N, N)

    # Perform batched matmul: left_flat @ right_flatᵀ
    out_flat = _batched_gemm_triton(left_flat, right_flat,
                                    block_m=64, block_n=64, block_k=64)  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
