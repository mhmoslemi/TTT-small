TriMul (outgoing) forward pass.

Algorithm
---------
1. LayerNorm over the last channel dimension of the input tensor.
2. Two bias‑free linear projections (left/right) from dim → hidden_dim.
3. Optional mask (broadcasted on hidden_dim) applied to the projected tensors.

   The left/right gates are applied to the projected tensors.
5. Core O(N³) contraction:
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication:
      out = left @ rightᵀ for each (b,d).
   Implemented as a fused Triton GEMM kernel that works on FP16 inputs
   and accumulates in FP32.
6. LayerNorm over hidden_dim, multiply by the out‑gate and apply a final
   bias‑free linear projection back to dim.

All heavy compute (step 5) is performed in Triton; the remaining
operations use PyTorch (which is already highly optimized for the
preceding linear layers and normalizations).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the batched GEMM: out = left @ right.T
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to left, right, out
    M, N, K,                           # matrix dimensions (all = seq_len)
    stride_ah, stride_am, stride_ak,   # strides for a: batch*head, row(i), col(k)
    stride_bh, stride_bk, stride_bn,   # strides for b: batch*head, row(k), col(j)
    stride_ch, stride_cm, stride_cn,   # strides for c: batch*head, row(i), col(j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_batch = tl.program_id(0)   # batch * head index
    pid_m = tl.program_id(1)       # block row (i)
    pid_n = tl.program_id(2)       # block col (j)

    # Compute offsets for the block.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Base pointers for this batch*head.
    a_batch_ptr = a_ptr + pid_batch * stride_ah
    b_batch_ptr = b_ptr + pid_batch * stride_bh
    c_batch_ptr = c_ptr + pid_batch * stride_ch

    # Accumulator in FP32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads.
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Compute pointers for the tiles.
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (half‑precision). Out‑of‑bounds elements are zeroed.
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Rank‑K update.
        acc += tl.dot(a, b)   # acc is FP32

    # Write back the result (FP32).
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: Tuple of (input_tensor, mask_tensor_or_None, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict mapping weight names to torch.Tensor
            - config: dict containing at least:
                * "dim" (int)          – input channel dimension
                * "hidden_dim" (int)   – hidden dimension for projections

    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # Ensure mask is float and broadcastable.
        mask = mask.unsqueeze(-1).type_as(x)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core O(N³) contraction via Triton GEMM
    # ------------------------------------------------------------------
    # Cast to half for compute; keep the original for gating (out_gate was already computed).
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to (B, hidden, N, N) and make right transposed along the N axes.
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_h.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) where last two dims are swapped

    B, H, N, _ = left_perm.shape

    # Collapse batch and head dims for a single GEMM launch.
    left_mat = left_perm.view(B * H, N, N)   # (B*H, M, K)
    right_mat = right_perm.view(B * H, N, N) # (B*H, K, N)

    # Output buffer (FP32 accumulation).
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=x.device)

    # Choose block sizes: larger tiles when N is big to reduce kernel launch overhead.
    if N >= 512:
        BLOCK_M = 128
        BLOCK_N = 128
    else:
        BLOCK_M = 64
        BLOCK_N = 64
    BLOCK_K = 32  # small enough for shared memory, good for tensor‑cores

    # Grid dimensions: (batch*head, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        left_mat.stride(0),   # stride_ah (batch*head)
        left_mat.stride(1),   # stride_am (row i)
        left_mat.stride(2),   # stride_ak (col k)
        right_mat.stride(0),  # stride_bh
        right_mat.stride(1),  # stride_bk (row k)
        right_mat.stride(2),  # stride_bn (col j)
        out_mat.stride(0),    # stride_ch
        out_mat.stride(1),    # stride_cm (row i)
        out_mat.stride(2),    # stride_cn (col j)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # back to [B, N, N, dim]

    return out
```
