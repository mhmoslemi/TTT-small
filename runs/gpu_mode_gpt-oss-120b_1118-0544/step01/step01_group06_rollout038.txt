    stride_bb, stride_bh, stride_bk, stride_bn,

Now inside kernel:

```
pid = tl.program_id(0)
pid_m = tl.program_id(1)
pid_n = tl.program_id(2)

# compute batch and head
batch_idx = pid // H
head_idx = pid % H

# compute base offsets 
a_base = a_ptr + batch_idx * stride_ab + head_idx * stride_ah
b_base = b_ptr + batch_idx * stride_bb + head_idx * stride_bh
c_base = c_ptr + batch_idx * stride_cb + head_idx * stride_ch

# compute offsets for i and j within block
offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

# Load accumulator zero
acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

# K loop
for k in range(0, tl.cdiv(N, BLOCK_K)):
    offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

    # Mask for bounds
    a_mask = (offs_m[:, None] < N) & (offs_k[None, :] < N)
    b_mask = (offs_k[:, None] < N) & (offs_n[None, :] < N)

    a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    a = tl.load(a_ptrs, mask=a_mask, other=0.0)
    b = tl.load(b_ptrs, mask=b_mask, other=0.0)

    acc += tl.dot(a, b)
# store
c_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)
c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
tl.store(c_ptrs, acc, mask=c_mask)
```

"""
TriMul (outgoing) forward pass with a fused batched GEMM kernel in Triton.

Algorithm
---------
1. LayerNorm the input over the last dimension.
2. Cast to float16 and apply linear projections (left/right) in half precision.
3. Apply optional mask and sigmoid gates (left/right) in half precision.
4. Perform the core N³ contraction:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batch of matrix multiplications (left @ rightᵀ) for each
   head `d`.  The GEMM is implemented in Triton, tiling the (i, j, k)
   axes (64×64×64) and accumulating in float32 while the inputs are
   float16.
5. LayerNorm the result over the hidden dimension, apply the output gate
   and a final linear projection back to the original `dim`.
All heavy work (the N³ reduction) is done in the custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    B, H, N,
    stride_ah, stride_am, stride_ak,   # left strides: batch, row(i), col(k)
    stride_bh, stride_bk, stride_bn,   # right strides: batch, row(k), col(j)
    stride_ch, stride_cm, stride_cn,   # out strides: batch, row(i), col(j)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM for one head:  out[b,i,j] = Σₖ left[b,i,k] * right[b,k,j]
    The kernel is launched with a 3‑D grid:
        dim0 : B * H  (combined batch & head)
        dim1 : ceil(N / BLOCK_M)   (tile rows)
        dim2 : ceil(N / BLOCK_N)   (tile cols)
    """
    pid = tl.program_id(0)     # combined batch*head index
    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile col

    # Decompose combined index into batch and head
    batch_idx = pid // H
    head_idx = pid % H

    # Base pointers for this (batch, head)
    a_base = a_ptr + batch_idx * stride_ah + head_idx * stride_am  # stride_am == stride of hidden dim (contiguous)
    b_base = b_ptr + batch_idx * stride_bh + head_idx * stride_bn
    c_base = c_ptr + batch_idx * stride_ch + head_idx * stride_cm

    # Offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for OOB loads
        a_mask = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        b_mask = (offs_k[:, None] < N) & (offs_n[None, :] < N)

        # Load tiles from A and B
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # FP16 × FP16 → accumulate in FP32
        acc += tl.dot(a, b)

    # Write result
    c_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with all required weight tensors
            - config: dict with 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Cast to half and linear projections (half precision)
    # -----------------------------------------------------------------
    x_h = x.to(torch.float16)

    # Weight tensors in half precision (they are small, conversion is cheap)
    left_proj_w_h = weights["left_proj.weight"].half()
    right_proj_w_h = weights["right_proj.weight"].half()
    left_gate_w_h = weights["left_gate.weight"].half()
    right_gate_w_h = weights["right_gate.weight"].half()

    # Project
    left = F.linear(x_h, left_proj_w_h)   # [B, N, N, hidden_dim] (fp16)
    right = F.linear(x_h, right_proj_w_h)  # [B, N, N, hidden_dim] (fp16)

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden_dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid) in half precision
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w_h))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w_h))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction via Triton kernel
    # -----------------------------------------------------------------
    # Permute to (B, hidden, N, N) layout required by the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k before j)

    B, H, N, _ = left_perm.shape

    # Output buffer (float32)
    out_buf = torch.empty((B, H, N, N), dtype=torch.float32, device=x.device)

    # Strides (in elements, not bytes)
    stride_ah = left_perm.stride(0)
    stride_am = left_perm.stride(2)
    stride_ak = left_perm.stride(3)
    stride_bh = right_perm.stride(0)
    stride_bk = right_perm.stride(2)
    stride_bn = right_perm.stride(3)
    stride_ch = out_buf.stride(0)
    stride_cm = out_buf.stride(2)
    stride_cn = out_buf.stride(3)

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _tri_mul_gemm_kernel[grid](
        left_perm.data_ptr(),
        right_perm.data_ptr(),
        out_buf.data_ptr(),
        B, H, N,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Restore original layout: [B, N, N, hidden_dim]
    # -----------------------------------------------------------------
    out = out_buf.permute(0, 2, 3, 1).contiguous()  # float32

    # -----------------------------------------------------------------
    # 7. Output LayerNorm, output gate and final projection
    # -----------------------------------------------------------------
    # Output gate (computed in float32 from the original normed tensor)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B, N, N, hidden_dim]

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
