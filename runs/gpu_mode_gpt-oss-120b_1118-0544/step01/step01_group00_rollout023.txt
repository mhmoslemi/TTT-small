@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    batch: tl.constexpr,
    M: tl.constexpr,
    N: tl.constexpr,
    K: tl.constexpr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)  # Batch dim
    pid_m = tl.program_id(1)  # Row tile
    pid_n = tl.program_id(2)  # Col tile

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer offsets for base
    a_base = a_ptr + pid_b * stride_a_batch
    b_base = b_ptr + pid_b * stride_b_batch
    c_base = c_ptr + pid_b * stride_c_batch

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Initialize k-loop
    num_k = tl.cdiv(K, BLOCK_K)

    # Preload first tiles for double buffering
    k0 = 0
    k_offset = k0 * BLOCK_K + tl.arange(0, BLOCK_K)
    mask_k = k_offset < K
    a_ptrs = a_base + (offs_m[:, None] * stride_a_m) + (k_offset[None, :] * stride_a_k)
    a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
    b_ptrs = b_base + (k_offset[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
    b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

    # Loop over k tiles
    for kk in range(num_k):
        # compute next k offsets
        if kk + 1 < num_k:
            k_next_offset = (kk + 1) * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k_next = k_next_offset < K
            a_ptrs_next = a_base + (offs_m[:, None] * stride_a_m) + (k_next_offset[None, :] * stride_a_k)
            a_next = tl.load(a_ptrs_next, mask=mask_m[:, None] & mask_k_next[None, :], other=0.0)
            b_ptrs_next = b_base + (k_next_offset[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
            b_next = tl.load(b_ptrs_next, mask=mask_k_next[:, None] & mask_n[None, :], other=0.0)
        else:
            a_next = tl.zeros((BLOCK_M, BLOCK_K), dtype=tl.float16)
            b_next = tl.zeros((BLOCK_K, BLOCK_N), dtype=tl.float16)
        
        # compute dot for current tile
        acc += tl.dot(a, b).to(tl.float32)

        # shift next to current

"""
Outgoing Triangle Multiplicative Update (TriMul) kernel.

The heavy O(N³) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication C = A @ Bᵀ where each batch corresponds
to one hidden channel (hidden_dim).  This implementation
* performs the LayerNorms, linear projections and gates in PyTorch,
* casts the projected tensors to FP16,
* uses a highly‑tuned Triton kernel with double‑buffered K‑loop and
  configurable block sizes to compute the batched GEMM,
* accumulates in FP32 and stores the result in FP16,
* finally applies the output LayerNorm, gating and linear projection back
  to the original dimension.

The kernel runs with a 2‑D tile grid (M‑tiles × N‑tiles) and a 3‑rd axis
for the (batch × hidden_dim) dimension.  Block sizes 128×128×64 work well
for the target sequence lengths (256‑1024).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,                         # left  tensor pointer  (batch, M, K)
    b_ptr,                         # right tensor pointer (batch, N, K) – read as transposed
    c_ptr,                         # output pointer (batch, M, N)
    batch: tl.constexpr,           # total # of independent matrices = B * hidden_dim
    M: tl.constexpr,               # rows of A (also rows of output)
    N: tl.constexpr,               # rows of B (also cols of output)
    K: tl.constexpr,               # shared reduction dimension
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile column
    pid_b = tl.program_id(2)  # batch*hidden index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k = tl.cdiv(K, BLOCK_K)

    # ----------------------------------------------------------------
    # Double‑buffered K‑loop: preload first tiles
    # ----------------------------------------------------------------
    k0 = 0
    k_offset = k0 * BLOCK_K + tl.arange(0, BLOCK_K)
    mask_k = k_offset < K

    # A tile (M × K)
    a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (k_offset[None, :] * stride_a_k)
    a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

    # Bᵀ tile (K × N) – we load B as (K, N) by swapping its strides
    b_ptrs = b_batch + (k_offset[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
    b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

    # ----------------------------------------------------------------
    # Main reduction loop
    # ----------------------------------------------------------------
    for kk in range(num_k):
        # prefetch next K‑tile (if any)
        if kk + 1 < num_k:
            k_next = (kk + 1) * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k_next = k_next < K

            a_ptrs_next = a_batch + (offs_m[:, None] * stride_a_m) + (k_next[None, :] * stride_a_k)
            a_next = tl.load(a_ptrs_next, mask=mask_m[:, None] & mask_k_next[None, :], other=0.0).to(tl.float16)

            b_ptrs_next = b_batch + (k_next[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
            b_next = tl.load(b_ptrs_next, mask=mask_k_next[:, None] & mask_n[None, :], other=0.0).to(tl.float16)
        else:
            a_next = tl.zeros((BLOCK_M, BLOCK_K), dtype=tl.float16)
            b_next = tl.zeros((BLOCK_K, BLOCK_N), dtype=tl.float16)

        # multiply‑accumulate for the current tile (M×N)
        acc += tl.dot(a, b).to(tl.float32)

        # advance buffers
        a = a_next
        b = b_next

    # write the result (cast back to FP16 for storage)
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float) – may be ignored
            - weights: dict of model weights (see skeleton)
            - config: dict containing:
                * "dim" (int) – input feature dimension
                * "hidden_dim" (int) – projection dimension
                * "nomask" (bool) – if True the mask is not applied
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------------------------------------------------
    # unpack arguments
    # -------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------------------
    # unpack model weights
    # -------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -------------------------------------------------------------
    # 1) Input LayerNorm
    # -------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # -------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # -------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------------------
    # 4) Gating (sigmoid of linear projections)
    # -------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # 5) Cast to FP16 and reshape for the Triton GEMM
    # -------------------------------------------------------------
    B, N, _, _ = left.shape
    left_h = left.to(torch.float16).permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right.to(torch.float16).permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # free intermediate FP32 tensors early
    del left, right, left_gate, right_gate, x

    # -------------------------------------------------------------
    # 6) Allocate output buffer (FP16)
    # -------------------------------------------------------------
    out_h = torch.empty_like(left_h, dtype=torch.float16, device=device)

    # -------------------------------------------------------------
    # 7) Triton kernel launch configuration
    # -------------------------------------------------------------
    # Block sizes – tuned for the target hardware
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    batch_dim = left_h.shape[0]        # B * hidden_dim

    grid = (num_pid_m, num_pid_n, batch_dim)

    # Strides (in elements)
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)

    stride_b_batch = right_h.stride(0)
    stride_b_k = right_h.stride(2)   # column stride after view = 1
    stride_b_n = right_h.stride(1)   # row stride = N

    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    # -----------------------------------------------------------------
    # Launch the kernel
    # -----------------------------------------------------------------
    tri_mul_outgoing_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        batch=batch_dim,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------------------
    # 8) Reshape back to [B, N, N, hidden_dim] and convert to FP32
    # -------------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden_dim]

    # free the half‑precision buffers
    del left_h, right_h, out_h

    # -------------------------------------------------------------
    # 9) Final LayerNorm, gating and linear projection to original dim
    # -------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # element‑wise gate
    out = F.linear(out, to_out_weight)       # [B, N, N, dim]

    return out
```
