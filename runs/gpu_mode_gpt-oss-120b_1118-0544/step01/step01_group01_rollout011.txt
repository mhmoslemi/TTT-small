"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The core of TriMul is the 4‑D Einstein summation:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

We implement this reduction as a batched matrix multiplication
(left @ rightᵀ) using a fused Triton kernel.  The kernel processes
each (batch, hidden) slice independently (grid dimension Z) and tiles
the (i, j) output plane with BLOCK_M × BLOCK_N tiles.  The reduction
over k is tiled with BLOCK_K and accumulated in FP32 for numerical
stability while keeping the input tensors in FP16 for bandwidth.
All other operations (LayerNorm, linear projections, gating, final
projection) are performed with PyTorch.  The final result is returned
as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_l_m, stride_l_k,
    stride_r_k, stride_r_n,
    stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul: out = left @ rightᵀ  for each hidden slice."""
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col
    pid_z = tl.program_id(2)  # batch * hidden index

    # Compute start offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for the (batch, hidden) slice
    left_base  = left_ptr  + pid_z * (M * K)
    right_base = right_ptr + pid_z * (K * N)
    out_base   = out_ptr   + pid_z * (M * N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_tile * BLOCK_K
        k_range = tl.arange(0, BLOCK_K) + k_offset

        # Pointers for current K‑block
        a_ptrs = left_base + offs_m[:, None] * stride_l_m + k_range[None, :] * stride_l_k
        b_ptrs = right_base + k_range[:, None] * stride_r_k + offs_n[None, :] * stride_r_n

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (k_range[None, :] < K)
        b_mask = (k_range[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)  # fp16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)  # fp16

        # Accumulate in FP32
        acc += tl.dot(a, b)  # TL automatically up‑casts to FP32 when added

    # Store the result (cast back to fp16)
    c_ptrs = out_base + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor_or_None, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor   : torch.Tensor of shape [B, N, N] or None
        - weights       : dict with the following keys (torch.Tensor):
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config        : dict containing at least
            * "dim"        – input channel dimension
            * "hidden_dim" – projection dimension

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    B, N, _, _ = x.shape

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weights["norm.weight"], weights["norm.bias"])

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias) + optional mask
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    if mask is not None:
        # broadcast mask over the hidden dimension
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 3) Gating (sigmoid) and apply gates
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4) Batched matrix multiplication via Triton (left @ rightᵀ)
    # -----------------------------------------------------------------
    # Convert to half for the kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange to (B, H, N, N) for contiguous slices per hidden channel
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch and hidden dimensions: each slice is an independent GEMM
    left_mat = left.view(-1, N, N)   # (B*H, N, N)
    right_mat = right.view(-1, N, N)

    # Output placeholder
    out_mat = torch.empty_like(left_mat)  # fp16

    # Strides in elements (not bytes)
    stride_l_m = left_mat.stride(1)   # stride for M dimension (i)
    stride_l_k = left_mat.stride(2)   # stride for K dimension (k)

    stride_r_k = right_mat.stride(2)  # stride for K dimension (k) after transposition
    stride_r_n = right_mat.stride(1)  # stride for N dimension (j)

    stride_o_m = out_mat.stride(1)    # stride for M dimension (i)
    stride_o_n = out_mat.stride(2)    # stride for N dimension (j)

    # Tile sizes – chosen for good occupancy on NVIDIA H100/B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles along M (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles along N (j)
        B * hidden_dim,                 # one program per (batch, hidden)
    )

    _trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,                # M
        N,                # N
        N,                # K
        stride_l_m,
        stride_l_k,
        stride_r_k,
        stride_r_n,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim) and promote to fp32 for the rest
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # -----------------------------------------------------------------
    # 5) Output LayerNorm, gate, final linear projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate   # gating after norm

    out = F.linear(out_hidden, weights["to_out.weight"])   # [B,N,N,dim]
    return out
