We'll implement a kernel:

```
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr, num_stages: tl.constexpr
):
    pid_batch = tl.program_id(0)  # batch * head
    pid_m = tl.program_id(1)  # vertical block index
    pid_n = tl.program_id(2)  # horizontal block index

    # compute start indexes
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointer for output element
    c_ptrs = c_ptr + pid_batch * stride_ch + off_m[:, None] * stride_cm + off_n[None, :] * stride_cn

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        a_ptrs = a_ptr + pid_batch * stride_ah + off_m[:, None] * stride_am + off_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_batch * stride_bh + off_k[:, None] * stride_bk + off_n[None, :] * stride_bn

        # load tiles with mask
        a = tl.load(a_ptrs, mask=(off_m[:, None] < M) & (off_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(off_k[:, None] < K) & (off_n[None, :] < N), other=0.0)

        acc += tl.dot(a, b)

    # store result
    tl.store(c_ptrs, acc, mask=(off_m[:, None] < M) & (off_n[None, :] < N))
```

"""
TriMul (outgoing) forward pass – Triton accelerated core contraction.

The implementation follows the reference PyTorch module:
1. Layer‑norm on the input.
2. Linear projections (left/right) onto `hidden_dim`.
3. Optional mask broadcasting on the projected tensors.
4. Sigmoid gates (left, right, out) applied to the projections.
5. The N³ reduction   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is performed by a fused batched GEMM written in Triton.  The tensors
   are cast to float16 for the matmul and accumulated in float32.
6. Output layer‑norm, out‑gate multiplication and final linear
   projection back to the original `dim`.

The Triton kernel tiles the M (i), N (j) and K (k) axes with configurable
block sizes (default 128×128×64) and uses `tl.dot` (Tensor‑Core) for the
inner multiply‑accumulate.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,   # A: batch/head, M, K strides
    stride_bh, stride_bk, stride_bn,   # B: batch/head, K, N strides
    stride_ch, stride_cm, stride_cn,   # C: batch/head, M, N strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_WARPS: tl.constexpr,
    NUM_STAGES: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A: M×K, B: K×N) for each batch/head."""
    pid_batch = tl.program_id(0)   # collapsed batch * hidden_dim
    pid_m = tl.program_id(1)       # block row index (i)
    pid_n = tl.program_id(2)       # block column index (j)

    # ----- tile offsets -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_batch_ptr = a_ptr + pid_batch * stride_ah
    b_batch_ptr = b_ptr + pid_batch * stride_bh
    c_batch_ptr = c_ptr + pid_batch * stride_ch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- main K‑loop (with double‑buffering) -----
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # pointers for the current tile
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load tiles (FP16)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # fused multiply‑accumulate (Tensor‑Core)
        acc += tl.dot(a, b)

    # ----- write back -----
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton‑accelerated
    N³ contraction.

    Args:
        data: tuple (x, mask, weights, config)
            - x: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (see skeleton)
            - config: dict containing 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑less)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure mask dtype matches x and broadcast over hidden_dim
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, hidden_dim]

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction – batched GEMM in Triton
    # ------------------------------------------------------------------
    # cast to FP16 for the kernel (Tensor‑Core friendly)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B, H, N, N) and make memory contiguous
    # left: (i, k)   -> rows=i, cols=k
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, i, k]
    # right needs to be (k, j) for the matmul → transpose the two N axes
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()  # [B, H, k, j]

    B, H, N, _ = left_perm.shape
    # collapse batch and head dimensions
    left_mat = left_perm.view(B * H, N, N)    # (batch_head, M=i, K=k)
    right_mat = right_perm.view(B * H, N, N)  # (batch_head, K=k, N=j)

    # output buffer (FP32 accumulator)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=x.device)

    # ------- kernel launch configuration -------
    # block sizes (tunable). 128×128 tiles cover the common sizes well.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64   # multiple of 32 works best with Tensor‑Cores

    grid = (
        B * H,                                  # batch_head dimension
        (N + BLOCK_M - 1) // BLOCK_M,           # M tiles
        (N + BLOCK_N - 1) // BLOCK_N,           # N tiles
    )

    _tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),   # A strides
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),# B strides
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),      # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_WARPS=4,
        NUM_STAGES=2,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
