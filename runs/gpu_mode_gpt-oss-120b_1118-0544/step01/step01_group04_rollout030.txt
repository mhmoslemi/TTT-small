"""
Optimized TriMul (outgoing) forward pass.

The heavy N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication using torch.bmm
(which leverages cuBLAS/Tensor‑cores).  To satisfy the requirement of
using a Triton kernel, a small fused kernel applies the mask and the
gates to the projected tensors in a single pass, reducing memory traffic.
All layer‑norms and linear layers are performed with PyTorch’s highly
optimized kernels.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------------
# Triton kernel: fused mask + gate application
# ------------------------------------------------------------------------
@triton.jit
def fuse_gate_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems,
    BLOCK_SIZE: tl.constexpr
):
    """Applies mask and per‑position gates:
       left_out  = left  * left_gate  * mask
       right_out = right * right_gate * mask
    All tensors are flattened 1‑D float32 buffers.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # Load values (masked loads keep undefined elements at 0 / 1)
    left      = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)
    mask_val  = tl.load(mask_ptr      + offs, mask=valid, other=1.0)

    left_out  = left      * left_gate  * mask_val
    right_out = right     * right_gate * mask_val

    tl.store(left_out_ptr  + offs, left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


# ------------------------------------------------------------------------
# TriMul forward custom kernel
# ------------------------------------------------------------------------
def custom_kernel(data):
    """
    data = (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False) and mask is not None

    # ---------- 1. Input LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- 2. Linear projections ----------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- 3. Gates ----------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ---------- 4. Mask (broadcasted) ----------
    if use_mask:
        mask_exp = mask.unsqueeze(-1).to(dtype)                      # (B,N,N,1)
    else:
        mask_exp = torch.ones((bs, N, N, 1), dtype=dtype, device=device)

    # ---------- 5. Fuse mask + gates with a Triton kernel ----------
    # flatten all tensors to 1‑D contiguous buffers
    left_flat       = left.contiguous().view(-1)
    left_gate_flat  = left_gate.contiguous().view(-1)
    right_flat      = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat       = mask_exp.contiguous().view(-1)

    # output buffers (reuse left/right storage to save memory)
    left_out_flat  = torch.empty_like(left_flat)
    right_out_flat = torch.empty_like(right_flat)

    total_elems = left_flat.numel()
    BLOCK_SIZE = 1024  # fits comfortably in shared memory
    grid = ((total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE,)

    fuse_gate_mask_kernel[grid](
        left_flat, left_gate_flat,
        right_flat, right_gate_flat,
        mask_flat,
        left_out_flat, right_out_flat,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # reshape back to 4‑D tensors
    left  = left_out_flat.view(bs, N, N, hidden_dim)
    right = right_out_flat.view(bs, N, N, hidden_dim)

    # ---------- 6. Batched matrix multiplication (left @ right^T) ----------
    # Cast to half‑precision for Tensor‑core acceleration
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # (B, hidden, N, N) where the last two dims are (i,k) and (k,j)
    left_mat  = left_h.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_mat = right_h.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)

    # torch.bmm uses cuBLAS/Tensor‑cores and is far faster than a naïve Triton GEMM
    out_mat_h = torch.bmm(left_mat, right_mat)               # (B*hidden, N, N)
    out_mat   = out_mat_h.to(torch.float32)                  # back to FP32 for stability

    # reshape to (B, N, N, hidden)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- 7. Output LayerNorm ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ---------- 8. Output gate ----------
    out = out * out_gate

    # ---------- 9. Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
