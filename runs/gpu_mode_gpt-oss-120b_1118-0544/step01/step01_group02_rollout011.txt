@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_am, stride_ak, stride_ah,
                          stride_bk, stride_bn, stride_bh,
                          stride_cm, stride_cn, stride_ch,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0) # block id in M dimension
    pid_n = tl.program_id(1) # block id in N dimension
    pid_b = tl.program_id(2) # batch id
    # compute block start indices
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute mask for edges
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers with base stride per batch
    a_ptrs = A_ptr + pid_b * stride_ah + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + pid_b * stride_bh + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # mask for K
        cur_k = k * BLOCK_K
        k_offs = tl.arange(0, BLOCK_K) + cur_k
        k_mask = k_offs < K

        a = tl.load(a_ptrs, mask=mask_m[:, None] & k_mask[None, :], other=0.0, eviction_policy='evict_last')
        b = tl.load(b_ptrs, mask=k_mask[:, None] & mask_n[None, :], other=0.0, eviction_policy='evict_last')
        # cast to fp16 if needed
        # assume a,b already fp16
        acc += tl.dot(a, b).to(tl.float32)  

        # advance pointers for next k tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = C_ptr + pid_b * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

def batched_gemm_triton(A, B):
    assert A.shape == B.shape
    batch, M, K = A.shape
    N = M  # reduction shape N = M = seq len
    # allocate output
    C = torch.empty((batch, M, M), dtype=A.dtype, device=A.device)
    # heuristics for block sizes
    # Use 128*128 for M,N >= 256 else 64*64
    BLOCK_M = 128 if M > 256 else 64
    BLOCK_N = 128 if M > 256 else 64
    BLOCK_K = 32
    grid = ( (M + BLOCK_M - 1)//BLOCK_M,
             (M + BLOCK_N - 1)//BLOCK_N,
             batch )
    _batched_gemm_kernel[grid](
        A, B, C,
        M, M, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(2), B.stride(1),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # mask
    if mask is not None and (not nomask):
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f
    # gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate
    # reshape left/right to batch*hidden_dim
    B,N,_,_ = left.shape
    # reorder to B,H,N,N
    left = left.permute(0,3,1,2).contiguous()
    right = right.permute(0,3,1,2).contiguous()

    right_flat = right.view(B*hidden_dim, N, N)

    right_fp16 = right_flat.to(torch.float16)

"""
TriMul outgoing update (forward only) – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑norm the input tensor of shape [B, N, N, C].
2. Linear projections (left/right) and gating (left_gate, right_gate, out_gate)
   are computed with the provided weights.  An optional mask (if present and
   ``nomask`` is False) is applied before the gates.
3. The hidden dimension H (``hidden_dim``) is moved into the batch axis:
   tensors become (B·H, N, N).  They are cast to float16 to exploit Tensor
   Cores.
4. A custom Triton kernel computes, for each hidden slice,
   ``out[b, i, j] = Σ_k left[b, i, k] * right[b, j, k]``
   i.e. a batched matrix multiplication with the second operand implicitly
   transposed.  The kernel tiles the M, N dimensions (BLOCK_M×BLOCK_N) and
   the reduction K dimension (BLOCK_K), loads tiles in FP16 and accumulates
   in FP32.
5. The result is cast back to float32, reshaped to [B, N, N, H] and
   processed by a second LayerNorm, the output gate and a final linear
   projection back to the original channel dimension C.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,                           # matrix dimensions
    stride_ah, stride_am, stride_ak,   # A strides: batch, row, col
    stride_bh, stride_bk, stride_bn,   # B strides: batch, col (K), row (N) – B is accessed as (k, n)
    stride_ch, stride_cm, stride_cn,   # C strides: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]  (B is treated as transposed)"""
    pid_b = tl.program_id(axis=2)    # batch*hidden dimension
    pid_m = tl.program_id(axis=0)    # tile row (i)
    pid_n = tl.program_id(axis=1)    # tile col (j)

    # Tile start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch slice
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # Pointers to the current (M×K) and (K×N) tiles
    a_ptrs = A_ptr + a_batch + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        # K‑axis validity mask
        k_valid = (k * BLOCK_K + tl.arange(0, BLOCK_K)) < K

        # Load A[i,k] and B[k,j] tiles (A/B are FP16)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_n[None, :],
                    other=0.0)

        # Tile‑wise dot product, accumulate in FP32
        acc += tl.dot(a, b, out_dtype=tl.float32)

        # Advance to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the result
    c_ptrs = C_ptr + c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A, B: (batch, M, K) with identical shapes.  Returns C: (batch, M, M).
    The reduction is performed over the last dimension (K).
    """
    assert A.ndim == 3 and B.ndim == 3
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2
    assert M == N, "TriMul expects square M=N (seq_len)."

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Heuristic block sizes – larger tiles for larger sequences
    if M <= 256:
        BLOCK_M, BLOCK_N = 64, 64
    else:
        BLOCK_M, BLOCK_N = 128, 128
    BLOCK_K = 32 if K >= 64 else 16

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    _batched_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(2), B.stride(1),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float) or None
            - weights: dict of model weights (LayerNorm, Linear weight tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (left/right) – no bias as per model definition
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask application (before gating)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Reshape for the batched GEMM (move hidden dim into batch)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left.view(B * hidden_dim, N, N)    # (B*H, N, N)
    right_flat = right.view(B * hidden_dim, N, N)  # (B*H, N, N)

    # Cast to FP16 for the Triton kernel (Tensor Cores)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # -----------------------------------------------------------------
    # 6. Core O(N³·H) reduction using Triton
    # -----------------------------------------------------------------
    out_fp16 = _batched_gemm_triton(left_fp16, right_fp16)   # (B*H, N, N)

    # Back to FP32
    out = out_fp16.to(torch.float32)

    # -----------------------------------------------------------------
    # 7. Reshape back to [B, N, N, H]
    # -----------------------------------------------------------------
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # -----------------------------------------------------------------
    # 8. Output LayerNorm, gating, and final projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
