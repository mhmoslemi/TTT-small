import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(a_ptr, b_ptr, c_ptr,
                   M, N, K,
                   stride_a_batch, stride_a_row, stride_a_col,
                   stride_b_batch, stride_b_row, stride_b_col,
                   stride_c_batch, stride_c_row, stride_c_col,
                   BLOCK_M: tl.constexpr,
                   BLOCK_N: tl.constexpr,
                   BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the current batch (hidden) matrix
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile transposed (K x N)
        b_ptrs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # matrix multiply-accumulate
        acc += tl.dot(a, b)

    # Write results
    c_ptrs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.
    Implements the heavy O(N³) contraction `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
    as a batched half‑precision GEMM (A @ Bᵀ) where each hidden dimension is treated as a
    separate batch entry. All other operations (LayerNorm, linear projections,
    gating, final LayerNorm and linear) are performed with PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm ----------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    # ---------- Mask ----------
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ---------- Apply gates ----------
    left = (left_proj * left_gate).to(torch.float16)
    right = (right_proj * right_gate).to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # ----------- Prepare tensors for the Triton GEMM -----------
    # [B, hidden_dim, N, N] layout for batched GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous()
    right_mat = right.permute(0, 3, 1, 2).contiguous()
    # flatten batch*hidden_dim dimension
    left_mat = left_mat.view(B * hidden_dim, N, N)
    right_mat = right_mat.view(B * hidden_dim, N, N)

    # Output buffer (float32 accumulator)
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # Strides (in elements)
    stride_a_batch, stride_a_row, stride_a_col = left_mat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_mat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_mat.stride()

    # Tile sizes – choose larger tiles for better throughput
    # Use 128×128 tiles when N is at least 256, otherwise fall back to 64×64.
    if N >= 256:
        BLOCK_M = 128
        BLOCK_N = 128
    else:
        BLOCK_M = 64
        BLOCK_N = 64
    BLOCK_K = 64  # keep K tile moderate to fit registers

    grid = (
        B * hidden_dim,                 # batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,   # rows tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # cols tiles
    )

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr,
    b_ptr,
    c_ptr,
    M,
    N,
    K,
    stride_a_batch,
    stride_a_row,
    stride_a_col,
    stride_b_batch,
    stride_b_row,
    stride_b_col,
    stride_c_batch,
    stride_c_row,
    stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM kernel computing C = A @ Bᵀ.
    A : (batch, M, K)
    B : (batch, N, K) – accessed transposed on‑the‑fly
    C : (batch, M, N)
    """
    pid_batch = tl.program_id(0)  # which matrix in the batch (B * hidden)
    pid_m = tl.program_id(1)      # tile row index
    pid_n = tl.program_id(2)      # tile col index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointer for this matrix
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_offs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(
            a_offs, mask=mask_m[:, None] & mask_k[None, :], other=0.0
        )  # fp16

        # Bᵀ tile: (BLOCK_K, BLOCK_N) from B (N, K)
        b_offs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(
            b_offs, mask=mask_k[:, None] & mask_n[None, :], other=0.0
        )  # fp16

        # accumulate
        acc += tl.dot(a, b)

    # store result
    c_offs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_offs, acc, mask=mask_m[:, None] & mask_n[:, None])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” variant, with the cubic contraction
    performed by a Triton kernel.

    Steps
    -------
    1. LayerNorm over the last channel dimension.
    2. Linear projections to `hidden_dim` (left/right) and to the three gates.
    3. Optional mask multiplication.
    4. Apply sigmoid gates to the projected tensors.
    5. Fuse the O(N³) contraction into a batched GEMM:
           out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
       implemented as `left_mat @ right_matᵀ` for each hidden slice.
    6. LayerNorm over the hidden dimension, multiply by the output gate,
       and a final linear projection back to `dim`.

    All heavy matrix‑multiply work is done in half‑precision with fp32
    accumulation for speed and numerical stability.
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # -----------------------------------------------------------------
    # 4. Gates (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 5. Apply gates and cast to half‑precision for the kernel
    # -----------------------------------------------------------------
    left = (left_proj * left_gate).to(torch.float16)
    right = (right_proj * right_gate).to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # -----------------------------------------------------------------
    # 6. Prepare tensors for batched GEMM
    # -----------------------------------------------------------------
    # Convert from [B,N,N,hidden] -> [B, hidden, N, N] and flatten
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (fp32 accumulator)
    out_mat = torch.empty(
        (B * hidden_dim, N, N), device=device, dtype=torch.float32
    )

    # strides in elements (not bytes)
    stride_a_batch, stride_a_row, stride_a_col = left_mat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_mat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_mat.stride()

    # -----------------------------------------------------------------
    # 7. Kernel launch – choose tile size based on sequence length
    # -----------------------------------------------------------------
    if N >= 256:
        BLOCK_M = 128
        BLOCK_N = 128
    else:
        BLOCK_M = 64
        BLOCK_N = 64
    BLOCK_K = 64  # moderate K‑tile for good register usage

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # -----------------------------------------------------------------
    # 8. Reshape back to [B, N, N, hidden]
    # -----------------------------------------------------------------
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 9. Output LayerNorm, gate, final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
