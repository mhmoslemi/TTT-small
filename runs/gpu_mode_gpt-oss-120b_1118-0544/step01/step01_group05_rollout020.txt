def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # Phase 1: Normalize input
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"]) # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Mask
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1) # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"])) # same shape
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Apply gating to left/right
    left = left * left_gate
    right = right * right_gate

    # Convert halves
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # reshape to (B*hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Perform batched GEMM: left_mat @ right_mat.transpose(1,2)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # shape (B*hidden_dim, N, N), dtype float16? Actually torch.bmm with half returns half? but accumulation default to half, but we may convert to float32 after.
    out_mat = out_mat.to(torch.float32)

    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,hidden_dim]

    # Second layernorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(x_ptr, y_ptr, out_ptr, n_elements, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    y = tl.load(y_ptr + offs, mask=mask, other=0.0)
    prod = x * y
    tl.store(out_ptr + offs, prod, mask=mask)

def apply_gate_triton(tensor, gate):
    # element‑wise multiply tensor *= gate using Triton (in‑place)
    assert tensor.shape == gate.shape
    tensor = tensor.contiguous()
    gate = gate.contiguous()
    n = tensor.numel()
    BLOCK = 1024
    grid = ( (n + BLOCK - 1) // BLOCK, )
    elementwise_mul_kernel[grid](
        tensor,
        gate,
        tensor,
        n,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return tensor

def custom_kernel(data):
    """
    Triton‑accelerated Triangle Multiplicative Update (outgoing variant).
    Steps:
      1. Layer‑norm + linear projections & gating (PyTorch, float32).
      2. Cast projected tensors to fp16 and perform a batched GEMM
         C = A @ Bᵀ for each hidden‑dim slice using torch.bmm (highly‑optimized cuBLAS).
      3. Second Layer‑norm (float32) and out‑gate application via a tiny Triton kernel.
      4. Final linear projection back to `dim`.
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1️⃣ LayerNorm on the input
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # 2️⃣ Linear projections (float32)
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # 3️⃣ Optional mask
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # 4️⃣ Gating (float32)
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # 5️⃣ Cast to fp16 for the heavy O(N³) contraction
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    B, N, _, _ = left_f16.shape  # B, seq_len, seq_len, hidden_dim

    # Reshape to (B*hidden_dim, N, N) – each slice will be multiplied with its transpose
    left_mat = left_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # 6️⃣ Batched GEMM: C = A @ Bᵀ  (float16 accumulation, then convert to float32)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out_mat = out_mat.to(torch.float32)

    # 7️⃣ Restore original 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # 8️⃣ Second LayerNorm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # 9️⃣ Apply the output gate using a tiny Triton kernel (in‑place)
    out = apply_gate_triton(out, out_gate)

    # 🔟 Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Tiny Triton kernel: element‑wise multiplication (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _elementwise_mul_kernel(x_ptr, y_ptr, out_ptr, n_elements,
                            BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    y = tl.load(y_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, x * y, mask=mask)


def _apply_gate_triton(tensor: torch.Tensor, gate: torch.Tensor) -> torch.Tensor:
    """
    In‑place element‑wise multiplication `tensor *= gate` using Triton.
    Both tensors must have the same shape and be contiguous on the same device.
    """
    assert tensor.shape == gate.shape
    tensor = tensor.contiguous()
    gate = gate.contiguous()
    n = tensor.numel()
    BLOCK = 1024                     # max threads per block on B200
    grid = ((n + BLOCK - 1) // BLOCK,)
    _elementwise_mul_kernel[grid](
        tensor,
        gate,
        tensor,
        n,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return tensor
