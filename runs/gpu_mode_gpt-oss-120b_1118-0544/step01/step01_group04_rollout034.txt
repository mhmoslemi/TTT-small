def custom_kernel(data):
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False) and mask is not None

    # normalized input
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # Linear projections
    left_raw = F.linear(x, weights["left_proj.weight"])          # shape [B,N,N,H]
    right_raw = F.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left_raw = left_raw.to(torch.float16)
    right_raw = right_raw.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    # Apply gates (and mask) using Triton kernel when mask present
    if use_mask:

        left_raw = left_raw.contiguous()
        right_raw = right_raw.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask = mask.to(torch.float16).contiguous()

        left = torch.empty_like(left_raw)
        right = torch.empty_like(right_raw)

        total_elems = left_raw.numel()
        BLOCK = 131072  # maybe 8192? Keep dynamic

        def fuse_kernel(
            left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
            mask_ptr,
            out_left_ptr, out_right_ptr,
            total_elements, hidden_dim,

            a = tl.load(left_ptr + offs, mask=mask, other=0.0)

"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections produce `left` and `right` tensors of shape
   [B, N, N, H] (H = hidden_dim) together with three gating tensors.
3. Fuse element‑wise multiplication of `left * left_gate` and
   `right * right_gate` (and optional mask) in a Triton kernel.
   The kernel works on flattened contiguous FP16 tensors and
   broadcasts the 3‑D mask across the hidden dimension.
4. Perform the heavy contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched matrix multiplication using `torch.bmm` on FP16 data
   (which maps to Tensor‑cores).  This is equivalent to
   `out = left @ right.transpose(-2,-1)`.
5. Apply a second LayerNorm, the output gate, and the final linear
   projection back to the original dimension.

Only the gating+mask step is implemented in Triton; the dominant
contraction uses cuBLAS via `torch.bmm`, giving a large speed‑up over
the previous pure‑Triton implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse gating and optional mask.
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_fuse_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr, mask_ptr,
    out_left_ptr, out_right_ptr,
    total_elements, hidden_dim,
    BLOCK_SIZE: tl.constexpr
):
    """
    Flattened tensors are processed element‑wise:
        out_left  = left      * left_gate  * mask
        out_right = right     * right_gate * mask
    `mask` has shape [B, N, N] and is broadcast over the hidden
    dimension; its index is obtained by integer division
    `offset // hidden_dim`.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load values (FP16)
    a  = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    ag = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    b  = tl.load(right_ptr      + offs, mask=mask, other=0.0)
    bg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Mask index (broadcast over hidden dimension)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Compute fused results
    out_l = a * ag * m
    out_r = b * bg * m

    # Store results
    tl.store(out_left_ptr  + offs, out_l, mask=mask)
    tl.store(out_right_ptr + offs, out_r, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    """
    input_tensor, mask, weights, config = data

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = (not config.get("nomask", False)) and (mask is not None)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left_raw = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right_raw = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # Work in FP16 for the heavy tensors
    left_raw = left_raw.to(torch.float16)
    right_raw = right_raw.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    if use_mask:
        # ----------------------------------------------------------
        # 4. Fuse gating and mask with a Triton kernel
        # ----------------------------------------------------------
        mask = mask.to(torch.float16).contiguous()
        # Ensure contiguous layout before flattening
        left_raw = left_raw.contiguous()
        right_raw = right_raw.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask = mask.contiguous()

        # Flatten tensors to 1‑D views (contiguous memory stride = 1)
        left_flat = left_raw.view(-1)
        right_flat = right_raw.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask.view(-1)   # shape B*N*N

        out_left_flat = torch.empty_like(left_flat)
        out_right_flat = torch.empty_like(right_flat)

        total_elems = left_flat.numel()
        BLOCK_SIZE = 8192  # reasonable block size for the kernel
        num_blocks = (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE

        # Launch the Triton kernel
        gate_mask_fuse_kernel[(num_blocks,)](
            left_flat, left_gate_flat,
            right_flat, right_gate_flat,
            mask_flat,
            out_left_flat, out_right_flat,
            total_elems, hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
        )

        # Reshape back to [B, N, N, H]
        left = out_left_flat.view(bs, N, N, hidden_dim)
        right = out_right_flat.view(bs, N, N, hidden_dim)
    else:
        # No mask – simple element‑wise multiplication
        left = left_raw * left_gate
        right = right_raw * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction via batched GEMM (torch.bmm on FP16)
    # --------------------------------------------------------------
    # (B, N, N, H) -> (B*H, N, N) and transpose right for matmul
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    # Batched matrix multiply: C = A @ B^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(-2, -1))   # [B*H, N, N]

    # Reshape back to [B, N, N, H]
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # LayerNorm works in FP32

    # --------------------------------------------------------------
    # 6. Output LayerNorm over hidden_dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 7. Output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 8. Final linear projection back to original dimension
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
