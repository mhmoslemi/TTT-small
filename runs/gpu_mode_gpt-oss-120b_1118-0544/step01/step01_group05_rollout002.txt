```
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # Compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to current batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # load A block
        a_ptrs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B^T block
        b_ptrs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b, out_dtype=tl.float32)

    # store acc
    c_ptrs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bsz, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dim = config["dim"]   # Though dim from shape may match.
    nomask = config.get("nomask", False)

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if any
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # broadcast to dim
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to (B*hidden_dim, N, N)
    left_mat = left_h.permute(0, 3, 1, 2).contiguous().view(bsz * hidden_dim, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(bsz * hidden_dim, N, N)

    # Output placeholder
    out_mat = torch.empty((bsz * hidden_dim, N, N), device=device, dtype=torch.float32)

    # Determine block sizes automatically based on N maybe
    BLOCK_M = 128 if N >= 256 else 64
    BLOCK_N = 128 if N >= 256 else 64
    BLOCK_K = 32 if N >= 256 else 64

    # Strides
    stride_a_batch = N * N
    stride_a_row = N
    stride_a_col = 1

    stride_b_batch = N * N
    stride_b_row = N
    stride_b_col = 1

    stride_c_batch = N * N
    stride_c_row = N
    stride_c_col = 1

    grid = (
        bsz * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(bsz, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply out_gate
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) optimized with Triton.

The core contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is implemented as a batch of matrix multiplications
    left_slice @ right_slice.T
for each (batch, hidden) slice.  The kernel works on FP16 inputs
and accumulates in FP32, then the result is post‑processed with
LayerNorm, gating and a final linear projection (all done in PyTorch).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ

    A, B: (batch, M, K) and (batch, N, K)  (row‑major)
    C:   (batch, M, N)
    """
    pid_batch = tl.program_id(0)      # batch * hidden index
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A block (BLOCK_M, BLOCK_K)
        a_offs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_offs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Bᵀ block (BLOCK_K, BLOCK_N) – load B transposed on‑the‑fly
        b_offs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_offs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot product, accumulate in FP32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # store the result
    c_offs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_offs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (already on the same device)
            - config: dict containing at least "dim" and "hidden_dim",
                      optionally "nomask" (bool)
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # LayerNorm on the input
    # ------------------------------------------------------------------ #
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------ #
    # Linear projections (no bias) to hidden dimension
    # ------------------------------------------------------------------ #
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------ #
    # Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------ #
    if (not nomask) and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)           # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------ #
    # Gating (sigmoid) – same shape as left/right: [B, N, N, hidden_dim]
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ------------------------------------------------------------------ #
    # Apply gates
    # ------------------------------------------------------------------ #
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # Prepare tensors for the Triton GEMM (FP16 inputs, FP32 accumulator)
    # ------------------------------------------------------------------ #
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*hidden, N, N)   (batch*hidden acts as the batch dimension)
    left_mat = left_h.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    # output buffer (FP32)
    out_mat = torch.empty((bs * hidden_dim, N, N), device=device, dtype=torch.float32)

    # ------------------------------------------------------------------ #
    # Kernel launch configuration
    # ------------------------------------------------------------------ #
    # block sizes – chosen to give good occupancy for the target N range
    BLOCK_M = 128 if N >= 256 else 64
    BLOCK_N = 128 if N >= 256 else 64
    BLOCK_K = 64 if N >= 256 else 32

    stride_a_batch = N * N
    stride_a_row = N
    stride_a_col = 1

    stride_b_batch = N * N
    stride_b_row = N
    stride_b_col = 1

    stride_c_batch = N * N
    stride_c_row = N
    stride_c_col = 1

    grid = (
        bs * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------ #
    # Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------ #
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
