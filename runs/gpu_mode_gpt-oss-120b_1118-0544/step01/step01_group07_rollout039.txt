@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bn, stride_bk, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # for M tiles
    pid_n = tl.program_id(1)  # for N tiles
    pid_b = tl.program_id(2)  # batch

    # compute offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # pointer offsets
        a_ptrs = A_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = B_ptr + pid_b * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=(mask_m[:, None] & (offs_k[None, :] < K)), other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=((offs_k[:, None] < K) & mask_n[None, :]), other=0.0, dtype=tl.float16)

        acc += tl.dot(a, b)

    # store
    c_ptrs = C_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)

    right_gate_h = right_gate.to(torch.float16)

        right = right * mask_h

    right = right * right_gate_h

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton implementation.

The kernel performs the core O(N³) operation:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication of shape
    (batch*hidden, N, N) @ (batch*hidden, N, N)^T.
All linear projections, layer‑norms and gating are done in PyTorch,
while the heavy N³ reduction is executed by a custom Triton kernel
using half‑precision inputs and FP32 accumulation for speed.
The final output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ   (A,B : [B, M, K] , B stored as [B, N, K])
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr,                     # *float16   [batch, M, K]
    B_ptr,                     # *float16   [batch, N, K] (interpreted as K×N)
    C_ptr,                     # *float32   [batch, M, N]
    M, N, K,                   # i32 dimensions
    stride_a_batch, stride_a_m, stride_a_k,   # strides for A (in elements)
    stride_b_batch, stride_b_n, stride_b_k,   # strides for B (in elements)
    stride_c_batch, stride_c_m, stride_c_n,   # strides for C (in elements)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(2)      # batch dimension
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col

    # ----- offsets for the output tile -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over K dimension -----
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # masks for the current tile
        mask_a = mask_m[:, None] & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & mask_n[None, :]

        # pointers to the current tiles of A and B (B is read as K×N)
        a_ptrs = (
            A_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            B_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k      # K dimension (row of Bᵀ)
            + offs_n[None, :] * stride_b_n      # N dimension (col of Bᵀ)
        )

        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16)

        # accumulate in FP32 (tensor‑core friendly)
        acc += tl.dot(a, b, dtype=tl.float32)

    # ----- store the result tile -----
    c_ptrs = (
        C_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)


def batched_matmul_triton(A_fp16: torch.Tensor, B_fp16: torch.Tensor) -> torch.Tensor:
    """
    Batched matmul using the Triton kernel defined above.
    A_fp16 : [batch, M, K]   (float16, contiguous)
    B_fp16 : [batch, N, K]   (float16, contiguous) – interpreted as K×N
    Returns C : [batch, M, N] (float32)
    """
    assert A_fp16.is_contiguous() and B_fp16.is_contiguous()
    batch, M, K = A_fp16.shape
    _, N, K2 = B_fp16.shape
    assert K == K2, "Inner dimensions must match."

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A_fp16.device)

    # strides are in *elements* (not bytes)
    stride_a_batch, stride_a_m, stride_a_k = A_fp16.stride()
    stride_b_batch, stride_b_n, stride_b_k = B_fp16.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tile sizes – chosen for good occupancy on B200 (FP16 tensor cores)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch                      # batch
    )

    batched_matmul_kernel[grid](
        A_fp16,
        B_fp16,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    return C


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N]  (can be all‑ones)
        - weights      : dict of pre‑loaded model parameters
        - config       : dict with "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Compute gates (float32)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3. Linear projections (float32 → cast to fp16 for the heavy part)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Apply mask (if present) – also in fp16
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5. Apply gating (cast gates to fp16 first)
    # ------------------------------------------------------------------
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)
    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 6. Core TriMul – batched matmul A @ Bᵀ using Triton
    # ------------------------------------------------------------------
    Bsz, N, _, H = left.shape  # Bsz = batch size, H = hidden_dim
    # reshape to [batch*hidden, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    BH = Bsz * H
    left_flat = left.view(BH, N, N)    # [BH, N, N]
    right_flat = right.view(BH, N, N)  # [BH, N, N]

    # Triton matmul (FP16 inputs, FP32 accumulator)
    out_fp32 = batched_matmul_triton(left_flat, right_flat)  # [BH, N, N] (float32)

    # reshape back to [B, N, N, hidden]
    out = out_fp32.view(Bsz, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm + gating (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    output = F.linear(out, weights["to_out.weight"])  # (dim, hidden) weight

    return output
