@triton.jit
def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    N,  # seq_len
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # row block
    pid_n = tl.program_id(1) # col block
    pid_batch = tl.program_id(2)

    # offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        mask_a = mask_m[:, None] & (offs_k[None, :] < N)
        mask_b = (offs_k[:, None] < N) & mask_n[None, :]

        a_ptrs = A_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = B_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # a,b are float16? keep loading as float16 but compute as float32
        acc += tl.dot(a, b).to(tl.float32)

    c_ptrs = C_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n

    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16) if C_ptr.dtype == tl.float16 else acc, mask=mask_c)

"""
TriMul (outgoing) forward pass implemented with Triton.

Algorithm
---------
1. Layer‑norm the input tensor `x` of shape [B,N,N,dim].
2. Linear projections `left_proj` and `right_proj` (no bias) → shape [...,hidden_dim].
3. Optional mask multiplication (broadcasted over the hidden dimension).
4. Gating: three sigmoid‑gated linear layers (left_gate, right_gate, out_gate).
   Apply `left_gate` and `right_gate` element‑wise to the projected tensors.
5. Core O(N³) update:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication A @ Bᵀ where A and B have shape
   (B*hidden_dim, N, N).  A custom Triton kernel computes this efficiently.
6. Layer‑norm on the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.

Only the heavy matrix multiplication is performed in Triton; all other
operations use PyTorch (they are memory bound and cheap compared to the
O(N³) compute).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    A_ptr,                # *float16  [batch, M, K]
    B_ptr,                # *float16  [batch, N, K] (read as K×N)
    C_ptr,                # *float32  [batch, M, N]
    N,                    # i32      sequence length (M == N == K)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul C = A @ Bᵀ  (A,B: [B, N, N])"""
    pid_m = tl.program_id(0)            # tile row
    pid_n = tl.program_id(1)            # tile col
    pid_b = tl.program_id(2)            # batch * hidden

    # Offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Masks for the current K‑tile
        mask_a = mask_m[:, None] & (offs_k[None, :] < N)
        mask_b = (offs_k[:, None] < N) & mask_n[None, :]

        # Pointers to the current tile of A and Bᵀ
        a_ptrs = (
            A_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            B_ptr
            + pid_b * stride_b_batch
            + offs_k[:, None] * stride_b_k   # K dimension (row of Bᵀ)
            + offs_n[None, :] * stride_b_n   # N dimension (col of Bᵀ)
        )

        # Load tiles (half‑precision)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # Accumulate (dot returns fp16, cast to fp32)
        acc += tl.dot(a, b).to(tl.float32)

    # Store the result tile
    c_ptrs = (
        C_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for tensors A,B ∈ [BH, N, N] using Triton.
    A,B must be contiguous, dtype torch.float16.
    Returns C of dtype torch.float32.
    """
    assert A.is_contiguous() and B.is_contiguous()
    BH, N, K = A.shape
    _, N2, K2 = B.shape
    assert N == N2 and K == K2, "A and B must be square and have matching inner dimension"

    C = torch.empty((BH, N, N), dtype=torch.float32, device=A.device)

    # Strides in elements (not bytes)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_n, stride_b_k = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tunable tile sizes – good occupancy on NVIDIA B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        BH,                        # batch * hidden
    )

    _tri_mul_kernel[grid](
        A,
        B,
        C,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,   # more warps for better utilization
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of pre‑loaded model parameters
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Extract weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) multiplication via Triton
    # ------------------------------------------------------------------
    # Cast to half‐precision for the heavy kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to [B, hidden, N, N] and make contiguous
    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    B, H = left_h.shape[0], left_h.shape[1]
    N = left_h.shape[2]

    # Collapse batch and hidden into a single dimension for the kernel
    left_mat = left_h.view(B * H, N, N)   # [BH, N, N]
    right_mat = right_h.view(B * H, N, N) # [BH, N, N]

    # Triton batched matmul (produces FP32 output)
    out_mat = _batched_matmul_triton(left_mat, right_mat)   # [BH, N, N]

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    # Final linear projection back to ‘dim’
    output = F.linear(out, to_out_weight)   # (dim, hidden_dim) weight

    return output
