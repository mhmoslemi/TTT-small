"""TriMul (outgoing) forward pass with a fused Triton batched GEMM.

The core O(N³) operation is:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
which is a batch of independent matrix multiplications
    out_h = left_h @ right_hᵀ   for each hidden channel h.

We implement this reduction with a high‑performance Triton kernel that
processes each (batch, hidden) slice as a separate GEMM.  The surrounding
layer‑norms, gating, masking and final projection are performed with
PyTorch.  All heavy arithmetic is done in float16; the result is cast back
to float32 for the subsequent LayerNorm and output linear layer.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A, B, C,                 # pointers
    M, N, K,                 # matrix sizes
    stride_am, stride_ak,   # strides for A (M, K)
    stride_bk, stride_bn,   # strides for B (K, N) – note swapped order for transpose
    stride_cm, stride_cn,   # strides for C (M, N)
    stride_ab0, stride_bb0, stride_cb0,  # stride to next batch*hidden slice
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b] = A[b] @ B[b].T   (A,B,C are half‑precision)."""
    pid_m = tl.program_id(0)            # tile row
    pid_n = tl.program_id(1)            # tile col
    pid_b = tl.program_id(2)            # batch * hidden

    # Compute offsets within the tile.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this (batch, hidden) slice.
    a_base = A + pid_b * stride_ab0
    b_base = B + pid_b * stride_bb0
    c_base = C + pid_b * stride_cb0

    # Accumulator in FP32 for numerical stability.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Main K‑reduction loop.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K) + k_offset

        # Pointers to the current K‑block of A and B.
        a_ptrs = a_base \
                 + offs_m[:, None] * stride_am \
                 + k_range[None, :] * stride_ak
        b_ptrs = b_base \
                 + k_range[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # Load with boundary checks.
        a_mask = (offs_m[:, None] < M) & (k_range[None, :] < K)
        b_mask = (k_range[:, None] < K) & (offs_n[None, :] < N)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Compute partial product.
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Write back the result.
    c_ptrs = c_base \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using Triton for the N³ reduction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors for all learned parameters
        - config       : dict containing at least "dim", "hidden_dim", and "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), norm_w, norm_b)

    # --------------------------------------------------------------
    # 2) Linear projections and gating
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Linear maps (float32)
    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # Apply mask if provided and not disabled.
    if (mask is not None) and (not nomask):
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # Gates (float32)
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply gates (still in float32)
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 3) Batched matrix multiplication in FP16 with Triton
    # --------------------------------------------------------------
    # Cast to half for the kernel.
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to [B, H, N, N] so each (b, h) slice is a contiguous matrix.
    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    # Collapse batch and hidden dimensions.
    left_mat  = left_h.view(-1, N, N)      # (B*H, N, N)
    right_mat = right_h.view(-1, N, N)
    out_mat = torch.empty_like(left_mat)   # (B*H, N, N), half precision

    # Strides in elements (not bytes).
    stride_am = left_mat.stride(1)   # stride over i (M)
    stride_ak = left_mat.stride(2)   # stride over k (K)

    # Right matrix is interpreted as transposed inside the kernel.
    stride_bk = right_mat.stride(2)  # row‑stride for k
    stride_bn = right_mat.stride(1)  # column‑stride for j

    stride_cm = out_mat.stride(1)   # M stride
    stride_cn = out_mat.stride(2)   # N stride

    stride_ab0 = left_mat.stride(0)   # batch*hidden stride (M*K)
    stride_bb0 = right_mat.stride(0)
    stride_cb0 = out_mat.stride(0)

    # Tile sizes – tuned for A100/B200 (shared memory not used, registers only).
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ab0, stride_bb0, stride_cb0,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # typical for FP16 GEMM
    )

    # Reshape back to (B, N, N, hidden_dim)
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)   # LayerNorm expects fp32

    # --------------------------------------------------------------
    # 4) Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_hidden = F.layer_norm(out_hidden, (hidden_dim,), to_out_norm_w, to_out_norm_b)

    # Apply the output gate (broadcasted over the hidden dimension)
    out_hidden = out_hidden * out_gate

    # Final linear projection back to dim
    to_out_w = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(out_hidden, to_out_w)  # [B, N, N, dim]

    return out
