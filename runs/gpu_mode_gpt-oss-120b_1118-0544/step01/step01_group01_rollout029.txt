"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy part of TriMul is the 3‑way reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which can be seen as a batched matrix multiplication of N×N matrices for each
(hidden) channel.  The implementation below avoids any large tensor transposes;
instead the kernel works directly on the original layout
    (B, N, N, H)  with strides (b,i,j,h).
The kernel tiles the (i,j) plane (BLOCK_M×BLOCK_N) and accumulates over k
in BLOCK_K‑sized chunks using `tl.dot`, letting the GPU Tensor‑Cores perform the
inner product in FP16 while accumulating in FP32.  The rest of the TriMul
operations (LayerNorms, linear projections, gating, mask) are handled with
standard PyTorch kernels.  This design removes the costly permute+copy of the
previous version and brings the runtime well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matmul for the TriMul outgoing update.

    Computes for each (batch, hidden) slice:
        out[i, j] = Σ_k left[i, k] * right[j, k]   (i,j,k ∈ [0, N))
    where left and right have shape (N, N) and are stored in the original
    (batch, i, k, hidden) / (batch, j, k, hidden) layout.
    """
    pid_m = tl.program_id(0)  # i‑tile
    pid_n = tl.program_id(1)  # j‑tile
    pid_bh = tl.program_id(2)  # combined batch*hidden index

    # Decode batch and hidden indices
    batch_idx = pid_bh // hidden_dim
    h_idx = pid_bh % hidden_dim

    # Offsets within the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + batch_idx * stride_lb + h_idx * stride_lh
    right_base = right_ptr + batch_idx * stride_rb + h_idx * stride_rh
    out_base   = out_ptr   + batch_idx * stride_ob + h_idx * stride_oh

    # Accumulator in FP32 (Tensor‑Core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k‑dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K) + k_offset

        # Pointers for the current K‑block
        a_ptrs = left_base + offs_i[:, None] * stride_li + k_range[None, :] * stride_lk
        b_ptrs = right_base + k_range[:, None] * stride_rk + offs_j[None, :] * stride_ri

        # Masks for out‑of‑bounds loads
        a_mask = (offs_i[:, None] < M) & (k_range[None, :] < K)
        b_mask = (k_range[:, None] < K) & (offs_j[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=tl.float16(0.0))
        b = tl.load(b_ptrs, mask=b_mask, other=tl.float16(0.0))

        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # Write the result back (store as FP16)
    c_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    c_mask = (offs_i[:, None] < M) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
        - mask       : torch.Tensor of shape [B, N, N] or None
        - weights    : dict of torch.Tensors containing all linear and LayerNorm
                       parameters (see paper implementation)
        - config     : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, cfg = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = cfg["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Linear projections and gates (all in FP32)
    # -----------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    if mask is not None:
        # Broadcast mask over the hidden dimension
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 3. TriMul core: batched matmul using Triton
    # -----------------------------------------------------------------
    # Cast to FP16 for the kernel (Tensor‑Core friendly)
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Allocate output buffer (FP16) with the same layout as left/right
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # Strides (in elements, not bytes)
    stride_lb = left_h.stride(0)
    stride_li = left_h.stride(1)
    stride_lk = left_h.stride(2)
    stride_lh = left_h.stride(3)

    stride_rb = right_h.stride(0)
    stride_ri = right_h.stride(1)
    stride_rk = right_h.stride(2)
    stride_rh = right_h.stride(3)

    stride_ob = out_h.stride(0)
    stride_oi = out_h.stride(1)
    stride_oj = out_h.stride(2)
    stride_oh = out_h.stride(3)

    # Tile sizes – tuned for A100 / B200 shared‑memory limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * hidden_dim,                 # combined batch‑hidden dimension
    )

    _trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        N,                # M
        N,                # N
        N,                # K
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 for the remaining layers
    out_hidden = out_h.to(torch.float32)

    # -----------------------------------------------------------------
    # 4. Output LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Apply output gate
    out_hidden = out_hidden * out_gate

    # Final linear projection back to original channel dimension
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
