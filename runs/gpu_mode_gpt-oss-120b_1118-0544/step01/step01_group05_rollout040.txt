@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    H,  # hidden dimension
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute batch and hidden index
    # Using integer division
    b_idx = pid // H
    h_idx = pid - b_idx * H

    # compute offsets for each slice start
    a_base = a_ptr + b_idx * stride_a_batch + h_idx
    b_base = b_ptr + b_idx * stride_b_batch + h_idx
    c_base = c_ptr + b_idx * stride_c_batch + h_idx

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A (left) block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B block transposed: we need B^(T) i.e., row=cur_k, col=offs_n
        b_ptrs = b_base + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_base + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) implementation with a Triton kernel.

Algorithm
---------
1. Layer‑normalize the input tensor `x` (shape [B, N, N, dim]).
2. Compute linear projections:
    left  = x @ left_proj.weight.T
    right = x @ right_proj.weight.T
3. (Optional) apply the pairwise mask.
4. Compute three sigmoid gates (left_gate, right_gate, out_gate) and
   multiply the projected tensors by their respective gates.
5. Cast `left` and `right` to float16 and launch a Triton kernel that
   performs, for every (batch, hidden) pair, the matrix multiplication

        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

   i.e. a batched GEMM `A @ Bᵀ` where A and B have shape (N, N).
   The kernel tiles the computation with BLOCK_M×BLOCK_N tiles and
   iterates over K in BLOCK_K steps, using fp16 for loads and fp32
   accumulation.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and linearly project back to `dim`.
All heavy arithmetic (the O(N³) contraction) is performed on the GPU
by the Triton kernel; the remaining lightweight operations stay in
PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    H,                              # hidden dimension (constexpr when launching)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ Bᵀ for many (batch, hidden) slices.

    A and B have shape (batch, N, N, H) with layout:
        stride_a_batch = N*N*H
        stride_a_row   = N*H        (i dimension)
        stride_a_col   = H          (k dimension)

    The kernel processes one (batch, hidden) slice per program_id(0).
    """
    pid = tl.program_id(0)      # combined batch * hidden index
    pid_m = tl.program_id(1)    # tile row index
    pid_n = tl.program_id(2)    # tile column index

    # Decompose pid into batch and hidden indices
    b_idx = pid // H
    h_idx = pid - b_idx * H

    # Base pointers for this (batch, hidden) slice
    a_base = a_ptr + b_idx * stride_a_batch + h_idx
    b_base = b_ptr + b_idx * stride_b_batch + h_idx
    c_base = c_ptr + b_idx * stride_c_batch + h_idx

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K in BLOCK_K chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block (shape BLOCK_M × BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B block transposed on‑the‑fly, i.e. Bᵀ with shape BLOCK_K × BLOCK_N
        b_ptrs = b_base + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate the dot product
        acc += tl.dot(a, b)   # fp16 inputs, fp32 accumulation

    # Write the result
    c_ptrs = c_base + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None])


def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (already on the correct device)
            - config: dict containing at least "dim" and "hidden_dim"
                     may also contain "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional pairwise mask
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # Cast to half for the kernel (memory bandwidth reduction)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape          # H == hidden_dim

    # ------------------------------------------------------------------
    # 6. Allocate output buffer for the contraction
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, H), device=device, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 7. Strides (in elements, not bytes)
    # ------------------------------------------------------------------
    stride_a_batch = left.stride(0)   # N * N * H
    stride_a_row   = left.stride(1)   # N * H
    stride_a_col   = left.stride(2)   # H

    stride_b_batch = right.stride(0)
    stride_b_row   = right.stride(1)
    stride_b_col   = right.stride(2)

    stride_c_batch = out.stride(0)
    stride_c_row   = out.stride(1)
    stride_c_col   = out.stride(2)

    # ------------------------------------------------------------------
    # 8. Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        B * H,                      # one program per (batch, hidden) slice
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 9. Output LayerNorm, gate, and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])

    return out
