def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = x.device
    # 1. LayerNorm
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # 2. Projections
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])
    # 3. Gates: compute gating
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))
    # 4. Apply mask and gates
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)
        # compute left' and right' as gating and mask
        # We'll use Triton kernel for this
        left = left * left_gate
        right = right * right_gate
        left = left * mask_f
        right = right * mask_f
    else:
        left = left * left_gate
        right = right * right_gate
    # 5. Cast to half
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    # 6. Reshape to (B*hidden_dim, N, N)
    B, N, _, _ = left.shape
    left_mat = left.permute(0, 3, 1, 2).reshape(B*hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B*hidden_dim, N, N)
    # 7. Batch matmul: left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # 8. Reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # shape [B, N, N, hidden_dim]

"""
TriMul outgoing variant (AlphaFold3) optimized implementation.

Algorithm
---------
1. Layer‑norm the input pair representation.
2. Linear projections produce left/right tensors of shape [B,N,N,hidden].
3. Compute three sigmoid gates (left, right, out) from the same normalized input.
4. Fuse mask, left‑gate and right‑gate multiplication in a single Triton kernel
   (element‑wise: left = left * left_gate * mask, right = right * right_gate * mask).
5. Convert left/right to fp16 and reshape to (B*hidden, N, N).
6. Perform the heavy O(N³) contraction as a batched GEMM:
      out_slice = left_slice @ right_sliceᵀ
   using torch.bmm (cuBLAS/Tensor‑Core accelerated).
7. Reshape back to [B,N,N,hidden] and apply a second layer‑norm.
8. Multiply by the output gate and a final linear projection back to the original
   channel dimension.

The only custom Triton kernel is the gate+mask fusion; the matrix multiplication
leverages cuBLAS for maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def apply_gate_mask_kernel(
    left_ptr,                # fp16 *
    right_ptr,               # fp16 *
    left_gate_ptr,           # fp16 *
    right_gate_ptr,          # fp16 *
    mask_ptr,                # fp16 *
    total_elements,          # int64 scalar
    hidden_dim,              # int64 scalar
    BLOCK_SIZE: tl.constexpr,  # compile‑time tile size
):
    """
    In‑place elementwise fusion:
    left  = left  * left_gate  * mask
    right = right * right_gate * mask
    mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    base = pid * BLOCK_SIZE
    offs = base + tl.arange(0, BLOCK_SIZE)

    # guard against out‑of‑bounds
    mask = offs < total_elements

    # mask offset = floor(offset / hidden_dim)  (broadcast)
    mask_offs = offs // hidden_dim

    # load values (fp16)
    l  = tl.load(left_ptr + offs,              mask=mask, other=tl.float16(0.0))
    r  = tl.load(right_ptr + offs,             mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + offs,         mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + offs,        mask=mask, other=tl.float16(0.0))
    m  = tl.load(mask_ptr + mask_offs,         mask=mask, other=tl.float16(1.0))

    # fused multiplication
    l = l * lg * m
    r = r * rg * m

    # write back
    tl.store(left_ptr + offs,  l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:        torch.Tensor [B, N, N] (bool / float) or None
            - weights: dict of model parameters (torch tensors on the same device)
            - config: dict containing at least "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (float32)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (float32 -> fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Cast everything needed for the contraction to fp16
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 4. Fuse mask + gates with a Triton kernel (if a mask is present)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        # mask: [B,N,N] -> fp16, contiguous, no extra dim
        mask_fp16 = mask.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)               # length B*N*N

        total_elements = left.numel()                # B*N*N*hidden
        BLOCK_SIZE = 1024                            # reasonable tile size
        grid = ((total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE,)

        apply_gate_mask_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            mask_flat,
            total_elements,
            hidden,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=4,
            num_stages=2,
        )
    else:
        # simple elementwise multiplication when no mask is used
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched GEMM using torch.bmm (fp16 inputs, fp32 accumulation)
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # out_mat = left @ rightᵀ  (shape B*hidden, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out_mat = out_mat.to(torch.float32)      # back to fp32 for the rest

    # ------------------------------------------------------------------
    # 6. Reshape to original 4‑D layout
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1)   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 7. Output LayerNorm + output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate    # out_gate is float32, same shape as out

    # ------------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
