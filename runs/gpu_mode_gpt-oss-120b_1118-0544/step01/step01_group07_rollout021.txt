def batched_matmul(A_fp16, B_fp16):
    # A,B: [batch, M, K] as fp16 (torch.float16)
    batch, M, K = A_fp16.shape
    _, N, K2 = B_fp16.shape
    assert K == K2
    C = torch.empty((batch, M, N), dtype=torch.float32, device=A_fp16.device)

    # compute pointers and strides
    a_batch_stride, a_m_stride, a_k_stride = A_fp16.stride()
    b_batch_stride, b_n_stride, b_k_stride = B_fp16.stride() # note dims: (batch, N, K)
    c_batch_stride, c_m_stride, c_n_stride = C.stride()

    # block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N), batch)
    batched_matmul_kernel[grid](
        A_fp16, B_fp16, C,
        M, N, K,
        a_batch_stride, a_m_stride, a_k_stride,
        b_batch_stride, b_n_stride, b_k_stride,
        c_batch_stride, c_m_stride, c_n_stride,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4, num_stages=2
    )
    return C

"""
Highly‑optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Apply LayerNorm to the input tensor.
2. Project the normalized tensor to a hidden dimension with two linear layers
   (left_proj, right_proj).  Optionally mask the projections.
3. Compute three sigmoid gates (left_gate, right_gate, out_gate) from the same
   normalized input and apply them element‑wise to the projected tensors.
4. Core O(N³·D) computation:
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication A @ Bᵀ for each hidden channel d.
   We fuse the batch and hidden dimensions (B·D) and run a custom Triton
   kernel that tiles the M/N dimensions (i/j) and the reduction dimension K.
   The kernel loads FP16 data, converts to FP32 for the dot‑product, and
   accumulates in FP32 before storing the result.
5. Apply a second LayerNorm on the hidden dimension, multiply by out_gate,
   and finally project back to the original dimension.
All operations are performed on the GPU; the heavy O(N³) step is completely
implemented in Triton.

The kernel is tuned for sequence lengths up to 1024 and hidden dimension 128.
Block sizes are chosen adaptively based on N to keep high occupancy on an
NVIDIA B200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
#   A : [Bhd, M, K]   (float16)
#   B : [Bhd, N, K]   (float16)   – accessed as (K, N) to achieve the transpose
#   C : [Bhd, M, N]   (float32)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr,                    # *fp16  [batch, M, K]
    B_ptr,                    # *fp16  [batch, N, K]   (read as K×N)
    C_ptr,                    # *fp32  [batch, M, N]
    M, N, K,                  # i32
    stride_ah, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bh, stride_bn, stride_bk,   # strides for B (batch, N, K)
    stride_ch, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)          # batch*hidden index
    pid_m = tl.program_id(0)              # tile row
    pid_n = tl.program_id(1)              # tile col

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Masks for current tile
        mask_a = mask_m[:, None] & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & mask_n[None, :]

        # Pointers to A[i,k] and B[k,j] (note B is accessed transposed)
        a_ptrs = (
            A_ptr
            + pid_batch * stride_ah
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        b_ptrs = (
            B_ptr
            + pid_batch * stride_bh
            + offs_k[:, None] * stride_bk   # K dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bn   # N dimension (col of Bᵀ)
        )

        # Load tiles, zero‑pad out‑of‑bounds
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # Up‑cast to FP32 for the dot product
        a_f32 = a.to(tl.float32)
        b_f32 = b.to(tl.float32)

        # Accumulate
        acc += tl.dot(a_f32, b_f32)

    # Write the result tile
    c_ptrs = (
        C_ptr
        + pid_batch * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)


def batched_matmul(A_fp16: torch.Tensor, B_fp16: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for a batch of matrices using Triton.
    A, B : [batch, M, K] and [batch, N, K] (FP16, contiguous)
    Returns C : [batch, M, N] (FP32)
    """
    assert A_fp16.is_contiguous() and B_fp16.is_contiguous()
    batch, M, K = A_fp16.shape
    _, N, K2 = B_fp16.shape
    assert K == K2, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A_fp16.device)

    # Strides in elements (not bytes)
    stride_ah, stride_am, stride_ak = A_fp16.stride()
    stride_bh, stride_bn, stride_bk = B_fp16.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    # Choose block sizes adaptively (must be compile‑time constants per launch)
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
    elif N <= 512:
        BLOCK_M = 128
        BLOCK_N = 128
    else:
        BLOCK_M = 256
        BLOCK_N = 256
    BLOCK_K = 32

    grid = (triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N), batch)

    batched_matmul_kernel[grid](
        A_fp16,
        B_fp16,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    return C


# ----------------------------------------------------------------------
# TriMul custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (may be None or all‑ones)
        - weights      : dict of model parameters (torch.Tensors)
        - config       : dict with at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Extract weights
    # ------------------------------------------------------------------
    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    # Gating weights
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    # Output layers
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape [B, N, N] -> [B, N, N, 1] and broadcast
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul: out = Σ_k left[..., i, k, d] * right[..., j, k, d]
    #    Implemented as batched matmul A @ Bᵀ using a Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Rearrange to [B, hidden_dim, N, N] and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, D, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, D, N, N]

    # Fuse batch and hidden dimensions for the kernel
    batch_hidden = B * hidden_dim
    left_flat_fp16 = left_perm.view(batch_hidden, N, N).to(torch.float16)
    right_flat_fp16 = right_perm.view(batch_hidden, N, N).to(torch.float16)

    # Batched matrix multiplication A @ Bᵀ  (result in FP32)
    out_flat_fp32 = batched_matmul(left_flat_fp16, right_flat_fp16)  # [BH, N, N]

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    # Final linear projection back to original dim
    output = F.linear(out, to_out_weight)   # (dim, hidden_dim) weight

    return output
