"""
TriMul outgoing variant (AlphaFold3) optimized with a Triton batched GEMM kernel.

The core operation contracts over the sequence dimension k:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a matrix multiplication A @ Bᵀ for each batch*hidden
slice where A = left[b,:, :, d] and B = right[b,:, :, d].

We
  * perform LayerNorms and linear projections in PyTorch (high‑performance cuBLAS),
  * apply masks and sigmoid gates (element‑wise),
  * fuse the expensive O(N³) contraction into a single Triton kernel that computes
    B * hidden_dim independent GEMMs of size (N, N) @ (N, N)ᵀ.
The kernel works on FP16 inputs (half‑precision) while accumulating in FP32 to
preserve numerical stability.  Output is written as FP16 to halve memory traffic,
then cast back to FP32 for the subsequent LayerNorm and final linear layer.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr,               # half*   left matrix batch
    b_ptr,               # half*   right matrix batch
    c_ptr,               # half*   output batch (will be cast to FP32 later)
    M, N, K,             # dimensions of each matrix (all equal to seq_len)
    stride_a_batch,      # batch stride for A
    stride_a_row,        # row stride for A (i dimension)
    stride_a_col,        # column stride for A (k dimension)
    stride_b_batch,      # batch stride for B
    stride_b_row,        # row stride for B (j dimension)
    stride_b_col,        # column stride for B (k dimension)
    stride_c_batch,      # batch stride for C
    stride_c_row,        # row stride for C
    stride_c_col,        # column stride for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched compute C = A @ Bᵀ.

    Each program (block) processes a tile of size BLOCK_M × BLOCK_N for a single
    (batch * hidden_dim) matrix.  The K‑dimension is tiled with BLOCK_K.
    """
    pid_batch = tl.program_id(0)   # which (batch * hidden_dim) matrix
    pid_m = tl.program_id(1)       # tile row index
    pid_n = tl.program_id(2)       # tile column index

    # Offsets in the output matrix C
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current matrix
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in chunks of BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A block: rows i, cols k
        a_offs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_offs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B block transposed on the fly: we need Bᵀ[k, j] = B[j, k]
        b_offs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_offs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Write result as FP16 to reduce bandwidth
    c_offs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_offs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (tensors on same device)
            - config: dict containing at least "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ----------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----------------------------------------------------------------------
    # 2. Linear projections (left/right) – keep in FP32 for precision
    # ----------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])      # [B, N, N, hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])    # [B, N, N, hidden_dim]

    # ----------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    # ----------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------------------
    # 4. Gating signals (sigmoid activations)
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ----------------------------------------------------------------------
    # 5. Apply gates (still FP32)
    # ----------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 6. Cast to FP16 and reshape for the Triton GEMM
    # ----------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Rearrange to (B, hidden_dim, N, N) -> (B*hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (FP16 to halve memory traffic)
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float16)

    # ----------------------------------------------------------------------
    # 7. Triton kernel launch configuration
    # ----------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 128
    BLOCK_K = 128

    # Strides expressed in element counts (not bytes)
    stride_a_batch = N * N
    stride_a_row = N
    stride_a_col = 1

    stride_b_batch = N * N
    stride_b_row = N
    stride_b_col = 1

    stride_c_batch = N * N
    stride_c_row = N
    stride_c_col = 1

    grid = (
        B * hidden_dim,                     # program ID 0: batch * hidden_dim
        (N + BLOCK_M - 1) // BLOCK_M,       # program ID 1: tile rows
        (N + BLOCK_N - 1) // BLOCK_N,       # program ID 2: tile cols
    )

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # ----------------------------------------------------------------------
    # 8. Cast back to FP32 and restore original layout
    # ----------------------------------------------------------------------
    out = out_mat.to(torch.float32)                                   # [B*hidden_dim, N, N]
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)           # [B, N, N, hidden_dim]

    # ----------------------------------------------------------------------
    # 9. Output LayerNorm, gate, and final linear projection
    # ----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])

    return out
