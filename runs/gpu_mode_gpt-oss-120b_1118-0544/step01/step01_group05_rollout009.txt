"""
TriMul (outgoing) forward implementation with Triton.
----------------------------------------------------
The heavy O(N³) contraction 
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] 
is rewritten as a batched matrix multiplication 
    out = left @ rightᵀ 
for every hidden‐dim slice d (and batch).  
All cheap operations (layer‑norm, linear projections, gates) are done in PyTorch,
while the batched GEMM is performed by a custom Triton kernel that
* uses FP16 for A/B to halve memory traffic,
* accumulates in FP32,
* leverages Tensor‑Core friendly tile sizes (128×128×32),
* pipelines loads with `num_stages=4` and `num_warps=8`.

The kernel computes C = A @ Bᵀ for each (batch * hidden_dim) matrix:
    A : (M, K)  with M = N, K = N
    B : (N, K)  (transposed on‑the‑fly to (K, N))
    C : (M, N)  = (N, N)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr,                # left matrix pointer   (batch*hdim, M, K)
    b_ptr,                # right matrix pointer  (batch*hdim, N, K)
    c_ptr,                # output pointer        (batch*hdim, M, N)
    M, N, K,              # dimensions
    stride_a_batch,       # stride to next A matrix
    stride_a_row,         # stride between rows of A (M axis)
    stride_a_col,         # stride between cols of A (K axis)
    stride_b_batch,       # stride to next B matrix
    stride_b_row,         # stride between rows of B (N axis)
    stride_b_col,         # stride between cols of B (K axis)
    stride_c_batch,       # stride to next C matrix
    stride_c_row,         # stride between rows of C (M axis)
    stride_c_col,         # stride between cols of C (N axis)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)  # which (batch * hidden_dim) slice
    pid_m = tl.program_id(1)      # tile over rows (M)
    pid_n = tl.program_id(2)      # tile over cols (N)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the first element of the current A/B/C matrices
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A block: shape (BLOCK_M, BLOCK_K)
        a_offs = a_batch + \
                 offs_m[:, None] * stride_a_row + \
                 cur_k[None, :] * stride_a_col
        a = tl.load(a_offs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Bᵀ block: shape (BLOCK_K, BLOCK_N)
        # Original B layout is (N, K); we fetch as (K, N)
        b_offs = b_batch + \
                 cur_k[:, None] * stride_b_col + \
                 offs_n[None, :] * stride_b_row
        b = tl.load(b_offs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Matrix multiply tile (uses Tensor Cores, accumulates in FP32)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Write out the result tile
    c_offs = c_batch + \
             offs_m[:, None] * stride_c_row + \
             offs_n[None, :] * stride_c_col
    tl.store(c_offs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (tensors on same device)
            - config: dict containing at least "dim", "hidden_dim" and optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask (applied before gating)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4) Gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Cast to half for the expensive O(N³) contraction
    # -----------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Reshape to (B * hidden_dim, N, N) – contiguous layout for Triton
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (float32 accumulator)
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # -----------------------------------------------------------------
    # 6) Triton GEMM: out_mat = left_mat @ right_matᵀ
    # -----------------------------------------------------------------
    # Tile sizes tuned for H100 / B200 (Tensor Core friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Strides in element units (not bytes)
    stride_a_batch = left_mat.stride(0)
    stride_a_row = left_mat.stride(1)   # row stride (M axis)
    stride_a_col = left_mat.stride(2)   # column stride (K axis)

    stride_b_batch = right_mat.stride(0)
    stride_b_row = right_mat.stride(1)  # row stride (N axis)
    stride_b_col = right_mat.stride(2)  # column stride (K axis)

    stride_c_batch = out_mat.stride(0)
    stride_c_row = out_mat.stride(1)    # row stride (M axis)
    stride_c_col = out_mat.stride(2)    # column stride (N axis)

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # -----------------------------------------------------------------
    # 7) Reshape back to [B, N, N, hidden_dim]
    # -----------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 8) Output LayerNorm, gating, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
