Outgoing Triangle Multiplicative Update (TriMul) – Triton implementation.

The forward pass consists of:
1️⃣ Layer‑norm the (B, N, N, D) input.
2️⃣ Linear projections to a hidden dimension H (left/right) without bias.
3️⃣ Optional mask applied to the projected tensors.
4️⃣ Gating tensors (left_gate, right_gate, out_gate) computed with sigmoid.
5️⃣ Core O(N³) operation:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication A @ Bᵀ where
   A = left.permute(0,3,1,2).view(B*H, N, N) and
   B = right.permute(0,3,1,2).view(B*H, N, N).
   Implemented with a custom Triton kernel that tiles the computation
   (BLOCK_M=128, BLOCK_N=128, BLOCK_K=64) and accumulates in FP32.
6️⃣ Output layer‑norm, multiply by out_gate, and final linear projection
   back to the original dimension D.

All heavy compute (step 5) is performed on the GPU via Triton; the rest
uses PyTorch which is already memory‑bound and negligible compared to the
N³ kernel.
"""

import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ  (A,B: [batch, M, K] float32)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr,                     # *float32   [batch, M, K]
    B_ptr,                     # *float32   [batch, N, K] (read as K×N)
    C_ptr,                     # *float32   [batch, M, N]
    M, N, K,                   # i32 dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication: C = A @ Bᵀ.

    A : (batch, M, K)
    B : (batch, N, K)   (interpreted as K×N when loading)
    C : (batch, M, N)
    """

    pid_batch = tl.program_id(2)          # batch index (including hidden dim)
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block column

    # Offsets for output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)  # K‑dimension offsets

        # Masks for loading
        mask_a = mask_m[:, None] & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & mask_n[None, :]

        # Pointers for current tile of A and (transposed) B
        a_ptrs = (
            A_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            B_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k   # K dimension (row of Bᵀ)
            + offs_n[None, :] * stride_b_n   # N dimension (col of Bᵀ)
        )

        # Load tiles, zero‑pad OOB elements
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Write back the result tile
    c_ptrs = (
        C_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)


def batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for batches of matrices using Triton.
    A, B : [batch, N, N] (float32, contiguous)
    Returns C : [batch, N, N] (float32)
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Strides in element counts (not bytes)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_n, stride_b_k = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tunable block sizes – chosen for good occupancy on Nvidia B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(M, BLOCK_M),   # dim 0: rows
        triton.cdiv(N, BLOCK_N),   # dim 1: cols
        batch,                     # dim 2: batch*hidden
    )

    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return C


# ----------------------------------------------------------------------
# Entry point for the custom kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, D] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (torch.Tensors)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Configuration & weight extraction
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Projection weights (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    # Output norm & projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, D]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Apply mask (if supplied)
    # ------------------------------------------------------------------
    if mask is not None:
        # Ensure mask is broadcastable and same dtype as tensors
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gating to the projected tensors
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) TriMul via batched matmul (A @ Bᵀ)
    # ------------------------------------------------------------------
    B_, N, _, _ = left.shape   # B_ = batch size, N = seq_len
    # Rearrange to [batch, hidden, N, N] and flatten batch*hidden dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_hidden = B_ * hidden_dim
    left_mat = left_perm.view(batch_hidden, N, N)   # [BH, N, N]
    right_mat = right_perm.view(batch_hidden, N, N) # [BH, N, N]

    # Batched matmul: result [BH, N, N]
    out_mat = batched_matmul(left_mat, right_mat)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B_, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out_gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    # Final linear projection back to original dimension D
    output = F.linear(out, to_out_weight)  # (D, H) weight

    return output
