```python
@triton.jit
def tri_mul_kernel(a_ptr, b_ptr, c_ptr,
                   M, N, K,
                   stride_a_batch, stride_a_row, stride_a_col,
                   stride_b_batch, stride_b_row, stride_b_col,
                   stride_c_batch, stride_c_row, stride_c_col,
                   BLOCK_M: tl.constexpr,
                   BLOCK_N: tl.constexpr,
                   BLOCK_K: tl.constexpr):
    pid = tl.program_id(0) # batch index
    pid_m = tl.program_id(1) # row tile
    pid_n = tl.program_id(2) # col tile

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers
    a_batch = a_ptr + pid * stride_a_batch
    b_batch = b_ptr + pid * stride_b_batch
    c_batch = c_ptr + pid * stride_c_batch

    # accumulation buffer
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # B transposed: we want B^T (BLOCK_K, BLOCK_N)
        # B layout: row dimension (N) -> stride_b_row, col dimension (K) -> stride_b_col
        b_ptrs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – accelerated core contraction with Triton.

The heavy part of the TriMul forward pass is the 4‑D tensor contraction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

for every batch `b` and hidden dimension `d`.  This can be written as a
batched matrix multiplication

    out_flat = left_flat @ right_flat.T

where `left_flat` and `right_flat` are reshaped to (B·D, N, N) and the
multiplication is performed independently for each of the `B·D` batches.
The Triton kernel below computes this GEMM in FP16 (tensor‑core) while
accumulating in FP32, giving a fast forward pass.  All other operations
(LayerNorm, linear projections, gating, output projection) are done in
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM C = A @ Bᵀ.
    A : (batch, M, K)   – fp16
    B : (batch, N, K)   – fp16 (we load its transpose on‑the‑fly)
    C : (batch, M, N)   – fp32 accumulator, written as fp32
    """
    pid_batch = tl.program_id(0)          # which batch*hidden slice
    pid_m = tl.program_id(1)              # tile over output rows (M)
    pid_n = tl.program_id(2)              # tile over output cols (N)

    # Coordinates of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)               # fp16 -> fp32 automatically in dot

        # Load Bᵀ block: we need B[j, k] where j is output column, k is reduction dim
        # B layout is (N, K); loading as (K, N) for the dot
        b_ptrs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Dot product (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # Write the result
    c_ptrs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    TriMul forward pass (outgoing variant).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (already on the correct device)
            - config: dict with at least "dim" and "hidden_dim"; optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm ----------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if (not nomask) and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare for Triton GEMM ----------
    # Half precision for the heavy matmul
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Bring hidden dimension to the second axis and flatten batch*hidden
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_mat = right.permute(0, 3, 1, 2).contiguous()

    batch_flat = B * hidden_dim
    left_mat = left_mat.view(batch_flat, N, N)
    right_mat = right_mat.view(batch_flat, N, N)

    # Output buffer (fp32 accumulator)
    out_mat = torch.empty((batch_flat, N, N), device=device, dtype=torch.float32)

    # Strides (in elements)
    stride_a_batch, stride_a_row, stride_a_col = left_mat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_mat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_mat.stride()

    # Tuning parameters – these work well on B200 for fp16 tensor‑core GEMM
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_flat,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the kernel
    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # ---------- Reshape back ----------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to original dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
