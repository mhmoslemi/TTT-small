"""
custom_kernel for the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B, N, N, dim]).
2. Cast to fp16 and compute the four linear projections (left/right and their
   gates) in a fused manner using torch.nn.functional.linear.  The mask (if
   any) is applied in fp16 before gating.
3. Apply the sigmoid gates and combine with the projected tensors.
4. Core O(N³·H) operation:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication `left @ rightᵀ` with batch size
   `B*H`.  The multiplication is performed by a custom Triton kernel that
   tiles the matrices (BLOCK_M×BLOCK_N tiles, reduction over BLOCK_K) and
   accumulates in FP32 while loading in FP16.
5. Apply a second LayerNorm on the hidden dimension, multiply by the
   output‑gate, and project back to the original dimension with a linear
   layer.
6. Return a float32 tensor of shape [B, N, N, dim].

The Triton kernel is tuned for square matrices up to N=1024, hidden
dimension H≤128 and batch size ≤2.  Block sizes (128,128,32) together with
8 warps give good occupancy on an NVIDIA B200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched matmul: C = A @ Bᵀ   (A,B : [batch, M, K] , [batch, N, K])
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(2)          # batch * hidden
    pid_m = tl.program_id(0)              # row tile
    pid_n = tl.program_id(1)              # col tile

    # ---- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over reduction dimension K -------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # masks for current tile
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # pointers for A (shape: batch, M, K) and B (shape: batch, N, K)
        a_ptrs = (
            A_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            B_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k   # K dimension becomes row index
            + offs_n[None, :] * stride_b_n   # N dimension is column index
        )

        a = tl.load(a_ptrs, mask=mask_a, other=tl.float16(0.0))
        b = tl.load(b_ptrs, mask=mask_b, other=tl.float16(0.0))

        # tiled gemm
        acc += tl.dot(a, b)

    # ---- write back ----------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)


def batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ Bᵀ using Triton.
    A, B : [batch, M, K] and [batch, N, K] (fp16, contiguous)
    Returns C : [batch, M, N] (fp32)
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # ---- strides in elements (not bytes) --------------------------------
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_n, stride_b_k = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # ---- tunable block sizes -------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )

    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    return C


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (may be None)
        - weights      : dict of model parameters
        - config       : dict containing at least "dim", "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. LayerNorm on the input
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Cast to fp16 for the heavy linear work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # Linear weights (cast to fp16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 3. Projections & gates (fp16)
    # --------------------------------------------------------------
    left_proj = F.linear(x_f16, left_proj_w)          # [B, N, N, H]
    right_proj = F.linear(x_f16, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x_f16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))

    # --------------------------------------------------------------
    # 4. Apply mask (if provided) in fp16
    # --------------------------------------------------------------
    if mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # --------------------------------------------------------------
    # 5. Apply element‑wise gates (still fp16)
    # --------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6. Core TriMul: batched matmul left @ rightᵀ
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # batch size and sequence length

    # reshape to [B*H, N, N] (contiguous)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_hidden = B * hidden_dim
    left_mat = left_perm.view(batch_hidden, N, N)      # fp16
    right_mat = right_perm.view(batch_hidden, N, N)    # fp16

    # batched matmul using Triton (output is fp32)
    out_mat = batched_matmul_triton(left_mat, right_mat)  # [BH, N, N] fp32

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Second LayerNorm + out‑gate
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # out_gate was fp16; promote to fp32 for final multiplication
    out = out * out_gate.to(out.dtype)

    # --------------------------------------------------------------
    # 8. Final projection back to dim
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"]          # (dim, hidden_dim), float32
    output = F.linear(out, to_out_w)             # [B, N, N, dim]

    return output
```
