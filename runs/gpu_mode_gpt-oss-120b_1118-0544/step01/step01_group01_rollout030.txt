"""
custom_kernel: Triton‑accelerated outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1️⃣ Layer‑norm the input tensor.
2️⃣ Linear projections → left, right and three gating tensors (left_gate,
   right_gate, out_gate).  Apply optional mask and element‑wise sigmoid gates.
3️⃣ Cast left/right to FP16, permute & reshape to (B*H, N, N) where H = hidden_dim.
   Right is additionally transposed along the two sequence axes so that the
   Einstein summation reduces to a batched matrix multiplication:
        out_h = left_h @ right_hᵀ   for each hidden slice h.
4️⃣ A Triton kernel performs the batched GEMM in FP16 (accumulator FP32) with
   128×128 tiles and K‑tiles of 32, fully exploiting Tensor‑cores.
5️⃣ Reshape the result back to (B, N, N, H), apply the output LayerNorm,
   multiply by out_gate and project back to the original dimension.
All heavy arithmetic stays on the GPU; the final tensor is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A, B, C,               # pointers
    M, N, K,               # matrix dims (all int32)
    stride_am, stride_ak,  # strides for A (row, col)
    stride_bk, stride_bn,  # strides for B (row, col)  - note B is already transposed
    stride_cm, stride_cn,  # strides for C (row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_z = tl.program_id(2)          # batch * hidden slice

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this (batch, head) slice
    a_base = A + pid_z * (M * K)      # contiguous layout: stride0 = M*K
    b_base = B + pid_z * (K * N)      # stride0 = K*N
    c_base = C + pid_z * (M * N)      # stride0 = M*N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K steps
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)

        # Compute pointers for the current K‑block
        a_ptrs = a_base \
                 + (offs_m[:, None] * stride_am) \
                 + (k_range[None, :] * stride_ak)
        b_ptrs = b_base \
                 + (k_range[:, None] * stride_bk) \
                 + (offs_n[None, :] * stride_bn)

        # OOB masks
        a_mask = (offs_m[:, None] < M) & (k_range[None, :] < K)
        b_mask = (k_range[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Tensor‑core dot (fp16 * fp16 -> fp32)
        acc += tl.dot(a, b)

    # Store the computed tile
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor[B, N, N, dim]
        - mask         : torch.Tensor[B, N, N] or None
        - weights      : dict of model parameters
        - config       : dict containing "dim" and "hidden_dim" (and optional "nomask")

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, cfg = data
    device = x.device
    dtype = x.dtype

    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    # -------------------- 1. Input LayerNorm --------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # -------------------- 2. Linear projections & gates --------------------
    left_proj_w = weights["left_proj.weight"]        # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left = F.linear(x, left_proj_w)                # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    if (mask is not None) and (not nomask):
        mask_f = mask.to(dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -------------------- 3. Batched matmul in FP16 (Triton) --------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # left: (B, N, N, H) → (B, H, N, N)   rows = i, cols = k
    left = left.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, H) → (B, H, k, j)   (transpose the two sequence axes)
    right = right.permute(0, 3, 2, 1).contiguous()

    # Merge batch and hidden dimension for the kernel
    left_mat = left.view(-1, N, N)          # (Z, M, K) where Z = B*H
    right_mat = right.view(-1, N, N)        # (Z, K, N)
    out_mat = torch.empty_like(left_mat)    # (Z, M, N)   fp16

    Z = left_mat.shape[0]                   # B * hidden_dim

    # Strides (in elements) for the kernel
    stride_am = left_mat.stride(1)   # M‑stride (row)
    stride_ak = left_mat.stride(2)   # K‑stride (col)
    stride_bk = right_mat.stride(1)  # K‑stride (row of transposed right)
    stride_bn = right_mat.stride(2)  # N‑stride (col of transposed right)
    stride_cm = out_mat.stride(1)    # M‑stride
    stride_cn = out_mat.stride(2)    # N‑stride

    # Tile sizes – chosen for Tensor‑core efficiency
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        Z,
    )

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,                   # M
        N,                   # N
        N,                   # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------- 4. Restore shape & post‑processing --------------------
    # out_mat: (Z, N, N) → (B, hidden_dim, N, N) → (B, N, N, hidden_dim)
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # Output LayerNorm + gating
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_hidden = F.layer_norm(out_hidden, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out_hidden = out_hidden * out_gate

    # Final projection back to original dimension
    to_out_w = weights["to_out.weight"]        # (dim, hidden_dim)
    out = F.linear(out_hidden, to_out_w)       # [B, N, N, dim]

    return out
```
