"""
TriMul outgoing kernel (AlphaFold3) – fast batched GEMM with Triton.

The heavy part of TriMul is the contraction
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
for every hidden channel d.  By reshaping the tensors we can view each
(d,b) slice as a matrix multiplication A @ Bᵀ where
    A = left[b,…,:,d]   of shape (N, N)
    B = right[b,…,:,d]  of shape (N, N)

The kernel computes a batched GEMM for all (batch * hidden_dim) slices.
It loads A and Bᵀ in tiles, uses tl.dot (fp16 → fp32 accumulation)
and writes the result as fp32.  Tile sizes (128×128×64) and 8 warps
give good occupancy on an A200.  All other parts (LayerNorm, Linear,
gating) are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_az, stride_ai, stride_ak,
    stride_bz, stride_bi, stride_bk,
    stride_cz, stride_ci, stride_ck,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ

    A : (batch, M, K)   float16
    B : (batch, N, K)   float16  (interpreted as transposed on‑the‑fly)
    C : (batch, M, N)   float32
    """
    pid_z = tl.program_id(0)   # batch * hidden_dim
    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile col

    # row / col offsets of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch slice
    a_batch = a_ptr + pid_z * stride_az
    b_batch = b_ptr + pid_z * stride_bz
    c_batch = c_ptr + pid_z * stride_cz

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_offs = a_batch + offs_m[:, None] * stride_ai + cur_k[None, :] * stride_ak
        a = tl.load(a_offs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Bᵀ tile: original B layout is (N, K)
        # we load it transposed so that the inner dimension matches K
        b_offs = b_batch + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bi
        b = tl.load(b_offs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_offs = c_batch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_ck
    tl.store(c_offs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (already on the correct device)
            - config: dict with at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------- LayerNorm (input) -------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------- Linear projections -------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------- Optional mask -------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------- Gates -------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------- Apply gates -------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------- Cast to half for GEMM -------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # ---- reshape to (B*hidden_dim, N, N) ----
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # output buffer (float32)
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # ---- kernel launch configuration ----
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    stride_a_batch, stride_a_row, stride_a_col = left_mat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_mat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_mat.stride()

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_az=stride_a_batch,
        stride_ai=stride_a_row,
        stride_ak=stride_a_col,
        stride_bz=stride_b_batch,
        stride_bi=stride_b_row,
        stride_bk=stride_b_col,
        stride_cz=stride_c_batch,
        stride_ci=stride_c_row,
        stride_ck=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ---- reshape back to [B, N, N, hidden_dim] ----
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # ------------------- Output LayerNorm -------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ------------------- Apply output gate -------------------
    out = out * out_gate

    # ------------------- Final linear projection -------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
