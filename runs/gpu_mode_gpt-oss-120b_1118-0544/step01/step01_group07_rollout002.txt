@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_Ab, stride_Am, stride_Ak,
                          stride_Bb, stride_Bk, stride_Bn,
                          stride_Cb, stride_Cm, stride_Cn,
                          BLOCK_M: tl.constexpr,
                          BLOCK_N: tl.constexpr,
                          BLOCK_K: tl.constexpr,
                          ):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offset
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out of bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        b_ptrs = B_ptr + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # store
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)

"""
custom_kernel for the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Apply a first LayerNorm to the 4‑D input [B,N,N,dim].
2. Linear projections (left/right) and three gating projections are computed
   with bias‑free Linear layers, followed by sigmoid gates.
3. The core O(N³) operation
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is implemented as a batched matrix multiplication A @ Bᵀ using a custom
   Triton kernel that processes tiles of size (BLOCK_M, BLOCK_N, BLOCK_K)
   and accumulates in FP32.  The tensors are reshaped to
   [B*hidden, N, N] before the multiplication.
4. A second LayerNorm (over the hidden dimension) is applied, multiplied by the
   output gate and finally projected back to the original dimension.

The kernel works for batch size 1‑2, sequence length up to 1024 and hidden
dimension up to 384.  All operations are performed in float32 and the final
output has shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
# A : [batch, M, K]   (contiguous)
# B : [batch, N, K]   (contiguous) -> we pass B_T = B.permute(0,2,1) with shape [batch, K, N]
# C : [batch, M, N]
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,            # pointers
    M, N, K,                        # matrix dimensions (int32)
    stride_a_batch, stride_a_m, stride_a_k,   # strides for A (elements)
    stride_b_batch, stride_b_k, stride_b_n,   # strides for B_T (elements)
    stride_c_batch, stride_c_m, stride_c_n,   # strides for C (elements)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the tile of C we are computing
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Masks for K dimension
        mask_k = offs_k < K

        # Pointers to A[batch, m, k] and B_T[batch, k, n]
        a_ptrs = (
            A_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            B_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )

        # Load tiles (zero‑pad out‑of‑bounds)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate dot product over the K dimension
        acc += tl.dot(a, b)

    # Store the result tile
    c_ptrs = (
        C_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)


def batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for a batch of matrices using Triton.
    A: [batch, M, K] (contiguous)
    B: [batch, N, K] (contiguous)
    Returns C: [batch, M, N] (float32, same device as A)
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2, "Inner dimensions must match for matmul"

    # Prepare B transposed (so we can access Bᵀ as [batch, K, N])
    B_T = B.permute(0, 2, 1).contiguous()

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Strides in *elements* (not bytes)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B_T.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tunable tile sizes – work well on modern NVIDIA GPUs (B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        A,
        B_T,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )
    return C


# ----------------------------------------------------------------------
# Entry point: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of model parameters (tensors)
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Configuration and hyper‑parameters
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load weights (all expected to be on the same device as input)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. First LayerNorm over the ``dim`` dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # Ensure mask is float and has shape [B, N, N, 1]
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating tensors (sigmoid activations)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates to the projected tensors
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul operation: out = Σ_k left[..., i, k, d] * right[..., j, k, d]
    #    Implemented as batched matrix multiplication A @ Bᵀ using Triton.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B = batch size, N = sequence length

    # Rearrange to [B, hidden_dim, N, N] and flatten batch*hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]

    left_flat = left_perm.view(B * hidden_dim, N, N)    # [B*H, N, N]
    right_flat = right_perm.view(B * hidden_dim, N, N) # [B*H, N, N]

    # Batched matmul: (B*H, N, N) @ (B*H, N, N)ᵀ  -> (B*H, N, N)
    out_flat = batched_matmul_triton(left_flat, right_flat)  # [B*H, N, N]

    # Restore original layout: [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Second LayerNorm, output gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    # Final linear projection back to ``dim``
    output = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return output
