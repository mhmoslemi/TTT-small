    M, N, K,

    stride_b_batch, stride_b_n, stride_b_k,

    pid_m = tl.program_id(1)  # block row index
    pid_n = tl.program_id(2)  # block col index

    mask_n = offs_n < N

"""
TriMul outgoing variant (AlphaFold3) optimized with a custom Triton kernel.

The heavy O(N³) contraction:
    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]
is implemented as a batched matrix multiplication A @ Bᵀ for each
(b * d) slice, where A and B are (N, N) matrices of half precision.
The kernel tiles the multiplication with BLOCK_M = BLOCK_N = 128 and
BLOCK_K = 64, uses 8 warps and 3 pipeline stages, and accumulates in
float32 for numerical stability. All other operations (LayerNorm,
linear projections, gating) are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 0‑th axis indexes the (batch * hidden_dim) slice
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)  # tile row
    pid_n = tl.program_id(2)  # tile column

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_offs = a_batch + offs_m[:, None] * stride_a_m + cur_k[None, :] * stride_a_k
        a = tl.load(a_offs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load Bᵀ block: shape (BLOCK_K, BLOCK_N)
        # Original B is (N, K); we stream it as (K, N)
        b_offs = b_batch + cur_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_offs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # Store the result
    c_offs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_offs, acc, mask=mask_m[:, None] & mask_n[:, None])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (all on the same device)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm ----------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ---------- Apply mask (if any) ----------
    if (not nomask) and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ---------- Apply gates ----------
    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to half for the Triton kernel ----------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # left shape [B, N, N, hidden_dim]

    # Rearrange to (B * hidden_dim, N, N) contiguous tensors
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (float32 accumulator)
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # ---------- Triton kernel launch ----------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Strides in elements (row‑major layout)
    stride_a_batch = N * N
    stride_a_m = N
    stride_a_k = 1

    stride_b_batch = N * N
    stride_b_n = N   # stride for the original row index (j)
    stride_b_k = 1   # stride for the k dimension

    stride_c_batch = N * N
    stride_c_m = N
    stride_c_n = 1

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ---------- Reshape back to [B, N, N, hidden_dim] ----------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
