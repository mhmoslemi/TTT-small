"""
TriMul outgoing variant (AlphaFold3) with a Triton kernel.

Core computation:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
which is a batched matrix multiplication C = A @ Bᵀ for every
(batch, hidden_dim) slice where A and B have shape (N, N)
and the hidden dimension is processed in parallel as an extra batch axis.

All LayerNorms, linear projections and sigmoid gates are performed in PyTorch.
The heavy O(N³) contraction is executed by a fused Triton kernel that
computes many independent GEMMs (A @ Bᵀ) with fp16 inputs and fp32 accumulation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched multiplication C = A @ Bᵀ.

    A: (batch, M, K)   (float16)
    B: (batch, N, K)   (float16)
    C: (batch, M, N)   (float32 accumulator)
    """
    pid_batch = tl.program_id(0)  # batch * hidden_dim index
    pid_m = tl.program_id(1)      # tile over rows   (M)
    pid_n = tl.program_id(2)      # tile over cols   (N)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_out = mask_m[:, None] & mask_n[None, :]

    # Base pointers for this batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_offset = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_offset,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B block transposed on‑the‑fly:
        # original B layout is (N, K). We need Bᵀ shape (K, N).
        b_offset = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_offset,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Matrix multiply: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

    # Store the result
    c_offset = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_offset, acc, mask=mask_out)


def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (tensors on same device)
            - config: dict with at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input channel dimension
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid on linear projections)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Prepare tensors for the Triton GEMM (A @ Bᵀ)
    # -----------------------------------------------------------------
    # Cast to fp16 for bandwidth reduction; accumulation stays fp32
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Layout: [B, hidden_dim, N, N] -> flatten batch*hidden_dim
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (float32 accumulator)
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # -----------------------------------------------------------------
    # 6. Triton kernel launch
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Strides in number of elements (not bytes)
    stride_a_batch = left_mat.stride(0)
    stride_a_row = left_mat.stride(1)
    stride_a_col = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_row = right_mat.stride(1)
    stride_b_col = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_row = out_mat.stride(1)
    stride_c_col = out_mat.stride(2)

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # -----------------------------------------------------------------
    # 7. Reshape back to 4‑D tensor [B, N, N, hidden_dim]
    # -----------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # -----------------------------------------------------------------
    # 8. Output LayerNorm, gate, and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
