import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_fp16_kernel(
    A_ptr,  # *fp16
    B_ptr,  # *fp16
    C_ptr,  # *fp32
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(2)  # batch*hidden
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # mask for tile loads
        mask_a = mask_m[:, None] & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & mask_n[None, :]

        # compute pointer offsets
        a_ptrs = (
            A_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            B_ptr
            + pid_b * stride_b_batch
            + offs_k[:, None] * stride_b_k   # K dim (row of Bᵀ)
            + offs_n[None, :] * stride_b_n   # N dim (col of Bᵀ)
        )

        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16)

        # dot over K dimension, accumulate in fp32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # write back
    c_ptrs = (
        C_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)

def batched_matmul_fp16(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for a batch of matrices.
    A, B: [batch, M, K] and [batch, N, K] (half tensors, contiguous).
    Returns C: [batch, M, N] (float32).
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # strides in elements
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_n, stride_b_k = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )

    batched_matmul_fp16_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return C

def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.
    Implements the core O(N³) contraction using a fused Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    # config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # Load weights
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1. LayerNorm on input
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------
    # 3. Optional mask
    # --------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gates (sigmoid) applied before multiplication
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Cast to half for the heavy O(N³) contraction
    # --------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # --------------------------------------------------------------
    # 6. Fold batch and hidden dimensions together and launch Triton kernel
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to [B, H, N, N] then view as [B*H, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_flat = right.permute(0, 3, 1, 2).contiguous()

    batch_hidden = B * H
    left_mat = left_flat.view(batch_hidden, N, N)   # [BH, N, N]
    right_mat = right_flat.view(batch_hidden, N, N)

    # core contraction: out = left @ rightᵀ
    out_mat = batched_matmul_fp16(left_mat, right_mat)   # [BH, N, N] (float32)

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    # final linear back to original dim
    output = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return output

"""
Outgoing Triangle Multiplicative Update (TriMul) – forward only.

The heavy O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is expressed as a batched matmul C = A @ Bᵀ where each batch corresponds to a
pair (batch, hidden_dim).  A and B are stored in half‑precision, the kernel
loads tiles, uses `tl.dot` with FP32 accumulation and writes the result in
float32.  The rest of the pipeline (LayerNorms, linear projections, gates)
is performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_fp16_kernel(
    A_ptr,  # *fp16   [batch, M, K]
    B_ptr,  # *fp16   [batch, N, K]  (read as K×N for the transpose)
    C_ptr,  # *fp32   [batch, M, N]
    M, N, K,                     # i32
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,

    BLOCK_N: tl.constexpr,

    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask:        torch.Tensor [B, N, N] or None
            - weights:     dict of model parameters
            - config:      dict with at least "dim" and "hidden_dim"

        torch.Tensor: output tensor of shape [B, N, N, dim] (float32)
    """

    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Extract weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Apply mask if present (mask shape [B, N, N])
    # ------------------------------------------------------------------
    if mask is not None:

        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # 5. Cast to half‑precision for the O(N³) part
    # ------------------------------------------------------------------
    left = left.to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Fold batch and hidden dimensions and launch the Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # [B, H, N, N] → contiguous layout
    left_flat = left.permute(0, 3, 1, 2).contiguous()
    right_flat = right.permute(0, 3, 1, 2).contiguous()

    batch_hidden = B * H
    left_mat = left_flat.view(batch_hidden, N, N)   # [BH, N, N] (fp16)
    right_mat = right_flat.view(batch_hidden, N, N)  # [BH, N, N] (fp16)

    # Core contraction: out = left @ rightᵀ
    out_mat = _batched_matmul_fp16(left_mat, right_mat)  # [BH, N, N] (fp32)

    # Restore original layout [B, N, N, hidden]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    output = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return output
