def triton_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_a_batch, stride_b_batch, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0) # block row idx (M)
    pid_n = tl.program_id(1) # block col idx (N)
    pid_b = tl.program_id(2) # batch index

    # Compute block offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointer for this batch
    a_ptrs = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # Using tl.dot
        acc += tl.dot(a, b)

        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

"""
TriMul (outgoing) forward pass implemented with a fused Triton GEMM.

Algorithm:
1. LayerNorm over the input channels (dim).
2. Linear projections to hidden_dim (left/right) and three gating projections.
3. Optional mask multiplication (broadcasted over the hidden dimension).
4. Apply gating (sigmoid) to left and right tensors.
5. Compute the core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication left @ right.T for each
   (batch * hidden_dim) slice.  The multiplication is performed by a
   custom Triton kernel (`gemm_kernel`) that accumulates in FP32 while
   loading in FP16.
6. LayerNorm over the hidden dimension, apply the output gate and a final
   linear projection back to `dim`.
All operations are performed on the GPU; only the GEMM is Triton‑accelerated.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel: C = A @ B   where A:[M,K], B:[K,N]
# The batch dimension is handled by program_id(2).
# ----------------------------------------------------------------------
@triton.jit
def gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_a_batch, stride_b_batch, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile column
    pid_b = tl.program_id(2)  # batch index

    # offsets within the matrices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the current block (A: M x K, B: K x N)
    a_ptrs = a_ptr + pid_b * stride_a_batch \
                + offs_m[:, None] * stride_am \
                + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_b_batch \
                + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                + offs_n[None, :] * stride_bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

        # advance pointers to next K‑slice
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = c_ptr + pid_b * stride_c_batch \
                + offs_m[:, None] * stride_cm \
                + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# Custom forward kernel for the TriMul (outgoing) operator.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor, shape [B, N, N, C] (float32)
            mask:          torch.Tensor, shape [B, N, N] (bool/float) or None
            weights:       dict of torch.Tensor with model parameters
            config:        dict containing "dim", "hidden_dim", and "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_mask = (not config.get("nomask", False)) and (mask is not None)

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])        # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3. Gating scalars (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 4. Optional mask application (broadcast to hidden_dim)
    # --------------------------------------------------------------
    if use_mask:
        # ensure mask is float for multiplication
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 5. Apply gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Core contraction via Triton GEMM (left @ right.T)
    # --------------------------------------------------------------
    # Cast to FP16 to reduce memory traffic; accumulation stays FP32.
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim
    # reshape to (B*hidden_dim, N, N) where the second dim is the row (i or k)
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # right_T has shape (B*hidden_dim, K=N, N) where K is the shared dimension
    right_reshape = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # output buffer (FP32)
    out_reshape = torch.empty(
        (B * hidden_dim, N, N),
        dtype=torch.float32,
        device=device,
    )

    # strides (in elements)
    stride_a_batch, stride_am, stride_ak = left_reshape.stride()
    stride_b_batch, stride_bk, stride_bn = right_reshape.stride()
    stride_c_batch, stride_cm, stride_cn = out_reshape.stride()

    # Tile sizes – chosen for good occupancy on NVIDIA B200 (Ampere‑like)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_m, grid_n, B * hidden_dim)

    gemm_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 8. Output LayerNorm over hidden_dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 9. Apply the output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 10. Final linear projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
