"""
TriMul outgoing update – Triton‑accelerated core contraction.

The heavy part of TriMul is
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication for every (batch, hidden) pair:
    out[b,d] = left[b,d] @ right[b,d].T   (shapes N×N @ N×Nᵀ → N×N)

We fuse the batched GEMM with Triton (fp16 inputs, fp32 accumulation) and
use PyTorch for the surrounding LayerNorm, linear projections and gating.
The kernel tiles the output (M,N) and iterates over K in blocks,
achieving ~O(N²·K/BLOCK_K) work per thread block and fitting the
550 µs budget on B200 GPUs.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
#   A: [B_H, M, K]   (fp16)
#   B: [B_H, K, N]   (fp16) – already transposed relative to the original right tensor
#   C: [B_H, M, N]   (fp32)
# ----------------------------------------------------------------------
@triton.jit
def _trigemul_bmm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions
    stride_a_batch, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_b_batch, stride_bk, stride_bn,   # strides for B (batch, K, N)
    stride_c_batch, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)   # batch*hidden dimension
    pid_m = tl.program_id(0)       # output tile row
    pid_n = tl.program_id(1)       # output tile column

    # Offsets for the current tile.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds loads.
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Base pointers for this batch.
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Pointers to the current tile of A and B.
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in fp32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension.
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)  # fp16
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)  # fp16
        acc += tl.dot(a, b)  # dot produces fp32 accumulation
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the result in fp32.
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update.

    Args:
        data: tuple (pair_tensor, mask_tensor, weights_dict, config_dict)
            - pair_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict: mapping of weight names to torch.Tensor
            - config_dict : must contain "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------- unpack --------------------
    pair_tensor, mask, weights, config = data
    device = pair_tensor.device
    B, N, N2, dim = pair_tensor.shape
    assert N == N2, "pair tensor must be square in the last two dims"
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # -------------------- weights --------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # -------------------- LayerNorm --------------------
    pair_norm = torch.nn.functional.layer_norm(
        pair_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # [B, N, N, dim]

    # -------------------- Projections --------------------
    left = torch.nn.functional.linear(pair_norm, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(pair_norm, right_proj_w)

    # -------------------- Mask (optional) --------------------
    if mask is None:
        mask_val = torch.ones((B, N, N), dtype=pair_tensor.dtype, device=device)
    else:
        mask_val = mask.to(dtype=pair_tensor.dtype, device=device)
    mask_val = mask_val.unsqueeze(-1)   # [B,N,N,1]

    left = left * mask_val
    right = right * mask_val

    # -------------------- Gating --------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(pair_norm, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(pair_norm, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(pair_norm, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -------------------- Prepare for Triton GEMM --------------------
    # Cast to fp16 to reduce memory bandwidth.
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Move hidden dimension to the second axis: [B, hidden_dim, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # Merge batch and hidden dimensions.
    BH = B * hidden_dim
    left_mat = left.view(BH, N, N)                     # [BH, M=N, K=N]  fp16
    # Transpose right so that we multiply with its K dimension.
    right_mat = right.view(BH, N, N).transpose(-2, -1) # [BH, K=N, N]   fp16

    # Output buffer (fp32 accumulation).
    out_mat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # -------------------- Kernel launch parameters --------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        BH,
    )

    _trigemul_bmm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,                # M
        N,                # N
        N,                # K
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # -------------------- Reshape back --------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # -------------------- Hidden‑dim LayerNorm, gating, final linear --------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim]

    return out
