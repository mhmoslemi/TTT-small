@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_a_batch, stride_b_batch, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(axis=0)  # M tile
    pid_n = tl.program_id(axis=1) # N tile
    pid_b = tl.program_id(axis=2) # batch tile

    # compute offsets:
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # masked out-of-bounds:
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute base pointers
    a_base = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_base = b_ptr + pid_b * stride_b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_base, mask=mask_m[:, None] & mask_k[None,:], other=0.0) # a shape (BLOCK_M, BLOCK_K)
        b = tl.load(b_base, mask=mask_k[:, None] & mask_n[None,:], other=0.0) # b shape (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)

        # advance ptrs
        a_base += BLOCK_K * stride_ak
        b_base += BLOCK_K * stride_bk

    # store result
    c_ptrs = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel
==============

Implements the forward pass of the “outgoing” TriMul operator (AlphaFold3).

Algorithm
---------
1. Layer‑norm the 4‑D input tensor.
2. Linear projection to a hidden dimension (left/right) without bias.
3. Optional mask multiplication (broadcast over the hidden dimension).
4. Compute three sigmoid gates (left, right, out) from the normalized input.
5. Apply the left/right gates.
6. Batched matrix multiplication:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is performed as a batch of GEMMs over the hidden dimension using a
   Triton kernel (half‑precision inputs, fp32 accumulation).
7. Layer‑norm over the hidden dimension.
8. Multiply by the out‑gate.
9. Final linear projection back to the original dimension.

All heavy arithmetic (the N³ contraction) is executed on GPU via Triton,
while the surrounding linear layers / layer‑norms are handled with
torch.nn.functional for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_a_batch, stride_b_batch, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ B, where A:[batch,M,K], B:[batch,K,N]."""
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # batch index (batch * hidden_dim)

    # row / column offsets for the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # pointers for the current K‑slice
        a_ptrs = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_b * stride_b_batch + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load tiles, applying masks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # accumulate (fp16 inputs → fp32 accumulation)
        acc += tl.dot(a, b)

    # store the result tile
    c_ptrs = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of torch.Tensor containing all linear / norm weights
            - config: dict with keys:
                * "dim" (int)
                * "hidden_dim" (int)
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False) and mask is not None

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Mask (if any)
    # --------------------------------------------------------------
    if use_mask:
        # broadcast mask over hidden dimension
        mask_exp = mask.unsqueeze(-1).type_as(left)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # --------------------------------------------------------------
    # 5. Apply gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Heavy contraction via Triton GEMM
    # --------------------------------------------------------------
    # Cast to half for memory bandwidth, keep accumulation in fp32
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to (B, H, N, N) and fuse batch*H dimension.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose the inner N axes -> [B, H, N, N]

    batch_dim = bs * hidden_dim
    left_view = left_perm.view(batch_dim, N, N)
    right_view = right_perm.view(batch_dim, N, N)

    out_view = torch.empty(
        (batch_dim, N, N),
        dtype=torch.float32,
        device=device,
    )

    # strides in elements
    stride_a_batch, stride_am, stride_ak = left_view.stride()
    stride_b_batch, stride_bk, stride_bn = right_view.stride()
    stride_c_batch, stride_cm, stride_cn = out_view.stride()

    # Heuristic block sizes (tuned for typical N on NVIDIA B200)
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch_dim,
    )

    batched_gemm_kernel[grid](
        left_view,
        right_view,
        out_view,
        N,          # M
        N,          # N
        N,          # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    out = out_view.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 9. Apply out‑gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 10. Final projection back to original dimension
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
