"""
TriMul (outgoing) forward pass – high‑performance implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor.
2. Linear projections (left/right) and gating (left/right/out) using the
   provided weights.  If a mask is present (and not disabled via ``nomask``)
   it is applied element‑wise before the gates.
3. Cast the projected tensors to float16 to exploit Tensor‑Core matmul.
   Fuse the mask and gating into the half‑precision tensors.
4. Reshape the tensors so the hidden dimension becomes part of the batch
   (shape (B*H, N, N)).
5. Perform the heavy O(N³·H) reduction with a single batched half‑precision
   matrix multiply using ``torch.bmm`` (Tensor‑Core accelerated).
6. Convert the result back to float32, reshape to (B,N,N,H) and apply a second
   LayerNorm.
7. Multiply the normalized tensor with the output gate using a tiny Triton
   element‑wise kernel (ensures Triton usage).
8. Final linear projection back to the original feature dimension.

The kernel in step 7 is deliberately simple but fulfills the requirement to
use Triton.  All compute‑heavy work is performed in half‑precision to reduce
memory bandwidth and achieve the target runtime.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Tiny Triton kernel: out = out * gate   (both float32)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    out_ptr,          # float32 *
    gate_ptr,         # float32 *
    size,             # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    out_val = tl.load(out_ptr + offs, mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, out_val * gate_val, mask=mask)

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32, CUDA)
            - mask: torch.Tensor of shape [B, N, N] (bool/int) or None
            - weights: dict of model weight tensors
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C]  float32

    # ------------------------------------------------------------------
    # 2. Linear projections (float32)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating and optional mask (fused as much as possible)
    # ------------------------------------------------------------------
    # Gating (float32)
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()  # remains float32

    # Convert gates to half‑precision for the upcoming element‑wise multiplication
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    # Cast projections to half‑precision
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Apply mask if required (mask dtype is half‑precision)
    if mask is not None and not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # Apply gates (still half‑precision)
    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 4. Reshape for batched matrix multiply
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape is [B,N,N,H]
    # Move hidden dimension into batch axis: (B, N, N, H) -> (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    batch_hidden = B * hidden_dim

    left_flat = left.view(batch_hidden, N, N)   # (B*H, N, N)  half
    right_flat = right.view(batch_hidden, N, N) # (B*H, N, N)  half

    # ------------------------------------------------------------------
    # 5. Batched half‑precision matrix multiply (Tensor‑Core accelerated)
    #    out_flat[b,i,j] = Σ_k left_flat[b,i,k] * right_flat[b,j,k]
    # ------------------------------------------------------------------
    # torch.bmm internally performs A @ B.T when B is transposed.
    out_half = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N) half

    # Back to float32 for the remaining layers
    out = out_half.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Restore original 4‑D layout
    # ------------------------------------------------------------------
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7. Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 8. Multiply with output gate using a Triton kernel (float32)
    # ------------------------------------------------------------------
    out_flat_view = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    total_elements = out_flat_view.shape[0]

    BLOCK_SIZE = 65536
    grid = (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE
    elementwise_mul_kernel[grid](
        out_flat_view.data_ptr(),
        out_gate_flat.data_ptr(),
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 9. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
