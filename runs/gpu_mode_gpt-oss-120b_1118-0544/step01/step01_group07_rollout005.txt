TriMul (outgoing) forward pass implemented with Triton.
----------------------------------------------------
The module normalises the input, projects it to a hidden dimension,
applies mask and gated element‑wise modulation, then performs the core
O(N³) triangle multiplicative update:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

The cubic term is a batched matrix multiplication A @ Bᵀ where each
head (batch * hidden_dim) computes an (N×N) @ (N×N)ᵀ.  This is
implemented with a custom Triton kernel that tiles the matrices in
blocks (128×128×32) and accumulates in FP32.  The rest of the
operations (LayerNorms, gating, linear projections) are performed in
PyTorch.  The kernel works for B∈{1,2}, N up to 1024, hidden_dim=128
and runs in FP32 (you may switch to FP16 by casting the inputs to
half before the kernel call).

The public entry point is `custom_kernel(data)` where `data` is a tuple:
(input_tensor, mask, weights, config).  The function returns the
processed tensor of shape [B, N, N, dim] in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr,               # *float32  [batch, M, K]
    B_ptr,               # *float32  [batch, N, K] (read as K×N for transpose)
    C_ptr,               # *float32  [batch, M, N]
    M, N, K,             # i32 dimensions
    stride_ah, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bh, stride_bn, stride_bk,   # strides for B (batch, N, K)
    stride_ch, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_batch = tl.program_id(2)   # batch dimension
    pid_m = tl.program_id(0)       # row block
    pid_n = tl.program_id(1)       # col block

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds checks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # masks for the current tile
        mask_a = mask_m[:, None] & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & mask_n[None, :]

        # pointers to tiles
        a_ptrs = (
            A_ptr
            + pid_batch * stride_ah
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        b_ptrs = (
            B_ptr
            + pid_batch * stride_bh
            + offs_k[:, None] * stride_bk       # K dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bn       # N dimension (col of Bᵀ)
        )

        # load with zero padding
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # dot‑product accumulation
        acc += tl.dot(a, b)

    # write back the result tile
    c_ptrs = (
        C_ptr
        + pid_batch * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)


def batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for batches of matrices using the Triton kernel.
    A : [batch, M, K]  (float32, contiguous)
    B : [batch, N, K]  (float32, contiguous)
    Returns C : [batch, M, N] (float32)
    """
    assert A.is_contiguous() and B.is_contiguous(), "inputs must be contiguous"
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # element‑wise strides (in elements, not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    # Tunable block sizes (chosen for B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),   # program id 0
        triton.cdiv(N, BLOCK_N),   # program id 1
        batch,                     # program id 2
    )

    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return C


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (float32/bool) or None
        - weights      : dict of model parameters (torch.Tensor)
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Extract weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure mask is broadcastable and same dtype
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul: batched matmul over the 'k' dimension
    #    Compute out = Σ_k left[..., i, k, d] * right[..., j, k, d]
    #    This is equivalent to A @ Bᵀ for each head (batch*hidden_dim).
    # ------------------------------------------------------------------
    B, N, _, HH = left.shape  # HH == hidden_dim
    # Rearrange to [batch, hidden, N, N] and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Merge batch and hidden dimensions
    batch_hidden = B * HH
    left_mat = left_perm.view(batch_hidden, N, N)   # [BH, N, N]
    right_mat = right_perm.view(batch_hidden, N, N) # [BH, N, N]

    # Batched matmul A @ Bᵀ via Triton
    out_mat = batched_matmul(left_mat, right_mat)   # [BH, N, N] (FP32)

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, HH, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    # Final linear projection back to original dim
    output = F.linear(out, to_out_weight)  # (dim, hidden_dim) weight

    return output
