"""
TriMul (outgoing) forward – Triton‑accelerated batched matrix multiplication.

The heavy part of the TriMul module is:
    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
which is a batch of GEMMs (left @ rightᵀ) over the hidden dimension `c`.
This implementation:
1. Applies LayerNorm, linear projections and sigmoid gates in PyTorch.
2. Packs the resulting tensors to [B, H, N, N] (H = hidden_dim) and casts to FP16.
3. Launches a custom Triton kernel that computes, for every (b, h) pair,
   the matrix product left[b,h] @ right[b,h].T using tiled
   (BLOCK_M × BLOCK_N × BLOCK_K) blocks.  The kernel accumulates in FP32
   for numerical stability and writes the result in FP16.
4. The output is reshaped back, normalized, gated and projected to the
   original feature dimension.

The kernel uses separate strides for the transposed `right` matrix,
so no explicit transpose is performed in memory.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the outgoing TriMul (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # dimensions (all equal to seq_len)
    stride_am, stride_ak,              # A: row‑stride, col‑stride
    stride_br, stride_bc,              # B (transposed): row‑stride, col‑stride
    stride_cm, stride_cn,              # C: row‑stride, col‑stride
    stride_a_batch, stride_b_batch, stride_c_batch,   # batch‑head stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_z = tl.program_id(2)   # batch * head index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this (batch, head) slice
    a_base = A_ptr + pid_z * stride_a_batch
    b_base = B_ptr + pid_z * stride_b_batch
    c_base = C_ptr + pid_z * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K‑dimension in BLOCK_K chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K) + k_offset

        # Pointers for A (row i, col k)
        a_ptrs = a_base + offs_m[:, None] * stride_am + k_range[None, :] * stride_ak
        # Pointers for Bᵀ (row k, col j) – note the swapped strides
        b_ptrs = b_base + k_range[:, None] * stride_br + offs_n[None, :] * stride_bc

        # Bounds checking
        a_mask = (offs_m[:, None] < M) & (k_range[None, :] < K)
        b_mask = (k_range[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Dot‑product accumulation
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # Write the result
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict with the module parameters
        - config       : dict with at least 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections + gating + optional mask
    # ------------------------------------------------------------------
    # Projections (float32)
    left_proj = F.linear(x, weights["left_proj.weight"])      # [B,N,N,H]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # Gates (float32, sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply mask if present (mask broadcasted over hidden dim)
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)               # [B,N,N,1]
        left = left_proj * mask_f * left_gate
        right = right_proj * mask_f * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Cast to FP16 for the GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Rearrange to [B, H, N, N] and flatten batch*head
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left.view(-1, N, N)     # (B*H, N, N)
    right_mat = right.view(-1, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat)

    # ------------------------------------------------------------------
    # 4) Strides (in elements, not bytes)
    # ------------------------------------------------------------------
    stride_a_batch = left_mat.stride(0)
    stride_am = left_mat.stride(1)     # row stride (i)
    stride_ak = left_mat.stride(2)     # col stride (k)

    stride_b_batch = right_mat.stride(0)
    # For the transposed view of B we swap the last two strides
    stride_br = right_mat.stride(2)    # row stride for Bᵀ (original col stride)
    stride_bc = right_mat.stride(1)    # col stride for Bᵀ (original row stride)

    stride_c_batch = out_mat.stride(0)
    stride_cm = out_mat.stride(1)      # row stride
    stride_cn = out_mat.stride(2)      # col stride

    # ------------------------------------------------------------------
    # 5) Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    _tri_mul_outgoing_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_br,
        stride_bc,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Reshape back to [B, N, N, H] (float32)
    # ------------------------------------------------------------------
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate  # apply output gate

    # Final linear projection back to `dim`
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
