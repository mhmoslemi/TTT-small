"""
custom_kernel for the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Input LayerNorm over the channel dimension (dim).
2. Linear projections (left/right) from dim → hidden_dim, without bias.
3. Optional pairwise mask (broadcasted on hidden_dim).
4. Compute three sigmoid gates (left, right, out) and apply the first two to the
   projected tensors.
5. Core contraction (the O(N³) part):
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batch of matrix multiplications:
       out[b,d] = left[b,d] @ right[b,d].T
   The multiplication is performed by a fused Triton kernel that tiles the
   i, j and k axes (default tile size 64) and accumulates in FP32 while the
   inputs are in FP16.
6. Output LayerNorm over hidden_dim, multiply by the output gate,
   and final linear projection back to dim.
The kernel works directly on the original (B,N,N,hidden_dim) layout,
avoiding costly permutes/copies.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, D,                                    # batch, sequence length, hidden dim
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    using a tiled GEMM for each (b,d) pair.
    """
    pid_head = tl.program_id(0)   # flattened (batch, hidden_dim) index
    pid_i    = tl.program_id(1)   # block row (i)
    pid_j    = tl.program_id(2)   # block column (j)

    # recover batch and hidden indices
    d = pid_head % D
    b = pid_head // D

    # base pointers for this (b,d) slice
    left_base  = left_ptr  + b * stride_left_batch  + d * stride_left_d
    right_base = right_ptr + b * stride_right_batch + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_batch   + d * stride_out_d

    # tile offsets for i and j
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # reduction over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,i,k,d]   shape (BLOCK_I, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & mask_k[None, :]),
                    other=0.0)

        # right[b,j,k,d] transposed -> right_T[k,j,d] (BLOCK_K, BLOCK_J)
        # original layout: right[b, j, k, d]
        b_ptrs = right_base + offs_j[None, :] * stride_right_i + offs_k[:, None] * stride_right_k
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_j[None, :]),
                    other=0.0)

        # accumulate (FP16 -> FP32)
        acc += tl.dot(a, b)

    # write result
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=(mask_i[:, None] & mask_j[None, :]))


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask: torch.Tensor or None, shape [B, N, N]
            - weights: dict of torch.Tensor containing all model parameters.
            - config: dict with keys "dim" and "hidden_dim".

    Returns:
        torch.Tensor: output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Gating tensors
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Apply gates (and mask if present)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    if mask is not None:
        # broadcast mask over hidden_dim
        mask_f = mask.unsqueeze(-1).type_as(x)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton kernel
    # ------------------------------------------------------------------
    # Cast to half to exploit Tensor Cores
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, D = left_h.shape  # B, N, N, hidden_dim

    # output buffer (float32 accumulator)
    out = torch.empty((B, N, N, D), dtype=torch.float32, device=x.device)

    # strides
    s_lb, s_li, s_lk, s_ld = left_h.stride()
    s_rb, s_ri, s_rk, s_rd = right_h.stride()
    s_ob, s_oi, s_oj, s_od = out.stride()

    # tiling parameters – 64 works well for the tested sizes
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    grid = (
        B * D,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    _tri_mul_outgoing_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        D,
        s_lb,
        s_li,
        s_lk,
        s_ld,
        s_rb,
        s_ri,
        s_rk,
        s_rd,
        s_ob,
        s_oi,
        s_oj,
        s_od,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output processing (LayerNorm, gating, final linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
