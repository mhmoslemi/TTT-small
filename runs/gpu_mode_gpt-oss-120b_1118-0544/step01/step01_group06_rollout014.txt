"""
TriMul (outgoing) forward pass with a fused batched GEMM kernel in Triton.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension (dim).
2. Linear projections (left/right) from dim → hidden_dim (no bias).
3. Apply an optional mask (broadcasted on hidden_dim).
4. Compute three sigmoid gates (left, right, out) from the original input.
   Multiply the left/right projections by their respective gates.
5. Compute the core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication for each head d:
       out[b,d] = left[b,d] @ right[b,d].T
   Implemented as a tiled FP16 GEMM (accumulating in FP32) using Triton.
6. Layer‑norm the result over hidden_dim, multiply by the output gate,
   and apply a final linear projection back to dim.

The kernel tiles the M (i), N (j) and K (k) axes using configurable
block sizes and autotunes for the best configuration.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 64, 'BLOCK_K': 32},
                      num_stages=3, num_warps=8),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 64},
                      num_stages=2, num_warps=8),
        triton.Config({'BLOCK_M': 32, 'BLOCK_N': 32, 'BLOCK_K': 32},
                      num_stages=4, num_warps=4),
    ],
    key=['M', 'N', 'K']
)
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                     # matrix dimensions
    stride_ah, stride_am, stride_ak,   # a: (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # b: (batch, K, N)
    stride_ch, stride_cm, stride_cn,   # c: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B  (A: MxK, B: KxN) for each batch/head."""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute global offsets of the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers offset to the current batch/head
    a_batch_ptr = a_ptr + pid_batch * stride_ah
    b_batch_ptr = b_ptr + pid_batch * stride_bh
    c_batch_ptr = c_ptr + pid_batch * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Compute masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Load A tile (M x K) and B tile (K x N)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # fp16

        # Multiply‑accumulate (fp16 * fp16 -> fp32)
        acc += tl.dot(a, b)

    # Write back the result (fp32)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


# ----------------------------------------------------------------------
# Entry point: custom Triton‑accelerated TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: Tuple containing
            - input tensor (B, N, N, dim)      torch.Tensor
            - mask tensor (B, N, N) or None    torch.Tensor or None
            - weights dict with all required parameters
            - config dict with keys "dim" and "hidden_dim"

    Returns:
        Tensor of shape (B, N, N, dim) (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs (mask may be omitted)
    # --------------------------------------------------------------
    if len(data) == 4:
        x, mask, weights, config = data
    else:  # mask omitted
        x, weights, config = data
        mask = None

    device = x.device
    dtype = x.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B, N, N, H)
    right = F.linear(x, weights["right_proj.weight"])   # (B, N, N, H)

    # --------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden dimension)
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.to(dtype=x.dtype).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # (B, N, N, H)

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction via fused batched GEMM (Triton)
    # --------------------------------------------------------------
    # Cast to FP16 for the kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Re‑order dimensions so that the reduction axis 'k' is contiguous:
    # left  -> (B, H, i, k)
    # right -> (B, H, k, j)  (i.e. transposed for matmul)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape

    # Collapse batch and head dimensions for the kernel
    left_mat = left_perm.view(B * H, N, N)   # (BH, M=i, K=k)
    right_mat = right_perm.view(B * H, N, N) # (BH, K=k, N=j)

    # Output buffer in FP32 (accumulator)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # Choose a reasonable block size; autotuning will pick the best.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (batch*head, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch kernel
    tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                                # M, N, K
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gate and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
