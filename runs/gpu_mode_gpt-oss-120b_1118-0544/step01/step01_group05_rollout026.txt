```python
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ac,
    stride_bm, stride_bn, stride_bc,
    stride_cm, stride_cn, stride_cc,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_am
    b_batch = b_ptr + pid_batch * stride_bm
    c_batch = c_ptr + pid_batch * stride_cm

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K
        # load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_ak + cur_k[None, :] * stride_ac
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B^T (K,N) -> we need B's row stride and col stride swapped
        b_ptrs = b_batch + cur_k[:, None] * stride_bc + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_batch + offs_m[:, None] * stride_cn + offs_n[None, :] * stride_cc
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ac,
    stride_bm, stride_bn, stride_bc,
    stride_cm, stride_cn, stride_cc,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)  # batch * hidden_dim
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers
    a_batch = a_ptr + pid_batch * stride_am
    b_batch = b_ptr + pid_batch * stride_bm
    c_batch = c_ptr + pid_batch * stride_cm

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

def custom_kernel(data):
    """
    Triton‐accelerated TriMul (outgoing) forward pass.
    The O(N³) contraction is expressed as a batched GEMM:
        out[b, d] = left[b, d] @ right[b, d].t()
    where left/right are reshaped to (B*D, N, N) and the
    multiply is performed by a Triton kernel.
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm over dim
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # Linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half for the contraction
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # reshape to (B*hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous()
    left_mat = left_mat.view(B * hidden_dim, N, N)

    right_mat = right.permute(0, 3, 1, 2).contiguous()
    right_mat = right_mat.view(B * hidden_dim, N, N)

    # Output buffer
    out_mat = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    stride_am = N * N
    stride_ak = N
    stride_ac = 1

    stride_bm = N * N
    stride_bn = N
    stride_bc = 1

    stride_cm = N * N
    stride_cn = N
    stride_cc = 1

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_ac=stride_ac,
        stride_bm=stride_bm,
        stride_bn=stride_bn,
        stride_bc=stride_bc,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_cc=stride_cc,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # Second LayerNorm over hidden_dim
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass accelerated with Triton.

The heavy O(N³) contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is rewritten as a batched matrix multiplication
    C = A @ Bᵀ
with A and B shaped (B·D, N, N) where D = hidden_dim.
Each batch‑hidden slice performs a GEMM of size N×N.
The kernel below computes this GEMM in half precision (fp16) while
accumulating in fp32 for numerical stability.
All other operations (LayerNorms, linear projections, gating) are done in
PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ac,
    stride_bm, stride_bn, stride_bc,
    stride_cm, stride_cn, stride_cc,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ  (A,B : fp16, C : fp32).

    Shapes (per batch*hidden slice):
        A : (M, K)   row‑major
        B : (N, K)   row‑major (will be accessed as K×N = Bᵀ)
        C : (M, N)   row‑major
    """
    pid_batch = tl.program_id(0)      # batch * hidden_dim index
    pid_m = tl.program_id(1)          # tile over rows   (M)
    pid_n = tl.program_id(2)          # tile over cols   (N)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch slice
    a_batch = a_ptr + pid_batch * stride_am
    b_batch = b_ptr + pid_batch * stride_bm
    c_batch = c_ptr + pid_batch * stride_cm

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_offs = a_batch + offs_m[:, None] * stride_ak + cur_k[None, :] * stride_ac
        a = tl.load(a_offs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Bᵀ tile: (BLOCK_K, BLOCK_N)
        # original B is (N, K); we index as (k, n) → stride_bc for k, stride_bn for n
        b_offs = b_batch + cur_k[:, None] * stride_bc + offs_n[None, :] * stride_bn
        b = tl.load(b_offs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # store the result
    c_offs = c_batch + offs_m[:, None] * stride_cn + offs_n[None, :] * stride_cc
    tl.store(c_offs, acc, mask=mask_m[:, None] & mask_n[None, :])

    # reshape to (B*hidden_dim, N, N) with hidden_dim as the batch axis
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # output buffer (fp32 accumulator)
    out_mat = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------------------
    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # strides in number of elements (row‑major layout)
    stride_am = N * N        # batch stride for A
    stride_ak = N           # row stride for A (M dimension)
    stride_ac = 1           # column stride for A (K dimension)

    stride_bm = N * N        # batch stride for B
    stride_bn = N           # row stride for B (N dimension)
    stride_bc = 1           # column stride for B (K dimension)

    stride_cm = N * N        # batch stride for C
    stride_cn = N           # row stride for C (M dimension)
    stride_cc = 1           # column stride for C (N dimension)

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_ac=stride_ac,
        stride_bm=stride_bm,
        stride_bn=stride_bn,
        stride_bc=stride_bc,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_cc=stride_cc,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 6) Reshape back to the original 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # 7) Second LayerNorm over the hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # 8) Apply the output gate and final linear projection back to `dim`
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
