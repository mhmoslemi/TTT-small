def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # Linear Layers weights shapes:
    # left_proj_weight: (hidden_dim, dim)
    # right_proj_weight: (hidden_dim, dim)
    # left_gate_weight: (hidden_dim, dim)
    # right_gate_weight: (hidden_dim, dim)
    # out_gate_weight: (hidden_dim, dim)
    # to_out_norm_weight/bias: shape hidden_dim
    # to_out_weight: (dim, hidden_dim)
    # norm weight: shape dim
    # norm bias: shape dim

    # Compute LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Compute projections
    left = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Mask if needed
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gating
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape # B,N,N,hidden_dim

    # Reshape for kernel: (B * hidden_dim, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()
    left_h = left_h.view(B * hidden_dim, N, N)
    right_h = right_h.view(B * hidden_dim, N, N)

    # Allocate output
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # Define grid
    BLOCK_M = 64; BLOCK_N = 64; BLOCK_K = 32
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    batch = B * hidden_dim
    grid = (num_pid_m, num_pid_n, batch)

    # Launch kernel
    tri_mul_outgoing_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        batch=batch,
        M=N,
        N=N,
        K=N,
        stride_a_batch=left_h.stride(0),
        stride_a_m=left_h.stride(1),
        stride_a_k=left_h.stride(2),
        stride_b_batch=right_h.stride(0),
        stride_b_k=right_h.stride(2),  # B's K dimension stride
        stride_b_n=right_h.stride(1),  # B's N dimension stride (row)
        stride_c_batch=out_h.stride(0),
        stride_c_m=out_h.stride(1),
        stride_c_n=out_h.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K
    )

    # Reshape back to [B,N,N,hidden_dim]
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # Final layernorm and gating
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)
    out = out * out_gate   # out_gate shape [B,N,N,hidden_dim] from earlier
    out = F.linear(out, to_out_weight)   # final projection to dim

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) kernel.
The core operation computes, for each batch and hidden channel,
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiplication
    C = A @ Bᵀ
with A,B ∈ ℝ^{(B·D) × N × N} (D = hidden_dim, N = seq_len).
We perform layer‑norms, linear projections and gating in PyTorch and
use a fused Triton kernel for the heavy N³ reduction, operating in
float16 with float32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left tensor [batch, M, K] (float16)
    b_ptr,               # right tensor [batch, N, K] (float16) – treated as transposed
    c_ptr,               # output tensor [batch, M, N] (float16)
    batch: tl.constexpr, # total independent matrices = B * hidden_dim
    M: tl.constexpr,     # rows of A (also rows of output)
    N: tl.constexpr,     # rows of B (also cols of output)
    K: tl.constexpr,     # reduction dimension (seq_len)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul C = A @ Bᵀ for outgoing TriMul."""
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # batch index (B * hidden_dim)

    # Offsets within this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch slice
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_offs = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offs < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (k_offs[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load Bᵀ tile (reading B as [K, N])
        b_ptrs = b_batch + (k_offs[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute partial dot product; cast to FP32 before accumulation
        acc += tl.dot(a, b).to(tl.float32)

    # Write the result
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float) – may be unused
            - weights: dict of torch.Tensors for all linear and layer‑norm parameters
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm on input (over last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Linear projections (float32)
    # ------------------------------------------------------------------
    proj_left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    proj_right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        proj_left = proj_left * mask_f
        proj_right = proj_right * mask_f

    # ------------------------------------------------------------------
    # Gating values (float32)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))     # used after the matmul

    # ------------------------------------------------------------------
    # Apply gates and cast to half for the reduction
    # ------------------------------------------------------------------
    left = (proj_left * left_gate).to(torch.float16)   # [B,N,N,hidden_dim] half
    right = (proj_right * right_gate).to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # ------------------------------------------------------------------
    # Rearrange tensors to (batch * hidden_dim, N, N) for the kernel
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    left = left.view(B * hidden_dim, N, N)
    right = right.view(B * hidden_dim, N, N)

    # Output buffer (half)
    out_h = torch.empty_like(left, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Triton kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    batch = B * hidden_dim

    grid = (num_pid_m, num_pid_n, batch)

    # Strides in number of elements
    stride_a_batch = left.stride(0)
    stride_a_m = left.stride(1)
    stride_a_k = left.stride(2)

    stride_b_batch = right.stride(0)
    # Treat right as transposed inside the kernel
    stride_b_k = right.stride(2)   # column stride (fastest)
    stride_b_n = right.stride(1)   # row stride

    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    # ------------------------------------------------------------------
    # Launch the batched matmul kernel
    # ------------------------------------------------------------------
    tri_mul_outgoing_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out_h,
        batch=batch,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] and convert to float32
    # ------------------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)           # [B, hidden_dim, N, N]
        .permute(0, 2, 3, 1)                     # [B, N, N, hidden_dim]
        .contiguous()
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # Final LayerNorm, gating and projection to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate               # element‑wise gating (float32)
    out = F.linear(out, to_out_weight) # [B, N, N, dim]

    return out
