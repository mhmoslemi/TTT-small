TriMul (outgoing) forward pass with a fused Triton kernel.

The core contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication for each hidden dimension d.
We:
1. Apply LayerNorm and linear projections (PyTorch).
2. Apply optional mask and sigmoid gates (PyTorch).
3. Run a custom Triton kernel that computes
       C = A @ Bᵀ  (A = left, B = right)
   where A and B are reshaped to (B*hidden_dim, N, N) and B is already
   transposed, so a single GEMM per hidden slice produces the desired
   tensor.
4. Apply the output LayerNorm, output gate and final linear projection
   (PyTorch).

The kernel works with half‑precision inputs (fp16) and accumulates in
float32, achieving high bandwidth while keeping final results in fp32.

# Triton GEMM kernel for the TriMul contraction.
# It computes C = A @ B  where A and B are (batch, M, K) and (batch, K, N)
# respectively.  All tensors are contiguous in row‑major order.
# ----------------------------------------------------------------------
@triton.jit
def trmul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions
    stride_ah, stride_am, stride_ak,   # A strides: batch, row (M), col (K)
    stride_bh, stride_bk, stride_bn,   # B strides: batch, row (K), col (N)
    stride_ch, stride_cm, stride_cn,   # C strides: batch, row (M), col (N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_h = tl.program_id(2)   # batch * hidden_dim

    # ---------------------------------------------------------------
    # Tile offsets
    # ---------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---------------------------------------------------------------
    # Accumulator
    # ---------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over K dimension
    # ---------------------------------------------------------------
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M x K)
        a = tl.load(
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_am
            + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (K x N)
        b = tl.load(
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate
        acc += tl.dot(a, b)

    # ---------------------------------------------------------------
    # Store C tile
    # ---------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the TriMul (“outgoing”) operator.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of TriMul (outgoing) using a fused Triton GEMM.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = (not config.get("nomask", False)) and (mask is not None)

    # ---------------------------------------------------------------
    # 1. Input LayerNorm
    # ---------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ---------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ---------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ---------------------------------------------------------------
    # 4. Optional mask
    # ---------------------------------------------------------------
    if use_mask:
        mask_exp = mask.unsqueeze(-1).type_as(left)  # [bs, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------------------------------------------------------------
    # 5. Apply gates
    # ---------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------------
    # 6. Cast to fp16 for the GEMM (reduce bandwidth)
    # ---------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ---------------------------------------------------------------
    # 7. Reshape to (B*hidden_dim, N, N) – left stays as‑is,
    #    right must be transposed on the last two axes.
    # ---------------------------------------------------------------
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    # right_T[i,k,j] = right[i,j,k]; we achieve this by swapping the last two axes
    right_reshape = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)

    # ---------------------------------------------------------------
    # 8. Allocate output buffer (float32 accumulator)
    # ---------------------------------------------------------------
    out_reshape = torch.empty(
        (bs * hidden_dim, N, N),
        dtype=torch.float32,
        device=device,
    )

    # ---------------------------------------------------------------
    # 9. Extract strides (in element units, not bytes)
    # ---------------------------------------------------------------
    stride_a_batch, stride_am, stride_ak = left_reshape.stride()
    stride_b_batch, stride_bk, stride_bn = right_reshape.stride()
    stride_c_batch, stride_cm, stride_cn = out_reshape.stride()

    # ---------------------------------------------------------------
    # 10. Choose block sizes (tuned for B200 FP16 tensor cores)
    # ---------------------------------------------------------------
    # Reasonable defaults that work for the whole test matrix‑size range.
    if N <= 128:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32

    # Number of warps needed for the chosen tile size.
    num_warps = 4 if N >= 256 else 2

    # ---------------------------------------------------------------
    # 11. Grid configuration
    # ---------------------------------------------------------------
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )

    # ---------------------------------------------------------------
    # 12. Launch Triton kernel
    # ---------------------------------------------------------------
    trmul_gemm_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N,
        N,
        N,
        stride_a_batch,
        stride_am,
        stride_ak,
        stride_b_batch,
        stride_bk,
        stride_bn,
        stride_c_batch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # ---------------------------------------------------------------
    # 13. Reshape back to [bs, N, N, hidden_dim]
    # ---------------------------------------------------------------
    out = out_reshape.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------------------------------------------------
    # 14. Output LayerNorm over hidden dimension
    # ---------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ---------------------------------------------------------------
    # 15. Output gate
    # ---------------------------------------------------------------
    out = out * out_gate

    # ---------------------------------------------------------------
    # 16. Final linear projection back to original dim
    # ---------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
