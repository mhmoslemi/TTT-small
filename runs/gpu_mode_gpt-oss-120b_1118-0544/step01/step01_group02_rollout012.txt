import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, N,
                           BLOCK_SIZE: tl.constexpr):
    # a_ptr,b_ptr,out_ptr are 1‑D contiguous pointers
    idx = tl.program_id(0) * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = idx < N
    a = tl.load(a_ptr + idx, mask=mask, other=0.0)
    b = tl.load(b_ptr + idx, mask=mask, other=0.0)
    tl.store(out_ptr + idx, a * b, mask=mask)

def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    This implementation fuses the O(N³·H) reduction with a cuBLAS batched matrix multiplication
    (performed in half‑precision) and uses a tiny Triton kernel for the final elementwise
    out‑gate multiplication. All other operations (LayerNorm, linear projections, gating) are
    expressed with PyTorch primitives.
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B, N, N, dim]

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])    # [B,N,N,hidden]

    # ---------- Optional masking ----------
    if mask is not None and not nomask:
        # mask shape [B,N,N] -> broadcast to hidden dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to half for the heavy O(N³·H) reduction ----------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ---------- Prepare batched matmul (B*H, N, N) ----------
    B, N, _, H = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)
    batch_hidden = B * H
    left_flat = left.view(batch_hidden, N, N)       # (B*H, N, N)
    right_flat = right.view(batch_hidden, N, N)     # (B*H, N, N)

    # ---------- Batched matrix multiplication (half‑precision) ----------
    # Compute left @ rightᵀ for each hidden slice.
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Cast back to float32 and reshape to (B,N,N,H)
    out = out_flat.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Output layer‑norm ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ---------- Apply output gate via Triton ----------
    # Both tensors are contiguous 1‑D views.
    out_flat_view = out.view(-1).contiguous()
    out_gate_flat_view = out_gate.view(-1).contiguous()
    numel = out_flat_view.numel()
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(numel, BLOCK_SIZE),)
    elementwise_mul_kernel[grid](
        out_flat_view,
        out_gate_flat_view,
        out_flat_view,
        numel,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # Restore shape after in‑place multiplication.
    out = out_flat_view.view(B, N, N, hidden_dim)

    # ---------- Final linear back to original dimension ----------
    out = F.linear(out, weights["to_out.weight"])

    return out

import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, N, BLOCK_SIZE: tl.constexpr):
    """
    Simple element‑wise multiplication kernel:
        out[i] = a[i] * b[i]
    All tensors are assumed to be 1‑D contiguous buffers.
    """
    idx = tl.program_id(0) * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = idx < N
    a = tl.load(a_ptr + idx, mask=mask, other=0.0)
    b = tl.load(b_ptr + idx, mask=mask, other=0.0)
    tl.store(out_ptr + idx, a * b, mask=mask)


def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    The O(N³·H) reduction is performed with a half‑precision batched matrix
    multiplication (cuBLAS via torch.bmm). A tiny Triton kernel handles the
    final element‑wise out‑gate multiplication. All remaining operations
    (LayerNorm, linear projections, gating) are expressed with PyTorch
    primitives.
    """
    # ----------------------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------------------
    # 1. Input LayerNorm
    # ----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ----------------------------------------------------------------------
    # 2. Linear projections (left / right)
    # ----------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"]) # [B, N, N, hidden]

    # ----------------------------------------------------------------------
    # 3. Optional mask application (before gating)
    # ----------------------------------------------------------------------
    if mask is not None and not nomask:

        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------------------
    # 4. Gating
    # ----------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5. Cast to half‑precision for the heavy reduction
    # ----------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ----------------------------------------------------------------------
    # 6. Batched matrix multiplication (B*H, N, N) -> (B*H, N, N)

    # ----------------------------------------------------------------------
    B, N, _, H = left.shape
