@triton.jit
def tri_mul_kernel(a_ptr, b_ptr, c_ptr,
                   M, N, K,
                   stride_a_batch, stride_a_row, stride_a_col,
                   stride_b_batch, stride_b_row, stride_b_col,
                   stride_c_batch, stride_c_row, stride_c_col,
                   BLOCK_M: tl.constexpr,
                   BLOCK_N: tl.constexpr,
                   BLOCK_K: tl.constexpr):
    # Get program IDs
    pid_batch = tl.program_id(0)   # which hidden dimension slice (including batch)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Determine mask for outs-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=tl.float16(0.0))

        # Load B block (N x K), but we need transpose (K x N) for dot
        # B original layout: (N, K), we want (K, N) -> using stride swapped
        b_ptrs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=tl.float16(0.0))

        # Dot product
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    # Retrieve dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if any
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel (except we need to keep fp32 for gating already used)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B,N,N,hidden_dim

    # Permute to (B, hidden_dim, N, N) and flatten batch*hidden_dim to (B*hidden_dim, N, N)
    left_mat = left_h.permute(0, 3, 1, 2).contiguous()
    right_mat = right_h.permute(0, 3, 1, 2).contiguous()
    # now shape (B, hidden_dim, N, N)
    left_mat = left_mat.view(B * hidden_dim, N, N)
    right_mat = right_mat.view(B * hidden_dim, N, N)

    # Output buffer
    out_mat = torch.empty((B * hidden_dim, N, N), device=input_tensor.device, dtype=torch.float32)

    # Choose block sizes based on N
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
    elif N <= 512:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 128

    # Strides (in elements)
    stride_a_batch = N * N
    stride_a_row = N
    stride_a_col = 1

    stride_b_batch = N * N
    stride_b_row = N
    stride_b_col = 1

    stride_c_batch = N * N
    stride_c_row = N
    stride_c_col = 1

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm over hidden_dim
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply out_gate
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) Triton implementation.

Computes, for each batch b, sequence indices i, j and channel d:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where
    left  = (x_norm @ left_proj.T) * left_gate  (masked if needed)
    right = (x_norm @ right_proj.T) * right_gate (masked if needed)

The contraction is a batched matrix multiplication C = A @ Bᵀ with
A and B of shape (batch*hidden_dim, N, N).  A single Triton kernel
tiles the output in 2‑D (M×N) blocks and iterates over K in
BLOCK_K steps, using fp16 loads and fp32 accumulation.
All other operations (LayerNorm, linear layers, gating) are performed
with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_batch = tl.program_id(0)  # batch * hidden_dim
    pid_m = tl.program_id(1)      # tile over rows (i)
    pid_n = tl.program_id(2)      # tile over cols (j)

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this (batch, hidden) slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_offs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(
            a_offs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
            dtype=tl.float16,
        )

        # Load B block transposed on the fly: (BLOCK_K, BLOCK_N)
        # Original B layout is (N, K); we access as (K, N)
        b_offs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(
            b_offs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0.0),
            dtype=tl.float16,
        )

        # Matrix multiply–accumulate
        acc += tl.dot(a, b)

    # Store the result tile
    c_offs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_offs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update.

    Arguments:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (all torch.Tensors on same device)
            - config: dict containing at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1) LayerNorm on input
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask (applied before gating)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid after linear)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Prepare for Triton: cast to fp16 and reshape for batched GEMM
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, seq_len, seq_len, hidden_dim

    # reshape to (B*hidden_dim, N, N) where each slice is a matrix to multiply
    left_mat = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # output buffer (fp32 accumulator)
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # -----------------------------------------------------------------
    # 6) Launch Triton kernel for the O(N³) contraction
    # -----------------------------------------------------------------
    # Choose block sizes based on sequence length (heuristic)
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
    elif N <= 512:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 128

    # Strides in number of elements (tensors are contiguous after view)
    stride_a_batch = N * N
    stride_a_row = N
    stride_a_col = 1

    stride_b_batch = N * N
    stride_b_row = N
    stride_b_col = 1

    stride_c_batch = N * N
    stride_c_row = N
    stride_c_col = 1

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # -----------------------------------------------------------------
    # 7) Reshape back to original layout [B, N, N, hidden_dim]
    # -----------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8) Output LayerNorm, gating, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
