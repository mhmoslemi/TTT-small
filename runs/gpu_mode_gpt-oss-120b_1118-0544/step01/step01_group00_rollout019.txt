"""
Outgoing Triangle Multiplicative Update (TriMul) – optimized forward pass.

The heavy O(N³·hidden_dim) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication C = A @ Bᵀ with a
custom Triton kernel that works on FP16 inputs and accumulates in FP32.
All other operations (LayerNorm, linear projections, gates) are performed
with PyTorch (cuBLAS), which is already highly tuned.  The kernel tiling
uses 128×128 tiles in the output matrix and 64‑wide reduction blocks,
giving good occupancy on NVIDIA B200 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr,                     # left  tensor,  (batch, M, K)   – FP16
    b_ptr,                     # right tensor, (batch, N, K)   – FP16 (will be read transposed)
    c_ptr,                     # output tensor, (batch, M, N) – FP16
    M: tl.constexpr,           # rows of A / output
    N: tl.constexpr,           # rows of B / output cols
    K: tl.constexpr,           # reduction dimension
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # batch index (B * hidden_dim)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=tl.float16(0.0))

        # Bᵀ tile: we read B as (K, N) by swapping strides
        b_ptrs = b_batch + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=tl.float16(0.0))

        # dot product, accumulation in FP32
        acc += tl.dot(a, b).to(tl.float32)

    # write out tile, cast back to FP16
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32, GPU)
            - mask: torch.Tensor of shape [B, N, N] (bool or float, optional)
            - weights: dict of model parameters (see skeleton in the prompt)
            - config: dict with keys
                * "dim" (int) – input feature dimension
                * "hidden_dim" (int) – projection dimension
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (over the last dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # same shape

    # ------------------------------------------------------------------
    # 3) Apply mask if it exists
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> broadcast to [B, N, N, 1]
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to FP16 for the reduction (memory‑bandwidth friendly)
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N_seq, _, _ = left_h.shape  # B, N, N, hidden_dim

    # reshape to (batch * hidden_dim, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N_seq, N_seq)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N_seq, N_seq)

    # allocate output buffer (FP16)
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 6) Triton batched matmul: out_h = left_h @ right_hᵀ
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # compute grid dimensions
    num_pid_m = (N_seq + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N_seq + BLOCK_N - 1) // BLOCK_N
    batch_dim = left_h.shape[0]                     # B * hidden_dim
    grid = (num_pid_m, num_pid_n, batch_dim)

    # strides in elements
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)

    stride_b_batch = right_h.stride(0)
    # for transposed access we need stride over K (original column) and N (original row)
    stride_b_k = right_h.stride(2)   # column stride (1)
    stride_b_n = right_h.stride(1)   # row stride (N_seq)

    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    # launch the kernel
    batched_matmul_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        M=N_seq,
        N=N_seq,
        K=N_seq,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7) Convert back to FP32 and restore original shape
    # ------------------------------------------------------------------
    out = out_h.to(torch.float32)
    out = (
        out.view(B, hidden_dim, N_seq, N_seq)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 8) Final LayerNorm, gating and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # elementwise gating
    out = F.linear(out, to_out_weight)       # [B, N, N, dim]

    return out
