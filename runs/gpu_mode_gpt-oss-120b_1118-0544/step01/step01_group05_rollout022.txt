"""
TriMul outgoing variant (AlphaFold3) – Triton implementation.

The core O(N³) contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is expressed as a batched GEMM C = A @ Bᵀ where each slice
A = left[b, :, :, d] and B = right[b, :, :, d] are N×N matrices.
A Triton kernel performs the batched GEMM in FP16 (Tensor‑Core friendly)
while accumulating in FP32. All surrounding LayerNorms, Linear layers
and gating are executed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions (all = seq_len)
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_bk, stride_bn,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ  (A,B shape: [batch, M, K] and [batch, N, K]).
    The kernel works on fp16 inputs with fp32 accumulation."""
    pid_batch = tl.program_id(0)   # batch * hidden slice
    pid_m = tl.program_id(1)       # row tile
    pid_n = tl.program_id(2)       # col tile

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A block: (BLOCK_M, BLOCK_K)
        a_offs = a_batch + \
                 offs_m[:, None] * stride_a_row + \
                 cur_k[None, :] * stride_a_col
        a = tl.load(a_offs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # B block transposed on‑the‑fly: (BLOCK_K, BLOCK_N)
        b_offs = b_batch + \
                 cur_k[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_offs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Multiply‑accumulate (Tensor‑Core path for fp16)
        acc += tl.dot(a, b)

    # Write result
    c_offs = c_batch + \
             offs_m[:, None] * stride_c_row + \
             offs_n[None, :] * stride_c_col
    tl.store(c_offs,
             acc,
             mask=mask_m[:, None] & mask_n[:, None])


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (tensors on the same device)
            - config: dict containing at least "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. First LayerNorm (per‑channel)
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)    # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Prepare for the Triton batched GEMM
    # -----------------------------------------------------------------
    # Cast to fp16 to reduce memory traffic (accumulate in fp32)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N_seq, _, _ = left.shape          # B, seq_len, seq_len, hidden
    assert N_seq == N_seq  # just to keep the name clear

    # Reorder to (B, hidden, N, N) and flatten batch*hidden
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N_seq, N_seq)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N_seq, N_seq)

    # Output buffer (fp32 accumulator)
    out_mat = torch.empty((B * hidden_dim, N_seq, N_seq), device=device, dtype=torch.float32)

    # -----------------------------------------------------------------
    # 6. Launch the Triton kernel
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    stride_a_batch = N_seq * N_seq
    stride_a_row = N_seq
    stride_a_col = 1

    stride_b_batch = N_seq * N_seq
    stride_bk = 1          # K‑dimension stride in B (column stride)
    stride_bn = N_seq      # N‑dimension stride in B (row stride)

    stride_c_batch = N_seq * N_seq
    stride_c_row = N_seq
    stride_c_col = 1

    grid = (
        B * hidden_dim,
        (N_seq + BLOCK_M - 1) // BLOCK_M,
        (N_seq + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N_seq,
        N=N_seq,
        K=N_seq,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # -----------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden]
    # -----------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N_seq, N_seq).permute(0, 2, 3, 1)  # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 8. Second LayerNorm, output gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
