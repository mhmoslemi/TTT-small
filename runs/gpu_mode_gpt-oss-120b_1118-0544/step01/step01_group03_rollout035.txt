"""
TriMul outgoing update (AlphaFold3) – Triton‑accelerated batched matmul.

The core operation is
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
where `left` and `right` are the linearly projected pair representations.
For each hidden channel `h` this is a matrix multiplication
    (N × N) @ (N × N).T
We pack the batch and hidden dimensions into a single batch axis and
perform a tiled batched GEMM with a custom Triton kernel.  All surrounding
layer‑norms, linear projections and gating are handled in PyTorch.

The kernel tiles the output (`BLOCK_M` × `BLOCK_N`) and iterates over the
reduction axis `K` in `BLOCK_K` chunks using `tl.dot`.  Out‑of‑bounds loads
are masked, and the accumulator is kept in float32 for numerical stability.
The final result is reshaped back to `[B, N, N, hidden_dim]`, layer‑normed,
gated, and projected to the original channel dimension.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                 # pointers
    M, N, K,                             # matrix dimensions
    stride_a_batch, stride_am, stride_ak,  # A strides (batch, M, K)
    stride_b_batch, stride_bk, stride_bn,  # B strides (batch, K, N) (B already transposed)
    stride_c_batch, stride_cm, stride_cn,  # C strides (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ B where A: [batch, M, K],
    B: [batch, K, N] and C: [batch, M, N].
    The GEMM is tiled with (BLOCK_M, BLOCK_N) output tiles and iterated over
    the reduction axis K in blocks of size BLOCK_K.
    """
    pid_batch = tl.program_id(2)   # batch index (combined batch * hidden)
    pid_m = tl.program_id(0)       # tile row index
    pid_n = tl.program_id(1)       # tile col index

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Base pointers for the current batch
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    # Pointers to the current tile of A and B
    a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K sized chunks
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                     # (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)                          # (BLOCK_M, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative
    Update (TriMul).

    Parameters
    ----------
    data : tuple
        (pair_tensor, mask_tensor, weights_dict, config_dict)
        - pair_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : mapping from string names to torch.Tensor weights
        - config_dict  : contains "dim" and "hidden_dim" (ints)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) after the TriMul update.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    pair_tensor, mask, weights, config = data
    device = pair_tensor.device
    B, N, N2, C = pair_tensor.shape
    assert N == N2, "pair_tensor must be square in the two sequence dimensions."
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]

    right_gate_w = weights["right_gate.weight"]

    to_out_norm_b = weights["to_out_norm.bias"]

    # Layer‑norm over the channel dimension C
    # ------------------------------------------------------------------
    pair_norm = torch.nn.functional.layer_norm(
        pair_tensor,
        (C,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(pair_norm, left_proj_w)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(pair_norm, right_proj_w)

    # ------------------------------------------------------------------

    mask_val = mask_val.unsqueeze(-1)   # [B, N, N, 1]

    left = left * mask_val
    right = right * mask_val

    # ------------------------------------------------------------------
    # Gating tensors (computed from the *unmasked* normed input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(pair_norm, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(pair_norm, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(pair_norm, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM kernel
    # ------------------------------------------------------------------
    # Move hidden dimension to axis 1 and make tensors contiguous.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Collapse batch and hidden dimensions: (B*H, N, N)
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)                # A : [BH, M=N, K=N]
    # Right needs to be transposed on the last two axes so that we compute
    # A @ Bᵀ (i.e. sum over the shared K dimension).
    right_T = right_perm.view(BH, N, N).transpose(-2, -1)  # B : [BH, K=N, N]

    # Output buffer
    out_flat = torch.empty_like(left_flat)   # [BH, N, N]

    # ------------------------------------------------------------------
    # Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks in M dimension
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks in N dimension
        BH,                             # batch*hidden
    )

    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        N,               # M
        N,               # N
        N,               # K
        # A strides
        left_flat.stride(0),   # batch stride
        left_flat.stride(1),   # M stride
        left_flat.stride(2),   # K stride
        # B strides (B already transposed)
        right_T.stride(0),     # batch stride
        right_T.stride(1),     # K stride
        right_T.stride(2),     # N stride
        # C strides
        out_flat.stride(0),    # batch stride
        out_flat.stride(1),    # M stride
        out_flat.stride(2),    # N stride
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to the original layout: [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Post‑processing: hidden‑dim layer‑norm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # back to original dim

    return out
