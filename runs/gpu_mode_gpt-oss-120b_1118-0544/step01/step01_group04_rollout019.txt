                          M, N, K,

                          stride_cm, stride_cn,

    pid_n = tl.program_id(1) # column block

    """
    TriMul outgoing operator forward pass.

    The kernel fuses the core cubic contraction out_{b,i,j,d} = sum_k left_{b,i,k,d} * right_{b,j,k,d}
    using a Triton batched matrix multiplication. All other linear layers,
    layer‑norms and gating are performed with PyTorch.
    """
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = (not config.get("nomask", False)) and mask is not None

    # 1. Input LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # 2. Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 3. Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # 4. Mask (optional)
    if use_mask:
        mask_exp = mask.unsqueeze(-1).type_as(x)
        left = left * mask_exp
        right = right * mask_exp

    # 5. Apply gates
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # 6. Contract via Triton batched GEMM (left @ right^T per hidden channel)
    # Reshape to (batch * hidden_dim, N, N) and transpose right for the matmul.
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)

        dtype=torch.float32,
        device=left_reshape.device,

"""
TriMul (outgoing) forward pass.

The heavy cubic contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is performed with a single Triton batched GEMM kernel.  The kernel
computes, for each hidden channel, a matrix multiplication
    left_reshape @ right_reshape.T
where `left_reshape` and `right_reshape` are (B*hidden, N, N) tensors
containing the projected and gated pair representations.  All other
operations (LayerNorm, linear projections, sigmoid gates, masking) are
executed with PyTorch for simplicity.

The kernel uses fp16 for the large tensors and accumulates in fp32.
The final output is cast back to fp32, passes through a second
LayerNorm, an output gate and a final linear projection back to the
original channel dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_a_batch, stride_b_batch, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication C = A @ B where
    A : (batch, M, K)  (fp16)
    B : (batch, K, N)  (fp16)
    C : (batch, M, N)  (fp32 accumulator, stored as fp32)
    """
    pid_b = tl.program_id(2)        # batch index (B * hidden_dim)
    pid_m = tl.program_id(0)        # row‑tile index
    pid_n = tl.program_id(1)        # col‑tile index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_am
    b_base = b_ptr + pid_b * stride_b_batch + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load a tile of A (M×K) and B (K×N)
        a = tl.load(
            a_base + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_base + cur_k[:, None] * stride_bk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)  # fp16 × fp16 → fp32 accumulation

    # write back
    c_ptrs = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of torch.Tensor, contains all LayerNorm and Linear weights
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor : shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = (not config.get("nomask", False)) and (mask is not None)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4. Optional mask
    # ------------------------------------------------------------------
    if use_mask:
        mask_exp = mask.unsqueeze(-1).type_as(x)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5. Apply gates and cast to fp16 (memory‑bandwidth reduction)
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Contract via Triton batched GEMM:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as (B*D) independent matmuls: left @ rightᵀ
    # ------------------------------------------------------------------
    # reshape to (B*D, N, N) and transpose right for the matmul
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)

    out_reshape = torch.empty(
        (bs * hidden_dim, N, N),
        dtype=torch.float32,
        device=left_reshape.device,
    )

    # strides: (batch, row, col)
    stride_a_batch, stride_am, stride_ak = left_reshape.stride()
    stride_b_batch, stride_bk, stride_bn = right_reshape.stride()
    stride_c_batch, stride_cm, stride_cn = out_reshape.stride()

    # simple heuristic for tile size
    if N <= 128:
        BLOCK_M = BLOCK_N = BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = BLOCK_K = 64

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    num_batch = bs * hidden_dim
    grid = (num_m, num_n, num_batch)

    _batched_matmul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N,          # M dimension
        N,          # N dimension
        N,          # K dimension
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_reshape.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 9. Output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10. Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
