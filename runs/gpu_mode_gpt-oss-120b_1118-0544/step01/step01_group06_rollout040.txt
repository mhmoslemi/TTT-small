def custom_kernel(data):
    x, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5

    # layernorm input
    x = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # cast to fp16 for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape

    # allocate output
    out_fp32 = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=x.device)

    # strides
    stride_lb, stride_li, stride_lk, stride_ld = left_h.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_h.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_fp32.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # compute grid
    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _trilu_mul_kernel[grid](
        left_h,
        right_h,
        out_fp32,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # continue with out processing
    # out shape = (B, N, N, hidden_dim)
    out = out_fp32

    # out layernorm:
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # gating
    out = out * out_gate

    # final projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dim.
2. Linear projections (left/right) from dim → hidden_dim (no bias).
3. Optional mask is applied (broadcast on hidden_dim).
4. Three sigmoid gates (left, right, out) are computed from the
   normalized input and applied to the projected tensors.
5. Core contraction:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batch of matrix multiplications
        out_head = left_head @ right_headᵀ
   where left_head ∈ ℝ^{N×N} and right_headᵀ ∈ ℝ^{N×N}.
   The computation is performed by a custom Triton kernel that
   loads the two operands with arbitrary (non‑contiguous) strides,
   tiles the M/N axes (BLOCK_M/BLOCK_N) and reduces over K
   (BLOCK_K).  Inputs are FP16, accumulation is FP32.
6. Output layer‑norm, multiplication by the output gate and a final
   linear projection back to the original dim.

At least the dominant O(N³) contraction is executed on Triton, while
the surrounding linear/LN/gating steps use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 pointer    [B, N, N, H]
    right_ptr,         # fp16 pointer    [B, N, N, H]
    out_ptr,           # fp32 pointer    [B, N, N, H]
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_head = tl.program_id(0)   # combined batch * head index
    pid_i    = tl.program_id(1)   # block row (i)
    pid_j    = tl.program_id(2)   # block col (j)

    # recover batch and head indices
    b = pid_head // H
    d = pid_head % H

    # base pointers for this (b, d)
    left_base  = left_ptr  + b * stride_lb + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd
    out_base   = out_ptr   + b * stride_ob + d * stride_od

    # offsets within the current block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction axis k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right tile transposed: we need (k, j) order
        #   row‑stride  = stride_rk (original k dimension)
        #   column‑stride = stride_rj (original j dimension)
        b_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # fused FP16×FP16→FP32 accumulation
        acc += tl.dot(a, b)

    # write back the result
    c_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data (tuple):
            - input_tensor (torch.Tensor): [B, N, N, dim]
            - mask (torch.Tensor or None): [B, N, N]
            - weights (dict of torch.Tensor): model parameters
            - config (dict): contains 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction using Triton (FP16 inputs, FP32 accumulation)
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape

    # Output buffer (float32 as required by subsequent LayerNorm)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=x.device)

    # Strides (in elements)
    stride_lb, stride_li, stride_lk, stride_ld = left_h.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_h.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # Tunable block sizes – work well for the given dimensions on a B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                         # batch * head
        (N + BLOCK_M - 1) // BLOCK_M,  # rows
        (N + BLOCK_N - 1) // BLOCK_N   # cols
    )

    _tri_mul_kernel[grid](
        left_h,
        right_h,
        out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
