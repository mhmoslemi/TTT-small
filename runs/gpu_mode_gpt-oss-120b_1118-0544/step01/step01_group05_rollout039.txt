"""
TriMul outgoing variant (AlphaFold3) – fast Triton implementation.

Algorithm
---------
1. LayerNorm + linear projections + optional mask + gating are performed in PyTorch.
2. The core O(N³) contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is expressed as a batched GEMM A @ Bᵀ for each (batch, hidden_dim) slice.
   Tensors keep the native layout [B, N, N, D] (no costly permute‑copy).  Strides
   are passed to Triton, a program ID encodes (batch, hidden) slice and a
   tiled GEMM kernel computes the result directly in FP16 with FP32 accumulation.
3. The result is LayerNorm‑ed, gated and linearly projected back to the original
   feature dimension in PyTorch.

The kernel uses BLOCK_M/N = 128, BLOCK_K = 32, 8 warps and 5 stages – a
configuration that works well on B200 GPUs for the tested sequence lengths.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix dimensions (all = seq_len)
    stride_a_batch, stride_a_row, stride_a_col,  # left tensor strides
    stride_b_batch, stride_b_row, stride_b_col,  # right tensor strides
    stride_c_batch, stride_c_row, stride_c_col,  # output tensor strides
    hidden_dim: tl.constexpr,               # number of hidden channels (per batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched A @ Bᵀ where A and B have shape (batch, hidden, M, K) and
    (batch, hidden, N, K) stored in layout [B, M, K, hidden] (no permute)."""

    # PROGRAM IDs
    pid = tl.program_id(0)          # combined batch * hidden slice
    pid_m = tl.program_id(1)        # tile over rows   (M)
    pid_n = tl.program_id(2)        # tile over cols   (N)

    # Decode batch and hidden‑dim index from pid
    batch = pid // hidden_dim
    d = pid % hidden_dim

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this (batch, d) slice
    a_base = a_ptr + batch * stride_a_batch + d
    b_base = b_ptr + batch * stride_b_batch + d
    c_base = c_ptr + batch * stride_c_batch + d

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_offs = a_base + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_offs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=tl.float16(0))

        # Load Bᵀ block on the fly (original B[j,k] accessed as (k,j))
        b_offs = b_base + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_offs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=tl.float16(0))

        # Matrix multiply‑accumulate
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # Write the tile to the output tensor
    c_offs = c_base + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_offs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul outgoing module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (tensors on the same device)
            - config: dict containing at least "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm on the input
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on the hidden dimension)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4️⃣ Gating signals (sigmoid after a projection)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 5️⃣ Apply gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Cast to FP16 (reduce memory traffic) and make contiguous
    # -----------------------------------------------------------------
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    B, N, _, _ = left.shape  # left.shape == [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 7️⃣ Allocate output buffer (float32 accumulator)
    # -----------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # 8️⃣ Extract strides (in element counts, not bytes)
    # -----------------------------------------------------------------
    a_stride_batch, a_stride_i, a_stride_k, a_stride_d = left.stride()
    b_stride_batch, b_stride_i, b_stride_k, b_stride_d = right.stride()
    c_stride_batch, c_stride_i, c_stride_k, c_stride_d = out.stride()

    # -----------------------------------------------------------------
    # 9️⃣ Launch the Triton kernel
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                     # one program per (batch, hidden) slice
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        M=N,
        N=N,
        K=N,
        stride_a_batch=a_stride_batch,
        stride_a_row=a_stride_i,
        stride_a_col=a_stride_k,
        stride_b_batch=b_stride_batch,
        stride_b_row=b_stride_i,
        stride_b_col=b_stride_k,
        stride_c_batch=c_stride_batch,
        stride_c_row=c_stride_i,
        stride_c_col=c_stride_k,
        hidden_dim=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=5,
    )

    # -----------------------------------------------------------------
    # 10️⃣ Output LayerNorm, gating and final projection (PyTorch)
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
