"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication C = A @ Bᵀ where
    A = left  (batch*hidden, N, N)   (# rows = i, # cols = k)
    B = right (batch*hidden, N, N)   (# rows = j, # cols = k)
and each batch corresponds to a hidden channel.

We implement this reduction with a fused Triton kernel that
processes a BLOCK_M×BLOCK_N tile per program (≤1024 threads) and
blocks the reduction over K using BLOCK_K.  The rest of the TriMul
operations (LayerNorm, linear projections, gating) are performed in
PyTorch.  Computation is carried out in FP16 for bandwidth, while
the accumulation is in FP32 to preserve numerical quality.  The final
output is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  tensor (batch, M, K)  - fp16
    b_ptr,               # right tensor (batch, N, K)  - fp16, we read it transposed
    c_ptr,               # output tensor (batch, M, N) - fp16 (accumulated in fp32)
    batch: tl.constexpr, # number of independent matrices = B * hidden_dim
    M: tl.constexpr,     # rows of A (i dimension)
    N: tl.constexpr,     # rows of B (j dimension)
    K: tl.constexpr,     # reduction dimension (k)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Program IDs
    pid_m = tl.program_id(0)   # tile row index
    pid_n = tl.program_id(1)   # tile col index
    pid_b = tl.program_id(2)   # batch (hidden) index

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction axis K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A tile (M x K) – each thread loads a row of length BLOCK_K
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (k_offset[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load Bᵀ tile (K x N) – each thread loads a column of length BLOCK_K
        b_ptrs = b_batch + (k_offset[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Dot product over K for the (M,N) tile
        acc += tl.dot(a, b)

    # Store the result (still fp16–> cast before store)
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float)
            - weights: dict of model parameters
            - config: dict with keys
                * "dim" (int)
                * "hidden_dim" (int)
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Unpack weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # Input LayerNorm (last dimension = dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # Linear projections to hidden dimension (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)    # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # Optional mask (broadcast on hidden)
    # -----------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; ensure proper dtype
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # Gating (sigmoid of linear maps over the input x)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Cast to FP16 for the heavy reduction
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape   # B: batch size, N: sequence length

    # Reshape to (batch*hidden, N, N) where batch = B * hidden_dim
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (FP16)
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # -----------------------------------------------------------------
    # Choose block sizes (product must be <= 1024)
    # -----------------------------------------------------------------
    # Simple heuristic based on N
    if N <= 64:
        BLOCK_M, BLOCK_N, BLOCK_K = 32, 32, 16
    elif N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 16, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 16, 32  # works for up to N=1024
    # Ensure thread count does not exceed 1024
    assert BLOCK_M * BLOCK_N <= 1024, "BLOCK_M * BLOCK_N must be <= 1024"

    # -----------------------------------------------------------------
    # Compute strides (in elements, not bytes)
    # -----------------------------------------------------------------
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)

    stride_b_batch = right_h.stride(0)
    stride_b_k = right_h.stride(2)   # third dimension (k) stride
    stride_b_n = right_h.stride(1)   # second dimension (j) stride

    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    # -----------------------------------------------------------------
    # Grid dimensions
    # -----------------------------------------------------------------
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    batch_dim = left_h.shape[0]   # B * hidden_dim
    grid = (num_pid_m, num_pid_n, batch_dim)

    # -----------------------------------------------------------------
    # Launch Triton kernel
    # -----------------------------------------------------------------
    tri_mul_outgoing_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        batch=batch_dim,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # Reshape back to [B, N, N, hidden] and convert to FP32
    # -----------------------------------------------------------------
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # -----------------------------------------------------------------
    # Final LayerNorm (over hidden), gating and output projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                       # apply outgoing gate
    out = F.linear(out, to_out_weight)         # project back to dim

    return out
