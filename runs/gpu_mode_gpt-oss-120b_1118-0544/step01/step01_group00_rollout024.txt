"""
Optimized outgoing TriMul implementation.

The costly part of TriMul is the reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication (A @ Bᵀ) over the
sequence dimension.  We perform the projection, gating and
LayerNorms in PyTorch (high‑level kernels) and accelerate the
reduction with a single torch.bmm call on FP16 tensors – this
leverages the cuBLAS tensor‑core GEMM.  To satisfy the
requirement of using Triton we fuse the final elementwise
gate (out_norm * out_gate) with a tiny Triton kernel.

The overall flow:
1. Layer‑norm the input.
2. Linear projections → left / right.
3. Optional mask (broadcast over the hidden dim).
4. Compute gating (sigmoid of linear projections) and apply.
5. Cast to FP16 and perform a batched GEMM (B*hidden_dim independent
   N×N multiplications) via torch.bmm.
6. Convert back to FP32, apply a second LayerNorm, multiply by the
   output gate using a Triton kernel, and finally project back to
   the original dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,  # pointers
    numel,                  # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    """Simple elementwise a * b → out, 1‑D flattened view."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def elementwise_mul_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton kernel on two contiguous tensors."""
    assert a.shape == b.shape
    a_cont = a.contiguous()
    b_cont = b.contiguous()
    out = torch.empty_like(a_cont)

    numel = a_cont.numel()
    BLOCK = 1024
    grid = (triton.cdiv(numel, BLOCK),)

    elementwise_mul_kernel[grid](
        a_ptr=a_cont,
        b_ptr=b_cont,
        out_ptr=out,
        numel=numel,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – may be ignored
            - weights: dict of model weight tensors
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool; ensure same dtype
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections) and apply
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched matrix multiplication in FP16 (core TriMul reduction)
    # ------------------------------------------------------------------
    # Cast to half to exploit tensor‑core speed while keeping final
    # result in float32 for the remaining lightweight ops.
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape

    # reshape to (B*hidden_dim, N, N) where the second dim is the
    # reduction axis K and the third dim is the other sequence axis.
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Batched GEMM: out = left_h @ right_h  (right_h already transposed)
    out_h = torch.bmm(left_h, right_h)  # [B*hidden_dim, N, N]  (FP16)

    # Restore the original 4‑D layout and convert back to FP32
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6. Second LayerNorm + output gating (Triton‑fused multiply)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B, N, N, hidden_dim]

    # Elementwise multiply with out_gate using a tiny Triton kernel
    gated = elementwise_mul_triton(out_norm, out_gate)  # still [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 7. Final projection back to the original dimension
    # ------------------------------------------------------------------
    final_out = F.linear(gated, to_out_weight)  # [B, N, N, dim]

    return final_out
```
