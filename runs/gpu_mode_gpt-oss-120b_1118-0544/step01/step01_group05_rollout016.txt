```python
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out of bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for batch slice
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    # accumulator as fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # main loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=tl.float16(0.0))

        b_ptrs = b_batch_ptr + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=tl.float16(0.0))

        # dot
        acc += tl.dot(a, b)
    # store
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    device = x.device
    dtype = x.dtype

    # LN
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # Linear projections
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # Mask (if any)
    if not config.get("nomask", False):
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # Rearrange tensors to (B*hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous()
    right_mat = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_mat.view(B * hidden_dim, N, N)
    right_mat = right_mat.view(B * hidden_dim, N, N)

    # output allocation (FP32)
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # compute grid
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    tri_mul_gemm_kernel[grid](
        left_mat, right_mat, out_mat,
        M=N, N=N, K=N,
        stride_a_batch=N * N,
        stride_a_row=N,
        stride_a_col=1,
        stride_b_batch=N * N,
        stride_b_row=N,
        stride_b_col=1,
        stride_c_batch=N * N,
        stride_c_row=N,
        stride_c_col=1,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # To_out_norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)

    # Apply out gate
    out = out * out_gate

    # Final linear
    out = F.linear(out, weights["to_out.weight"])
    return out
```

"""
TriMul (outgoing) forward pass with a fused Triton GEMM kernel.

Algorithm
---------
1. LayerNorm over the C dimension.
2. Linear projections (left/right) and gating (left/right/out) are done in PyTorch.
3. Apply mask (if present) and element‑wise gates to the projected tensors.
4. Cast the gated tensors to float16 and reshape them to
   (B * hidden_dim, N, N) → a batch of independent matrices.
5. Compute for each batch slice d:
        out[d] = left[d] @ right[d].T
   using a custom Triton kernel that performs a batched GEMM with
   half‑precision inputs and FP32 accumulation (Tensor‑Core friendly).
6. Reshape back to [B, N, N, hidden_dim] and apply the output LayerNorm,
   the output gate and the final linear projection.

The Triton kernel (tri_mul_gemm_kernel) tiles the multiplication with
BLOCK_M×BLOCK_N output tiles and BLOCK_K inner‑product tiles.
It is optimized for GPUs (e.g. NVIDIA B200) with 8 warps and 3 pipeline
stages to hide memory latency.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ
    A/B shapes: (batch, M, K) and (batch, N, K) (row‑major).
    C shape: (batch, M, N).
    All tensors are stored column‑major in memory (standard torch layout).
    """
    pid_batch = tl.program_id(0)   # batch * hidden_dim index
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A block: (BLOCK_M, BLOCK_K)
        a_offs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_offs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))

        # Bᵀ block: (BLOCK_K, BLOCK_N)
        b_offs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_offs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=tl.float16(0.0))

        # dot product accumulation
        acc += tl.dot(a, b)

    # write out
    c_offs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_offs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (all tensors on the same device)
            - config: dict containing at least "dim", "hidden_dim" and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections
    # -----------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask
    # -----------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))  # keep fp32 for later

    left = left.to(torch.float16) * left_gate
    right = right.to(torch.float16) * right_gate

    # -----------------------------------------------------------------
    # 5. Prepare tensors for the batched GEMM
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim
    # reshape to (B * hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # output buffer (float32 accumulator)
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # -----------------------------------------------------------------
    # 6. Triton GEMM launch
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                              # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,                # tiles along M
        (N + BLOCK_N - 1) // BLOCK_N,                # tiles along N
    )

    # Since tensors are contiguous after the view, strides are simple
    stride_a_batch = N * N
    stride_a_row = N
    stride_a_col = 1

    stride_b_batch = N * N
    stride_b_row = N
    stride_b_col = 1

    stride_c_batch = N * N
    stride_c_row = N
    stride_c_col = 1

    tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # -----------------------------------------------------------------
    # 7. Reshape back to 4‑D layout
    # -----------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 8. Output LayerNorm, output gate and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    out = out * out_gate.to(out.dtype)   # broadcast over hidden_dim

    out = F.linear(out, weights["to_out.weight"])

    return out
