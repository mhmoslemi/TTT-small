def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.
    Implements the heavy N^3 einsum via a fused Triton kernel.
    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of model parameters
            - config: dict with keys "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    import torch.nn.functional as F

    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), norm_w, norm_b)

    # 2. Projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # 2a. Apply mask if present
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # 3. Gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    out_hidden = out_hidden * out_gate

    # 6. Final projection back to dim
    out = F.linear(out_hidden, weights["to_out.weight"])
    return out

"""
Outgoing TriMul (Triangle Multiplicative Update) forward pass.

The expensive O(N³) einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a fused batched matrix multiplication
(left @ rightᵀ) using a Triton kernel.  The rest of the
TriMul pipeline (LayerNorms, linear projections, gating) is
performed with PyTorch operations.  The kernel works on
FP16 tensors (accumulating in FP32) and supports batch size 1‑2,
sequence lengths up to 1024 and hidden dimensions up to 768.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _tri_mul_kernel(
    A, B, C,                # pointers
    M, N, K,                # matrix dimensions (all equal to seq_len)
    stride_am, stride_ak,   # A row / col strides
    stride_bk, stride_bn,   # B row / col strides (B is conceptually transposed)
    stride_cm, stride_cn,   # C row / col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for many independent (batch, head) slices."""
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile column
    pid_b = tl.program_id(2)  # batch * head index

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this (b,head) slice
    a_base = A + pid_b * (M * K)      # each slice is M×K
    b_base = B + pid_b * (K * N)      # each slice is K×N
    c_base = C + pid_b * (M * N)      # each slice is M×N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        # Pointers to the current K‑tile
        a_ptrs = a_base + offs_m[:, None] * stride_am + (cur_k + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        b_ptrs = b_base + (cur_k + tl.arange(0, BLOCK_K))[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # OOB masks
        a_mask = (offs_m[:, None] < M) & ((cur_k + tl.arange(0, BLOCK_K))[None, :] < K)
        b_mask = ((cur_k + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Accumulate (BLOCK_M×BLOCK_K)·(BLOCK_K×BLOCK_N)
        acc += tl.dot(a, b)

    # Write back, casting to fp16
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
        - mask        : torch.Tensor of shape [B, N, N] or None
        - weights     : dict mapping parameter names to tensors
        - config      : dict containing at least 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weights["norm.weight"], weights["norm.bias"])

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias) + optional mask
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    if mask is not None:
        # broadcast mask over hidden dim
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 3. Gating
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Batched matrix multiplication via Triton
    #    out_h = left_h @ right_hᵀ for every hidden channel h
    # -----------------------------------------------------------------
    # Cast to half for the kernel (still keep gradients disabled)
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # Move hidden dimension to axis 1 and make memory contiguous
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Collapse batch & head into a single GEMM batch dimension
    left_mat  = left.view(-1, N, N)    # (B*H, N, N)
    right_mat = right.view(-1, N, N)   # (B*H, N, N)
    out_mat   = torch.empty_like(left_mat)

    # Strides in element counts (not bytes)
    stride_am = left_mat.stride(1)   # row stride for A
    stride_ak = left_mat.stride(2)   # col stride for A
    stride_bk = right_mat.stride(2)  # row stride for B (k dimension after transpose)
    stride_bn = right_mat.stride(1)  # col stride for B (j dimension)
    stride_cm = out_mat.stride(1)    # row stride for C
    stride_cn = out_mat.stride(2)    # col stride for C

    # Tile sizes – tuned for the NVIDIA H100 / B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: [B, N, N, H]
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # -----------------------------------------------------------------
    # 5. Output LayerNorm, output gate and final projection
    # -----------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
