    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Unpack weights
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Flatten and concatenated linear
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)

    # Concatenate weight matrices
    weight_cat = torch.cat([left_proj_weight, right_proj_weight,
                            left_gate_weight, right_gate_weight,
                            out_gate_weight], dim=0)  # shape (5*hidden_dim, dim)

    # Compute all projections in one matmul
    proj_flat = F.linear(x_flat, weight_cat)  # shape (B*N*N, 5*hidden_dim)

        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # final layernorm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # gating
    out = out * out_gate

    # final projection
    out = F.linear(out, to_out_weight)

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) kernel.

The heavy computation is the batched matrix product
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched GEMM C = A @ Bᵀ where each batch
represents one (hidden_dim) channel slice of the pairwise tensor.
The kernel below performs this GEMM in FP16 (inputs) with FP32
accumulation.  All other operations (LayerNorm, linear projections,
gating, final projection) are done in PyTorch, but the multiple
linear layers are fused into a single matrix multiply to cut launch
overhead and memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A: M×K, B: N×K)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  tensor pointer  (batch, M, K)
    b_ptr,               # right tensor pointer  (batch, N, K) – we treat it as transposed
    c_ptr,               # output tensor pointer (batch, M, N)
    batch: tl.constexpr, # number of independent matrices = B * hidden_dim
    M: tl.constexpr,     # rows of A
    N: tl.constexpr,     # rows of B (output columns)
    K: tl.constexpr,     # reduction dimension
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # batch index

    # Offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K in blocks
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A tile (M, K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (k_offset[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load Bᵀ tile (K, N).  B is stored as (batch, N, K) row‑major,
        # so we read using stride_k = 1 (contiguous over K) and stride_n = N.
        b_ptrs = b_batch + (k_offset[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Matrix multiply for the tile
        acc += tl.dot(a, b).to(tl.float32)

    # Write back, casting to FP16 to save bandwidth
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Public API
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim]   (float32, GPU)
            mask:          torch.Tensor [B, N, N]       (bool or float)
            weights:       dict of model weights, keys as in the reference module
            config:        dict with at least "dim" and "hidden_dim";

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)                       # [B, N, N, dim]

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # Fuse the five linear projections (left, right, three gates) into a
    # single matmul to reduce launch overhead.
    # ------------------------------------------------------------------
    # Flatten to (B * N * N, dim)
    x_flat = x.view(-1, dim)                               # [B*N*N, dim]

    # Concatenate weight matrices: each is (hidden_dim, dim)
    weight_cat = torch.cat([left_proj_weight, right_proj_weight,
                            left_gate_weight, right_gate_weight,
                            out_gate_weight], dim=0)       # [(5*hidden_dim), dim]

    # One big linear layer (no bias)
    proj_flat = F.linear(x_flat, weight_cat)               # [B*N*N, 5*hidden_dim]

    # Reshape back to (B, N, N, 5*hidden_dim)
    proj = proj_flat.view(B, N, N, 5 * hidden_dim)

    # ------------------------------------------------------------------
    # Slice the concatenated results
    # ------------------------------------------------------------------
    hid = hidden_dim
    left      = proj[..., 0:hid]
    right     = proj[..., hid:2*hid]
    left_gate = proj[..., 2*hid:3*hid]
    right_gate= proj[..., 3*hid:4*hid]
    out_gate  = proj[..., 4*hid:5*hid]

    # ------------------------------------------------------------------
    # Masking (if required) and gating
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(x.dtype)            # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the Triton GEMM: cast to FP16, reshape to
    # (B*hidden_dim, N, N) where the hidden dimension becomes the batch
    # ------------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Move hidden dim to the batch axis for the kernel
    left_h  = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Triton kernel launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # tiles over M
        (N + BLOCK_N - 1) // BLOCK_N,          # tiles over N
        B * hidden_dim                         # batch dimension
    )

    # Strides in element units (not bytes)
    stride_a_batch = left_h.stride(0)
    stride_a_m     = left_h.stride(1)
    stride_a_k     = left_h.stride(2)

    stride_b_batch = right_h.stride(0)
    # right is (batch, N, N) – we need it as (batch, K, N) for Bᵀ.
    # K (the reduction axis) corresponds to the last dimension → stride 1.
    stride_b_k     = right_h.stride(2)    # = 1
    stride_b_n     = right_h.stride(1)    # = N

    stride_c_batch = out_h.stride(0)
    stride_c_m     = out_h.stride(1)
    stride_c_n     = out_h.stride(2)

    # Launch the kernel
    tri_mul_outgoing_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        batch=B * hidden_dim,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] and convert to FP32
    # ------------------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
             .permute(0, 2, 3, 1)          # [B, N, N, hidden_dim]
             .contiguous()
             .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # Final layer‑norm, gating and projection back to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate                               # element‑wise gate
    out = F.linear(out, to_out_weight)                 # [B, N, N, dim]

    return out
