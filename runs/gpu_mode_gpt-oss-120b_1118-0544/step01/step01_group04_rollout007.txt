def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Setup
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False) and mask is not None

    # LN
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Proj
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Mask
    if use_mask:
        mask_exp = mask.unsqueeze(-1)  # (bs, N, N, 1)
        left = left * mask_exp
        right = right * mask_exp

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Convert to fp16 for matmul
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Reshape for batched GEMM
    left_reshape = left_fp16.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_reshape = right_fp16.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)

    # Output buffer
    out_reshape = torch.empty((bs * hidden_dim, N, N), dtype=torch.float32, device=left_reshape.device)

    # Strides
    stride_a_batch, stride_am, stride_ak = left_reshape.stride()
    stride_b_batch, stride_bk, stride_bn = right_reshape.stride()
    stride_c_batch, stride_cm, stride_cn = out_reshape.stride()

    # Kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_m, num_n, bs * hidden_dim)

    matmul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [bs, N, N, hidden_dim]
    out = out_reshape.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Output gate
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

"""
TriMul (outgoing) forward pass implemented with a Triton batched GEMM kernel.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×C}.
2. Linear projections left/right : ℝ^{C→H}.
3. Compute sigmoid gates (left, right, out) from x.
4. (Optional) apply the mask to left/right before gating.
5. Multiply left/right by their respective gates.
6. Compute the core contraction
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication:
      left_reshaped (B·H, N, N)  @  right_reshaped^T (B·H, N, N)
   performed by a custom Triton kernel (fp16 inputs, fp32 accumulation).
7. Reshape to (B,N,N,H), apply a second LayerNorm over H,
   multiply by the out‑gate, and project back to the original C dimension.

Only the large O(N³) contraction is done in Triton; all other ops use
torch.nn.functional for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_a_batch, stride_b_batch, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   where
       A : (batch, M, K)  B : (batch, K, N)  C : (batch, M, N)
       A/B are fp16, C is fp32.
    """
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # batch index

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers for the current batch
    a_ptrs = a_ptr + pid_b * stride_a_batch \
                + offs_m[:, None] * stride_am \
                + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_b_batch \
                + offs_k[:, None] * stride_bk \
                + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = offs_k
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)               # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the result
    c_ptrs = c_ptr + pid_b * stride_c_batch \
                  + offs_m[:, None] * stride_cm \
                  + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (tensors)
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = (not config.get("nomask", False)) and mask is not None

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4. Optional mask
    # ------------------------------------------------------------------
    if use_mask:
        mask_exp = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Batched GEMM via Triton (equivalent to the einsum over k)
    # ------------------------------------------------------------------
    # Cast to fp16 to reduce bandwidth; accumulation stays in fp32.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Reshape to (B*H, N, N) with right transposed on its last two axes.
    left_reshape = left_fp16.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_reshape = right_fp16.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)

    # Output buffer (fp32)
    out_reshape = torch.empty(
        (bs * hidden_dim, N, N),
        dtype=torch.float32,
        device=left_reshape.device,
    )

    # Strides required by the Triton kernel
    stride_a_batch, stride_am, stride_ak = left_reshape.stride()
    stride_b_batch, stride_bk, stride_bn = right_reshape.stride()
    stride_c_batch, stride_cm, stride_cn = out_reshape.stride()

    # Tile size – chosen for good occupancy on NVIDIA B200.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles over rows (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles over cols (j)
        bs * hidden_dim,                # batch * hidden_dim
    )

    matmul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_reshape.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 9. Output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10. Final linear projection back to original dimension (C)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
