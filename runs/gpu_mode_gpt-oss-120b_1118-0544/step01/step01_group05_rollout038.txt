"""
TriMul outgoing kernel (AlphaFold3) – fast batched GEMM.
The heavy O(N³·C_hidden) contraction

    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]

is performed by a custom Triton kernel that treats each (b,c) pair as
an independent matrix‑multiply of shape (N,N).  The kernel works
directly on the original layout (B,N,N,C) to avoid the costly
contiguous‑permute copies used previously.  All LayerNorms, linear
projections and gating are kept in PyTorch (highly optimized cuBLAS).
Mixed‑precision (FP16 for the multiply, FP32 accumulation) reduces
memory traffic while preserving accuracy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                     # M=N rows, N=N cols, K=N reduction dim
    stride_a_batch, stride_a_row, stride_a_col, stride_a_hidden,
    stride_b_batch, stride_b_row, stride_b_col, stride_b_hidden,
    stride_c_batch, stride_c_row, stride_c_col, stride_c_hidden,
    hidden_dim,                  # runtime hidden dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ where A,B are (B,hidden,N,N) stored as
    (B,N,N,hidden) (row‑major).  The hidden index is folded into the
    batch dimension (pid = b*hidden + h)."""
    pid = tl.program_id(0)        # combined batch*hidden index
    pid_m = tl.program_id(1)      # tile over rows (i)
    pid_n = tl.program_id(2)      # tile over cols (j)

    # Decompose combined index into batch and hidden offsets
    b = pid // hidden_dim
    h = pid % hidden_dim

    # Base pointers for this (b,h) slice
    a_base = a_ptr + b * stride_a_batch + h * stride_a_hidden
    b_base = b_ptr + b * stride_b_batch + h * stride_b_hidden
    c_base = c_ptr + b * stride_c_batch + h * stride_c_hidden

    # Offsets within the N×N matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in chunks of BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_offs = a_base + \
                 offs_m[:, None] * stride_a_row + \
                 cur_k[None, :] * stride_a_col
        a = tl.load(a_offs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B block transposed on‑the‑fly: we need Bᵀ, i.e. B[j,k] →
        # accessed as (k,j).  Original B layout is (B,N,N,hidden) with
        # row‑stride = stride_b_row (j) and col‑stride = stride_b_col (k).
        b_offs = b_base + \
                 cur_k[:, None] * stride_b_col + \
                 offs_n[None, :] * stride_b_row
        b = tl.load(b_offs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Dot product and accumulate (FP32 accumulator)
        acc += tl.dot(a, b)

    # Write output tile
    c_offs = c_base + \
             offs_m[:, None] * stride_c_row + \
             offs_n[None, :] * stride_c_col
    tl.store(c_offs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (tensors on same device)
            - config: dict containing at least "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2. Linear projections (float32 -> float16 for speed)
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Optional mask
    if not nomask and mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)        # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gates (computed in float32, then cast)
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))  # keep float32

    # --------------------------------------------------------------
    # 5. Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Triton batched GEMM: out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
    B, N_seq, _, _ = left.shape  # B, N, N, hidden_dim

    # Allocate output (float32 accumulator)
    out = torch.empty((B, N_seq, N_seq, hidden_dim), dtype=torch.float32, device=device)

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides (in elements) of the source tensors
    stride_a_batch = left.stride(0)          # B stride
    stride_a_row = left.stride(1)            # i stride (N*hidden)
    stride_a_col = left.stride(2)            # k stride (hidden)
    stride_a_hidden = left.stride(3)         # h stride (1)

    stride_b_batch = right.stride(0)
    stride_b_row = right.stride(1)
    stride_b_col = right.stride(2)
    stride_b_hidden = right.stride(3)

    stride_c_batch = out.stride(0)
    stride_c_row = out.stride(1)
    stride_c_col = out.stride(2)
    stride_c_hidden = out.stride(3)

    grid = (
        B * hidden_dim,
        (N_seq + BLOCK_M - 1) // BLOCK_M,
        (N_seq + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        M=N_seq,
        N=N_seq,
        K=N_seq,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_a_hidden=stride_a_hidden,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_b_hidden=stride_b_hidden,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        stride_c_hidden=stride_c_hidden,
        hidden_dim=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # --------------------------------------------------------------
    # 7. Output LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # 8. Apply output gate (broadcasted)
    out = out * out_gate

    # 9. Final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out
