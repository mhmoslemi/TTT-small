"""
TriMul (outgoing) forward pass with a fused Triton GEMM kernel.

The heavy contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication:
    out_batch = left_batch @ right_batch.T
where each batch corresponds to a (batch * hidden_dim) pair.
All other operations (LayerNorm, Linear, Sigmoid, masking) are performed
with PyTorch. The kernel works on fp16 inputs and accumulates in fp32,
yielding a float32 output that matches the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction.
# Computes C = A @ B.T with A and B in fp16 and accumulation in fp32.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all = seq_len)
    stride_a_batch, stride_am, stride_ak,   # strides for A
    stride_b_batch, stride_bk, stride_bn,   # strides for B (already transposed)
    stride_c_batch, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch * hidden_dim index
    pid_m = tl.program_id(0)   # tile row index
    pid_n = tl.program_id(1)   # tile col index

    # -------------------- tile offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch element (no K offset yet)
    a_ptr_base = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_am
    b_ptr_base = b_ptr + pid_b * stride_b_batch + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- reduction over K --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M x K) and B tile (K x N)
        a = tl.load(
            a_ptr_base + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
        )
        b = tl.load(
            b_ptr_base + cur_k[:, None] * stride_bk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0.0),
        )
        # Tensor‑core dot product (fp16 -> fp32 accumulation)
        acc += tl.dot(a, b)

    # -------------------- write out --------------------
    c_ptrs = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entrypoint used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of TriMul (outgoing) using a fused Triton GEMM.
    `data` is a tuple:
        (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False) and mask is not None

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on channel dim)
    # ------------------------------------------------------------------
    if use_mask:
        mask_exp = mask.unsqueeze(-1)          # [bs, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()   # [bs, N, N, hidden_dim]

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to fp16 for the heavy contraction
    # ------------------------------------------------------------------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Prepare tensors for batched GEMM: A @ B^T
    # ------------------------------------------------------------------
    # A: (bs, N, N, hidden_dim) -> (bs*hidden_dim, N, N)
    a = left_f16.permute(0, 3, 1, 2).contiguous().view(-1, left_f16.shape[1], left_f16.shape[2])
    # B must be transposed on the last two dimensions before view
    b = right_f16.permute(0, 3, 2, 1).contiguous().view(-1, right_f16.shape[1], right_f16.shape[2])

    # Output buffer (FP32)
    c = torch.empty(a.shape[0], a.shape[1], a.shape[2], dtype=torch.float32, device=a.device)

    # ------------------------------------------------------------------
    # 7. Launch Triton kernel
    # ------------------------------------------------------------------
    seq_len = a.shape[1]        # == N
    batch_hidden = a.shape[0]   # bs * hidden_dim

    # Strides (in elements)
    stride_a_batch, stride_am, stride_ak = a.stride()
    stride_b_batch, stride_bk, stride_bn = b.stride()
    stride_c_batch, stride_cm, stride_cn = c.stride()

    # Tile sizes – chosen to fit Tensor Cores and stay within register limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,
        (seq_len + BLOCK_N - 1) // BLOCK_N,
        batch_hidden,
    )

    tri_mul_gemm_kernel[grid](
        a, b, c,
        seq_len, seq_len, seq_len,
        stride_a_batch, stride_am, stride_ak,
        stride_b_batch, stride_bk, stride_bn,
        stride_c_batch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8. Reshape back to [bs, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = c.view(-1, hidden_dim, seq_len, seq_len) \
           .permute(0, 2, 3, 1).contiguous()   # [bs, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 9. Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 10. Output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
