```
def _batched_outgoing_matmul_triton(A, B):
    # A: [batch, M, K], B: [batch, N, K] (float16)
    # returns C: [batch, M, N] (float16)
    assert A.shape[0] == B.shape[0]
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2
    dtype = A.dtype
    # Determine block sizes
    BLOCK_M = 64 if M <= 256 else 128
    BLOCK_N = 64 if N <= 256 else 128
    BLOCK_K = 32  # can adjust
    # allocate output
    C = torch.empty((batch, M, N), dtype=dtype, device=A.device)
    grid = ( (M + BLOCK_M -1)//BLOCK_M,
             (N + BLOCK_N -1)//BLOCK_N,
             batch )

        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
```

```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # Gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape

    # Rearrange to (B*H, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous()
    batch_hidden = B * hidden_dim
    left_flat = left_h.view(batch_hidden, N, N)
    right_flat = right_h.view(batch_hidden, N, N)

    # core O(N³) via triton
    out_flat_h = _batched_outgoing_matmul_triton(left_flat, right_flat)  # (B*H, N, N) half

    # reshape back to (B,N,N,H)
    out = out_flat_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # Second layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

```
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bn, stride_Bk,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(axis=2)
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
```

"""
TriMul (outgoing) forward pass with a Triton kernel for the heavy
O(N³·H) reduction.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Linear projections (left/right) and gating (left/right/out) using the
   provided weight tensors. If a mask is present it is applied after the
   projections and before the gates.
3. Cast the gated tensors to float16 and reshape them so that the hidden
   dimension becomes part of the batch dimension:
       left/right ∈ [B, N, N, H] → [B·H, N, N].
4. Compute the batched matrix multiplication
       out[b, i, j] = Σₖ left[b, i, k] * right[b, j, k]
   with a custom Triton kernel.  The kernel tiles the i‑ and j‑axes
   (BLOCK_M × BLOCK_N) and iterates over k in BLOCK_K‑sized chunks,
   using `tl.dot(..., trans_b=True)` which maps to Tensor‑Core matmul
   and accumulates in float32.
5. Reshape the result back to [B, N, N, H] and apply a second
   LayerNorm, the output gate, and the final linear projection back to
   the original channel dimension C.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _outgoing_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bn, stride_Bk,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul:
    C[b, i, j] = Σₖ A[b, i, k] * B[b, j, k]
    A : (batch, M, K)   (float16)
    B : (batch, N, K)   (float16)
    C : (batch, M, N)   (float16, stored from float32 accumulator)
    """
    pid_b = tl.program_id(axis=2)      # batch * hidden
    pid_m = tl.program_id(axis=0)      # tile over i
    pid_n = tl.program_id(axis=1)      # tile over j

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current batch slice
    a_ptrs = A_ptr + pid_b * stride_Ab + (offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak)
    b_ptrs = B_ptr + pid_b * stride_Bb + (offs_n[:, None] * stride_Bn + tl.arange(0, BLOCK_K)[None, :] * stride_Bk)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        k_valid = (k_offset + tl.arange(0, BLOCK_K)) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0)           # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_n[None, :],
                    other=0.0)           # (BLOCK_N, BLOCK_K)

        # B is loaded as (j, k); we need (k, j) → trans_b=True
        acc += tl.dot(a, b, trans_b=True)

        # Advance k‑dimension pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # Store the result (cast accumulator back to half)
    c_ptrs = C_ptr + pid_b * stride_Cb + (offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_outgoing_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Wrapper for the Triton kernel.
    A, B : (batch, N, N) float16
    Returns C : (batch, N, N) float16
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "Kernel expects float16 tensors"
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2, "Reduction dimension mismatch"

    # Heuristic block sizes
    BLOCK_M = 64 if M <= 256 else 128
    BLOCK_N = 64 if N <= 256 else 128
    BLOCK_K = 32

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    _outgoing_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict with all required weight tensors
            - config: dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- 1️⃣ LayerNorm ----------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- 2️⃣ Linear projections ----------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- 3️⃣ Optional masking ----------
    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ---------- 4️⃣ Gating ----------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- 5️⃣ Prepare for Triton reduction ----------
    # Cast to half for the heavy matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B=batch size, N=seq_len

    # Move hidden dimension into the batch axis
    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch_hidden = B * hidden_dim
    left_flat = left_h.view(batch_hidden, N, N)    # (B*H, N, N)
    right_flat = right_h.view(batch_hidden, N, N)  # (B*H, N, N)

    # ---------- 6️⃣ Core O(N³·H) reduction on GPU ----------
    out_flat_h = _batched_outgoing_matmul_triton(left_flat, right_flat)  # (B*H, N, N) half

    # ---------- 7️⃣ Reshape back ----------
    out = out_flat_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ---------- 8️⃣ Output LayerNorm ----------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ---------- 9️⃣ Apply output gate ----------
    out = out * out_gate

    # ---------- 🔟 Final linear back to original dimension ----------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
