"""
TriMul (outgoing) forward pass using Triton.

The heavy O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM:
    left_reshape (B*H, N, N) @ right_reshape (B*H, N, N).T
where H = hidden_dim.  The GEMM runs in FP16 (inputs) with FP32 accumulation
via a custom Triton kernel, while all LayerNorms, linear projections,
gates and optional masking are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ B) with FP16 inputs, FP32 acc.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_a_batch, stride_b_batch, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # batch * hidden_dim

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_am
    b_base = b_ptr + pid_b * stride_b_batch + offs_n[None, :] * stride_bn

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(
            a_base + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_base + cur_k[:, None] * stride_bk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16 dot, accumulate in FP32
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the TriMul (outgoing) operator.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of TriMul (outgoing) using Triton for the core contraction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of torch.Tensors with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict containing at least "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected float32

    # configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_mask = (not config.get("nomask", False)) and (mask is not None)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4. Optional mask
    # ------------------------------------------------------------------
    if use_mask:
        mask_exp = mask.unsqueeze(-1).to(dtype)        # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction via Triton (batched GEMM)
    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy MatMul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, N, N, hidden_dim

    # reshape to (B*hidden_dim, N, N) for GEMM
    # left: (b,i,k,d) -> (batch, M=N, K=N)
    left_reshape = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # right: (b,j,k,d) -> (batch, K=N, N=j)
    right_reshape = right_h.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # output buffer (FP32)
    out_reshape = torch.empty(
        (B * hidden_dim, N, N), dtype=torch.float32, device=device
    )

    # strides (in elements)
    stride_a_batch, stride_am, stride_ak = left_reshape.stride()
    stride_b_batch, stride_bk, stride_bn = right_reshape.stride()
    stride_c_batch, stride_cm, stride_cn = out_reshape.stride()

    # tile sizes (tuned for A100 / B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # grid dimensions
    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    num_batch = B * hidden_dim
    grid = (num_m, num_n, num_batch)

    # launch kernel
    batched_gemm_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # sufficient parallelism for B200
    )

    # reshape back to [B, N, N, hidden_dim]
    out = (
        out_reshape.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden_dim] float32

    # ------------------------------------------------------------------
    # 7. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 8. Output gate
    # ------------------------------------------------------------------
    out = out * out_gate  # elementwise

    # ------------------------------------------------------------------
    # 9. Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
