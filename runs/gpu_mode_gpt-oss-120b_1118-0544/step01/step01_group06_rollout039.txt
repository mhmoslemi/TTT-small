def mask_kernel(left_ptr, right_ptr, mask_ptr,
                B, N, H,
                stride_left_b, stride_left_i, stride_left_j, stride_left_h,
                stride_right_b, stride_right_i, stride_right_j, stride_right_h,
                stride_mask_b, stride_mask_i, stride_mask_j,
                BLOCK_M, BLOCK_N, BLOCK_H):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, BLOCK_H)

    # bounds check
    mask_offs = (pid_b * stride_mask_b) + (offs_i[:, None] * stride_mask_i) + (offs_j[None, :] * stride_mask_j)
    # shape (BLOCK_M, BLOCK_N)
    mask = tl.load(mask_ptr + mask_offs, mask = (offs_i[:,None] < N) & (offs_j[None,:] < N), other = 0.0)  # (M,N)

    # broadcast mask to hidden dim: convert mask to shape (M,N,1) then broadcast.

    for hk in range(0, tl.cdiv(H, BLOCK_H)):
        offs_hk = hk * BLOCK_H + offs_h
        h_mask = (offs_hk < H)
        # compute offsets for left and right
        left_offs = pid_b*stride_left_b + offs_i[:,None,None]*stride_left_i + offs_j[None,:,None]*stride_left_j + offs_hk[None,None,:]*stride_left_h
        right_offs = pid_b*stride_right_b + offs_i[:,None,None]*stride_right_i + offs_j[None,:,None]*stride_right_j + offs_hk[None,None,:]*stride_right_h

        left_val = tl.load(left_ptr + left_offs, mask = (offs_i[:,None,None] < N) & (offs_j[None,:,None] < N) & h_mask, other = 0.0)
        right_val = tl.load(right_ptr + right_offs, mask = (offs_i[:,None,None] < N) & (offs_j[None,:,None] < N) & h_mask, other = 0.0)

        # multiply by mask (broadcasted)
        left_val = left_val * mask[...,None]
        right_val = right_val * mask[...,None]

        # store back (in-place)
        tl.store(left_ptr + left_offs, left_val, mask = (offs_i[:,None,None] < N) & (offs_j[None,:,None] < N) & h_mask)
        tl.store(right_ptr + right_offs, right_val, mask = (offs_i[:,None,None] < N) & (offs_j[None,:,None] < N) & h_mask)

@triton.jit
def apply_mask_kernel(left_ptr, right_ptr, mask_ptr,
                      B, N, H,
                      stride_left_b, stride_left_i, stride_left_j, stride_left_h,
                      stride_right_b, stride_right_i, stride_right_j, stride_right_h,
                      stride_mask_b, stride_mask_i, stride_mask_j,
                      BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr):
    # program ids
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute start offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_off_i = offs_i[:, None] < N
    mask_off_j = offs_j[None, :] < N
    # compute mask loads (broadcast)
    mask_ptrs = mask_ptr + pid_b * stride_mask_b + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j
    mask = tl.load(mask_ptrs, mask = mask_off_i & mask_off_j, other = tl.constant(0.0, tl.float16))

    # broadcast mask to H dimension

    for k in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = k * BLOCK_H + tl.arange(0, BLOCK_H)
        h_valid = offs_h < H
        # compute addresses for left
        left_ptrs = left_ptr + pid_b * stride_left_b \
                    + offs_i[:, None, None] * stride_left_i \
                    + offs_j[None, :, None] * stride_left_j \
                    + offs_h[None, None, :] * stride_left_h
        right_ptrs = right_ptr + pid_b * stride_right_b \
                    + offs_i[:, None, None] * stride_right_i \
                    + offs_j[None, :, None] * stride_right_j \
                    + offs_h[None, None, :] * stride_right_h

        left = tl.load(left_ptrs, mask = mask_off_i[:, None] & mask_off_j[None, :] & h_valid[None, None, :], other = tl.float16(0.0))
        right = tl.load(right_ptrs, mask = mask_off_i[:, None] & mask_off_j[None, :] & h_valid[None, None, :], other = tl.float16(0.0))

        # broadcast mask to shape (M,N,1) then broadcast multiplication by mask[..., None]
        # mask shape M,N thus can be broadcast: left = left * mask[..., None]; but mask is FP16
        left = left * mask[:, :, None]
        right = right * mask[:, :, None]

        # store back
        tl.store(left_ptrs, left, mask = mask_off_i[:, None, None] & mask_off_j[None, :, None] & h_valid[None, None, :])
        tl.store(right_ptrs, right, mask = mask_off_i[:, None, None] & mask_off_j[None, :, None] & h_valid[None, None, :])

"""
TriMul (outgoing) forward pass using Triton for optional masking.
Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Linear projections (left / right) from dim → hidden_dim.
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Apply left / right gates, then (optionally) broadcast a pair mask over the
   hidden dimension using a small Triton kernel that multiplies the projected
   tensors in‑place.
5. Cast the gated tensors to FP16 and perform the core O(N³) contraction:
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   which is a batched GEMM over the hidden dimension (left @ rightᵀ).
   The GEMM is performed with torch.bmm on FP16 tensors (Tensor‑cores).
6. Layer‑norm the result over hidden_dim, multiply by the output gate,
   and apply a final linear projection back to dim.
The function returns a float32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _apply_mask_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr
):
    """In‑place element‑wise mask multiplication for left/right tensors.
    left/right : [B, N, N, H] (FP16)
    mask       : [B, N, N]       (FP16, broadcasted on H)
    """
    pid_b = tl.program_id(0)      # batch index
    pid_i = tl.program_id(1)      # block row (i)
    pid_j = tl.program_id(2)      # block col (j)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    i_valid = offs_i < N
    j_valid = offs_j < N
    ij_valid = i_valid[:, None] & j_valid[None, :]

    # Load mask tile (no H dimension)
    mask_offs = pid_b * stride_mask_b + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j
    mask_tile = tl.load(mask_ptr + mask_offs, mask=ij_valid, other=tl.float16(0.0))

    for k in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = k * BLOCK_H + tl.arange(0, BLOCK_H)
        h_valid = offs_h < H

        # Pointers for left and right
        left_offs = (
            pid_b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_j[None, :, None] * stride_left_j
            + offs_h[None, None, :] * stride_left_h
        )
        right_offs = (
            pid_b * stride_right_b
            + offs_i[:, None, None] * stride_right_i
            + offs_j[None, :, None] * stride_right_j
            + offs_h[None, None, :] * stride_right_h
        )

        # Load tiles
        left_tile = tl.load(
            left_ptr + left_offs,
            mask=ij_valid[:, :, None] & h_valid[None, None, :],
            other=tl.float16(0.0),
        )
        right_tile = tl.load(
            right_ptr + right_offs,
            mask=ij_valid[:, :, None] & h_valid[None, None, :],
            other=tl.float16(0.0),
        )

        # Apply mask (broadcast mask over H)
        left_tile = left_tile * mask_tile[:, :, None]
        right_tile = right_tile * mask_tile[:, :, None]

        # Store back
        tl.store(
            left_ptr + left_offs,
            left_tile,
            mask=ij_valid[:, :, None] & h_valid[None, None, :],
        )
        tl.store(
            right_ptr + right_offs,
            right_tile,
            mask=ij_valid[:, :, None] & h_valid[None, None, :],
        )

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: Tuple containing:
            - input_tensor (torch.Tensor): [B, N, N, dim] (float32)
            - mask (torch.Tensor or None): [B, N, N] (float32, 0/1)
            - weights (dict of torch.Tensor): model parameters (float32)
            - config (dict): contains 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]) # [B, N, N, hidden_dim]

    # --------------------------------------------------------------------- #
    # 3. Gating (computed from the normalized input)
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, hidden_dim]

    # Apply left/right gates (float32)
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 4. Cast to half precision for the heavy contraction
    # --------------------------------------------------------------------- #
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 5. Optional mask (broadcasted on hidden_dim) – Triton kernel
    # --------------------------------------------------------------------- #
    if mask is not None:
        # Cast mask to half‑precision
        mask_h = mask.to(torch.float16)

        B, N, _, H = left.shape  # H == hidden_dim
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_H = 32

        grid = (
            B,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
        )

        _apply_mask_kernel[grid](
            left,
            right,
            mask_h,
            B,
            N,
            H,
            left.stride(0),
            left.stride(1),
            left.stride(2),
            left.stride(3),
            right.stride(0),
            right.stride(1),
            right.stride(2),
            right.stride(3),
            mask_h.stride(0),
            mask_h.stride(1),
            mask_h.stride(2),
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_H=BLOCK_H,
        )

    # --------------------------------------------------------------------- #
    # 6. Core O(N³) contraction via batched GEMM (Tensor‑core accelerated)
    # --------------------------------------------------------------------- #
    #   left:  [B, N, N, H]  --> [B, H, N, N]
    #   right: [B, N, N, H]  --> transpose last two dims then [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()        # [B, H, N, N]
    right_t = right_perm.transpose(-2, -1).contiguous()       # [B, H, N, N] (k‑j)

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)   # (B·H, N, N)
    right_mat = right_t.view(B * H, N, N)    # (B·H, N, N)

    # Batched half‑precision matrix multiplication (accumulates in FP16/F32 via Tensor Cores)
    out_mat = torch.bmm(left_mat, right_mat)   # (B·H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # back to FP32 for the remaining ops

    # --------------------------------------------------------------------- #
    # 7. Output LayerNorm, output‑gate, final projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
