"""
TriMul outgoing update (AlphaFold3) – Triton‑accelerated core multiplication.

Algorithm
---------
1. Layer‑norm the pair representation (shape [B, N, N, C]).
2. Linear projections (left/right) and three gating projections (left_gate,
   right_gate, out_gate) – all bias‑free.
3. Apply optional pair mask and element‑wise gating.
4. Core update:    out[b, i, j, h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   → a batched matrix multiplication A @ Bᵀ for each hidden channel h.
   The multiplication is performed by a custom Triton kernel that tiles the
   (M, N) output and iterates over K in blocks, using fp16 arithmetic with
   fp32 accumulation.
5. Hidden‑dim layer‑norm, out‑gate, and final linear projection back to the
   original channel dimension.

Only the matmul (step 4) is written in Triton; all other operations use
native PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
import math
from torch.nn import functional as F


@triton.jit
def _trigemul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # tensor pointers
    M, N, K,                           # matrix dimensions
    stride_a_batch, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_b_batch, stride_bk, stride_bn,   # strides for B (batch, K, N) – B is already transposed
    stride_c_batch, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ B  (A:[batch,M,K], B:[batch,K,N])"""
    pid_z = tl.program_id(2)  # batch index (B * hidden_dim)
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col

    # --- tile offsets ---
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # --- base pointers for this batch ---
    a_batch_ptr = a_ptr + pid_z * stride_a_batch
    b_batch_ptr = b_ptr + pid_z * stride_b_batch
    c_batch_ptr = c_ptr + pid_z * stride_c_batch

    # --- pointers to the current tile of A and B ---
    a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K chunks
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)               # fp16 * fp16 → fp32 accumulation
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the tile
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    pair_tensor, mask, weights, config = data
    device = pair_tensor.device
    B, N, N2, C = pair_tensor.shape
    assert N == N2, "pair tensor must be square in the sequence dimensions"
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension C
    # ------------------------------------------------------------------
    x = F.layer_norm(pair_tensor, (C,), weight=norm_w, bias=norm_b, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    # shape after projection: [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_w)   # left_proj_w: [hidden_dim, dim]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask_val = torch.ones((B, N, N), dtype=pair_tensor.dtype, device=device)
    else:
        mask_val = mask.to(dtype=pair_tensor.dtype, device=device)
        if mask_val.dim() != 3:
            raise ValueError("Mask must have shape [B, N, N]")
    mask_val = mask_val.unsqueeze(-1)        # [B, N, N, 1]
    left = left * mask_val
    right = right * mask_val

    # ------------------------------------------------------------------
    # 4. Gating tensors (sigmoid on linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core MatMul (batched GEMM) using Triton
    #    left : [B, N, N, hidden_dim]  →  (B, hidden_dim, N, N)
    #    right: same shape, transposed on the last two axes before matmul.
    # ------------------------------------------------------------------
    # Re‑arrange so hidden dimension becomes batch dimension
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    # Combine batch and hidden_dim into a single batch axis for Triton
    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)               # [BH, M=N, K=N]
    # transpose last two axes of right (K ↔ N) for A @ Bᵀ
    right_T = right.view(BH, N, N).transpose(-2, -1).contiguous()  # [BH, K=N, N]

    # Cast to fp16 for the kernel (reduce memory bandwidth)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_T.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)   # accumulator

    # --------------------------------------------------------------
    # Triton launch configuration
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles over M (i dimension)
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles over N (j dimension)
        BH,                             # batch * hidden_dim
    )

    _trigemul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N, N, N,
        # strides for A
        left_fp16.stride(0),   # batch stride
        left_fp16.stride(1),   # M stride (i)
        left_fp16.stride(2),   # K stride (k)
        # strides for B (transposed)
        right_fp16.stride(0),  # batch stride
        right_fp16.stride(1),  # K stride (k)
        right_fp16.stride(2),  # N stride (j)
        # strides for C
        out_fp16.stride(0),    # batch stride
        out_fp16.stride(1),    # M stride
        out_fp16.stride(2),    # N stride
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to float32 and reshape to original layout
    out = out_fp16.to(torch.float32)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w,
                       bias=to_out_norm_b, eps=eps)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # back to original dim (C == dim)

    return out
```
