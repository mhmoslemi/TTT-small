"""
Triton implementation of the outgoing TriMul operator (AlphaFold3).
The heavy O(N³) contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is performed as a batched half‑precision GEMM with fp32 accumulation.
All other linear layers, LayerNorms and gating are executed with PyTorch.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (A and B are [batch, M, K] and [batch, K, N])
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_a_batch, stride_b_batch, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col
    pid_b = tl.program_id(2)  # batch (batch * hidden_dim)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_am
    b_base = b_ptr + pid_b * stride_b_batch + offs_n[None, :] * stride_bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block (M x K)
        a = tl.load(
            a_base + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B block (K x N)
        b = tl.load(
            b_base + cur_k[:, None] * stride_bk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Tensor‑core dot (fp16 inputs, fp32 accumulation)
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# Custom forward kernel for the TriMul “outgoing” operator.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul outgoing operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [bs, N, N, dim] (float32)
            mask:        torch.Tensor of shape [bs, N, N]  (float32 or bool) or None
            weights:     dict of torch.Tensors for all linear/ln weights
            config:      dict with keys:
                         - "dim": input channel dimension
                         - "hidden_dim": projection dimension
                         - "nomask": bool (True => ignore mask)
    Returns:
        out: torch.Tensor of shape [bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # convenience aliases for weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [bs, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [bs, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [bs, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # ------------------------------------------------------------------
    # 4. Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [bs, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Prepare tensors for the fused GEMM (fp16 inputs)
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    bs, N, _, _ = left.shape  # N is seq_len

    # left  -> (bs, hidden_dim, N, N)  then (batch_h, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    # right -> (bs, hidden_dim, N, N) with inner dims transposed
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(-1, N, N)

    # Output buffer (FP32 for higher precision)
    out_mat = torch.empty(
        left_mat.shape[0], N, N,
        dtype=torch.float32,
        device=left_mat.device,
    )

    # ------------------------------------------------------------------
    # 7. Extract strides for the kernel
    # ------------------------------------------------------------------
    stride_a_batch, stride_am, stride_ak = left_mat.stride()
    stride_b_batch, stride_bk, stride_bn = right_mat.stride()
    stride_c_batch, stride_cm, stride_cn = out_mat.stride()

    # ------------------------------------------------------------------
    # 8. Launch the Triton kernel (batched GEMM)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # multiple of 16 → tensor‑core friendly

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    batch_h = left_mat.shape[0]

    grid = (num_m, num_n, batch_h)

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,          # M
        N,          # N
        N,          # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9. Reshape back to [bs, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 10. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    # 11. Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12. Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # no bias

    return out
```
