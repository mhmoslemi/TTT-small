"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The heavy contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication using a Triton kernel,
where `right` is transposed so that a single GEMM per (batch, d) yields the
desired result. All other operations (LayerNorm, Linear projections, gates)
are performed with PyTorch for simplicity. The kernel runs in FP16 with FP32
accumulation and supports optional masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for batched GEMM: C = A @ B   (A: (M,K), B: (K,N))
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_a_batch, stride_b_batch, stride_c_batch,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile row index
    pid_n = tl.program_id(1)  # tile col index
    pid_b = tl.program_id(2)  # batch (batch * hidden_dim) index

    # Compute the absolute offsets of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks for M and N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load a block of A
        a = tl.load(
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a block of B
        b = tl.load(
            b_ptr
            + pid_b * stride_b_batch
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16 * FP16 -> FP32 accumulation via tl.dot
        acc += tl.dot(a, b)

    # Write the result back to C
    c = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of weight tensors (see skeleton)
            - config: dict containing at least keys:
                * "dim" (input channel dimension)
                * "hidden_dim" (intermediate channel dimension)
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False) and mask is not None

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcasted on channel dim)
    # ------------------------------------------------------------------
    if use_mask:
        mask_f = mask.to(dtype)  # ensure same dtype as x
        mask_f = mask_f.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5. Apply per‑channel gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Cast to FP16 for the heavy contraction
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 7. Rearrange tensors for batched matmul: (B*H, N, N)
    #    left:   (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    #    right:  (B, N, N, H) -> transpose last two dims -> (B, H, N, N)
    # ------------------------------------------------------------------
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)

    # Output buffer (FP32 accumulator)
    out_reshape = torch.empty(
        (bs * hidden_dim, N, N), dtype=torch.float32, device=device
    )

    # ------------------------------------------------------------------
    # 8. Strides for the kernel
    # ------------------------------------------------------------------
    stride_a_batch, stride_am, stride_ak = left_reshape.stride()
    stride_b_batch, stride_bk, stride_bn = right_reshape.stride()
    stride_c_batch, stride_cm, stride_cn = out_reshape.stride()

    # Block sizes (tuned for NVIDIA H100 / B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid dimensions
    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_m, num_n, bs * hidden_dim)

    # ------------------------------------------------------------------
    # 9. Launch the Triton kernel
    # ------------------------------------------------------------------
    tri_mul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 10. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = (
        out_reshape.view(bs, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 11. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 12. Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 13. Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
