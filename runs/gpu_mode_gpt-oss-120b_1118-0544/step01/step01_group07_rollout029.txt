def fused_mul_kernel(
    left_ptr, left_gate_ptr, mask_ptr,
    right_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    B, N, H,
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_g0, stride_g1, stride_g2, stride_g3,
    stride_m0, stride_m1, stride_m2,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total
    if not tl.any(mask):
        return

    b = idx // N

"""
TriMul (outgoing) forward pass with a fused elementwise Triton kernel and
cuBLAS batched GEMM. The heavy O(N³) part is performed by torch.bmm on
float16 tensors (tensor‑core accelerated). A small Triton kernel fuses the
mask, left‑gate and right‑gate multiplications and stores the result in
float16, reducing memory traffic. The implementation respects the API:
custom_kernel(data) -> output tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask and gates into the projected tensors.
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    left_ptr,            # float32  [B, N, N, H]
    left_gate_ptr,       # float32  [B, N, N, H]
    right_ptr,           # float32  [B, N, N, H]
    right_gate_ptr,      # float32  [B, N, N, H]
    mask_ptr,            # float32  [B, N, N]
    left_out_ptr,        # float16  [B, N, N, H]
    right_out_ptr,       # float16  [B, N, N, H]
    B, N, H,
    # strides (in elements)
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_g0, stride_g1, stride_g2, stride_g3,
    stride_m0, stride_m1, stride_m2,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total
    if not tl.any(mask):
        return

    # Decode linear index into (b, i, j, h)
    idx = offs
    h = idx % H
    idx = idx // H
    j = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # Compute flat offsets for each tensor
    off_l = (
        b * stride_l0
        + i * stride_l1
        + j * stride_l2
        + h * stride_l3
    )
    off_g = (
        b * stride_g0
        + i * stride_g1
        + j * stride_g2
        + h * stride_g3
    )
    off_m = b * stride_m0 + i * stride_m1 + j * stride_m2

    # Load values
    l = tl.load(left_ptr + off_l, mask=mask, other=0.0)          # [BLOCK]
    lg = tl.load(left_gate_ptr + off_g, mask=mask, other=0.0)
    r = tl.load(right_ptr + off_l, mask=mask, other=0.0)        # same layout as left
    rg = tl.load(right_gate_ptr + off_g, mask=mask, other=0.0)
    m = tl.load(mask_ptr + off_m, mask=mask, other=1.0)

    # Apply mask & gates, cast to float16 for storage
    out_l = tl.cast(l * lg * m, tl.float16)
    out_r = tl.cast(r * rg * m, tl.float16)

    # Store results
    tl.store(left_out_ptr + off_l, out_l, mask=mask)
    tl.store(right_out_ptr + off_l, out_r, mask=mask)

        stride_m2,

        num_warps=4,

    # reshape to (B*hidden, N, N) and keep half precision for tensor cores

        left_fused.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    )  # [BH, N, N]  (half)
    right_mat = (
        right_fused.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    )  # [BH, N, N]  (half)

    out_mat = torch.bmm(left_mat, right_mat.transpose(-2, -1))  # [BH, N, N] half

    # reshape back to [B, N, N, hidden_dim] and convert to float32
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .float()
    )  # [B, N, N, hidden_dim]

    # ---- Output LayerNorm, gating and final projection -----------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, to_out_weight)  # (dim, hidden_dim) weight, no bias

    return output
```
