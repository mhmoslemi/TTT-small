def custom_kernel(data):

    input_tensor, mask, weights, config = data

    bs, N, _, dim = input_tensor.shape

    hidden_dim = config["hidden_dim"]
    use_mask = mask is not None and not config.get("nomask", False)

    # load all weight tensors
    # typical weights shape: for linear layers: weight shape [hidden_dim, dim] etc.
    # for layernorm: shape [dim] or [hidden_dim].

    # Do layernorm on input
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Left and right projection
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply mask
    if use_mask:
        mask_exp = mask.unsqueeze(-1).type_as(left)  # broadcast to same dtype
        left = left * mask_exp
        right = right * mask_exp

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for matmul
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Reshape for batched matmul
    left_resh = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_resh = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)

    # Allocate output
    out_resh = torch.empty((bs * hidden_dim, N, N), dtype=torch.float32, device=left_resh.device)

    # Strides
    stride_a_batch, stride_am, stride_ak = left_resh.stride()
    stride_b_batch, stride_bk, stride_bn = right_resh.stride()
    stride_c_batch, stride_cm, stride_cn = out_resh.stride()

    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )

    matmul_kernel[grid](
        left_resh,
        right_resh,
        out_resh,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back
    out = out_resh.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # To_out_norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # Output gate

"""
TriMul (outgoing) forward pass implemented with a fused Triton batched matmul.

Algorithm:
1. Layer‑norm the input tensor over the feature dimension.
2. Linear projections (left, right) and three sigmoidal gates (left, right, out).
3. Apply optional mask and gates to the projected tensors.
4. Compute the heavy contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batch of independent matrix multiplications
   left_d @ right_d.T for each hidden channel d.
   The multiplication is performed in fp16 with fp32 accumulation
   by a custom Triton kernel.
5. Reshape the result, apply a second LayerNorm, multiply by out_gate,
   and finally a linear projection back to the original feature dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_a_batch, stride_b_batch, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication C = A @ B  (FP16 inputs, FP32 accumulation)."""
    pid_m = tl.program_id(0)      # tile row
    pid_n = tl.program_id(1)      # tile col
    pid_b = tl.program_id(2)      # batch*channel

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for kk in range(0, K, BLOCK_K):
        cur_k = kk + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A sub‑matrix
        a = tl.load(
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_am
            + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B sub‑matrix
        b = tl.load(
            b_ptr
            + pid_b * stride_b_batch
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate dot product
        acc += tl.dot(a, b)

    # Store the result
    c = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of TriMul (outgoing) using Triton for the N³ contraction.
    Input:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Output:
        Tensor of shape [bs, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # Configuration & shapes
    # ------------------------------------------------------------------
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = (mask is not None) and (not config.get("nomask", False))

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Optional mask
    # ------------------------------------------------------------------
    if use_mask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # [bs, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Batched matmul using Triton (left @ rightᵀ per hidden channel)
    # ------------------------------------------------------------------
    # Cast to fp16 for bandwidth‑limited matmul; accumulation stays fp32.
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Reshape to (batch*hidden_dim, N, N) where the second tensor is transposed.
    left_resh = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_resh = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)

    # Output buffer in FP32 (higher precision for the sum over K).
    out_resh = torch.empty(
        (bs * hidden_dim, N, N), dtype=torch.float32, device=left_resh.device
    )

    # Strides for the three tensors
    stride_a_batch, stride_am, stride_ak = left_resh.stride()
    stride_b_batch, stride_bk, stride_bn = right_resh.stride()
    stride_c_batch, stride_cm, stride_cn = out_resh.stride()

    # Tiling parameters – 64 is a safe choice for most modern GPUs.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # tiles over M (i dimension)
        (N + BLOCK_N - 1) // BLOCK_N,          # tiles over N (j dimension)
        bs * hidden_dim,                       # batch * channel
    )

    _matmul_kernel[grid](
        left_resh,
        right_resh,
        out_resh,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [bs, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_resh.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 9. Output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
