"""
TriMul (outgoing) forward kernel.

The heavy part is the 4‑D contraction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
for each hidden dimension *d*.  By reshaping to (B*D, N, N) we can view it as
a batched matrix multiplication `C = A @ Bᵀ` where each batch corresponds to a
different hidden channel.  The Triton kernel below computes this GEMM with
FP16 inputs and FP32 accumulation, using tiling (BLOCK_M×BLOCK_N) and a K‑loop
with tile size BLOCK_K.  LayerNorms, linear projections and gating are kept in
PyTorch; only the O(N³) contraction is off‑loaded to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ Bᵀ.
    A: (batch, M, K)   (float16)
    B: (batch, N, K)   (float16) – accessed as transpose on the fly
    C: (batch, M, N)   (float32 accumulator)
    """
    pid_batch = tl.program_id(0)   # batch * hidden dim
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile col

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K‑loop
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + \
                 offs_m[:, None] * stride_a_row + \
                 cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load Bᵀ block: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + \
                 cur_k[:, None] * stride_b_col + \
                 offs_n[None, :] * stride_b_row
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Matrix multiply‐accumulate
        acc += tl.dot(a, b)

    # Write C block
    c_ptrs = c_batch + \
             offs_m[:, None] * stride_c_row + \
             offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (tensors on same device)
            - config: dict with at least "dim" and "hidden_dim" and optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------- LayerNorm ----------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -------- Linear projections ----------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -------- Mask (if any) ----------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1]
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # -------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # -------- Apply gates ----------
    left = left * left_gate
    right = right * right_gate

    # -------- Cast to half for the kernel ----------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # -------- Prepare tensors for Triton GEMM ----------
    # Shape: (B, N, N, hidden_dim) -> (B, hidden_dim, N, N) -> (B*hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (FP32 accumulator)
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # Strides (in elements)
    stride_a_batch = left_mat.stride(0)
    stride_a_row = left_mat.stride(1)
    stride_a_col = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_row = right_mat.stride(1)
    stride_b_col = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_row = out_mat.stride(1)
    stride_c_col = out_mat.stride(2)

    # Tile sizes – choose based on N for a modest balance
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch Triton kernel
    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # -------- Reshape back to 4‑D ----------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # -------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # -------- Apply output gate ----------
    out = out * out_gate

    # -------- Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
