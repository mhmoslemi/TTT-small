"""
TriMul (outgoing) forward pass.
-------------------------------------------------
1. Layer‑norm the input.
2. Linear projections → left, right (shape [B,N,N,H]).
3. Optional pair‑wise mask applied to left/right.
4. Gating: sigmoid(linear(x_norm)) → left_gate, right_gate, out_gate.
5. Apply gates to left/right.
6. Batched GEMM with Triton: for each batch‑hidden slice compute
       out_slice = left_slice @ right_slice.T
   (shape [H*B, N, N]) using fp16 inputs, fp32 accumulation.
7. Reshape back to [B,N,N,H].
8. Output layer‑norm, multiply by out_gate, final linear to original dim.
All heavy O(N³·H) work is done in the Triton kernel.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr,
    b_ptr,
    c_ptr,
    M,
    N,
    K,
    stride_a_batch,
    stride_a_row,
    stride_a_col,
    stride_b_batch,
    stride_b_row,
    stride_b_col,
    stride_c_batch,
    stride_c_row,
    stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ.
    A, B: (batch, M, K) and (batch, N, K) in row‑major layout.
    C: (batch, M, N)
    """
    pid_batch = tl.program_id(0)  # which (batch*hidden) matrix
    pid_m = tl.program_id(1)      # tile row index
    pid_n = tl.program_id(2)      # tile col index

    # Compute the row/col offsets for this tile.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch slice.
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in BLOCK_K chunks.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a_offs = a_batch + (offs_m[:, None] * stride_a_row) + (cur_k[None, :] * stride_a_col)
        a = tl.load(
            a_offs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
        )

        # Load Bᵀ tile: we need Bᵀ of shape (BLOCK_K, BLOCK_N)
        b_offs = b_batch + (cur_k[:, None] * stride_b_col) + (offs_n[None, :] * stride_b_row)
        b = tl.load(
            b_offs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0.0),
        )

        # Accumulate the matrix product.
        acc += tl.dot(a, b)

    # Write the result tile.
    c_offs = c_batch + (offs_m[:, None] * stride_c_row) + (offs_n[None, :] * stride_c_col)
    tl.store(c_offs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float) or None
            - weights: dict with the model parameters (tensors on the same device)
            - config: dict containing at least "dim" and "hidden_dim".
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input layer‑norm
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional pair‑wise mask
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape -> [B, N, N, 1] for broadcasting
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # float mask
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid of a linear projection)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 5. Apply gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Convert to fp16 and reshape for batched GEMM
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, seq_len, seq_len, hidden_dim

    # Rearrange to [B*hidden, N, N] – each slice is a matrix to multiply.
    left_mat = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (float32 accumulator)
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # -----------------------------------------------------------------
    # 7. Triton GEMM launch parameters
    # -----------------------------------------------------------------
    # Choose block sizes based on sequence length (heuristic)
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    # Strides in number of elements (row‑major layout)
    stride_a_batch = left_mat.stride(0)
    stride_a_row = left_mat.stride(1)
    stride_a_col = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_row = right_mat.stride(1)   # original rows (j dimension)
    stride_b_col = right_mat.stride(2)   # original cols (k dimension)

    stride_c_batch = out_mat.stride(0)
    stride_c_row = out_mat.stride(1)
    stride_c_col = out_mat.stride(2)

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the Triton kernel
    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # -----------------------------------------------------------------
    # 8. Reshape back to [B, N, N, hidden_dim]
    # -----------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden]

    # -----------------------------------------------------------------
    # 9. Output layer‑norm, gate, final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])

    return out
