    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute batch offset
    a_batch_ptr = a_ptr + pid * stride_ah
    b_batch_ptr = b_ptr + pid * stride_bh
    c_batch_ptr = c_ptr + pid * stride_ch

    # offsets for M,N block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundary
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # initial block pointers for A and B
    a_block_ptr = tl.make_block_ptr(
        base=a_batch_ptr,
        shape=(M, K),
        strides=(stride_am, stride_ak),
        offsets=(pid_m * BLOCK_M, 0),
        block_shape=(BLOCK_M, BLOCK_K),
        order=(0, 1)
    )
    b_block_ptr = tl.make_block_ptr(
        base=b_batch_ptr,
        shape=(K, N),
        strides=(stride_bk, stride_bn),
        offsets=(0, pid_n * BLOCK_N),
        block_shape=(BLOCK_K, BLOCK_N),
        order=(0, 1)
    )
    # prefetch first iteration
    a = tl.load(a_block_ptr, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K), other=0.0)
    b = tl.load(b_block_ptr, mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute stage
        # Load next block (if not final)
        if k < tl.cdiv(K, BLOCK_K) - 1:
            a_next = tl.load(tl.advance(a_block_ptr, (0, BLOCK_K)), mask=mask_m[:, None] & ((tl.arange(0, BLOCK_K) + (k+1)*BLOCK_K) < K), other=0.0)

        a_block_ptr = tl.advance(a_block_ptr, (0, BLOCK_K))
        b_block_ptr = tl.advance(b_block_ptr, (BLOCK_K, 0))

            b = b_next

        strides=(stride_cm, stride_cn),

    )

"""
TriMul (outgoing) forward pass with a fused Triton kernel for the
N³ contraction. The kernel performs a batched matrix multiplication:
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
where left and right are the linearly‑projected and gated tensors.
All other operations (layer‑norms, gates, final projection) are done in
PyTorch. The kernel is a straightforward tiled GEMM (block‑size 64) that
accumulates in FP32 while loading FP16, achieving high throughput on
GPU B200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_ah, stride_am, stride_ak,   # A strides: batch, row (i), col (k)
    stride_bh, stride_bk, stride_bn,   # B strides: batch, row (k), col (j)
    stride_ch, stride_cm, stride_cn,   # C strides: batch, row (i), col (j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched FP16 GEMM with FP32 accumulation (no boundary checks, all
    dimensions are assumed to be multiples of the block sizes)."""

    # Program IDs
    pid_batch = tl.program_id(0)          # batch * head index
    pid_m = tl.program_id(1)              # output row block
    pid_n = tl.program_id(2)              # output column block

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # (BLOCK_M,)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # (BLOCK_N,)

    # Base pointers for this batch/head
    a_batch = a_ptr + pid_batch * stride_ah
    b_batch = b_ptr + pid_batch * stride_bh
    c_batch = c_ptr + pid_batch * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, K, BLOCK_K):
        k_offsets = k + tl.arange(0, BLOCK_K)                      # (BLOCK_K,)

        # Load a tile of A and B (FP16)
        a = tl.load(
            a_batch + offs_m[:, None] * stride_am + k_offsets[None, :] * stride_ak
        )
        b = tl.load(
            b_batch + k_offsets[:, None] * stride_bk + offs_n[None, :] * stride_bn
        )

        # Matrix‑multiply‑accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) ← (BLOCK_M, BLOCK_K)×(BLOCK_K, BLOCK_N)

    # Write the FP32 result back
    tl.store(
        c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
        acc
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: Tuple of (input_tensor, mask_tensor_or_None, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (optional)
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    eps = 1e-5

    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (bias‑less)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden_dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)       # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core N³ contraction via Triton GEMM
    # -----------------------------------------------------------------
    # Cast to half for the GEMM
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to (B, head, N, N) where head = hidden_dim
    # left: [B, N, N, H] -> [B, H, N, N]
    # right: [B, N, N, H] -> [B, H, N, N] and transpose last two axes
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 2, 1).contiguous()   # note the extra transpose

    B, H, N, _ = left_h.shape

    # Collapse batch and head dimensions: (B*H, N, N)
    left_mat = left_h.view(B * H, N, N)
    right_mat = right_h.view(B * H, N, N)

    # Output buffer (FP32 accumulator)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=x.device)

    # Tile sizes – 64 works well for the target hardware
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute strides for the kernel
    a_stride_batch = left_mat.stride(0)
    a_stride_m = left_mat.stride(1)   # stride over i (row)
    a_stride_k = left_mat.stride(2)   # stride over k (col)

    b_stride_batch = right_mat.stride(0)
    b_stride_k = right_mat.stride(1)   # stride over k (row)
    b_stride_n = right_mat.stride(2)   # stride over j (col)

    c_stride_batch = out_mat.stride(0)
    c_stride_m = out_mat.stride(1)
    c_stride_n = out_mat.stride(2)

    # Grid: (batch_head, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        a_stride_batch,
        a_stride_m,
        a_stride_k,
        b_stride_batch,
        b_stride_k,
        b_stride_n,
        c_stride_batch,
        c_stride_m,
        c_stride_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # adequate parallelism on B200
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
