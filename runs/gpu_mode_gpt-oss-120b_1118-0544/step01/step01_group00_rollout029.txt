Outgoing Triangle Multiplicative Update (TriMul) kernel.

The core of TriMul is a batched matrix multiplication:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
For each hidden dimension `d` (and batch) this is a GEMM A @ Bᵀ where
A and B have shape (N, N).  The kernel below implements a high‑performance
blocked GEMM in FP16 (inputs) with FP32 accumulation.  All surrounding
LayerNorms, linear projections and gating are performed with PyTorch.
"""

import triton.language as tl


@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    batch, M, N, K,                           # dimensions (compile‑time consts)

    stride_c_batch, stride_c_m, stride_c_n,   # strides of C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

    """Batched GEMM C = A @ Bᵀ  (A: M×K, B: N×K)"""
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_z = tl.program_id(2)          # batch dimension (B * hidden_dim)

    # ---------------------------------------------------------
    # block offsets
    # ---------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ---------------------------------------------------------
    # base pointers for this batch
    # ---------------------------------------------------------
    a_batch = a_ptr + pid_z * stride_a_batch
    b_batch = b_ptr + pid_z * stride_b_batch
    c_batch = c_ptr + pid_z * stride_c_batch

    # ---------------------------------------------------------
    # accumulator in FP32
    # ---------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------
    # loop over reduction axis K in blocks
    # ---------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)

        # ----- load A tile (BLOCK_M x BLOCK_K) ----- #
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (k_offset[None, :] * stride_a_k)
        a_mask = (offs_m[:, None] < M) & (k_offset[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # ----- load B tile as Bᵀ (BLOCK_K x BLOCK_N) ----- #
        b_ptrs = b_batch + (k_offset[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b_mask = (k_offset[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # ----- MAC -----
        acc += tl.dot(a, b)

    # ---------------------------------------------------------
    # write back C tile
    # ---------------------------------------------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – may be None
        - weights      : dict of torch.Tensor containing model parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ---------------------------------------------------------
    # unpack inputs
    # ---------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------------------------------------------------------
    # unpack model weights
    # ---------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ---------------------------------------------------------
    # 1) LayerNorm over channel dimension
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ---------------------------------------------------------
    # 2) Linear projections (no bias)
    # ---------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ---------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ---------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ---------------------------------------------------------
    # 4) Gating
    # ---------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))     # used after the matmul

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 5) Cast to FP16 for the heavy reduction
    # ---------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, seq_len, seq_len, hidden_dim

    # reshape to (batch*hidden_dim, N, N) where each slice is a matrix
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # output buffer (FP16)
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # ---------------------------------------------------------
    # 6) Triton batched GEMM launch
    # ---------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    # grid: (tiles over M, tiles over N, batch*hidden_dim)
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    batch_dim = left_h.shape[0]

    grid = (num_pid_m, num_pid_n, batch_dim)

    # strides in element units
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)

    stride_b_batch = right_h.stride(0)
    stride_b_k = right_h.stride(2)   # innermost
    stride_b_n = right_h.stride(1)   # next

    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    tri_mul_matmul_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        batch=batch_dim,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion
    torch.cuda.synchronize()

    # ---------------------------------------------------------
    # 7) Restore shape -> [B, N, N, hidden_dim] (float32)
    # ---------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)                 # [B, N, N, hidden_dim]
        .contiguous()
        .to(torch.float32)
    )

    # ---------------------------------------------------------
    # 8) Final LayerNorm, gating and projection back to original dim
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                         # element‑wise gating
    out = F.linear(out, to_out_weight)           # [B, N, N, dim]

    return out
