"""
TriMul (outgoing) forward pass with a custom Triton batched GEMM.

The heavy O(N³·hidden_dim) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a Triton kernel that multiplies, for each (batch, d),
the matrix A = left[b,d,i,k] (i × k) with Bᵀ = right[b,d,k,j] (k × j).
All other operations (LayerNorm, Linear projections, gates, mask handling)
are done with PyTorch (cuBLAS). The kernel uses FP16 for inputs and FP32
accumulation, tiled with configurable block sizes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _triton_trimul_gemm(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dims
    stride_am, stride_ak,              # A strides (M stride, K stride)
    stride_bk, stride_bn,              # B strides (K stride, N stride)
    stride_cm, stride_cn,              # C strides (M stride, N stride)
    stride_a_batch, stride_a_hidden,   # A batch & hidden strides
    stride_b_batch, stride_b_hidden,   # B batch & hidden strides
    stride_c_batch, stride_c_hidden,   # C batch & hidden strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid_m = tl.program_id(0)      # tile row
    pid_n = tl.program_id(1)      # tile col
    pid_bh = tl.program_id(2)     # combined batch * hidden index

    # split combined index into batch and hidden
    batch_idx = pid_bh // HIDDEN_DIM
    hidden_idx = pid_bh % HIDDEN_DIM

    # base pointers for this (batch, hidden) slice
    a_batch = a_ptr + batch_idx * stride_a_batch + hidden_idx * stride_a_hidden
    b_batch = b_ptr + batch_idx * stride_b_batch + hidden_idx * stride_b_hidden
    c_batch = c_ptr + batch_idx * stride_c_batch + hidden_idx * stride_c_hidden

    # tile offsets within the matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # looping over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # pointers for the current K‑tile
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (cur_k[None, :] * stride_ak)
        b_ptrs = b_batch + (cur_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # write result
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    The core contraction is executed by a Triton kernel;
    all other linear layers / LayerNorms are performed with PyTorch.
    """
    input_tensor, mask, weights, config = data

    # --------------------------------------------------------------
    # unpack sizes and configuration
    # --------------------------------------------------------------
    bs, N, N2, dim = input_tensor.shape
    assert N == N2, "input must be square on the last two axes"
    hidden_dim = config["hidden_dim"]
    use_mask = mask is not None and not config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # --------------------------------------------------------------
    # 4. Optional mask
    # --------------------------------------------------------------
    if use_mask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # [bs, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 5. Apply gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Cast to fp16 for the heavy contraction
    # --------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # --------------------------------------------------------------
    # 7. Prepare tensors for the batched GEMM
    #    left:  (bs, hidden_dim, N, N)   –> A[i,k]
    #    right: (bs, hidden_dim, N, N)   –> B[j,k] → need Bᵀ (k,j)
    # --------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2)          # (bs, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 1, 2)        # (bs, hidden_dim, N, N)
    right_T = right_perm.transpose(-2, -1)      # (bs, hidden_dim, N, N), now (k,j)

    # Output buffer (fp32)
    out_buf = torch.empty_like(left_perm, dtype=torch.float32)

    # --------------------------------------------------------------
    # 8. Extract strides for the kernel
    # --------------------------------------------------------------
    stride_a_batch = left_perm.stride(0)
    stride_a_hidden = left_perm.stride(1)
    stride_am = left_perm.stride(2)
    stride_ak = left_perm.stride(3)

    stride_b_batch = right_T.stride(0)
    stride_b_hidden = right_T.stride(1)
    stride_bk = right_T.stride(2)   # K dimension stride
    stride_bn = right_T.stride(3)   # N dimension stride

    stride_c_batch = out_buf.stride(0)
    stride_c_hidden = out_buf.stride(1)
    stride_cm = out_buf.stride(2)
    stride_cn = out_buf.stride(3)

    # --------------------------------------------------------------
    # 9. Launch the Triton kernel
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )

    _triton_trimul_gemm[grid](
        left_perm,
        right_T,
        out_buf,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_a_hidden,
        stride_b_batch,
        stride_b_hidden,
        stride_c_batch,
        stride_c_hidden,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
    )

    # --------------------------------------------------------------
    # 10. Reshape back to [bs, N, N, hidden_dim]
    # --------------------------------------------------------------
    out = out_buf.permute(0, 2, 3, 1).contiguous()   # (bs, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 11. Output LayerNorm over hidden_dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 12. Output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 13. Final linear projection back to original dimension
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
