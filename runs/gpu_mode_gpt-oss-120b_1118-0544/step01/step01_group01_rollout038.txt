def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # Input layernorm
    x = torch.nn.functional.layer_norm(input_tensor,
                                       (dim,),
                                       weight=weights["norm.weight"],
                                       bias=weights["norm.bias"])

    # Flatten for combined linear
    x_flat = x.view(-1, dim)  # (B*N*N, dim)

    # Concatenate projection weights
    proj_weights = [
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ]
    combined_weight = torch.cat(proj_weights, dim=0)  # (5*hidden_dim, dim)

    # Combined linear
    combined = torch.nn.functional.linear(x_flat, combined_weight)  # (M, 5*hidden)

    # Split
    split = torch.split(combined, hidden_dim, dim=1)  # list of 5 tensors each (M, hidden)
    left_proj_flat, right_proj_flat, left_gate_pre_flat, right_gate_pre_flat, out_gate_pre_flat = split

    # Apply sigmoid
    left_gate_flat = torch.sigmoid(left_gate_pre_flat)
    right_gate_flat = torch.sigmoid(right_gate_pre_flat)
    out_gate = torch.sigmoid(out_gate_pre_flat).view(B, N, N, hidden_dim)

    # Reshape projection outputs
    left_proj = left_proj_flat.view(B, N, N, hidden_dim)
    right_proj = right_proj_flat.view(B, N, N, hidden_dim)

    # Prepare mask
    if mask is None:
        mask_exp = torch.ones((B, N, N, hidden_dim), dtype=left_proj.dtype, device=left_proj.device)
    else:
        mask_exp = mask.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()

    # Flatten for kernel
    left_proj_flat_cont = left_proj.contiguous().view(-1)
    left_gate_flat_cont = left_gate_flat.contiguous().view(-1)
    mask_flat_cont = mask_exp.contiguous().view(-1)   # float32

    right_proj_flat_cont = right_proj.contiguous().view(-1)
    right_gate_flat_cont = right_gate_flat.contiguous().view(-1)

    total_elements = left_proj_flat_cont.numel()  # B*N*N*hidden

    # Allocate outputs
    left_half = torch.empty_like(left_proj_flat_cont, dtype=torch.float16)
    right_half = torch.empty_like(right_proj_flat_cont, dtype=torch.float16)

    # launch kernel for left and right
    BLOCK_SIZE = 65536
    grid = ( (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    _apply_gate_mask_kernel[grid](
        left_proj_flat_cont,
        left_gate_flat_cont,
        mask_flat_cont,
        left_half,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    _apply_gate_mask_kernel[grid](
        right_proj_flat_cont,
        right_gate_flat_cont,
        mask_flat_cont,
        right_half,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # reshape for bmm
    left_h = left_half.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_h = right_half.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()

    # flatten batch*hidden
    left_mat = left_h.reshape(B * hidden_dim, N, N)
    right_mat = right_h.reshape(B * hidden_dim, N, N)

    # batched matmul in half precision
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # convert to float32 for layernorm & gating
    out_hidden_f32 = out_hidden.to(torch.float32)

    # output layer norm
    out_hidden_norm = torch.nn.functional.layer_norm(
        out_hidden_f32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # apply out_gate
    out_hidden_gate = out_hidden_norm * out_gate

    # final projection
    output = torch.nn.functional.linear(out_hidden_gate, weights["to_out.weight"])

    return output

"""
TriMul (outgoing) forward pass – high‑performance implementation.

Algorithm
---------
1. Input LayerNorm (FP32).
2. Fuse the five linear maps (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single GEMM:   y = x_flat @ Wᵀ,
   where W concatenates the five weight matrices along the output dim.
   Split the result and apply sigmoid to the three gate tensors.
3. Fuse mask, gate and conversion to FP16 with a tiny Triton kernel:
   out = (proj * gate * mask).to(fp16)   – performed for left and right.
4. Batched matrix multiply in FP16 using torch.bmm:
   for each hidden channel h:  out_h = left_h @ right_hᵀ .
5. Convert back to FP32, apply the output LayerNorm, the output gate
   and the final linear projection back to the original dimension.
6. Return a tensor of shape [B, N, N, dim] (float32).

Only the element‑wise mask/gate fusion is written in Triton; the
computationally heavy GEMMs are executed with cuBLAS (via torch.nn.functional
and torch.bmm), which is optimal on the B200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _apply_gate_mask_kernel(
    a_ptr,      # left_proj or right_proj   (float32*)
    g_ptr,      # corresponding gate        (float32*)
    m_ptr,      # mask (broadcasted)       (float32*)
    out_ptr,    # output FP16 buffer        (float16*)
    total_n: tl.constexpr,   # total number of elements
    BLOCK_SIZE: tl.constexpr,  # compile‑time tile size
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_n

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)   # float32
    g = tl.load(g_ptr + offs, mask=mask, other=0.0)   # float32
    m = tl.load(m_ptr + offs, mask=mask, other=1.0)   # float32 (1.0 if no mask)

    out = a * g * m                                   # still float32
    tl.store(out_ptr + offs, out.to(tl.float16), mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of model weight tensors
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    B, N, _, _ = x.shape
    device = x.device
    dtype = x.dtype

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (FP32)

    # -----------------------------------------------------------------
    # 2) Fused linear projections (5 * hidden_dim outputs)
    # -----------------------------------------------------------------
    # Flatten the pairwise dimension for a single GEMM
    x_flat = x.view(-1, dim)                         # (M, dim),  M = B*N*N

    # Concatenate the five projection weight matrices
    # each weight has shape (hidden, dim)
    combined_W = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )                                                # (5*hidden, dim)

    # Single GEMM
    combined = F.linear(x_flat, combined_W)          # (M, 5*hidden)

    # Split the combined result
    left_proj_flat, right_proj_flat, left_gate_pre_flat, right_gate_pre_flat, out_gate_pre_flat = torch.split(
        combined, hidden, dim=1
    )  # each: (M, hidden)

    # Gates – sigmoid
    left_gate_flat = torch.sigmoid(left_gate_pre_flat)   # (M, hidden)
    right_gate_flat = torch.sigmoid(right_gate_pre_flat) # (M, hidden)
    out_gate = torch.sigmoid(out_gate_pre_flat).view(B, N, N, hidden)  # (B, N, N, hidden)

    # Reshape projection tensors back to 4‑D
    left_proj = left_proj_flat.view(B, N, N, hidden)
    right_proj = right_proj_flat.view(B, N, N, hidden)

    # -----------------------------------------------------------------
    # 3) Fuse mask, gate and FP16 conversion via Triton
    # -----------------------------------------------------------------
    if mask is None:
        mask_expanded = torch.ones(
            (B, N, N, hidden), dtype=dtype, device=device
        )
    else:
        # mask shape [B, N, N] -> broadcast over hidden
        mask_expanded = (
            mask.to(dtype)
            .unsqueeze(-1)
            .expand(-1, -1, -1, hidden)
            .contiguous()
        )

    # Flatten everything for the 1‑D kernel
    left_proj_f = left_proj.contiguous().view(-1)
    right_proj_f = right_proj.contiguous().view(-1)
    left_gate_f = left_gate_flat.contiguous().view(-1)
    right_gate_f = right_gate_flat.contiguous().view(-1)
    mask_f = mask_expanded.contiguous().view(-1)

    total_elements = left_proj_f.numel()   # = B * N * N * hidden
    BLOCK = 65536
    grid = ( (total_elements + BLOCK - 1) // BLOCK, )

    # Output buffers in FP16
    left_fp16 = torch.empty_like(left_proj_f, dtype=torch.float16)
    right_fp16 = torch.empty_like(right_proj_f, dtype=torch.float16)

    # Left side
    _apply_gate_mask_kernel[grid](
        left_proj_f,
        left_gate_f,
        mask_f,
        left_fp16,
        total_n=total_elements,
        BLOCK_SIZE=BLOCK,
    )
    # Right side
    _apply_gate_mask_kernel[grid](
        right_proj_f,
        right_gate_f,
        mask_f,
        right_fp16,
        total_n=total_elements,
        BLOCK_SIZE=BLOCK,
    )

    # -----------------------------------------------------------------
    # 4) Batched matrix multiplication (FP16)
    # -----------------------------------------------------------------
    # Reshape to (B, hidden, N, N) and then to (B*hidden, N, N)
    left_h = left_fp16.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()
    right_h = right_fp16.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()

    left_mat = left_h.reshape(B * hidden, N, N)
    right_mat = right_h.reshape(B * hidden, N, N)

    # BMM in FP16 – uses tensor cores on the B200
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*hidden, N, N)

    # Restore (B, N, N, hidden) layout
    out_hidden = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden_f32 = out_hidden.to(torch.float32)

    # -----------------------------------------------------------------
    # 5) Output LayerNorm, output gate and final projection
    # -----------------------------------------------------------------
    out_hidden_norm = F.layer_norm(
        out_hidden_f32,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out_hidden_gated = out_hidden_norm * out_gate

    # Final linear projection back to original dimension (FP32)
    out = F.linear(out_hidden_gated, weights["to_out.weight"])

    return out
