    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # cast to fp16
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # reshape for GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # output buffer
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # strides
    stride_a_batch, stride_a_row, stride_a_col = left_mat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_mat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_mat.stride()

    # block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )

    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,hidden)

    # to_out_norm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # apply out_gate
    out = out * out_gate

    # final linear
    out = F.linear(out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm:
1. Layer‑normalize the input tensor.
2. Apply two linear projections (left/right) and optionally a mask.
3. Compute three sigmoid gates (left, right, out) and apply the left/right gates.
4. Cast left/right to fp16 and reshape to (B*H, N, N) where H = hidden_dim.
5. Launch a Triton kernel that computes, for each (batch, hidden) slice,
   C = A @ Bᵀ   with   A = left_mat, B = right_mat.
   This implements   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
6. Reshape the result back to (B, N, N, H), layer‑normalize,
   multiply by the out‑gate, and apply the final linear projection.
All heavy O(N³) work is done in the fused Triton GEMM kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ.
    A: (S, M, K)   B: (S, N, K)   C: (S, M, N)
    where S = B * hidden_dim.
    """
    pid_s = tl.program_id(0)  # slice index (batch * hidden)
    pid_m = tl.program_id(1)  # tile over rows (M)
    pid_n = tl.program_id(2)  # tile over cols (N)

    # Coordinate ranges for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this slice
    a_batch = a_ptr + pid_s * stride_a_batch
    b_batch = b_ptr + pid_s * stride_b_batch
    c_batch = c_ptr + pid_s * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load a block of A (M x K)
        a_offs = (
            a_batch
            + offs_m[:, None] * stride_a_row
            + cur_k[None, :] * stride_a_col
        )
        a = tl.load(
            a_offs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # Load a block of Bᵀ (K x N) on‑the‑fly
        b_offs = (
            b_batch
            + cur_k[:, None] * stride_b_col   # K dimension
            + offs_n[None, :] * stride_b_row   # N dimension
        )
        b = tl.load(
            b_offs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        acc += tl.dot(a, b)  # accumulate in fp32

    # Store the result tile
    c_offs = (
        c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(
        c_offs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )

        right_mat,

        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,

        BLOCK_K=BLOCK_K,

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B, N, N, hidden)

        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ----- Final linear projection back to dim -----
    out = F.linear(out, weights["to_out.weight"])
