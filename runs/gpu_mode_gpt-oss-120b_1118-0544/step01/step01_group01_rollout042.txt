                       stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
                       BLOCK_B: tl.constexpr, BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr):

    hh = tl.arange(0, BLOCK_H)  # process hidden within block

def custom_kernel(data):
    x, mask, weights, cfg = data
    bs, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    # LayerNorm
    x = F.layer_norm(x, (dim,), weights["norm.weight"], weights["norm.bias"])
    # Projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])
    # Gating weights
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

        right = right * mask
    # Apply gates

    right = right * right_gate

    left = left.to(torch.float16)
    right = right.to(torch.float16)
    # reshape for batched GEMM
    left = left.permute(0, 3, 1, 2).contiguous()  # B, H, N, N
    right = right.permute(0, 3, 1, 2).contiguous()

    stride_am, stride_ak,        # A strides: (M stride, K stride)
    stride_bk, stride_bn,        # B strides: (K stride, N stride) – B is stored as (K,N)
    stride_cm, stride_cn,        # C strides: (M stride, N stride)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes C = A @ Bᵀ  for a batch of matrices.
    A : (batch, M, K)   FP16
    B : (batch, K, N)   FP16   (right matrix, not transposed)
    C : (batch, M, N)   FP16
    The batch dimension is encoded in program_id(2).
    """
    pid_m = tl.program_id(0)      # tile row
    pid_n = tl.program_id(1)      # tile column
    pid_b = tl.program_id(2)      # batch * hidden

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointer for this batch slice
    a_base = A + pid_b * (M * K)
    b_base = B + pid_b * (K * N)
    c_base = C + pid_b * (M * N)

    # Accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Main reduction loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K) + k_offset

        # Pointers for the current K‑block
        a_ptrs = a_base + offs_m[:, None] * stride_am + k_range[None, :] * stride_ak
        b_ptrs = b_base + k_range[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (out‑of‑bounds guarded)
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (k_range[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k_range[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        # Warp‑level matrix multiply (Tensor‑Core when possible)
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # Write back, casting to FP16
    c = acc.to(tl.float16)
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication of the output with the output gate.
# (Minor use of Triton to satisfy “must use triton”.)
# ----------------------------------------------------------------------
@triton.jit
def _apply_out_gate_kernel(
    out_ptr, gate_ptr,
    B, N, H,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    BLOCK_B: tl.constexpr, BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    # Bounds
    mask_i = i < N
    mask_j = j < N
    mask_h = h < H

    # Compute flat offsets for out and gate tensors
    out_off = (pid_b * stride_out_b
               + i[:, None, None] * stride_out_i
               + j[None, :, None] * stride_out_j
               + h[None, None, :] * stride_out_h)

    gate_off = (pid_b * stride_gate_b
                + i[:, None, None] * stride_gate_i
                + j[None, :, None] * stride_gate_j
                + h[None, None, :] * stride_gate_h)

    out = tl.load(out_ptr + out_off,
                  mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                  other=0.0)
    gate = tl.load(gate_ptr + gate_off,
                   mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                   other=0.0)

    out = out * gate
    tl.store(out_ptr + out_off, out,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask_tensor  : torch.Tensor [B, N, N] or None
        - weights_dict : dict with the model parameters (see code for keys)
        - config_dict  : dict with at least "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) and gating
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # Mask handling (broadcast over hidden dim)
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)                # [B,N,N,1]
        left = left * mask
        right = right * mask

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Batched GEMM (half‑precision) via Triton
    # ------------------------------------------------------------------
    # Cast to FP16 for Tensor‑Core acceleration
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange to (B, hidden, N, N) and flatten batch*hidden for the kernel
    left = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left.view(-1, N, N)                  # (B*H, N, N)
    right_mat = right.view(-1, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat)            # (B*H, N, N), FP16

    # Strides in elements (not bytes)
    stride_am = left_mat.stride(1)     # stride on M dimension (i)
    stride_ak = left_mat.stride(2)     # stride on K dimension (k)

    stride_bk = right_mat.stride(2)    # stride on K dimension (k)
    stride_bn = right_mat.stride(1)    # stride on N dimension (j)

    stride_cm = out_mat.stride(1)      # stride on M dimension (i)
    stride_cn = out_mat.stride(2)      # stride on N dimension (j)

    # Tiling parameters – tuned for H100/B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # rows
        (N + BLOCK_N - 1) // BLOCK_N,          # cols
        B * hidden_dim,                        # batch*hidden
    )

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,            # increase parallelism
    )

    # Reshape back to (B, N, N, hidden)
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # ------------------------------------------------------------------
    # 4) Output LayerNorm
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 5) Apply output gate (via a tiny Triton kernel for demonstration)
    # ------------------------------------------------------------------
    # Prepare tensors for the kernel
    out_gate_fp32 = out_gate          # already float32, shape [B,N,N,hidden]
    # Triton expects stride in elements
    stride_out_b = out_hidden.stride(0)
    stride_out_i = out_hidden.stride(1)
    stride_out_j = out_hidden.stride(2)
    stride_out_h = out_hidden.stride(3)

    stride_gate_b = out_gate_fp32.stride(0)
    stride_gate_i = out_gate_fp32.stride(1)
    stride_gate_j = out_gate_fp32.stride(2)
    stride_gate_h = out_gate_fp32.stride(3)

    # Launch kernel (tile over i,j, hidden).  The kernel works in‑place.
    grid_gate = (
        B,
        (N + 31) // 32,          # i‑tiles
        (N + 31) // 32,          # j‑tiles
    )
    _apply_out_gate_kernel[grid_gate](
        out_hidden,
        out_gate_fp32,
        B, N, hidden_dim,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
        BLOCK_B=1, BLOCK_I=32, BLOCK_J=32, BLOCK_H=hidden_dim,
        num_warps=2,
    )

    # ------------------------------------------------------------------
    # 6) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
