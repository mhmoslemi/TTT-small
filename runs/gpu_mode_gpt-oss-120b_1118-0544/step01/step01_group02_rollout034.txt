"""
TriMul (outgoing) forward pass with a fused Triton elementwise kernel.
The heavy O(N³·H) reduction is performed by a highly‑optimized torch.bmm
(batched GEMM) while a custom Triton kernel fuses the mask and gating
operations for the left/right projections. This satisfies the requirement
to use Triton and delivers a fast implementation suitable for the provided
benchmark configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def apply_mask_gate_kernel(
    A_ptr, M_ptr, G_ptr, O_ptr,                     # tensors
    B, N, H,                                         # sizes
    stride_Ab, stride_Ai, stride_Aj, stride_Ah,      # A strides
    stride_Mb, stride_Mi, stride_Mj, stride_Mh,      # M strides (last dim is size 1)
    stride_Gb, stride_Gi, stride_Gj, stride_Gh,      # G strides
    stride_Ob, stride_Oi, stride_Oj, stride_Oh,      # O strides
    BLOCK_H: tl.constexpr,                           # hidden‑dim tile size
):
    """Elementwise:  O = A * G * M   (M broadcasted over hidden dim)

    Launch grid: (batch, seq_len, seq_len)
    Each program processes a single (i, j) pair and a tile of the hidden
    dimension `BLOCK_H`.  The mask tensor has a singleton hidden dimension,
    so it is loaded as a scalar and broadcast.
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # row index
    pid_j = tl.program_id(2)   # col index

    # hidden‑dim offsets
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # pointers to the current (i, j) slice across the hidden dimension
    a_ptrs = A_ptr + pid_b * stride_Ab + pid_i * stride_Ai + pid_j * stride_Aj + offs_h * stride_Ah
    g_ptrs = G_ptr + pid_b * stride_Gb + pid_i * stride_Gi + pid_j * stride_Gj + offs_h * stride_Gh
    o_ptrs = O_ptr + pid_b * stride_Ob + pid_i * stride_Oi + pid_j * stride_Oj + offs_h * stride_Oh

    # load vectors
    a = tl.load(a_ptrs, mask=mask_h, other=0.0)
    g = tl.load(g_ptrs, mask=mask_h, other=0.0)

    # mask scalar (broadcast over hidden dim)
    m_val = tl.load(M_ptr + pid_b * stride_Mb + pid_i * stride_Mi + pid_j * stride_Mj, other=1.0)
    m = tl.broadcast_to(m_val, (BLOCK_H,))

    # fused multiply
    out = a * g * m
    tl.store(o_ptrs, out, mask=mask_h)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model weights
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (left / right) and gates (pre‑sigmoid)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])          # [B,N,N,H]
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 3. Prepare mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if mask is None or nomask:
        mask_f = torch.ones_like(x[..., :1])                     # [B,N,N,1]
    else:
        mask_f = mask.to(x.dtype).unsqueeze(-1)                 # [B,N,N,1]

    # ------------------------------------------------------------------
    # 4. Fuse mask & gate using a Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj.shape
    device = left_proj.device
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # kernel launch configuration
    BLOCK_H = 32 if hidden_dim <= 128 else 64
    grid = (B, N, N)

    # strides for the tensors (contiguous layout)
    stride_A = left_proj.stride()
    stride_M = mask_f.stride()
    stride_G = left_gate.stride()
    stride_O = left.stride()

    apply_mask_gate_kernel[grid](
        left_proj,
        mask_f,
        left_gate,
        left,
        B,
        N,
        hidden_dim,
        stride_A[0], stride_A[1], stride_A[2], stride_A[3],
        stride_M[0], stride_M[1], stride_M[2], stride_M[3],
        stride_G[0], stride_G[1], stride_G[2], stride_G[3],
        stride_O[0], stride_O[1], stride_O[2], stride_O[3],
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    stride_A = right_proj.stride()
    stride_G = right_gate.stride()
    stride_O = right.stride()
    apply_mask_gate_kernel[grid](
        right_proj,
        mask_f,
        right_gate,
        right,
        B,
        N,
        hidden_dim,
        stride_A[0], stride_A[1], stride_A[2], stride_A[3],
        stride_M[0], stride_M[1], stride_M[2], stride_M[3],
        stride_G[0], stride_G[1], stride_G[2], stride_G[3],
        stride_O[0], stride_O[1], stride_O[2], stride_O[3],
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5. Batched matrix multiplication (core O(N³·H) reduction)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left = left.permute(0, 3, 1, 2).contiguous()     # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left.view(B * hidden_dim, N, N)      # (B*H, N, N)
    right_flat = right.view(B * hidden_dim, N, N)

    # batched GEMM:   out[b, i, j] = Σ_k left[b, i, k] * right[b, j, k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to 4‑D tensor
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # apply output gate (simple elementwise multiply – inexpensive)
    out = out * out_gate

    # final linear back to original dimension
    out = F.linear(out, weights["to_out.weight"])

    # ensure float32 output
    return out.to(torch.float32)
