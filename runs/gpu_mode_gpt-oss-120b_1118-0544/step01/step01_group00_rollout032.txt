"""
TriMul outgoing update (forward only).

The heavy reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication C = A @ Bᵀ where each batch
corresponds to one hidden‑dim slice.  We perform this reduction with
torch.bmm (FP16 → TensorCores) for speed, while a small Triton kernel
is used to apply the optional pairwise mask (broadcast across the
hidden dimension).  All other linear / layer‑norm / gating steps are
implemented with PyTorch functional ops.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: elementwise multiplication with a broadcast mask.
# src : [B, N, N, H]   (contiguous)
# mask: [B, N, N]      (broadcast over H)
# -------------------------------------------------------------------------
@triton.jit
def mask_apply_kernel(
    src_ptr,          # pointer to src (float32)
    mask_ptr,         # pointer to mask (float32)
    dst_ptr,          # pointer to dst (float32)
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)                     # (BLOCK,)

    total = B * N * N * H
    mask = offs < total

    # index of the un‑broadcasted mask element (flattened)
    mask_off = offs // H

    a = tl.load(src_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=mask, other=0.0)

    tl.store(dst_ptr + offs, a * m, mask=mask)


def _apply_mask_triton(src: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Apply ``src = src * mask`` where mask is broadcast over the hidden dim.
    """
    B, N, _, H = src.shape
    total = B * N * N * H
    BLOCK = 1024  # threads per program
    grid = ((total + BLOCK - 1) // BLOCK,)

    dst = torch.empty_like(src)
    # ensure contiguous inputs for pointer arithmetic
    src_c = src.contiguous()
    mask_c = mask.contiguous()

    mask_apply_kernel[grid](
        src_c,
        mask_c,
        dst,
        B,
        N,
        H,
        BLOCK=BLOCK,
    )
    return dst


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32 / bool) – may be ignored
            - weights: dict with required weight tensors
            - config: dict with keys:
                * "dim" (int)
                * "hidden_dim" (int)
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # layer‑norm on the input (over last dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        # mask may be bool; convert to float32 and keep on the same device
        mask_f = mask.to(x.dtype).contiguous()
        left = _apply_mask_triton(left, mask_f)   # left * mask
        right = _apply_mask_triton(right, mask_f) # right * mask

    # -----------------------------------------------------------------
    # gating (sigmoid of linear projections of x)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # reduction: batched matmul C = A @ Bᵀ  (FP16, Tensor Cores)
    # -----------------------------------------------------------------
    Bsz, N, _, H = left.shape
    # cast to half for the GEMM
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B*H, N, N)   -> each hidden slice is an independent matrix
    left_mat = left_h.permute(0, 3, 1, 2).contiguous().view(Bsz * H, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(Bsz * H, N, N)

    # batched GEMM: (B*H, N, N) @ (B*H, N, N).T  -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden]
    out = out_mat.view(Bsz, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # post‑processing: layer‑norm, out‑gate, final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
