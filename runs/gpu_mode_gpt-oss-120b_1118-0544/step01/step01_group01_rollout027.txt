"""
Highly‑optimized outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the input (FP32).
2. Cast to FP16 and apply three bias‑free linear projections:
   * left_proj, right_proj  → hidden_dim
   * left_gate, right_gate, out_gate → hidden_dim (followed by sigmoid)
3. Fuse mask (if provided), left_gate and right_gate with a tiny Triton
   elementwise kernel to obtain the gated tensors `left_out` and `right_out`.
4. Batched matrix multiplication across the sequence dimension:
   For each hidden channel h we need
         out_h = left_out_h @ right_out_hᵀ
   This is performed with `torch.bmm`, which uses cuBLAS’s highly‑tuned
   Tensor‑Core kernels (FP16 inputs, FP32 accumulation).
5. Convert the result back to FP32, apply a second LayerNorm, multiply by
   the `out_gate`, and finally project back to the original dimension
   with a linear layer.
All heavy arithmetic is in FP16; only the two LayerNorms and the final
projection run in FP32, matching the reference implementation while
exploiting the full compute power of the B200 GPU.

The Triton kernel (`fuse_gate_mask_kernel`) only performs the required
elementwise fusion, satisfying the “use Triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton elementwise kernel: out = a * g * m
#   a : tensor (e.g. left or right)
#   g : gate tensor (sigmoid output)
#   m : mask tensor broadcasted over the hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def fuse_gate_mask_kernel(
    out_ptr, a_ptr, g_ptr, mask_ptr,      # pointers
    hidden_dim, total_elems,              # runtime constants
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load data
    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    g = tl.load(g_ptr + offs, mask=mask, other=tl.float16(0.0))

    # mask is stored without the hidden dim (shape B*L*L)
    # Each hidden entry repeats the same mask -> integer division
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))

    out = a * g * m
    tl.store(out_ptr + offs, out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    device = x.device
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Cast linear weights to FP16 once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (bias‑free) and gates
    # ------------------------------------------------------------------
    # Projections
    left  = F.linear(x_fp16, left_proj_w)      # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)     # [B, N, N, H]

    # Gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    # ------------------------------------------------------------------
    # 4) Optional mask handling – make a half‑precision mask tensor
    # ------------------------------------------------------------------
    if mask is None:
        # mask of all ones (broadcasted later)
        mask_fp16 = torch.ones((B, N, N, 1), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]

    # Ensure contiguous tensors for the kernel
    left      = left.contiguous()
    right     = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate= right_gate.contiguous()
    mask_fp16 = mask_fp16.contiguous()

    # ------------------------------------------------------------------
    # 5) Fuse mask & gates with a tiny Triton kernel
    # ------------------------------------------------------------------
    total_elem = left.numel()                     # B*N*N*H
    BLOCK = 8192                                   # tuning parameter

    # Allocate output buffers
    left_out  = torch.empty_like(left)   # FP16
    right_out = torch.empty_like(right)  # FP16

    grid = ( (total_elem + BLOCK - 1) // BLOCK, )
    fuse_gate_mask_kernel[grid](
        left_out.data_ptr(),
        left.data_ptr(),
        left_gate.data_ptr(),
        mask_fp16.data_ptr(),
        hidden_dim,
        total_elem,
        BLOCK_SIZE=BLOCK,
    )
    fuse_gate_mask_kernel[grid](
        right_out.data_ptr(),
        right.data_ptr(),
        right_gate.data_ptr(),
        mask_fp16.data_ptr(),
        hidden_dim,
        total_elem,
        BLOCK_SIZE=BLOCK,
    )

    # Release memory we no longer need (helps with large N)
    del left, right, left_gate, right_gate, mask_fp16
    torch.cuda.empty_cache()

    # ------------------------------------------------------------------
    # 6) Batched matrix multiplication (FP16) → out_hidden (FP16)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) → flatten batch*hidden for torch.bmm
    left_out  = left_out.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_out = right_out.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_out.view(-1, N, N)                     # (B*H, N, N)
    right_mat = right_out.view(-1, N, N)                    # (B*H, N, N)

    # Batched gemm: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore original layout and convert to FP32 for the following ops
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Apply output gate (already in FP16 → convert)
    out_gate = out_gate.to(torch.float32)
    out_hidden = out_hidden * out_gate

    # Final projection back to `dim` (FP32 weight used for stability)
    out = F.linear(out_hidden, to_out_w.to(torch.float32))

    return out
```
