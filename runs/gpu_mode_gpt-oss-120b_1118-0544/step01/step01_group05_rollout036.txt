    x, mask, weights, config = data

    dim = config["dim"]

    nomask = config.get("nomask", False)

    # LayerNorm
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # Projections
    left = F.linear(x_norm, weights["left_proj.weight"])

We'll define kernel:

```
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)               # batch index
    pid_m = tl.program_id(1)             
    pid_n = tl.program_id(2)             

    # Compute offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for valid rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch slice
    a_batch = a_ptr + pid * stride_a_batch
    b_batch = b_ptr + pid * stride_b_batch
    c_batch = c_ptr + pid * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block (M x K)
        a = tl.load(
            a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B block transposed (K x N) i.e., B^T
        b = tl.load(
            b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # Store result
    c = acc
    c_ptrs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

Make sure the stride naming matches our earlier. For B, stride_b_col is column stride (k dimension) and stride_b_row is row stride (n dimension). Since we use B transposed we loaded as cur_k (k) * stride_b_col + offs_n * stride_b_row. That's correct.

Now we need to compute stride arguments:

- For A: `stride_a_batch = left_mat.stride(0)` (elements)
- `stride_a_row = left_mat.stride(1)`
- `stride_a_col = left_mat.stride(2)`

- For B: `stride_b_batch = right_mat.stride(0)`
- `stride_b_row = right_mat.stride(1)` (original row stride)

"""
TriMul (outgoing) implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Apply linear projections (left, right) and three gating linear layers.
3. Multiply left/right by their respective gates and optional boolean mask.
4. Cast the gated tensors to float16 and reshape them to
   (B*hidden_dim, N, N) → a batched matrix‑multiplication problem.
5. The core contraction
   out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
   is computed as a batched GEMM C = A @ Bᵀ using a Triton kernel.
   The kernel processes blocks of size 128×128×32 (M,N,K) with
   8 warps and 4 stages, using FP16 inputs and FP32 accumulation
   (tensor‑core friendly on B200 GPUs).
6. Reshape the result back to [B, N, N, hidden_dim].
7. Apply a second LayerNorm, an output gate, and a final linear projection
   back to the original dimension.

The heavy O(N³) contraction is therefore offloaded to an efficient
custom Triton GEMM while the surrounding linear and normalization steps
remain in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ  (A,B: fp16, C: fp32).

    A : (batch, M, K)  – row‑major
    B : (batch, N, K)  – row‑major (accessed transposed)
    C : (batch, M, N)
    """
    pid = tl.program_id(0)      # batch * hidden_dim index
    pid_m = tl.program_id(1)    # tile over M
    pid_n = tl.program_id(2)    # tile over N

    # Block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch slice
    a_batch = a_ptr + pid * stride_a_batch
    b_batch = b_ptr + pid * stride_b_batch
    c_batch = c_ptr + pid * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block (M x K)
        a = tl.load(
            a_batch
            + offs_m[:, None] * stride_a_row
            + cur_k[None, :] * stride_a_col,
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0),
            dtype=tl.float16,
        )

        # Load B block transposed (K x N)
        b = tl.load(
            b_batch
            + cur_k[:, None] * stride_b_col
            + offs_n[None, :] * stride_b_row,
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0),
            dtype=tl.float16,
        )

        # Rank‑K update
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = (
        c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (tensors on the same device)
            - config: dict containing at least "dim" and "hidden_dim".
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------
    # 1) Input LayerNorm
    # -------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -------------------------------------------------
    # 2) Linear projections
    # -------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3) Optional mask
    # -------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------
    # 4) Gating
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5) Cast to half‑precision for the contraction
    # -------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim (H == hidden_dim)

    # Reshape to (B*H, N, N) – each hidden channel becomes an independent matrix
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (FP32 accumulation)
    out_mat = torch.empty((B * H, N, N), device=device, dtype=torch.float32)

    # -------------------------------------------------
    # 6) Triton batched GEMM: C = A @ Bᵀ
    # -------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Strides in number of elements
    stride_a_batch = left_mat.stride(0)
    stride_a_row   = left_mat.stride(1)
    stride_a_col   = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_row   = right_mat.stride(1)   # original row stride (N)
    stride_b_col   = right_mat.stride(2)   # original column stride (1)

    stride_c_batch = out_mat.stride(0)
    stride_c_row   = out_mat.stride(1)
    stride_c_col   = out_mat.stride(2)

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # -------------------------------------------------
    # 7) Reshape back to [B, N, N, hidden_dim]
    # -------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # (B, N, N, hidden_dim)

    # -------------------------------------------------
    # 8) Second LayerNorm over hidden dimension
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # -------------------------------------------------
    # 9) Output gate
    # -------------------------------------------------
    out = out * out_gate

    # -------------------------------------------------
    # 10) Final linear projection back to `dim`
    # -------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
