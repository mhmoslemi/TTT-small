@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0) # batch index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        a_ptrs = a_ptr + pid * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        # load
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)

        acc += tl.dot(a, b)
    # Store result
    c_ptrs = c_ptr + pid * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    batch_stride_a, stride_i_a, stride_k_a,
    batch_stride_b, stride_k_b, stride_n_b,
    batch_stride_c, stride_i_c, stride_n_c,
    hidden_dim: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)  # batch_head index
    pid_m = tl.program_id(1)  # row tile
    pid_n = tl.program_id(2)  # col tile

    # compute batch and head index
    batch_idx = pid0 // hidden_dim
    head_idx = pid0 - batch_idx * hidden_dim

    # base pointer offset for this batch/head
    a_base = a_ptr + batch_idx * batch_stride_a + head_idx
    b_base = b_ptr + batch_idx * batch_stride_b + head_idx
    c_base = c_ptr + batch_idx * batch_stride_c + head_idx

    # offsets for this tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out of bounds rows/cols
    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # mask for k dimension
        mask_k = offs_k < K

        # load A tile: rows i, columns k
        a_ptrs = a_base + offs_i[:, None] * stride_i_a + offs_k[None, :] * stride_k_a
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B tile: rows k, cols j
        b_ptrs = b_base + offs_k[:, None] * stride_k_b + offs_j[None, :] * stride_n_b
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product (M x K) * (K x N) -> (M x N)
        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_base + offs_i[:, None] * stride_i_c + offs_j[None, :] * stride_n_c
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    hidden_dim = config['hidden_dim']
    eps = 1e-5

    # Layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=eps)

    # Projections
    left = F.linear(x, weights['left_proj.weight'])
    right = F.linear(x, weights['right_proj.weight'])

    # mask
    if mask is not None:
        mask = mask.unsqueeze(-1).type_as(x)
        left = left * mask
        right = right * mask

    # Gating
    left_gate = torch.sigmoid(F.linear(x, weights['left_gate.weight']))
    right_gate = torch.sigmoid(F.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(F.linear(x, weights['out_gate.weight']))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # shape [B,N,N,H]
    # Output tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_h.device)

    # Strides (in elements)
    stride_a_batch = left_h.stride(0)
    stride_i_a = left_h.stride(1)
    stride_k_a = left_h.stride(2)
    stride_batch_a = stride_a_batch  # for whole batch offset

    stride_b_batch = right_h.stride(0)
    stride_k_b = right_h.stride(2)
    stride_j_b = right_h.stride(1)

    stride_c_batch = out.stride(0)
    stride_i_c = out.stride(1)
    stride_j_c = out.stride(2)

    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_outgoing_kernel[grid](
        left_h,
        right_h,
        out,
        N, N, N,
        stride_a_batch, stride_i_a, stride_k_a,
        stride_b_batch, stride_k_b, stride_j_b,
        stride_c_batch, stride_i_c, stride_j_c,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Output norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'], eps=eps)

    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, weights['to_out.weight'])
    return out

"""
TriMul (outgoing) forward pass – Triton implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input over its last dimension (dim).
2. Linear projections (left / right) from dim → hidden_dim.
3. Optional mask (broadcast on hidden_dim) is applied to the projected tensors.
4. Compute three sigmoid gates (left, right, out) and fuse the first two with the
   projected tensors.
5. Core O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batch of GEMMs over the hidden dimension.  A custom Triton kernel
   computes the batched matrix multiplications directly on the original memory
   layout, avoiding costly permutes or copies.
6. Layer‑norm the result, multiply by the output gate and apply a final linear
   projection back to the original channel dimension.

The Triton kernel tiles the M (i), N (j) and K (k) axes (default block size
64) and accumulates in FP32 while the inputs are FP16.  The batch and head
dimensions are fused into a single program‑id and split inside the kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ah, stride_ai, stride_ak,      # A: batch, i, k
    stride_bh, stride_bk, stride_bn,      # B: batch, k, j (j is the second N axis)
    stride_ch, stride_ci, stride_cj,      # C: batch, i, j
    hidden_dim,                           # runtime hidden‑dim (H)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: out = left @ rightᵀ  (per hidden head)."""
    pid0 = tl.program_id(0)   # combined batch*head axis
    pid_m = tl.program_id(1)  # block row (i)
    pid_n = tl.program_id(2)  # block column (j)

    # ---- split combined axis into (batch, head) ----
    batch_idx = pid0 // hidden_dim
    head_idx = pid0 - batch_idx * hidden_dim

    # base pointer for the current (batch, head)
    a_base = a_ptr + batch_idx * stride_ah + head_idx
    b_base = b_ptr + batch_idx * stride_bh + head_idx
    c_base = c_ptr + batch_idx * stride_ch + head_idx

    # ---- tile offsets ----
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- reduction over K dimension ----
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: rows = i, cols = k
        a_ptrs = a_base + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile: rows = k, cols = j
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # FP16 matmul, acc in FP32
        acc += tl.dot(a, b)

    # ---- write back ----
    c_ptrs = c_base + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of torch.Parameters (see keys in the stub)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------
    # 1. Input LayerNorm
    # -------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # -------------------------------------------------
    # 2. Linear projections (bias‑less)
    # -------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # -------------------------------------------------
    # 3. Optional mask
    # -------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).type_as(x)         # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -------------------------------------------------
    # 4. Gating (sigmoid)
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5. Core contraction via Triton
    # -------------------------------------------------
    # Cast to half for speed
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # left_h shape = [B, N, N, hidden_dim]

    # Output buffer (FP32 accumulator)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=x.device)

    # Strides (in elements)
    stride_ah = left_h.stride(0)   # batch stride
    stride_ai = left_h.stride(1)   # i (row) stride
    stride_ak = left_h.stride(2)   # k (col) stride

    stride_bh = right_h.stride(0)  # batch stride (same as A)
    stride_bk = right_h.stride(2)  # k (row) stride
    stride_bn = right_h.stride(1)  # j (col) stride

    stride_ch = out.stride(0)      # batch stride
    stride_ci = out.stride(1)      # i (row) stride
    stride_cj = out.stride(2)      # j (col) stride

    # Tuning parameters – 64 works well across the given sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                       # combined batch * head axis
        (N + BLOCK_M - 1) // BLOCK_M,         # tile rows (i)
        (N + BLOCK_N - 1) // BLOCK_N,         # tile cols (j)
    )

    _tri_mul_outgoing_kernel[grid](
        left_h,
        right_h,
        out,
        N, N, N,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_ci, stride_cj,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------
    # 6. Output normalization, gating and final projection
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
