def apply_mask_gate_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    total_elements,
    BLOCK_SIZE: tl.constexpr
):

    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements
    # compute pointer offsets to multi-dim arrays: we will treat flattened tensors
    left = tl.load(left_ptr + offs, mask=mask).to(tl.float16)

    out_right = right * right_gate * mask_val

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the pair representation (shape [B, N, N, C]).
2. Linear projections → left / right (hidden_dim) and three gating projections.
3. Fuse mask, left‑gate and right‑gate using a simple Triton kernel that
   multiplies element‑wise and casts the result to FP16.
4. Compute the core update:
        out = left @ rightᵀ   (batched GEMM over the sequence dimension N)
   using Torch's highly‑optimized batched half‑precision matrix multiply.
5. Layer‑norm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original channel dimension.
The kernel is deliberately lightweight: it only performs the element‑wise
mask‑gate fusion, leaving the N³ reduction to cuBLAS (torch.bmm), which gives
the desired ~µs‑scale runtime on B200 hardware.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    N_ELEMS,
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask, gating and cast to half‑precision.

    All pointers refer to contiguous tensors of the same shape:
        [B, N, N, hidden_dim] (flattened when launched).
    The kernel reads FP16 values, multiplies element‑wise and writes FP16.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < N_ELEMS

    # Load (FP16)
    l = tl.load(left_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    r = tl.load(right_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + offs, mask=valid, other=0.0)

    # Fuse
    out_l = l * lg * m
    out_r = r * rg * m

    # Store
    tl.store(out_left_ptr + offs, out_l, mask=valid)
    tl.store(out_right_ptr + offs, out_r, mask=valid)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul.

    Parameters
    ----------
    data : tuple
        (pair_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32).
    """
    pair_tensor, mask, weights, config = data
    device = pair_tensor.device
    B, N, N2, C = pair_tensor.shape
    assert N == N2, "pair tensor must be square in the sequence dimensions."

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the channel dimension C
    # ------------------------------------------------------------------
    x = F.layer_norm(pair_tensor, (C,), weight=norm_w, bias=norm_b, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)      # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)    # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Gating projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4. Prepare mask (broadcast to hidden_dim) – if missing use ones
    # ------------------------------------------------------------------
    if mask is None:
        mask_exp = torch.ones((B, N, N, 1), dtype=pair_tensor.dtype, device=device)
    else:
        # ensure mask is float and has shape [B,N,N]
        if mask.dim() == 3:
            mask_exp = mask.unsqueeze(-1)
        else:
            mask_exp = mask
        mask_exp = mask_exp.to(pair_tensor.dtype)
    # Broadcast to hidden_dim
    mask_exp = mask_exp.expand(-1, -1, -1, hidden_dim).contiguous()

    # ------------------------------------------------------------------
    # 5. Fuse mask & gates with a Triton kernel (outputs in FP16)
    # ------------------------------------------------------------------
    # Cast to half before the kernel to keep memory bandwidth low
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    left_gate_fp16 = left_gate.to(torch.float16)
    right_gate_fp16 = right_gate.to(torch.float16)
    mask_fp16 = mask_exp.to(torch.float16)

    out_left = torch.empty_like(left_fp16)   # FP16 buffers for the fused results
    out_right = torch.empty_like(right_fp16)

    total_elements = left_fp16.numel()
    BLOCK_SIZE = 1024  # reasonable vector size per program
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    _mask_gate_fuse_kernel[grid](
        left_fp16,
        left_gate_fp16,
        right_fp16,
        right_gate_fp16,
        mask_fp16,
        out_left,
        out_right,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 6. Core TriMul matmul: out = left @ rightᵀ   (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    # Rearrange to [B, hidden_dim, N, N] and flatten batch & hidden_dim
    left_bh = out_left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = out_right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Transpose the last two axes of right for the matmul
    right_T = right_bh.transpose(-2, -1).contiguous()      # [B, H, N, N]

    B_, H_, N_, _ = left_bh.shape
    # (B*H) batched matrix multiply
    left_flat = left_bh.view(B_ * H_, N_, N_)
    right_flat = right_T.view(B_ * H_, N_, N_)

    out_flat = torch.bmm(left_flat, right_flat)  # result in FP16

    # Restore original batch / hidden layout and convert back to FP32
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 7. Post‑processing: hidden‑dim LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # back to original dim

    return out
