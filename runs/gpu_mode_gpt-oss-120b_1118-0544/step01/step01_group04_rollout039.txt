"""
Trident implementation of the “outgoing” TriMul operator.

The heavy O(N³·C) contraction
    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
is performed with a highly‑optimized batched GEMM (torch.bmm) on
FP16 tensors.  Element‑wise gating and optional masking are fused
into tiny Triton kernels that operate on the flattened tensors.
All other linear layers and LayerNorms are carried out with PyTorch
so that we keep full‑precision semantics while still meeting the
sub‑millisecond runtime target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Tiny elementwise kernels (required by the problem statement)
# ----------------------------------------------------------------------
@triton.jit
def mul2_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr):
    """out = a * b  (all tensors are 1‑D flattened FP16)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    out = a * b
    tl.store(out_ptr + offs, out, mask=mask)


@triton.jit
def mul3_kernel(a_ptr, b_ptr, c_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr):
    """out = a * b * c  (all tensors are 1‑D flattened FP16)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim] (torch.float32)
        - mask         : [B, N, N] (torch.float32 or torch.bool) or None
        - weights      : dict of torch tensors (see code)
        - config       : dict containing at least "dim" and "hidden_dim".
    Returns
    -------
    out : torch.Tensor
        [B, N, N, dim] (torch.float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2. Linear projections (still FP32, cast to FP16 afterwards)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"]).to(torch.float16)
    right_proj = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid, then FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"])).to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Fuse mask (if any) and gates with tiny Triton kernels
    # ------------------------------------------------------------------
    # allocate output tensors for gated values
    left = torch.empty_like(left_proj, dtype=torch.float16, device=device)
    right = torch.empty_like(right_proj, dtype=torch.float16, device=device)

    # flatten tensors for the kernels (contiguous 1‑D view)
    left_proj_f = left_proj.contiguous().view(-1)
    left_gate_f = left_gate.contiguous().view(-1)
    right_proj_f = right_proj.contiguous().view(-1)
    right_gate_f = right_gate.contiguous().view(-1)
    left_f = left.view(-1)
    right_f = right.view(-1)

    total = left_proj_f.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    if (not nomask) and (mask is not None):
        # mask is broadcast over the channel dimension
        mask_f = mask.unsqueeze(-1).to(torch.float16).contiguous().view(-1)
        # left = left_proj * left_gate * mask
        mul3_kernel[grid](
            left_proj_f,
            left_gate_f,
            mask_f,
            left_f,
            total,
            BLOCK_SIZE=BLOCK,
        )
        # right = right_proj * right_gate * mask
        mul3_kernel[grid](
            right_proj_f,
            right_gate_f,
            mask_f,
            right_f,
            total,
            BLOCK_SIZE=BLOCK,
        )
    else:
        # no mask – simple elementwise product
        mul2_kernel[grid](
            left_proj_f,
            left_gate_f,
            left_f,
            total,
            BLOCK_SIZE=BLOCK,
        )
        mul2_kernel[grid](
            right_proj_f,
            right_gate_f,
            right_f,
            total,
            BLOCK_SIZE=BLOCK,
        )

    # ------------------------------------------------------------------
    # 5. Batched GEMM: out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
    #    Implemented with PyTorch's highly‑optimized batched GEMM (FP16)
    # ------------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N) and transpose right on the K dim
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication (FP16 inputs, FP32 accumulation internally)
    out_mat = torch.bmm(left_mat, right_mat)  # [B*hidden_dim, N, N] (FP16)

    # reshape back to [B, N, N, hidden_dim] and cast to FP32 for the rest
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 7. Apply output gate (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    out_gate_f32 = out_gate.to(torch.float32)
    out = out * out_gate_f32

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dimension (FP32)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
