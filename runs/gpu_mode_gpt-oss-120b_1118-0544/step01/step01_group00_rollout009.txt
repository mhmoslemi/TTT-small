"""
Outgoing Triangle Multiplicative Update (TriMul) kernel.

This implementation follows the PyTorch reference but off‑loads the
O(N³) reduction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
to a custom Triton kernel that performs a batched matrix multiplication
C = A @ Bᵀ for each (batch * hidden) slice.  The heavy projection,
gating and final layer‑norm / output projection remain in PyTorch.
Mixed‑precision (FP16 storage, FP32 accumulation) is used to reduce
memory traffic while preserving numerical quality.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ   (A: M×K, B: N×K)
# -------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left tensor   (batch, M, K)  – FP16
    b_ptr,               # right tensor  (batch, N, K)  – FP16 (will be read as K×N)
    c_ptr,               # output tensor (batch, M, N)   – FP16
    batch: tl.constexpr, # total # of independent matrices = B * hidden_dim
    M: tl.constexpr,     # rows of A   (= N)
    N: tl.constexpr,     # rows of B   (= N)
    K: tl.constexpr,     # reduction dimension (= N)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Program IDs
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile column
    pid_b = tl.program_id(2)          # which matrix in the batch

    # Offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the selected batch element
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K in BLOCK_K sized chunks
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (k_offset[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load Bᵀ tile: we want (BLOCK_K, BLOCK_N) = Bᵀ[:, :]
        b_ptrs = b_batch + (k_offset[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Dot product + accumulate (FP32)
        acc += tl.dot(a, b).to(tl.float32)

    # Write back the result
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# -------------------------------------------------------------
# Public entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool or float) – may be ignored
            - weights      : dict of model weights (see skeleton)
            - config       : dict containing at least
                * "dim" (int)
                * "hidden_dim" (int)
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Unpack weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # LayerNorm on the input (last dimension)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # Optional mask (broadcast on hidden_dim)
    # -----------------------------------------------------------------
    if not nomask:
        # mask may be bool; ensure same dtype and expand last dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # Gating (sigmoid of linear projects)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Cast to half for the O(N³) reduction
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, seq_len, seq_len, hidden_dim

    # Reshape to (B * hidden_dim, N, N) where hidden_dim is the innermost dim
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (half)
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # -----------------------------------------------------------------
    # Triton kernel launch configuration
    # -----------------------------------------------------------------
    # Tile sizes – tuned for typical N (256‑1024) on NVIDIA B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    batch_dim = B * hidden_dim

    grid = (num_pid_m, num_pid_n, batch_dim)

    # Strides expressed in element counts (not bytes)
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)

    stride_b_batch = right_h.stride(0)
    stride_b_k = right_h.stride(2)   # column stride of original (K)
    stride_b_n = right_h.stride(1)   # row stride (N)

    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    # Launch kernel
    tri_mul_outgoing_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        batch=batch_dim,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] and convert to FP32
    # -----------------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)                 # [B, N, N, hidden_dim]
        .contiguous()
        .to(torch.float32)
    )

    # -----------------------------------------------------------------
    # Final LayerNorm, gating and projection back to original dim
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # apply output gate
    out = F.linear(out, to_out_weight)       # [B, N, N, dim]

    return out
