    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"])   # shape [B,N,N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))

        right = right * mask_f[..., None]

    left = left * left_gate
    right = right * right_gate

    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Reshape for batched matmul
    B, N, _, _ = left_h.shape

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_mul_kernel(
    x_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_x_b, stride_x_i, stride_x_k, stride_x_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    total_elements,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Compute multi‑dimensional indices from linear offset
    h = offs % H
    tmp = offs // H
    k = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # Compute flat pointers
    x_off = b * stride_x_b + i * stride_x_i + k * stride_x_k + h * stride_x_h
    m_off = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k

    x = tl.load(x_ptr + x_off, mask=mask, other=0.0)
    m = tl.load(mask_ptr + m_off, mask=mask, other=0.0)
    out = x * m
    tl.store(out_ptr + x_off, out, mask=mask)

def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) – forward pass.
    Performs LayerNorm, linear projections, optional mask, gating,
    a batched matmul reduction (implemented with torch.bmm for speed),
    final LayerNorm, gating and projection back to input dimension.
    A small Triton kernel is used to apply the mask element‑wise.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm over last dimension
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # Optionally apply mask using Triton (in‑place)
    if not nomask:
        # Ensure mask is float32 for multiplication
        mask_f = mask.to(torch.float32)
        B, N, _, _ = left.shape
        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        # left = left * mask_f[..., None]
        mask_mul_kernel[grid](
            x_ptr=left,
            mask_ptr=mask_f,
            out_ptr=left,
            B=B,
            N=N,
            H=hidden_dim,
            stride_x_b=left.stride(0),
            stride_x_i=left.stride(1),
            stride_x_k=left.stride(2),
            stride_x_h=left.stride(3),
            stride_mask_b=mask_f.stride(0),
            stride_mask_i=mask_f.stride(1),
            stride_mask_k=mask_f.stride(2),
            total_elements=total,
            BLOCK=BLOCK,
        )
        # right = right * mask_f[..., None]
        total = right.numel()
        grid = (triton.cdiv(total, BLOCK),)
        mask_mul_kernel[grid](
            x_ptr=right,
            mask_ptr=mask_f,
            out_ptr=right,
            B=B,
            N=N,
            H=hidden_dim,
            stride_x_b=right.stride(0),
            stride_x_i=right.stride(1),
            stride_x_k=right.stride(2),
            stride_x_h=right.stride(3),
            stride_mask_b=mask_f.stride(0),
            stride_mask_i=mask_f.stride(1),
            stride_mask_k=mask_f.stride(2),
            total_elements=total,
            BLOCK=BLOCK,
        )

    # Gating (sigmoid of linear projections)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for the expensive reduction
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape

    # Reshape to (B*hidden_dim, N, N) for batched matmul
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication: C = A @ Bᵀ   (A: M×K, B: N×K)
    out_h = torch.bmm(left_h, right_h.transpose(1, 2))

    # Reshape back to [B, N, N, hidden_dim]
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Final LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply outgoing gate
    out = out * out_gate

    # Linear projection back to original dimension
    out = F.linear(out, weights["to_out.weight"])

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, total_elements, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul forward implementation.
    Steps:
      1. LayerNorm over the input channel dimension.
      2. Linear projections to hidden_dim (left/right) and gating projections.
      3. Optional mask multiplication (handled by PyTorch).
      4. Gating (element‑wise multiply).
      5. Cast to float16 and perform a batched matrix multiplication
         C = A @ Bᵀ using torch.bmm (highly optimized cuBLAS).
      6. Reshape back, LayerNorm over hidden_dim.
      7. Apply output gate using a tiny Triton kernel.
      8. Final linear projection back to the original dimensionality.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input layer‑norm
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # 2. Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # 3. Optional mask (broadcast on hidden_dim)
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # 4. Gating (sigmoid of linear projections)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # 5. Cast to half and batched matmul C = A @ Bᵀ
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape
    # reshape to (B*hidden_dim, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matmul (half precision)
    out_h = torch.bmm(left_h, right_h.transpose(1, 2))

    # 6. Restore shape [B, N, N, hidden_dim]
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # final LayerNorm over hidden dim
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # 7. Apply out_gate via Triton kernel (in‑place)
    out_gate = out_gate.contiguous()
    out = out.contiguous()
    total = out.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    elementwise_mul_kernel[grid](
        a_ptr=out,
        b_ptr=out_gate,
        out_ptr=out,
        total_elements=total,
        BLOCK=BLOCK,
    )

    # 8. Linear projection back to dim (no bias)
    out = F.linear(out, weights["to_out.weight"])
    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Tiny Triton kernel used to apply the output gate (element‑wise product).
# It works on a flat view of the tensor (contiguous memory) and therefore
# needs only the total number of elements.
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, total_elements, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) – forward pass.

    1. Layer‑norm over the last dimension of the 4‑D input.
    2. Linear projections to ``hidden_dim`` (left/right) and to three gates.
    3. Optional mask multiplication (broadcast on the hidden dimension) – done with PyTorch.
    4. Gating (sigmoid) and element‑wise multiplication.
    5. Cast to float16 and compute the heavy reduction
       ``out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]``
       as a batched matmul C = A @ Bᵀ using ``torch.bmm`` (highly tuned cuBLAS).
    6. Reshape back, LayerNorm over ``hidden_dim``.
    7. Apply the output gate with a tiny Triton kernel (in‑place).
    8. Final linear projection back to the original ``dim``.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched matmul reduction (FP16 for bandwidth)
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape
    # reshape to (B*hidden_dim, N, N) for torch.bmm
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # C = A @ Bᵀ, both A and B are (M=N, K=N)
    out_h = torch.bmm(left_h, right_h.transpose(1, 2))

    # ------------------------------------------------------------------
    # 6. Restore shape and final LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 7. Apply output gate via a tiny Triton kernel (in‑place)
    # ------------------------------------------------------------------
    out_gate = out_gate.contiguous()
    out = out.contiguous()
    total = out.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    elementwise_mul_kernel[grid](
        a_ptr=out,
        b_ptr=out_gate,
        out_ptr=out,
        total_elements=total,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 8. Project back to the original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])
    return out
