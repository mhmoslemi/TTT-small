```
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_ah, stride_ai, stride_aj, stride_ad,
    stride_bh, stride_bi, stride_bj, stride_bd,
    stride_ch, stride_ci, stride_cj, stride_cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Determine batch and hidden dim from pid
    hid = pid % H
    b = pid // H

    # Compute offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointer for this batch and hidden dim
    a_base = a_ptr + b * stride_ah + hid * stride_ad
    b_base = b_ptr + b * stride_bh + hid * stride_bd
    c_base = c_ptr + b * stride_ch + hid * stride_cd

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # mask for OOB
        a_mask = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        b_mask = (offs_k[:, None] < N) & (offs_n[None, :] < N)

        # Compute pointers
        # left: a_base + offs_m * stride_ai + offs_k * stride_aj
        a_ptrs = a_base + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_aj
        # right: b_base + offs_k * stride_bi + offs_n * stride_bj
        b_ptrs = b_base + offs_k[:, None] * stride_bi + offs_n[None, :] * stride_bj

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

    # Store result
    c_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    c_ptrs = c_base + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=c_mask)
```

"""
TriMul (outgoing) forward pass – Triton‐accelerated core contraction.

Algorithm
---------
1. Layer‑norm the input over the feature dimension.
2. Linear projections (left/right) and three gating linear layers.
3. Apply optional mask and gates to the projected tensors.
4. Core O(N³) contraction:
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   Implemented as a fused batched GEMM using a Triton kernel that
   works on tiles of size (BLOCK_M, BLOCK_N, BLOCK_K) and accumulates
   in FP32 while the inputs are FP16.  The batch and hidden dimensions
   are collapsed into a single program‑id dimension to avoid any data
   copies (no permute/contiguous needed).
5. Layer‑norm over the hidden dimension, apply the output gate and a
   final linear projection back to the original feature dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers to left, right, output
    B, N, H,                         # batch size, sequence length, hidden dim
    stride_ah, stride_ai, stride_aj, stride_ad,   # strides of left  (batch,i,k,dim)
    stride_bh, stride_bi, stride_bj, stride_bd,   # strides of right (batch,i,k,dim)
    stride_ch, stride_ci, stride_cj, stride_cd,   # strides of output (batch,i,j,dim)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the outgoing TriMul contraction.

    Computes for a given batch `b` and hidden index `h`:
        C[i,j] = Σₖ A[i,k] * B[j,k]   (note B is accessed with its first index = k)
    """
    pid0 = tl.program_id(0)   # collapsed (batch, hidden)
    pid_m = tl.program_id(1)  # block row (i)
    pid_n = tl.program_id(2)  # block col (j)

    # Decode batch and hidden indices
    hid = pid0 % H
    bid = pid0 // H

    # Offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this (batch, hidden) pair
    a_base = a_ptr + bid * stride_ah + hid * stride_ad
    b_base = b_ptr + bid * stride_bh + hid * stride_bd
    c_base = c_ptr + bid * stride_ch + hid * stride_cd

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        b_mask = (offs_k[:, None] < N) & (offs_n[None, :] < N)

        # Load tiles.  left: A[i,k]; right: B[k,j]
        a_ptrs = a_base + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_aj
        b_ptrs = b_base + offs_k[:, None] * stride_bi + offs_n[None, :] * stride_bj

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # fp16

        # Matrix‑multiply tile using Tensor Cores (fp16 * fp16 -> fp32)
        acc += tl.dot(a, b)

    # Write the result back (fp32)
    c_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    c_ptrs = c_base + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation using a Triton kernel for
    the O(N³) contraction.

    Args:
        data (Tuple[torch.Tensor, torch.Tensor|None, dict, dict]):
            - input_tensor:   [B, N, N, dim] (float32)
            - mask:           [B, N, N]          (float32) or None
            - weights: dict of model parameters (torch.Tensors)
            - config:  dict containing 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------------------
    # 1. Input LayerNorm
    # -------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # -------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # -------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden_dim)
    # -------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).type_as(x)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -------------------------------------------------------------
    # 4. Gating (sigmoid activations)
    # -------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # 5. Core contraction – Triton kernel
    # -------------------------------------------------------------
    B, N, _, _ = left.shape
    # Cast to half‑precision for the kernel (inputs) – accumulation stays FP32
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Output buffer (FP32 accumulator)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=x.device)

    # Tile sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides for the three tensors
    stride_ah = left_h.stride(0)
    stride_ai = left_h.stride(1)
    stride_aj = left_h.stride(2)
    stride_ad = left_h.stride(3)

    stride_bh = right_h.stride(0)
    stride_bi = right_h.stride(1)
    stride_bj = right_h.stride(2)
    stride_bd = right_h.stride(3)

    stride_ch = out.stride(0)
    stride_ci = out.stride(1)
    stride_cj = out.stride(2)
    stride_cd = out.stride(3)

    # Grid: (batch*hidden, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_outgoing_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        hidden_dim,
        stride_ah,
        stride_ai,
        stride_aj,
        stride_ad,
        stride_bh,
        stride_bi,
        stride_bj,
        stride_bd,
        stride_ch,
        stride_ci,
        stride_cj,
        stride_cd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # reasonable default for 64×64 tiles
    )

    # -------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # -------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
