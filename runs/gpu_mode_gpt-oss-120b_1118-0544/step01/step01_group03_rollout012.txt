"""
TriMul outgoing update (AlphaFold3) – Triton‑accelerated core multiplication.
The function performs:
1. LayerNorm over the channel dimension.
2. Linear projections (left/right) + gating & optional mask.
3. Batched matrix multiplication C = A @ Bᵀ where
   A = left[b, d, i, k] and B = right[b, d, j, k].
   This O(N³·hidden) step is executed by a custom Triton kernel.
4. LayerNorm over the hidden dimension, out‑gate scaling and final linear
   projection back to the original channel dimension.
All operations run on the provided CUDA device; only the N³ multiplication
is off‑loaded to Triton for speed.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _trigemul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix sizes (int64)
    stride_a_batch, stride_am, stride_ak,    # A strides: (batch, M, K)
    stride_b_batch, stride_bk, stride_bn,    # B strides: (batch, K, N)
    stride_c_batch, stride_cm, stride_cn,    # C strides: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul C = A @ B where A:[B, M, K] and B:[B, K, N]"""
    pid_batch = tl.program_id(2)   # batch (including hidden dim)
    pid_m = tl.program_id(0)       # row tile
    pid_n = tl.program_id(1)       # col tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # pointers to tiles
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)          # dot over the K dimension

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul update.

    Parameters
    ----------
    data : tuple
        (pair_tensor, mask_tensor, weights_dict, config_dict)
        - pair_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict mapping weight names to torch.Tensor
        - config_dict  : contains "dim" and "hidden_dim" (ints)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] after the TriMul update, same dtype/device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    pair_tensor, mask, weights, config = data
    device = pair_tensor.device
    B, N, N2, C = pair_tensor.shape
    assert N == N2, "Input must be square in the sequence dimensions."
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # ------------------------------------------------------------------
    # Extract weights
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input layer‑norm over channel C
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        pair_tensor,
        (C,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask_val = torch.ones((B, N, N), dtype=pair_tensor.dtype, device=device)
    else:
        mask_val = mask.to(dtype=pair_tensor.dtype, device=device)
        if mask_val.dim() != 3:
            raise ValueError("Mask must have shape [B, N, N]")
    mask_val = mask_val.unsqueeze(-1)      # [B,N,N,1]

    left = left * mask_val
    right = right * mask_val

    # ------------------------------------------------------------------
    # 4. Gating tensors (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul: batched matmul C = A @ Bᵀ
    #    A = left (i,k), B = right (j,k) --> out[i,j] = Σ_k left[i,k]*right[j,k]
    # ------------------------------------------------------------------
    # Move hidden dim to second axis and make tensors contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, i, k]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, hidden, j, k]

    # Collapse batch and hidden dim for the kernel
    B_H = B * hidden_dim
    left_flat = left.view(B_H, N, N)               # [B_H, M=N, K=N]
    right_T = right.view(B_H, N, N).transpose(-2, -1)  # [B_H, K=N, N]

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # 6. Triton kernel launch
    # ------------------------------------------------------------------
    # Tune block sizes (they are compile‑time constants)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B_H,
    )

    _trigemul_matmul_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        N,          # M
        N,          # N
        N,          # K
        # strides for A
        left_flat.stride(0),   # batch stride
        left_flat.stride(1),   # M stride
        left_flat.stride(2),   # K stride
        # strides for B (already transposed)
        right_T.stride(0),     # batch stride
        right_T.stride(1),     # K stride
        right_T.stride(2),     # N stride
        # strides for C
        out_flat.stride(0),    # batch stride
        out_flat.stride(1),    # M stride
        out_flat.stride(2),    # N stride
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Post‑processing: hidden‑dim layer norm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # back to original dim C

    return out
