def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Implements the heavy contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    as a batched half‑precision matrix multiplication fused in a
    Triton kernel. All surrounding layer‑norms, linear projections and
    gating are performed with PyTorch.
    """
    input_tensor, mask, weights, config = data

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = (not config.get("nomask", False)) and (mask is not None)

    # LayerNorm over the channel dimension `dim`
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Gating values (sigmoid)
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # Optional mask
    if use_mask:
        # mask shape: [bs, N, N] -> [bs, N, N, 1]
        mask_exp = mask.unsqueeze(-1).type_as(left)
        left = left * mask_exp
        right = right * mask_exp

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half‑precision for the large GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Re‑order tensors to (batch*hidden_dim, M, K) and (batch*hidden_dim, K, N)
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    # right needs a transpose on the two sequence axes
    right_reshape = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)

    # Output buffer (FP32)
    out_reshape = torch.empty(
        (bs * hidden_dim, N, N), dtype=torch.float32, device=left_reshape.device
    )

    # Strides for the kernel (already contiguous)
    stride_am, stride_ak, stride_a_batch = left_reshape.stride()
    stride_bk, stride_bn, stride_b_batch = right_reshape.stride()
    stride_cm, stride_cn, stride_c_batch = out_reshape.stride()

    # Tile size selection – tuned for A100/B200 tensor‑core performance
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )

    batched_matmul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_a_batch,
        stride_bk,
        stride_bn,
        stride_b_batch,
        stride_cm,
        stride_cn,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [bs, N, N, hidden_dim]
    out = out_reshape.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm over the hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to the original `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out

        c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )

def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    The heavy contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is performed as a batched half‑precision matrix multiplication
    fused in a Triton kernel. The surrounding LayerNorms, linear
    projections and gating are executed with PyTorch.
    """
    input_tensor, mask, weights, config = data

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = (not config.get("nomask", False)) and (mask is not None)

    # ---------- 1. Input LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- 2. Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- 3. Gating (sigmoid) ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ---------- 4. Optional mask ----------
    if use_mask:
        mask_exp = mask.unsqueeze(-1).type_as(left)   # [bs, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ---------- 5. Apply gates ----------
    left = left * left_gate
    right = right * right_gate

    # ---------- 6. Cast to FP16 for the large GEMM ----------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ---------- 7. Re‑order tensors for batched GEMM ----------
    # left: (B, N, N, H) -> (B, H, N, N)
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    # right needs transpose of the two sequence axes: (B, N, N, H) -> (B, H, N, N) with (K, J) order
    right_reshape = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)

    # Output buffer (FP32 for later ops)
    out_reshape = torch.empty(
        (bs * hidden_dim, N, N), dtype=torch.float32, device=left_reshape.device
    )

    # Strides (batch, M, K) / (batch, K, N) / (batch, M, N)
    stride_a_batch, stride_am, stride_ak = left_reshape.stride()
    stride_b_batch, stride_bk, stride_bn = right_reshape.stride()
    stride_c_batch, stride_cm, stride_cn = out_reshape.stride()

    # Tile sizes – tuned for tensor‑core efficiency on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )

    # Launch the Triton kernel
    batched_matmul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_a_batch,
        stride_bk,
        stride_bn,
        stride_b_batch,
        stride_cm,
        stride_cn,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ---------- 8. Reshape back to [bs, N, N, hidden_dim] ----------
    out = out_reshape.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- 9. Output LayerNorm ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ---------- 10. Output gate ----------
    out = out * out_gate

    # ---------- 11. Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
