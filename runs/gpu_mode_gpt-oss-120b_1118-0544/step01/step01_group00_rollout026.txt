"""
Outgoing Triangle Multiplicative Update (TriMul) implementation.

The kernel computes, for each batch and hidden channel d:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

where left and right are linear projections of the normalized input,
optionally masked and gated. The heavy batched matrix multiplication
(C = A @ Bᵀ) is performed in a fused Triton kernel using FP16 inputs
with FP32 accumulation. LayerNorms, gating, masking and the final
projection are done in PyTorch. The projection + gate linear layers are
fused to reduce memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A: M×K, B: N×K)
# -------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile column
    pid_b = tl.program_id(2)          # batch * hidden id

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the current (batch, hidden) matrix
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (k_offset[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Bᵀ tile: we read B as (K, N) using its stride layout
        b_ptrs = b_batch + (k_offset[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # accumulate dot product; promote to FP32 for accuracy
        acc += tl.dot(a, b).to(tl.float32)

    # write the result tile
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------
# Public entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float) – may be ignored
            - weights: dict of model weights (see skeleton)
            - config: dict with at least:
                * "dim" (int) – input channel dimension
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True mask is ignored (default False)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Fused projection + gating for left and right branches
    # ------------------------------------------------------------------
    # concatenate projection and gate weights (out_features stacked)
    left_comb_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0)
    right_comb_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0)

    # combined linear: produces (proj | gate) for each hidden dim
    left_comb = F.linear(x, left_comb_weight)   # [B, N, N, 2*hidden]
    right_comb = F.linear(x, right_comb_weight)

    # split
    left_proj = left_comb[..., :hidden_dim]
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])
    right_proj = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])

    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 3. Outgoing gate (applied after the matrix multiplication)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Cast to FP16 and reshape for the Triton kernel
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape
    batch_dim = B * hidden_dim  # total number of independent matrices

    # (B, hidden, N, N) -> (batch_dim, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(batch_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(batch_dim, N, N)

    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 5. Launch Triton kernel for batched matmul C = A @ Bᵀ
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m, num_pid_n, batch_dim)

    # strides are expressed in elements (not bytes)
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)

    stride_b_batch = right_h.stride(0)
    stride_b_k = right_h.stride(2)   # inner-most (col) stride
    stride_b_n = right_h.stride(1)   # row stride

    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    tri_mul_outgoing_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Reshape back to [B, N, N, hidden] (float32)
    # ------------------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)          # [B, N, N, hidden]
        .contiguous()
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 7. Final LayerNorm, gating, and projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
