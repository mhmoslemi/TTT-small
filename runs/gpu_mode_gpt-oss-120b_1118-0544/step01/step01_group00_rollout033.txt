```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,   # note B is transposed
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_STAGES: tl.constexpr = 2):
    pid = tl.program_id(0)   # over M blocks
    pid_n = tl.program_id(1) # over N blocks
    pid_b = tl.program_id(2) # batch index

    # Compute offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for within dimensions
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of each batch
    left_batch_ptr = left_ptr + pid_b * stride_l_batch
    right_batch_ptr = right_ptr + pid_b * stride_r_batch
    out_batch_ptr = out_ptr + pid_b * stride_o_batch

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # load A tile (M, K)
        a_ptrs = left_batch_ptr + (offs_m[:, None] * stride_l_m) + (k_offset[None, :] * stride_l_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)  # fp16

        # load B tile (N, K) transposed orientation:
        # we need B[j,k] but we want B_T[k,j] shape KxN
        # We can load B as (K,N) using strides: (k_offset[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_n)
        b_ptrs = right_batch_ptr + (k_offset[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot
        acc += tl.dot(a, b)   # result float32

    # write out
    out_ptrs = out_batch_ptr + (offs_m[:, None] * stride_o_m) + (offs_n[None, :] * stride_o_n)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get('nomask', False)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])
    # Cast to half for projection speed
    x_half = x.to(torch.float16)

    # Linear projections
    left = F.linear(x_half, weights['left_proj.weight'].to(torch.float16))
    right = F.linear(x_half, weights['right_proj.weight'].to(torch.float16))

    # Apply mask if needed
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # use float dtype same as original dtype (float32)
        left = left * mask_f
        right = right * mask_f

    # Compute gates (full precision? half)
    # gating uses original normalized x (float32) for linear then sigmoid, then cast
    left_gate = torch.sigmoid(F.linear(x, weights['left_gate.weight']))
    right_gate = torch.sigmoid(F.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(F.linear(x, weights['out_gate.weight']))

    # Cast gates to half and apply
    left = left * left_gate.to(torch.float16)
    right = right * right_gate.to(torch.float16)

    B, N, _, _ = left.shape # left shape [B,N,N, hidden_dim]

    # reshape to (B*hidden_dim, N, N) for GEMM
    left_h = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # allocate out
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    batch_dim = left_h.shape[0]   # B * hidden_dim

    grid = (num_m, num_n, batch_dim)

    # strides
    stride_l_batch, stride_l_m, stride_l_k = left_h.stride()
    stride_r_batch, stride_r_m, stride_r_k = right_h.stride()
    # for B transposed we need stride_k and stride_n
    stride_r_k = stride_r_k  # original stride of last dim (k)
    stride_r_n = stride_r_m  # original stride of second dim (N)
    stride_o_batch, stride_o_m, stride_o_n = out_h.stride()

    tri_mul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        batch_dim,
        N,
        N,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_k,
        stride_r_n,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape output
    out = out_h.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous().to(torch.float32)

    # final LN and gating and projection

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton implementation.

The heavy part of TriMul is the cubic reduction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication C = A @ Bᵀ for every
(batch * hidden_dim) slice (A,B have shape N×N).

The implementation follows these steps:
1. Layer‑norm the input.
2. Linear projections → left / right (FP16) and three gating linear layers.
3. Apply mask (if present), gating and cast to fp16.
4. Reshape to (B·hidden_dim, N, N) and launch a custom Triton kernel that
   computes C = A @ Bᵀ using tiled fp16 loads and fp32 accumulation.
5. Reshape back, apply the final layer‑norm, out‑gate and linear projection
   to the original dimension.

The Triton kernel uses 64×64 tiles with a 64‑element K‑block and
leverages `tl.dot` which maps to Tensor‑Core FP16 × FP16 → FP32
multiply‑add.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,                # left  tensor (batch, M, K)
    b_ptr,                # right tensor (batch, N, K) – treated as transposed
    c_ptr,                # output tensor (batch, M, N)
    batch: tl.constexpr,  # total number of independent matrices (B * hidden_dim)
    M: tl.constexpr,      # rows of A (also rows of output)
    N: tl.constexpr,      # rows of B (also cols of output)
    K: tl.constexpr,      # reduction dimension
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile column
    pid_b = tl.program_id(2)          # which matrix in the batch

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    num_k = tl.cdiv(K, BLOCK_K)
    for kk in range(num_k):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A tile (M x K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (k_offset[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load Bᵀ tile (K x N) – B is stored as (N, K) so we flip the strides
        b_ptrs = b_batch + (k_offset[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)   # fp32 accumulation

    # Write the result
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32 or bool); may be unused if nomask=True
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with at least "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------
    # 1. LayerNorm on the input (over the last dim)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # -------------------------------------------------
    # 2. Linear projections (FP16 for speed)
    # -------------------------------------------------
    # Cast to half‑precision once and keep the half tensors
    x_half = x.to(torch.float16)

    left = F.linear(x_half, weights["left_proj.weight"].to(torch.float16))   # [B, N, N, hidden_dim]
    right = F.linear(x_half, weights["right_proj.weight"].to(torch.float16))

    # -------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # -------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1] float32
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # float32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, hidden_dim]

    # Cast gates to half and apply
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    left = left * left_gate_h
    right = right * right_gate_h

    B, N, _, _ = left.shape

    # -------------------------------------------------
    # 5. Reshape to (B*hidden_dim, N, N) – contiguous layout for GEMM
    # -------------------------------------------------
    left_h = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (FP16)
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # -------------------------------------------------
    # 6. Launch the Triton kernel
    # -------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    batch_dim = left_h.shape[0]               # B * hidden_dim

    grid = (num_m, num_n, batch_dim)

    # Strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = left_h.stride()
    stride_b_batch, stride_b_m, stride_b_k = right_h.stride()
    # For Bᵀ we need stride over the reduction axis (k) and the output axis (n)
    stride_b_k = stride_b_k                     # original innermost stride (k)
    stride_b_n = stride_b_m                     # original stride of the N dimension

    stride_c_batch, stride_c_m, stride_c_n = out_h.stride()

    # Kernel launch (asynchronous)
    tri_mul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        batch_dim,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel has finished before proceeding
    torch.cuda.synchronize()

    # -------------------------------------------------
    # 7. Reshape back to the original layout
    # -------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)          # [B, N, N, hidden_dim]
        .contiguous()
        .to(torch.float32)
    )

    # -------------------------------------------------
    # 8. Final layer‑norm, out‑gate and projection back to dim
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate               # gating (both are float32)
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
