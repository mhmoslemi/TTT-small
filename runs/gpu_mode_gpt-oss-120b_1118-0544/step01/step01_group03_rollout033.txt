"""
TriMul outgoing update – Triton‑accelerated batched matrix multiplication.

The core operation is
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which can be expressed as a batched GEMM:
    out_flat = left_flat @ right_T_flat
where left_flat ∈ (B·H, N, N) and right_T_flat = right_flat.transpose(-2,-1).

This file implements:
- Layer‑norm and linear projections in PyTorch.
- Optional masking and gating.
- A custom Triton kernel for the batched GEMM.
- Post‑norm, gating and final linear projection back to the input dimension.
"""
import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes (all equal to seq_len)
    stride_a_batch, stride_am, stride_ak,   # A strides: (batch, M, K)
    stride_b_batch, stride_bk, stride_bn,   # B strides: (batch, K, N)   (B is already transposed)
    stride_c_batch, stride_cm, stride_cn,   # C strides: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ B  (A:[batch,M,K], B:[batch,K,N]).

    The kernel tiles the output matrix in blocks of size BLOCK_M×BLOCK_N and
    iterates over the K dimension in chunks of BLOCK_K.
    """
    pid_m = tl.program_id(0)      # tile index in M dimension
    pid_n = tl.program_id(1)      # tile index in N dimension
    pid_batch = tl.program_id(2)  # batch (B·H) index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds loads
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Base pointers for current batch
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Pointers to the tiles of A and B
    a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)                       # dot -> same dtype as loads (float32)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the result tile
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (pair_tensor, mask_tensor, weights_dict, config_dict)
        - pair_tensor : torch.Tensor of shape [B, N, N, C] (float32, on CUDA)
        - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict: mapping from string names to torch.Tensor weights
        - config_dict : contains "dim", "hidden_dim" and optionally "layernorm_eps"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] after the TriMul update (float32, on CUDA).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    pair_tensor, mask, weights, config = data
    device = pair_tensor.device
    B, N, N2, C = pair_tensor.shape
    assert N == N2, "Pair tensor must be square in the sequence dimensions."

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over channel dimension C)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        pair_tensor,
        (C,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask_val = torch.ones((B, N, N), dtype=pair_tensor.dtype, device=device)
    else:
        mask_val = mask.to(dtype=pair_tensor.dtype, device=device)
        if mask_val.dim() != 3:
            raise ValueError("Mask must have shape [B, N, N]")
    mask_val = mask_val.unsqueeze(-1)  # [B, N, N, 1]
    left = left * mask_val
    right = right * mask_val

    # ------------------------------------------------------------------
    # 4. Gating tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul – batched GEMM via Triton
    # ------------------------------------------------------------------
    # Move hidden dimension before the two sequence dimensions for easier flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]  (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, hidden, N, N]  (k,j)

    # Optionally reduce precision for the heavy matmul
    left_flat = left_perm.view(B * hidden_dim, N, N).contiguous()
    right_flat = right_perm.view(B * hidden_dim, N, N).contiguous()
    # Cast to half‑precision to improve bandwidth (still safe for the model)
    left_flat = left_flat.half()
    right_flat = right_flat.half()

    out_flat = torch.empty_like(left_flat)  # half‑precision result

    # Strides (in number of elements)
    a_batch_stride = left_flat.stride(0)
    a_m_stride = left_flat.stride(1)
    a_k_stride = left_flat.stride(2)

    b_batch_stride = right_flat.stride(0)
    b_k_stride = right_flat.stride(1)
    b_n_stride = right_flat.stride(2)

    c_batch_stride = out_flat.stride(0)
    c_m_stride = out_flat.stride(1)
    c_n_stride = out_flat.stride(2)

    # Choose block sizes (simple heuristic)
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        a_batch_stride, a_m_stride, a_k_stride,
        b_batch_stride, b_k_stride, b_n_stride,
        c_batch_stride, c_m_stride, c_n_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # reasonable default
    )

    # Reshape back to [B, N, N, hidden_dim] and cast to float32 for following ops
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().float()

    # ------------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm, gate, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # back to original dim

    return out
