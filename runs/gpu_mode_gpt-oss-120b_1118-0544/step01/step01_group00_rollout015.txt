"""
Outgoing triangle multiplicative update (TriMul) optimized with a Triton
batched matrix‑multiply kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Linear projections → left/right tensors (B,N,N,hidden_dim).
3. Optional mask multiplication (broadcast over the hidden dim).
4. Compute gating values (sigmoid of linear projections) and apply them:
   left  = left  * left_gate
   right = right * right_gate
5. Cast left/right to float16 and reshape to (B*hidden, N, N) so that each
   hidden channel becomes an independent matrix‑multiply.
6. Compute C = A @ Bᵀ for every (batch * hidden) matrix with a custom
   Triton kernel that tiles the M, N and K dimensions (block sizes can be
   tuned).  The kernel works in FP16 and accumulates in FP32.
7. Reshape the result back to (B, N, N, hidden_dim) and convert to FP32.
8. Layer‑norm over the hidden dimension, apply the output gate and a final
   linear projection back to the original feature dimension.

Only the heavy O(N³) reduction (step 6) is performed in Triton; all other
operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ  (A,B : (batch, N, N))
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,                # left  ptr  (batch, M, K)
    b_ptr,                # right ptr  (batch, N, K) – accessed transposed
    c_ptr,                # output ptr (batch, M, N)
    M: tl.constexpr,      # rows of A
    N: tl.constexpr,      # rows of B (also cols of C)
    K: tl.constexpr,      # reduction dim
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # batch element (B * hidden)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K in blocks
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A tile (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (k_offset[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load Bᵀ tile (BLOCK_K, BLOCK_N) – we read B as (K, N)
        b_ptrs = b_batch + (k_offset[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Rank‑K update
        acc += tl.dot(a, b)          # fp16 dot → fp32 accumulate

    # Store result
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask        : torch.Tensor [B, N, N] (bool/float) – optional
            weights     : dict of model weight tensors
            config      : dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------------------------------------------------------
    # 1. Input layer‑norm
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ---------------------------------------------------------
    # 2. Linear projections (no bias)
    # ---------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ---------------------------------------------------------
    if not nomask:
        # mask may be bool or float; ensure same dtype as tensors
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ---------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ---------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 5. Cast to FP16 for the O(N³) reduction
    # ---------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape

    # Reshape to (B * hidden_dim, N, N) – each hidden channel is an
    # independent matrix for the batched GEMM.
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (FP16)
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # ---------------------------------------------------------
    # 6. Triton kernel launch (batched matmul)
    # ---------------------------------------------------------
    # Block sizes – tuned for typical B200 GPUs; adapt for very small N.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    batch_dim = left_h.shape[0]                     # B * hidden_dim

    grid = (num_pid_m, num_pid_n, batch_dim)

    # Strides in element units (not bytes)
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)

    stride_b_batch = right_h.stride(0)
    stride_b_k = right_h.stride(2)   # fast‑changing (col) dimension
    stride_b_n = right_h.stride(1)   # row dimension

    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    tri_mul_outgoing_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ---------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim] (FP32)
    # ---------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden_dim]

    # ---------------------------------------------------------
    # 8. Final LayerNorm, gating and projection to original dim
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate          # apply output gate (float32)
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return out
