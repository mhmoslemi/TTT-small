"""
TriMul (outgoing) – high‑performance Triton implementation.

The module performs:
  1. LayerNorm on the input [B, N, N, C].
  2. Linear projections to a hidden dimension H.
  3. Optional masking, and per‑position gating (sigmoid).
  4. Core contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
     which is a batched matrix multiplication left_slice @ right_sliceᵀ.
  5. LayerNorm + output gating + final linear projection back to C.

The O(N³) contraction is executed by a fused Triton kernel that
processes each (batch, hidden) slice independently, using FP16
loads & Tensor‑Core dot products (accumulated in FP32).  The remaining
operations (LayerNorm, Linear, sigmoid) stay in PyTorch – they are
light compared to the dense contraction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to left, right, output
    M, N, K,                           # matrix dimensions (all equal to seq_len)
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel computing C = A @ Bᵀ.

    A has shape (batch, M, K), B has shape (batch, N, K) and the result
    C has shape (batch, M, N).  The kernel is launched with a 3‑D grid:
      pid_batch = program_id(0)   # which (batch * hidden) slice
      pid_m     = program_id(1)   # tile row index
      pid_n     = program_id(2)   # tile column index
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute the absolute offsets for the current tile.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current (batch, hidden) slice.
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Accumulator in FP32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K steps.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B block transposed on‑the‑fly: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Perform the dot product (Tensor‑Core if possible).
        acc += tl.dot(a, b)

    # Write the result tile.
    c_ptrs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Entry point for the Triton‑accelerated TriMul (outgoing) module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (already on the same device)
            - config: dict containing at least
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – bool, True if mask should be ignored
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over the hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)        # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – computed on the original normalized tensor
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_norm, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        F.linear(x_norm, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        F.linear(x_norm, weights["out_gate.weight"])
    )

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the heavy contraction
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # B, N, N, hidden_dim

    # Rearrange to (B, hidden_dim, N, N) → (batch_hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    batch_hidden, M, K = left.shape[0], left.shape[2], left.shape[3]  # M = K = N

    left_mat = left.view(batch_hidden, M, K)
    right_mat = right.view(batch_hidden, M, K)

    # Output buffer (float32 accumulator)
    out_mat = torch.empty((batch_hidden, M, K), device=device, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 6. Choose block sizes (simple heuristic)
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128
    BLOCK_K = 64  # works well for both precisions

    # Strides in number of elements (tensor is contiguous after view)
    stride_a_batch = M * K
    stride_a_row = K
    stride_a_col = 1

    stride_b_batch = M * K
    stride_b_row = K
    stride_b_col = 1

    stride_c_batch = M * K
    stride_c_row = K
    stride_c_col = 1

    grid = (
        batch_hidden,
        (M + BLOCK_M - 1) // BLOCK_M,
        (K + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the kernel
    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M,
        K,
        K,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply output gate (out_gate is float32, out is float32)
    out = out * out_gate

    # Final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out
