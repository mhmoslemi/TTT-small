import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_cast_kernel(out_fp16_ptr, gate_fp32_ptr, out_fp32_ptr, N_ELEM: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM
    # load fp16, convert to fp32
    a = tl.load(out_fp16_ptr + offs, mask=mask).to(tl.float32)
    b = tl.load(gate_fp32_ptr + offs, mask=mask, other=0.0)
    c = a * b
    tl.store(out_fp32_ptr + offs, c, mask=mask)

def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    This implementation:
    - Performs LayerNorm, linear projections, and gating in PyTorch.
    - Executes the core reduction `out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]`
      using a batched half‑precision matrix multiplication (torch.bmm) for high
      throughput.
    - Applies the final gating, LayerNorm and output projection, with the
      elementwise multiplication and fp16→fp32 cast fused in a tiny Triton kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim], float32
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5) Apply gates and cast to half precision for the reduction
    # ------------------------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 6) Batched matrix multiplication: out_h = left @ rightᵀ
    #    We treat each hidden channel as an independent batch element.
    # ------------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batch matmul (FP16 input, FP32 accumulation on tensor cores)
    out_h = torch.bmm(left_flat, right_flat.transpose(1, 2))  # [B*hidden_dim, N, N], fp16

    # reshape back to [B, N, N, hidden_dim] (still fp16)
    out_h = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7) Fuse fp16→fp32 cast and element‑wise multiplication with out_gate
    #    using a tiny Triton kernel.
    # ------------------------------------------------------------------
    total_elements = out_h.numel()
    out = torch.empty_like(out_h, dtype=torch.float32)  # allocate fp32 output

    # launch kernel
    BLOCK = 8192  # number of elements processed per program
    num_warps = 1
    grid = ( (total_elements + BLOCK - 1) // BLOCK, )
    elementwise_mul_cast_kernel[grid](
        out_h_ptr=out_h,
        gate_ptr=out_gate.contiguous().view(-1),
        out_ptr=out,
        N_ELEM=total_elements,
        BLOCK=BLOCK,
        num_warps=num_warps,
    )
    # ------------------------------------------------------------------
    # 8) Final LayerNorm, gating and projection back to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate  # out_gate already float32
    out = F.linear(out, to_out_weight)  # -> [B, N, N, dim]

    return out

Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched half‑precision matrix multiplication (torch.bmm)
combined with a tiny Triton kernel that only casts the FP16 result to FP32.
All LayerNorms, linear projections and gating are done in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def cast_fp16_to_fp32_kernel(
    in_ptr,          # FP16 source
    out_ptr,         # FP32 destination
    N_ELEM: tl.constexpr,   # total number of elements
    BLOCK: tl.constexpr,    # block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM
    # Load FP16, promote to FP32 and store
    val = tl.load(in_ptr + offs, mask=mask).to(tl.float32)
    tl.store(out_ptr + offs, val, mask=mask)

            - mask: torch.Tensor [B, N, N] (bool or float) – may be ignored
            - weights: dict of model weights (see skeleton)
            - config: dict containing at least:
                * "dim" (int)
                * "hidden_dim" (int)

    Returns:

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1) LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(

        normalized_shape=(dim,),

    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim], float32

    # ------------------------------------------------------------------

        left = left * mask_f
        right = right * mask_f

    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # used later

    )
    out = out * out_gate                     # apply the output gate
    out = F.linear(out, to_out_weight)       # [B, N, N, dim]

    return out
```
