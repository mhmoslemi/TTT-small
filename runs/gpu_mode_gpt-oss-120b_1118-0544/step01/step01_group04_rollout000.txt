"""
TriMul (outgoing) forward pass implemented with Triton.

Algorithm:
1. LayerNorm on the input tensor.
2. Linear projections to hidden dimension and gated signals (all in PyTorch).
3. Optional mask applied to the projected tensors.
4. Core contraction:
   out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
   Implemented as a batched GEMM for each hidden channel using a
   Triton kernel (fp16 inputs, fp32 accumulation).
5. Output LayerNorm, output gate, and final linear projection back to the
   original dimension (all in PyTorch).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_a_batch, stride_b_batch, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM kernel: C = A @ B   (A: M×K, B: K×N)

    A and B are fp16, C is fp32. Batch dimension is the 3rd program ID.
    """
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # batch (bs * hidden_dim)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch_ptr = a_ptr + pid_b * stride_a_batch
    b_batch_ptr = b_ptr + pid_b * stride_b_batch
    c_batch_ptr = c_ptr + pid_b * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k_offset in range(0, K, BLOCK_K):
        cur_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile: (M x K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # B tile: (K x N)
        b_ptrs = b_batch_ptr + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # dot product, accumulate in fp32
        acc += tl.dot(a, b).to(tl.float32)

    # store result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple containing
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask: torch.Tensor or None, shape [B, N, N]
            - weights: dict of torch.Tensor with model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = (not config.get("nomask", False)) and (mask is not None)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [bs, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Gate computations (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # --------------------------------------------------------------
    # 4. Apply optional mask
    # --------------------------------------------------------------
    if use_mask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [bs, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 5. Apply gates to projected tensors
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Core contraction via Triton GEMM (left @ right^T over K)
    # --------------------------------------------------------------
    # Cast to fp16 for the heavy multiply to improve bandwidth
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Reshape to (batch*hidden_dim, N, N) and arrange right as (K, N)
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    # right needs its K dimension first: (batch, hidden_dim, K, N)
    right_reshape = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)

    # Output buffer (fp32 accumulation)
    out_reshape = torch.empty(
        (bs * hidden_dim, N, N), dtype=torch.float32, device=device
    )

    # Extract strides (in elements, not bytes)
    stride_a_batch, stride_am, stride_ak = left_reshape.stride()
    stride_b_batch, stride_bk, stride_bn = right_reshape.stride()
    stride_c_batch, stride_cm, stride_cn = out_reshape.stride()

    # --------------------------------------------------------------
    # 7. Triton launch configuration
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    num_batch = bs * hidden_dim

    grid = (num_m, num_n, num_batch)

    _tri_mul_gemm_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N,  # M
        N,  # N
        N,  # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8. Restore original tensor shape
    # --------------------------------------------------------------
    out = out_reshape.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 9. Output LayerNorm over hidden_dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 10. Apply output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 11. Final linear projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
