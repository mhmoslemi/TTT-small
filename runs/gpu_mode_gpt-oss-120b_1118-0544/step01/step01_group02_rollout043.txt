def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Optional mask
    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1) # [B,N,N,1]
    else:
        mask_f = None

    # Convert to half for speed
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    # Fuse linear projection with gates (and mask)
    left = elementwise_mul(left, left_gate)
    right = elementwise_mul(right, right_gate)

    if mask_f is not None:
        mask_f = mask_f.to(torch.float16)
        left = left * mask_f
        right = right * mask_f

    # Reshape for batched GEMM
    B, N, _, H = left.shape  # H==hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]
    left_flat = left.view(B * H, N, N)   # [B*H, N, N]
    right_flat = right.view(B * H, N, N)

    # Compute batched matrix multiplication using torch.bmm
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # [B*H, N, N]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # Output layer norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

highly‑optimized cuBLAS batched matrix multiplication.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size,
                         BLOCK_SIZE: tl.constexpr):
    """Element‑wise product of two 1‑D tensors (a * b).

    Args:
        a_ptr, b_ptr, out_ptr: pointers to the flattened tensors.
        size: total number of elements.
        BLOCK_SIZE: compile‑time tile size.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Helper that launches the Triton kernel to compute a * b."""
    assert a.shape == b.shape, "Shapes must match for elementwise mul"
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](
        a.view(-1), b.view(-1), out.view(-1), size,
        BLOCK_SIZE=BLOCK,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model weights (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------------
    # 2) Linear projections (left/right) – keep in float16 for speed
    # --------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # --------------------------------------------------------------------
    # 3) Gating (left/right/out) – also in float16 for the first two gates
    # --------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid().to(torch.float16)
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid().to(torch.float16)
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()   # keep float32

    # --------------------------------------------------------------------
    # 4) Fuse projection with gate (and optional mask) using Triton
    # --------------------------------------------------------------------
    left = _elemwise_mul(left, left_gate)   # left := left * left_gate
    right = _elemwise_mul(right, right_gate)  # right := right * right_gate

    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        mask_f = mask_f.to(torch.float16)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------
    # 5) Prepare tensors for batched matrix multiplication:
    #    left/right : [B, N, N, H] -> [B*H, N, N] (row‑major, contiguous)
    # --------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left.view(B * H, N, N)   # [B*H, N, N]
    right_flat = right.view(B * H, N, N) # [B*H, N, N]

    # --------------------------------------------------------------------
    # 6) Core O(N³·H) reduction – batched GEMM (cuBLAS)
    # --------------------------------------------------------------------
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # [B*H, N, N]

    # --------------------------------------------------------------------
    # 7) Reshape back to [B, N, N, H] and float32 for the rest of the pipeline
    # --------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().float()  # [B,N,N,H]

    # --------------------------------------------------------------------
    # 8) Output LayerNorm
    # --------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------------
    # 9) Apply output gate
    # --------------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------------
    # 10) Final linear projection back to original dimension
    # --------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
