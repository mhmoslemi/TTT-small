"""
TriMul (outgoing) forward pass implemented in Triton.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension (dim).
2. Linear projections (left/right) from dim → hidden_dim (no bias).
3. Optional mask (broadcasted on the hidden dimension).
4. Compute three sigmoid gates (left, right, out) from the *pre‑norm* input
   and apply the first two to the projected tensors.
5. Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication: for each (b,d) compute
        out = left @ rightᵀ
   The multiplication is performed by a custom Triton kernel that tiles the
   M (i), N (j) and K (k) axes (default 64×64×64) and accumulates in FP32
   while the inputs are FP16.
6. Layer‑norm over the hidden dimension, multiply by the output gate,
   and finally linearly project back to the original dim.

The function `custom_kernel` follows the signature required by the tests.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers to A, B, C
    M, N, K,                      # matrix dimensions (all equal to seq_len)
    stride_ah, stride_am, stride_ak,   # strides for A: batch, M, K
    stride_bh, stride_bk, stride_bn,   # strides for B: batch, K, N
    stride_ch, stride_cm, stride_cn,   # strides for C: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ Bᵀ  with shapes
        A: (batch, M, K)
        B: (batch, N, K)   (note: K is the reduction axis for both)
        C: (batch, M, N)
    The kernel processes one batch (i.e. one (hidden_dim) slice) per
    program_id(0) and tiles M/N with BLOCK_M/N.
    """
    pid_batch = tl.program_id(0)   # collapsed (batch * hidden_dim)
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block col

    # Compute absolute offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for the current batch slice
    a_batch_ptr = a_ptr + pid_batch * stride_ah
    b_batch_ptr = b_ptr + pid_batch * stride_bh
    c_batch_ptr = c_ptr + pid_batch * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Pointers for the current tile
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (FP16)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Accumulate (FP32)
        acc += tl.dot(a, b)

    # Write back the result (FP32)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: Tuple containing
            - input_tensor (torch.Tensor): shape [B, N, N, dim]
            - mask (torch.Tensor or None): shape [B, N, N] (optional)
            - weights (dict of torch.Tensor): model parameters
            - config (dict): contains 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor: shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden_dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).type_as(x)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction via Triton (batched GEMM)
    # -----------------------------------------------------------------
    # Cast to float16 for the GEMM
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to (B, hidden_dim, N, N) where the K axis is the last dim
    # left: (i, k)    -> we keep as (i, k)
    # right: (j, k)   -> we need its transpose so that we can use dot(a,b)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) (transposed on last two dims)

    B, H, N, _ = left_perm.shape
    # Collapse batch and head dimensions → treat each (b, h) as an independent GEMM
    left_mat = left_perm.view(B * H, N, N)   # (B*H, M, K)
    right_mat = right_perm.view(B * H, N, N)  # (B*H, K, N)

    # Output buffer (FP32 accumulator)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=x.device)

    # Tunable block sizes – 64 works well for the supported sequence lengths.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # batch*head dimension
        (N + BLOCK_M - 1) // BLOCK_M,          # tiles over M (i)
        (N + BLOCK_N - 1) // BLOCK_N,          # tiles over N (j)
    )

    _tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        left_mat.stride(0),
        left_mat.stride(1),
        left_mat.stride(2),
        right_mat.stride(0),
        right_mat.stride(1),
        right_mat.stride(2),
        out_mat.stride(0),
        out_mat.stride(1),
        out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to [B, N, N, dim]

    return out
