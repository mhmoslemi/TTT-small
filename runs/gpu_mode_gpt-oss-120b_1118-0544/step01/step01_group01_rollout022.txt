def custom_kernel(data):
    """
    Triton-accelerated outgoing TriMul forward pass.

    The heavy operation is the batched matrix multiplication
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d],
    which we implement as a fused Triton kernel over the
    (batch * hidden) dimension.
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), norm_weight, norm_bias
    )
    # linear proj
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # mask
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f
    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate
    # Cast to fp16 and prepare for batched matmul
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    # Permute to (B, H, N, N) to make each slice contiguous
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    # Flatten batch*hidden dimension
    left_mat = left.view(-1, N, N)
    right_mat = right.view(-1, N, N)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)
    # Strides in elements
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    stride_bk = right_mat.stride(2)
    stride_bn = right_mat.stride(1)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    # Tile sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Reshape back
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)
    # output layernorm
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden, (hidden_dim,), weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    )
    out_hidden = out_hidden * out_gate
    out = torch.nn.functional.linear(out_hidden, weights["to_out.weight"])
    return out

"""
Trition‑accelerated outgoing TriMul (AlphaFold3).

The heavy computation is the batched matrix multiplication
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
which is equivalent to a batch of N×N matmuls over the
(batch * hidden_dim) dimension.  This kernel materialises each
(b, d) slice as a separate matrix multiply and uses FP16 Tensor
Core dot products with FP32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A, B, C,                 # pointers
    M, N, K,                 # matrix dimensions (all == seq_len)
    stride_am, stride_ak,    # strides for A (M‑row, K‑col)
    stride_bk, stride_bn,    # strides for B (K‑row, N‑col) – B is transposed
    stride_cm, stride_cn,    # strides for C (M‑row, N‑col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_m = tl.program_id(0)               # tile row
    pid_n = tl.program_id(1)               # tile column
    pid_hb = tl.program_id(2)              # batch * hidden index

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this (batch,hidden) slice
    # layout is contiguous: each slice occupies M*K, K*N and M*N elements
    a_base = A + pid_hb * (M * K)
    b_base = B + pid_hb * (K * N)
    c_base = C + pid_hb * (M * N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K) + k_offset

        # A: (M, K)
        a_ptrs = a_base + (offs_m[:, None] * stride_am
                           + k_range[None, :] * stride_ak)
        # B (transposed): (K, N)
        b_ptrs = b_base + (k_range[:, None] * stride_bk
                           + offs_n[None, :] * stride_bn)

        a_mask = (offs_m[:, None] < M) & (k_range[None, :] < K)
        b_mask = (k_range[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)                 # fp16 * fp16 -> fp32 accumulation

    # store result
    c_ptrs = c_base + (offs_m[:, None] * stride_cm
                       + offs_n[None, :] * stride_cn)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------- 1) Input LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), norm_weight, norm_bias)

    # ---------- 2) Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- 3) Optional mask ----------
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------- 4) Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- 5) Batched matmul (FP16) ----------
    # Cast to half precision for the heavy reduction
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Make each (batch, hidden) slice a contiguous N×N matrix
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch & hidden
    left_mat = left.view(-1, N, N)                # (B*H, N, N)
    right_mat = right.view(-1, N, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Strides in elements (not bytes)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    stride_bk = right_mat.stride(2)   # transposed view
    stride_bn = right_mat.stride(1)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # Tile sizes – tuned for B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ---------- 6) Restore original layout ----------
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # ---------- 7) Output LayerNorm ----------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )

    # ---------- 8) Output gate and final projection ----------
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])   # [B,N,N,dim]

    return out
