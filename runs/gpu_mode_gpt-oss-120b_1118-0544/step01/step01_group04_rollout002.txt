"""
TriMul outgoing implementation with Triton acceleration.

The heavy contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is realized as a batched matrix multiplication L @ Rᵀ for each
(batch * hidden_dim) slice using a custom Triton kernel.
All other operations (LayerNorm, linear layers, gating, mask handling)
are performed with PyTorch. The final output type is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                        # problem size
    stride_am, stride_ak,           # A: (batch, M, K)
    stride_bk, stride_bn,           # B: (batch, K, N)
    stride_cm, stride_cn,           # C: (batch, M, N)
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a batch of matrix products:
        C[b] = A[b] @ B[b]
    where A has shape (M, K) and B has shape (K, N).
    All tensors are in row‑major order.
    """
    pid_m = tl.program_id(0)   # tile over M
    pid_n = tl.program_id(1)   # tile over N
    pid_b = tl.program_id(2)   # batch dimension (batch * hidden_dim)

    # Offsets for the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of this block (batch offset included)
    a_batch_ptr = a_ptr + pid_b * stride_a_batch
    b_batch_ptr = b_ptr + pid_b * stride_b_batch
    c_batch_ptr = c_ptr + pid_b * stride_c_batch

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for cur_k in range(0, K, BLOCK_K):
        cur_k_off = cur_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k_off < K

        # Load A tile (M × K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (cur_k_off[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile (K × N)
        b_ptrs = b_batch_ptr + (cur_k_off[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Store results
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Arguments:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict containing at least "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = (not config.get("nomask", False)) and mask is not None

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))    # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if use_mask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Batched matmul via Triton (L @ Rᵀ)
    # ------------------------------------------------------------------
    # Cast to half to reduce memory traffic; accumulation stays FP32.
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Reshape to (B*hidden_dim, N, N) where each slice is a matrix.
    #   left  : (i, k)   -> direct
    #   right : (j, k)   -> permute to (k, j) before view.
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)

    # Output buffer (FP32)
    out_reshape = torch.empty(
        (bs * hidden_dim, N, N),
        dtype=torch.float32,
        device=device,
    )

    # Strides for the three tensors (after view)
    stride_a_batch, stride_am, stride_ak = left_reshape.stride()
    stride_b_batch, stride_bk, stride_bn = right_reshape.stride()
    stride_c_batch, stride_cm, stride_cn = out_reshape.stride()

    # Tuning parameters – reasonable defaults for B200.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over M dimension
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over N dimension
        bs * hidden_dim,                # batch * hidden_dim
    )

    batched_matmul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N, N, N,                     # M, N, K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_reshape.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 9. Output gating
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10. Final projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
