    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)

    out = a * b * c

def custom_kernel(data):
    x, mask, weights, config = data

    # extraction
    device = x.device
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', False)

    # layernorm
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights['norm.weight'],
        bias=weights['norm.bias'],
        eps=1e-5,
    )

    # linear projections
    left_proj = torch.nn.functional.linear(x_norm, weights['left_proj.weight'])
    right_proj = torch.nn.functional.linear(x_norm, weights['right_proj.weight'])

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['left_gate.weight']))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['right_gate.weight']))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['out_gate.weight']))

    # mask handling: create mask_f (if present and not nomask)
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left_proj * left_gate * mask_f
        right = right_proj * right_gate * mask_f
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Cast to half for GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # reshape to batch_hidden dimension for GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer
    out_mat = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # Kernel launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # As K dimension may be N up to 1024

    # Compute strides
    stride_a_batch = left_mat.stride(0)
    stride_a_row = left_mat.stride(1)
    stride_a_col = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_row = right_mat.stride(1)  # N dimension (row)
    stride_b_col = right_mat.stride(2)  # K dimension (col)

    stride_c_batch = out_mat.stride(0)
    stride_c_row = out_mat.stride(1)
    stride_c_col = out_mat.stride(2)

    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # shape [B,N,N,hidden_dim] float32

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply out_gate
    out = out * out_gate

    # final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,

    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # accumulator

    # K loop
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a_offs = a_batch \
                 + offs_m[:, None] * stride_a_row \
                 + cur_k[None, :] * stride_a_col

                 + offs_n[None, :] * stride_b_row

             + offs_n[None, :] * stride_c_col

"""
TriMul outgoing variant implemented with Triton.
The expensive contraction is realized as a batched GEMM kernel that computes
C[b*d, i, j] = sum_k A[b*d, i, k] * B[b*d, j, k] (i.e. A @ Bᵀ)
for each hidden‑dim slice.  All other ops (layer‑norm, linear projections,
gating) are performed with PyTorch.  The kernel loads A and the transposed B
in tiles and accumulates in fp32 while the inputs are fp16.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ.
    A : (batch, M, K)   (fp16)
    B : (batch, N, K)   (fp16)  – will be accessed transposed.
    C : (batch, M, N)   (fp32 accumulator)
    """
    pid_batch = tl.program_id(0)   # which matrix in the batch
    pid_m = tl.program_id(1)       # tile row index
    pid_n = tl.program_id(2)       # tile column index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the current batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A block: (BLOCK_M, BLOCK_K)
        a_offs = a_batch \
                 + offs_m[:, None] * stride_a_row \
                 + cur_k[None, :] * stride_a_col
        a = tl.load(a_offs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B block transposed on‑the‑fly to get shape (BLOCK_K, BLOCK_N)
        b_offs = b_batch \
                 + cur_k[:, None] * stride_b_col \
                 + offs_n[None, :] * stride_b_row
        b = tl.load(b_offs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product, accumulation in fp32

"""
TriMul outgoing variant implemented with Triton.

The heavy O(N³) contraction is expressed as a batched GEMM:
    C[b*d, i, j] = Σ_k A[b*d, i, k] * B[b*d, j, k]   (i.e. A @ Bᵀ)

A and B have shape (batch * hidden_dim, N, N) and are stored in fp16.
The kernel tiles the multiplication, accumulates in fp32 and writes the
result back in fp32.  All other operations (LayerNorm, Linear, gating) are
performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ.

    A : (batch, M, K)   fp16
    B : (batch, N, K)   fp16   (accessed transposed)
    C : (batch, M, N)   fp32 accumulator
    """
    pid_batch = tl.program_id(0)   # which matrix in the batch
    pid_m = tl.program_id(1)       # tile row index
    pid_n = tl.program_id(2)       # tile column index

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block (BLOCK_M, BLOCK_K)
        a_offs = a_batch \
                 + offs_m[:, None] * stride_a_row \
                 + cur_k[None, :] * stride_a_col
        a = tl.load(a_offs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B block transposed on‑the‑fly to get (BLOCK_K, BLOCK_N)
        b_offs = b_batch \
                 + cur_k[:, None] * stride_b_col \
                 + offs_n[None, :] * stride_b_row
        b = tl.load(b_offs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b)

    # Write back the result
    c_offs = c_batch \
             + offs_m[:, None] * stride_c_row \
             + offs_n[None, :] * stride_c_col
    tl.store(c_offs, acc,
             mask=mask_m[:, None] & mask_n[None, :])
