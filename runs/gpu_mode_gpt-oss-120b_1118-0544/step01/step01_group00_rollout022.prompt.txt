import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
Outgoing Triangle Multiplicative Update (TriMul) kernel.

The heavy part of TriMul is the reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication C = A @ Bᵀ where each batch
corresponds to a (batch * hidden_dim) slice of the projected tensors.
The Triton kernel below computes this reduction in FP32 while the
projections, layer‑norms and gating are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ   (A: M×K, B: N×K)
# -------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  tensor pointer  (batch, M, K)
    b_ptr,               # right tensor pointer  (batch, N, K) – we treat it as transposed inside
    c_ptr,               # output tensor pointer (batch, M, N)
    batch: tl.constexpr, # number of independent matrices = B * hidden_dim
    M: tl.constexpr,     # rows of A (also rows of output)
    N: tl.constexpr,     # rows of B (also cols of output)
    K: tl.constexpr,     # shared reduction dimension
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # which matrix in the batch

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis K in blocks
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (k_offset[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Bᵀ tile: we read B as (K, N) using its original strides (col -> k, row -> n)
        b_ptrs = b_batch + (k_offset[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product for the tile (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b).to(tl.float32)

    # write results
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# Public entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32 or bool) – may be ignored
            - weights: dict of model weights (see skeleton)
            - config: dict containing at least:
                * "dim" (int)
                * "hidden_dim" (int)
                * "nomask" (bool) – if True the mask is not applied
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------------------------------------------------------
    # unpack weights
    # ---------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ---------------------------------------------------------
    # LayerNorm on the input (over the last dim)
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ---------------------------------------------------------
    # Linear projections (no bias)
    # ---------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ---------------------------------------------------------
    # Optional mask (broadcast on the hidden dimension)
    # ---------------------------------------------------------
    if not nomask:
        # mask may be bool; convert to same dtype as tensors
        mask_f = mask.to(x.dtype).unsqueeze(-1)           # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ---------------------------------------------------------
    # Gating (sigmoid of linear projects)
    # ---------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # Cast to half for the reduction to improve bandwidth
    # ---------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape

    # Reshape to (B*hidden, N, N) and make contiguous
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Allocate output buffer (half)
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # ---------------------------------------------------------
    # Triton kernel launch parameters
    # ---------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    batch_dim = left_h.shape[0]                     # B * hidden_dim

    grid = (num_pid_m, num_pid_n, batch_dim)

    # Strides (in elements, not bytes)
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)

    stride_b_batch = right_h.stride(0)
    # For Bᵀ we need stride over k (original column) and n (original row)
    stride_b_k = right_h.stride(2)   # col stride = 1
    stride_b_n = right_h.stride(1)   # row stride = N

    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    # Launch kernel
    tri_mul_outgoing_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        batch=batch_dim,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ---------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] and convert to fp32
    # ---------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden_dim]

    # ---------------------------------------------------------
    # Final LayerNorm, gating and projection back to dim
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                               # gating
    out = F.linear(out, to_out_weight)                 # [B, N, N, dim]

    return out

**Approach and Rationale**
We are to implement the "outgoing" version of the Triangle Multiplicative Update (TriMul) operator, which requires processing a 4D tensor [B, N, N, C]. The operation involves batched linear projections, elementwise masking, gating mechanisms, and a core triply nested loop reduction. The bottleneck is the reduction operation, which involves N^3 operations per element and must be optimized using tiling and cache blocking. We need to compute:

   out[b, i, j, d] = σ_{k} [left[b, i, k, d] * right[b, j, k, d]]

where left and right are projected and gated tensors from the input.

**Mathematical Specification**
The input tensor `x` is of shape [B, N, N, C]. Its structure is:

   x[b, i, j, c]: for batch b, sequence positions i and j, and channel c.

We define:

   - `M = N`: The sequence length.
   - `K = C`: The channel dimension.
   - `N_out = hidden_dim`: The output channel dimension (which is a parameter).

The core reduction is:

   out[b, i, j, d] = sum_{k=0}^{N-1} (left[b, i, k, d] * right[b, j, k, d])

for each batch b, output index i, output index j, and output channel d.

**Implementation Procedure**
We must ensure all operations are correctly aligned with the input dimensions. Here's the step-by-step procedure:

1. **Tensor Loading and Initialization**
   In the kernel, we expect two inputs (left and right) and need to create the output tensor with the same shape. The dimensions for the kernel are:
   - Global grid: (N_out, N, N) for the output's dimensions.
   - Batch dimension: B * N_out * N, but we'll treat it differently.

2. **Index Definitions**
   Define inner and outer loop indices:
   - `b_idx`: Batch index.
   - `i_idx`: Row index of the output matrix.
   - `j_idx`: Column index of the output matrix for the `...` results.
   - `k_idx`: Reduction/dimension index.

3. **Access Sizes**
   We use:
   - `BLOCK_M = 64`
   - `BLOCK_N = 64`
   - `BLOCK_K = 64`

4. **Memory Layout**
   All input tensors are stored with channel dimensions first, then sequentially by row and column. The left and right tensors are 2D arrays representing:

        left[ batch, hidden_dim*k1, row ] for k1=0 to 1? Actually, our tensors are [B, N, N, hidden_dim]. 

   We'll assume that the left and right tensors are stored in format `CHW` (channel, height, width) with contiguous dimensions along channel, width, and height.

5. **Kernel Function**
   The function signature matters. We are defining:

        @triton.jit
        def tri_mul_kernel(
            left_ptr, right_ptr, out_ptr,
            batch, n, hidden_dim,
            stride_left, stride_between_left_seq,
            stride_left_channels,
            n_outer, hidden_outer,
            BLOCK_N: tl.constexpr,   # This is BLOCK_K but used for blocking?
            BLOCK_M: tl.constexpr,   # We'll stick to 64
            BLOCK: tl.constexpr,   # We'll let it be?
        ):

   The strides:
   - `stride_left`: For batch (leading dimension).
   - `stride_between_left_seq`: For moving from one batch to the next.
   - `stride_left_channels`: For moving from row to next row (unroll in channel).

   But note: our tensors have shape [B, N, N, hidden_dim] and we are using 4D indexing. 

   Alternative approach: reorder the dimensions to [B * hidden_dim * N, N] so that the sequence length is accessible contiguously. 

   We'll do:

        left_ptr = (Batch * hidden_dim) + (i_idx * stride_inner) + k_idx

   But let's reassess:

   The operation is: 

        out[b, i, j, d] = 0 if mask[b,i,j] else sum_{k} (proj_left[b,i,k,d] * proj_right[b,j,k,d])

   We are to compute this using a blocked matrix multiplication.

   Steps:

   a. In the forward pass, we have a 4D tensor [B, N, N, hidden_dim]. 

   b. We want to compute the output [B, N, N, hidden_dim] by: 

        out[b,i,j,d] = sum_{k} (left_deg[b,i,k,d] * right_deg[b,j,k,d])

   (where left_deg and right_deg are the gated activations)

   But note: the gatings are also applied elementwise on the hidden_dim channels.

   We'll assume the following dimension layout:

        Input: [B, N, N, hidden_dim]
        We project to [B, N, N, hidden_dim] (same shape) but the reduction is triangular.

   However, the paper says "outgoing" and it's a bioinformatics problem.

   After re-examining the PyTorch code:

        x: [B, N, N, C] -> normalized to [B, N, N, C]
        left_proj(x) -> [B, N, N, hidden_dim]
        right_proj(x) -> [B, N, N, hidden_dim]

        then masked and gated.

   We'll use the big batch dimension:

        batch_size = B * hidden_dim

        and the two projected tensors are [batch_size, N, N]

   Then, we want to compute:

        out = 0.0   # initially

        for k in range(N):
            out += left_gate[b_idx, k, d_idx] * right_gate[b_idx, k, d_idx]

   We can do:

        We treat the output as needing to be computed for every j_idx (0 to N-1). 

        But note the gating is applied after the reduction, so gating in the kernel will be:

            gating_tensor = some_tensor.sigmoid()   # not gated in kernel because the gating is done on the original projection.

   Actually, the gating in the TriMul operation is applied to the original projection. 

   But we are to implement the forward pass only and not compute gradients.

   We'll follow:

        We have texture (input) and mask.
        We project to left_proj and right_proj (and also compute gating weights) using linear algebra.
