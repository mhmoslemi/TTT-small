@triton.jit
def apply_gate_mask_kernel(
    a_ptr, a_gate_ptr, b_ptr, b_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_a_b, stride_a_i, stride_a_j, stride_a_h,
    stride_g_b, stride_g_i, stride_g_j, stride_g_h,
    stride_b_b, stride_b_i, stride_b_j, stride_b_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr
):
    pid_b = tl.program_id(0) # batch dimension
    pid_i = tl.program_id(1) # i dimension tile
    pid_j = tl.program_id(2) # j dimension tile

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    # For each tile we need to handle H dimension
    offs_h = tl.arange(0, BLOCK_H)

    # Create mask for i and j within bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # broadcast
    # compute pointers
    a_ptrs = a_ptr + pid_b * stride_a_b + offs_i[:, None, None] * stride_a_i + offs_j[None, :, None] * stride_a_j + offs_h[None, None, :] * stride_a_h
    a_gate_ptrs = a_gate_ptr + pid_b * stride_g_b + offs_i[:, None, None] * stride_g_i + offs_j[None, :, None] * stride_g_j + offs_h[None, None, :] * stride_g_h

    # same for b
    b_ptrs = b_ptr + pid_b * stride_b_b + offs_i[:, None, None] * stride_b_i + offs_j[None, :, None] * stride_b_j + offs_h[None, None, :] * stride_b_h
    b_gate_ptrs = b_gate_ptr + pid_b * stride_g_b + offs_i[:, None, None] * stride_g_i + offs_j[None, :, None] * stride_g_j + offs_h[None, None, :] * stride_g_h # same stride as gate

    # load values (fp16)
    a = tl.load(a_ptrs, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0)
    a_gate = tl.load(a_gate_ptrs, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0)
    b = tl.load(b_ptrs, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0)
    b_gate = tl.load(b_gate_ptrs, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0)

    if HAS_MASK:
        mask_ptrs = mask_ptr + pid_b * stride_mask_b + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j
        mask = tl.load(mask_ptrs, mask=mask_i[:, None] & mask_j[None, :], other=0.0)  # shape (BLOCK_I, BLOCK_J)
        mask = mask[:, :, None]  # broadcast to H
        a = a * a_gate * mask
        b = b * b_gate * mask
    else:
        a = a * a_gate
        b = b * b_gate

    # store back
    tl.store(a_ptrs, a, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])
    tl.store(b_ptrs, b, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

import torch.nn.functional as F
import triton

Define kernel:

@triton.jit
def apply_gate_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    # Limit within bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Process hidden as loops
    # accumulators for left/right slices will be stored directly

    # We'll loop over hidden dim in blocks
    # output pointers for left and right
    # For each hidden block, compute multiplication and write back.

    # Outer loop over h-blocks
    h_start = 0
    while h_start < H:
        cur_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = cur_h < H

        # Compute pointers for left
        left_ptrs = left_ptr + pid_b * stride_left_b + offs_i[:, None] * stride_left_i + offs_j[None, :] * stride_left_j + cur_h[None, None, :] * stride_left_h
        left_gate_ptrs = left_gate_ptr + pid_b * stride_gate_b + offs_i[:, None] * stride_gate_i + offs_j[None, :] * stride_gate_j + cur_h[None, None, :] * stride_gate_h
        # right
        right_ptrs = right_ptr + pid_b * stride_right_b + offs_i[:, None] * stride_right_i + offs_j[None, :] * stride_right_j + cur_h[None, None, :] * stride_right_h
        right_gate_ptrs = right_gate_ptr + pid_b * stride_gate_b + offs_i[:, None] * stride_gate_i + offs_j[None, :] * stride_gate_j + cur_h[None, None, :] * stride_gate_h

        # Load
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_j[None, :] & mask_h[None, None, :], other=0.0)
        left_gate = tl.load(left_gate_ptrs, mask=mask_i[:, None] & mask_j[None, :] & mask_h[None, None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_i[:, None] & mask_j[None, :] & mask_h[None, None, :], other=0.0)
        right_gate = tl.load(right_gate_ptrs, mask=mask_i[:, None] & mask_j[None, :] & mask_h[None, None, :], other=0.0)

        if HAS_MASK:
            # load mask scalar per (i,j); broadcast to H block
            mask_ptrs = mask_ptr + pid_b * stride_mask_b + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j
            mask_val = tl.load(mask_ptrs, mask=mask_i[:, None] & mask_j[None, :], other=0.0)  # shape (BLOCK_I, BLOCK_J)
            mask_val = mask_val[:, :, None]
            left = left * left_gate * mask_val
            right = right * right_gate * mask_val
        else:
            left = left * left_gate
            right = right * right_gate

        # store back
        tl.store(left_ptrs, left, mask=mask_i[:, None] & mask_j[None, :] & mask_h[None, None, :])
        tl.store(right_ptrs, right, mask=mask_i[:, None] & mask_j[None, :] & mask_h[None, None, :])

        h_start += BLOCK_H

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,D]).
2. Linear projections to hidden dimension H:
   left = x @ W_left  ,  right = x @ W_right      (float32)
3. Compute three sigmoid gates (float32):
   left_gate  = sigmoid(x @ W_lgate)
   right_gate = sigmoid(x @ W_rgate)
   out_gate   = sigmoid(x @ W_outgate)
4. Cast all tensors to fp16 and fuse the element‑wise
   multiplication `left *= left_gate` (and optional mask)
   `right *= right_gate` (and optional mask) in a Triton kernel.
   The kernel processes a 3‑D tile (i, j, hidden) and loops over
   the hidden dimension in blocks.
5. Perform the heavy O(N³·H) contraction as a batched matrix
   multiplication:
       out_ijh = Σ_k left_ik h * right_jk h   (i.e. left @ rightᵀ)
   using torch.bmm on the reshaped tensors (fp16 inputs,
   fp32 accumulation by default in cuBLAS).
6. Reshape back to [B,N,N,H] and apply a second LayerNorm.
7. Multiply by `out_gate` (float32) and a final linear projection
   back to the original dimension D.
8. Return the result [B,N,N,D] (float32).

The Triton kernel fuses the gate application and optional mask,
reducing intermediate memory traffic and guaranteeing at least
part of the computation runs on custom GPU code.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse left/right gates with optional mask.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,
    # strides for left/right tensors (shape B,N,N,H)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    # strides for mask (shape B,N,N)
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr
):
    pid_b = tl.program_id(0)               # batch dimension
    pid_i = tl.program_id(1)               # i‑tile
    pid_j = tl.program_id(2)               # j‑tile

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Loop over hidden dimension in blocks
    for h in range(0, H, BLOCK_H):
        cur_h = h + tl.arange(0, BLOCK_H)
        mask_h = cur_h < H

        # Pointers for left and its gate
        left_ptrs = left_ptr + pid_b * stride_left_b \
                    + offs_i[:, None, None] * stride_left_i \
                    + offs_j[None, :, None] * stride_left_j \
                    + cur_h[None, None, :]   * stride_left_h
        left_gate_ptrs = left_gate_ptr + pid_b * stride_gate_b \
                         + offs_i[:, None, None] * stride_gate_i \
                         + offs_j[None, :, None] * stride_gate_j \
                         + cur_h[None, None, :]   * stride_gate_h

        # Pointers for right and its gate
        right_ptrs = right_ptr + pid_b * stride_right_b \
                     + offs_i[:, None, None] * stride_right_i \
                     + offs_j[None, :, None] * stride_right_j \
                     + cur_h[None, None, :]   * stride_right_h
        right_gate_ptrs = right_gate_ptr + pid_b * stride_gate_b \
                          + offs_i[:, None, None] * stride_gate_i \
                          + offs_j[None, :, None] * stride_gate_j \
                          + cur_h[None, None, :]   * stride_gate_h

        # Load tensors (fp16)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                       other=0.0)
        left_gate = tl.load(left_gate_ptrs,
                            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                            other=0.0)
        right = tl.load(right_ptrs,
                        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                        other=0.0)
        right_gate = tl.load(right_gate_ptrs,
                             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                             other=0.0)

        if HAS_MASK:
            # Load mask (scalar per (i,j)), broadcast to hidden block
            mask_ptrs = mask_ptr + pid_b * stride_mask_b \
                        + offs_i[:, None] * stride_mask_i \
                        + offs_j[None, :] * stride_mask_j
            m = tl.load(mask_ptrs,
                        mask=mask_i[:, None] & mask_j[None, :],
                        other=0.0)               # shape (BLOCK_I, BLOCK_J)
            m = m[:, :, None]                   # broadcast to (i,j,hidden)
            left = left * left_gate * m
            right = right * right_gate * m
        else:
            left = left * left_gate
            right = right * right_gate

        # Store back the fused results
        tl.store(left_ptrs,
                 left,
                 mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])
        tl.store(right_ptrs,
                 right,
                 mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, D] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (see code)
            - config: dict containing "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False) and mask is not None

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,D), float32

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])        # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()   # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 4. Cast to fp16 for the heavy contraction
    # --------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    # --------------------------------------------------------------
    # 5. Fuse gate application (and optional mask) via Triton
    # --------------------------------------------------------------
    # Ensure tensors are contiguous for correct stride interpretation
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    if use_mask:
        mask_f16 = mask.to(torch.float16).contiguous()
    else:
        # dummy tensor – never accessed when HAS_MASK=0
        mask_f16 = torch.empty(1, dtype=torch.float16, device=device)

    # Compile‑time tile sizes (tuned for B200)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (bs,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _gate_mask_kernel[grid](
        left_ptr=left,
        left_gate_ptr=left_gate,
        right_ptr=right,
        right_gate_ptr=right_gate,
        mask_ptr=mask_f16,
        B=bs,
        N=N,
        H=hidden_dim,
        stride_left_b=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_j=left.stride(2),
        stride_left_h=left.stride(3),
        stride_gate_b=left_gate.stride(0),
        stride_gate_i=left_gate.stride(1),
        stride_gate_j=left_gate.stride(2),
        stride_gate_h=left_gate.stride(3),
        stride_right_b=right.stride(0),
        stride_right_i=right.stride(1),
        stride_right_j=right.stride(2),
        stride_right_h=right.stride(3),
        stride_mask_b=mask_f16.stride(0) if use_mask else 0,
        stride_mask_i=mask_f16.stride(1) if use_mask else 0,
        stride_mask_j=mask_f16.stride(2) if use_mask else 0,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        HAS_MASK=1 if use_mask else 0,
    )

    # --------------------------------------------------------------
    # 6. Batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Perform as a batched matmul (left @ rightᵀ) on fp16 inputs.
    # --------------------------------------------------------------
    # Reshape to (B*H, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)

    # torch.bmm uses cuBLAS; accumulation is in fp32 when inputs are fp16
    out_mat = torch.bmm(left_mat, right_mat)          # (B*H, N, N), fp16
    out = out_mat.to(torch.float32)                   # cast back to fp32 for next layers

    # --------------------------------------------------------------
    # 7. Restore original 4‑D layout
    # --------------------------------------------------------------
    out = out.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 8. Output LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 9. Output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 10. Final linear projection back to original dimension
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
