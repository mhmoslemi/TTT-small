@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Compute program ids
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out of bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointer offsets
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr \
                 + offs_m[:, None] * stride_a_row \
                 + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B block transposed: (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr \
                 + cur_k[:, None] * stride_b_col \
                 + offs_n[None, :] * stride_b_row
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch_ptr \
             + offs_m[:, None] * stride_c_row \
             + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul outgoing variant (AlphaFold3) – Triton accelerated contraction.

The heavy O(N³) operation
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is rewritten as a batched matrix multiplication
    C = A @ Bᵀ
for each (batch, hidden) slice, where A and B have shape (N, N) and the
result C has shape (N, N).  The kernel processes one (batch × hidden) matrix
at a time using a tiled GEMM (BLOCK_M × BLOCK_N tiles, BLOCK_K accumulation)
with fp16 inputs and fp32 accumulation.  All other TriMul components
(LayerNorms, linear projections, gating) are performed in PyTorch.

The implementation supports optional masking, arbitrary batch size (1‑2),
sequence lengths up to 1024 and hidden dimensions of 128.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ.

    A: (batch, M, K)  – fp16
    B: (batch, N, K)  – fp16
    C: (batch, M, N)  – fp32 (accumulator)
    """
    pid_batch = tl.program_id(0)      # index over (batch * hidden_dim)
    pid_m = tl.program_id(1)          # tile over M (rows)
    pid_n = tl.program_id(2)          # tile over N (cols)

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K steps
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block (BLOCK_M, BLOCK_K)
        a_offs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_offs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B block transposed on‑the‑fly (BLOCK_K, BLOCK_N)
        b_offs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_offs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # GEMM accumulate
        acc += tl.dot(a, b)

    # Write the result
    c_offs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_offs, acc,
             mask=mask_m[:, None] & mask_n[None, :])

    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----- LayerNorm on input -----
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ----- Linear projections (dim -> hidden_dim) -----
    left = F.linear(x, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----- Optional mask (applied before gating) -----
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----- Gating (sigmoid) -----
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ----- Apply gates -----
    left = left * left_gate
    right = right * right_gate

    # ----- Cast to FP16 for the contraction -----
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape   # B, seq_len, seq_len, hidden_dim (H == hidden_dim)

    # Reshape to (B*H, N, N) for batched GEMM
    left_mat = left_h.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (fp32 accumulator)
    out_mat = torch.empty((B * H, N, N), device=device, dtype=torch.float32)

    # ----- Tiling parameters -----
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides in number of elements (contiguous layout)
    stride_a_batch = left_mat.stride(0)
    stride_a_row   = left_mat.stride(1)
    stride_a_col   = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_row   = right_mat.stride(1)
    stride_b_col   = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_row   = out_mat.stride(1)
    stride_c_col   = out_mat.stride(2)

    # Grid: (batch*hidden, tiles over M, tiles over N)
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch kernel
    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    # Ensure completion before next PyTorch ops
    torch.cuda.synchronize()

    # ----- Reshape back to [B, N, N, hidden_dim] -----
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # ----- Output LayerNorm (over hidden_dim) -----
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    # ----- Apply output gate -----
    out = out * out_gate

    # ----- Final linear projection (hidden_dim -> dim) -----
    out = F.linear(out, to_out_weight)

    return out
