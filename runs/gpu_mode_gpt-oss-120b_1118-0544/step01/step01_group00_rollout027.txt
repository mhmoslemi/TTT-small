"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm over the channel dimension of the input.
2. Linear projections to a hidden dimension (no bias).  Both projections are
   computed in FP16 for bandwidth efficiency.
3. Optional mask (broadcast over the hidden dimension) applied to the projected
   tensors.
4. Gating vectors (sigmoid of linear projections) are computed in FP32.
5. A tiny Triton kernel fuses the element‑wise product
        fused = (proj * gate)   (with mask already applied)
   and casts the result back to FP16.
6. The core reduction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix multiplication C = A @ Bᵀ where each batch corresponds to
   one (batch × hidden_dim) slice.  This is performed with torch.bmm on FP16
   tensors (leveraging Tensor‑cores).
7. The result is reshaped back to [B, N, N, hidden_dim] and converted to FP32.
8. Layer‑norm over the hidden dimension, gating with `out_gate`, and a final
   linear projection back to the original channel dimension produce the output.

Only the element‑wise fusion step (5) is implemented in Triton; the heavy
batched GEMM is delegated to cuBLAS via torch.bmm for maximal performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse projection with its gate (and optional mask already applied)
# ----------------------------------------------------------------------
@triton.jit
def fuse_proj_gate_kernel(
    proj_ptr,          # FP16 projected tensor (flattened)
    gate_ptr,          # FP16 gate tensor (flattened) – will be cast to FP32
    out_ptr,           # FP16 output tensor (flattened)
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load projection (FP16) and cast to FP32 for multiplication
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    # Load gate (stored as FP16) and cast to FP32
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0).to(tl.float32)

    # Elementwise product, cast back to FP16
    out = (proj * gate).to(tl.float16)

    tl.store(out_ptr + offs, out, mask=mask)


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float) – may be ignored
            - weights: dict of model weights (see skeleton)
            - config: dict containing at least:
                * "dim" (int)
                * "hidden_dim" (int)
                * "nomask" (bool) – if True the mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (FP16 for bandwidth)
    # ------------------------------------------------------------------
    # Cast normalized input and weights to half precision
    x_half = x_norm.to(torch.float16)

    left_proj_weight_h = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight_h = weights["right_proj.weight"].to(torch.float16)

    left_proj = F.linear(x_half, left_proj_weight_h)   # [B, N, N, hidden_dim], FP16
    right_proj = F.linear(x_half, right_proj_weight_h)  # same shape

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; convert to half and broadcast
        mask_h = mask.to(torch.float16).unsqueeze(-1)          # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 4. Gating vectors (computed in FP32 for numerical stability)
    # ------------------------------------------------------------------
    # Gates are based on the *original* normalized tensor (pre‑mask)
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))   # [B,N,N,hidden_dim], FP32
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))    # used later

    # Cast gates to FP16 to fit the fusion kernel (still safe after sigmoid)
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Fuse projection + gate (and cast back to FP16) via Triton
    # ------------------------------------------------------------------
    total_elements = left_proj.numel()  # B * N * N * hidden_dim
    BLOCK_SIZE = 1024

    # Prepare flattened tensors
    left_proj_flat = left_proj.contiguous().view(-1)
    left_gate_flat = left_gate_h.contiguous().view(-1)
    left_fused_flat = torch.empty_like(left_proj_flat, dtype=torch.float16)

    right_proj_flat = right_proj.contiguous().view(-1)
    right_gate_flat = right_gate_h.contiguous().view(-1)
    right_fused_flat = torch.empty_like(right_proj_flat, dtype=torch.float16)

    # Grid for 1‑D launch
    grid = ( (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    # Launch fusion kernels
    fuse_proj_gate_kernel[grid](
        left_proj_flat,
        left_gate_flat,
        left_fused_flat,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    fuse_proj_gate_kernel[grid](
        right_proj_flat,
        right_gate_flat,
        right_fused_flat,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # Reshape back to [B, N, N, hidden_dim]
    left_fused = left_fused_flat.view_as(left_proj)   # FP16
    right_fused = right_fused_flat.view_as(right_proj)

    # ------------------------------------------------------------------
    # 6. Core reduction: batched matmul C = A @ Bᵀ  (FP16 → FP32 accumulation)
    # ------------------------------------------------------------------
    B, N, _, _ = left_fused.shape
    # Rearrange to [B, hidden, N, N] then merge batch & hidden
    left_perm = left_fused.permute(0, 3, 1, 2).contiguous()    # [B, hidden, N, N]
    right_perm = right_fused.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * hidden_dim, N, N)          # [B*hidden, N, N]
    right_mat = right_perm.view(B * hidden_dim, N, N)        # same

    # Batched matrix multiply (half‑precision, Tensor‑core accelerated)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*hidden, N, N], FP16

    # Reshape back to [B, N, N, hidden_dim] and convert to FP32 for subsequent ops
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 7. Final LayerNorm, gating and projection back to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )  # [B, N, N, hidden_dim], FP32

    out = out * out_gate  # apply output gate (both FP32)

    # Final linear projection to `dim` (no bias)
    out = F.linear(out, weights["to_out.weight"])

    return out
