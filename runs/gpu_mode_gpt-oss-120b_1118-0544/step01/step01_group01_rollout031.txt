We fuse the mask and gating step into a small Triton kernel and perform
the batched GEMM with PyTorch's cuBLAS (highly tuned, tensor‑core capable).
All other operations (LayerNorms, linear projections) stay in PyTorch.
The final result is a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr,            # [B,N,N,H] contiguous half tensors
    left_gate_ptr, right_gate_ptr,  # same shape, half
    mask_ptr,                       # [B,N,N] half, broadcast over H
    total_elems, hidden_dim,        # scalars
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    `mask` is broadcasted over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # linear offsets
    valid = offs < total_elems

    # load data
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # compute mask index (mask is H‑wise broadcast)
    mask_idx = offs // hidden_dim                               # integer division
    mask_len = total_elems // hidden_dim                         # B*N*N
    mask_valid = mask_idx < mask_len
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    # apply
    left = left * left_gate * mask_val
    right = right * right_gate * mask_val

    # store back in‑place
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict with the tensors described in the prompt
            - config: dict containing at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, cfg = data
    dim, hidden_dim = cfg["dim"], cfg["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), norm_w, norm_b)

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]    # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]   # (hidden_dim, dim)

    left  = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 3) Gates (sigmoid)
    # -----------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 4) Fuse mask and gates with Triton (half‑precision)
    # -----------------------------------------------------------------
    # Cast everything to fp16 for the kernel
    left      = left.to(torch.float16).contiguous()
    right     = right.to(torch.float16).contiguous()
    left_gate = left_gate.to(torch.float16).contiguous()
    right_gate = right_gate.to(torch.float16).contiguous()

    if mask is None:
        mask_fp16 = torch.ones(B, N, N, dtype=torch.float16, device=left.device)
    else:
        mask_fp16 = mask.to(torch.float16).contiguous()

    total_elems = left.numel()                     # B * N * N * hidden_dim
    BLOCK_SIZE = 1024
    grid = ( (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    _gate_mask_fuse_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        left_gate.data_ptr(),
        right_gate.data_ptr(),
        mask_fp16.data_ptr(),
        total_elems,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # left_gate / right_gate no longer needed – let them be freed
    del left_gate, right_gate, mask_fp16

    # -----------------------------------------------------------------
    # 5) Batched matrix multiplication (fp16) using cuBLAS via torch.bmm
    #    left  : [B, N, N, H] -> (i,k)
    #    right : [B, N, N, H] -> (j,k) -> we need (k,j) => transpose last two dims
    # -----------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # batched GEMM (tensor‑core fp16)
    out_mat = torch.bmm(left_mat, right_mat)       # (B*H, N, N), fp16

    # reshape back to (B, N, N, H) and cast to fp32 for the rest
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, output gate and final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_hidden = F.layer_norm(out_hidden, (hidden_dim,), to_out_norm_w, to_out_norm_b)

    # apply output gate (broadcasted over last dim)
    out_gate = out_gate.to(torch.float32)
    out_hidden = out_hidden * out_gate

    # final linear projection back to `dim`
    to_out_w = weights["to_out.weight"]            # (dim, hidden_dim)
    out = F.linear(out_hidden, to_out_w)           # [B, N, N, dim]

    return out
