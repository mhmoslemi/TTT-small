def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps = 1e-5

    # layer norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Mask appropriate
    if mask is None:
        mask_f = torch.ones((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]), dtype=dtype, device=device)
    else:
        mask_f = mask.to(dtype=dtype)

    # Prepare mask for broadcast (unsqueeze dim)
    mask_f = mask_f.unsqueeze(-1)  # shape [B,N,N,1]
    # By default keep as contiguous
    # Ensure memory contiguous for all
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_f = mask_f.contiguous()

    # Allocate output tensors
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Launch fused kernel
    B, N, _, _ = left_proj.shape
    # block sizes
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32  # hidden dim chunk
    grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)

    # get strides in elements
    lp_s_b, lp_s_i, lp_s_j, lp_s_h = left_proj.stride()
    lg_s_b, lg_s_i, lg_s_j, lg_s_h = left_gate.stride()
    rp_s_b, rp_s_i, rp_s_j, rp_s_h = right_proj.stride()
    rg_s_b, rg_s_i, rg_s_j, rg_s_h = right_gate.stride()
    mask_s_b, mask_s_i, mask_s_j = mask_f.stride()
    out_s_b, out_s_i, out_s_j, out_s_h = left.stride()  # same as left_proj

    fused_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask_f,
        left,
        right,
        B,
        N,
        hidden_dim,
        lp_s_b, lp_s_i, lp_s_j, lp_s_h,
        lg_s_b, lg_s_i, lg_s_j, lg_s_h,
        rp_s_b, rp_s_i, rp_s_j, rp_s_h,
        rg_s_b, rg_s_i, rg_s_j, rg_s_h,
        mask_s_b, mask_s_i, mask_s_j,
        out_s_b, out_s_i, out_s_j, out_s_h, 
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # Now compute core multiplication using torch.bmm
    # Rearrange
    left_mat = left.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    right_mat = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch and hidden dim
    B, H, N, _ = left_mat.shape
    left_flat = left_mat.view(B * H, N, N)
    right_flat = right_mat.view(B * H, N, N)

    # Compute out (batch matmul) using torch.bmm
    out_mat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # shape [B*H, N, N]

    # Reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Gate multiplication
    out = out * out_gate

    # Final linear projection
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output

"""
Custom Triton‑accelerated implementation of the outgoing Triangle Multiplicative
Update (TriMul) used in AlphaFold3 and related models.

The heavy O(N³) contraction is expressed as a batched matrix multiplication
(A @ Bᵀ) and executed with PyTorch’s highly‑optimized cuBLAS (torch.bmm).  To
satisfy the requirement of using at least one Triton kernel we fuse the
projection, masking and gating steps into a single elementwise kernel.  This
kernel writes the gated tensors `left` and `right` that are later multiplied.

The overall algorithm follows the reference PyTorch implementation:

1. Layer‑norm the input.
2. Linear projection to a hidden dimension (left_proj, right_proj).
3. Compute three sigmoid gates (left_gate, right_gate, out_gate).
4. Fuse mask, projection and gates with a Triton kernel ⇒ `left`, `right`.
5. Compute out = Σ_k left[..., i, k, d] * right[..., j, k, d] via a
   batched matrix multiplication (torch.bmm).
6. Layer‑norm the result, multiply by `out_gate`, and project back to the
   original dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_gate_mask_kernel(
    # Input tensors
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    # Output tensors
    left_out_ptr, right_out_ptr,
    # Shapes
    B, N, H,
    # Strides (in elements)
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # Compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # Masks for out‑of‑bounds i / j
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]          # (BLOCK_I, BLOCK_J)

    # ----- Load the mask (shape [B,N,N]) and broadcast later -----
    base_mask = mask_ptr + pid_b * stride_mask_b
    mask_tile = tl.load(
        base_mask + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j,
        mask=mask_ij,
        other=1.0,
    )                       # (BLOCK_I, BLOCK_J)
    mask_tile = mask_tile[:, :, None]                    # broadcast to (BLOCK_I,BLOCK_J,1)

    # Base pointers for the four input tensors (no hidden dim yet)
    base_lp = left_proj_ptr + pid_b * stride_lp_b
    base_lg = left_gate_ptr  + pid_b * stride_lg_b
    base_rp = right_proj_ptr + pid_b * stride_rp_b
    base_rg = right_gate_ptr + pid_b * stride_rg_b

    # Base pointers for the outputs
    base_lo = left_out_ptr  + pid_b * stride_out_b
    base_ro = right_out_ptr + pid_b * stride_out_b

    # Number of hidden‑dim tiles
    num_h_tiles = tl.cdiv(H, BLOCK_H)

    for t in range(num_h_tiles):
        h_start = t * BLOCK_H
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Pointers to the current hidden‑dim slice
        lp_ptrs = (
            base_lp
            + offs_i[:, None, None] * stride_lp_i
            + offs_j[None, :, None] * stride_lp_j
            + offs_h[None, None, :] * stride_lp_h
        )
        lg_ptrs = (
            base_lg
            + offs_i[:, None, None] * stride_lg_i
            + offs_j[None, :, None] * stride_lg_j
            + offs_h[None, None, :] * stride_lg_h
        )
        rp_ptrs = (
            base_rp
            + offs_i[:, None, None] * stride_rp_i
            + offs_j[None, :, None] * stride_rp_j
            + offs_h[None, None, :] * stride_rp_h
        )
        rg_ptrs = (
            base_rg
            + offs_i[:, None, None] * stride_rg_i
            + offs_j[None, :, None] * stride_rg_j
            + offs_h[None, None, :] * stride_rg_h
        )

        # Load with proper masking
        mask_load = mask_ij[:, :, None] & mask_h[None, None, :]
        lp = tl.load(lp_ptrs, mask=mask_load, other=0.0)
        lg = tl.load(lg_ptrs, mask=mask_load, other=0.0)
        rp = tl.load(rp_ptrs, mask=mask_load, other=0.0)
        rg = tl.load(rg_ptrs, mask=mask_load, other=0.0)

        # Apply gates and mask
        left_val  = lp * lg * mask_tile
        right_val = rp * rg * mask_tile

        # Store the results
        lo_ptrs = (
            base_lo
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        ro_ptrs = (
            base_ro
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(lo_ptrs, left_val,  mask=mask_load)
        tl.store(ro_ptrs, right_val, mask=mask_load)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N]  (may be None)
        - weights      : dict of pre‑loaded model parameters
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Broadcast mask (if missing create all‑ones)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=dtype,
            device=device,
        )
    mask = mask.to(dtype=dtype)          # [B, N, N]

    # ------------------------------------------------------------------
    # 5. Fuse projection, mask and gating with a Triton kernel
    # ------------------------------------------------------------------
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask.contiguous()

    B, N, _, _ = left_proj.shape
    H = hidden_dim

    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    # Tile sizes – tuned for typical B200 resources
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_H = 32  # hidden‑dim chunk

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    # Strides (elements, not bytes)
    lp_s = left_proj.stride()
    lg_s = left_gate.stride()
    rp_s = right_proj.stride()
    rg_s = right_gate.stride()
    mask_s = mask.stride()
    out_s = left_out.stride()

    fused_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        B,
        N,
        H,
        lp_s[0], lp_s[1], lp_s[2], lp_s[3],
        lg_s[0], lg_s[1], lg_s[2], lg_s[3],
        rp_s[0], rp_s[1], rp_s[2], rp_s[3],
        rg_s[0], rg_s[1], rg_s[2], rg_s[3],
        mask_s[0], mask_s[1], mask_s[2],
        out_s[0], out_s[1], out_s[2], out_s[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6. Core O(N³) contraction via batched matrix multiplication
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Rearrange to [B, hidden, N, N] and flatten batch*hidden
    left_mat = left_out.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat = right_out.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_mat.shape
    left_flat = left_mat.view(B * H, N, N)
    right_flat = right_mat.view(B * H, N, N)

    # batched matmul: A @ Bᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # [B*H, N, N]

    # Reshape back to [B, N, N, hidden]
    out = (
        out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
