"""
TriMul outgoing update (AlphaFold3) – Triton‑accelerated core.

The module computes:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are linear projections of the normalized pair
representation, masked and gated.  The triple sum is a batch of matrix
multiplications A @ Bᵀ for each hidden channel, which we implement with a
custom Triton kernel that tiles the N×N output and iterates over K in
blocks.  All surrounding LayerNorm/Linear/Gating steps use native PyTorch.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _trigemul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_a_batch, stride_am, stride_ak,  # A strides: (batch, M, K)
    stride_b_batch, stride_bk, stride_bn,  # B strides: (batch, K, N)  (B already transposed)
    stride_c_batch, stride_cm, stride_cn,  # C strides: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ B  where
       A : [batch, M, K]   (float16)
       B : [batch, K, N]   (float16, already transposed from original right)
       C : [batch, M, N]   (float32 accumulator, stored as float32)
    """
    pid_batch = tl.program_id(2)      # batch index (B * hidden_dim)
    pid_m = tl.program_id(0)          # tile row index
    pid_n = tl.program_id(1)          # tile col index

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # OOB masks
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Base pointers for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    # Pointers to current block of A and B
    a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in chunks of BLOCK_K
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # tl.dot returns FP16; cast to FP32 before accumulation
        acc += tl.dot(a, b).to(tl.float32)
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the result tile
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative
    Update (TriMul).

    Parameters
    ----------
    data : tuple
        (pair_tensor, mask_tensor, weights_dict, config_dict)
        - pair_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : mapping from string names to torch.Tensor weights
        - config_dict  : contains "dim" (C) and "hidden_dim" (H)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] after the TriMul update (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    pair_tensor, mask, weights, config = data
    device = pair_tensor.device
    B, N, N2, C = pair_tensor.shape
    assert N == N2, "Input must be square in the sequence dimensions."
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    pair_norm = torch.nn.functional.layer_norm(
        pair_tensor,
        (C,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(pair_norm, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(pair_norm, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional masking
    # ------------------------------------------------------------------
    if mask is None:
        mask_val = torch.ones((B, N, N), dtype=pair_tensor.dtype, device=device)
    else:
        mask_val = mask.to(dtype=pair_tensor.dtype, device=device)
    mask_val = mask_val.unsqueeze(-1)   # [B, N, N, 1]

    left = left * mask_val
    right = right * mask_val

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(pair_norm, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(pair_norm, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(pair_norm, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul kernel (batched matmul A @ Bᵀ)
    # ------------------------------------------------------------------
    # Transform to [B, hidden_dim, N, N] where last two axes are (i,k) and (j,k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, H, i, k]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, j, k]
    # Transpose right to bring k as leading axis for the GEMM
    right_T = right_perm.transpose(2, 3).contiguous()   # [B, H, k, j]

    # Merge batch and hidden_dim for a single batched GEMM
    B_H = B * hidden_dim
    left_flat = left_perm.reshape(B_H, N, N).to(torch.float16)
    right_T_flat = right_T.reshape(B_H, N, N).to(torch.float16)

    # Output buffer (FP32 accumulator)
    out_flat = torch.empty((B_H, N, N), dtype=torch.float32, device=device)

    # Triton launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B_H,
    )

    _trigemul_matmul_kernel[grid](
        left_flat,
        right_T_flat,
        out_flat,
        N,        # M
        N,        # N
        N,        # K
        # Strides for A (left)
        left_flat.stride(0),   # batch stride
        left_flat.stride(1),   # M stride (i)
        left_flat.stride(2),   # K stride (k)
        # Strides for B (right_T)
        right_T_flat.stride(0),  # batch stride
        right_T_flat.stride(1),  # K stride (k)
        right_T_flat.stride(2),  # N stride (j)
        # Strides for C (output)
        out_flat.stride(0),    # batch stride
        out_flat.stride(1),    # M stride
        out_flat.stride(2),    # N stride
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Post‑processing: hidden‑dim LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # back to original dim

    return out
