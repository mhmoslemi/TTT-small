        right = right * mask_f

    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Reshape
    B,N,_,_ = left.shape
    left = left.permute(0,3,1,2).contiguous()  # [B, H, N, N]
    right = right.permute(0,3,1,2).contiguous()  # [B, H, N, N]

    right_flat = right.view(batch_hidden, N, N)

    # cast to half for speed

    right_flat_h = right_flat.half()

    # MatMul: left_flat_h @ right_flat_h^T
    out_flat_h = torch.bmm(left_flat_h, right_flat_h.transpose(1,2))   # shape (batch_hidden, N, N)

    # Cast back to float32
    out_flat = out_flat_h.float()

    # Reshape back to [B,N,N,H]
    out = out_flat.view(B, hidden, N, N).permute(0,2,3,1).contiguous()

    # Second LayerNorm
    out = torch.nn.functional.layer_norm(out, (hidden,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # Apply out_gate via Triton kernel
    out_gate = out_gate  # shape B,N,N,H same as out
    out = elementwise_mul_triton(out, out_gate)  # use custom kernel

    # Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input (C‑dim).
2. Two linear projections (left / right) to hidden_dim.
3. Optional mask is applied before gating.
4. Three gating projections (left, right, out) with sigmoid.
5. Fuse gates with the projections.
6. Reshape (B,N,N,hidden) → (B*hidden,N,N) and cast to FP16.
7. Use a cuBLAS batched GEMM (torch.bmm) to compute
      out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
   which is exactly the Einstein‑summation in the reference.
8. Cast the result back to FP32, reshape to (B,N,N,hidden).
9. Second LayerNorm over the hidden dimension.
10. Apply the output gate with a tiny Triton element‑wise kernel.
11. Final linear projection back to the original dimension.
All heavy arithmetic is performed in FP16 (Tensor‑Core) while the
surrounding operations stay in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, N, BLOCK_SIZE: tl.constexpr):
    """Element‑wise multiply a and b (both 1‑D contiguous tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton kernel for a * b (both of same shape)."""
    assert a.shape == b.shape, "Shapes must match for elementwise multiply"
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](
        a, b, out, size,
        BLOCK_SIZE=BLOCK,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/int) or None
            - weights: dict with the model weights (all float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"]) # (B,N,N,hidden)

    # -----------------------------------------------------------------
    # 3. Optional mask (mask is broadcast on the hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Reshape for batched GEMM: (B, N, N, H) -> (B*H, N, N)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch_hidden = B * hidden_dim
    left_flat = left.view(batch_hidden, N, N)      # (B*H, N, N)
    right_flat = right.view(batch_hidden, N, N)    # (B*H, N, N)

    # -----------------------------------------------------------------
    # 6. Cast to FP16 and perform batched GEMM with cuBLAS (torch.bmm)
    #    out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
    # -----------------------------------------------------------------
    left_fp16 = left_flat.half()
    right_fp16 = right_flat.half()
    # right needs to be transposed on the reduction axis
    out_fp16 = torch.bmm(left_fp16, right_fp16.transpose(1, 2))
    out_fp32 = out_fp16.float()

    # -----------------------------------------------------------------
    # 7. Restore original layout (B,N,N,hidden)
    # -----------------------------------------------------------------
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8. Second LayerNorm over hidden dimension
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -----------------------------------------------------------------
    # 9. Apply the output gate with a small Triton kernel
    # -----------------------------------------------------------------
    out = _elemwise_mul_triton(out, out_gate)

    # -----------------------------------------------------------------
    # 10. Final linear projection back to original dimension
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
