    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # 1. LayerNorm
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # 2. Linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # 3. Mask

        right = right * mask

@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ah, stride_bh, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # batch*head
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointer for this batch/head
    a_ptrs = a_ptr + pid * stride_ah
    b_ptrs = b_ptr + pid * stride_bh
    c_ptrs = c_ptr + pid * stride_ch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # masks for out-of-bounds
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # pointers
        a_tile_ptr = a_ptrs + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_tile_ptr = b_ptrs + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # loads
        a = tl.load(a_tile_ptr, mask=a_mask, other=0.0)
        b = tl.load(b_tile_ptr, mask=b_mask, other=0.0)

        # dot
        acc += tl.dot(a, b)

    # store
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_tile_ptr = c_ptrs + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr, acc, mask=c_mask)

TriMul (outgoing) forward pass using Triton for the N³ reduction.

Algorithm
---------
1. Layer‑norm the input on the last dimension (dim).
2. Linear projections `left` and `right` (dim → hidden_dim), optional mask,
   and element‑wise gates (sigmoid of linear layers).
3. Fuse the batched matrix multiplication
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   as a GEMM per (b,d) using a custom Triton kernel.
   The tensors are reshaped to (B*H, N, N) and the kernel computes
   A @ B (FP16 inputs, FP32 accumulation).
4. Layer‑norm on the hidden dimension, apply the output gate, and a final
   linear projection back to `dim`.

The Triton kernel is a standard tiled GEMM (BLOCK_M=128, BLOCK_N=128,
BLOCK_K=32) that uses `tl.dot` (Tensor‑cores) and accumulates in FP32.

    """
    Batched GEMM:
        C[batch, i, j] = Σₖ A[batch, i, k] * B[batch, k, j]
    All tensors are row‑major.  A and B are FP16, the accumulator is FP32.
    """
    pid_batch = tl.program_id(0)  # collapsed batch*head dimension
    pid_m = tl.program_id(1)      # block row
    pid_n = tl.program_id(2)      # block column

    # Offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch/head
    a_batch_ptr = a_ptr + pid_batch * stride_ah
    b_batch_ptr = b_ptr + pid_batch * stride_bh
    c_batch_ptr = c_ptr + pid_batch * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out-of-bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Pointers to the current tile
        a_tile_ptr = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_tile_ptr = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (FP16)
        a = tl.load(a_tile_ptr, mask=a_mask, other=0.0)
        b = tl.load(b_tile_ptr, mask=b_mask, other=0.0)

        # Rank‑K update
        acc += tl.dot(a, b)

    # Write back the result (FP32)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_tile_ptr = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of torch.Tensor containing model parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden_dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).type_as(x)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid of bias‑free linear layers)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Cast to FP16 for the GEMM
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # -----------------------------------------------------------------
    # 6. Prepare tensors for the batched GEMM
    #    We need the multiplication left[i,k] * right[k,j],
    #    i.e. left @ right where right is already transposed.
    # -----------------------------------------------------------------
    #   left : (B, N, N, H) -> (B, H, N, N)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    #   right: (B, N, N, H) -> (B, H, N, N) with K and N swapped
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape

    # Collapse the batch & head dimensions
    left_mat = left_perm.view(B * H, N, N)    # (B*H, M, K)
    right_mat = right_perm.view(B * H, N, N)  # (B*H, K, N)

    # Output buffer (FP32 accumulator)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=x.device)

    # -----------------------------------------------------------------
    # 7. Launch the Triton GEMM kernel
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                                     # batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,              # M blocks
        (N + BLOCK_N - 1) // BLOCK_N,              # N blocks
    )

    _tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(1), left_mat.stride(2),   # stride_am, stride_ak
        right_mat.stride(1), right_mat.stride(2), # stride_bk, stride_bn
        out_mat.stride(1), out_mat.stride(2),     # stride_cm, stride_cn
        left_mat.stride(0), right_mat.stride(0), out_mat.stride(0),  # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 8. Reshape back to [B, N, N, hidden_dim]
    # -----------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 9. Output LayerNorm, output gate, and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
