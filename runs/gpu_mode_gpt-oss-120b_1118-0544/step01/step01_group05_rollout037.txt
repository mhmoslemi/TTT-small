@triton.jit
def tri_mul_kernel(
    lhs_ptr, rhs_ptr, out_ptr,
    B, M, N, K, D,  # all sizes
    stride_lhs_batch, stride_lhs_i, stride_lhs_k, stride_lhs_d,
    stride_rhs_batch, stride_rhs_i, stride_rhs_k, stride_rhs_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for i and j tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # Base pointers for this batch
    lhs_batch_ptr = lhs_ptr + pid_batch * stride_lhs_batch
    rhs_batch_ptr = rhs_ptr + pid_batch * stride_rhs_batch
    out_batch_ptr = out_ptr + pid_batch * stride_out_batch

    for d_tile in range(0, tl.cdiv(D, BLOCK_D)):
        # Determine d indices
        cur_d = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = cur_d < D

        # Accumulator for D tile
        acc = tl.zeros((BLOCK_D, BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
            cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = cur_k < K

            # Load A block shape (BLOCK_M, BLOCK_K, BLOCK_D)
            a_ptrs = lhs_batch_ptr \
                + offs_i[:, None, None] * stride_lhs_i \
                + cur_k[None, :, None] * stride_lhs_k \
                + cur_d[None, None, :] * stride_lhs_d

            a = tl.load(a_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

            # Load B block transposed shape (BLOCK_K, BLOCK_N, BLOCK_D)
            b_ptrs = rhs_batch_ptr \
                + cur_k[:, None, None] * stride_rhs_k \
                + offs_j[None, :, None] * stride_rhs_i \
                + cur_d[None, None, :] * stride_rhs_d

            b = tl.load(b_ptrs, mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :], other=0.0)

            # Compute dot for each d
            for d_idx in range(0, BLOCK_D):
                acc[d_idx] += tl.dot(a[:, :, d_idx], b[:, :, d_idx])

        # Store results for each d
        for d_idx in range(0, BLOCK_D):
            # mask for valid d
            mask_store = mask_i[:, None] & mask_j[None, :] & mask_d[d_idx]

            # compute offset for this d
            d_off = cur_d[d_idx]  # scalar offset
            out_ptrs = out_batch_ptr \
                + offs_i[:, None] * stride_out_i \
                + offs_j[None, :] * stride_out_j \
                + d_off * stride_out_d

            tl.store(out_ptrs, acc[d_idx], mask=mask_store)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if present
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f

"""
TriMul (outgoing) optimized Triton kernel.

The core operation computes, for each batch `b` and hidden channel `d`:

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

where `left` and `right` are the projected and gated tensors
of shape [B, N, N, H] (H = hidden_dim).  The kernel performs a
batched matrix multiplication `A @ Bᵀ` for each hidden slice, handling
multiple hidden slices (`BLOCK_D`) per program to reduce launch overhead.
All LayerNorms, Linear projections and gating are executed in PyTorch,
while the O(N³) contraction is off‑loaded to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    lhs_ptr, rhs_ptr, out_ptr,
    batch, M, N, K, D,
    stride_lhs_batch, stride_lhs_i, stride_lhs_k, stride_lhs_d,
    stride_rhs_batch, stride_rhs_i, stride_rhs_k, stride_rhs_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Batched GEMM across hidden dimension.

    Arguments
    ---------
    lhs_ptr, rhs_ptr, out_ptr : pointers to left, right and output tensors.
    batch, M, N, K, D          : sizes (batch, rows, cols, inner, hidden).
    stride_*                   : strides (in elements) for each dimension.
    BLOCK_*                    : compile‑time tile sizes.
    """
    pid_batch = tl.program_id(0)   # which batch element
    pid_m = tl.program_id(1)       # tile over rows (i)
    pid_n = tl.program_id(2)       # tile over cols (j)

    # Tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # Base pointers for this batch
    lhs_batch = lhs_ptr + pid_batch * stride_lhs_batch
    rhs_batch = rhs_ptr + pid_batch * stride_rhs_batch
    out_batch = out_ptr + pid_batch * stride_out_batch

    # Loop over hidden‑dim tiles
    for d_tile in range(0, tl.cdiv(D, BLOCK_D)):
        cur_d = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = cur_d < D

        # Accumulator for this hidden tile (float32)
        acc = tl.zeros((BLOCK_D, BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over the reduction dimension K
        for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
            cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = cur_k < K

            # Load A block: shape (BLOCK_M, BLOCK_K, BLOCK_D)
            a_ptrs = lhs_batch \
                + offs_i[:, None, None] * stride_lhs_i \
                + cur_k[None, :, None] * stride_lhs_k \
                + cur_d[None, None, :] * stride_lhs_d
            a = tl.load(
                a_ptrs,
                mask=mask_i[:, None, None] &
                     mask_k[None, :, None] &
                     mask_d[None, None, :],
                other=0.0
            )

            # Load B block transposed (K, N, D)
            b_ptrs = rhs_batch \
                + cur_k[:, None, None] * stride_rhs_k \
                + offs_j[None, :, None] * stride_rhs_i \
                + cur_d[None, None, :] * stride_rhs_d
            b = tl.load(
                b_ptrs,
                mask=mask_k[:, None, None] &
                     mask_j[None, :, None] &
                     mask_d[None, None, :],
                other=0.0
            )

            # Accumulate dot products for each hidden slice
            for d_idx in range(0, BLOCK_D):
                acc[d_idx] += tl.dot(a[:, :, d_idx], b[:, :, d_idx])

        # Store the results for this hidden tile
        for d_idx in range(0, BLOCK_D):
            store_mask = mask_i[:, None] & mask_j[None, :] & mask_d[d_idx]
            d_off = cur_d[d_idx]  # scalar offset into hidden dim
            out_ptrs = out_batch \
                + offs_i[:, None] * stride_out_i \
                + offs_j[None, :] * stride_out_j \
                + d_off * stride_out_d
            tl.store(out_ptrs, acc[d_idx], mask=store_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (all on the same CUDA device)
        - config       : dict with at least "dim" and "hidden_dim"
                         optionally "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to FP16 for the GEMM kernel (accumulator stays FP32)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # [B, N, N, hidden_dim]

    # ---------- Allocate output buffer ----------
    out = torch.empty((B, N, N, hidden_dim), device=device, dtype=torch.float32)

    # ---------- Strides (in elements) ----------
    s_l_batch, s_l_i, s_l_k, s_l_d = left.stride()
    s_r_batch, s_r_i, s_r_k, s_r_d = right.stride()
    s_o_batch, s_o_i, s_o_j, s_o_d = out.stride()

    # ---------- Triton launch configuration ----------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    BLOCK_D = 2   # number of hidden channels processed per program

    grid = (
        B,                                    # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,          # row tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # column tiles
    )

    # ---------- Run the kernel ----------
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        N,
        N,
        hidden_dim,
        s_l_batch,
        s_l_i,
        s_l_k,
        s_l_d,
        s_r_batch,
        s_r_i,
        s_r_k,
        s_r_d,
        s_o_batch,
        s_o_i,
        s_o_j,
        s_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
