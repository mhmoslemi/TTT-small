"""
custom_kernel: Triton-accelerated outgoing TriMul (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C).
2. Linear projections → left/right (B,N,N,H), optional mask multiplication.
3. Compute three sigmoid gates (left, right, out) from the same normalized input.
4. Apply gates to left/right (mixed‑precision, fp16) to obtain gated tensors.
5. Perform the expensive contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched half‑precision GEMM: each (b,h) slice computes
   A (N×N) @ Bᵀ (N×N).  The matmul is implemented in Triton with
   tiling (BLOCK_M×BLOCK_N×BLOCK_K) and fp32 accumulation.
6. Reshape the result back to (B,N,N,H) (float32).
7. Layer‑norm over the hidden dimension, multiply by the out‑gate,
   and a final linear projection back to the original channel dimension.

Only the O(N³) contraction is executed on the GPU via Triton; all other
operations use PyTorch (high‑level, negligible cost).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ  (A,B: (batch, N, N), C: (batch, N, N))
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ.
    A, B, C are stored in row‑major order.
    """
    pid_batch = tl.program_id(0)      # batch * hidden_dim index
    pid_m = tl.program_id(1)          # block row index
    pid_n = tl.program_id(2)          # block col index

    # Offsets for the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block (M x K)
        a_offs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_offs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B block transposed on the fly (K x N) = Bᵀ
        b_offs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_offs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # Write out the result
    c_offs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_offs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (tensors on the same device)
            - config: dict containing at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # (B,N,N,H)
    right = F.linear(x_norm, weights["right_proj.weight"])  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Convert mask to fp16 for the upcoming half‑precision ops
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)         # (B,N,N,1), float32
        mask_f = mask_f.to(torch.float16)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))   # keep float32 for later

    # ------------------------------------------------------------------
    # 5) Apply gates (mixed precision)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # Cast to half for the kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6) Prepare for Triton GEMM: shape (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape: (B,N,N,H)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (float32 accumulator)
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 7) Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides in elements (row‑major layout)
    stride_a_batch = N * N
    stride_a_row = N
    stride_a_col = 1

    stride_b_batch = N * N
    stride_b_row = N          # original row stride of B (j dimension)
    stride_b_col = 1          # original column stride of B (k dimension)

    stride_c_batch = N * N
    stride_c_row = N
    stride_c_col = 1

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9) Output LayerNorm, gating, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Apply output gate (float32)
    out = out * out_gate

    # Final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out
