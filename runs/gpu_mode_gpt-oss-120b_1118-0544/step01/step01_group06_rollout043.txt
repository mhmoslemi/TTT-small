import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, mask_ptr, c_ptr,
    B, N, H,
    stride_ab, stride_ai, stride_ak, stride_ah,
    stride_bb, stride_bk, stride_bn, stride_bh,
    stride_mb, stride_mi, stride_mj, stride_mh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)      # batch index
    pid_m = tl.program_id(1)      # row tile
    pid_n = tl.program_id(2)      # col tile

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # we will iterate over heads inside loop
    
    for h in range(H):
        # pointers for a and b for this head
        a_batch_head_ptr = a_ptr + pid_b * stride_ab + h * stride_ah
        b_batch_head_ptr = b_ptr + pid_b * stride_bb + h * stride_bh
        c_batch_head_ptr = c_ptr + pid_b * stride_mb + h * stride_mh

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            # mask offsets for left and right
            mask_a_ptr = mask_ptr + pid_b * stride_mb + offs_m[:, None] * stride_mi + offs_k[None, :] * stride_mj
            mask_b_ptr = mask_ptr + pid_b * stride_mb + offs_k[:, None] * stride_mi + offs_n[None, :] * stride_mj

            a_ptrs = a_batch_head_ptr + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
            b_ptrs = b_batch_head_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

            a = tl.load(a_ptrs)
            b = tl.load(b_ptrs)
            mask_a = tl.load(mask_a_ptr)
            mask_b = tl.load(mask_b_ptr)
            a = a * mask_a
            b = b * mask_b

            acc += tl.dot(a, b)

        # store
        c_ptrs = c_batch_head_ptr + offs_m[:, None] * stride_mi + offs_n[None, :] * stride_mj
        # note: strides for out: same as left (since output same shape [B,N,N,H])
        mask_c = (offs_m[:, None] < N) & (offs_n[None, :] < N)
        tl.store(c_ptrs, acc, mask=mask_c)

"""
TriMul (outgoing) forward pass optimized with Triton.

Algorithm
---------
1. Layer‑norm the input over the last channel dimension.
2. Linear projections (left/right) from `dim` → `hidden_dim`.
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Fuse the left/right gates into the projected tensors (still fp32),
   then cast to fp16 for the heavy N³ contraction.
5. Triton kernel computes, for each batch and each head `h`:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] * mask[b,i,k] * mask[b,j,k]
   The kernel iterates over heads (compile‑time unrolled) and the reduction
   axis `k` using a blocked GEMM (tile sizes 64×64×64) with fp16 loads
   and fp32 accumulation.
6. Apply a second LayerNorm, multiply by the output gate and a final linear
   projection back to `dim`.  The final tensor is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, mask_ptr, c_ptr,               # pointers
    M, N, K,                                     # sequence lengths (all equal)
    stride_ab, stride_ai, stride_ak, stride_ah,  # left strides: batch, i, k, head
    stride_bb, stride_bi, stride_bk, stride_bh,  # right strides (same layout)
    stride_cb, stride_ci, stride_cj, stride_ch,  # output strides
    stride_mb, stride_mi, stride_mj,             # mask strides: batch, i, j
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_HEADS: tl.constexpr
):
    # Program IDs (grid): 0 = batch, 1 = block row i, 2 = block col j
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Loop over heads (unrolled at compile time)
    for head in range(NUM_HEADS):
        # Base pointers for the current batch and head
        a_head = a_ptr + pid_b * stride_ab + head * stride_ah
        b_head = b_ptr + pid_b * stride_bb + head * stride_bh
        c_head = c_ptr + pid_b * stride_cb + head * stride_ch

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Reduction over k dimension
        for kk in range(0, tl.cdiv(K, BLOCK_K)):
            offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)

            # Masks for out‑of‑bounds loads
            a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
            b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

            # Pointers for left and right tiles
            a_ptrs = a_head + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
            b_ptrs = b_head + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bi

            # Load tiles (fp16)
            a = tl.load(a_ptrs, mask=a_mask, other=0.0)
            b = tl.load(b_ptrs, mask=b_mask, other=0.0)

            # Load mask tiles (fp16) and apply
            mask_a = tl.load(mask_ptr + pid_b * stride_mb
                              + offs_m[:, None] * stride_mi
                              + offs_k[None, :] * stride_mj,
                             mask=a_mask, other=0.0)
            mask_b = tl.load(mask_ptr + pid_b * stride_mb
                              + offs_k[:, None] * stride_mi
                              + offs_n[None, :] * stride_mj,
                             mask=b_mask, other=0.0)
            a = a * mask_a
            b = b * mask_b

            # Accumulate
            acc += tl.dot(a, b)

        # Store the result (fp32)
        c_ptrs = c_head + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
        c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
        tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: Tuple containing:
            - input_tensor (torch.Tensor): [B, N, N, dim] (float32)
            - mask (torch.Tensor or None): [B, N, N] (float32 or bool)
            - weights (dict of torch.Tensor): model parameters
            - config (dict): contains 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 4. Apply gates (still float32) and cast to fp16 for the kernel
    # --------------------------------------------------------------
    left = (left_proj * left_gate).to(torch.float16)
    right = (right_proj * right_gate).to(torch.float16)
    # out_gate will be used later; keep it float32 for the final multiplication
    out_gate = out_gate.to(torch.float32)

    # --------------------------------------------------------------
    # 5. Prepare mask (fp16). If no mask, use an all‑ones tensor.
    # --------------------------------------------------------------
    if mask is None:
        mask_h = torch.ones(
            (x.shape[0], x.shape[1], x.shape[2]), dtype=torch.float16, device=x.device
        )
    else:
        mask_h = mask.to(torch.float16)

    # --------------------------------------------------------------
    # 6. Allocate output buffer (float32)
    # --------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float32)  # shape [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 7. Triton kernel launch
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    _tri_mul_kernel[grid](
        left,
        right,
        mask_h,
        out,
        N,                # M
        N,                # N
        N,                # K
        left.stride(0),   # stride_ab (batch)
        left.stride(1),   # stride_ai (i)
        left.stride(2),   # stride_ak (k)
        left.stride(3),   # stride_ah (head)
        right.stride(0),  # stride_bb
        right.stride(1),  # stride_bi (i) – used as column stride after transpose
        right.stride(2),  # stride_bk (k)
        right.stride(3),  # stride_bh (head)
        out.stride(0),    # stride_cb
        out.stride(1),    # stride_ci (i)
        out.stride(2),    # stride_cj (j)
        out.stride(3),    # stride_ch (head)
        mask_h.stride(0), # stride_mb (mask batch)
        mask_h.stride(1), # stride_mi (mask i)
        mask_h.stride(2), # stride_mj (mask j)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_HEADS=hidden_dim,
    )

    # --------------------------------------------------------------
    # 8. Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
