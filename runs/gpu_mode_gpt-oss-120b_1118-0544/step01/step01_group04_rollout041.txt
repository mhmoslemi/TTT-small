def custom_kernel(data):
    """
    Triton-accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Steps:
    1. LayerNorm on input.
    2. Linear projections and gates (all in PyTorch).
    3. Optional mask multiplication.
    4. Core contraction: out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
       executed as a batched GEMM per hidden channel using a custom Triton kernel
       that works directly on the [B,N,N,hidden] layout to avoid costly transposes.
    5. Output LayerNorm, output gate, final linear projection.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False) and mask is not None

    # LayerNorm on the last dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Optional mask (broadcast over hidden dim)
    if use_mask:
        mask_exp = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Convert to half for the heavy contraction
    left_fp16 = left.contiguous().to(torch.float16)
    right_fp16 = right.contiguous().to(torch.float16)

    B, N, _, _ = left_fp16.shape

    # Output tensor (float32)
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)

    # Strides for the flattened (B, hidden) batch dimension
    stride_batch = left_fp16.stride(0)           # B stride (N*N*hidden)
    stride_i = left_fp16.stride(1)               # row stride (N*hidden)
    stride_k = left_fp16.stride(2)               # column stride (hidden)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_m, num_n, B * hidden_dim)

    tri_mul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B,
        N,
        hidden_dim,
        stride_batch,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Output now has shape [B, N, N, hidden_dim] (float32)
    out = out_fp32

    # Output LayerNorm over hidden_dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Output gate
    out = out * out_gate

    # Final projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×C}.
2. Linear projections left = x·W_left, right = x·W_right (no bias).
3. Apply sigmoid gates left_gate, right_gate, out_gate.
4. Optional mask multiplication.
5. Fuse gates: left ← left·left_gate, right ← right·right_gate.
6. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM for each hidden channel d.
   The Triton kernel works directly on the contiguous layout
   [B,N,N,hidden] to avoid costly permutes / copies.
7. Output layer‑norm over hidden, multiply by out_gate, final linear
   projection back to C dimensions.

Only the O(N³·hidden) contraction is performed on the GPU with Triton;
all other operations use regular PyTorch (which is already highly
optimized on the GPU).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,          # const half*   (input left)
    right_ptr,         # const half*   (input right)
    out_ptr,           #       float*  (output)
    batch_size: tl.int32,
    N: tl.int32,
    hidden_dim: tl.int32,
    stride_batch: tl.int32,   # stride for the batch dimension (B dimension)
    stride_i: tl.int32,       # stride for the first N dimension (i / row)
    stride_k: tl.int32,       # stride for the second N dimension (k / col)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Tile indices
    pid_m = tl.program_id(0)          # row tile index (i)
    pid_n = tl.program_id(1)          # col tile index (j)
    pid_b = tl.program_id(2)          # flattened (batch, hidden) index

    # Decode batch and hidden indices from the flattened id
    b = pid_b // hidden_dim
    h = pid_b - b * hidden_dim          # equivalent to pid_b % hidden_dim

    # Base pointers for this (b, h) slice
    base_left = left_ptr + b * stride_batch + h
    base_right = right_ptr + b * stride_batch + h
    base_out = out_ptr + b * stride_batch + h

    # Offsets for the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load left tile A[i,k]  (shape BLOCK_M×BLOCK_K)
        a_ptrs = base_left + offs_i[:, None] * stride_i + cur_k[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load right tile B_T[k,j] where B_T is the transpose of right
        # Original right is indexed as right[b, j, k, h]
        b_ptrs = base_right + cur_k[:, None] * stride_k + offs_j[None, :] * stride_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # dot performs FP32 accumulation for FP16 inputs

    # Store the result tile C[i,j]
    c_ptrs = base_out + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False) and mask is not None

    # ------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------
    if use_mask:
        mask_exp = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------
    # 6. Heavy contraction using Triton
    # ------------------------------------------------------------
    # Cast to FP16 to halve memory traffic
    left_fp16 = left.contiguous().to(torch.float16)
    right_fp16 = right.contiguous().to(torch.float16)

    B, N, _, _ = left_fp16.shape

    # Output tensor (float32 as required by subsequent layers)
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)

    # Strides for the [B,N,N,hidden] layout (contiguous case)
    stride_batch = left_fp16.stride(0)   # B‑stride = N*N*hidden
    stride_i = left_fp16.stride(1)       # row stride = N*hidden
    stride_k = left_fp16.stride(2)       # col stride = hidden
    # hidden_dim stride is 1 (last dimension)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_m, num_n, B * hidden_dim)

    tri_mul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B,
        N,
        hidden_dim,
        stride_batch,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------
    # 7. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------
    out = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------
    # 8. Output gate and final projection
    # ------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
