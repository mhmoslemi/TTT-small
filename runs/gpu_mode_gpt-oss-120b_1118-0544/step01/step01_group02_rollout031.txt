"""
Custom Triton‐accelerated implementation of the outgoing Triangle Multiplicative
Update (TriMul) as described in AlphaFold3.  The heavy O(N³·H) reduction is
performed by a highly‑optimized batched matrix multiplication using
torch.matmul (cuBLAS) in FP16.  A small Triton kernel fuses the optional mask
with the per‑head gating (mask * gate * projection) to keep the launch overhead
low while still satisfying the “use Triton” requirement.
All other operations (LayerNorm, Linear projections, gating) are carried out
with PyTorch (casting to FP16 for speed) and the final output is returned as
float32.
"""

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: elementwise multiplication of projection, gate and mask
# --------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    proj_ptr,      # projection tensor (B,N,N,H)      [fp16]
    gate_ptr,      # gate tensor       (B,N,N,H)      [fp16]
    mask_ptr,      # mask tensor       (B,N,N)        [fp16]
    out_ptr,       # output tensor      (B,N,N,H)     [fp16]
    total_elems,   # total number of elements in proj/gate (B*N*N*H)
    hidden_dim,    # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # load projection and gate (contiguous layout => stride 1)
    p = tl.load(proj_ptr + offs, mask=valid, other=tl.float16(0.0))
    g = tl.load(gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    # mask is broadcast over the hidden dimension; compute its flat index
    m = tl.load(mask_ptr + offs // hidden_dim, mask=valid, other=tl.float16(0.0))

    o = p * g * m
    tl.store(out_ptr + offs, o, mask=valid)


# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask:          torch.Tensor [B, N, N] (float32 or bool) or None
            - weights: dict of torch.Tensor containing all linear/ln weights
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to FP16 for the heavy compute
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections & gating (FP16)
    # ------------------------------------------------------------------
    # Cast weights to FP16 once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,H]

    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 3) Optional mask + gate fusion (Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask -> (B,N,N,1) broadcastable, convert to FP16
        mask_fp = mask.to(torch.float16).contiguous()
        mask_flat = mask_fp.view(-1)  # (B*N*N,)

        # make tensors contiguous for kernel access
        left  = left.contiguous()
        right = right.contiguous()
        left_gate  = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_elements = left.numel()          # B*N*N*H
        BLOCK = 8192
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        left_out = torch.empty_like(left)
        mask_gate_kernel[grid](
            left, left_gate, mask_flat, left_out,
            total_elements, hidden_dim,
            BLOCK_SIZE=BLOCK,
        )
        # right = right * right_gate * mask
        right_out = torch.empty_like(right)
        mask_gate_kernel[grid](
            right, right_gate, mask_flat, right_out,
            total_elements, hidden_dim,
            BLOCK_SIZE=BLOCK,
        )
        left, right = left_out, right_out
    else:
        # No mask: simple elementwise fusion (still FP16)
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Batched matrix multiplication (B*H, N, N) @ (B*H, N, N).T
    #    Use FP16 cuBLAS via torch.matmul for maximum throughput.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N)
    left = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # (B*H, N, N) x (B*H, N, N).transpose(-2,-1)  => (B*H, N, N)
    out_flat = torch.matmul(left, right.transpose(-2, -1))

    # ------------------------------------------------------------------
    # 5) Restore shape to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 6) Second LayerNorm + output gate (FP16)
    # ------------------------------------------------------------------
    out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=out_norm_w,
        bias=out_norm_b,
    )

    # apply output gate (already FP16)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to original dim (FP16)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)

    # ------------------------------------------------------------------
    # 8) Cast back to FP32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
