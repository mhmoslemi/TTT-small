"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Linear projections (left/right) from dim → hidden_dim.
3. Optional mask (broadcasted on hidden_dim) and three sigmoid gates.
4. Apply the first two gates to the projected tensors.
5. Core O(N³) contraction:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM: for each (b,d),
       out = left @ rightᵀ
   performed by a Triton kernel that tiles the M (i), N (j) and K (k) axes.
   The kernel works directly on the original layout (no permute/copy) and
   uses half‑precision inputs with FP32 accumulation.
6. Layer‑norm the hidden result, multiply by the output gate,
   and project back to the original dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,
    stride_bh, stride_bk, stride_bn,
    stride_ch, stride_cm, stride_cn,
    stride_a_head, stride_b_head, stride_c_head,
    hidden_dim: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ Bᵀ  where A,B shape = (B, H, N, N)."""
    # program id 0 == (batch, head)
    pid = tl.program_id(0)
    batch_idx = pid // hidden_dim
    head_idx = pid % hidden_dim

    # base pointers for this (batch, head) slice
    a = a_ptr + batch_idx * stride_ah + head_idx * stride_a_head
    b = b_ptr + batch_idx * stride_bh + head_idx * stride_b_head
    c = c_ptr + batch_idx * stride_ch + head_idx * stride_c_head

    # program ids for tile coordinates
    pid_m = tl.program_id(1)  # i‑block
    pid_n = tl.program_id(2)  # j‑block

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a_ptrs = a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a_tile = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b_tile = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a_tile, b_tile)

    # write out tile
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation using Triton for the
    N³ reduction.  Returns a tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(dtype=x.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton (half‑precision with FP32 accumulation)
    # ------------------------------------------------------------------
    # Convert to half and ensure contiguity (no extra copies if already contiguous)
    left_h = left.contiguous().to(torch.float16)
    right_h = right.contiguous().to(torch.float16)
    del left, right  # free FP32 memory

    B, N_seq, _, _ = left_h.shape
    assert B == x.shape[0]
    # Output buffer: hidden result before the final projection
    out_hidden = torch.empty((B, N_seq, N_seq, hidden_dim), dtype=torch.float32, device=x.device)

    # Strides (in element units)
    stride_ah = left_h.stride(0)   # batch stride
    stride_am = left_h.stride(1)   # i stride
    stride_ak = left_h.stride(2)   # k (j) stride
    stride_a_head = left_h.stride(3)  # hidden dim stride (==1)

    # Right operand: we need Bᵀ, i.e. rows = k (original j), cols = j (original i)
    stride_bh = right_h.stride(0)          # batch stride
    stride_bk = right_h.stride(2)          # row stride for k (original j) -> D
    stride_bn = right_h.stride(1)          # col stride for j (original i) -> N*D
    stride_b_head = right_h.stride(3)      # hidden dim stride (==1)

    stride_ch = out_hidden.stride(0)   # batch stride
    stride_cm = out_hidden.stride(1)   # i stride
    stride_cn = out_hidden.stride(2)   # j stride
    stride_c_head = out_hidden.stride(3)  # hidden dim stride (==1)

    # Block sizes – can be tuned, 64 works well for the provided ranges
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        (N_seq + BLOCK_M - 1) // BLOCK_M,
        (N_seq + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_gemm_kernel[grid](
        left_h,
        right_h,
        out_hidden,
        N_seq,
        N_seq,
        N_seq,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bk,
        stride_bn,
        stride_ch,
        stride_cm,
        stride_cn,
        stride_a_head,
        stride_b_head,
        stride_c_head,
        hidden_dim=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate, and final projection back to dim
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
