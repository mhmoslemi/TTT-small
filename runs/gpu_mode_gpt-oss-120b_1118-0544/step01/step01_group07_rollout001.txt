"""
TriMul (outgoing) implementation with Triton acceleration.
-------------------------------------------------------
The core O(N³) operation:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is realized as a batched matrix multiplication:
    out = (left_flat) @ (right_flat)ᵀ
where left_flat and right_flat are reshaped to [B*H, N, N] (H = hidden_dim).
A custom Triton kernel computes C = A @ Bᵀ with tiled loads, fp16
matrix elements and fp32 accumulation (TensorCore friendly).  All other
operations (LayerNorm, Linear projections, gating) are performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ  (A,B : [batch, M, K], C : [batch, M, N])
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr,                # *fp16  [batch, M, K]
    B_ptr,                # *fp16  [batch, N, K]  (read as K×N)
    C_ptr,                # *fp32  [batch, M, N]
    M, N, K,              # i32 dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)
    pid_m = tl.program_id(0)      # row tile
    pid_n = tl.program_id(1)      # col tile

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Masks for the current tile
        mask_a = mask_m[:, None] & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & mask_n[None, :]

        # Pointers for A and transposed B
        a_ptrs = (A_ptr
                  + pid_batch * stride_a_batch
                  + offs_m[:, None] * stride_a_m
                  + offs_k[None, :] * stride_a_k)

        b_ptrs = (B_ptr
                  + pid_batch * stride_b_batch
                  + offs_k[:, None] * stride_b_k   # K dimension (row of Bᵀ)
                  + offs_n[None, :] * stride_b_n)  # N dimension (col of Bᵀ)

        # Load tiles (zero‑pad out‑of‑bounds)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # Accumulate (fp16*fp16 → fp32)
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = (C_ptr
              + pid_batch * stride_c_batch
              + offs_m[:, None] * stride_c_m
              + offs_n[None, :] * stride_c_n)
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)


def batched_matmul_fp16(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for batches of matrices using the Triton kernel above.
    A, B : [batch, M, K] and [batch, N, K] (fp16, contiguous)
    Returns C : [batch, M, N] (fp32)
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides in element units (not bytes)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_n, stride_b_k = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tunable tile sizes – good for fp16 TensorCores on NVIDIA B200/A100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )

    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return C


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation (no gradient needed).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of model parameters
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # Configuration & weight extraction
    # --------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm (input)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    # Gating linear weights
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    # Output LayerNorm & final projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------
    # 3. Optional masking (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Compute gating tensors (sigmoid) and apply
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core TriMul: out = Σ_k left[...,i,k,d] * right[...,j,k,d]
    #    Implemented as batched matmul A @ Bᵀ using the Triton kernel.
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # batch size, sequence length

    # Rearrange to [B, hidden, N, N] and flatten batch*hidden
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    batch_hidden = B * hidden_dim
    left_flat = left.view(batch_hidden, N, N)      # [BH, N, N]
    right_flat = right.view(batch_hidden, N, N)    # [BH, N, N]

    # Cast to fp16 for the kernel (TensorCore friendly)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Batched matmul: result shape [BH, N, N] (fp32)
    out_mat = batched_matmul_fp16(left_fp16, right_fp16)

    # Restore original layout: [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    # Final projection back to original dimension
    output = F.linear(out, to_out_weight)   # (dim, hidden_dim) weight

    return output
