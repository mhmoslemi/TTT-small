```
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # batch*hidden dimension
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch_ptr = a_ptr + pid * stride_a_batch
    b_batch_ptr = b_ptr + pid * stride_b_batch
    c_batch_ptr = c_ptr + pid * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        b_ptrs = b_batch_ptr + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row   # load B^T
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)
    # store result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul outgoing variant (AlphaFold3) with a fused Triton GEMM.

The kernel computes, for each batch and hidden dimension slice,
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d],
which is a batched matrix multiplication C = A @ Bᵀ.
All LayerNorms, linear projections and sigmoid gates are performed with
PyTorch (highly optimized cuBLAS/cuDNN).  The costly O(N³) contraction is
off‑loaded to a custom Triton kernel that uses fp16 inputs with fp32
accumulation and Tensor‑Core‑friendly block sizes (128×128×32).

The implementation supports optional masking and works for the
configuration space required by the benchmark (B∈{1,2}, N up to 1024,
dim∈{128,384}, hidden_dim∈{128,384}).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ Bᵀ
    A: (batch, M, K)   fp16
    B: (batch, N, K)   fp16
    C: (batch, M, N)   fp32
    """
    pid_batch = tl.program_id(0)   # which (batch * hidden_dim) matrix
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile col

    # Offsets for the tiles
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the current batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in chunks of BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block (M x K)
        a_offs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_offs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Load B block transposed on‑the‑fly (K x N) -> original B is (N x K)
        b_offs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_offs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Accumulate the dot product
        acc += tl.dot(a, b)

    # Write back C block
    c_offs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_offs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass for the Triangle Multiplicative Update (outgoing) using a
    Triton kernel for the expensive contraction.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (torch tensors on the same device)
            - config: dict containing at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm ----------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1] broadcast
        left = left * mask_f
        right = right * mask_f

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare for GEMM ----------
    # Cast to half for the kernel, keep fp32 for accumulation
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange to (B, hidden_dim, N, N) and flatten batch*hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left.shape  # H == hidden_dim

    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # Output buffer (fp32 accumulator)
    out_mat = torch.empty((B * H, N, N), device=device, dtype=torch.float32)

    # ---------- Triton kernel launch ----------
    # Block sizes tuned for Tensor Cores on Ampere/A100 (B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    stride_a_batch = N * N
    stride_a_row = N
    stride_a_col = 1

    stride_b_batch = N * N
    stride_b_row = N   # row stride of original B (N)
    stride_b_col = 1   # column stride of original B (1)

    stride_c_batch = N * N
    stride_c_row = N
    stride_c_col = 1

    grid = (
        B * H,                                   # batch*hidden
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ---------- Reshape back ----------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)   # [B,N,N,hidden_dim]

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
