```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # layer norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # cast to fp16
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # reshape for batched GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # strides in elements
    stride_a_batch = left_mat.stride(0)
    stride_a_row = left_mat.stride(1)
    stride_a_col = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_row = right_mat.stride(1)
    stride_b_col = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_row = out_mat.stride(1)
    stride_c_col = out_mat.stride(2)

    # Block sizes - tune for Tesla B200 (A100-like?). Use 128
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64  # could also be 128 but ensure fits inside shared memory.

    # Grid
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    triton_trimul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # reshape back: out [B,N,N,hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # apply out_gate
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
```

```python
@triton.jit
def triton_trimul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # each slice
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # coordinate offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # load A block (M,K)
        a = tl.load(
            a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B block transposed (K,N)
        b = tl.load(
            b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # store result
    c = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul outgoing variant (AlphaFold3) Triton implementation.

The core operation is a batched matrix multiplication:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

For each hidden dimension `d` we compute a matrix product
    left_mat[d] @ right_mat[d].T   (shape N×N @ N×N → N×N)

We pack the batch and hidden dimensions into a single batch axis,
use fp16 for the inputs (to exploit Tensor‑cores) and fp32 for the
accumulator.  The Triton kernel tiles the multiplication with
BLOCK_M×BLOCK_N tiles and a BLOCK_K reduction dimension.
All layer‑norms, linear projections and gating are performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def triton_trimul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  (A,B shape: batch × M × K, batch × N × K)."""
    pid_batch = tl.program_id(0)   # which (batch*hidden) matrix
    pid_m = tl.program_id(1)       # tile over rows  (M)
    pid_n = tl.program_id(2)       # tile over cols  (N)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Reduction over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block (M × K)
        a = tl.load(
            a_batch
            + offs_m[:, None] * stride_a_row
            + cur_k[None, :] * stride_a_col,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B block transposed on‑the‑fly (K × N)
        b = tl.load(
            b_batch
            + cur_k[:, None] * stride_b_col
            + offs_n[None, :] * stride_b_row,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Tensor‑core friendly dot
        acc += tl.dot(a, b)

    # Store the result
    c = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (tensors on same device)
            - config: dict containing at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to FP16 for the GEMM ----------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Reshape to (batch*hidden, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (FP32 accumulator)
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # Strides in elements (contiguous layout)
    stride_a_batch = left_mat.stride(0)
    stride_a_row = left_mat.stride(1)
    stride_a_col = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_row = right_mat.stride(1)
    stride_b_col = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_row = out_mat.stride(1)
    stride_c_col = out_mat.stride(2)

    # Tile sizes – chosen to fit into 64 KB of shared memory per SM
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (batch*hidden, tiles_M, tiles_N)
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the Triton kernel
    triton_trimul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # ---------- Reshape back to [B, N, N, hidden_dim] ----------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
