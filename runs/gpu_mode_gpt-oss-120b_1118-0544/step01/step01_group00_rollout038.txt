"""
Outgoing Triangle Multiplicative Update (TriMul) optimized with Triton.

The heavy part of TriMul is the reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication C = A @ Bᵀ for each
(b * hidden) pair.  This implementation:
  * Performs LayerNorm, linear projections and gating in PyTorch.
  * Casts the gated tensors to fp16 and reshapes them to
    (batch * hidden, N, N) where the inner dimension N is contiguous.
  * Launches a Triton kernel that computes the batched GEMM using
    tiling (BLOCK_M×BLOCK_N) and a K‑tile (BLOCK_K).  Accumulation is
    performed in fp32 for numerical stability and the result is stored
    back as fp16.
  * Reshapes the result, applies the final LayerNorm, output gate and
    projection to obtain the tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,                 # left  tensor pointer  (batch, M, K)
    b_ptr,                 # right tensor pointer  (batch, N, K) – accessed as transposed
    c_ptr,                 # output tensor pointer (batch, M, N)
    batch: tl.constexpr,   # total number of independent matrices = B * hidden_dim
    M: tl.constexpr,       # rows of A (and output)
    N: tl.constexpr,       # cols of Bᵀ (and output)
    K: tl.constexpr,       # reduction dimension
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)      # tile row
    pid_n = tl.program_id(1)      # tile column
    pid_b = tl.program_id(2)      # which matrix in the batch

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch element
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load a tile of A (M × K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (k_offset[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load a tile of Bᵀ (K × N) by reading B with swapped offsets
        b_ptrs = b_batch + (k_offset[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # FP32 accumulation of the dot product
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Store the result as FP16
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass using a fused Triton kernel.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask:          torch.Tensor of shape [B, N, N] (bool or float) – may be ignored
            - weights:       dict of model weights
            - config:        dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – if True the mask is not applied (optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ---------------------------------------------------------
    # Unpack inputs
    # ---------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------------------------------------------------------
    # Unpack model weights
    # ---------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ---------------------------------------------------------
    # Input LayerNorm
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ---------------------------------------------------------
    # Linear projections (no bias)
    # ---------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ---------------------------------------------------------
    # Optional mask (broadcast on hidden dimension)
    # ---------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ---------------------------------------------------------
    # Gating (sigmoid)
    # ---------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # Cast to fp16 for the reduction
    # ---------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, N, N, hidden_dim

    # Rearrange to (B, hidden, N, N) → flatten batch*hidden as independent matrices
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]

    batch_dim = B * hidden_dim
    left_mat = left_perm.view(batch_dim, N, N)   # (batch, M=N, K=N)
    right_mat = right_perm.view(batch_dim, N, N) # (batch, K=N, N) – accessed as transposed inside kernel

    # Allocate output buffer (fp16)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # ---------------------------------------------------------
    # Triton kernel launch configuration
    # ---------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m, num_pid_n, batch_dim)

    # Strides (in elements)
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)  # over rows (i)
    stride_a_k = left_mat.stride(2)  # over cols (k)

    stride_b_batch = right_mat.stride(0)
    # For Bᵀ we read B with k as fast axis and n as slow axis
    stride_b_k = right_mat.stride(2)  # stride over k (fast) – 1 for contiguous layout
    stride_b_n = right_mat.stride(1)  # stride over n (row) – N

    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    # Launch the GEMM kernel
    tri_mul_outgoing_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        batch=batch_dim,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # more warps per block for higher occupancy
    )
    # Ensure kernel completion before proceeding (optional but safe)
    torch.cuda.synchronize()

    # ---------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] and promote to fp32
    # ---------------------------------------------------------
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)      # [B, N, N, hidden_dim]
        .contiguous()
        .to(torch.float32)
    )

    # ---------------------------------------------------------
    # Final LayerNorm, gating and projection back to original dim
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # apply output gate
    out = F.linear(out, to_out_weight)       # [B, N, N, dim]

    return out
