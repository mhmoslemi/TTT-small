def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # Linear projections
    left = F.linear(x_norm, weights["left_proj.weight"])   # shape [B,N,N,hidden]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if present
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # Compute gates
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B,N,_ , _ = left.shape  # shape [B,N,N,hidden]

    # Permute and flatten batch*hidden
    left_mat = left.permute(0,3,1,2).contiguous()  # [B, hidden, N, N]
    right_mat = right.permute(0,3,1,2).contiguous()
    # flatten to (B*hidden, N, N)
    left_mat = left_mat.view(B*hidden_dim, N, N)
    right_mat = right_mat.view(B*hidden_dim, N, N)

    # Allocate output buffer
    out_mat = torch.empty((B*hidden_dim, N, N), dtype=torch.float32, device=x.device)

    # Kernel config
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    # compute grid
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # strides (in elements)
    stride_a_batch = N * N
    stride_a_row = N
    stride_a_col = 1
    stride_b_batch = N * N
    stride_b_row = N
    stride_b_col = 1
    stride_c_batch = N * N
    stride_c_row = N
    stride_c_col = 1

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # shape [B,N,N,hidden]

    # to_out_norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)

    # apply out_gate
    out = out * out_gate

    # final linear projection
    out = F.linear(out, weights["to_out.weight"])

    return out

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for edges
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to base for batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # load A tile (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B tile transposed: we need B_T (K,N) => read original B (N,K) but transpose
        b_ptrs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

The kernel computes, for each batch element b and hidden channel h,
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

which is a batched matrix‑multiply of the form A @ Bᵀ where
A = left (shape N×N) and B = right (shape N×N).  The heavy O(N³)
computation is performed by a custom Triton kernel that tiles the
matrices (BLOCK_M × BLOCK_N) while traversing K in BLOCK_K steps.
All other operations (LayerNorms, linear projections, gating) are
handled by PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ Bᵀ
    A shape : (batch, M, K)   – row‑major, K contiguous
    B shape : (batch, N, K)   – row‑major, K contiguous (will be accessed transposed)
    C shape : (batch, M, N)   – row‑major, N contiguous
    All tensors are stored in half‑precision; accumulation is float32.
    """
    pid_batch = tl.program_id(0)  # batch * hidden_dim index
    pid_m = tl.program_id(1)      # tile over M (i dimension)
    pid_n = tl.program_id(2)      # tile over N (j dimension)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (BLOCK_M x BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Load Bᵀ tile on‑the‑fly (BLOCK_K x BLOCK_N)
        # Original B is (N,K).  The indexing below yields B[j, k] => Bᵀ[k, j].
        b_ptrs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Accumulate dot product
        acc += tl.dot(a, b)

    # Write the result back
    c_ptrs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (all tensors on the same device)
            - config: dict containing at least "dim" and "hidden_dim"
                      optional key "nomask" (bool, default False)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------
    # 1. LayerNorm over the feature dimension
    # -------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3. Optional mask
    # -------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------
    # 4. Gating (sigmoid)
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # -------------------------------------------------
    # 5. Apply gates
    # -------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6. Cast to fp16 for the heavy contraction
    # -------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Rearrange to (B, hidden_dim, N, N) and flatten batch*hidden_dim
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (FP32 accumulator)
    out_mat = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # -------------------------------------------------
    # 7. Triton kernel launch
    # -------------------------------------------------
    # Tile sizes – chosen to be multiples of 16 for good tensor‑core utilization
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                     # batch * hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # tiles over rows (i)
        (N + BLOCK_N - 1) // BLOCK_N,       # tiles over cols (j)
    )

    # Compute strides in number of elements (not bytes)
    stride_a_batch, stride_a_row, stride_a_col = left_mat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_mat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_mat.stride()

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # -------------------------------------------------
    # 8. Reshape back to the original 4‑D layout
    # -------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # -------------------------------------------------
    # 9. Output LayerNorm, output gating, and final linear
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
