    A, B, C,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,

    pid_n = tl.program_id(1) # tile col
    pid_b = tl.program_id(2) # batch * hidden

    # Compute offsets

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy O(N³) part is the batched matrix multiplication
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
which is equivalent to a batch of GEMMs:
    out_h = left_h @ right_hᵀ   for each hidden channel h.
The kernel computes many independent GEMMs (batch*hidden) in FP16
with accumulation in FP32.  All other operations (LayerNorm, gating,
masking, final projection) are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_gemm_kernel(
    A, B, C,               # pointers
    M, N, K,               # matrix dims
    stride_am, stride_ak,  # A strides (M, K)
    stride_bk, stride_bn,  # B strides (K, N)
    stride_cm, stride_cn,  # C strides (M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel: C = A @ B   where A:(M,K), B:(K,N)."""
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # batch*hidden index

    # Offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch slice
    a_base = A + pid_b * (M * K)
    b_base = B + pid_b * (K * N)
    c_base = C + pid_b * (M * N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A block
        a = tl.load(
            a_base + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak,
            mask=(offs_m[:, None] < M) & (cur_k[None, :] < K),
            other=0.0,
        )
        # Load B block
        b = tl.load(
            b_base + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=(cur_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        # Accumulate
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # Write back the result in FP16
    c = acc.to(tl.float16)
    tl.store(
        c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using Triton for the
    N³ reduction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (see skeleton)
            - config       : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), norm_w, norm_b)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left = F.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply gates to the projected tensors
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched GEMM in FP16 via Triton
    # ------------------------------------------------------------------
    # Convert to half for the kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange to (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()               # [B, H, i, k]
    # Right must be transposed on its last two axes: (k, j)
    right = right.permute(0, 3, 2, 1).contiguous()            # [B, H, k, j]

    # Merge batch and channel dimensions for the kernel
    left_mat = left.view(-1, N, N)        # (B*H, M=N, K=N)
    right_mat = right.view(-1, N, N)      # (B*H, K=N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat)  # (B*H, N, N)

    # Strides (in elements)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bk = right_mat.stride(1)   # rows (K dimension)
    stride_bn = right_mat.stride(2)   # columns (N dimension)

    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # Tile sizes – tuned for typical B200 SMEM limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows
        (N + BLOCK_N - 1) // BLOCK_N,   # cols
        B * hidden_dim,                 # batch * hidden
    )

    _trimul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,          # M
        N,          # N
        N,          # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_hidden = F.layer_norm(out_hidden, (hidden_dim,), to_out_norm_w, to_out_norm_b)

    # Apply output gate (broadcasted)
    out_hidden = out_hidden * out_gate

    # Final linear projection back to original dim
    to_out_w = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(out_hidden, to_out_w)  # [B, N, N, dim]

    return out
