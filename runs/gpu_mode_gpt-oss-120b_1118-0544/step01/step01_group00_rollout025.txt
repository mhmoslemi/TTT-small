def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Norm x
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Projs
    left = F.linear(x, left_proj_weight)  # (bs,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    if not config.get("nomask", False):
        m = mask.to(x.dtype).unsqueeze(-1) # (bs,N,N,1)
        left = left * m
        right = right * m

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    del left, right, x

    # reshape and view
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()
    # flatten batch and hidden
    batch_dim = bs * hidden_dim
    left_h = left_h.view(batch_dim, N, N)
    right_h = right_h.view(batch_dim, N, N)

    # allocate output
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # Launch Triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # could be 64 but must be multiple of 16 maybe

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (num_pid_m, num_pid_n, batch_dim)

    # compute strides
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)

    stride_b_batch = right_h.stride(0)
    stride_b_k = right_h.stride(2)  # 1
    stride_b_n = right_h.stride(1)

    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    tri_mul_outgoing_kernel[grid](

        b_ptr=right_h,
        c_ptr=out_h,
        batch=batch_dim,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    out = out_h.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

"""
Outgoing Triangle Multiplicative Update (TriMul) kernel.

The kernel implements the costly reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
as a batched matrix multiplication C = A @ Bᵀ where each batch
corresponds to one (batch * hidden_dim) slice of the projected tensors.
All other operations (LayerNorm, linear projections, gating, final
normalisation) are performed with PyTorch.  The heavy matmul is
accelerated with a tiled Triton kernel that works in FP16 with
Tensor‑Core accumulation (FP32) and supports arbitrary sequence lengths
up to 1024.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A, B: (batch, N, N), C: (batch, N, N))
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  tensor (batch, M, K)   – here M = N, K = N
    b_ptr,               # right tensor (batch, N, K) – we treat it as transposed inside
    c_ptr,               # output tensor (batch, M, N)
    batch: tl.constexpr, # total independent matrices = B * hidden_dim
    M: tl.constexpr,     # rows of A (also rows of C) = N
    N: tl.constexpr,     # rows of B (also cols of C) = N
    K: tl.constexpr,     # reduction dimension = N
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # batch index (B * hidden_dim)

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K in blocks
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (k_offset[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Bᵀ tile: we read B as (K, N) using original strides (col = k, row = n)
        b_ptrs = b_batch + (k_offset[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product; result promoted to FP32
        acc += tl.dot(a, b).to(tl.float32)

    # write result
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None])

# ----------------------------------------------------------------------

    """

            - weights     : dict of model weights (see skeleton in the prompt)
            - config      : dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # Initial LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)    # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # Apply gates (Element‑wise multiplication)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the reduction (Tensor‑Core friendly)
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # free intermediate FP32 tensors early
    del left, right, x, left_gate, right_gate

    # ------------------------------------------------------------------
    # Rearrange to (batch * hidden, N, N) – contiguous layout required by the kernel
    # ------------------------------------------------------------------
    # shape after permute: (B, hidden_dim, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_h.shape  # B = batch, H = hidden_dim
    batch_dim = B * H
    left_h = left_h.view(batch_dim, N, N)
    right_h = right_h.view(batch_dim, N, N)

    # ------------------------------------------------------------------
    # Allocate output buffer (FP16)
    # ------------------------------------------------------------------
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Triton kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m, num_pid_n, batch_dim)  # (row tiles, col tiles, batch)

    # strides (in elements)
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)

    stride_b_batch = right_h.stride(0)
    # for Bᵀ we need stride over the reduction axis (k) and over n
    stride_b_k = right_h.stride(2)   # 1
    stride_b_n = right_h.stride(1)   # N

    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    # launch Triton kernel
    tri_mul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        batch_dim,
        N, N, N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] and convert to FP32
    # ------------------------------------------------------------------
    out = (
        out_h.view(B, H, N, N)
        .permute(0, 2, 3, 1)          # (B, N, N, hidden_dim)
        .contiguous()
        .to(torch.float32)
    )
    # free the half‑precision buffers
    del left_h, right_h, out_h

    # ------------------------------------------------------------------
    # Final LayerNorm, gating and projection back to the original dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
```
