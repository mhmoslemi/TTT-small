import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ B  where
        A : (batch, M, K)   (float16)
        B : (batch, K, N)   (float16)
        C : (batch, M, N)   (float32 accumulator)
    The batch dimension includes the original batch size multiplied by the hidden
    dimension (i.e. B * hidden_dim).  The kernel tiles over M and N with
    BLOCK_M×BLOCK_N tiles and over K with BLOCK_K.
    """
    pid_batch = tl.program_id(0)      # which batch*hidden matrix
    pid_m = tl.program_id(1)          # tile row index
    pid_n = tl.program_id(2)          # tile col index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M, K)
        a_offs = a_batch \
            + offs_m[:, None] * stride_a_row \
            + cur_k[None, :] * stride_a_col
        a = tl.load(a_offs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (K, N) – already stored transposed
        b_offs = b_batch \
            + cur_k[:, None] * stride_b_row \
            + offs_n[None, :] * stride_b_col
        b = tl.load(b_offs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c_offs = c_batch \
        + offs_m[:, None] * stride_c_row \
        + offs_n[None, :] * stride_c_col
    tl.store(c_offs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator using a fused Triton GEMM.
    Arguments:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (all on the same device)
            - config: dict containing at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # LayerNorm on the input channel dimension
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # Linear projections
    # -----------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # Optional mask (broadcasted on hidden dimension)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)           # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # Gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Cast to float16 for the heavy O(N³) contraction
    # -----------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # -----------------------------------------------------------------
    # Prepare tensors for batched GEMM:
    #   left : (B, hidden, N, N)   -> (batch', M, K)
    #   right: (B, hidden, N, N) transposed on the last two dims -> (batch', K, N)
    # -----------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, hidden, N, N) transposed

    left_mat = left_perm.view(B * hidden_dim, N, N)            # (batch', M, K)
    right_mat = right_perm.view(B * hidden_dim, N, N)          # (batch', K, N)

    out_mat = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # Strides (in elements, not bytes)
    # -----------------------------------------------------------------
    stride_a_batch, stride_a_row, stride_a_col = left_mat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_mat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_mat.stride()

    # -----------------------------------------------------------------
    # Kernel launch configuration
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # -----------------------------------------------------------------
    # Reshape back to [B, N, N, hidden]
    # -----------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # (B, N, N, hidden)

    # -----------------------------------------------------------------
    # Output LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
