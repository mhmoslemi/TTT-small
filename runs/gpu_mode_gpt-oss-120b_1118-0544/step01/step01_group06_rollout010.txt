"""
TriMul (outgoing) forward pass with Triton‐accelerated N³ contraction.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Linear projections (left/right) from dim → hidden_dim (no bias).
3. Optional mask broadcasted on the hidden dimension.
4. Three sigmoid gates (left, right, out); left/right gates are applied to the
   projected tensors.
5. Core contraction:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batch of GEMMs: for each (b,d) compute
        out = left @ rightᵀ
   Implemented with a fused Triton kernel that tiles M, N and K (default
   BLOCK_M=N_BLOCK, BLOCK_N=N_BLOCK, BLOCK_K=K_BLOCK) and accumulates in FP32
   while loading FP16.
6. Output LayerNorm, multiplication by the out‑gate and final linear projection
   back to the original dimension.
The function returns a float32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,   # a: (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # b: (batch, K, N)
    stride_ch, stride_cm, stride_cn,   # c: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for the TriMul core contraction.
    a_ptr: (B*H, M, K)   left  tensor
    b_ptr: (B*H, K, N)   right tensor (already transposed)
    c_ptr: (B*H, M, N)   output
    """
    pid_batch = tl.program_id(0)   # collapsed batch * hidden_dim
    pid_m = tl.program_id(1)       # block row (i)
    pid_n = tl.program_id(2)       # block col (j)

    # Offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for the current batch/head
    a_batch = a_ptr + pid_batch * stride_ah
    b_batch = b_ptr + pid_batch * stride_bh
    c_batch = c_ptr + pid_batch * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Pointers to the current tiles
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (FP16)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Write back the result (FP32)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of parameter tensors
            - config: dict containing 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction via Triton
    # ------------------------------------------------------------------
    # Cast to FP16 for the GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to bring hidden_dim to a leading axis
    # left: (B, H, N, N)   where axes are (i, k)
    # right: (B, H, N, N) where axes are (k, j) -> transpose last two dims
    left_perm = left.permute(0, 3, 1, 2).contiguous()       # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()     # (B, H, N, N) = (B, H, K, J)

    B, H, N, _ = left_perm.shape
    # Flatten batch and hidden dimensions for the kernel
    left_mat = left_perm.view(B * H, N, N)   # (B*H, M=N, K=N)
    right_mat = right_perm.view(B * H, N, N)  # (B*H, K=N, N=N)

    # Output buffer (FP32 accumulator)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=x.device)

    # Block sizes – tuned for typical N (≤1024)
    # Larger blocks improve occupancy for larger N.
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_outgoing_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        left_mat.stride(0),
        left_mat.stride(1),
        left_mat.stride(2),
        right_mat.stride(0),
        right_mat.stride(1),
        right_mat.stride(2),
        out_mat.stride(0),
        out_mat.stride(1),
        out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
