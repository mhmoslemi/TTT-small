@triton.jit
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_am, stride_ak, stride_ab,
                        stride_bk, stride_bn, stride_bb,
                        stride_cm, stride_cn, stride_cb,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # A shape: (batch, M, K) with strides: stride_ab, stride_am, stride_ak
    # B shape: (batch, K, N) with strides: stride_bb, stride_bk, stride_bn
    # C shape: (batch, M, N) with strides: stride_cb, stride_cm, stride_cn

    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute base pointers for this batch
    a_ptr = A_ptr + batch_id * stride_ab
    b_ptr = B_ptr + batch_id * stride_bb
    c_ptr = C_ptr + batch_id * stride_cb

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # mask out-of-bound
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # compute pointers
        a = tl.load(a_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak, mask=a_mask, other=0.0)
        b = tl.load(b_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

    # store
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn, acc, mask=c_mask)

"""
TriMul (outgoing) forward pass with a fused batched GEMM kernel in Triton.

The heavy part of TriMul is the contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which is a batch of matrix multiplies (left @ rightᵀ) over the hidden
dimension.  This implementation:
  * Performs LayerNorms and linear projections in PyTorch.
  * Casts the projected tensors to float16.
  * Permutes them so that the hidden dimension becomes the batch axis.
  * Executes a custom Triton kernel that computes many independent
    GEMMs (one per (batch,head) pair) in FP16 with FP32 accumulation.
  * Returns the result in float32 after output LayerNorm, gating and
    final linear projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM: C = A @ B
    A : (batch, M, K)   (fp16)
    B : (batch, K, N)   (fp16)
    C : (batch, M, N)   (fp32 accumulator, stored as fp32)
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = a_ptr + batch_id * stride_ab
    b_batch = b_ptr + batch_id * stride_bb
    c_batch = c_ptr + batch_id * stride_cb

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)

        # masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # load A and B tiles (fp16)
        a = tl.load(
            a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=a_mask,
            other=0.0,
        )
        b = tl.load(
            b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=b_mask,
            other=0.0,
        )

        # dot product accumulation
        acc += tl.dot(a, b)

    # store result (fp32)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(
        c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
        acc,
        mask=c_mask,
    )


def custom_kernel(data):
    """
    TriMul forward pass.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model weight tensors
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).type_as(x)      # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ contraction via Triton batched GEMM
    # ------------------------------------------------------------------
    # Cast to half for speed
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Bring hidden dimension to the batch axis
    # left: (B, hidden, N, K) where K == N
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    # right: (B, hidden, K, N) where K == N
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()  # [B, hidden, N, N]

    B, H, N, _ = left_perm.shape

    # Collapse batch & head dimensions for the kernel
    left_mat = left_perm.view(-1, N, N)   # (B*H, M=N, K=N)
    right_mat = right_perm.view(-1, N, N)  # (B*H, K=N, N=N)

    # Output buffer in fp32
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=x.device)

    # Tile sizes – tuned for typical GPU shared‑memory limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (batch*head, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch kernel
    _tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        # strides for A (batch, M, K)
        left_mat.stride(1),   # stride_am (row stride = K)
        left_mat.stride(2),   # stride_ak (col stride = 1)
        left_mat.stride(0),   # stride_ab (batch stride = M*K)
        # strides for B (batch, K, N)
        right_mat.stride(1),  # stride_bk (row stride = N)
        right_mat.stride(2),  # stride_bn (col stride = 1)
        right_mat.stride(0),  # stride_bb (batch stride = K*N)
        # strides for C (batch, M, N)
        out_mat.stride(1),    # stride_cm (row stride = N)
        out_mat.stride(2),    # stride_cn (col stride = 1)
        out_mat.stride(0),    # stride_cb (batch stride = M*N)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
