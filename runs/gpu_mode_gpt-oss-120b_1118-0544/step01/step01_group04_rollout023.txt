def triang_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_j,

        right = right * mask_exp

    right = right * right_gate

    right = right.to(torch.float16)

"""
TriMul “outgoing” forward pass (AlphaFold3 style).

The heavy 4‑D tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a per‑head matrix multiplication:
    left (i,k) @ right.T (k,j)  → (i,j)

Implementation details
----------------------
1. LayerNorm, linear projections and gating are executed with PyTorch in FP32.
2. The projected tensors are cast to FP16 to lower memory traffic.
3. `left` is kept as (i,k) while `right` is transposed to (k,j) via a
   `permute`.  Both tensors are reshaped to (B·H, N, N) where H = hidden_dim.
4. A custom Triton kernel performs a batched GEMM on these flattened
   matrices (FP16 × FP16 → FP32 accumulation).
5. The result is reshaped back to (B,N,N,hidden_dim) and passed through the
   output LayerNorm, output gate and final linear projection (FP32).

The kernel follows the classic block‑wise GEMM pattern with
BLOCK_M = BLOCK_N = 128 and BLOCK_K = 32, which provides good occupancy on
NVIDIA B200 GPUs.
"""

    # problem size
    N,
    # strides (batch, row, col for each tensor)
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_i, stride_c_j,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program identifiers
    pid_batch = tl.program_id(0)      # batch * hidden_dim
    pid_m = tl.program_id(1)          # tile over output rows (i)
    pid_n = tl.program_id(2)          # tile over output cols (j)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulation buffer in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left: shape (i, k)
        a = tl.load(
            left_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_i
            + cur_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right (transposed): shape (k, j)
        b = tl.load(
            right_ptr
            + pid_batch * stride_b_batch
            + cur_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_j,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # multiply‑accumulate
        acc += tl.dot(a, b)

    # write back
    out = out_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(out, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C]
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict with all linear‑layer and LayerNorm parameters
            - config: dict containing at least "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # nomask == True  →  mask is not used
    use_mask = (not config.get("nomask", False)) and (mask is not None)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2) Linear projections (no bias) – keep FP32 for accuracy
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # --------------------------------------------------------------
    # 4) Optional mask
    # --------------------------------------------------------------
    if use_mask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)          # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 5) Apply per‑head gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6) Cast to FP16 and prepare for the Triton GEMM
    # --------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    #   left : [B, N, N, H] → (batch, head, i, k)
    #   right: [B, N, N, H] → (batch, head, j, k) → transpose to (batch, head, k, j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)  i‑k
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N)  k‑j

    B, H, N, _ = left_perm.shape
    batch_hidden = B * H

    # flatten the first two dimensions for the kernel
    left_flat = left_perm.view(batch_hidden, N, N)
    right_flat = right_perm.view(batch_hidden, N, N)

    # output buffer (FP32 accumulation)
    out_flat = torch.empty((batch_hidden, N, N), dtype=torch.float32, device=left_flat.device)

    # strides for the flattened tensors
    stride_a_batch, stride_a_i, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_j = right_flat.stride()
    stride_c_batch, stride_c_i, stride_c_j = out_flat.stride()

    # --------------------------------------------------------------
    # 7) Batched GEMM via Triton
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_j,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8) Reshape back to [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)          # (B, N, N, H)
        .contiguous()
    )

    # --------------------------------------------------------------
    # 9) Output LayerNorm over hidden_dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 10) Output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 11) Final linear projection back to original dimension
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
