def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Get weights.

    # Normalization
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gating and cast to fp16 for matmul
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # Reshape for batched matmul
    B, N, _, _ = left.shape
    left_f = left.permute(0, 3, 1, 2).contiguous()
    right_f = right.permute(0, 3, 1, 2).contiguous()
    batch_hidden = B * hidden_dim

    left_mat = left_f.view(batch_hidden, N, N)
    right_mat = right_f.view(batch_hidden, N, N)

    # Compute A @ B^T with Triton
    out_mat_fp16 = batched_matmul_triton(left_mat, right_mat)

    # Reshape back to (B,N,N,hidden)
    out = out_mat_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # Cast back to float32 for subsequent ops
    out = out.to(torch.float32)

    # Output layer norm
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    # Apply out gate
    out = out * out_gate

    # Final projection
    output = F.linear(out, weights["to_out.weight"])

    return output

"""
custom_kernel for the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor `x` of shape [B, N, N, dim].
2. Linear projections:
   * left = Linear(x, left_proj_weight)   # -> [B, N, N, hidden]
   * right = Linear(x, right_proj_weight)
3. (Optional) mask is applied element‑wise to `left` and `right`.
4. Compute three sigmoid gates from `x`:
   left_gate, right_gate, out_gate  (shape [B, N, N, hidden]).
   Apply the first two gates to `left` and `right`.
5. Core O(N³) operation:
   out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
   This is a batched matrix multiplication C = A @ Bᵀ where
   A and B have shape (B·hidden, N, N).  The multiplication is
   performed by a Triton kernel that tiles the matrices in the
   M, N and K dimensions, accumulates in FP32 and writes FP32.
6. Reshape the result back to [B, N, N, hidden] and apply a second
   LayerNorm on the hidden dimension.
7. Multiply by `out_gate` and project back to the original `dim` with a
   final Linear layer.
8. Return the tensor of shape [B, N, N, dim] (float32).

Only the expensive N³ matrix multiplication is implemented in Triton;
all other operations use optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
# A : [batch, M, K]   (float32)
# B : [batch, N, K]   (float32)   – the kernel reads B as K×N
# C : [batch, M, N]   (float32)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                      # problem sizes (int32)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(2)               # batch index
    pid_m = tl.program_id(0)                   # block row of C
    pid_n = tl.program_id(1)                   # block col of C

    # ------------------------------------------------------------------
    # Compute offsets for the output tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Masks for the current K‑tile
        mask_a = mask_m[:, None] & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & mask_n[None, :]

        # Pointers for the current tile of A and (transposed) B
        a_ptrs = (
            A_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            B_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k   # K dimension (row of Bᵀ)
            + offs_n[None, :] * stride_b_n   # N dimension (col of Bᵀ)
        )

        # Load tiles (zero‑pad out‑of‑bounds)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float32)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float32)

        # Accumulate the product
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)


def batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for a batch of matrices using Triton.
    A, B : [batch, N, N] (float32, contiguous)
    Returns C : [batch, N, N] (float32)
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_n, stride_b_k = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tunable block sizes – chosen for good occupancy on NVIDIA B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),   # program_id 0
        triton.cdiv(N, BLOCK_N),   # program_id 1
        batch,                     # program_id 2
    )

    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return C


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict mapping weight names to tensors
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Extract weight tensors (all are already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the last (dim) dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply the first two gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul: batched matrix multiplication A @ Bᵀ
    #    Reshape to combine batch and hidden dimensions.
    # ------------------------------------------------------------------
    B_sz, N, _, _ = left.shape  # B_sz = batch size
    # Permute to [B, hidden, N, N] and make contiguous for the view
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch and hidden into a single dimension
    batch_hidden = B_sz * hidden_dim
    left_mat = left_perm.view(batch_hidden, N, N)    # [BH, N, N]
    right_mat = right_perm.view(batch_hidden, N, N)  # [BH, N, N]

    # Perform C = left_mat @ right_matᵀ via Triton
    out_mat = batched_matmul_triton(left_mat, right_mat)  # [BH, N, N] float32

    # Restore original layout [B, N, N, hidden]
    out = out_mat.view(B_sz, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm on hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7. Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    output = F.linear(out, to_out_weight)   # (dim, hidden) weight → [B,N,N,dim]

    return output
```
