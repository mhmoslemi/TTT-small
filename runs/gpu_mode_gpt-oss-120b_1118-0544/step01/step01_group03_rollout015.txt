"""
Triton‑accelerated forward pass for the outgoing Triangle Multiplicative Update (TriMul).

The algorithm follows the reference implementation:
1. Layer‑norm over the channel dimension of the input pair representation.
2. Linear projections to a hidden dimension (left / right) and three gating
   linear layers (left_gate, right_gate, out_gate). All linear layers have
   no bias.
3. Optional pair mask is applied to the projected tensors.
4. Gating values (sigmoid) are multiplied element‑wise.
5. Core reduction:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   This is equivalent to a batched matrix multiplication
   `left @ rightᵀ` for each hidden channel h.  A custom Triton kernel
   performs the batched GEMM with fp16 operands and fp32 accumulation.
6. Hidden‑dim layer‑norm, multiplication by the out‑gate, and a final
   linear projection back to the original channel dimension.

Only the N³ reduction (step 5) is implemented in Triton; the surrounding
operations use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all = seq_len)
    stride_a_batch, stride_am, stride_ak,   # A strides (batch, M, K)
    stride_b_batch, stride_bk, stride_bn,   # B strides (batch, K, N)
    stride_c_batch, stride_cm, stride_cn,   # C strides (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ B where
        A: (batch, M, K)   – fp16
        B: (batch, K, N)   – fp16 (already transposed for TriMul)
        C: (batch, M, N)   – fp32 accumulator, stored as fp32
    """
    pid_batch = tl.program_id(2)   # batch * hidden_dim
    pid_m = tl.program_id(0)       # row tile
    pid_n = tl.program_id(1)       # column tile

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = offs_k + k_offset

        # Masks for current K block
        mask_k = cur_k < K

        # Pointers for this block
        a_ptrs = a_ptr + pid_batch * stride_a_batch \
                  + offs_m[:, None] * stride_am \
                  + cur_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_batch * stride_b_batch \
                  + cur_k[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

        # Load with masks
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=tl.float16(0.0))

        # Accumulate (dot returns fp16, cast to fp32 on addition)
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
              + offs_m[:, None] * stride_cm \
              + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update.

    Parameters
    ----------
    data : tuple
        (pair_tensor, mask_tensor, weights_dict, config_dict)
        - pair_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping from string names to torch.Tensor weights
        - config_dict  : contains "dim" (C) and "hidden_dim" (hd)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] after the TriMul update (float32).
    """
    pair_tensor, mask, weights, config = data
    device = pair_tensor.device
    B, N, _, C = pair_tensor.shape
    assert N == pair_tensor.shape[2], "Input must be square on the sequence dimensions."

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # ------------------------------------------------------------------
    # Load weights (all expected to be float32 tensors)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over channel dimension C
    # ------------------------------------------------------------------
    x = F.layer_norm(pair_tensor,
                     (C,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=eps)          # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask handling
    # ------------------------------------------------------------------
    if mask is None:
        mask_val = torch.ones((B, N, N), dtype=pair_tensor.dtype, device=device)
    else:
        mask_val = mask.to(dtype=pair_tensor.dtype, device=device)
    mask_val = mask_val.unsqueeze(-1)      # [B, N, N, 1]

    left = left * mask_val
    right = right * mask_val

    # ------------------------------------------------------------------
    # 4. Gating tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul reduction via Triton batched GEMM
    #    Compute out = left @ rightᵀ for each hidden channel.
    # ------------------------------------------------------------------
    # Convert to half precision for the GEMM
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Permute to bring hidden_dim to a batch-like axis
    # left: (B, N, N, H) -> (B, H, N, N)   rows=i, cols=k
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    # right: need transposed last two dims (k, j)
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()

    # Merge batch and hidden dimension
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)          # (batch, M=N, K=N)
    right_flat = right_perm.view(BH, N, N)        # (batch, K=N, N)

    # Output buffer (float32)
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # Kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),      # number of tiles in M dimension
        triton.cdiv(N, BLOCK_N),      # number of tiles in N dimension
        BH,                           # batch * hidden_dim
    )

    _batched_matmul_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        M=N, N=N, K=N,
        stride_a_batch=left_flat.stride(0),
        stride_am=left_flat.stride(1),
        stride_ak=left_flat.stride(2),
        stride_b_batch=right_flat.stride(0),
        stride_bk=right_flat.stride(1),
        stride_bn=right_flat.stride(2),
        stride_c_batch=out_flat.stride(0),
        stride_cm=out_flat.stride(1),
        stride_cn=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Post‑processing: hidden‑dim LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # back to original dim

    return out
