"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the reference PyTorch module:
1️⃣ Layer‑norm over the channel dimension.
2️⃣ Linear projections (left/right) and three gating projections.
3️⃣ Optional pair mask applied to the projected tensors.
4️⃣ Element‑wise gating.
5️⃣ Core operation: for each hidden channel d,
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication A @ Bᵀ.
   This reduction is performed by a custom Triton kernel that tiles the
   (i,j) output and iterates over k in blocks.
6️⃣ Layer‑norm over the hidden dimension, multiply by out_gate and a final
   linear projection back to the original channel dimension.

The Triton kernel works on tensors of shape (batch*hidden_dim, N, N) and
uses block sizes chosen at launch based on the sequence length.
All surrounding linear / norm / gating work is done with PyTorch for
clarity and simplicity.
"""

import torch
import triton
import triton.language as tl
import math
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel for batch matmul C = A @ B, where A:[B,M,K] and B:[B,K,N]
# (B = batch * hidden_dim, M = N = seq_len, K = seq_len).
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to A, B, C
    M, N, K,                           # matrix sizes (all int64)
    stride_a_batch, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_b_batch, stride_bk, stride_bn,   # strides for B (batch, K, N)
    stride_c_batch, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Program IDs
    pid_batch = tl.program_id(2)   # which (batch*hidden) slice
    pid_m = tl.program_id(0)       # tile row
    pid_n = tl.program_id(1)       # tile col

    # Compute global offsets for the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks to avoid out‑of‑bounds loads
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Base pointers for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    # Pointers to the current tile of A and B
    a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Main loop over K dimension in BLOCK_K chunks
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        # dot returns same dtype as inputs, promote to FP32 explicitly
        acc += tl.dot(a, b)

        # Move pointers to next K‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the result tile
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative
    Update.

    Parameters
    ----------
    data : tuple
        (pair_tensor, mask_tensor, weights_dict, config_dict)
        - pair_tensor : torch.Tensor, shape [B, N, N, C]   (float32)
        - mask_tensor : torch.Tensor, shape [B, N, N] or None
        - weights_dict : dict of torch.Tensor weights used by the module
        - config_dict  : contains "dim", "hidden_dim" and optionally
                         "layernorm_eps"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    pair_tensor, mask, weights, config = data
    device = pair_tensor.device
    B, N, N2, C = pair_tensor.shape
    assert N == N2, "pair_tensor must be square in the sequence dimensions."

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # --------------------------------------------------------------
    # Load model weights
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1. LayerNorm over the channel dimension C
    # --------------------------------------------------------------
    pair_norm = F.layer_norm(pair_tensor,
                             (C,),
                             weight=norm_w,
                             bias=norm_b,
                             eps=eps)                     # [B, N, N, C]

    # --------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # --------------------------------------------------------------
    left = F.linear(pair_norm, left_proj_w)    # [B, N, N, hidden_dim]
    right = F.linear(pair_norm, right_proj_w)  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dimension)
    # --------------------------------------------------------------
    if mask is None:
        mask_val = torch.ones((B, N, N), dtype=pair_tensor.dtype, device=device)
    else:
        mask_val = mask.to(dtype=pair_tensor.dtype, device=device)
        # ensure shape [B, N, N]
        if mask_val.dim() != 3:
            raise ValueError("mask must have shape [B, N, N]")
    mask_val = mask_val.unsqueeze(-1)          # [B, N, N, 1]

    left = left * mask_val
    right = right * mask_val

    # --------------------------------------------------------------
    # 4. Gating (sigmoid after a linear projection)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(pair_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(pair_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(pair_norm, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core TriMul reduction via Triton kernel
    # --------------------------------------------------------------
    # Move hidden dimension to the second axis: [B, hidden_dim, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # For the right operand we need the k‑dimension first: (B, H, N, N) -> (B, H, N, N) with swapped last two axes
    right_T = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) where dim2 is k, dim3 is j

    # Collapse batch and hidden dimensions: treat each hidden slice as a separate batch
    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)               # (BH, M=N, K=N)
    right_flat = right_T.view(BH, N, N)           # (BH, K=N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)        # (BH, N, N)

    # Choose block sizes based on sequence length to keep occupancy high
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:  # N up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    # Grid definition: one block per tile in M and N, plus batch dimension
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        BH,
    )

    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,          # M
        N,          # N
        N,          # K
        # strides for A (left)
        left_flat.stride(0),   # batch stride
        left_flat.stride(1),   # M‑stride (i)
        left_flat.stride(2),   # K‑stride (k)
        # strides for B (right_T)
        right_flat.stride(0),  # batch stride
        right_flat.stride(1),  # K‑stride (k)
        right_flat.stride(2),  # N‑stride (j)
        # strides for C (out)
        out_flat.stride(0),    # batch stride
        out_flat.stride(1),    # M‑stride
        out_flat.stride(2),    # N‑stride
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Post‑processing: hidden‑dim LayerNorm, out_gate, final linear
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)                     # [B, N, N, hidden_dim]

    out = out * out_gate
    out = F.linear(out, to_out_w)                # back to original dim

    return out
