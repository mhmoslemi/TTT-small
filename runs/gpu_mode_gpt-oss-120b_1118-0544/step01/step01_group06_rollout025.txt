"""
TriMul (outgoing) forward pass – Triton accelerated N³ contraction.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension (dim).
2. Linear projections (left/right) from dim → hidden_dim (bias‑free).
3. Optional mask is broadcast‑multiplied on the hidden dimension.
4. Compute three sigmoid gates (left, right, out) and apply the first two.
5. Core contraction:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batch of GEMMs (left @ rightᵀ) for each (b,d).  The heavy
   N³ work is performed by a fused Triton kernel that tiles the M (i),
   N (j) and K (k) axes (auto‑tuned block sizes) and accumulates in
   FP32 while the inputs are FP16.
6. Layer‑norm over hidden_dim, apply the output gate and a final linear
   projection back to dim.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) with FP16 inputs, FP32 accumulation
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({"BLOCK_M": 64,  "BLOCK_N": 64,  "BLOCK_K": 64}, num_warps=4),
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 64}, num_warps=8),
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 64,  "BLOCK_K": 64}, num_warps=8),
        triton.Config({"BLOCK_M": 64,  "BLOCK_N": 128, "BLOCK_K": 64}, num_warps=8),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions
    stride_ab, stride_am, stride_ak,  # A strides: batch, rows, cols
    stride_bb, stride_bk, stride_bn,  # B strides: batch, rows(K), cols(N)
    stride_cb, stride_cm, stride_cn,  # C strides: batch, rows, cols
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # collapsed batch * hidden_dim
    pid_m = tl.program_id(1)              # block row (i)
    pid_n = tl.program_id(2)              # block col (j)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this (batch, head) slice
    a_batch = a_ptr + pid_batch * stride_ab
    b_batch = b_ptr + pid_batch * stride_bb
    c_batch = c_ptr + pid_batch * stride_cb

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Pointers for the current tile
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (FP16) with masking
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # FP16×FP16→FP32 dot product
        acc += tl.dot(a, b)

    # Write result (FP32) with masking
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=c_mask)


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    Args:
        data: tuple (input_tensor, mask_tensor_or_None, weights_dict, config_dict)
    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.unsqueeze(-1).type_as(x)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. N³ contraction – fused batched GEMM in Triton
    # ------------------------------------------------------------------
    # Cast to half‑precision for the GEMM
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Permute so that each head can be treated as a separate matrix:
    #   left: (B, hidden_dim, i, k)
    #   right (transposed): (B, hidden_dim, k, j)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()  # transpose the two N axes

    B, H, N, _ = left_perm.shape

    # Collapse batch and head dimensions
    left_mat = left_perm.view(B * H, N, N)   # (batch_head, M=i, K=k)
    right_mat = right_perm.view(B * H, N, N) # (batch_head, K=k, N=j)

    # Output buffer (FP32 accumulator)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=x.device)

    # Strides (in elements)
    stride_ab = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bb = right_mat.stride(0)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)

    stride_cb = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # Grid configuration
    #   dim0: batch_head, dim1: M‑tiles, dim2: N‑tiles
    grid = (
        B * H,
        (N + 63) // 64,   # placeholder, will be overridden by autotune
        (N + 63) // 64,
    )

    # Launch the kernel
    _tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,  # M
        N,  # N
        N,  # K
        stride_ab,
        stride_am,
        stride_ak,
        stride_bb,
        stride_bk,
        stride_bn,
        stride_cb,
        stride_cm,
        stride_cn,
        # let Triton pick the optimal block size via autotune
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
