                num_stages: tl.constexpr, dtype: tl.constexpr):

   pid_n = tl.program_id(1)
   pid_batch = tl.program_id(2)

   offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype # assume float32

    # Get dim, hidden_dim
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Extract weights
    # Convert weights to appropriate dtype (float32)
    for each weight: w = weights[key].to(device=device, dtype= torch.float32)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left = F.linear(x, left_proj_weight)
    right = F.linear(x, right_proj_weight)

    if mask is not None:
        mask_f = mask.to(dtype)  # float32
        mask_f = mask_f.unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast left,right to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # Permute to (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch*hidden
    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)
    right_flat = right.view(BH, N, N)

    # Compute batched gemm
    out_flat = batched_gemm(left_flat, right_flat)  # returns float32 [BH, N, N]

    # Reshape back to (B,N,N,hidden)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output Layer norm
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)
    out = out * out_gate

    # Final projection
    output = F.linear(out, to_out_weight)

    return output

"""
custom_kernel for the outgoing Triangle Multiplicative Update (TriMul).

The kernel implements the O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
as a batched GEMM (A @ Bᵀ) where each hidden channel is treated as an
independent batch.  Left and right tensors are cast to float16 to reduce
memory traffic, while the accumulation is performed in float32 using
Tensor‑Core friendly `tl.dot`.  The rest of the TriMul pipeline
(LayerNorm, linear projections, gating, output LayerNorm and final
projection) is handled in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton GEMM kernel: C = A @ Bᵀ   (A: [batch, M, K], B: [batch, N, K])
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr
):
    pid_batch = tl.program_id(2)          # batch (B * hidden)
    pid_m = tl.program_id(0)              # output tile row   (i)
    pid_n = tl.program_id(1)              # output tile col   (j)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # pointers for the current batch
    a_batch_ptr = A_ptr + pid_batch * stride_a_batch
    b_batch_ptr = B_ptr + pid_batch * stride_b_batch
    c_batch_ptr = C_ptr + pid_batch * stride_c_batch

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # masks for the current tile
        mask_a = (mask_m[:, None] & (offs_k[None, :] < K))
        mask_b = ((offs_k[:, None] < K) & mask_n[None, :])

        # load A tile   (M x K)
        a_ptrs = a_batch_ptr + \
                 offs_m[:, None] * stride_a_m + \
                 offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16)

        # load B tile   (K x N) – note transposed view
        b_ptrs = b_batch_ptr + \
                 offs_k[:, None] * stride_b_k + \
                 offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16)

        # accumulate
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch_ptr + \
             offs_m[:, None] * stride_c_m + \
             offs_n[None, :] * stride_c_n
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)


def _batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for a batch of matrices using the Triton kernel above.
    A, B : [batch, M, K] and [batch, N, K] (A, B are float16, contiguous)
    Returns C : [batch, M, N] (float32)
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2, "Inner dimensions must match."

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_m, stride_b_k = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # For Bᵀ we need: row stride = stride over original K dimension (col)
    # col stride = stride over original N dimension (row)
    stride_b_k_tile = stride_b_k          # original column stride (usually 1)
    stride_b_n_tile = stride_b_m          # original row stride (N)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )

    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k_tile,
        stride_b_n_tile,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_stages=3,          # pipelining depth
        num_warps=4,
    )
    return C


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict with at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    x, mask, weights, cfg = data
    device = x.device
    dtype = x.dtype  # assumed float32

    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    # ----- weight extraction ------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=dtype)
    norm_bias   = weights["norm.bias"].to(device=device, dtype=dtype)

    left_proj_weight  = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_weight = weights["right_proj.weight"].to(device=device, dtype=dtype)

    left_gate_weight  = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_weight = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_weight   = weights["out_gate.weight"].to(device=device, dtype=dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device=device, dtype=dtype)

    to_out_weight = weights["to_out.weight"].to(device=device, dtype=dtype)

    # ----- 1. Input LayerNorm -----------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ----- 2. Linear projections (no bias) -----------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden]

    # ----- 3. Optional mask --------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=dtype, device=device).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----- 4. Gates (sigmoid) ------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----- 5. Core TriMul (batched GEMM) ------------------------------------
    B, N, _, _ = left.shape

    # transpose to (B, hidden, N, N) for GEMM friendliness
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # cast to half for the kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # flatten batch*hidden dimension
    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)   # [BH, N, N]  (float16)
    right_flat = right.view(BH, N, N) # [BH, N, N]  (float16)

    # compute A @ Bᵀ for each hidden channel
    out_flat = _batched_gemm(left_flat, right_flat)  # [BH, N, N] float32

    # reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----- 6. Output LayerNorm, gating, final projection --------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, to_out_weight)  # (dim, hidden) weight, no bias

    return output
