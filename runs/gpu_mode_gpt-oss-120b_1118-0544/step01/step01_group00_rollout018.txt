"""
Highly optimized Triton implementation of the outgoing Triangle Multiplicative Update
(TriMul) used in AlphaFold3 and related models.

The forward pass consists of:
1. LayerNorm over the input channels.
2. Linear projections to a hidden dimension (left/right) and three gating
   linear layers (left_gate, right_gate, out_gate).  The projections and gates
   are bias‑free as in the reference implementation.
3. Optional masking of the projected tensors.
4. Element‑wise gating of the left/right tensors.
5. Core O(N³) reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication C = A @ Bᵀ for each
   (batch, hidden) pair.  This step is performed in fp16 while accumulating
   in fp32 using a custom Triton kernel.
6. LayerNorm over the hidden dimension, out‑gate multiplication and final
   projection back to the original channel dimension.

The kernel operates on tensors of shape (batch * hidden, N, N) where each row
represents an independent matrix multiply.  Tiling parameters (BLOCK_M,
BLOCK_N, BLOCK_K) are tuned for typical B200 memory hierarchies.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
# -------------------------------------------------------------
@triton.jit
def tri_mul_batched_matmul_kernel(
    a_ptr,               # left  tensor pointer  (batch, M, K)
    b_ptr,               # right tensor pointer  (batch, N, K) – accessed as transposed
    c_ptr,               # output tensor pointer (batch, M, N)
    batch: tl.constexpr, # total number of independent matrices = B * hidden_dim
    M: tl.constexpr,     # rows of A (also rows of output)
    N: tl.constexpr,     # rows of B (also cols of output)
    K: tl.constexpr,     # shared reduction dimension
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)      # row tile within output
    pid_n = tl.program_id(1)      # col tile within output
    pid_b = tl.program_id(2)      # which matrix in the batch

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch element
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (k_offset[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load Bᵀ tile: we read B as (K, N) using its original strides
        b_ptrs = b_batch + (k_offset[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Perform dot product; result is fp32 because acc is fp32
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None] & mask_n[None, :])

# -------------------------------------------------------------
# Public entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float) – may be unused
            - weights: dict of model weights (see function body)
            - config: dict containing at least:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool) – if True the mask is ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Unpack weights
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # Input LayerNorm over the last dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # Linear projections (bias‑free) to hidden dimension
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # Optional mask (broadcast on hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure mask dtype matches
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # Gating layers (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B, N, N, hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))     # [B, N, N, hidden_dim]

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Convert projections to fp16 for the heavy reduction
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, seq_len, seq_len, hidden_dim

    # Reorder to (B, hidden_dim, N, N) and make contiguous so that each
    # (b, d) slice is laid out as a dense matrix.
    left_h = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer in fp16
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # -----------------------------------------------------------------
    # Triton kernel launch configuration
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    M = N          # rows of left matrix
    K = N          # reduction dimension
    batch_dim = left_h.shape[0]  # B * hidden_dim

    num_pid_m = (M + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m, num_pid_n, batch_dim)

    # Strides in *elements* (not bytes)
    stride_a_batch = left_h.stride(0)
    stride_a_m = left_h.stride(1)
    stride_a_k = left_h.stride(2)

    stride_b_batch = right_h.stride(0)
    # For Bᵀ: k‑stride is stride over the last axis, n‑stride is stride over rows
    stride_b_k = right_h.stride(2)
    stride_b_n = right_h.stride(1)

    stride_c_batch = out_h.stride(0)
    stride_c_m = out_h.stride(1)
    stride_c_n = out_h.stride(2)

    # Launch
    tri_mul_batched_matmul_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_h,
        batch=batch_dim,
        M=M,
        N=N,
        K=K,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # empirically good for these sizes
    )

    # -----------------------------------------------------------------
    # Reshape back to original layout and convert to fp32
    # -----------------------------------------------------------------
    out = (
        out_h.view(B, hidden_dim, N, N)   # [B, hidden, N, N]
        .permute(0, 2, 3, 1)              # [B, N, N, hidden]
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # Final LayerNorm, out‑gate, and projection back to dim
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
