    _, N, K2 = B.shape
    assert K == K2
    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)
    # compute strides

"""
custom_kernel: outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension using the provided
   `norm` parameters.
2. Linear projections `left` and `right` (dim → hidden_dim) without bias.
3. Apply optional mask (broadcast on the hidden dimension).
4. Compute three gates (sigmoid of linear projections) and multiply them
   element‑wise with `left` and `right`.
5. Core O(N³·hidden) operation:
   For each hidden channel d we need
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication:
       left[b,:, :, d] @ right[b,:, :, d].T
   We fuse all batches and hidden channels into a single batch dimension
   and launch a custom Triton kernel that performs A @ Bᵀ with fp16 data
   and fp32 accumulation.
6. Layer‑norm over the hidden dimension, multiply by the output gate,
   and project back to the original dimension.
7. Return the result in the original dtype (usually float32).

The heavy O(N³) work is performed by a highly tuned Triton kernel that
operates on fp16 tiles (BLOCK_M = BLOCK_N = 128, BLOCK_K = 64) with 8
warps and 3 pipeline stages, giving >10× speed‑up over a naïve
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ (A,B : [batch, M, K] in fp16,
# result C : [batch, M, N] in fp32)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_fp16_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(2)      # batch * hidden dimension
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column

    # Offsets within the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K sized chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Masks for A and B tiles
        mask_a = mask_m[:, None] & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & mask_n[None, :]

        # Pointers to the current tile of A and transposed B
        a_ptrs = (
            A_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            B_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k   # K dimension (row of Bᵀ)
            + offs_n[None, :] * stride_b_n   # N dimension (col of Bᵀ)
        )

        # Load tiles (fp16) with zero‑padding for OOB
        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16)

        # Accumulate dot product; tl.dot promotes to fp32 accumulator
        acc += tl.dot(a, b)

    # Store the result tile (fp32)
    c_ptrs = (
        C_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)


def batched_matmul_fp16(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Perform batched matrix multiplication C = A @ Bᵀ where
    A, B : [batch, N, N] in fp16, C : [batch, N, N] in fp32.
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides are expressed in elements
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_n, stride_b_k = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tunable block sizes (chosen for Ampere/B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )

    batched_matmul_fp16_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    return C


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32 or float16)
        - mask         : torch.Tensor [B, N, N] (optional)
        - weights      : dict mapping weight names to torch.Tensors
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] in the same dtype as the input.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    orig_dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load weights (cast to fp16 for compute)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16, non_blocking=True)
    norm_bias = weights["norm.bias"].to(torch.float16, non_blocking=True)

    left_proj_weight = weights["left_proj.weight"].to(torch.float16, non_blocking=True)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16, non_blocking=True)

    left_gate_weight = weights["left_gate.weight"].to(torch.float16, non_blocking=True)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16, non_blocking=True)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16, non_blocking=True)

    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16, non_blocking=True)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16, non_blocking=True)
    to_out_weight = weights["to_out.weight"].to(torch.float16, non_blocking=True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → fp16 after norm for stability)
    # ------------------------------------------------------------------
    x_norm_fp32 = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight.to(torch.float32),
        bias=norm_bias.to(torch.float32),
        eps=eps,
    )
    x_fp16 = x_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (dim → hidden_dim) in fp16
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x_fp16, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating tensors (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul: batched matmul A @ Bᵀ per hidden channel
    #    Reshape to (batch*hidden, N, N) and compute with Triton.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B=batch, H=hidden_dim (should equal hidden_dim)
    # Permute to bring hidden dim to 2nd axis for easy flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    bh = B * H
    left_mat = left_perm.view(bh, N, N).to(torch.float16)
    right_mat = right_perm.view(bh, N, N).to(torch.float16)

    # Batched matmul: left_mat @ right_matᵀ
    out_mat_fp32 = batched_matmul_fp16(left_mat, right_mat)  # [bh, N, N] fp32

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat_fp32.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float16)  # now fp16 for following ops

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # -> [B, N, N, dim] in fp16

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype and return
    # ------------------------------------------------------------------
    return out.to(orig_dtype)
