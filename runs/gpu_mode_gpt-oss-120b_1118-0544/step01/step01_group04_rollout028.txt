3. Optionally apply an input mask, then multiply left/right by their gates.
4. Cast left/right to fp16 and reshape:
   * left  → (B*H, N, N)  (row dimension i, column k)
   * right → (B*H, N, N)  (row dimension k, column j)  – already transposed.
5. Compute the core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched matrix multiplication using a Triton kernel that
   performs fp16×fp16 → fp32 accumulation (tensor‑core friendly).
6. Reshape the result back to (B, N, N, H), apply a second layer‑norm,
   multiply by the output gate and project back to the original dimension.

The Triton kernel is a straightforward tiled GEMM with block sizes
BLOCK_M, BLOCK_N, BLOCK_K (tuned for the NVIDIA B200).  It is the only
non‑Python part of the forward pass.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched matrix multiplication for the TriMul outgoing operation:
    out[b, i, j] = Σ_k left[b, i, k] * right[b, k, j]
    left : (B, M, K)      – fp16
    right: (B, K, N)      – fp16
    out  : (B, M, N)      – fp32
    """
    pid_b = tl.program_id(2)   # batch index (B * hidden_dim)
    pid_m = tl.program_id(0)   # output row block
    pid_n = tl.program_id(1)   # output col block

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch
    left_batch_ptr = left_ptr + pid_b * stride_l_batch
    right_batch_ptr = right_ptr + pid_b * stride_r_batch
    out_batch_ptr = out_ptr + pid_b * stride_o_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load a tile of A (left)
        a = tl.load(
            left_batch_ptr
            + offs_m[:, None] * stride_l_m
            + cur_k[None, :] * stride_l_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a tile of B (right)
        b = tl.load(
            right_batch_ptr
            + cur_k[:, None] * stride_r_k
            + offs_n[None, :] * stride_r_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # GEMM accumulation (fp16 * fp16 -> fp32)
        acc += tl.dot(a, b)

    # Write back the result
    out_ptrs = out_batch_ptr + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, dim]
        - mask        : torch.Tensor of shape [B, N, N] or None
        - weights     : dict with all linear / layer‑norm parameters
        - config      : dict containing at least "dim" and "hidden_dim",
                        optionally "nomask"
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_mask = (not config.get("nomask", False)) and mask is not None

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4. Optional mask
    # ------------------------------------------------------------------
    if use_mask:
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5. Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction via Triton (batched GEMM)
    # ------------------------------------------------------------------
    # Cast to fp16 to exploit tensor cores
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # Reshape for batched GEMM:
    #   left  -> (B*H, N, N)   (i,k)
    #   right -> (B*H, N, N)   (k,j)   (already transposed)
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (FP32 for the accumulated result)
    out_reshape = torch.empty(
        (B * hidden_dim, N, N),
        dtype=torch.float32,
        device=left_reshape.device,
    )

    # Strides (in elements) for the three tensors
    stride_l_batch, stride_l_m, stride_l_k = left_reshape.stride()
    stride_r_batch, stride_r_k, stride_r_n = right_reshape.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_reshape.stride()

    # Tile sizes – good for the B200 (ampere) architecture
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_m, grid_n, B * hidden_dim)

    _tri_mul_outgoing_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N,          # M
        N,          # N
        N,          # K
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_k,
        stride_r_n,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = (
        out_reshape.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 8. Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 9. Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10. Final linear projection back to the original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
