def tri_mul_kernel(a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr):

    pid_m = tl.program_id(1) # row tile id
    pid_n = tl.program_id(2) # col tile id

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

        mask_k = cur_k < K

        a_ptrs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col

   x_norm = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

   # Linear projections
   left = F.linear(x_norm, weights["left_proj.weight"])
   right = F.linear(x_norm, weights["right_proj.weight"])

   # Apply mask (if any)
   if not nomask and mask is not None:
       mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # shape [B,N,N,1]

       right = right * mask_f

   # Gates
   left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))

   # Apply gates to left/right
   left = left * left_gate
   right = right * right_gate

   left = left.to(torch.float16)

"""
TriMul (outgoing) – Triton‑accelerated contraction

The core of the module is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
for a 4‑D tensor [B, N, N, H] (H = hidden_dim).  By reshaping each
hidden slice into a (N×N) matrix we can compute the contraction as a
batched matrix multiplication C = A @ Bᵀ with A = left[b,d] and B = right[b,d].
The kernel below performs this batched GEMM in half‑precision (fp16) while
accumulating in fp32.  All surrounding layer‑norms, linear layers and gates
are kept in PyTorch; only the O(N³) part is offloaded to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ

    A : (batch, M, K)      (fp16)
    B : (batch, N, K)      (fp16)
    C : (batch, M, N)      (fp32)

    The kernel computes one (M,N) tile of one batch slice.
    """
    pid_batch = tl.program_id(0)          # which (batch * hidden_dim) matrix
    pid_m = tl.program_id(1)              # tile row index
    pid_n = tl.program_id(2)              # tile col index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load a tile of A : (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load a tile of Bᵀ : we need (BLOCK_K, BLOCK_N)
        # Original B layout is (N, K); transposed => (K, N)
        b_ptrs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate dot product
        acc += tl.dot(a, b)

    # Write back the result tile
    c_ptrs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” variant.

    Args:
        data (tuple):
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (tensors on the same device)
            - config: dict containing at least "dim" and "hidden_dim",
                      optional "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # ----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ----------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # ----------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------------------
    # 4. Gates (sigmoid after linear)
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ----------------------------------------------------------------------
    # 5. Apply gates to projected tensors
    # ----------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 6. Cast to fp16 for the heavy contraction
    # ----------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # [B, N, N, hidden_dim]

    # ----------------------------------------------------------------------
    # 7. Reshape to (B*hidden_dim, N, N) for batched GEMM
    #    Layout after permute: (B, hidden_dim, N, N) → view → (B*hidden_dim, N, N)
    # ----------------------------------------------------------------------
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (float32 accumulator)
    out_mat = torch.empty(
        (B * hidden_dim, N, N), device=device, dtype=torch.float32
    )

    # ----------------------------------------------------------------------
    # 8. Kernel launch configuration
    # ----------------------------------------------------------------------
    # Choose tile sizes based on sequence length to keep shared memory
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Strides in element counts (row‑major)
    stride_a_batch = N * N
    stride_a_row = N
    stride_a_col = 1

    stride_b_batch = N * N
    stride_b_row = N
    stride_b_col = 1

    stride_c_batch = N * N
    stride_c_row = N
    stride_c_col = 1

    # Launch the Triton kernel
    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ----------------------------------------------------------------------
    # 9. Reshape back to the original 4‑D layout (hidden dim -> last axis)
    # ----------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,hidden_dim]

    # ----------------------------------------------------------------------
    # 10. Output layer‑norm, gating, and final linear projection
    # ----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    out = out * out_gate  # element‑wise gate

    out = F.linear(out, weights["to_out.weight"])  # back to original dim

    return out
