@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # batch index (B * hidden_dim)
    pid_m = tl.program_id(1)  # row tile
    pid_n = tl.program_id(2)  # col tile

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch = a_ptr + pid * stride_a_batch
    b_batch = b_ptr + pid * stride_b_batch
    c_batch = c_ptr + pid * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a_ptrs = a_batch + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        b_ptrs = b_batch + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – fast Triton implementation of the O(N³) contraction

The heavy part of TriMul is:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
which is a batched matrix multiply L @ Rᵀ for every hidden dimension d.
We keep all cheap ops (LayerNorm, linear projections, gating) in PyTorch
and launch a Triton kernel that computes a large number of
independent GEMMs (batch = B * hidden_dim).  The kernel works on
half‑precision inputs (to reduce traffic) while accumulating in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ
    A : (batch, M, K)  (row‑major)
    B : (batch, N, K)  (row‑major) – we read it transposed on‑the‑fly
    C : (batch, M, N)  (row‑major)
    """
    pid_batch = tl.program_id(0)   # index over B*hidden_dim
    pid_m = tl.program_id(1)       # tile over rows (M)
    pid_n = tl.program_id(2)       # tile over cols (N)

    # Offsets of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch slice
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A[i, k] block  (BLOCK_M, BLOCK_K)
        a_offs = a_batch + \
                 offs_m[:, None] * stride_a_row + \
                 cur_k[None, :] * stride_a_col
        a = tl.load(a_offs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Bᵀ[j, k] is accessed as B[k, j]  (BLOCK_K, BLOCK_N)
        b_offs = b_batch + \
                 cur_k[:, None] * stride_b_col + \
                 offs_n[None, :] * stride_b_row
        b = tl.load(b_offs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate product
        acc += tl.dot(a, b)

    # Write C[i, j]
    c_offs = c_batch + \
             offs_m[:, None] * stride_c_row + \
             offs_n[None, :] * stride_c_col
    tl.store(c_offs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask: torch.Tensor [B, N, N] (bool/float) or None
            weights: dict of model parameters (all on the same device)
            config: dict with at least "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------
    # 1. LayerNorm on the input
    # -------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3. Optional mask (broadcasted over the channel dim)
    # -------------------------------------------------
    if (not nomask) and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------
    # 4. Gates (sigmoid of linear projections of x)
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -------------------------------------------------
    # 5. Apply gates
    # -------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6. Prepare for the Triton GEMM (cast to half for traffic)
    # -------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Bring hidden dimension to the batch axis and make tensors contiguous.
    # Shape after view: (B*hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (FP32 accumulator)
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # -------------------------------------------------
    # 7. Triton kernel launch configuration
    # -------------------------------------------------
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
        num_warps = 4
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 128
        num_warps = 8

    stride_a_batch = left_mat.stride(0)
    stride_a_row   = left_mat.stride(1)
    stride_a_col   = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_row   = right_mat.stride(1)
    stride_b_col   = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_row   = out_mat.stride(1)
    stride_c_col   = out_mat.stride(2)

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=3,
    )

    # -------------------------------------------------
    # 8. Reshape back to (B, N, N, hidden_dim)
    # -------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # -------------------------------------------------
    # 9. Output LayerNorm, output gate, final linear projection
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
