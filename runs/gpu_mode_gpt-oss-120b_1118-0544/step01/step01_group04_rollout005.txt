"""
TriMul outgoing operator (AlphaFold3 style) with a fused Triton kernel.

The core contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is equivalent to a batched matrix multiplication
    left[b, d] @ right[b, d].T
for each batch `b` and hidden channel `d`.  This kernel performs the
batched GEMM in FP16 (leveraging Tensor Cores) while accumulating in
FP32 for numerical stability.  All surrounding LayerNorm, linear
projections and gating are kept in PyTorch.

The implementation follows these steps:
1. Layer‑norm the input.
2. Linear projections: left_proj, right_proj.
3. Compute sigmoid gates (left_gate, right_gate, out_gate).
4. Apply optional mask and the gates.
5. Convert left/right to FP16 and reshape to (B*H, N, N).
   Right tensor is transposed on the last two dimensions so a standard
   GEMM yields the desired contraction.
6. Run the Triton `batched_matmul_kernel` (FP16·FP16 → FP32) over the
   batch dimension B*hidden_dim.
7. Reshape back, apply the output LayerNorm, multiply by `out_gate`,
   and a final linear projection to the original `dim`.

The kernel processes one (M,N) tile per program instance and loops over
the K dimension in blocks; out‑of‑range accesses are masked.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B.T)
# A: (batch, M, K)  B: (batch, K, N)   C: (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # strides for A (batch, M, K)
    stride_bk, stride_bn,          # strides for B (batch, K, N)
    stride_cm, stride_cn,          # strides for C (batch, M, N)
    stride_a_batch, stride_b_batch, stride_c_batch,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile index over M
    pid_n = tl.program_id(1)   # tile index over N
    pid_b = tl.program_id(2)   # batch index (B * hidden_dim)

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch (without K offset)
    a_base = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_am
    b_base = b_ptr + pid_b * stride_b_batch + offs_n[None, :] * stride_bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K‑dimension loop
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # load A block (M x K_tile)
        a = tl.load(
            a_base + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B block (K_tile x N)
        b = tl.load(
            b_base + cur_k[:, None] * stride_bk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # matrix multiply‑accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) in FP32

    # write back result
    c_ptrs = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel for the TriMul (outgoing) operator.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Heavy contraction is performed by a fused Triton batched GEMM.
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # Extract shapes / config
    # ------------------------------------------------------------------
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    use_mask = (not config.get("nomask", False)) and (mask is not None)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if use_mask:
        mask_exp = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Heavy contraction via Triton (batched GEMM)
    # ------------------------------------------------------------------
    # Convert to half for Tensor‑Core speed
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Reshape to (B*hidden_dim, N, N)
    # left: (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    # right needs to be transposed on the last two axes to implement the
    # sum over k: right_T = right_perm.permute(0, 3, 2, 1)
    right_t = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)

    # Output buffer (FP32 accumulator)
    out_reshape = torch.empty(
        (bs * hidden_dim, N, N),
        dtype=torch.float32,
        device=left_reshape.device,
    )

    # Strides: (batch, M, K) for left; (batch, K, N) for right_t; (batch, M, N) for out
    stride_a_batch, stride_am, stride_ak = left_reshape.stride()
    stride_b_batch, stride_bk, stride_bn = right_t.stride()
    stride_c_batch, stride_cm, stride_cn = out_reshape.stride()

    # Tile sizes – tuned for modern NVIDIA GPUs (Tensor‑Core friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # small K block fits well in shared memory

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )

    # Launch the kernel
    batched_matmul_kernel[grid](
        left_reshape,
        right_t,
        out_reshape,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7. Reshape back to [bs, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_reshape.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 9. Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
