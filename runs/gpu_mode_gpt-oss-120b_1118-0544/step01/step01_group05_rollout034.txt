```
def custom_kernel(data):
    x, mask, weights, config = data
    # device, dtype
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # projections
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # cast to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # allocate output buffer
    out = torch.empty((B, N, N, hidden_dim), device=device, dtype=torch.float32)

    # compute strides
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_d = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_i = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_d = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_d = out.stride(3)

    # block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        N,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_d,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_d,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
        num_warps=8,
        num_stages=3,
    )
    # After kernel, out shape [B,N,N,hidden_dim]

    # Output layer norm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Apply out gate
    out = out * out_gate

    # Final linear: dim output features
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul outgoing kernel (AlphaFold3) – fast batched GEMM in Triton.

Given tensors
    left  ∈ ℝ^{B×N×N×H}   (half)
    right ∈ ℝ^{B×N×N×H}   (half)
the kernel computes for every batch b and hidden channel h
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

which is mathematically a batched matrix multiplication
    C = A @ Bᵀ   with A = left[b, :, :, h], B = right[b, :, :, h].

The implementation works directly on the original
[B, N, N, H] layout (no costly permute + contiguous copy),
uses fp16 inputs with fp32 accumulation, and fuses the
N³ contraction into a single Triton kernel.  The surrounding
LayerNorms, linear projections and gating are performed in
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched A @ Bᵀ where A and B have shape (B, N, N, H).
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to left, right, out
    N,                                   # sequence length
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_d,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    """Batched matmul C = A @ Bᵀ for tensors of shape (B,N,N,H)."""
    pid0 = tl.program_id(0)      # combined batch*hidden index
    pid_i = tl.program_id(1)      # tile over output rows (i)
    pid_j = tl.program_id(2)      # tile over output cols (j)

    # Compute the batch and hidden‑dim indices
    batch_idx = pid0 // HIDDEN_DIM
    hidden_idx = pid0 % HIDDEN_DIM

    # Base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch_idx * stride_a_batch + hidden_idx * stride_a_d
    b_base = b_ptr + batch_idx * stride_b_batch + hidden_idx * stride_b_d
    c_base = c_ptr + batch_idx * stride_c_batch + hidden_idx * stride_c_d

    # Offsets within the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k in blocks of size BLOCK_K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load A block: shape (BLOCK_M, BLOCK_K)  -> left[b,i,k,h]
        a_offs = a_base \
                 + offs_i[:, None] * stride_a_i \
                 + cur_k[None, :] * stride_a_k
        a = tl.load(a_offs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B block transposed on‑the‑fly:
        # original layout right[b,j,k,h]; we need (k,j) -> shape (BLOCK_K, BLOCK_N)
        b_offs = b_base \
                 + cur_k[:, None] * stride_b_k \
                 + offs_j[None, :] * stride_b_i
        b = tl.load(b_offs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Accumulate dot product (fp16 inputs, fp32 accumulation)
        acc += tl.dot(a, b)

    # Store result block: out[b,i,j,h]
    c_offs = c_base \
             + offs_i[:, None] * stride_c_i \
             + offs_j[None, :] * stride_c_j
    tl.store(c_offs, acc, mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Entry point used by the testing harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” variant.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (torch tensors on same device)
            - config: dict containing at least "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) LayerNorm over the feature dimension
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2) Linear projections (float32) – cheap relative to the O(N³) term
    # --------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4) Gating (sigmoid of a linear projection)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Cast to half‑precision for the heavy contraction
    # --------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # --------------------------------------------------------------
    # 6) Allocate output buffer (float32 accumulator)
    # --------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), device=device, dtype=torch.float32)

    # --------------------------------------------------------------
    # 7) Compute strides (in elements, not bytes)
    # --------------------------------------------------------------
    stride_a_batch = left.stride(0)
    stride_a_i     = left.stride(1)
    stride_a_k     = left.stride(2)
    stride_a_d     = left.stride(3)

    stride_b_batch = right.stride(0)
    stride_b_i     = right.stride(1)
    stride_b_k     = right.stride(2)
    stride_b_d     = right.stride(3)

    stride_c_batch = out.stride(0)
    stride_c_i     = out.stride(1)
    stride_c_j     = out.stride(2)
    stride_c_d     = out.stride(3)

    # --------------------------------------------------------------
    # 8) Triton kernel launch parameters
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # combined batch‑hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,       # tiles over j
    )

    # Launch kernel
    tri_mul_kernel[grid](
        left,
        right,
        out,
        N,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_a_d,
        stride_b_batch,
        stride_b_i,
        stride_b_k,
        stride_b_d,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        stride_c_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
        num_warps=8,
        num_stages=3,
    )

    # --------------------------------------------------------------
    # 9) Post‑contraction processing (LayerNorm, gating, final projection)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
