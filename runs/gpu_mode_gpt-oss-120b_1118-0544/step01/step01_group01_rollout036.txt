```
@triton.jit
def fuse_gate_mask_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    mask_ptr,
    N_elems,
    has_mask: tl.constexpr,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_elems
    left = tl.load(left_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)
    if has_mask:
        mask_val = tl.load(mask_ptr + offs, mask=mask)
        left = left * mask_val
        right = right * mask_val
    # apply gates
    left = left * left_gate
    right = right * right_gate
    tl.store(out_left_ptr + offs, left, mask=mask)
    tl.store(out_right_ptr + offs, right, mask=mask)
```

"""
Fast outgoing TriMul implementation.

The operator first applies a LayerNorm and several linear projections.
A fused Triton kernel then multiplies the projected tensors with their
corresponding sigmoid‑gates and an optional mask, producing FP16 tensors.
The heavy N³ reduction is performed as a batched matrix multiplication
(C = A @ Bᵀ) with torch.bmm (FP16, tensor‑core accelerated).  Finally,
the result is Layer‑Normed, gated, and linearly projected back to the
original dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _fused_gate_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    mask_ptr,                     # may be 0 if no mask
    total_elems,
    B, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # Load FP32 values
    l = tl.load(left_ptr + offs, mask=valid, other=0.0, dtype=tl.float32)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0, dtype=tl.float32)
    r = tl.load(right_ptr + offs, mask=valid, other=0.0, dtype=tl.float32)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0, dtype=tl.float32)

    # Apply sigmoid‑gates (already applied outside, we just multiply)
    l = l * lg
    r = r * rg

    if HAS_MASK:
        # Compute the corresponding mask index.
        # Tensor layout: (B, H, N, N) -> index = ((b*H + h)*N + i)*N + j
        # Mask layout: (B, N, N)   -> index = (b*N + i)*N + j
        bn = H * N * N                     # elements per batch slice
        b = offs // bn                     # batch index
        rest = offs - b * bn               # remaining offset within batch slice
        i = (rest // N) % N                # row within N×N
        j = rest % N                       # column within N×N
        mask_idx = b * (N * N) + i * N + j
        m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0, dtype=tl.float32)
        l = l * m
        r = r * m

    # Cast to FP16 for the GEMM
    l = tl.cast(l, tl.float16)
    r = tl.cast(r, tl.float16)

    tl.store(left_out_ptr + offs, l, mask=valid)
    tl.store(right_out_ptr + offs, r, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N] or None
        - weights_dict : mapping name -> torch.Tensor (all on CUDA)
        - config_dict  : contains "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs ---------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    B, N, _, _ = x.shape

    # 1) Input LayerNorm ----------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), norm_w, norm_b)

    # 2) Linear projections --------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])            # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,hidden_dim]

    # 3) Fuse mask & gates into FP16 tensors via Triton --------------------
    has_mask = mask is not None
    if has_mask:
        mask = mask.to(x.dtype)  # float32 for multiplication

    # Bring channel dimension next to batch for contiguous layout
    left = left.permute(0, 3, 1, 2).contiguous()        # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous()

    total_elements = left.numel()                      # B*H*N*N

    # Allocate FP16 output buffers
    left_fp16 = torch.empty(total_elements, dtype=torch.float16, device=x.device)
    right_fp16 = torch.empty(total_elements, dtype=torch.float16, device=x.device)

    # Prepare kernel launch
    BLOCK_SIZE = 1024                                   # ≤ max threads per block
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    # Pointers
    left_ptr = left.data_ptr()
    left_gate_ptr = left_gate.data_ptr()
    right_ptr = right.data_ptr()
    right_gate_ptr = right_gate.data_ptr()
    left_out_ptr = left_fp16.data_ptr()
    right_out_ptr = right_fp16.data_ptr()
    mask_ptr = mask.data_ptr() if has_mask else 0

    _fused_gate_mask_kernel[grid](
        left_ptr, left_gate_ptr,
        right_ptr, right_gate_ptr,
        left_out_ptr, right_out_ptr,
        mask_ptr,
        total_elements,
        B, hidden_dim, N,
        HAS_MASK=1 if has_mask else 0,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # 4) Batched matrix multiplication (C = A @ Bᵀ) -------------------------
    left_fp16 = left_fp16.view(B, hidden_dim, N, N)          # [B, H, N, N]
    right_fp16 = right_fp16.view(B, hidden_dim, N, N)

    left_mat = left_fp16.view(B * hidden_dim, N, N)          # (B*H, N, N)
    right_mat = right_fp16.view(B * hidden_dim, N, N)

    # Transpose the second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N), FP16

    # Restore original layout and cast back to FP32
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)
    # out_hidden shape: [B, N, N, hidden_dim]

    # 5) Output LayerNorm, gating and final linear projection ----------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_hidden = F.layer_norm(out_hidden, (hidden_dim,), to_out_norm_w, to_out_norm_b)

    # Apply the output gate (already in float32)
    out_hidden = out_hidden * out_gate

    # Final projection back to original dimension
    out = F.linear(out_hidden, weights["to_out.weight"])   # [B, N, N, dim]

    return out
