"""
TriMul outgoing update (alphaFold3) – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the pair representation x ∈ [B, N, N, C].
2. Linear projections (bias‑free) → left, right (shape [B, N, N, H]).
3. Optional pair mask is applied, then three gates are computed from x
   (sigmoid(linear(...))) and multiplied into left/right.
4. Core update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication:
        left_flat  ∈ [B·H, N, N]   (M=N, K=N)
        right_T    ∈ [B·H, N, N]   (K=N, N)
        out_flat   = left_flat @ right_T   →  [B·H, N, N]
   Implemented with a custom Triton kernel that tiles the output
   (BLOCK_M × BLOCK_N) and iterates over K in BLOCK_K steps, using
   fp16 loads and fp32 accumulation (tensor‑cores).
5. Hidden‑dim layer‑norm, multiply by out_gate, final linear back to dim.

Only the heavy N³ reduction (step 4) is performed in Triton; all
other operations use PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _trigemul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix sizes (int)
    stride_a_batch, stride_am, stride_ak,   # A strides (batch, M, K)
    stride_b_batch, stride_bk, stride_bn,   # B strides (batch, K, N) (B is already transposed)
    stride_c_batch, stride_cm, stride_cn,   # C strides (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matmul C = A @ B where
        A: [batch, M, K]   (fp16)
        B: [batch, K, N]   (fp16, already transposed)
        C: [batch, M, N]   (fp32 accumulator, stored as fp32)
    """
    pid_batch = tl.program_id(2)      # batch dimension (B*H)
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Pointers for the current tile
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        b_ptrs = b_batch_ptr + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles (fp16) with out‑of‑bounds masking
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Matrix multiplication of the tile (uses tensor cores)
        acc += tl.dot(a, b)

    # Write the output tile (fp32)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative
    Update (TriMul).

    Parameters
    ----------
    data : tuple
        (pair_tensor, mask_tensor, weights_dict, config_dict)
        - pair_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : mapping from weight name to torch.Tensor
        - config_dict  : contains "dim" and "hidden_dim" (ints) and optionally
                         "layernorm_eps" (float)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) after the TriMul update.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    pair_tensor, mask, weights, config = data
    device = pair_tensor.device
    B, N, N2, C = pair_tensor.shape
    assert N == N2, "pair_tensor must be square in the sequence dimensions"
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm over channel dimension C
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        pair_tensor,
        (C,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.to(dtype=pair_tensor.dtype, device=device).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for batched matmul kernel
    #   left:  [B, N, N, H] → [B, H, N, N] → flatten batch*H
    #   right: same but transposed on the last two dims (K,N)
    # ------------------------------------------------------------------
    # Move hidden dim to second axis and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N]

    # Flatten batch and hidden dimension
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)        # [BH, M=N, K=N]
    # Transpose the inner dimensions for the right operand: K,N
    right_T = right_perm.transpose(-2, -1).contiguous().view(BH, N, N)

    # Cast to half‑precision for the kernel (tensor‑core friendly)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_T.to(torch.float16)

    # Output buffer (float32, because accumulator is fp32)
    out_flat = torch.empty_like(left_fp16, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 6. Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        BH,
    )

    _trigemul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_flat,
        N,               # M
        N,               # N
        N,               # K
        left_fp16.stride(0),  # batch stride for A
        left_fp16.stride(1),  # stride_am (row)
        left_fp16.stride(2),  # stride_ak (col)
        right_fp16.stride(0), # batch stride for B
        right_fp16.stride(1), # stride_bk (K)
        right_fp16.stride(2), # stride_bn (N)
        out_flat.stride(0),   # batch stride for C
        out_flat.stride(1),   # stride_cm (row)
        out_flat.stride(2),   # stride_cn (col)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Hidden‑dim layer‑norm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # back to original dim

    return out
