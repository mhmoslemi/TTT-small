"""
TriMul (outgoing) forward pass optimized with:
1. PyTorch for LayerNorm, Linear projections and gating.
2. A tiny Triton kernel that fuses mask application and gating
   (eliminating an extra pass over the large tensors).
3. Batched half‑precision GEMM (torch.bmm) for the O(N³) contraction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   performed as (B*D) independent matrix multiplications
   left_mat @ right_matᵀ with Tensor‑Core acceleration.
The implementation respects the required signature `custom_kernel(data)`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    lhs_ptr, lhs_gate_ptr,               # left & its gate
    rhs_ptr, rhs_gate_ptr,               # right & its gate
    mask_ptr,                            # mask of shape (B, N, N, 1) flattened
    lhs_out_ptr, rhs_out_ptr,            # output tensors (float32)
    total_elements, hidden_dim,          # scalar arguments
    BLOCK_SIZE: tl.constexpr,            # launch‑time block size
):
    """
    Compute
        lhs_out = lhs * lhs_gate * mask
        rhs_out = rhs * rhs_gate * mask
    where ``mask`` is broadcast over the hidden‑dim axis.
    All tensors are flattened, contiguous and have shape
    (B, N, N, hidden_dim) → total_elements = B*N*N*hidden_dim.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    in_range = offs < total_elements

    # load values
    lhs = tl.load(lhs_ptr + offs, mask=in_range, other=0.0)
    lhs_gate = tl.load(lhs_gate_ptr + offs, mask=in_range, other=0.0)
    rhs = tl.load(rhs_ptr + offs, mask=in_range, other=0.0)
    rhs_gate = tl.load(rhs_gate_ptr + offs, mask=in_range, other=0.0)

    # mask index = floor(idx / hidden_dim)
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=in_range, other=1.0)

    lhs_out = lhs * lhs_gate * mask_val
    rhs_out = rhs * rhs_gate * mask_val

    tl.store(lhs_out_ptr + offs, lhs_out, mask=in_range)
    tl.store(rhs_out_ptr + offs, rhs_out, mask=in_range)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (all on the same device)
            - config: dict containing at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- 1. Input LayerNorm ----------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- 2. Linear projections ----------
    left_raw = F.linear(x_norm, weights["left_proj.weight"])
    right_raw = F.linear(x_norm, weights["right_proj.weight"])

    # ---------- 3. Gating values ----------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Ensure all tensors are contiguous for the Triton kernel
    left_raw = left_raw.contiguous()
    right_raw = right_raw.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # ---------- 4. Apply mask + gates (fused Triton kernel) ----------
    if (not nomask) and (mask is not None):
        # mask: [B, N, N] -> [B, N, N, 1] and keep as float32
        mask_f = mask.to(x.dtype).unsqueeze(-1).contiguous()
        # allocate output buffers
        left = torch.empty_like(left_raw)
        right = torch.empty_like(right_raw)

        total = left_raw.numel()                     # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        _gate_mask_kernel[grid](
            left_raw,
            left_gate,
            right_raw,
            right_gate,
            mask_f,
            left,
            right,
            total,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Release unneeded tensors early
        del left_raw, right_raw, left_gate, right_gate, mask_f
        torch.cuda.empty_cache()
    else:
        # No mask: simple element‑wise multiplication
        left = left_raw * left_gate
        right = right_raw * right_gate
        del left_raw, right_raw, left_gate, right_gate
        torch.cuda.empty_cache()

    # ---------- 5. Cast to half‑precision for the heavy contraction ----------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # now hidden_dim == config["hidden_dim"]

    # reshape to (B*hidden_dim, N, N) – batch of independent GEMMs
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # ---------- 6. Batched GEMM:   out = left @ rightᵀ ----------
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N), dtype=half
    # convert back to float32 and reshape to original 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # release half‑precision buffers
    del left, right, left_mat, right_mat, out_mat
    torch.cuda.empty_cache()

    # ---------- 7. Output LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------- 8. Apply output gate ----------
    out = out * out_gate

    # ---------- 9. Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
