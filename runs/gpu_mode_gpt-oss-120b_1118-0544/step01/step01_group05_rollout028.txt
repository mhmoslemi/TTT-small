def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # Linear Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if needed
    if (not nomask) and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half for matmul kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # Rearrange to (B*hidden, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # Get strides

    stride_b_batch, stride_b_row, stride_b_col = right_mat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_mat.stride()

"""
Highly‑optimized Triton implementation of the TriMul “outgoing” update.

The expensive O(N³) contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is reduced to a batched half‑precision GEMM (A @ Bᵀ) where each
hidden‑dim slice d is a separate matrix multiplication.
The kernel uses 128×128 tiles, 64‑wide K‑tiles and Tensor‑Core
fp16·fp16→fp32 accumulation.  All LayerNorm, Linear and gating
operations are done in PyTorch; only the contraction is performed
in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _triton_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ.
    A and B: (batch, M, K) and (batch, N, K)  (row‑major)
    C: (batch, M, N)  (row‑major, fp32 accumulator)
    """
    pid_batch = tl.program_id(0)      # which (batch * hidden_dim) matrix
    pid_m = tl.program_id(1)          # tile over rows   (M)
    pid_n = tl.program_id(2)          # tile over cols   (N)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this matrix
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_offs = a_batch \
                 + offs_m[:, None] * stride_a_row \
                 + cur_k[None, :] * stride_a_col
        a = tl.load(a_offs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load Bᵀ tile on‑the‑fly: shape (BLOCK_K, BLOCK_N)
        # Original B layout is (N, K); we access (K, N) by swapping strides.
        b_offs = b_batch \
                 + cur_k[:, None] * stride_b_col \
                 + offs_n[None, :] * stride_b_row
        b = tl.load(b_offs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate block
        acc += tl.dot(a, b)

    # Write back C tile
    c_offs = c_batch \
             + offs_m[:, None] * stride_c_row \
             + offs_n[None, :] * stride_c_col
    tl.store(c_offs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (tensors on the same device)
            - config: dict with at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm ----------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------- Linear projections ----------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if (not nomask) and mask is not None:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gates ----------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ---------- Apply gates ----------
    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to half for the Triton GEMM ----------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape                         # B, seq_len, seq_len, hidden_dim

    # Rearrange to (B*hidden_dim, N, N) – each slice is a dense N×N matrix
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer for the batched GEMM
    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # Strides in element counts (not bytes)
    stride_a_batch, stride_a_row, stride_a_col = left_mat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_mat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_mat.stride()

    # ---------- Triton kernel launch ----------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                      # batch * hidden_dim
        (N + BLOCK_M - 1) // BLOCK_M,        # rows
        (N + BLOCK_N - 1) // BLOCK_N,        # cols
    )
    _triton_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ---------- Reshape back to [B, N, N, hidden_dim] ----------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
