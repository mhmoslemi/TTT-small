def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # everything is cuda
    device = input_tensor.device
    dtype = input_tensor.dtype

    # dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # fetch weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1. Input LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # 2. Linear projections
    left = F.linear(x, left_proj_weight)  # shape [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # 3. Mask
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(dtype)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # 4. Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # 5. Core tri-multiplicative update using Triton batched matrix multiplication
    B, N, _, _ = left.shape

    # Permute to (B, hidden, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_hidden = B * hidden_dim

    left_mat = left_perm.view(batch_hidden, N, N)    # shape (BH, N, N)
    right_mat = right_perm.view(batch_hidden, N, N)

    # Compute batched matmul: out = left_mat @ right_mat^T
    out_mat = batched_triton_matmul(left_mat, right_mat)  # shape (BH, N, N) float32

    # Reshape back to (B,N,N,hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output LayerNorm, gating, final linear
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    out = out * out_gate

    # Final projection
    output = F.linear(out, to_out_weight)  # (dim, hidden_dim)
    return output

"""
Optimized Triton implementation of the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (float32) on the last channel.
2. Linear projections (left/right) and gating linear layers are computed with
   torch.nn.functional.linear (no bias).  The mask (if present) is applied,
   followed by element‑wise sigmoid gating.
3. The core O(N³) operation:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is realised as a batched matrix multiplication A @ Bᵀ where
   A and B have shape (B·hidden, N, N).  The multiplication is performed
   by a custom Triton kernel that tiles the matrices (BLOCK_M×BLOCK_N×BLOCK_K),
   loads tiles in fp16, accumulates in fp32 and writes the result back in fp16.
   The kernel is launched with a 3‑D grid (M‑tiles, N‑tiles, batch‑tiles).
4. The result is reshaped back to (B, N, N, hidden), a second LayerNorm is
   applied, multiplied by the output‑gate, and finally projected back to the
   original channel dimension.
5. All heavy work (the N³ contraction) lives in Triton; the rest uses
   efficient PyTorch ops.  The implementation supports optional masking,
   arbitrary batch size (1‑2), sequence lengths up to 1024, and hidden
   dimensions as in the AlphaFold‑3 configs.

The kernel works with torch.float16 internally to reduce memory bandwidth,
while the accumulation is performed in float32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ Bᵀ for a batch of matrices.
    A : [batch, M, K]   (fp16)
    B : [batch, N, K]   (fp16) – will be read as K×N (i.e. transposed)
    C : [batch, M, N]   (fp16, accumulation in fp32)
    """
    pid_batch = tl.program_id(2)  # batch dimension
    pid_m = tl.program_id(0)      # row block
    pid_n = tl.program_id(1)      # col block

    # Offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols (used for loading and storing)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Masks for the current tile
        mask_a = mask_m[:, None] & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & mask_n[None, :]

        # Pointers to tiles of A and transposed B
        a_ptrs = (
            A_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            B_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k   # K dimension (row of Bᵀ)
            + offs_n[None, :] * stride_b_n   # N dimension (col of Bᵀ)
        )

        # Load tiles (zero‑pad out‑of‑bounds)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # Accumulate the dot product (fp16 inputs, fp32 accumulation)
        acc += tl.dot(a, b)

    # Write the resulting tile
    c_ptrs = (
        C_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)


def _batched_triton_matmul(A_fp32: torch.Tensor, B_fp32: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication A @ Bᵀ using the Triton kernel above.
    Input tensors are float32 with shape (batch, N, N).  Returns a float32
    tensor of the same shape containing the result.
    """
    # Cast to fp16 for the kernel (memory‑bandwidth reduction)
    A = A_fp32.to(torch.float16)
    B = B_fp32.to(torch.float16)

    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2, "Inner dimensions must match"

    # Output buffer in fp16, will be converted back after kernel
    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in elements (not bytes)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_n, stride_b_k = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Tunable block sizes – 64 works well for typical B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    # Convert back to fp32 for the remaining PyTorch pipeline
    return C.to(torch.float32)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32, on CUDA)
        - mask         : torch.Tensor of shape [B, N, N] (float32 or bool) or None
        - weights      : dict of model parameters (all float32 tensors on the same device)
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    # Layer‑norm before projection
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]     # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]   # (hidden_dim, dim)

    # Gating linear layers (no bias)
    left_gate_weight = weights["left_gate.weight"]     # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]   # (hidden_dim, dim)
    out_gate_weight = weights["out_gate.weight"]       # (hidden_dim, dim)

    # Output layers
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]           # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over channel dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast projected tensors to fp16 before the heavy O(N³) contraction
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Core triangle update using batched matmul (A @ Bᵀ)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # batch size, sequence length

    # Rearrange to [B, hidden, N, N] (contiguous) and flatten batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_hidden = B * hidden_dim
    left_mat = left_perm.view(batch_hidden, N, N)     # [BH, N, N] in fp16
    right_mat = right_perm.view(batch_hidden, N, N)   # [BH, N, N] in fp16

    # Perform batched matrix multiplication: out = left_mat @ right_matᵀ
    out_mat = _batched_triton_matmul(left_mat, right_mat)   # [BH, N, N] fp32

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    output = F.linear(out, to_out_weight)   # (dim, hidden_dim)

    return output
