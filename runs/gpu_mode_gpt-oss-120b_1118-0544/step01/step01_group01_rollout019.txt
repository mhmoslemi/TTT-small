"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy O(N³) part is the batched matrix multiplication
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
which is equivalent to a separate GEMM per hidden channel h:
    out_h = left_h @ right_hᵀ.
We fuse this computation across the (batch, hidden) dimension in a
single Triton kernel that operates on FP16 data (accumulating in FP32).
All other operations (LayerNorm, linear projections, gating, final
projection) are performed with PyTorch.  The kernel tiles the
output with BLOCK_M×BLOCK_N×BLOCK_K and uses `tl.dot` to exploit
Tensor‑Core hardware on NVIDIA B200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for left @ rightᵀ
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                      # matrix dimensions (all equal to seq_len)
    stride_am, stride_ak,         # strides for A (left)   : (batch_h, M, K)
    stride_bk, stride_bn,         # strides for B (rightᵀ) : (batch_h, K, N)
    stride_cm, stride_cn,         # strides for C (out)    : (batch_h, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_bh = tl.program_id(2)         # batch * hidden index

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this (batch, hidden) slice
    a_base = A_ptr + pid_bh * (M * K)
    b_base = B_ptr + pid_bh * (K * N)
    c_base = C_ptr + pid_bh * (M * N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K) + k_offset

        # Pointers for the current K‑block
        a_ptrs = a_base + offs_m[:, None] * stride_am + k_range[None, :] * stride_ak
        b_ptrs = b_base + k_range[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (k_range[None, :] < K)
        b_mask = (k_range[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float16)

        # Accumulate the product (FP16*FP16 -> FP32)
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/int) or None
        - weights_dict : dict with all weight & bias tensors
        - config_dict  : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections & gating (still in FP32)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # Optional mask (broadcast over hidden dim)
    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gating values (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Batched GEMM via Triton (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Cast to half for the kernel (Tensor cores)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to (B, hidden, N, N) – each (b,h) slice is a contiguous matrix
    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    # Collapse batch & hidden into a single dimension for the kernel
    left_mat = left_h.view(-1, N, N)    # (B*H, M=N, K=N)
    right_mat = right_h.view(-1, N, N)  # (B*H, K=N, N) – we will treat it as transposed inside the kernel
    out_mat = torch.empty_like(left_mat)  # (B*H, N, N), FP16

    # Strides (in elements, not bytes)
    stride_am = left_mat.stride(1)   # stride on M (rows) = N
    stride_ak = left_mat.stride(2)   # stride on K (cols) = 1

    # For the transposed view of right (K,N)
    stride_bk = right_mat.stride(2)  # stride on K (fast dim) = 1
    stride_bn = right_mat.stride(1)  # stride on N (row stride) = N

    stride_cm = out_mat.stride(1)    # output row stride = N
    stride_cn = out_mat.stride(2)    # output col stride = 1

    # Tile sizes – 64×64 work well on B200 with Tensor‑Core FP16
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,               # M
        N,               # N
        N,               # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden)
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # ------------------------------------------------------------------
    # 4) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply output gate (shape broadcastable to [B,N,N,hidden])
    out_hidden = out_hidden * out_gate

    # Final linear projection back to original dim
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
