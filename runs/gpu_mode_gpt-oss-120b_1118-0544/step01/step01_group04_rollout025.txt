"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy part of the operation is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalently a batched matrix multiplication:
    left_reshape (B*D, N, N) @ right_reshape (B*D, N, N).T
The kernel computes this product in FP16 (inputs) with FP32 accumulation.
All other linear layers and LayerNorms are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (batch, M, K)
    stride_bk, stride_bn,          # B: (batch, K, N)
    stride_cm, stride_cn,          # C: (batch, M, N)
    stride_a_batch, stride_b_batch, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ B  (A: MxK, B: KxN) for all batches."""
    pid_batch = tl.program_id(2)          # batch index (collapsed bs*hidden_dim)
    pid_m = tl.program_id(0)              # tile row index
    pid_n = tl.program_id(1)              # tile col index

    # Compute base pointers for the current batch
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load a tile of A and B
        a = tl.load(
            a_batch + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_batch + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Perform dot‑product (FP16 inputs → FP32 accumulation)
        acc += tl.dot(a, b)

        # Advance pointers for the next K‑tile
        # (Not strictly needed because we re‑compute a_batch each iteration,
        #  but keeps the pattern symmetric with many Triton examples.)
        # a_batch += BLOCK_K * stride_ak
        # b_batch += BLOCK_K * stride_bk

    # Store the result
    c = acc
    tl.store(
        c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask:        torch.Tensor [B, N, N] (float32/int) or None
            weights:     dict of model parameters (torch.Tensors)
            config:      dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    bs, N, _, _ = input_tensor.shape

    ########################################
    # 1. Input LayerNorm
    ########################################
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    ########################################
    # 2. Linear projections (no bias)
    ########################################
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    ########################################
    # 3. Gating (sigmoid)
    ########################################
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    ########################################
    # 4. Optional mask
    ########################################
    if (mask is not None) and (not nomask):
        mask_exp = mask.unsqueeze(-1)            # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    ########################################
    # 5. Apply gates
    ########################################
    left = left * left_gate
    right = right * right_gate

    ########################################
    # 6. Fuse the contraction with a Triton kernel
    ########################################
    # Switch to FP16 for the large pairwise tensors (memory‑bound)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange to (batch, hidden_dim, N, N) and collapse batch*hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()          # (B, D, N, N) -> A[i,k]
    right = right.permute(0, 3, 2, 1).contiguous()        # (B, D, N, N) -> B[k,j]
    left_reshape = left.view(bs * hidden_dim, N, N)       # (B*D, M=N, K=N)
    right_reshape = right.view(bs * hidden_dim, N, N)     # (B*D, K=N, N=N)

    # Output buffer (FP32 accumulator)
    out_reshape = torch.empty(
        (bs * hidden_dim, N, N), dtype=torch.float32, device=device
    )

    # Strides (in elements, not bytes)
    stride_a_batch, stride_am, stride_ak = left_reshape.stride()
    stride_b_batch, stride_bk, stride_bn = right_reshape.stride()
    stride_c_batch, stride_cm, stride_cn = out_reshape.stride()

    # Tile sizes – empirically good for B200 and the given problem sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_m, grid_n, bs * hidden_dim)

    # Launch the kernel
    _batched_gemm_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = (
        out_reshape.view(bs, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    ########################################
    # 7. Output LayerNorm over hidden_dim
    ########################################
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    ########################################
    # 8. Output gating
    ########################################
    out = out * out_gate

    ########################################
    # 9. Final linear projection back to original dim
    ########################################
    out = F.linear(out, weights["to_out.weight"])

    return out
