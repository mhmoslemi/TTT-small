class TriMul:

    def forward(self, x, mask=None):
        # LayerNorm on x
        mean = np.mean(x**2, axis=-1, keepdims=True)
        variance = np.mean((x - mean) ** 2, axis=-1, keepdimensions=True)
        normalized_x = (x - mean) / np.sqrt(variance + 1e-5)  # Subtracting mean and adding epsilon

        #Linearprojections
        left_weights = np.random.randn(self.hidden_dim, self.dim)
        right_weights = np.random.randn(self.hidden_dim, self.dim)
