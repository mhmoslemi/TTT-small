```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for batched GEMM: out = A @ B where B is already transposed
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute 3D index
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_id = pid // (num_blocks_m * num_blocks_n)
    pid = pid % (num_blocks_m * num_blocks_n)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    # offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointer offsets for A and B
    a_ptrs = a_ptr + batch_id * stride_Ab + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = b_ptr + batch_id * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # store result
    c_ptrs = c_ptr + batch_id * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    mask_c = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc, mask=mask_c)

def custom_kernel(data):
    """Outgoing Triangle Multiplicative Update (TriMul) forward pass using Triton.
    
    Implements the heavy N³ pairwise reduction with a fused batched GEMM kernel.
    All linear projections, gating and the core einsum are performed in fp16,
    while the final LayerNorm, gating and output projection are in fp32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ----------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim], float32

    # ----------------------------------------------------
    # 2) Cast to fp16 for heavy work
    # ----------------------------------------------------
    x_h = x.to(torch.float16)

    # ----------------------------------------------------
    # 3) Fuse projection + gate linear layers (fp16)
    # ----------------------------------------------------
    # Concatenate projection and gate weights for left/right side
    left_combined_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, device=device)
    right_combined_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, device=device)

    # Linear: [B,N,N,2*H]
    left_combined = F.linear(x_h, left_combined_weight)   # fp16
    right_combined = F.linear(x_h, right_combined_weight) # fp16

    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    # ----------------------------------------------------
    # 4) Optional mask (broadcast on hidden dim)
    # ----------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ----------------------------------------------------
    # 5) Gating (sigmoid, fp16)
    # ----------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = (left_proj * left_gate).to(torch.float16)
    right = (right_proj * right_gate).to(torch.float16)

    # ----------------------------------------------------
    # 6) Output gate (fp16)
    # ----------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(torch.float16, device=device)
    out_gate_pre = F.linear(x_h, out_gate_weight)   # [B,N,N,H] fp16
    out_gate = torch.sigmoid(out_gate_pre)          # fp16

    # ----------------------------------------------------
    # 7) Core N³ reduction via batched GEMM (fp16 -> fp32)
    # ----------------------------------------------------
    B, N, _, _ = left.shape  # left shape: [B, N, N, H]
    batch = B * hidden_dim   # total number of independent GEMMs

    # Rearrange for GEMM: [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) contiguous
    right_perm = right.permute(0, 3, 1, 2)              # (B, H, N, N) non‑contiguous
    right_t = right_perm.transpose(-2, -1)            # (B, H, N, N) rows=k, cols=j

    # Flatten batch dimension
    left_flat = left_perm.view(-1, N, N)   # (B*H, N, N) fp16
    right_flat = right_t.view(-1, N, N)    # (B*H, N, N) fp16

    # Allocate output in fp32
    out_flat = torch.empty((batch, N, N), dtype=torch.float32, device=device)

    # Strides for Triton
    stride_Ab = left_flat.stride(0)
    stride_Am = left_flat.stride(1)
    stride_Ak = left_flat.stride(2)

    stride_Bb = right_flat.stride(0)
    stride_Bk = right_flat.stride(1)
    stride_Bn = right_flat.stride(2)

    stride_Cb = out_flat.stride(0)
    stride_Cm = out_flat.stride(1)
    stride_Cn = out_flat.stride(2)

    # Block sizes (tuned for Hopper B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch * num_blocks_m * num_blocks_n,)

    batched_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

- Applies a per‑pair LayerNorm to the input.
- Projects the tensor into a hidden dimension twice (projection + gate) using
  fused linear layers and fp16 arithmetic.
- Optionally masks the projected tensors.
- Computes the core N³ reduction
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
  as a large batch of independent GEMMs (one per (batch, hidden) pair) via a
  custom Triton kernel that exploits Tensor‑cores (fp16 inputs, fp32 accumulation).
- Applies a final LayerNorm, output gating and a linear projection back to dim.
All heavy compute is in fp16; final result is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B  (B already transposed).

    A: (batch, M, K)   fp16
    B: (batch, K, N)   fp16   (row‑major after a transpose)
    C: (batch, M, N)   fp32   (accumulated in fp32)
    """
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_id = pid // (num_blocks_m * num_blocks_n)
    pid = pid % (num_blocks_m * num_blocks_n)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    # Offsets within the tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the start of the tile for A and B
    a_ptrs = a_ptr + batch_id * stride_Ab + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = b_ptr + batch_id * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K‑dimension loop
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=(offs_m < M)[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(tl.arange(0, BLOCK_K) < K)[:, None] & (offs_n < N)[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # Store result
    c_ptrs = c_ptr + batch_id * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    mask_c = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc, mask=mask_c)


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass using fp16 for the heavy work
    and a custom Triton kernel for the N³ reduction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model weights (see skeleton)
            - config: dict with at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ----------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ----------------------------------------------------
    # 2) Cast to fp16 for the heavy part
    # ----------------------------------------------------
    x_h = x.to(torch.float16)

    # ----------------------------------------------------
    # 3) Fuse projection + gate linear layers (fp16)
    # ----------------------------------------------------
    left_combined_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, device=device)
    right_combined_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, device=device)

    left_combined = F.linear(x_h, left_combined_weight)   # [B, N, N, 2*H]
    right_combined = F.linear(x_h, right_combined_weight)  # [B, N, N, 2*H]

    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    # ----------------------------------------------------
    # 4) Optional mask (broadcast on hidden dim)
    # ----------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16, device=device).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ----------------------------------------------------
    # 5) Gating (sigmoid, fp16)
    # ----------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = (left_proj * left_gate).to(torch.float16)   # [B, N, N, H]
    right = (right_proj * right_gate).to(torch.float16)  # [B, N, N, H]

    # ----------------------------------------------------
    # 6) Output gate (fp16)
    # ----------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(torch.float16, device=device)
    out_gate_pre = F.linear(x_h, out_gate_weight)   # [B, N, N, H]
    out_gate = torch.sigmoid(out_gate_pre)          # fp16

    # ----------------------------------------------------
    # 7) Core N³ reduction via batched GEMM (fp16 → fp32)
    # ----------------------------------------------------
    B, N, _, _ = left.shape
    batch = B * hidden_dim          # total independent GEMMs

    # Rearrange tensors for GEMM: (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2)            # non‑contiguous
    right_t = right_perm.transpose(-2, -1)          # rows=k, cols=j

    left_flat = left_perm.view(-1, N, N)            # (B*H, N, N) fp16
    right_flat = right_t.view(-1, N, N)             # (B*H, N, N) fp16
    out_flat = torch.empty((batch, N, N), dtype=torch.float32, device=device)

    # Strides for Triton (int values)
    stride_Ab = left_flat.stride(0)
    stride_Am = left_flat.stride(1)
    stride_Ak = left_flat.stride(2)

    stride_Bb = right_flat.stride(0)
    stride_Bk = right_flat.stride(1)
    stride_Bn = right_flat.stride(2)

    stride_Cb = out_flat.stride(0)
    stride_Cm = out_flat.stride(1)
    stride_Cn = out_flat.stride(2)

    # Tuned block sizes for Hopper B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch * num_blocks_m * num_blocks_n,)

    batched_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        batch=batch,
        M=N,
        N=N,
        K=N,
        stride_Ab=stride_Ab,
        stride_Am=stride_Am,
        stride_Ak=stride_Ak,
        stride_Bb=stride_Bb,
        stride_Bk=stride_Bk,
        stride_Bn=stride_Bn,
        stride_Cb=stride_Cb,
        stride_Cm=stride_Cm,
        stride_Cn=stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [B, N, N, H] (float32)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------
    # 8) Final LayerNorm, output gate, projection
    # ----------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )  # fp32

    out = out * out_gate.to(torch.float32)          # apply output gate
    out = F.linear(out, weights["to_out.weight"])   # final projection, fp32

    return out
