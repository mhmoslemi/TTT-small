"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) with the provided LayerNorm parameters.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections → left, right (shape B,N,N,H) using the provided weights.
4. Optional binary mask is applied to left/right before gating.
5. Compute three sigmoid gates (left_gate, right_gate, out_gate) from the
   normalized input and multiply them into the projected tensors.
6. Fold the hidden dimension H into the batch axis: tensors of shape (B*H,N,N).
   Perform the core reduction using a fused Triton batched GEMM:
        out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
   (implemented as a matmul of left with the transpose of right).
7. Reshape back to (B,N,N,H), apply a second LayerNorm over H,
   multiply by the output gate and linearly project back to C.
8. Cast the result back to the original dtype (float32 by default).

At least one part of the computation (the O(N³·H) reduction) is written
in Triton for high performance on the B200 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # B: (batch, K, N)
    stride_ch, stride_cm, stride_cn,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul for TriMul.
    Computes for each batch:
        C[i, j] = Σ_k A[i, k] * B[k, j]
    All tensors are FP16; accumulation is in FP32.
    """
    pid_b = tl.program_id(2)   # batch index (B*H)
    pid_m = tl.program_id(0)   # tile index along M (i)
    pid_n = tl.program_id(1)   # tile index along N (j)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute base pointers for this batch
    a_batch_off = pid_b * stride_ah
    b_batch_off = pid_b * stride_bh
    c_batch_off = pid_b * stride_ch

    # Pointers to the start of the tile for A and B
    a_ptrs = A_ptr + a_batch_off + (
        offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = B_ptr + b_batch_off + (
        tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        # Global k indices for this tile
        k_global = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_global < K

        # Load A and B tiles with masking
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write out the result (FP16) with proper masking
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch_off + (
        offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using Triton.
    A : (batch, M, K)  FP16
    B : (batch, K, N)  FP16
    Returns C : (batch, M, N) FP16
    """
    assert A.is_cuda and B.is_cuda
    batch, M, K = A.shape
    batch2, K2, N = B.shape
    assert batch == batch2 and K == K2

    # Heuristic block sizes (tuned for N up to 1024)
    if max(M, N) <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif max(M, N) <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    BLOCK_M = min(BLOCK_M, M)
    BLOCK_N = min(BLOCK_N, N)
    BLOCK_K = min(BLOCK_K, K)

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    _tri_mul_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        A.stride(0),  # stride_ah (batch)
        A.stride(1),  # stride_am (M)
        A.stride(2),  # stride_ak (K)
        B.stride(0),  # stride_bh (batch)
        B.stride(1),  # stride_bk (K)
        B.stride(2),  # stride_bn (N)
        C.stride(0),  # stride_ch (batch)
        C.stride(1),  # stride_cm (M)
        C.stride(2),  # stride_cn (N)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32)
            - config: dict containing:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask
    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # unpack
    x, mask, weights, config = data
    device = x.device
    orig_dtype = x.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm on input ----------
    norm_w = weights["norm.weight"].to(device).to(orig_dtype)
    norm_b = weights["norm.bias"].to(device).to(orig_dtype)
    x_norm = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ---------- Cast to FP16 for compute ----------
    compute_dtype = torch.float16
    x_fp16 = x_norm.to(compute_dtype) if x_norm.dtype != compute_dtype else x_norm

    # ---------- Load weight tensors ----------
    left_proj_w   = weights["left_proj.weight"].to(device).to(compute_dtype)
    right_proj_w  = weights["right_proj.weight"].to(device).to(compute_dtype)
    left_gate_w   = weights["left_gate.weight"].to(device).to(compute_dtype)
    right_gate_w  = weights["right_gate.weight"].to(device).to(compute_dtype)
    out_gate_w    = weights["out_gate.weight"].to(device).to(compute_dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(compute_dtype)
    to_out_w      = weights["to_out.weight"].to(device).to(compute_dtype)

    # ---------- Linear projections ----------
    left = F.linear(x_fp16, left_proj_w)   # (B, N, N, H)
    right = F.linear(x_fp16, right_proj_w)  # (B, N, N, H)

    # ---------- Optional mask ----------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(compute_dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp
        right = right * mask_fp

    # ---------- Gating ----------
    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare tensors for Triton GEMM ----------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (batch, M, K)
    # right needs to be transposed so that we have (batch, K, N)
    right_T = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)   # (batch, K, N)

    # ---------- Core O(N³·H) reduction ----------
    out_flat = _batched_matmul_triton(left_flat, right_T)   # (B*H, N, N)

    # ---------- Reshape back ----------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection back to dim ----------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ---------- Cast back to original dtype ----------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)
    return out
