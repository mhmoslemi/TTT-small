    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)   # tile i block index
    pid_n = tl.program_id(1)   # tile j block index
    pid_bh = tl.program_id(2)  # batch*hidden index

    # compute batch and hidden
    bh = pid_bh
    b = bh // H
    h = bh % H

    # Compute block offsets
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # Create range offsets
    i_offsets = i_start + tl.arange(0, BLOCK_M)
    j_offsets = j_start + tl.arange(0, BLOCK_N)

    # Masks for valid indices
    i_mask = i_offsets < N
    j_mask = j_offsets < N

    # Compute base pointer offset for this batch and hidden
    # left and right have same base offset for batch and h
    offset_bh = b * stride_bl + h * stride_hl  # assume stride_hl is 1

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        # left_ptrs shape: offset_bh + i_offsets[:,None]*stride_il + k_offsets[None,:]*stride_kl
        a_ptrs = left_ptr + offset_bh + i_offsets[:, None] * stride_il + k_offsets[None, :] * stride_kl
        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # Load B_T tile: shape (BLOCK_K, BLOCK_N)
        # right_ptrs shape: offset_bh + k_offsets[:,None]*stride_kr + j_offsets[None,:]*stride_jr
        b_ptrs = right_ptr + offset_bh + k_offsets[:, None] * stride_kr + j_offsets[None, :] * stride_jr
        b = tl.load(b_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write output
    # out_ptrs shape: offset_out = b*stride_out_b + h*stride_out_h + i_offsets[:,None]*stride_out_i + j_offsets[None,:]*stride_out_j
    out_offset = b * stride_out_b + h * stride_out_h
    out_ptrs = out_ptr + out_offset + i_offsets[:, None] * stride_out_i + j_offsets[None, :] * stride_out_j
    # Convert to fp16

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the supplied LN parameters.
2. Convert to fp16 for the compute‑heavy path.
3. Fuse the three linear projections on the left side (proj, gate,
   out_gate) into a single matmul.  Fuse the two linear projections on
   the right side (proj, gate) likewise.
4. Apply the optional binary mask (broadcast over the hidden dim) and
   fuse the gating (sigmoid) into the projected tensors.
5. Core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is implemented as a fused Triton GEMM where each program instance
   computes a BLOCK_M×BLOCK_N tile for a (batch, hidden) pair.
6. Apply a second LayerNorm over the hidden dimension.
7. Multiply by the output gate (element‑wise).
8. Final linear projection back to the original channel dimension.
9. Cast the result back to the input dtype and return.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the O(N³·H) reduction (matrix multiply of left and right
# transposed).  The batch and hidden dimensions are fused into a single
# program‑id axis to avoid a large launch overhead.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # *fp16 left tensor   (B,N,N,H) layout: (b,i,k,h)
    right_ptr,         # *fp16 right tensor  (B,N,N,H) layout: (b,j,k,h)
    out_ptr,           # *fp16 output tensor (B,N,N,H) layout: (b,i,j,h)
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,   # left strides
    stride_br, stride_jr, stride_kr, stride_hr,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ----- program ids -------------------------------------------------
    pid_i = tl.program_id(0)   # tile over output rows   (i)
    pid_j = tl.program_id(1)   # tile over output cols   (j)
    pid_bh = tl.program_id(2)  # fused batch * hidden

    # ----- decode batch and hidden index -------------------------------
    bh = pid_bh
    b = bh // H
    h = bh % H

    # ----- compute tile start indices ---------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # ----- range vectors ------------------------------------------------
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # ----- mask for out‑of‑bounds rows / cols -------------------------
    mask_i = i < N
    mask_j = j < N

    # ----- base offset for this (batch, hidden) -----------------------
    # left and right share the same batch/hidden base offset
    base_bh = b * stride_bl + h * stride_hl

    # ----- accumulator in fp32 ----------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over the reduction dimension k -------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kk * BLOCK_K
        k_off = k0 + k
        mask_k = k_off < N

        # Load left block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + base_bh \
                 + i[:, None] * stride_il \
                 + k_off[None, :] * stride_kl
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load right block transposed: shape (BLOCK_K, BLOCK_N)
        # right is accessed as right[b, j, k, h] -> we want B_T[k, j]
        b_ptrs = right_ptr + base_bh \
                 + k_off[:, None] * stride_kr \
                 + j[None, :] * stride_jr
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b)

    # ----- write result ------------------------------------------------
    c = acc.to(tl.float16)
    out_base = b * stride_ob + h * stride_oh
    out_ptrs = out_ptr + out_base \
               + i[:, None] * stride_oi \
               + j[None, :] * stride_oj
    tl.store(out_ptrs,
             c,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper: element‑wise multiplication with Triton (used for the out‑gate)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, c_ptr, numel,
                         BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton element‑wise multiplication kernel."""
    assert a.shape == b.shape
    total = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ( (total + BLOCK - 1) // BLOCK, )
    _elemwise_mul_kernel[grid](
        a, b, out, total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            mask: torch.Tensor of shape [B, N, N] (or None)
            weights: dict containing model weights
            config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 3) Prepare fused weight matrices (fp16)
    # ------------------------------------------------------------------
    # left side: proj, gate, out_gate  -> 3 * hidden_dim rows
    left_combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (H, C)
            weights["left_gate.weight"],   # (H, C)
            weights["out_gate.weight"],    # (H, C)
        ],
        dim=0,
    ).to(torch.float16)

    # right side: proj, gate -> 2 * hidden_dim rows
    right_combined_weight = torch.cat(
        [
            weights["right_proj.weight"],  # (H, C)
            weights["right_gate.weight"],  # (H, C)
        ],
        dim=0,
    ).to(torch.float16)

    # final projection back to original dimension
    to_out_weight = weights["to_out.weight"].to(torch.float16)  # (dim, H)

    # layer‑norm parameters for the output of the O(N³) step
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)      # (H,)

    # ------------------------------------------------------------------
    # 4) Linear projections (fused)
    # ------------------------------------------------------------------
    # We work on the flattened view (B*N*N, C)
    x_flat = x.view(-1, dim)                # (B*N*N, C)

    # left side: (B*N*N, 3*H)
    left_all = torch.nn.functional.linear(x_flat, left_combined_weight)
    left_all = left_all.view(B, N, N, 3 * hidden_dim)

    # split
    left_proj = left_all[..., :hidden_dim]                         # (B,N,N,H)
    left_gate_raw = left_all[..., hidden_dim:2 * hidden_dim]
    out_gate_raw = left_all[..., 2 * hidden_dim:]

    left_gate = left_gate_raw.sigmoid()
    out_gate = out_gate_raw.sigmoid()

    # right side: (B*N*N, 2*H)
    right_all = torch.nn.functional.linear(x_flat, right_combined_weight)
    right_all = right_all.view(B, N, N, 2 * hidden_dim)

    right_proj = right_all[..., :hidden_dim]
    right_gate_raw = right_all[..., hidden_dim:]
    right_gate = right_gate_raw.sigmoid()

    # ------------------------------------------------------------------
    # 5) Apply optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 6) Fuse gates into the projected tensors
    # ------------------------------------------------------------------
    left = (left_proj * left_gate).contiguous()
    right = (right_proj * right_gate).contiguous()

    # ------------------------------------------------------------------
    # 7) Core O(N³·H) reduction via custom Triton kernel
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # (B,N,N,H) fp16

    # strides for left, right, out (all contiguous in NHWC order)
    stride_bl, stride_il, stride_kl, stride_hl = left.stride()
    stride_br, stride_jr, stride_kr, stride_hr = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Triton block sizes (tuned for typical B200 memory hierarchy)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_jr, stride_kr, stride_hr,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 8) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    # 9) Apply output gate – element‑wise multiplication (Triton)
    # ------------------------------------------------------------------
    out = _elemwise_mul(out, out_gate)

    # ------------------------------------------------------------------
    # 10) Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_weight)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 11) Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
