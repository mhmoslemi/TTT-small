@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(out_ptr + offs, mask=mask)
    b = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

"""
Outgoing Triangle Multiplicative Update (TriMul) – fused forward implementation.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Fuse left/right projections and their gates into a single linear layer
   (weight shape (4·hidden, dim)).  Split the result into left_proj,
   left_gate, right_proj and right_gate.
3. (Optional) apply the pairwise mask to the two projections.
4. Apply sigmoid gates and multiply with the corresponding projections.
5. Perform the core O(N³) reduction as a batched matrix multiplication:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is realised as torch.bmm on tensors reshaped to (B·hidden, N, N).
6. Layer‑norm the intermediate result over the hidden dimension.
7. Compute the out‑gate (sigmoid of a linear projection of the normalized input)
   and multiply it element‑wise with the intermediate tensor using a tiny
   Triton kernel.
8. Final linear projection back to the original channel dimension.

All heavy arithmetic is carried out in float16 (FP16) to exploit tensor cores;
the final output is returned in float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """In‑place out = out * gate  (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(out_ptr + offs, mask=mask)
    b = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – may be ignored if config["nomask"] is True
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # -----------------------------------------------------------------
    # Load and cast weights (all linear layers have bias=False)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim], float32

    # Cast to FP16 for all subsequent heavy work
    x_fp16 = x.to(torch.float16)

    B, N, _, _ = x_fp16.shape

    # -----------------------------------------------------------------
    # 2️⃣ Fuse left/right projections and their gates
    #    Combined weight shape: (4*hidden_dim, dim)
    # -----------------------------------------------------------------
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    )  # (4*hidden_dim, dim)

    # Flatten spatial dimensions and apply the fused linear mapping
    x_flat = x_fp16.view(-1, dim)                     # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 4*hidden_dim)

    # Restore 4‑D layout and split
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each (B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 3️⃣ Optional masking (applied only to the projections)
    # -----------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # (B, N, N, 1)
        mask_f = mask_f.to(torch.float16)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # -----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and combine with projections
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate   # (B, N, N, hidden_dim)
    right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Batched GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented via torch.bmm on reshaped tensors.
    # -----------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    left_mat = left.view(B * hidden_dim, N, N)     # (B*hidden, N, N)
    right_mat = right.view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, hidden_dim), FP16

    # -----------------------------------------------------------------
    # 6️⃣ LayerNorm over hidden dimension
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden_dim), FP16

    # -----------------------------------------------------------------
    # 7️⃣ Compute out_gate and apply it (Triton kernel)
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # (B, N, N, hidden_dim)

    # Triton element‑wise multiplication: out *= out_gate (in‑place)
    total_elements = out.numel()
    BLOCK = 1024
    grid = ((total_elements + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        out,
        out_gate,
        total_elements,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Final projection back to original dimension
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # (B, N, N, dim), FP16
    out = out.to(torch.float32)          # return as float32
    return out
