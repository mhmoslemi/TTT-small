"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) with provided LN parameters.
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections to hidden dimension H and compute three sigmoid gates.
4. Apply an optional binary mask and the left/right gates using a small
   Triton kernel that fuses element‑wise multiplies.
5. Reshape the gated tensors to (B*H, N, N) and perform a batched matrix
   multiplication with torch.bmm (equivalent to the einsum in the reference):
        out[b*i,j] = Σ_k left[b,i,k] * right[b,j,k]
6. Reshape back to (B,N,N,H), apply a second LayerNorm over H,
   multiply by the output gate and linearly project back to C.
7. Cast the result back to the original dtype.

At least one operation (mask + gate application) is implemented in Triton,
while the O(N³·H) reduction leverages cuBLAS via torch.bmm for maximal speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems, mask_elems, hidden_dim,
    WITH_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Element‑wise fused kernel:
        left_out  = left  * left_gate  * mask
        right_out = right * right_gate * mask
    Mask has shape (B,N,N) and is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load the four input tensors
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if WITH_MASK:
        # compute index into mask: each hidden_dim elements share one mask entry
        mask_idx = offs // hidden_dim                # shape (BLOCK_SIZE,)
        mask_idx_ok = mask_idx < mask_elems
        # load mask values (broadcasted)
        m = tl.load(mask_ptr + mask_idx, mask=mask_idx_ok, other=0.0)
        # apply mask and gates
        l = l * lg * m
        r = r * rg * m
        # store only where both masks are valid
        store_mask = mask & mask_idx_ok
        tl.store(left_out_ptr + offs, l, mask=store_mask)
        tl.store(right_out_ptr + offs, r, mask=store_mask)
    else:
        # no mask – only apply gates
        l = l * lg
        r = r * rg
        tl.store(left_out_ptr + offs, l, mask=mask)
        tl.store(right_out_ptr + offs, r, mask=mask)

    use_mask = (not nomask) and (mask is not None)

    if use_mask:
        mask_fp = mask.to(device).to(compute_dtype).contiguous()
        mask_flat = mask_fp.view(-1)                     # (B*N*N,)
        mask_elems = mask_flat.numel()
    else:
        # dummy tensor – never read when WITH_MASK=False
        mask_flat = torch.empty(0, device=device, dtype=compute_dtype)
        mask_elems = 0

    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    BLOCK_SIZE = 1024  # threads per block (must be multiple of 32)
    num_blocks = (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE

    _gate_mask_kernel[(num_blocks,)](
        left,
        right,
        left_gate,
        right_gate,
        mask_flat,
        left_out,
        right_out,
        total_elems,
        mask_elems,
        H,
        WITH_MASK=use_mask,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    torch.cuda.synchronize()

    # ---------- Batched matrix multiplication (core O(N³·H) reduction) ----------
    # reshape to (B*H, N, N)
    left_flat = left_out.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # bmm performs Σ_k left[i,k] * right[j,k] after transposition
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection back to dim ----------
    out = F.linear(out, to_out_w)      # (B, N, N, C)

    # Cast back to original dtype if necessary
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
```
