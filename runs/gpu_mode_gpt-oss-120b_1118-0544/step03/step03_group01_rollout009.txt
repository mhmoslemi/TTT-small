"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The kernel computes the core O(N³·H) reduction:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

by folding the hidden dimension H into the batch dimension and performing a
batched matrix multiplication C = A @ Bᵀ with a custom Triton GEMM kernel.
All cheap operations (LayerNorm, linear projections, gating, final projection)
are done with PyTorch, while the heavy O(N³·H) part runs in fp16 with fp32
accumulation for speed and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM computing C = A @ Bᵀ
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({'BLOCK_M': 32, 'BLOCK_N': 32, 'BLOCK_K': 32}, num_warps=2),
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 64, 'BLOCK_K': 32}, num_warps=4),
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 64, 'BLOCK_K': 64}, num_warps=4),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 64}, num_warps=8),
    ],
    key=['M', 'N', 'K']
)
@triton.jit
def _outgoing_gemm_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes (constexpr)
    stride_ah, stride_am, stride_ak,    # strides of A: batch, row, col
    stride_bh, stride_bk, stride_bn,    # strides of B (accessed as K,N)
    stride_ch, stride_cm, stride_cn,    # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for the outgoing TriMul:
        C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]   (i.e. A @ Bᵀ)
    A, B, C are fp16 tensors, accumulation is fp32.
    """
    pid_b = tl.program_id(2)          # batch index (B * H)
    pid_m = tl.program_id(0)          # block row index
    pid_n = tl.program_id(1)          # block column index

    # --------------------------------------------------------------
    # Block coordinates
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Base pointers for this batch
    # --------------------------------------------------------------
    a_batch_off = pid_b * stride_ah
    b_batch_off = pid_b * stride_bh
    c_batch_off = pid_b * stride_ch

    # --------------------------------------------------------------
    # Pointer tiles for the current M×K and K×N blocks
    # --------------------------------------------------------------
    a_ptrs = A_ptr + a_batch_off + (
        offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = B_ptr + b_batch_off + (
        tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn
    )

    # --------------------------------------------------------------
    # Accumulator (fp32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k in range(num_k_tiles):
        # K‑mask for the current tile
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                     # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)                         # (BLOCK_M, BLOCK_N)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Write back result (convert to fp16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch_off + (
        offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_outgoing_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.  A and B must be fp16 tensors of shape
    (batch, N, N).  Returns C of the same shape and dtype.
    """
    assert A.shape == B.shape, "A and B must have identical shape"
    batch, N, K = A.shape
    assert N == K, "TriMul requires square sequence dimension"

    # Heuristic block size selection (fallback to autotune configs)
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
    elif N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks in M dimension
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks in N dimension
        batch,                          # folded batch * hidden_dim
    )

    C = torch.empty_like(A)
    _outgoing_gemm_kernel[grid](
        A,
        B,
        C,
        N, N, N,                         # M, N, K (constexpr)
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(2), B.stride(1),   # note B is accessed as (K,N)
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, mask is ignored

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # LayerNorm on the input (float32 is safe for statistics)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Work in half precision for the heavy path
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # Cast linear weights to fp16 (no bias terms)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # Linear projections (produces [B, N, N, H])
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w)  # (B,N,N,H)

    # --------------------------------------------------------------
    # Optional binary mask (applied before gating)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # Gates (sigmoid activation)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Fold hidden dimension into batch for the O(N³·H) reduction
    # --------------------------------------------------------------
    B, N, _, H = left.shape                # B, seq_len, seq_len, hidden_dim
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    left  = left.view(B * H, N, N)          # (B*H, N, N)
    right = right.view(B * H, N, N)         # (B*H, N, N)

    # --------------------------------------------------------------
    # Core O(N³·H) multiply‑accumulate using Triton
    # --------------------------------------------------------------
    out_flat = _batched_outgoing_gemm(left, right)   # (B*H, N, N)

    # --------------------------------------------------------------
    # Restore original layout: (B, N, N, H)
    # --------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # Output LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # Apply output gate and final linear projection back to dim
    # --------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_w)        # (B,N,N,dim)

    # --------------------------------------------------------------
    # Cast back to the original dtype of the input tensor
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
