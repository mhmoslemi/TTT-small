    right = right_proj * right_gate

    # compute batched GEMM (left[i,k] * right[j,k] sum over k)
    # Permute to bring hidden dim to batch
    left = left.permute(0, 3, 1, 2).contiguous()  # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, hidden, N, N)

    # reshape batch*hidden
    B_h = B * hidden_dim
    left_mat = left.view(B_h, N, N)  # (B*hidden, N, N)
    right_mat = right.view(B_h, N, N)  # (B*hidden, N, N)

    # batched matmul
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    # reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden)

    # to_out_norm
    to_out_norm_w = weights['to_out_norm.weight'].to(device).to(torch.float16)
    to_out_norm_b = weights['to_out_norm.bias'].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=1e-5)

    # compute out_gate
    out_gate_w = weights['out_gate.weight'].to(device).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))  # (B, N, N, hidden)

    # elementwise multiply out * out_gate using Triton kernel
    total_elements = out.numel()
    BLOCK = 65536  # choose a large block for fewer kernel launches
    grid = ( (total_elements + BLOCK - 1) // BLOCK, )
    # ensure contiguous
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    # launch kernel (in-place)
    elementwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        out_flat,
        total_elements,
        BLOCK=BLOCK
    )

    # final projection to dim using linear
    to_out_w = weights['to_out.weight'].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, dim) in fp16

    # convert to float32
    out = out.to(torch.float32)

    return out
```

"""
Outgoing Triangle Multiplicative Update (TriMul) – fast forward implementation.

The kernel fuses:
* Four linear projections (left/right + their gates) into a single matmul.
* An element‑wise multiplication with the output gate using a tiny Triton kernel.
All heavy computations are performed in float16 to exploit tensor‑cores.
The final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """c = a * b  (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – ignored if config["nomask"] is True
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim], float32

    # --------------------------------------------------------------
    # 2️⃣ Cast to fp16 and fuse four linear layers (proj + gate)
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)                     # [B, N, N, dim]
    B, N, _, _ = x_fp16.shape

    # gather projection & gate weights and cast to fp16
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)

    # combine into one big weight matrix: (4*hidden, dim)
    combined_w = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0)

    # flatten spatial dimensions and apply combined linear map
    x_flat = x_fp16.view(-1, dim)                               # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_w)                 # (B*N*N, 4*hidden)

    # reshape and split
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each: (B, N, N, hidden)

    # --------------------------------------------------------------
    # 3️⃣ Optional masking (applied only to the projections)
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and combine with projections
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate    # (B, N, N, hidden) fp16
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Batched GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched fp16 matmul.
    # --------------------------------------------------------------
    # move hidden to batch dimension for efficient bmm
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    B_h = B * hidden_dim
    left_mat = left.view(B_h, N, N)     # (B*hidden, N, N)
    right_mat = right.view(B_h, N, N)   # (B*hidden, N, N)

    # (B*hidden, N, N) @ (B*hidden, N, N)^T -> (B*hidden, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ to_out_norm (LayerNorm over hidden dimension)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device).to(torch.float16),
        bias=weights["to_out_norm.bias"].to(device).to(torch.float16),
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 7️⃣ Compute out_gate and apply it (Triton element‑wise kernel)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(
            x_fp16,
            weights["out_gate.weight"].to(device).to(torch.float16),
        )
    )  # (B, N, N, hidden) fp16

    total_el = out.numel()
    BLOCK = 65536
    grid = ((total_el + BLOCK - 1) // BLOCK,)

    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)

    elementwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        out_flat,
        total_el,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 8️⃣ Final projection back to original dimension
    # --------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(device).to(torch.float16),
    )  # (B, N, N, dim) fp16

    # return in float32 as required
    return out.to(torch.float32)
