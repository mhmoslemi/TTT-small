"""
TriMul (outgoing) forward implementation with heavy operations in fp16
and a tiny Triton element‑wise kernel.

Algorithm
---------
1. LayerNorm over the input channel dimension (dim).
2. Fuse the four linear projections (left/right + their gates) into a single
   matrix multiplication by concatenating the four weight matrices.
3. Optional masking applied to the left/right projections.
4. Apply sigmoid gates and multiply with the projections → left, right.
5. Batched half‑precision GEMM:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   implemented as a series of `torch.bmm` calls after reshaping
   (batch × hidden) × (N × N) matrices.
6. LayerNorm over the hidden dimension.
7. Compute out‑gate (sigmoid of a linear) and fuse element‑wise multiplication
   using a Triton kernel.
8. Final linear projection back to the original dimension and cast to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, SIZE: tl.constexpr, BLOCK: tl.constexpr):
    """c = a * b  (1‑D flat tensors)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < SIZE
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – optional
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # Cast weights to appropriate dtype
    # ------------------------------------------------------------------
    # LayerNorm on input
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    # Projection & gate weights (no bias)
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    # Output norm & final linear
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → keep precision for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # Cast to fp16 for all subsequent heavy work
    x_fp16 = x.to(torch.float16)

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Fuse four linear layers (proj + gates) into one matmul
    # ------------------------------------------------------------------
    combined_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w],
        dim=0,
    )  # (4*H, dim)

    # Flatten spatial dimensions, apply combined linear
    x_flat = x_fp16.view(-1, dim)                 # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_w)   # (B*N*N, 4*H)

    # Restore 4‑D layout and split
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each: (B, N, N, H)

    # ------------------------------------------------------------------
    # 3. Optional masking (applied only to the projections)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # (B, N, N, 1) float32
        mask_f = mask_f.to(torch.float16)                # (B, N, N, 1) fp16
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and combine with projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate      # (B, N, N, H)
    right = right_proj * right_gate   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5. Batched GEMM: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Move hidden to the second dimension and merge batch+hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_mat = left_perm.view(B * hidden_dim, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N) # (B*H, N, N)

    # Batch matrix‑multiply with transposed right operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 7. Compute out‑gate and fuse element‑wise multiplication via Triton
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))  # (B, N, N, H) fp16

    # Ensure contiguous tensors for the Triton kernel
    out = out.contiguous()
    out_gate = out_gate.contiguous()

    total_elems = out.numel()
    BLOCK = 1024
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    elementwise_mul_kernel[grid](
        out,
        out_gate,
        out,
        total_elems,
        BLOCK=BLOCK,
    )  # out = out * out_gate (in‑place)

    # ------------------------------------------------------------------
    # 8. Final projection back to original dimension and cast to float32
    # ------------------------------------------------------------------
    out = F.linear(out.view(-1, hidden_dim), to_out_w)  # (B*N*N, dim) fp16
    out = out.view(B, N, N, dim).to(torch.float32)

    return out
