"""Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the PyTorch reference:
1. Layer‑norm the input tensor.
2. Linear projections `left` and `right` (C → H) and three gates
   (C → H) with sigmoid activation.
3. Optional binary mask is applied before gating.
4. Element‑wise gate multiplication.
5. The core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM C = A @ Bᵀ where A and B have shape
   (B·H, N, N).  A custom Triton kernel tiles the matrices, loads in
   fp16 and accumulates in fp32 for high throughput.
6. Reshape back, Layer‑norm over the hidden dimension, multiply by the
   output gate and project back to the original channel dimension.
7. Return a float32 tensor of shape [B, N, N, C].
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dimensions
    stride_ah, stride_am, stride_ak,   # A: batch, row, col
    stride_bh, stride_bk, stride_bn,   # B: batch, row (k), col (n)
    stride_ch, stride_cm, stride_cn,   # C: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]  (A @ Bᵀ)."""
    pid_b = tl.program_id(2)   # batch index (B·H)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch offsets
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # pointers for the current tile
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of tiles along K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """Wrapper for the batched GEMM Triton kernel.

    A, B: fp16 tensors of shape (batch, N, N) with batch = B·H.
    Returns C of the same shape and dtype (fp16)."""
    assert A.shape == B.shape
    batch, N, K = A.shape
    assert N == K, "TriMul expects square sequence dimension."

    # Choose tile sizes heuristically based on N
    if N <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    C = torch.empty_like(A)

    _batched_gemm_kernel[grid](
        A, B, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),      # A: batch, row, col
        B.stride(0), B.stride(2), B.stride(1),      # B as (batch, K, N)
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B, N, N, C], dtype float32

    # ------------------------------------------------------------------
    # 2) Linear projections (float32)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Optional binary mask (float32)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # -> [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gates (float32)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Apply gates (float32)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Cast to fp16 for the heavy O(N³·H) reduction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # reshape to (B*H, N, N) for the batched GEMM
    B, N, _, H = left_fp16.shape
    left_flat = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # ------------------------------------------------------------------
    # 7) Core reduction with Triton
    # ------------------------------------------------------------------
    out_fp16 = _batched_gemm_triton(left_flat, right_flat)  # [B*H, N, N] fp16

    # ------------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H) and convert to float32
    # ------------------------------------------------------------------
    out = (
        out_fp16.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9) Output LayerNorm over hidden dimension (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate (float32)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to original dim (float32)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 12) Ensure final dtype is float32
    # ------------------------------------------------------------------
    if out.dtype != torch.float32:
        out = out.to(torch.float32)

    return out
