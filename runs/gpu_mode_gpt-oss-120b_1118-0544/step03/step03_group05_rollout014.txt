@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, N, H,
    stride_left_batch, stride_left_row, stride_left_col,
    stride_right_batch, stride_right_row, stride_right_col,
    stride_out_batch, stride_out_row, stride_out_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    total_blocks_m = tl.cdiv(N, BLOCK_M)
    total_blocks_n = tl.cdiv(N, BLOCK_N)
    total_batches = BATCH
    # compute batch index
    batch = pid // (total_blocks_m * total_blocks_n)
    block_index = pid % (total_blocks_m * total_blocks_n)
    block_m = block_index // total_blocks_n
    block_n = block_index % total_blocks_n

    # compute offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # within each block, we will accumulate over k
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # mask for out-of-bound k
        mask_k = offs_k < N

        # compute pointers
        a_ptrs = left_ptr + batch * stride_left_batch + offs_m[:, None] * stride_left_row + offs_k[None, :] * stride_left_col
        b_ptrs = right_ptr + batch * stride_right_batch + offs_n[:, None] * stride_right_row + offs_k[None, :] * stride_right_col

        a = tl.load(a_ptrs, mask=offs_m[:, None] < N & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=offs_n[:, None] < N & mask_k[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b)

        # increment pointers for next k? Not needed because we recompute with new offs_k.

    # write back
    c_ptrs = out_ptr + batch * stride_out_batch + offs_m[:, None] * stride_out_row + offs_n[None, :] * stride_out_col
    # apply mask for out-of-range
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor `[B, N, N, C]`.
2. Cast to fp16 and compute linear projections for “left” and “right”.
3. Apply optional binary mask (broadcast over the hidden dimension).
4. Compute three sigmoid gates (left, right, output) from the same normalized input.
5. Fuse the left/right gates into the projected tensors.
6. Fuse the hidden dimension (`H = hidden_dim`) into the batch dimension and run a
   custom Triton GEMM kernel that computes
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   i.e. a batched mat‑mul `A @ Bᵀ` for each `(b,h)` pair.
7. Reshape the result back to `[B, N, N, H]` and apply a second Layer‑Norm.
8. Multiply by the output gate (element‑wise) and project back to the original
   channel dimension `C`.
9. Cast the result back to the input dtype.

Only the heavy O(N³·H) reduction (step 6) is implemented in Triton; the remaining
operations use PyTorch but stay in fp16 to keep the memory traffic low.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_batch, stride_left_row, stride_left_col,
    stride_right_batch, stride_right_row, stride_right_col,
    stride_out_batch, stride_out_row, stride_out_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = left @ rightᵀ for a batch of (N×N) matrices.

    Arguments are raw pointers and strides for three contiguous tensors of shape
    (B, N, N) in fp16. `B` = batch_size * hidden_dim after flattening.
    """
    pid = tl.program_id(0)

    total_m = tl.cdiv(N, BLOCK_M)
    total_n = tl.cdiv(N, BLOCK_N)

    # batch index and tile coordinates
    batch = pid // (total_m * total_n)
    tile_id = pid % (total_m * total_n)
    tile_m = tile_id // total_n
    tile_n = tile_id % total_n

    # offsets within the matrix
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # valid‑index masks
        mask_m = offs_m < N
        mask_n = offs_n < N
        mask_k = offs_k < N

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + batch * stride_left_batch \
                 + offs_m[:, None] * stride_left_row \
                 + offs_k[None, :] * stride_left_col
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Bᵀ tile: shape (BLOCK_K, BLOCK_N)
        # (we load B using (k, n) order to obtain the transpose on‑the‑fly)
        b_ptrs = right_ptr + batch * stride_right_batch \
                 + offs_k[:, None] * stride_right_row \
                 + offs_n[None, :] * stride_right_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot‑product accumulation
        acc += tl.dot(a, b)

    # write back the result in FP16
    c_ptrs = out_ptr + batch * stride_out_batch \
             + offs_m[:, None] * stride_out_row \
             + offs_n[None, :] * stride_out_col
    store_mask = (offs_m < N)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=store_mask)


def _tri_mul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper for the Triton GEMM kernel.
    `left` and `right` must be FP16 tensors of shape (B, N, N) and be
    contiguous. Returns a tensor of the same shape and dtype.
    """
    assert left.shape == right.shape
    B, N, _ = left.shape
    out = torch.empty_like(left)

    # strides in element units (already in the correct datatype)
    stride_l_b, stride_l_i, stride_l_j = left.stride()
    stride_r_b, stride_r_i, stride_r_j = right.stride()
    stride_o_b, stride_o_i, stride_o_j = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # total number of program instances
    total_m = (N + BLOCK_M - 1) // BLOCK_M
    total_n = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = B * total_m * total_n
    grid = (total_tiles,)

    _tri_mul_gemm_kernel[grid](

        stride_l_i,
        stride_l_j,
        stride_r_b,
        stride_r_i,
        stride_r_j,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            mask: torch.Tensor of shape [B, N, N] or None
            weights: dict containing model weights
            config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 → keep precision for statistics)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Layout‑wise weight preparation (fp16)
    # ------------------------------------------------------------------
    w_left   = weights["left_proj.weight"].to(torch.float16)
    w_right  = weights["right_proj.weight"].to(torch.float16)
    w_lgate  = weights["left_gate.weight"].to(torch.float16)
    w_rgate  = weights["right_gate.weight"].to(torch.float16)
    w_outgate = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, w_left)   # (B, N, N, H)
    right = F.linear(x, w_right)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6) Gates (sigmoid) – derived from the same normalized input
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, w_lgate).sigmoid()
    right_gate = F.linear(x, w_rgate).sigmoid()
    out_gate   = F.linear(x, w_outgate).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Fuse gates into left/right
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate
    # free the gate tensors that are no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 8) Prepare tensors for the batched GEMM kernel
    #    Shape conversion: (B, N, N, H) → (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left = left.view(B * H, N, N)   # (B*H, N, N)
    right = right.view(B * H, N, N) # (B*H, N, N)

    # ------------------------------------------------------------------
    # 9) Core O(N³·H) reduction via the custom Triton GEMM kernel
    # ------------------------------------------------------------------
    out = _tri_mul_gemm(left, right)  # (B*H, N, N)

    # ------------------------------------------------------------------
    # 10) Restore original tensor layout (B, N, N, H)
    # ------------------------------------------------------------------
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 11) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=w_to_out_norm, bias=b_to_out_norm)

    # ------------------------------------------------------------------
    # 12) Apply the output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 13) Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, w_to_out)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 14) Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
