def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # layernorm
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # cast to half
    x = x.to(torch.float16)

    # cast weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # projections
    left = F.linear(x, left_proj_w)   # [B,N,N,H]
    right = F.linear(x, right_proj_w)

    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Core reduction using Triton matmul kernel
    B, N, _, H = left.shape

    # reshape left and right for batched matmul
    left_ = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_ = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch*head
    left_flat = left_.view(B * H, N, N)
    right_flat = right_.view(B * H, N, N)
    # allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # compute strides
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)  # from transposed view
    stride_b_n = right_flat.stride(1)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # define block sizes (tuned)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # compute grid
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    # launch kernel
    _matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]

    # to_out_norm
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # multiply with out_gate using Triton elementwise kernel
    out = _elemwise_mul_triton(out, out_gate)

    # final linear
    out = F.linear(out, to_out_w)

    # cast to original dtype
    out = out.to(x.dtype) # Actually x is half; but we need original dtype: input_tensor.dtype
    out = out.to(input_tensor.dtype)

    return out

```
@triton.jit
def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program ids
    pid = tl.program_id(0)
    batch = tl.program_id(1)

    # compute block indices
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute base pointers for this batch
    a_batch_ptr = a_ptr + batch * stride_ab
    b_batch_ptr = b_ptr + batch * stride_bb
    c_batch_ptr = c_ptr + batch * stride_cb

    # load a tile initially
    # we will accumulate across K
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_batch_ptr + (offs_m[:, None] * stride_am + (k*BLOCK_K + offs_k)[None, :] * stride_ak), mask=mask_m[:, None] & ((k*BLOCK_K + offs_k) < K), other=0.0)
        b = tl.load(b_batch_ptr + ((k*BLOCK_K + offs_k)[:, None] * stride_bk + offs_n[None, :] * stride_bn), mask=((k*BLOCK_K + offs_k) < K)[:, None] & mask_n[None, :], other=0.0)
        # dot
        acc += tl.dot(a, b)
    # store result
    c = acc.to(tl.float16)
    tl.store(c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn), c, mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the reference PyTorch version but fuses the heavy
O(N³·H) reduction with a custom Triton batched‑GEMM kernel
(left @ rightᵀ).  Linear projections for the left/right pathways are combined
with their corresponding gates to halve the number of input‑tensor reads.
All costly tensor math is performed in float16 with float32 accumulation,
while the final result is cast back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernels
# --------------------------------------------------------------------------- #

@triton.jit
def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dims (M×K)·(K×N)→(M×N)
    stride_am, stride_ak,              # strides for A (batch‑local)
    stride_bk, stride_bn,              # strides for B (interpreted as K×N)
    stride_cm, stride_cn,              # strides for C
    stride_ab, stride_bb, stride_cb,   # strides for the batch dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM: C = A @ Bᵀ  (A,B shape: batch×M×K)"""

    pid = tl.program_id(0)          # output tile id (i‑j)
    batch = tl.program_id(1)        # batch index (B·H)

    # output tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # linear offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the K‑dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = k_offset + offs_k
        mask_k = cur_k < K

        # A is (M, K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B is stored as (K, N) – we load it as Bᵀ
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)          # fp16·fp16 → fp32 accumulation

    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


@triton.jit
def _elemwise_mul_kernel(
    a_ptr, b_ptr, out_ptr, size,
    BLOCK: tl.constexpr,
):
    """Element‑wise multiplication of two 1‑D contiguous tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton element‑wise kernel on two tensors of identical shape."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication."
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------- #
    # unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # work in half precision for the heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 2. Cast model weights to half precision (on the same device)
    # ------------------------------------------------------------------- #
    device = x.device
    dtype = torch.float16
    # combine left projection + gate, right projection + gate to reduce input reads
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype)

    # ------------------------------------------------------------------- #
    # 3. Linear projections + gates (combined)
    # ------------------------------------------------------------------- #
    # left: (proj, gate), right: (proj, gate)
    left_comb = F.linear(x, left_comb_w)      # [B, N, N, 2*H]
    right_comb = F.linear(x, right_comb_w)    # [B, N, N, 2*H]

    # split projections and gates
    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # optional masking (mask is broadcast over the hidden dimension)
    if mask is not None and not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)      # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------- #
    # 4. Core O(N³·H) reduction (batched matmul) using Triton
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape to (batch = B*H, M = N, K = N) for A and (batch, K = N, N) for Bᵀ
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_t.view(B * H, N, N)   # (batch, M, K)
    right_flat = right_t.view(B * H, N, N) # (batch, K, N) – we will treat it as transposed in the kernel

    out_flat = torch.empty_like(left_flat)  # (batch, M, N), fp16

    # strides for the kernel
    stride_a_batch = left_flat.stride(0)            # M*K
    stride_a_m = left_flat.stride(1)                # K
    stride_a_k = left_flat.stride(2)                # 1

    stride_b_batch = right_flat.stride(0)           # K*N
    stride_b_k = right_flat.stride(2)               # 1 (K dimension for the transposed view)
    stride_b_n = right_flat.stride(1)               # N

    stride_c_batch = out_flat.stride(0)             # M*N
    stride_c_m = out_flat.stride(1)                 # N
    stride_c_n = out_flat.stride(2)                 # 1

    # block size – tuned for typical seq‑len ≤ 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,                     # M, N, K (all equal to seq‑len)
        stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 5. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------- #
    # 6. Output gate (element‑wise) – Triton kernel
    # ------------------------------------------------------------------- #
    out_gate = F.linear(x, out_gate_w).sigmoid()   # [B, N, N, H]
    out = _elemwise_mul(out, out_gate)

    # ------------------------------------------------------------------- #
    # 7. Final projection back to original channel dimension
    # ------------------------------------------------------------------- #
    out = F.linear(out, to_out_w)   # [B, N, N, C]

    # Cast back to the original dtype of the input tensor
    out = out.to(x.dtype)
    out = out.to(x.shape) if out.dtype != x.dtype else out
    out = out.to(x.dtype)  # ensure dtype matches input
    out = out.to(x.dtype) if out.dtype != x.dtype else out
    out = out.to(x.dtype)  # redundant safety

    return out
