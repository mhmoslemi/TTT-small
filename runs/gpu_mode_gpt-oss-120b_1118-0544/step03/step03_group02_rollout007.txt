import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(dst_ptr, a_ptr, b_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """
    Triton element‑wise multiplication: dst = a * b.
    Operates on a flattened 1‑D view of the tensors.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(dst_ptr + offs, a * b, mask=mask)

        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, dim] (float32)
            - mask:        torch.Tensor[B, N, N] (bool/float) – ignored if config["nomask"] is True
            - weights:     dict of model parameters (torch.Tensor)
            - config:      dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # Cast / cache weights (no dtype conversion here – autocast will handle)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over channel dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim], dtype float32

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2️⃣ Fuse four linear layers into a single matmul
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices (each hidden_dim x dim)
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    )  # (4*hidden_dim, dim)

    # Flatten spatial dimensions and apply the combined projection
    x_flat = x.view(-1, dim)                          # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 4*hidden_dim)

    # Restore shape and split
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each: (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (applied only to the projections)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: (B, N, N) -> (B, N, N, 1)
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left_proj  = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and combine with projections
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left  = left_proj  * left_gate   # (B, N, N, hidden)
    right = right_proj * right_gate  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched matmul over (B·hidden) batches.
    # ------------------------------------------------------------------
    # Move hidden to the batch dimension for efficient matmul
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, hidden, N, N)

    # Perform batched matrix multiplication: left @ rightᵀ
    # Shape after matmul: (B, hidden, N, N)
    out = torch.matmul(left, right.transpose(-2, -1))

    # Restore original layout: (B, N, N, hidden)
    out = out.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ to_out_norm (LayerNorm over hidden dimension)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 7️⃣ Compute out_gate and apply it (Triton kernel for element‑wise mult)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # (B, N, N, hidden)

    # Triton element‑wise multiplication: out *= out_gate
    total_elements = out.numel()
    BLOCK = 65536  # large enough to keep kernel launches low
    grid = ( (total_elements + BLOCK - 1) // BLOCK, )
    # The kernel works on flattened pointers; ensure contiguous storage.
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    elementwise_mul_kernel[grid](
        out.data_ptr(),
        out.data_ptr(),
        out_gate.data_ptr(),
        total_elements,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Final projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # (B, N, N, dim)
    return out.to(torch.float32)
