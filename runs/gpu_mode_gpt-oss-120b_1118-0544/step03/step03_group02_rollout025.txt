"""
Outgoing TriMul (Triangular Multiplicative Update) – fast forward pass.

Algorithm:
1. Layer‑norm the input (FP16).
2. Fuse the two projection‑linear layers (proj + gate) for the left and right
   paths into two matrix‑multiplications with weight shapes (2·H, C).
3. Apply the optional pairwise mask, sigmoid‑gate and element‑wise product to
   obtain tensors `left` and `right` of shape (B, N, N, H) in FP16.
4. Collapse the batch and hidden dimensions and perform a batched GEMM:
   out[b·H+i, :, :] = left[b·H+i] @ right[b·H+i].T   →   shape (B·H, N, N).
5. Reshape back to (B, N, N, H) and apply a second LayerNorm (FP16).
6. Compute the output‑gate from the normalized input, apply it element‑wise
   using a tiny Triton kernel, and finally project back to the original
   channel dimension (FP32 output).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, size, BLOCK: tl.constexpr):
    """c = a * b  (1‑D flattened tensors, FP16)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config["nomask"] is True
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension H
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # Cast everything to FP16 for speed
    # ------------------------------------------------------------------
    x = input_tensor.to(torch.float16)

    # LayerNorm over the input channel dimension
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias = weights["norm.bias"].to(device).to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ------------------------------------------------------------------
    # Linear projections (proj + gate) for left and right sides
    # ------------------------------------------------------------------
    # combine proj & gate weights (2·H × C)
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)          # (2*H, C)

    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)       # (2*H, C)

    B, N, _, _ = x.shape
    x_flat = x.view(B * N * N, dim)               # (B·N·N, C)

    # Compute combined linear maps
    left_comb = F.linear(x_flat, left_comb_w)     # (B·N·N, 2*H)
    right_comb = F.linear(x_flat, right_comb_w)   # (B·N·N, 2*H)

    # Split back to proj / gate
    left_proj, left_gate = torch.split(left_comb, hidden_dim, dim=-1)
    right_proj, right_gate = torch.split(right_comb, hidden_dim, dim=-1)

    # Restore 4‑D layout
    left_proj = left_proj.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_proj = right_proj.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # Optional mask (applied before gating, exactly as in the reference)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate          # (B, N, N, H)
    right = right_proj * right_gate       # (B, N, N, H)

    # ------------------------------------------------------------------
    # Batched GEMM: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Collapse batch & hidden into a single batch dimension
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, H)

    # ------------------------------------------------------------------
    # Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # (B, N, N, H)

    # ------------------------------------------------------------------
    # Output gate (computed from the previously normalized input)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # Element‑wise multiplication with the gate – Triton kernel
    # ------------------------------------------------------------------
    total_el = out.numel()
    BLOCK = 65536
    grid = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        out.ravel(),
        out_gate.ravel(),
        out.ravel(),
        total_el,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # Final projection back to original channel dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out.view(-1, hidden_dim), to_out_w)   # (B·N·N, C)
    out = out.view(B, N, N, dim).to(torch.float32)

    return out
