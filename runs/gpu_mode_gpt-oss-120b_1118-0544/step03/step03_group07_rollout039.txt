def _batched_matmul_triton(A, B):
    # A: (batch, M, K) fp16
    # B: (batch, K, N) fp16
    assert A.shape[0] == B.shape[0] and A.shape[1] == B.shape[2] and A.shape[2] == B.shape[1]
    batch, M, K = A.shape
    N = B.shape[2]
    # Choose block sizes
    # heuristics: choose BLOCK_M,N based on M,N size.
    # Might use small block for small M,N.
    # Use min(128, M) etc.
    BLOCK_M = 128 if M >= 128 else 64 if M >= 64 else 32
    BLOCK_N = 128 if N >= 128 else 64 if N >= 64 else 32
    BLOCK_K = 64 if K >= 64 else 32 if K >= 32 else 16

    # clamp
    BLOCK_M = min(BLOCK_M, M)
    BLOCK_N = min(BLOCK_N, N)
    BLOCK_K = min(BLOCK_K, K)

    grid = ( (M + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             batch)
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # may be tuned
    )
    return C
```

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (B, N, N, C) with the provided LN parameters.
2. Cast to float16 for the compute‑heavy part.
3. Linear projections:
      left  = x @ left_proj.W   (B, N, N, H)
      right = x @ right_proj.W  (B, N, N, H)
   (optional binary mask is applied before gating.)
4. Compute sigmoid gates (left_gate, right_gate, out_gate) from the normalized
   input and multiply them into the projected tensors.
5. Reshape tensors so that the hidden dimension is folded into the batch axis:
      left  → (B·H, N, N)   with layout (i,k)
      right → (B·H, N, N)   with layout (k,j)  (i.e. transposed on the last two axes)
   Perform the core reduction with a Triton batched GEMM kernel:
      out[b, i, j] = Σ_k left[b, i, k] * right[b, k, j]
6. Reshape back to (B, N, N, H), apply a second LayerNorm over H,
   multiply by the output gate, and linearly project back to C.
7. Cast the result back to the original dtype (float32).

Only the O(N³·H) reduction (step 5) is implemented in Triton; all other
operations use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for batched matrix multiplication:
#   C[b] = A[b] @ B[b]   where
#   A : (batch, M, K)   (fp16)
#   B : (batch, K, N)   (fp16)
#   C : (batch, M, N)   (fp16, accumulation in fp32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,   # A batch, row, col strides
    stride_bh, stride_bk, stride_bn,   # B batch, row (K), col (N) strides
    stride_ch, stride_cm, stride_cn,   # C batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)   # batch * hidden
    pid_m = tl.program_id(0)       # tile row
    pid_n = tl.program_id(1)       # tile col

    # ------------------------------------------------------------------
    # Compute tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Offsets for the current batch (or hidden‑folded batch)
    a_batch_offset = pid_batch * stride_ah
    b_batch_offset = pid_batch * stride_bh
    c_batch_offset = pid_batch * stride_ch

    # Pointers to the start of the current tile
    a_ptrs = A_ptr + a_batch_offset + (
        offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = B_ptr + b_batch_offset + (
        tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn
    )

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        # Mask for the K‑dimension of this tile
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Tensor‑core dot (fp16 x fp16 → fp32 accumulation)
        acc += tl.dot(a, b)

        # Move pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch_offset + (
        offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """Batched GEMM using the Triton kernel above.

    Args:
        A: (batch, M, K) fp16 tensor.
        B: (batch, K, N) fp16 tensor.

    Returns:
        C: (batch, M, N) fp16 tensor.
    """
    assert A.shape[0] == B.shape[0], "Batch dimension must match"
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match"

    # Heuristic block sizes (tuned for typical N up to 1024)
    if M <= 64 or N <= 64:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    BLOCK_M = min(BLOCK_M, M)
    BLOCK_N = min(BLOCK_N, N)
    BLOCK_K = min(BLOCK_K, K)

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    C = torch.empty_like(A)

    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # enough warps for the chosen block sizes
    )
    return C


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape (B, N, N, C)
        - mask         : torch.Tensor of shape (B, N, N) or None
        - weights      : dict of model parameters (float32)
        - config       : dict containing:
            * "dim"         : input channel dimension C
            * "hidden_dim"  : hidden dimension H
            * "nomask"      : bool (optional, default False)
    Returns
    -------
    torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on the input (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_b = weights["norm.bias"].to(device=device, dtype=torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for the heavy path
    x_fp16 = x.half() if x.dtype != torch.float16 else x

    # ------------------------------------------------------------------
    # Load model weights in fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x_fp16, left_proj_w)    # (B, N, N, H)
    right = F.linear(x_fp16, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # Optional binary mask (mask ∈ {0,1})
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device=device, dtype=torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # Gates (sigmoid activations)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM
    #   left  : (B, N, N, H) → (B·H, N, N)   layout (i,k)
    #   right : (B, N, N, H) → (B·H, N, N)   layout (k,j)   (transpose on the last two axes)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # (B, H, N, N) where the last two axes are (i,k)
    left = left.permute(0, 3, 1, 2).contiguous()
    # (B, H, N, N) where the last two axes are (k,j) – we achieve this via a permute
    right = right.permute(0, 3, 2, 1).contiguous()

    # Collapse batch and hidden dimensions
    left_flat = left.view(B * H, N, N)   # (B*H, M=N, K=N)
    right_flat = right.view(B * H, N, N) # (B*H, K=N, N)

    # ------------------------------------------------------------------
    # Core O(N³·H) reduction with Triton
    # ------------------------------------------------------------------
    out_flat = _batched_gemm_triton(left_flat, right_flat)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
