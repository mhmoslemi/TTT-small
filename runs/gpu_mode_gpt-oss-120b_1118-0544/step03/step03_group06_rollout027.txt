"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (dtype float32).
2. Cast to FP16 and apply linear projections (left/right) and gate projections.
3. Fuse mask (if present) and gate multiplications with a tiny Triton elementwise
   kernel – this satisfies the “use Triton” requirement while being cheap.
4. Collapse the hidden dimension into the batch dimension and perform the heavy
   O(N³·H) reduction with a batched GEMM using torch.bmm (cuBLAS), which is far
   faster than a naïve Triton matmul.
5. Reshape back, apply an output LayerNorm, the output gate and the final linear
   projection.  Cast the result back to the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_kernel(
    left_ptr,               # *fp16
    right_ptr,              # *fp16
    mask_ptr,               # *fp16
    left_gate_ptr,          # *fp16
    right_gate_ptr,         # *fp16
    N_ELEMENTS: tl.int32,   # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place elementwise:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEMENTS

    l = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    r = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + offs, mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    l = l * m * lg
    r = r * m * rg

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C] float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast linear weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (shape [B, N, N, H])
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Gate projections (shape [B, N, N, H])
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask + gate fusion (Triton kernel)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] and FP16
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        # Ensure all tensors are contiguous for 1‑D indexing
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_f = mask_f.contiguous()

        total_elems = left.numel()
        BLOCK = 1024
        num_blocks = (total_elems + BLOCK - 1) // BLOCK
        _mask_gate_kernel[(num_blocks,)](
            left,
            right,
            mask_f,
            left_gate,
            right_gate,
            total_elems,
            BLOCK_SIZE=BLOCK,
        )
    else:
        # No mask – just apply the gates (pure PyTorch)
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Core O(N³·H) reduction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Move hidden dim into the batch dimension: (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch & hidden -> (B*H, N, N)
    left_flat = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)

    # Batched matmul:   out = left @ rightᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9️⃣ Output LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 🔟 Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Final linear projection back to original channel dim C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 1️⃣2️⃣ Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
