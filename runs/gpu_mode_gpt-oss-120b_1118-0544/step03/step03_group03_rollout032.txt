def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH_HEAD: tl.constexpr,
    N: tl.constexpr,
    stride_l_batch: tl.constexpr,
    stride_l_i: tl.constexpr,
    stride_l_k: tl.constexpr,
    stride_r_batch: tl.constexpr,
    stride_r_i: tl.constexpr,
    stride_r_k: tl.constexpr,
    stride_o_batch: tl.constexpr,
    stride_o_i: tl.constexpr,
    stride_o_j: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_head_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Ensure within bounds
    mask_m = offs_m < N
    mask_n = offs_n < N
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # iterate over k tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N
        # load left tile (M x K)
        left = tl.load(
            left_ptr + batch_head_idx * stride_l_batch
                + offs_m[:, None] * stride_l_i
                + offs_k[None, :] * stride_l_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # load right tile (N x K) -> we need transposed for dot
        # Load as (K x N) to match dot (M,K) * (K,N)
        right_T = tl.load(
            right_ptr + batch_head_idx * stride_r_batch
                + offs_k[:, None] * stride_r_k
                + offs_n[None, :] * stride_r_i,  # note: for contiguous, stride_r_i = N (row stride)
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # Convert to float32
        left = left.to(tl.float32)
        right_T = right_T.to(tl.float32)
        # dot
        acc += tl.dot(left, right_T)
    # Store to out
    out = acc.to(tl.float16)
    tl.store(
        out_ptr + batch_head_idx * stride_o_batch
            + offs_m[:, None] * stride_o_i
            + offs_n[None, :] * stride_o_j,
        out,
        mask=mask_m[:, None] & mask_n[None, :]
    )

import triton.language as tl

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH_HEAD, N,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_head_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for boundaries
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left block (M x K)
        left = tl.load(
            left_ptr
            + batch_head_idx * stride_l_batch
            + offs_m[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # load right block transposed (K x N) -> shape (K,N)
        right_t = tl.load(
            right_ptr
            + batch_head_idx * stride_r_batch
            + offs_k[:, None] * stride_r_k
            + offs_n[None, :] * stride_r_i,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # convert to fp32 for accumulation
        left = left.to(tl.float32)
        right_t = right_t.to(tl.float32)

        acc += tl.dot(left, right_t)

    # store result
    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + batch_head_idx * stride_o_batch
        + offs_m[:, None] * stride_o_i
        + offs_n[None, :] * stride_o_j,
        out,
        mask=mask_m[:, None] & mask_n[None, :]
    )

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy N³ reduction is performed by a custom Triton kernel that
computes, for each batch and hidden dimension, the matrix product
    out[b, h, i, j] = Σ_k left[b, i, k, h] * right[b, j, k, h]
which is equivalent to the einsum used in the reference implementation.
All linear projections, gating, and the final LayerNorm + output projection
are kept in PyTorch (fp32/fp16) while the core GEMM is executed in fp16
using Tensor‑Core friendly blocks.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,               # fp16 pointer   [batch_head, N, N]
    right_ptr,              # fp16 pointer   [batch_head, N, N]
    out_ptr,                # fp16 pointer   [batch_head, N, N]
    BATCH_HEAD: tl.constexpr,  # total batch * hidden_dim
    N: tl.constexpr,           # sequence length
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_head_idx = tl.program_id(0)      # which (batch, head) we are processing
    pid_m = tl.program_id(1)               # tile index over i dimension
    pid_n = tl.program_id(2)               # tile index over j dimension

    # ---- offsets within the N×N matrix ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over the reduction dimension k ----
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block:   (M, K)
        left = tl.load(
            left_ptr
            + batch_head_idx * stride_l_batch
            + offs_m[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0)
        )
        # right block transposed: (K, N)
        # we load right as (k, j) so that tl.dot(left, right_T) yields (i, j)
        right_t = tl.load(
            right_ptr
            + batch_head_idx * stride_r_batch
            + offs_k[:, None] * stride_r_k
            + offs_n[None, :] * stride_r_i,
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0)
        )

        # cast to fp32 for accumulation
        left = left.to(tl.float32)
        right_t = right_t.to(tl.float32)

        acc += tl.dot(left, right_t)   # (M, K) @ (K, N) -> (M, N)

    # store the result back in fp16
    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + batch_head_idx * stride_o_batch
        + offs_m[:, None] * stride_o_i
        + offs_n[None, :] * stride_o_j,
        out,
        mask=mask_m[:, None] & mask_n[None, :]
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32/bool) or None
            - weights: dict with the following keys (all torch.Tensor):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict containing at least "dim" and "hidden_dim"
                and optionally "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    fp16 = torch.float16
    fp32 = torch.float32

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # -----------------------------------------------------------------
    # 2) Cast to fp16 for the heavy work
    # -----------------------------------------------------------------
    x_h = x.to(fp16)

    # -----------------------------------------------------------------
    # 3) Fuse projection + gate linear layers (fp16)
    # -----------------------------------------------------------------
    # combine left projection and left gate weights
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(fp16).to(device)                     # (2*H, dim)
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(fp16).to(device)                     # (2*H, dim)

    # single fused linear per side
    left_combined = F.linear(x_h, left_combined_w)   # [B, N, N, 2*H]
    right_combined = F.linear(x_h, right_combined_w)

    # split back into projection and gate pre‑activations
    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    # -----------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(fp16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # -----------------------------------------------------------------
    # 5) Gating (sigmoid, fp16)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate          # [B, N, N, H] fp16
    right = right_proj * right_gate       # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 6) Output gate (fp16)
    # -----------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(fp16).to(device)
    out_gate_pre = F.linear(x_h, out_gate_w)            # [B, N, N, H] fp16
    out_gate = torch.sigmoid(out_gate_pre)              # fp16

    # -----------------------------------------------------------------
    # 7) Core N³ reduction via custom Triton kernel
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # move hidden dimension to batch dimension for a packed GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_head = B * H
    left_flat = left_perm.view(batch_head, N, N)
    right_flat = right_perm.view(batch_head, N, N)

    # allocate output buffer (fp16)
    out_flat = torch.empty_like(left_flat, dtype=fp16)

    # strides for the flattened tensors (they are contiguous)
    s_l_batch, s_l_i, s_l_k = left_flat.stride()
    s_r_batch, s_r_i, s_r_k = right_flat.stride()
    s_o_batch, s_o_i, s_o_j = out_flat.stride()

    # tuning parameters – can be adjusted for the target GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_head,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_gemm_kernel[grid](
        left_ptr=left_flat,
        right_ptr=right_flat,
        out_ptr=out_flat,
        BATCH_HEAD=batch_head,
        N=N,
        stride_l_batch=s_l_batch,
        stride_l_i=s_l_i,
        stride_l_k=s_l_k,
        stride_r_batch=s_r_batch,
        stride_r_i=s_r_i,
        stride_r_k=s_r_k,
        stride_o_batch=s_o_batch,
        stride_o_i=s_o_i,
        stride_o_j=s_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to [B, N, N, H] and promote to fp32 for the final steps
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(fp32)   # [B, N, N, H] fp32

    # -----------------------------------------------------------------
    # 8) Final LayerNorm, gating and projection back to original dim
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device),
        bias=weights["to_out_norm.bias"].to(device),
        eps=1e-5,
    )
    # apply the output gate (convert gate to fp32 for the multiplication)
    out = out * out_gate.to(fp32)

    # final linear projection to `dim`
    out = F.linear(out, weights["to_out.weight"].to(device))

    return out
