"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the provided norm parameters.
2. Cast to fp16 and compute three linear projections (left, right, three gates)
   with no bias.  Gate values are passed through a sigmoid.
3. Apply an optional binary mask to the left/right projections.
4. Fuse left/right with their respective gates (element‑wise multiplication).
5. Collapse the hidden dimension H into the batch dimension and compute the
   core O(N³·H) operation:
        out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
   which is a batched matrix multiplication C = A @ Bᵀ for each hidden slice.
   This reduction is performed by a custom Triton kernel (fp16 inputs,
   fp32 accumulation, tensor‑core friendly dot).
6. Reshape back to (B,N,N,H), apply a second LayerNorm over the hidden dim,
   multiply by the output gate and project back to the original channel size.
7. Return the result in the original dtype (float32).

The heavy O(N³·H) reduction is the only part off‑loaded to Triton; all other
operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (both fp16, accumulate in fp32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    M, N, K,                                 # matrix sizes (all equal to seq_len)
    stride_ah, stride_am, stride_ak,         # strides for A: batch, row (M), col (K)
    stride_bh, stride_bk, stride_bn,         # strides for B: batch, K, N  (B is accessed as K×N)
    stride_ch, stride_cm, stride_cn,         # strides for C: batch, row (M), col (N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch index (B * H)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column

    # --------------------------------------------------------------
    # Compute offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # Pointers to tiles of A and B
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        # K‑wise mask for the current tile
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back result (convert back to fp16)
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton batched GEMM kernel.
    A, B: fp16 tensors of shape (batch, N, N)
    Returns C of shape (batch, N, N) and dtype fp16.
    """
    assert A.shape == B.shape, "A and B must have identical shapes"
    batch, N, K = A.shape
    assert N == K, "TriMul requires square sequence dimension"

    # Choose tile sizes heuristically based on N
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:  # up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along M
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along N
        batch,                          # batch dimension (B * hidden_dim)
    )

    C = torch.empty_like(A)

    _batched_matmul_kernel[grid](
        A, B, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),   # A: (batch, M, K)
        B.stride(0), B.stride(2), B.stride(1),   # B accessed as (batch, K, N)
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict containing model parameters
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 -> fp16 for the heavy path)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight.to(input_tensor.dtype),
        bias=norm_bias.to(input_tensor.dtype),
    )
    # Cast to fp16 for all subsequent ops (except final cast back)
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast linear weights to fp16 (no bias terms)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # shape: (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)   # shape: (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B,N,N) -> (B,N,N,1) and cast to fp16
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Compute gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply gates to projections
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Fold hidden dimension into batch for the O(N³·H) reduction
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim
    # Bring the hidden dimension next to batch for a contiguous view
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    left  = left.view(B * H, N, N)   # (B*H, N, N)
    right = right.view(B * H, N, N)  # (B*H, N, N)

    # ------------------------------------------------------------------
    # 7) Core reduction via Triton batched GEMM (C = A @ Bᵀ)
    # ------------------------------------------------------------------
    out_flat = _batched_matmul_triton(left, right)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8) Reshape back to (B,N,N,H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9) Output LayerNorm over hidden dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 12) Cast back to the original dtype (float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
