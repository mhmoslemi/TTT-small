def batch_matmul_triton(A, B):
    # A shape (batch, M, K)
    # B shape (batch, N, K) (so we treat B as transposed (K,N))
    assert A.shape[0] == B.shape[0] and A.shape[2] == B.shape[2]
    batch = A.shape[0]; M = A.shape[1]; N = B.shape[1]; K = A.shape[2]
    # allocate output
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)
    # get strides in element counts
    stride_Ab = A.stride(0)
    stride_Am = A.stride(1)
    stride_Ak = A.stride(2)
    stride_Bb = B.stride(0)
    # For B treat as (batch, K, N) for multiplication: we need stride over K (stride_Bk) and over N (stride_Bn)
    # B is stored as (batch, N, K). So B.stride(1) is stride_N, B.stride(2) is stride_K
    # In memory order (N, K). For our kernel we need to index B as B_ptr + k*stride_Bk + n*stride_Bn.
    # So we set:
    stride_Bk = B.stride(2)  # stride for K dimension
    stride_Bn = B.stride(1)  # stride for N dimension
    # Output strides:
    stride_Cb = C.stride(0)
    stride_Cm = C.stride(1)
    stride_Cn = C.stride(2)
    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64

    # Compute grid size
    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch * num_blocks_m * num_blocks_n, )
    # Call kernel
    batch_matmul_T_kernel[grid](

        stride_Am=stride_Am,

@triton.jit
def batch_matmul_T_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bn, stride_Bk,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute batch index and block indices
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # Compute offsets for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # use float32 for accumulation

    # Loop over K dimension in chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offsets for K
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + batch_idx * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B: shape (BLOCK_K, BLOCK_N) (since B is K x N for gemm)
        b_ptrs = B_ptr + batch_idx * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Dot product accumulate
        acc += tl.dot(a, b)

    # Write back to C
    c_ptrs = C_ptr + batch_idx * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    # Convert acc to output dtype
    acc = acc.to(tl.float16) if C_ptr.dtype == tl.float16 else acc
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

```python
import torch
import triton
import triton.language as tl

@triton.jit
def batch_matmul_T_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bn, stride_Bk,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # Determine batch and block indices
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A : shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + batch_idx * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B (as KxN) : shape (BLOCK_K, BLOCK_N)
        b_ptrs = B_ptr + batch_idx * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = C_ptr + batch_idx * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    # Cast back to fp16 (assume compute is fp16)
    acc = acc.to(tl.float16)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

def batch_matmul_triton(A, B, BLOCK_M=64, BLOCK_N=64, BLOCK_K=64):
    # A: (batch, M, K), B: (batch, N, K) (same K)
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)
    # Strides (in elements)
    stride_Ab = A.stride(0)
    stride_Am = A.stride(1)
    stride_Ak = A.stride(2)
    stride_Bb = B.stride(0)
    stride_Bn = B.stride(1)  # N dimension
    stride_Bk = B.stride(2)  # K dimension
    stride_Cb = C.stride(0)
    stride_Cm = C.stride(1)
    stride_Cn = C.stride(2)
    # Compute grid
    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch * num_blocks_m * num_blocks_n
    grid = (total_blocks, )
    batch_matmul_T_kernel[grid](
        A_ptr=A,
        B_ptr=B,
        C_ptr=C,
        batch=batch,
        M=M,
        N=N,
        K=K,
        stride_Ab=stride_Ab,
        stride_Am=stride_Am,
        stride_Ak=stride_Ak,
        stride_Bb=stride_Bb,
        stride_Bn=stride_Bn,
        stride_Bk=stride_Bk,
        stride_Cb=stride_Cb,

        stride_Cn=stride_Cn,

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on input (dim axis)
    x = torch.nn.functional.layer_norm(input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"])

    # Cast to fp16 for compute
    x = x.to(torch.float16)

    # Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)

    # Apply mask if needed
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Fuse gates
    left = left * left_gate
    right = right * right_gate

    # Core reduction using triton batched GEMM
    # reshape to (B*H, N, N)
    B, N, _, H = left.shape
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (batch_h, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Compute out_mat = left_mat @ right_mat.T using triton
    out_mat = batch_matmul_triton(left_mat, right_mat)  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layer norm
    out = torch.nn.functional.layer_norm(out,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b)

    # Apply output gate (elementwise)
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to input dtype (float32 typically)
    out = out.to(input_tensor.dtype)

    return out

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Cast to fp16 and compute linear projections (left/right) and their gates.
3. Apply optional binary mask, then fuse gates into the projected tensors.
4. Core O(N³·H) reduction is performed as a batched matrix multiplication:
   for each (batch, hidden) pair we compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a GEMM L_h @ R_hᵀ.  This is implemented with a
   custom Triton kernel that processes many (batch·hidden) matrices in parallel.
5. Apply a second Layer‑Norm over the hidden dimension.
6. Multiply the result with the output gate (element‑wise, still in fp16).
7. Project back to the original channel dimension.
8. Cast the result to the original dtype (usually float32) and return.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batch_matmul_T_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bn, stride_Bk,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """C[batch, M, N] = A[batch, M, K] @ B[batch, N, K].T  (B is stored as N×K)."""
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_idx = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A: (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + batch_idx * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # B (treated as K×N): (BLOCK_K, BLOCK_N)
        b_ptrs = B_ptr + batch_idx * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write back in fp16
    c_ptrs = C_ptr + batch_idx * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor,
                           BLOCK_M=64, BLOCK_N=64, BLOCK_K=64) -> torch.Tensor:
    """
    Batched matrix multiplication using Triton.

    Computes C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]   (i.e. A @ Bᵀ).

    Args:
        A: (batch, M, K) tensor (fp16, contiguous)
        B: (batch, N, K) tensor (fp16, contiguous) – stored as N×K
        BLOCK_*: tiling sizes

    Returns:
        C: (batch, M, N) tensor (fp16)
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Strides in element units
    stride_Ab = A.stride(0)
    stride_Am = A.stride(1)
    stride_Ak = A.stride(2)

    stride_Bb = B.stride(0)
    stride_Bn = B.stride(1)   # N dimension
    stride_Bk = B.stride(2)   # K dimension

    stride_Cb = C.stride(0)
    stride_Cm = C.stride(1)
    stride_Cn = C.stride(2)

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    _batch_matmul_T_kernel[grid](
        A_ptr=A,
        B_ptr=B,
        C_ptr=C,
        batch=batch,
        M=M,
        N=N,
        K=K,
        stride_Ab=stride_Ab,
        stride_Am=stride_Am,
        stride_Ak=stride_Ak,
        stride_Bb=stride_Bb,
        stride_Bn=stride_Bn,
        stride_Bk=stride_Bk,
        stride_Cb=stride_Cb,
        stride_Cm=stride_Cm,
        stride_Cn=stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model weights (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy compute path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)  # (H, C)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)  # (H, C)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, C)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)      # (C, H)

    # ------------------------------------------------------------------
    # 4) Linear projections (left/right)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # (B, N, N, H)
    right = torch.nn.functional.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 6) Gate projections and sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Fuse gates into left/right
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate
    # free intermediate gates
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 8) Core O(N³·H) reduction via a Triton batched GEMM:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    = L_h @ R_hᵀ for each (b, h) pair.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) where the hidden dim is merged into the batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H, N, N)

    # Compute batched matmul: left_mat @ right_matᵀ
    out_mat = _batched_matmul_triton(left_mat, right_mat)               # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 9) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 12) Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
