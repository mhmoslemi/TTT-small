import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """c = a * b  (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Optimized forward pass of outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float), may be ignored
        - weights      : dict of torch.Tensor weights (float32)
        - config       : dict with keys:
            * "dim" (int) – input channel dimension C
            * "hidden_dim" (int) – hidden projection dimension
            * "nomask" (bool, optional) – if True, mask is not applied

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # Load & cast weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 1. LayerNorm on input (keep fp16 to stay on tensor cores)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # cast to fp16 for the rest of the pipeline
    x = x.to(torch.float16)

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Fused linear projection for left/right and gates
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices (shape (4*H, dim))
    combined_weight = torch.cat(
        [left_proj_weight, left_gate_weight, right_proj_weight, right_gate_weight],
        dim=0,
    )
    # flatten spatial dimensions and apply linear map
    x_flat = x.view(-1, dim)                       # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 4*H)
    # reshape back to 4‑D tensor
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each: (B, N, N, H)

    # ------------------------------------------------------------------
    # 3. Optional mask on projections (mask is broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: (B, N, N) -> (B, N, N, 1) -> fp16
        mask_f = mask.to(x.dtype).unsqueeze(-1).to(torch.float16)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and combine with projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate   # (B, N, N, H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a large batched matmul.
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Compute batched matrix multiplication (left * right^T)
    # result shape: (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. to_out_norm (LayerNorm over hidden dimension)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 7. out_gate (linear + sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # (B, N, N, H) fp16

    # Triton element‑wise multiplication: out *= out_gate
    total_el = out.numel()
    BLOCK = 1024
    grid = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        out,
        out_gate,
        out,
        total_el,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 8. Final projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # (B, N, N, dim) fp16

    # Return float32 as required
    return out.to(torch.float32)

"""
Outgoing Triangle Multiplicative Update (TriMul) – fused forward pass.

Algorithm
---------
1. Layer‑norm the input (fp16) using the supplied norm weights.
2. Fuse the four linear layers (left/right projections + their gates) into a
   single matmul by concatenating the four weight matrices.
3. (Optional) apply the pair‑wise mask to the left/right projections.
4. Apply sigmoid gates and multiply them with the projections to obtain
   `left` and `right` tensors of shape (B, N, N, H).
5. Perform the O(N³) reduction as a batched GEMM:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is realized with a single torch.bmm over (B·H) matrices.
6. Layer‑norm the intermediate result over the hidden dimension.
7. Compute the output gate (linear + sigmoid) and fuse it with the tensor
   using a tiny Triton element‑wise kernel.
8. Final linear projection back to the original channel dimension and cast
   the result to float32.

At least one operation (the final element‑wise multiplication) is performed
with a custom Triton kernel to satisfy the requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """c = a * b   (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/float) – may be ignored
        - weights      : dict of torch.Tensor (float32)
        - config       : dict with keys
            * "dim" (int) – input channel dimension
            * "hidden_dim" (int) – hidden projection dimension
            * "nomask" (bool, optional) – if True, mask is not applied

    Returns
    -------
    torch.Tensor
        Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # Load and cast weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (keep fp16 for the rest)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    x = x.to(torch.float16)          # fp16 for subsequent matmuls

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Fused linear projection for left/right and their gates
    # ------------------------------------------------------------------
    combined_weight = torch.cat(
        [left_proj_weight, left_gate_weight, right_proj_weight, right_gate_weight],
        dim=0,
    )                                   # (4*H, dim)

    x_flat = x.view(-1, dim)            # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)   # (B*N*N, 4*H)

    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )                                   # each (B, N, N, H)

    # ------------------------------------------------------------------
    # 3. Optional mask on the projections (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # (B, N, N, H) fp16

    total_el = out.numel()

    elementwise_mul_kernel[grid](

        out_gate,
        out,
        total_el,
        BLOCK=BLOCK,
    )

    # 8. Final projection back to the original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # (B, N, N, dim) fp16
    return out.to(torch.float32)
