    # convert to fp16
    x = x.to(torch.float16)
    
    # Cast weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    
    # linear proj
    left = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w)
    
    # mask
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # broadcast over H
        left = left * mask_f
        right = right * mask_f
    
    # gating
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))
    
    left = left * left_gate
    right = right * right_gate
    
    B, N, _, H = left.shape
    
    # reshape for batched matmul
    left = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C).
2. Apply linear projections (left/right) and three gating linear layers (left,
   right, output) using fp16 weights; activations are sigmoid.
3. Optional binary mask (if provided) is applied to the projected tensors.
4. Multiply left/right by their respective gates.
5. Fold the hidden dimension H into the batch dimension:
      (B,N,N,H) -> (B*H,N,N).  Compute the core O(N³·H) reduction with a
   custom Triton kernel that evaluates a batched matrix multiplication
   C = A @ Bᵀ (A and B are (batch,N,N) fp16 tensors, accumulation in fp32).
6. Reshape the result back to (B,N,N,H), apply a second LayerNorm over H,
   multiply by the output gate and linearly project back to the original
   channel dimension C.
7. Cast the result to the original dtype of the input.

Only the heavy N³ reduction is performed in Triton; all other ops use
high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,           # pointers
    M, N, K,                       # matrix dims (all equal to seq_len)
    stride_ah, stride_am, stride_ak,   # A: batch, row, col strides
    stride_bh, stride_bk, stride_bn,   # B: batch, k (col of A), n (col of result) strides
    stride_ch, stride_cm, stride_cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b,i,j] = Σ_k A[b,i,k] * B[b,k,j] (i.e. A @ Bᵀ)."""
    pid_b = tl.program_id(2)            # batch index (B*H)
    pid_m = tl.program_id(0)            # block row index
    pid_n = tl.program_id(1)            # block column index

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # Pointers to the current tile (K dimension will be iterated)
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        # K‑dimension mask for the current tile
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back in fp16
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """Wrapper for the batched GEMM kernel (A @ Bᵀ)."""
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert M == N and K == K2, "M, N, K dimensions must match."

    # Heuristic block sizes (tuned for common seq_len ranges)
    if M <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif M <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
    elif M <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:  # up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,   # rows
        (N + BLOCK_N - 1) // BLOCK_N,   # cols
        batch,                          # batch dimension (B*H)
    )

    C = torch.empty_like(A)

    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),   # A strides
        B.stride(0), B.stride(2), B.stride(1),   # B: treat dim2 as K, dim1 as N
        C.stride(0), C.stride(1), C.stride(2),   # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Work in fp16 for the heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast linear weights to fp16 (no bias terms)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B,N,N,H)
    right = F.linear(x, right_proj_w)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4) Optional mask (binary) – broadcast over hidden dim
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Fold hidden dimension into batch for the kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N) layout, then view as (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # ------------------------------------------------------------------
    # 7) Core reduction C = left @ rightᵀ via Triton
    # ------------------------------------------------------------------
    out_flat = _batched_matmul_triton(left, right)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8) Reshape back to (B,N,N,H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 12) Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
