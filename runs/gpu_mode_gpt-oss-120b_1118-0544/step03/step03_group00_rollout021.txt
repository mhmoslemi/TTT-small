"""
TriMul (outgoing) forward pass with Triton acceleration.
  1. LayerNorm on input (FP32) → cast to fp16.
  2. Linear projections (left/right) and gates (left_gate/right_gate/out_gate)
     using fp16 weights.
  3. Optional mask+gate fusion via a small Triton kernel.
  4. Core O(N³·H) reduction: for each batch*head compute
        C = A @ Bᵀ   (A,B shape N×N, batch‑head flattened)
     Implemented as a batched fp16 matrix‑multiply kernel with fp32 accumulation.
  5. Second LayerNorm, out‑gate multiplication and final linear projection.
  6. Cast back to the original dtype.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Elementwise kernel: proj * gate * mask (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,       # (B,N,N,H) flattened → pointer
    gate_ptr,       # (B,N,N,H) flattened
    mask_ptr,       # (B,N,N) flattened (fp16)
    out_ptr,        # (B,N,N,H) flattened
    total_elems,    # B*N*N*H
    hidden_dim,     # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # index into mask (broadcast over hidden dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """Fuse element‑wise multiplication with optional mask."""
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Batched matrix multiplication kernel for the core reduction.
# Computes C = A @ Bᵀ  where A,B are (B*H,N,N) and C has the same shape.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_h, N,
    stride_ah, stride_ai, stride_ak,          # A: (batch_h,N,N) strides
    stride_bh, stride_br, stride_bc,          # B (transposed view) strides
    stride_ch, stride_ci, stride_ck,          # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)          # encodes batch_h * block_m
    pid_n = tl.program_id(axis=1)        # block_n

    # number of tiles along M dimension
    num_blocks_m = tl.cdiv(N, BLOCK_M)

    batch_idx = pid // num_blocks_m
    block_m = pid % num_blocks_m
    block_n = pid_n

    # offsets within the tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction axis K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load A tile (M x K) ----
        a_ptrs = a_ptr + \
                 batch_idx * stride_ah + \
                 offs_m[:, None] * stride_ai + \
                 offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0).to(tl.float32)

        # ---- load Bᵀ tile (K x N) ----
        # Bᵀ is accessed as rows=offs_k, cols=offs_n on the original B
        b_ptrs = b_ptr + \
                 batch_idx * stride_bh + \
                 offs_k[:, None] * stride_br + \
                 offs_n[None, :] * stride_bc
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptr + \
             batch_idx * stride_ch + \
             offs_m[:, None] * stride_ci + \
             offs_n[None, :] * stride_ck
    c = acc.to(tl.float16)
    tl.store(c_ptrs,
             c,
             mask=(mask_m[:, None] & mask_n[None, :]))


def tri_mul_matmul(left: torch.Tensor, right: torch.Tensor,
                   block_m: int = 64, block_n: int = 64, block_k: int = 64) -> torch.Tensor:
    """
    Batched matmul for the TriMul core:
        out[b*h, i, j] = Σ_k left[b*h, i, k] * right[b*h, j, k]
    left/right: (B, H, N, N) contiguous tensors (dtype fp16)
    Returns tensor of same shape (B, H, N, N) dtype fp16.
    """
    B, H, N, _ = left.shape
    batch_h = B * H
    # view as (batch_h, N, N)
    left_flat = left.view(batch_h, N, N)
    right_flat = right.view(batch_h, N, N)
    out_flat = torch.empty_like(left_flat)

    # strides in element units
    stride_batch = left_flat.stride(0)   # N * N
    stride_row = left_flat.stride(1)     # N
    stride_col = left_flat.stride(2)     # 1

    num_blocks_m = (N + block_m - 1) // block_m
    num_blocks_n = (N + block_n - 1) // block_n
    grid = (batch_h * num_blocks_m, num_blocks_n)

    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_h,
        N,
        # A strides
        stride_batch, stride_row, stride_col,
        # B (transposed) strides: row stride = stride_col, col stride = stride_row
        stride_batch, stride_col, stride_row,
        # C strides (same layout as A)
        stride_batch, stride_row, stride_col,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    # reshape back to (B, H, N, N)
    return out_flat.view(B, H, N, N)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3).
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (float32) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32 for stability)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for heavy compute
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Gate projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5. Apply mask and gates (fused via Triton if mask present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core reduction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched matmul using a Triton kernel.
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for contiguous batch‑head layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    out_perm = tri_mul_matmul(left_perm, right_perm,
                              block_m=64, block_n=64, block_k=64)  # (B, H, N, N)

    # Restore original layout (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    # LayerNorm on hidden dimension
    out = out.to(torch.float32)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
    )
    out = out.to(torch.float16)

    # Multiply by out‑gate (broadcast over dim)
    out = out * out_gate

    # Final linear projection back to original dim
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # Cast back to input dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
