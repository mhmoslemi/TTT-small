"""
TriMul (outgoing) forward implementation.

The kernel follows the reference PyTorch module but fuses several steps
and keeps the heavy O(N³) reduction as a batched half‑precision GEMM
performed by torch.matmul (cuBLAS).  The only mandatory Triton kernel
applies the final out‑gate element‑wise multiplication.
All other tensor work is done in fp16 for speed, and the final result
is up‑cast to float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """c = a * b  (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config["nomask"] is True
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # Load and cast weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    # Linear weights for proj/gates (no bias in the reference model)
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    # Output norm & projection
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the input channels (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 and perform fused projection (proj + gate)
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)                     # fp16 for the rest
    B, N, _, _ = x_fp16.shape

    # Combine the four linear transforms into one big matrix multiply
    combined_weight = torch.cat(
        [left_proj_weight,
         left_gate_weight,
         right_proj_weight,
         right_gate_weight],
        dim=0,
    )  # (4*hidden, dim)

    # Flatten spatial dims, apply linear mapping
    x_flat = x_fp16.view(-1, dim)                    # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 4*hidden)

    # Restore shape and split
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each: (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (applied only to the projections)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # (B, N, N, 1)
        mask_f = mask_f.to(torch.float16)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and combine with projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate      # (B, N, N, hidden)
    right = right_proj * right_gate   # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched matrix multiplication using cuBLAS.
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) for batched matmul
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    # Merge batch & hidden dimensions
    left_mat = left_mat.view(B * hidden_dim, N, N)
    right_mat = right_mat.view(B * hidden_dim, N, N)

    # Perform GEMM: (B*hidden, N, N) @ (B*hidden, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    # Reshape back to (B, N, N, hidden)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6️⃣ to_out_norm (LayerNorm over hidden dimension)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden) fp16

    # ------------------------------------------------------------------
    # 7️⃣ Compute out_gate and apply it (Triton kernel)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # (B, N, N, hidden) fp16

    # Triton element‑wise multiplication: out *= out_gate
    total_el = out.numel()
    BLOCK = 1024
    grid = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        out,
        out_gate,
        out,
        total_el,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Final projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # (B, N, N, dim) fp16
    out = out.to(torch.float32)         # return as float32
    return out
