def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_lbm, stride_lbk, stride_lb,
    stride_rbk, stride_rbn, stride_rb,
    stride_obm, stride_obn, stride_ob,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # accumulate in fp32

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):

        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load left tile: shape (BLOCK_M, BLOCK_K)

    left_ptr, right_ptr, out_ptr,
    BATCH, M, N, K,
    stride_lbatch, stride_lm, stride_lk,
    stride_rbatch, stride_rk, stride_rn,
    stride_obatch, stride_om, stride_on,

def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_lbatch, stride_lm, stride_lk,
    stride_rbatch, stride_rk, stride_rn,
    stride_obatch, stride_om, stride_on,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr, num_stages: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load left tile A: (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + batch_id * stride_lbatch + offs_m[:, None] * stride_lm + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile B_T: (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + batch_id * stride_rbatch + offs_k[:, None] * stride_rk + offs_n[None, :] * stride_rn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = out_ptr + batch_id * stride_obatch + offs_m[:, None] * stride_om + offs_n[None, :] * stride_on
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the input tensor (B, N, N, C) using the provided LN parameters.
2. Cast to fp16 for the heavy compute path.
3. Linear projections (left, right) and three gating projections are evaluated
   with `torch.nn.functional.linear`.  The optional binary mask is applied,
   then the gates are fused into the projected tensors.
4. The core O(N³·H) contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed by a custom Triton GEMM kernel that computes, for each
   (batch * hidden) matrix pair, the product `left @ rightᵀ`.  The kernel
   tiles the matrices (128×128 tiles, K‑tiles of size 32) and accumulates in
   fp32, storing the result as fp16.
5. The result is reshaped back to (B, N, N, H) and a second LayerNorm
   (`to_out_norm`) is applied.
6. The output gate (`out_gate`) is multiplied element‑wise (torch multiplication).
7. Final linear projection back to the original channel dimension.
8. Cast back to the original dtype (usually float32) and return.

Only the O(N³) matrix multiplication is implemented in Triton; all other
operations use fused PyTorch kernels, ensuring both correctness and high
performance on the target B200 GPU.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,          # *fp16  [batch, M, K]
    right_ptr,         # *fp16  [batch, N, K]
    out_ptr,           # *fp16  [batch, M, N]
    M, N, K,           # matrix dimensions (int64)
    stride_lbatch, stride_lm, stride_lk,   # left strides
    stride_rbatch, stride_rk, stride_rn,   # right strides (rk = 1, rn = K for transposed view)
    stride_obatch, stride_om, stride_on,   # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
    num_stages: tl.constexpr,
):
    """Compute out = left @ rightᵀ for each batch element."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block col

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load tile of left  : (BLOCK_M, BLOCK_K)
        a_ptrs = (left_ptr
                  + batch_id * stride_lbatch
                  + offs_m[:, None] * stride_lm
                  + offs_k[None, :] * stride_lk)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load tile of right transposed : (BLOCK_K, BLOCK_N)
        # right is stored as [batch, N, K]; we interpret it as [batch, K, N]
        b_ptrs = (right_ptr
                  + batch_id * stride_rbatch
                  + offs_k[:, None] * stride_rk   # row stride = 1
                  + offs_n[None, :] * stride_rn)  # col stride = K
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Matrix multiply‑accumulate
        acc += tl.dot(a, b)

    # Store the result as fp16
    c_ptrs = (out_ptr
              + batch_id * stride_obatch
              + offs_m[:, None] * stride_om
              + offs_n[None, :] * stride_on)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict with the model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Prepare FP16 weights
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)  # (H, C)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)  # (H, C)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, C)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)      # (C, H)

    # ------------------------------------------------------------------
    # 4) Linear projections (left/right) and gate projections
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)

    # Optional binary mask (broadcast over hidden dim)
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # Gates (sigmoid after linear)
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Fuse gates into projected tensors
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core O(N³·H) contraction via a fused Triton GEMM
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Rearrange to (B, H, N, N) and make contiguous for efficient tiling
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch = B * hidden_dim                     # total number of independent matrices
    left_mat = left.view(batch, N, N)          # (batch, M, K)  where M=K=N
    right_mat = right.view(batch, N, N)        # (batch, N, K)

    out_mat = torch.empty((batch, N, N), dtype=torch.float16, device=device)

    # Tile sizes – chosen for good occupancy on A100/B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                                   # M, N, K
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(2), right_mat.stride(1),  # rk=1, rn=K
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7) Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9) Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
