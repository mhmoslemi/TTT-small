def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0) # combined batch and hidden dimension
    batch_idx = pid // H
    hid = pid % H

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for off-grid
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (M, K)
        left_ptrs = left_ptr + batch_idx * stride_l_batch + hid * stride_l_h + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load right tile: shape (K, N) (because we need transposed)
        right_ptrs = right_ptr + batch_idx * stride_r_batch + hid * stride_r_h + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot
        acc += tl.dot(left, right)  # left (M,K), right (K,N) -> M,N

    # store result
    out_ptrs = out_ptr + batch_idx * stride_o_batch + hid * stride_o_h + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    # Cast to half for compute

        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x = x.to(compute_dtype)

    # 2. Combine linear weights for left/right projections and gates
    # All weights are shape (out_features, in_features)
    left_w = weights["left_proj.weight"]  # shape (hidden_dim, dim)
    right_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    right = combined[..., H:2*H]

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # combined batch and hidden dimension
    batch_idx = pid // H
    hid = pid % H

    pid_m = tl.program_id(1)  # tile index along i
    pid_n = tl.program_id(2)  # tile index along j

    # compute start offsets for the M (i) and N (j) dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M x K)
        left_ptrs = left_ptr + batch_idx * stride_l_batch + hid * stride_l_h + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load right tile transposed (K x N)
        right_ptrs = right_ptr + batch_idx * stride_r_batch + hid * stride_r_h + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Dot product
        acc += tl.dot(left, right)  # left (M,K), right (K,N) -> (M,N)

    # Write back
    out_ptrs = out_ptr + batch_idx * stride_o_batch + hid * stride_o_h + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    This implementation fuses the six per‑position linear projections into a
    single matrix multiplication, then performs the core O(N³·H) reduction
    with a custom Triton kernel that computes a batched GEMM for each hidden
    channel.  LayerNorm, gating, and the final linear projection use
    standard PyTorch kernels.
    """
    input_tensor, mask, weights, config = data
    dtype = input_tensor.dtype
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # all heavy compute in float16
    compute_dtype = torch.float16
    x = x.to(compute_dtype)

    # ------------------------------------------------------------------ #
    # 2. Fuse linear projections (left/right) and gates into one matmul
    # ------------------------------------------------------------------ #
    # concatenate weight matrices: each is (hidden_dim, dim)
    left_w = weights["left_proj.weight"]
    right_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # shape: (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [left_w, right_w, left_gate_w, right_gate_w, out_gate_w], dim=0
    ).to(compute_dtype)

    B, N, _, C = x.shape
    x_flat = x.reshape(-1, C)  # (B*N*N, C)
    combined = F.linear(x_flat, combined_weight)  # (B*N*N, 5*hidden_dim)
    combined = combined.view(B, N, N, hidden_dim * 5)

    H = hidden_dim
    left = combined[..., 0:H]
    right = combined[..., H:2*H]
    left_gate = combined[..., 2*H:3*H].sigmoid()
    right_gate = combined[..., 3*H:4*H].sigmoid()
    out_gate = combined[..., 4*H:5*H].sigmoid()

    # ------------------------------------------------------------------ #
    # 3. Apply mask (if any) and gates to produce gated tensors
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        mask_f = mask.to(compute_dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 4. Core O(N³·H) reduction via Triton kernel
    # ------------------------------------------------------------------ #
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left, dtype=compute_dtype)  # (B,N,N,H)
    _tri_mul_gpu(left, right, out)  # fills `out`

    # ------------------------------------------------------------------ #
    # 5. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------ #
    # 6. Apply output gate
    # ------------------------------------------------------------------ #
    out = out * out_gate

    # ------------------------------------------------------------------ #
    # 7. Final projection back to original channel dimension
    # ------------------------------------------------------------------ #
    to_out_w = weights["to_out.weight"].to(compute_dtype)
    out = F.linear(out, to_out_w)

    # Cast back to original dtype if necessary
    if out.dtype != dtype:
        out = out.to(dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation fuses the five per‑position linear projections (left/right
projections and three gates) into a single matmul, then performs the core
O(N³·H) reduction with a custom Triton kernel that computes a batched
GEMM for each hidden channel (out = left @ rightᵀ).  The remaining
LayerNorm, gating, and final projection use regular PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid = tl.program_id(0)                     # combined (batch, hidden) dimension
    batch_idx = pid // H
    hid = pid % H

    pid_m = tl.program_id(1)                   # tile index along i
    pid_n = tl.program_id(2)                   # tile index along j

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (M, K)
        left_ptrs = left_ptr + batch_idx * stride_l_batch + hid * stride_l_h \
                    + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile transposed: shape (K, N)  (rightᵀ)
        right_ptrs = right_ptr + batch_idx * stride_r_batch + hid * stride_r_h \
                     + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        acc += tl.dot(left, right)               # (M, K) • (K, N) → (M, N)

    out_ptrs = out_ptr + batch_idx * stride_o_batch + hid * stride_o_h \
               + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _tri_mul_gpu(left: torch.Tensor, right: torch.Tensor, out: torch.Tensor):
    """Launch the Triton kernel for the O(N³·H) reduction."""
    B, N, _, H = left.shape
    # strides in elements
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
        stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
        stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dtype = input_tensor.dtype
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float16 compute)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
    )
    compute_dtype = torch.float16
    x = x.to(compute_dtype)

    # ------------------------------------------------------------------ #
    # 2. Fuse all per‑position linear projections (left/right) and gates
    # ------------------------------------------------------------------ #
    # concatenate weights: each is (hidden_dim, dim)
    left_w = weights["left_proj.weight"]
    right_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    combined_weight = torch.cat(
        [left_w, right_w, left_gate_w, right_gate_w, out_gate_w], dim=0
    ).to(compute_dtype)  # shape (5*H, dim)

    B, N, _, C = x.shape
    x_flat = x.reshape(-1, C)                     # (B·N·N, C)
    combined = F.linear(x_flat, combined_weight) # (B·N·N, 5·H)
    combined = combined.view(B, N, N, hidden_dim * 5)

    H = hidden_dim
    left = combined[..., 0:H]
    right = combined[..., H:2 * H]
    left_gate = combined[..., 2 * H:3 * H].sigmoid()
    right_gate = combined[..., 3 * H:4 * H].sigmoid()
    out_gate = combined[..., 4 * H:5 * H].sigmoid()

    # ------------------------------------------------------------------ #
    # 3. Apply optional mask and element‑wise gates
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        mask_f = mask.to(compute_dtype).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 4. Core reduction via Triton (batched GEMM per hidden channel)
    # ------------------------------------------------------------------ #
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left, dtype=compute_dtype)   # (B, N, N, H)
    _tri_mul_gpu(left, right, out)                     # fills `out`

    # ------------------------------------------------------------------ #
    # 5. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(compute_dtype),
        bias=weights["to_out_norm.bias"].to(compute_dtype),
    )

    # ------------------------------------------------------------------ #
    # 6. Apply output gate
    # ------------------------------------------------------------------ #
    out = out * out_gate

    # ------------------------------------------------------------------ #
    # 7. Final projection back to original channel dimension
    # ------------------------------------------------------------------ #
    to_out_w = weights["to_out.weight"].to(compute_dtype)
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to the original dtype if needed
    if out.dtype != dtype:
        out = out.to(dtype)

    return out
