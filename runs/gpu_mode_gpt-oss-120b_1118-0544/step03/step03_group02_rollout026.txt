import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# define Triton kernel for batched GEMM
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_abatch,
    stride_bk, stride_bn, stride_bbatch,
    stride_cm, stride_cn, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for OOB
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_offset in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k_offset * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        # Load A tile
        a_ptr = A_ptr + batch_id * stride_abatch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # Load B tile (shape BLOCK_K x BLOCK_N)
        b_ptr = B_ptr + batch_id * stride_bbatch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # compute dot product
        acc += tl.dot(a, b)

    # Write back
    c_ptr = C_ptr + batch_id * stride_cbatch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    # apply mask for output OOB
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
Optimized outgoing TriMul (Triangle Multiplicative Update) forward pass.

Algorithm:
1. LayerNorm over the input channel dimension.
2. Fuse the four linear projections (left/right and their gates) into a single
   matrix multiplication to reduce memory traffic.
3. Apply optional mask to the left/right projections.
4. Apply sigmoid gates and form gated tensors `left` and `right`.
5. Compute the core N³ reduction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   using a custom Triton batched GEMM kernel (fp16 accumulation in fp32).
6. LayerNorm on the hidden dimension, element‑wise multiplication with the
   output gate and final linear projection back to the original channel size.
All heavy compute is performed in float16; the final result is returned as
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (A @ Bᵀ) where A,B have shape (batch, M, K)
# and B is provided already transposed as (batch, K, N).
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A, B, C,                # pointers
    M, N, K,                # matrix sizes
    stride_am, stride_ak, stride_ab,   # A strides
    stride_bk, stride_bn, stride_bb,   # B strides (B is K×N)
    stride_cm, stride_cn, stride_cb,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptr = A + pid_batch * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (K x N) – B is already transposed (K dimension first)
        b_ptr = B + pid_batch * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back the result in FP16
    c_ptr = C + pid_batch * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – may be ignored if config["nomask"] is True
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load weights, cast to fp16 where appropriate
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)

    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)  # shape (dim, hidden)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over input channels (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # still float32
    x = x.to(torch.float16)                     # fp16 for the rest

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2️⃣ Fuse four linear layers (proj + gate for left/right)
    # ------------------------------------------------------------------
    # Combine weights for left_proj, left_gate, right_proj, right_gate
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # (4*hidden, dim)

    x_flat = x.view(-1, dim)                    # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 4*hidden)

    combined_out = combined_out.view(B, N, N, 4 * hidden)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden, dim=-1
    )  # each: (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied only to projections)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # (B, N, N, 1)
        mask_f = mask_f.to(torch.float16)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate      # (B, N, N, hidden)
    right = right_proj * right_gate   # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ reduction via Triton batched GEMM
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Rearrange for batched GEMM: (B, hidden, N, N) -> (B*hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    bs = B * hidden
    left_mat = left.view(bs, N, N)                # (bs, M=N, K=N)
    right_mat = right.view(bs, N, N)

    # Transpose right for GEMM (K×N)
    right_t = right_mat.transpose(1, 2).contiguous()  # (bs, K=N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat)          # (bs, N, N), fp16

    # Strides
    stride_am = left_mat.stride(1)   # stride over M (N)
    stride_ak = left_mat.stride(2)   # stride over K (1)
    stride_ab = left_mat.stride(0)   # batch stride

    stride_bk = right_t.stride(1)    # stride over K (N)
    stride_bn = right_t.stride(2)    # stride over N (1)
    stride_bb = right_t.stride(0)    # batch stride

    stride_cm = out_mat.stride(1)    # stride over M (N)
    stride_cn = out_mat.stride(2)    # stride over N (1)
    stride_cb = out_mat.stride(0)    # batch stride

    # Triton block sizes (tuned for fp16 on L2/L1)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        bs,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_gemm_kernel[grid](
        left_mat,
        right_t,
        out_mat,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1)  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6️⃣ Output layer‑norm (over hidden dimension)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Output gate
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B, N, N, hidden), fp16
    out = out * out_gate                               # element‑wise, fp16

    # ------------------------------------------------------------------
    # 8️⃣ Final projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)  # (B, N, N, dim), fp16
    out = out.to(torch.float32)    # required output dtype

    return out
```
