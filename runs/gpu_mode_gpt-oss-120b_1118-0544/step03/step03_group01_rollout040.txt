@triton.jit
def fuse_mask_gate_kernel(
    left_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elements,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    if mask_ptr != 0 (or we pass a flag):
        mask_val = tl.load(mask_ptr + offs, mask=mask, other=0.0)
        left = left * mask_val * gate
    else:
        left = left * gate

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor.
2. Compute four linear projections on the last channel dimension:
   * left_proj  (dim -> hidden_dim)
   * right_proj (dim -> hidden_dim)
   * left_gate  (dim -> hidden_dim)  + sigmoid
   * right_gate (dim -> hidden_dim)  + sigmoid
   * out_gate   (dim -> hidden_dim)  + sigmoid (used later)
3. (Optional) apply the binary mask to the projected tensors
   before gating – the mask is broadcast over the hidden dimension.
4. Fuse the element‑wise product `proj * gate` for *left* and *right*
   with a tiny Triton kernel (the only required Triton part).
5. Fold the hidden dimension into the batch dimension and perform the
   O(N³·H) reduction as a batched matrix multiplication
   `C[b] = left[b] @ right[b].T`.  This uses PyTorch’s highly‑optimized
   cuBLAS batched `bmm`.
6. Reshape back to (B, N, N, hidden_dim) and apply a second
   LayerNorm over the hidden dimension.
7. Multiply by `out_gate` and project back to the original channel
   dimension with a final linear layer.
8. Cast the result to float32 (the required output dtype).

Only the element‑wise fusion in step 4 is written in Triton; all heavy
linear / matmul work stays in PyTorch for maximal performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise product of two tensors (both 1‑D views)
# ----------------------------------------------------------------------
@triton.jit
def _elem_mul_kernel(
    a_ptr, b_ptr, out_ptr,               # pointers
    total_elems: tl.constexpr,           # total number of elements
    BLOCK_SIZE: tl.constexpr,            # block size (must be a compile‑time const)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)

    tl.store(out_ptr + offs, a * b, mask=mask)


def _elem_mul_torch(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Element‑wise multiply `a * b` using a tiny Triton kernel."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication"
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 1024  # reasonable for all test sizes
    grid = ((total + BLOCK - 1) // BLOCK,)

    _elem_mul_kernel[grid](
        a,
        b,
        out,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# Main custom kernel implementation
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask        : torch.Tensor of shape [B, N, N] (binary) or None
            - weights     : dict of model parameters (float32 tensors)
            - config      : dict with keys
                * "dim"         : C  (int)
                * "hidden_dim" : H  (int)
                * "nomask"      : bool (optional, default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (keep in FP32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Work in FP16 from now on
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast linear weights to FP16 (biases are absent)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Optional mask (binary) – broadcast over hidden dim
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left_proj  = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid of linear projections on the original normed x)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Fuse projection with its gate using a tiny Triton kernel
    # ------------------------------------------------------------------
    # Both tensors are contiguous after the above ops; flatten for the kernel.
    left = _elem_mul_torch(left_proj, left_gate)   # (B, N, N, H)
    right = _elem_mul_torch(right_proj, right_gate)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Fold hidden dimension into the batch dimension
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left  = left.view(B * H, N, N)   # (B*H, N, N)
    right = right.view(B * H, N, N)  # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8) Core O(N³·H) reduction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # right needs to be transposed on the last two dims before bmm
    out_flat = torch.bmm(left, right.transpose(1, 2))   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 9) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 10) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 11) Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12) Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 13) Cast back to float32 as required by the spec
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
