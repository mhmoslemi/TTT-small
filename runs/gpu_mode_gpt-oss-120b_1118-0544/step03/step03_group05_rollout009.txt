```python
"""
TriMul out-of-place kernel.

Implements the outgoing triangle multiplicative update from AlphaFold3.
The heavy O(N³·H) contraction is performed with a batched GEMM (torch.bmm)
while the linear projections are fused (proj+gate) to reduce the number of
matrix multiplications. A small Triton kernel is used for the element‑wise
multiplication of the output with the output‑gate.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def _elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, n_elements,
                            BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)

def _elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch Triton kernel for element‑wise a*b."""
    assert a.shape == b.shape, "shape mismatch for elementwise mul"
    total = a.numel()
    c = torch.empty_like(a)
    BLOCK_SIZE = 32768  # adjust for GPU
    grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE,)
    _elementwise_mul_kernel[grid](
        a, b, c, total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    return c

@torch.no_grad()
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            mask: torch.Tensor of shape [B, N, N] (or None)
            weights: dict of model parameters (fp32)
            config: dict containing "dim" (C) and "hidden_dim" (H); optional
                    "nomask" bool.

    Returns:
        torch.Tensor of shape [B, N, N, C] with same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nominal_mask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Compute in fp16 for speed
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Fuse proj + gate for left and right branches
    # ------------------------------------------------------------------
    # Concatenate projection and gate weights (out_features stacked)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)               # (2*H, C)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)               # (2*H, C)

    # Flatten batch+seq dims for a single matmul
    B, N, _, C = x.shape
    x_flat = x.view(-1, C)                           # (B*N*N, C)

    # Combined linear layers
    left_concat = torch.nn.functional.linear(x_flat, left_concat_w)   # (B*N*N, 2*H)
    right_concat = torch.nn.functional.linear(x_flat, right_concat_w) # (B*N*N, 2*H)

    # Individual gates
    out_gate = torch.nn.functional.linear(
        x_flat, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()                                      # (B*N*N, H)

    # Reshape back
    left_concat = left_concat.view(B, N, N, 2 * hidden_dim)
    right_concat = right_concat.view(B, N, N, 2 * hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # Split proj/gate and apply sigmoid gate
    left_proj = left_concat[..., :hidden_dim]
    left_gate = left_concat[..., hidden_dim:].sigmoid()
    left = left_proj * left_gate

    right_proj = right_concat[..., :hidden_dim]
    right_gate = right_concat[..., hidden_dim:].sigmoid()
    right = right_proj * right_gate

    # Free temporary tensors early
    del left_concat, right_concat, left_proj, left_gate, right_proj, right_gate

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nominal_mask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Core O(N³·H) contraction via batched GEMM
    #    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # ------------------------------------------------------------------
    # Rearrange to (B*H, N, N) and use torch.bmm
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 6) Multiply by output gate (Triton kernel)
    # ------------------------------------------------------------------
    out = _elementwise_mul(out, out_gate)

    # ------------------------------------------------------------------
    # 7) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

    # --------------------------------------------------------------
    # 4) Core O(N³·H) contraction via batched GEMM
    #    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # --------------------------------------------------------------
    # Rearrange to (B·H, N, N) and invoke torch.bmm
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B·H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N) # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))            # (B·H, N, N)

    # Restore shape [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5) Output LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # --------------------------------------------------------------
    # 6) Multiply by the output gate (Triton kernel)
    # --------------------------------------------------------------
    out = _elementwise_mul(out, out_gate)

    # --------------------------------------------------------------
    # 7) Final projection back to original channel dimension
    # --------------------------------------------------------------
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )  # (B, N, N, C)

    # --------------------------------------------------------------
    # 8) Cast back to the original dtype
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
```
