def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device

    # Cast input to fp16 for performance
    x = input_tensor.to(torch.float16)

    # LayerNorm: use torch.nn.functional.layer_norm on fp16
    # weight and bias are FP32, cast to fp16
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias = weights["norm.bias"].to(device).to(torch.float16)

    x = torch.nn.functional.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections fused:
    # left_proj, left_gate, right_proj, right_gate
    # weight shapes: (hidden_dim, dim)
    # We will stack them: (4*hidden_dim, dim)
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)

    # combined weight: (4*hidden_dim, dim)
    combined_weight = torch.cat([left_proj_weight, left_gate_weight, right_proj_weight, right_gate_weight], dim=0)  # (4*hidden_dim, dim)

    # Apply linear to flattened input: shape (B*N*N, dim) @ (4*hidden_dim, dim).T? Actually torch.nn.functional.linear(x, weight) expects weight shape (out_features, in_features). So we can call F.linear(x_flat, combined_weight). This yields (B*N*N, 4*hidden_dim).
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)  # (B*N*N, dim)
    combined_out = torch.nn.functional.linear(x_flat, combined_weight)  # (B*N*N, 4*hidden_dim)

    # Reshape
    combined_out = combined_out.view(B, N, N, 4*hidden_dim)

    # Split
    left_proj, left_gate, right_proj, right_gate = torch.split(combined_out, hidden_dim, dim=-1)

    # Apply mask if needed: mask shape (B,N,N); unsqueeze(-1) and cast to fp16.
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Gating: sigmoid - use torch.sigmoid
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # Combine
    left = left_proj * left_gate  # (B,N,N,H), fp16
    right = right_proj * right_gate

    # Rearrange for batched matmul
    # left: (B,N,N,H) -> (B*H, N, N) after permute (B,H,N,N)
    left_permuted = left.permute(0, 3, 1, 2).contiguous()
    right_permuted = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_permuted.view(B * hidden_dim, N, N)
    right_mat = right_permuted.view(B * hidden_dim, N, N)

    # Compute batched matmul
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # shape (B*H, N, N)

    # Reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,H)

    # Apply to_out_norm (layernorm over hidden_dim). Use fp16 weights and bias.
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # Compute out_gate: linear from x (original x after LN) and sigmoid
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)
    # x is still fp16 after LN; compute linear
    out_gate = torch.nn.functional.linear(x_flat, out_gate_weight)  # shape (B*N*N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(out_gate)

    # Multiply out by out_gate elementwise using Triton kernel for demonstration (must be a kernel)
    # out and out_gate are fp16; implement a Triton kernel for elementwise multiplication.
    # This kernel will read two tensors and write product.

    # Use flatten to 1D pointers
    total = out.numel()
    BLOCK = 1024
    grid = lambda meta: ( (total + meta['BLOCK'] - 1) // meta['BLOCK'], )
    @triton.jit
    def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N, BLOCK: tl.constexpr):
        pid = tl.program_id(0)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < N
        a = tl.load(a_ptr + offs, mask=mask)
        b = tl.load(b_ptr + offs, mask=mask)
        tl.store(c_ptr + offs, a * b, mask=mask)

    # Launch kernel
    elementwise_mul_kernel[grid](out, out_gate, out, total, BLOCK=BLOCK)

    # final linear projection back to dim:
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)   # shape (dim, hidden_dim) (out_features, in_features)
    # flatten out (B,N,N,H) -> (B*N*N, hidden_dim)
    out_flat = out.view(-1, hidden_dim)
    out_proj = torch.nn.functional.linear(out_flat, to_out_weight)  # shape (B*N*N, dim)
    out_proj = out_proj.view(B, N, N, dim)

    # Cast back to float32
    out_proj = out_proj.to(torch.float32)
    return out_proj

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) kernel.

Algorithm
---------
1. Layer‑normalize the input tensor over the channel dimension (C) using the
   learned weight/bias (FP16 for speed).
2. Fuse the four linear projections (left_proj, left_gate, right_proj,
   right_gate) into a single matrix multiplication with a stacked weight
   matrix (shape (4·H, C)).  This reduces memory traffic and launches only one
   GEMM.
3. Apply the optional mask to the projection tensors (only to the
   left/right projections, not the gates).
4. Apply sigmoid to the gate tensors and multiply them with the
   corresponding projections, yielding ``left`` and ``right`` tensors of
   shape (B, N, N, H) in FP16.
5. Perform the core O(N³) reduction via a batched matrix multiplication:
   for each hidden channel d,
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].
   This is implemented as a single torch.bmm over the merged batch dimension
   (B·H), which leverages cuBLAS/Tensor‑cores.
6. Layer‑normalize the result over the hidden dimension (H) and compute the
   output gate (sigmoid of a linear projection of the normalized input).
7. Apply the output gate element‑wise using a tiny Triton kernel (required
   custom kernel usage).
8. Project back to the original channel dimension with a final linear layer
   and return the result in float32.

All heavy arithmetic is performed in FP16 to exploit Tensor‑cores on the
Ada‑Lovelace B200 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication (out *= out_gate)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, c_ptr, size, BLOCK: tl.constexpr):
    """c = a * b  (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, C] (float32)
            - mask: torch.Tensor[B, N, N] (bool/float) – may be ignored
            - weights: dict of model weights (float32 tensors)
            - config: dict containing:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension H
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor[B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # Cast everything to FP16 for the heavy path
    # --------------------------------------------------------------
    # Input (FP16)
    x = input_tensor.to(torch.float16)

    # LayerNorm parameters (FP16)
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias = weights["norm.bias"].to(device).to(torch.float16)

    # Apply LayerNorm over the channel dimension C
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, C] (fp16)

    # --------------------------------------------------------------
    # 1️⃣ Fuse four linear projections (left/right + gates)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)   # (H, C)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)   # (H, C)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16) # (H, C)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16) # (H, C)

    # Stacked weight: (4*H, C)
    stacked_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # (4*H, C)

    # Linear on flattened tensor
    x_flat = x.view(-1, dim)                                 # (B*N*N, C)
    proj_combined = F.linear(x_flat, stacked_weight)        # (B*N*N, 4*H)
    proj_combined = proj_combined.view(B, N, N, 4 * hidden_dim)

    # Split back into the four components
    left_proj, left_gate, right_proj, right_gate = torch.split(
        proj_combined, hidden_dim, dim=-1
    )  # each (B, N, N, H)

    # --------------------------------------------------------------
    # 2️⃣ Optional mask on the projections (not on gates)
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------------------
    # 3️⃣ Gating (sigmoid) and apply to projections
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate    # (B, N, N, H)
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 4️⃣ Core O(N³) reduction via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # Rearrange to (B, H, N, N) for batched matmul
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Merge batch and hidden dims
    left_mat = left_t.view(B * hidden_dim, N, N)      # (B*H, N, N)
    right_mat = right_t.view(B * hidden_dim, N, N)    # (B*H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ LayerNorm over hidden dimension
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # (B, N, N, H), fp16

    # --------------------------------------------------------------
    # 6️⃣ Output gate (sigmoid of linear projection of normalized input)
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    # Re‑use the flattened normalized input x_flat (same as before)
    out_gate = F.linear(x_flat, out_gate_w)                     # (B*N*N, H)
    out_gate = out_gate.view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(out_gate)                          # (B, N, N, H) fp16

    # --------------------------------------------------------------
    # 7️⃣ Element‑wise gate application via Triton
    # --------------------------------------------------------------
    total_elems = out.numel()
    BLOCK = 1024
    grid = ( (total_elems + BLOCK - 1) // BLOCK, )
    _elemwise_mul_kernel[grid](
        out,                # a_ptr
        out_gate,           # b_ptr
        out,                # c_ptr (in‑place)
        total_elems,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dimension C
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)  # (C, H)
    out_flat = out.view(-1, hidden_dim)                               # (B*N*N, H)
    out_proj = F.linear(out_flat, to_out_w)                           # (B*N*N, C)
    out_proj = out_proj.view(B, N, N, dim).to(torch.float32)

    return out_proj
