```python
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # compute program ids
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute range of row indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # guard for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # offsets for block's k dimension
    # accumulate over K using tiles
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in steps of BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offsets for this tile
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # guard for K dimension
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_batch * stride_a_batch + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)  # shape (BLOCK_M, BLOCK_K)

        # Load B_T tile (since we want B_T of shape (K,N)):
        # B tile shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + pid_batch * stride_b_batch + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

        # move pointers? Not needed because we compute offsets each iteration.

    # Write back to C
    c_ptrs = c_ptr + pid_batch * stride_c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    # store with mask
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input [B,N,N,C] (C = dim) using the provided LN weights.
2. Cast to float16 for the heavy compute path.
3. Linear projections (left/right) and gating projections (left_gate, right_gate,
   out_gate) are performed with torch.nn.functional.linear on the last dimension.
4. Optional binary mask is applied (broadcast over the hidden dimension).
5. Gates are fused into the projected tensors.
6. Core O(N³·H) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is implemented as a batched GEMM (A @ Bᵀ) using a custom Triton kernel.
   For each (batch, hidden) pair we multiply two N×N matrices.
7. Reshape back to [B,N,N,H] and apply a second Layer‑Norm over the hidden dim.
8. Multiply by the output gate (element‑wise).
9. Final linear projection back to the original channel dimension (dim).
10. Cast the result back to the input dtype.

At least the O(N³·H) reduction (the performance‑critical part) is executed in
pure Triton code.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ Bᵀ  (A,B shape: [batch, M, K] & [batch, N, K])"""
    pid_batch = tl.program_id(0)          # batch * hidden index
    pid_m = tl.program_id(1)              # tile row
    pid_n = tl.program_id(2)              # tile col

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for boundary handling
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_batch * stride_a_batch \
                       + (offs_m[:, None] * stride_a_m) \
                       + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load Bᵀ tile: shape (BLOCK_K, BLOCK_N)
        # B is provided as (batch, N, K); we treat it as (batch, K, N)
        b_ptrs = b_ptr + pid_batch * stride_b_batch \
                       + (offs_k[:, None] * stride_b_k) \
                       + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Gemm accumulation
        acc += tl.dot(a, b)

    # Write result back to C
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
                   + (offs_m[:, None] * stride_c_m) \
                   + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(a: torch.Tensor, b: torch.Tensor,
                         BLOCK_M=128, BLOCK_N=128, BLOCK_K=32,
                         num_warps=4) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for A,B shaped (batch, N, N) using Triton.
    A and B must be FP16, contiguous tensors.
    Returns C with same dtype (FP16) and shape (batch, N, N).
    """
    B_batch, M, K = a.shape
    _, N, _ = b.shape
    assert M == N == K, "TriMul expects square matrices (N×N)."

    # Strides for A
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    # Strides for B (original shape (batch, N, K))
    stride_b_batch, stride_b_n, stride_b_k = b.stride()
    # Output strides (contiguous)
    c = torch.empty((B_batch, M, N), dtype=a.dtype, device=a.device)
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    grid = (B_batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _batched_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        batch=B_batch,
        M=M,
        N=N,
        K=K,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,          # stride of the K dimension (last dim)
        stride_b_n=stride_b_n,          # stride of the N dimension (middle dim)
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )
    return c


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model weights (LayerNorm, Linear, etc.)
            - config: dict containing "dim" and "hidden_dim" (and optional "nomask")

    Returns:
        torch.Tensor of shape [B, N, N, dim] with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (left/right) and gates (fp16)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)  # (hidden_dim, dim)

    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4) Apply optional binary mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()   # [B,N,N,hidden_dim]
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()    # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6) Fuse gates into left/right
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Core O(N³·H) reduction via Triton batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # B, N, N, hidden_dim
    # Rearrange to (batch*hidden, N, N) for batched GEMM
    left_mat  = left.permute(0, 3, 1, 2).contiguous().reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().reshape(B * H, N, N)

    # Triton GEMM: C = left_mat @ right_matᵀ
    out_mat = _batched_gemm_triton(left_mat, right_mat,
                                   BLOCK_M=128, BLOCK_N=128, BLOCK_K=32,
                                   num_warps=8)   # result: (B*H, N, N), fp16

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 9) Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 11) Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
```
