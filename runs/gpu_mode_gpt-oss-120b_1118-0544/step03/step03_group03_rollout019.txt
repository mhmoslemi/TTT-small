"""Implementation of outgoing TriMul using Triton for the heavy N^3 reduction.
The kernel performs a batched matrix multiplication of shape (B*H, N, N) @ (B*H, N, N)^T
in fp16 using Tensor Cores, while the surrounding linear layers, LayerNorms,
and gating are done in PyTorch (fp32/fp16 as appropriate)."""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block column
    pid_b = tl.program_id(2)  # batch index

    # compute offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointer offsets for the current batch
    a_batch_offset = pid_b * stride_ab
    b_batch_offset = pid_b * stride_bb
    c_batch_offset = pid_b * stride_cb

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k-tiles
    num_k = tl.cdiv(K, BLOCK_K)

    for k in range(num_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # linear offsets for A and B tiles
        a_ptrs = a_ptr + a_batch_offset + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + b_batch_offset + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # mask out-of-bounds
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0).to(tl.float16)
        b = tl.load(b_ptrs, mask=b_mask, other=0).to(tl.float16)

        # accumulate
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptr + c_batch_offset + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Arguments:
        data (tuple): (input_tensor, mask, weights, config)
    Returns:
        torch.Tensor: output tensor [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # ------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # shape [B, N, N, dim], float32

    # Cast to fp16 for compute‑intensive parts
    x_h = x.to(torch.float16)

    # -------------------------------------------------
    # 2. Linear projections + gating (merged where possible)
    # -------------------------------------------------
    # Merge left projection and left gate weights
    left_combined_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16).to(device)  # shape [2*H, dim]

    right_combined_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16).to(device)  # shape [2*H, dim]

    # Combined linear
    left_combined = F.linear(x_h, left_combined_weight)   # [B, N, N, 2*H]
    right_combined = F.linear(x_h, right_combined_weight)  # [B, N, N, 2*H]

    # Split into projection and gate pre‑activations
    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    # Optional mask (broadcast on hidden dimension)
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Gating
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate   # [B, N, N, H] fp16
    right = right_proj * right_gate  # [B, N, N, H] fp16

    # Output gate
    out_gate_weight = weights["out_gate.weight"].to(torch.float16).to(device)
    out_gate_pre = F.linear(x_h, out_gate_weight)   # [B, N, N, H] fp16
    out_gate = torch.sigmoid(out_gate_pre)          # fp16

    # -------------------------------------------------
    # 3. N³ reduction via batched GEMM in Triton
    # -------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # transpose right for the correct multiplication order
    right_T = right_flat.transpose(-2, -1).contiguous()

    # allocate output
    out_flat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # strides (int64)
    stride_am = left_flat.stride(1)   # stride for M dimension
    stride_ak = left_flat.stride(2)   # stride for K dimension
    stride_ab = left_flat.stride(0)   # batch stride

    stride_bk = right_T.stride(1)     # stride for K dimension in B
    stride_bn = right_T.stride(2)     # stride for N dimension in B
    stride_bb = right_T.stride(0)     # batch stride for B

    stride_cm = out_flat.stride(1)    # stride for M dimension in C
    stride_cn = out_flat.stride(2)    # stride for N dimension in C
    stride_cb = out_flat.stride(0)    # batch stride for C

    # tuning parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    batched_matmul_fp16_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_T,
        c_ptr=out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_ab=stride_ab,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_bb=stride_bb,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_cb=stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to [B, N, N, H] (fp16)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 4. Final LayerNorm, gating and projection (FP32)
    # -------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_fp32 = out_fp32 * out_gate.to(torch.float32)  # apply output gate
    out_fp32 = F.linear(out_fp32, weights["to_out.weight"])

    return out_fp32

"""
Outgoing TriMul (AlphaFold3) forward pass.
The heavy N³ reduction is performed by a custom Triton kernel that
computes a batched matrix multiplication
    out[b*h, i, j] = Σ_k left[b, i, k, h] * right[b, j, k, h]
in fp16 using Tensor‑Cores.  All surrounding LayerNorms, linear layers,
and gating are executed in PyTorch (fp32/fp16 as appropriate), while the
final tensor is cast back to fp32 before the last LayerNorm and output
projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched (batch, M, K) @ (batch, K, N) → (batch, M, N) in fp16."""
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column
    pid_b = tl.program_id(2)          # batch index

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # batch offsets
    a_batch_off = pid_b * stride_ab
    b_batch_off = pid_b * stride_bb
    c_batch_off = pid_b * stride_cb

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # pointers to A and B tiles
        a_ptrs = a_ptr + a_batch_off + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + b_batch_off + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # mask out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0).to(tl.float16)
        b = tl.load(b_ptrs, mask=b_mask, other=0).to(tl.float16)

        # tensor‑core dot (accumulates to fp32)
        acc += tl.dot(a, b)

    # write result
    c_ptrs = c_ptr + c_batch_off + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model weights
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor: output [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # -------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # -------------------------------------------------
    # 2. Cast to fp16 for the heavy part
    # -------------------------------------------------
    x_h = x.to(torch.float16)

    # -------------------------------------------------
    # 3. Linear projections + gates (merged where possible)
    # -------------------------------------------------
    left_combined_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16).to(device)               # (2*H, dim)

    right_combined_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16).to(device)               # (2*H, dim)

    left_combined = F.linear(x_h, left_combined_weight)    # [B, N, N, 2*H]
    right_combined = F.linear(x_h, right_combined_weight) # [B, N, N, 2*H]

    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    # optional mask (broadcast on hidden dimension)
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # gating
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate    # [B, N, N, H] fp16
    right = right_proj * right_gate  # [B, N, N, H] fp16

    # output gate
    out_gate_weight = weights["out_gate.weight"].to(torch.float16).to(device)
    out_gate_pre = F.linear(x_h, out_gate_weight)   # [B, N, N, H] fp16
    out_gate = torch.sigmoid(out_gate_pre)          # fp16

    # -------------------------------------------------
    # 4. N³ reduction via batched GEMM (Triton)
    # -------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # right is transposed for the dot product
    right_T = right_flat.transpose(-2, -1).contiguous()

    out_flat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # strides (int64)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    stride_ab = left_flat.stride(0)

    stride_bk = right_T.stride(1)
    stride_bn = right_T.stride(2)
    stride_bb = right_T.stride(0)

    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    stride_cb = out_flat.stride(0)

    # tile sizes – chosen to map well onto Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    batched_matmul_fp16_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_T,
        c_ptr=out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_ab=stride_ab,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_bb=stride_bb,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_cb=stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to [B, N, N, H] (fp16)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 5. Final LayerNorm, gating & projection (float32)
    # -------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_fp32 = out_fp32 * out_gate.to(torch.float32)   # apply output gate
    out_fp32 = F.linear(out_fp32, weights["to_out.weight"])

    return out_fp32
