"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The forward pass consists of
1. LayerNorm over the channel dimension.
2. Two linear projections (left/right) and three per‑channel gates (left, right, out).
3. Optional binary mask applied before gating.
4. Element‑wise gating of the projected tensors.
5. Core O(N³·H) reduction:
   For each hidden channel `h` we compute
       out_h[b,i,j] = Σ_k left_h[b,i,k] * right_h[b,j,k]
   which is a batched matrix multiplication C = A @ Bᵀ with
   A, B shaped (B·H, N, N).  This is the only part written in Triton.
6. LayerNorm over the hidden dimension, gating by `out_gate`, and final linear
   projection back to the original channel dimension.

All heavy arithmetic is performed in FP16 with FP32 accumulation.
The final tensor is returned in the dtype of the input (float32 or float16).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dimensions (M=N=N_seq, K=N_seq)
    stride_ah, stride_am, stride_ak,   # strides for A: batch, row, col
    stride_bh, stride_bk, stride_bn,   # strides for B: batch, k (row), n (col) → B is viewed as (batch, K, N)
    stride_ch, stride_cm, stride_cn,   # strides for C: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM kernel for the TriMul reduction:
        C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]
    A has shape (batch, M, K)   (row‑major)
    B is interpreted as (batch, K, N)   (so we index B with (k, j))
    C has shape (batch, M, N)
    """
    pid_b = tl.program_id(2)   # batch index (B * hidden_dim)
    pid_m = tl.program_id(0)   # block row index
    pid_n = tl.program_id(1)   # block column index

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current batch slice
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # Pointers for the tile (M,N) at the current K‑tile start
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                              + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                              + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # (BLOCK_M, BLOCK_K) fp16
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # (BLOCK_K, BLOCK_N) fp16

        acc += tl.dot(a, b)            # fp16×fp16 → fp32 accumulation

        a_ptrs += BLOCK_K * stride_ak   # move to next K‑tile
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _tri_mul_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matmul for TriMul: C = A @ Bᵀ (A,B shape (batch, N, N), dtype fp16).
    Returns C of the same shape and dtype.
    """
    assert A.shape == B.shape, "A and B must have identical shapes"
    batch, N, K = A.shape
    assert N == K, "TriMul requires square sequence dimension"

    # Choose block sizes (multiples of 16 for WMMA)
    if N <= 256:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over M
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over N
        batch,                          # batch * hidden
    )

    C = torch.empty_like(A)

    # Strides: note that B is accessed as (batch, K, N)
    _tri_mul_matmul_kernel[grid](
        A,
        B,
        C,
        N,                     # M
        N,                     # N
        N,                     # K
        A.stride(0), A.stride(1), A.stride(2),   # A: batch, row, col
        B.stride(0), B.stride(2), B.stride(1),   # B interpreted as (batch, K, N)
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] or None (binary mask)
            - weights: dict of layer parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------
    # 1) Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 2) LayerNorm over channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Work in FP16 for the heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast weights to FP16 (no bias terms in linear layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Optional mask (binary) – broadcast over hidden dimension
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape (B, N, N) → (B, N, N, 1)
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 6) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Fold hidden dimension into the batch dimension for the kernel
    #    Shape conversion: (B, N, N, H) → (B·H, N, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Permute to bring hidden dim to the second axis and make contiguous
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_flat  = left.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right.view(B * H, N, N)  # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8) Core O(N³·H) reduction via Triton batched GEMM
    #    Compute out_flat[b,i,j] = Σ_k left_flat[b,i,k] * right_flat[b,j,k]
    # ------------------------------------------------------------------
    out_flat = _tri_mul_matmul_triton(left_flat, right_flat)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 9) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 10) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 11) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 13) Cast back to input dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
