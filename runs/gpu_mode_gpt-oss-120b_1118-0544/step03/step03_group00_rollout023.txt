Implementation of kernel:

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # Get program IDs
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # Compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # Compute pointers
        a_ptrs = a_ptr + pid_batch * stride_batch_a + offs_m[:, None] * stride_m_a + offs_k[None, :] * stride_k_a
        b_ptrs = b_ptr + pid_batch * stride_batch_b + offs_k[:, None] * stride_k_b + offs_n[None, :] * stride_n_b
        # Load A and B
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        # Accumulate
        acc += tl.dot(a, b)
    # Write back
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_batch_c + offs_m[:, None] * stride_m_c + offs_n[None, :] * stride_n_c
    tl.store(c_ptrs, c, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

"""
TriMul (outgoing) forward pass, fused with Triton.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the provided LN parameters.
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (left/right) and three gating projections are computed with
   fp16 weights and a sigmoid activation.
4. If a mask is supplied, a small Triton kernel fuses elementwise
   multiplication of projection * gate * mask for each of left/right.
5. Core reduction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication A @ Bᵀ where for every hidden
   dimension d we multiply two (N×N) matrices.  A Triton kernel performs the
   batched GEMM across the combined batch dimension B·hidden_dim.
6. The result is layer‑normed, multiplied by the output gate, and projected back
   to the original channel dimension.
7. Cast back to the input dtype (float32) and return.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ------------------------------------------------------------
# Triton kernel: proj * gate * mask  (mask broadcast over hidden)
# ------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,          # (B*N*N*H,) fp16
    gate_ptr,          # (B*N*N*H,) fp16
    mask_ptr,          # (B*N*N,)   fp16
    out_ptr,           # (B*N*N*H,) fp16
    total_elems,       # int, B*N*N*H
    hidden_dim,        # int, H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Element‑wise fused multiplication proj * gate * mask.
    proj, gate : (B,N,N,H)   fp16
    mask       : (B,N,N)     fp16
    Returns    : (B,N,N,H)   fp16
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
# A : (batch, M, K)   fp16
# B : (batch, K, N)   fp16   (already transposed so that multiplication
#                           yields the required sum over the k‑axis)
# C : (batch, M, N)   fp16
# ------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Pointers for A and B (B already transposed)
        a_ptrs = a_ptr + pid_batch * stride_batch_a \
                 + offs_m[:, None] * stride_m_a \
                 + offs_k[None, :] * stride_k_a
        b_ptrs = b_ptr + pid_batch * stride_batch_b \
                 + offs_k[:, None] * stride_k_b \
                 + offs_n[None, :] * stride_n_b

        # Load with masking
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & (offs_k[None, :] < K)),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=((offs_k[:, None] < K) & mask_n[None, :]),
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_batch * stride_batch_c \
             + offs_m[:, None] * stride_m_c \
             + offs_n[None, :] * stride_n_c
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def batched_gemm(A: torch.Tensor, B: torch.Tensor,
                 block_m: int = 64, block_n: int = 64, block_k: int = 32) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ where A,B are (batch, N, N) fp16 tensors.
    Returns C of same shape and dtype.
    """
    assert A.shape == B.shape
    batch, M, K = A.shape
    _, Kb, N = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty_like(A)

    grid = (batch,
            triton.cdiv(M, block_m),
            triton.cdiv(N, block_n))

    batched_gemm_kernel[grid](
        A, B, C,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        M, N, K,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return C

        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------

    norm_bias = weights["norm.bias"].to(torch.float16)

                     (dim,),
                     weight=norm_weight,

    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask + gates (fused via Triton when mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction: batched GEMM
    #    Compute left @ rightᵀ for every hidden slice.
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) – contiguous layout for fusion
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) – transposed on the last two dims

    B, H, N, _ = left_perm.shape
    batch = B * H

    # Collapse batch dimension for the kernel
    left_flat  = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)

    # Batched GEMM (fp16)
    out_flat = batched_gemm(left_flat, right_flat,
                            block_m=64, block_n=64, block_k=32)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
