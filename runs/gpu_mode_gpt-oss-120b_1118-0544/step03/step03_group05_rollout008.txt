    A_ptr, B_ptr, C_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    M, N, K,
    stride_ak, stride_am,
    stride_bk, stride_bn,
    stride_ck, stride_cm,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)

    mask_n = offs_n < N

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config['dim']; hidden_dim = config['hidden_dim']; nomask = config.get('nomask', False)
    # 1 layernorm
    x = torch.nn.functional.layer_norm(x, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])
    # cast to fp16 for compute
    x_fp16 = x.to(torch.float16)

    # linear projections
    left = torch.nn.functional.linear(x_fp16, weights['left_proj.weight'].to(torch.float16))
    right = torch.nn.functional.linear(x_fp16, weights['right_proj.weight'].to(torch.float16))

    # gates
    left_gate = torch.nn.functional.linear(x_fp16, weights['left_gate.weight'].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, weights['right_gate.weight'].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, weights['out_gate.weight'].to(torch.float16)).sigmoid()

    # apply mask if needed
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1) # shape (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # out = left @ right_T for each batch and hidden
    # prepare shapes
    B, N, _, H = left.shape
    # rearrange: view as (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B*H, N, N) # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B*H, N, N) # original right

    # allocate output
    out_mat = torch.empty_like(left_mat) # will hold (B*H,N,N) in fp16
    # call triton kernel
    _batched_gemm_kernel(...)

    # reshape out to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (H,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'])

    # apply out_gate: fused elementwise multiplication via triton kernel
    out = _elementwise_mul(out, out_gate)

    # final linear to C
    out = torch.nn.functional.linear(out, weights['to_out.weight'].to(torch.float16))

    # cast back to original dtype
    out = out.to(x.dtype)
    return out

"""
TriMul (outgoing) kernel – Triton implementation.

Algorithm
---------
1. Layer‑norm the input tensor (B, N, N, C).
2. Cast to fp16 for compute‑heavy ops.
3. Linear projections → left/right (B, N, N, H) and three gates
   (left_gate, right_gate, out_gate) of the same shape.
4. Optional binary mask applied to left/right before gating.
5. Fuse gates: left *= left_gate,  right *= right_gate.
6. Core O(N³·H) contraction:
      out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   Implemented as a batched GEMM (left @ rightᵀ) with a custom Triton
   kernel that processes the (B·H) batches in parallel.
7. Layer‑norm over the hidden dimension H.
8. Fuse output gate with an element‑wise Triton kernel.
9. Final linear projection back to the original channel dimension C.
10. Cast back to the input dtype (usually float32).

The heavy O(N³) step is fully realized in Triton, while the surrounding
operations use fast PyTorch primitives.  All compute is performed in fp16
with fp32 accumulation for the GEMM.
"""

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernels
# --------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_Ab, stride_Ai, stride_Aj,
    stride_Bb, stride_Bi, stride_Bj,
    stride_Cb, stride_Ci, stride_Cj,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  (all tensors fp16, accumulation fp32).

    A : (batch, M, K)
    B : (batch, N, K)   (original right matrix, we read it transposed)
    C : (batch, M, N)
    """
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col
    pid_b = tl.program_id(2)  # batch index (b * hidden_dim)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block (M x K)
        a_ptrs = (
            A_ptr
            + pid_b * stride_Ab
            + offs_m[:, None] * stride_Ai
            + offs_k[None, :] * stride_Aj
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B block transposed: we need Bᵀ[k, n] = B[n, k]
        # Original B layout: (batch, N, K) → stride_Bi = row‑stride (N), stride_Bj = col‑stride (1)
        b_ptrs = (
            B_ptr
            + pid_b * stride_Bb
            + offs_n[None, :] * stride_Bi   # n‑dimension
            + offs_k[:, None] * stride_Bj   # k‑dimension
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        # Accumulate block‑wise product
        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Ci
        + offs_n[None, :] * stride_Cj
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or fp16)
            - mask: torch.Tensor [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load model weights (cast to fp16 for compute)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)

    left_proj_weight = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    left_gate_weight = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(torch.float16)        # (C, H)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B, N, N, H)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Gate projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # ------------------------------------------------------------------
    # 6) Fuse gates into projected tensors
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate
    # free temporary gates
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 7) Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM: left @ rightᵀ
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) – each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Allocate output buffer
    out_mat = torch.empty_like(left_mat)  # fp16

    # Triton batched GEMM
    _batched_gemm(left_mat, right_mat, out_mat)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    # 9) Apply output gate (element‑wise Triton kernel)
    # ------------------------------------------------------------------
    out = _elemwise_mul(out, out_gate)

    # ------------------------------------------------------------------
    # 10) Final linear projection back to C dimensions
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_weight)

    # ------------------------------------------------------------------
    # 11) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
```
