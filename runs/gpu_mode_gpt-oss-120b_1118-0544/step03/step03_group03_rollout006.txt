def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # retrieving parameters
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # Convert to half
    x_h = x.to(torch.float16)

    # Linear projections
    left_proj = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Gates
    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].to(torch.float16))
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].to(torch.float16))
    out_gate_pre = F.linear(x_h, weights["out_gate.weight"].to(torch.float16))

    # Apply mask
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left_gate = torch.sigmoid(left_gate_pre)  # fp16
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Prepare for batched matmul
    B, N, _, _ = left.shape
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # flatten first two dims (batch*hidden)
    left_flat = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # Output container fp16
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Kernel launch
    BH = B * hidden_dim
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    matmul_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=left_flat.stride(0),
        stride_a_m=left_flat.stride(1),
        stride_a_k=left_flat.stride(2),
        stride_b_batch=right_flat.stride(0),
        stride_b_n=right_flat.stride(1),
        stride_b_k=right_flat.stride(2),
        stride_c_batch=out_flat.stride(0),
        stride_c_m=out_flat.stride(1),
        stride_c_n=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to (B,N,N,hidden_dim)
    out_h = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # Convert to fp32 for final LN and projection
    out_fp32 = out_h.float()

    # Final LayerNorm
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Apply output gate
    out_norm = out_norm * out_gate.float()
    # Final projection
    out = F.linear(out_norm, weights["to_out.weight"])

    return out

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,

"""
TriMul (outgoing) forward pass.

The heavy triple‑wise reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is equivalent to a batch of matrix multiplications
    out_h = left_h @ right_hᵀ
for each hidden dimension h.  This kernel performs the batched
matmul in FP16 using Triton (Tensor‑Core dot), accumulates in FP32,
stores FP16, and the surrounding logic (LayerNorm, linear
projections, gating) is handled in PyTorch.  The final output is
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched A @ Bᵀ where A,B ∈ (batch, M, K) and C ∈ (batch, M, N)."""
    batch_id = tl.program_id(0)          # which batch×hidden slice
    pid_m = tl.program_id(1)             # block row
    pid_n = tl.program_id(2)             # block col

    # Block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to the output tile
    c_tile_ptr = (
        c_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a_tile_ptr = (
            a_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=tl.float16(0))

        # Load B tile transposed (K × N) -> we need Bᵀ, so rows are k, cols are n
        b_tile_ptr = (
            b_ptr
            + batch_id * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )
        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=tl.float16(0))

        acc += tl.dot(a, b)   # FP16 dot → FP32 accumulation

    # Store the result (cast back to FP16)
    tl.store(c_tile_ptr,
             tl.cast(acc, tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask: torch.Tensor or None, shape [B, N, N]; broadcasted to hidden dim
            - weights: dict of model weight tensors (float32)
            - config: dict containing at least "dim" and "hidden_dim"; optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim], dtype float32
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections + gating (all in FP16)
    # ------------------------------------------------------------------
    # Projections
    left_proj = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Gates (pre‑sigmoid)
    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].to(torch.float16))
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].to(torch.float16))
    out_gate_pre = F.linear(x_h, weights["out_gate.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 5. Gating (sigmoid in FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)   # [B, N, N, hidden_dim]

    left = left_proj * left_gate            # [B, N, N, hidden_dim] fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Batched matrix multiplication via Triton:
    #    out_h[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Rearrange to (B, hidden_dim, N, N) → flatten to (BH, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(B * hidden_dim, N, N)    # [BH, N, N]
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # Output buffer (FP16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BH = B * hidden_dim

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _trimul_batched_matmul_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=left_flat.stride(0),
        stride_a_m=left_flat.stride(1),
        stride_a_k=left_flat.stride(2),
        stride_b_batch=right_flat.stride(0),
        stride_b_n=right_flat.stride(1),
        stride_b_k=right_flat.stride(2),
        stride_c_batch=out_flat.stride(0),
        stride_c_m=out_flat.stride(1),
        stride_c_n=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim] (still FP16)
    # ------------------------------------------------------------------
    out_h = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 8. Final LayerNorm, output gate and projection (float32)
    # ------------------------------------------------------------------
    out_fp32 = out_h.float()

    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate.float()
    out = F.linear(out_norm, weights["to_out.weight"])

    return out
