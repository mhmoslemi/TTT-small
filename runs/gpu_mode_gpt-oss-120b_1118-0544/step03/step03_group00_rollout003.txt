"""
TriMul (outgoing) forward pass implemented with Triton.
Algorithm
---------
1. Layer‑Norm the input tensor (B,N,N,C) using the provided LN parameters.
2. Cast to fp16 for compute‑heavy operations.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are done with fp16 weights.
4. If a mask is provided (and not disabled) a Triton kernel fuses the
   element‑wise multiplication of the projection, its gate and the mask:
        out = proj * gate * mask
   This reduces memory traffic compared to separate PyTorch ops.
5. Core update: for each batch and hidden channel d we need
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is exactly a batched matrix multiplication
        left_flat @ right_flatᵀ .
   The tensors are reshaped to (B*H, N, N) and torch.bmm (cuBLAS) is used,
   which leverages Tensor‑cores on modern GPUs.
6. Apply the second Layer‑Norm, multiply by the output gate and a final
   linear projection back to the original channel dimension.
7. Cast back to the original input dtype.

The implementation uses Triton for the mask‑gate fusion kernel and
relies on highly‑optimized cuBLAS for the O(N³·H) reduction.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused (proj * gate * mask) over a flattened view.
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,          # *float16, shape (B*N*N*H,)
    gate_ptr,          # *float16, shape (B*N*N*H,)
    mask_ptr,          # *float16, shape (B*N*N,)
    out_ptr,           # *float16, shape (B*N*N*H,)
    total_elems,       # int: total number of elements in proj/gate (B*N*N*H)
    hidden_dim,        # int: H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds.
    valid = offs < total_elems

    # Compute the corresponding mask index (mask is broadcast over H).
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def apply_gate_and_mask(proj: torch.Tensor,
                        gate: torch.Tensor,
                        mask: torch.Tensor,
                        block_size: int = 1024) -> torch.Tensor:
    """
    Fuse element‑wise multiplication of projection, gate and mask.
    All tensors must be fp16 and contiguous.
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive steps
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16 (no bias in all Linear layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,dim) -> (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)

    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask + gating (fused via Triton if mask present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16, contiguous and has the same batch/seq layout.
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        # No mask – simple elementwise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched matrix multiplication left @ rightᵀ
    #    Shapes: left, right -> (B, N, N, H)
    #    After permutation: (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Batched GEMM using cuBLAS (Tensor‑cores on fp16)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, multiply by out_gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
