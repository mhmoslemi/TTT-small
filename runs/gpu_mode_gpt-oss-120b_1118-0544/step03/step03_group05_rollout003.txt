@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Obtain program ids
    pid = tl.program_id(0)  # combine batch and hidden
    pid_i = tl.program_id(1)  # tile i
    pid_j = tl.program_id(2)  # tile j

    # Compute batch index and hidden index
    bh = pid
    b = bh // H
    h = bh % H

    # Compute start indices for i and j
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k
    for k in range(0, N, BLOCK_K):
        # Compute actual block size (maybe less at edges)
        k_end = tl.minimum(k + BLOCK_K, N)
        cur_k = k_end - k
        # Offsets
        i_range = i_start + tl.arange(0, BLOCK_I)
        j_range = j_start + tl.arange(0, BLOCK_J)
        k_range = k + tl.arange(0, BLOCK_K)
        # Masks for boundaries
        mask_i = i_range < N
        mask_j = j_range < N
        mask_k = k_range < N

        # Load left tile shape (BLOCK_I, cur_k)
        left_ptrs = left_ptr + \
            b * stride_l_b + \
            i_range[:, None] * stride_l_i + \
            k_range[None, :] * stride_l_k + \
            h * stride_l_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile shape (cur_k, BLOCK_J)
        right_ptrs = right_ptr + \
            b * stride_r_b + \
            k_range[:, None] * stride_r_k + \
            j_range[None, :] * stride_r_j + \
            h * stride_r_h
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Multiply-accumulate using dot (matrix multiply)
        # left: (BLOCK_I, cur_k), right: (cur_k, BLOCK_J)
        acc += tl.dot(left, right)

    # Write result to output
    # Compute output pointer offsets
    out_ptrs = out_ptr + \
        b * stride_o_b + \
        i_range[:, None] * stride_o_i + \
        j_range[None, :] * stride_o_j + \
        h * stride_o_h
    # Cast acc to fp16 (output dtype)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C).
2. Cast to float16 and apply linear projections for left/right tensors and their
   three gating tensors (left_gate, right_gate, out_gate).
3. Optional binary mask is applied to left/right before gating.
4. Fuse gates into left/right.
5. Core contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed with a custom Triton kernel that tiles the i‑ and j‑axes and
   accumulates over k using fp16 matrix‐multiply (Tensor‑Core) with fp32
   accumulation.
6. Layer‑norm over the hidden dimension, multiply by out_gate (element‑wise)
   and a final linear projection back to the original channel dimension.
7. Cast the result back to the input dtype.

The heavy O(N³·H) reduction is fully fused in Triton, avoiding the many
small cuBLAS calls of a naïve implementation.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # shape / config
    B, N, H,
    # left strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # right strides (j dimension replaces i)
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    # output strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    with tiling over (i,j) and reduction over k.
    """
    pid0 = tl.program_id(0)      # combined batch * hidden index
    pid_i = tl.program_id(1)     # i‑tile index
    pid_j = tl.program_id(2)     # j‑tile index

    # decode batch and hidden indices
    b = pid0 // H
    h = pid0 % H

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # i/j coordinate vectors
    i_range = i_start + tl.arange(0, BLOCK_I)
    j_range = j_start + tl.arange(0, BLOCK_J)

    mask_i = i_range < N
    mask_j = j_range < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k dimension
    for k_start in range(0, N, BLOCK_K):
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # left tile: (BLOCK_I, BLOCK_K)
        left_offs = (
            b * stride_l_b
            + i_range[:, None] * stride_l_i
            + k_range[None, :] * stride_l_k
            + h * stride_l_h
        )
        left = tl.load(
            left_ptr + left_offs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right tile: (BLOCK_K, BLOCK_J)  (k as rows, j as cols)
        right_offs = (
            b * stride_r_b
            + k_range[:, None] * stride_r_k
            + j_range[None, :] * stride_r_j
            + h * stride_r_h
        )
        right = tl.load(
            right_ptr + right_offs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # matrix multiply – use fp16 inputs, fp32 accumulation
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # write result back (cast to fp16)
    out_offs = (
        b * stride_o_b
        + i_range[:, None] * stride_o_i
        + j_range[None, :] * stride_o_j
        + h * stride_o_h
    )
    tl.store(
        out_ptr + out_offs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            mask: torch.Tensor of shape [B, N, N] (or None)
            weights: dict containing model weights
            config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for compute‑intensive path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Prepare fp16 weight tensors
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)  # (H, C)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)  # (H, C)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)    # (H, C)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    to_out_w = weights["to_out.weight"].to(torch.float16)            # (C, H)

    # ------------------------------------------------------------------
    # 3) Linear projections (left/right) and gates
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)

    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()   # (B,N,N,H)
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4) Fuse gates into projections
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate
    # free gating tensors that are no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5) Core O(N³·H) contraction via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # fp16 output buffer

    # strides (in elements, not bytes)
    l_stride_b, l_stride_i, l_stride_k, l_stride_h = left.stride()
    # right uses the same layout; the second N dimension is accessed via the
    # stride that corresponds to `i` in the left tensor.
    r_stride_b = l_stride_b
    r_stride_j = l_stride_i
    r_stride_k = l_stride_k
    r_stride_h = l_stride_h

    o_stride_b, o_stride_i, o_stride_j, o_stride_h = out.stride()

    # tiling parameters (tuned for B200 / fp16 Tensor Cores)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = (
        B * H,                                 # batch * hidden
        (N + BLOCK_I - 1) // BLOCK_I,          # i‑tiles
        (N + BLOCK_J - 1) // BLOCK_J,          # j‑tiles
    )

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        l_stride_b,
        l_stride_i,
        l_stride_k,
        l_stride_h,
        r_stride_b,
        r_stride_j,
        r_stride_k,
        r_stride_h,
        o_stride_b,
        o_stride_i,
        o_stride_j,
        o_stride_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7) Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9) Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)

    return out
