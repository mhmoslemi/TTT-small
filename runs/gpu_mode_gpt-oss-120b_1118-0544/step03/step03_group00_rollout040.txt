@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a,
    batch_stride_b,
    batch_stride_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    BATCH = tl.program_id(0)
    # Compute tile offsets
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    
    # Compute starting offsets in each dimension
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    
    # Compute pointers
    a_ptrs = a_ptr + BATCH * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + BATCH * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
    
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=offs_m[:, None] < M and offs_k[None, :] < K, other=0.0)
        b = tl.load(b_ptrs, mask=offs_k[:, None] < K and offs_n[None, :] < N, other=0.0)
        acc += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
    # Convert to fp16 if needed
    c = acc.to(tl.float16)
    # store
    c_ptrs = c_ptr + BATCH * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
```

    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    # linear projections (no bias)
    left_proj = torch.nn.functional.linear(x, left_proj_w)  # shape (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x, right_proj_w)
    # gate projections
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()
    # apply mask+gate
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()  # shape (B,N,N,1)
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
    # core reduction using batched GEMM kernel
    # Permute and reshape to (batch, hidden_dim, N, N) -> (batch*hidden_dim, N, N)
    B, N, _, H = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # flatten batch-head
    batch_head = B * H
    left_flat = left_perm.view(batch_head, N, N)
    right_flat = right_perm.view(batch_head, N, N)
    # allocate output
    out_flat = torch.empty_like(left_flat)  # same dtype fp16
    
    # compute strides for kernel
    # a_ptr points to left_flat, b_ptr to right_flat, c_ptr to out_flat
    # get strides
    a_stride_batch = left_flat.stride(0)
    a_stride_m = left_flat.stride(1)   # stride for M dimension (row)
    a_stride_k = left_flat.stride(2)   # stride for K dimension (col)
    b_stride_batch = right_flat.stride(0)
    b_stride_n = right_flat.stride(1)   # row stride (original N dimension)
    b_stride_k = right_flat.stride(2)   # col stride
    c_stride_batch = out_flat.stride(0)
    c_stride_m = out_flat.stride(1)
    c_stride_n = out_flat.stride(2)
    # For transposed B operation, swap b strides
    stride_bk = b_stride_n  # row stride becomes K stride (original row)
    stride_bn = b_stride_k  # col stride becomes N stride
    # Kernel constants
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    # grid dimensions
    grid = (batch_head,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        a_stride_batch,
        b_stride_batch,
        c_stride_batch,
        a_stride_m,
        a_stride_k,
        stride_bk,
        stride_bn,
        c_stride_m,
        c_stride_n,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # output layernorm
    out = torch.nn.functional.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)
    # Cast back to original dtype
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)
    return out
```

"""
TriMul forward (outgoing) with Triton acceleration.

Algorithm
---------
1. Layer‑norm the (B,N,N,C) input.
2. Cast to fp16 for the compute‑heavy path.
3. Linear projections (left/right) and gate projections (left, right, out) are
   performed with fp16 weights.
4. If a mask is present, a tiny Triton kernel fuses mask‑application and gating:
   out = proj * gate * mask   (broadcast over the hidden dimension).
5. Core O(N³·H) reduction is realised as a batched GEMM:
       out_{b,h,i,j} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   which is equivalent to a matrix multiplication of left with the transpose of
   right for each (batch, hidden) pair.  This is implemented in a custom
   Triton kernel to avoid the extra memory passes incurred by torch.bmm.
6. Apply a second LayerNorm, multiply by the output gate and a final linear
   projection back to the original channel dimension.
7. Cast back to the input dtype.

All heavy compute is done in fp16 with float32 accumulation.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Elementwise kernel: proj * gate * mask (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,          # (B,N,N,H) fp16
    gate_ptr,          # (B,N,N,H) fp16
    mask_ptr,          # (B,N,N)   fp16
    out_ptr,           # (B,N,N,H) fp16
    total_elems,       # int = B*N*N*H
    hidden_dim,        # int = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def apply_gate_and_mask(proj: torch.Tensor,
                        gate: torch.Tensor,
                        mask: torch.Tensor,
                        block_size: int = 1024) -> torch.Tensor:
    """
    proj, gate: (B,N,N,H) fp16
    mask:      (B,N,N)   fp16
    Returns proj * gate * mask (broadcast over H)
    """
    total = proj.numel()
    hidden_dim = proj.shape[-1]
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        hidden_dim,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Batched GEMM kernel for the TriMul reduction:
# out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h]
#   left:  (B,H,N,N) -> (BH,N,N)
#   right: (B,H,N,N) -> (BH,N,N) (used transposed)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch/head
    a_base = a_ptr + pid_batch * batch_stride_a
    b_base = b_ptr + pid_batch * batch_stride_b
    c_base = c_ptr + pid_batch * batch_stride_c

    # pointers to tiles
    a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        a = tl.load(a_ptrs)      # (BLOCK_M, BLOCK_K) fp16
        b = tl.load(b_ptrs)      # (BLOCK_K, BLOCK_N) fp16
        acc += tl.dot(a, b)      # fp32 accumulation
        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c = acc.to(tl.float16)
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, c, mask=store_mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the compute‑intensive path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask (if any) together with gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()     # (B,N,N)
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM (left @ rightᵀ) per head
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B,N,N,H) -> (B,H,N,N) -> (BH,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_head = B * H
    left_flat = left_perm.view(batch_head, N, N)
    right_flat = right_perm.view(batch_head, N, N)
    out_flat = torch.empty_like(left_flat)

    # strides for the GEMM (all in element units)
    a_batch_stride = left_flat.stride(0)
    a_stride_m = left_flat.stride(1)   # = N
    a_stride_k = left_flat.stride(2)   # = 1

    b_batch_stride = right_flat.stride(0)
    b_stride_n = right_flat.stride(1)  # = N (row stride)
    b_stride_k = right_flat.stride(2)  # = 1 (col stride)

    # For the transposed view of B we swap the row/col strides:
    stride_bk = b_stride_n   # K‑dimension stride in the transposed view
    stride_bn = b_stride_k   # N‑dimension stride in the transposed view

    c_batch_stride = out_flat.stride(0)
    c_stride_m = out_flat.stride(1)   # = N
    c_stride_n = out_flat.stride(2)   # = 1

    # Choose block sizes that tile the N dimension (N is a multiple of 128 in all tests)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_head,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        a_batch_stride,
        b_batch_stride,
        c_batch_stride,
        a_stride_m,
        a_stride_k,
        stride_bk,
        stride_bn,
        c_stride_m,
        c_stride_n,
        N,               # M
        N,               # N
        N,               # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
