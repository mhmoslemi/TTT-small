@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, N,
    stride_ab, stride_a_row, stride_a_col,
    stride_bb, stride_b_row, stride_b_col,
    stride_cb, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    off_m = pid_m * BLOCK_M
    off_n = pid_n * BLOCK_N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loops over K dimension
    # compute number of blocks in K
    num_blocks_k = tl.cdiv(N, BLOCK_K)

    for k in range(num_blocks_k):
        off_k = k * BLOCK_K

        # Create range indices
        row_idx = off_m + tl.arange(0, BLOCK_M)
        col_idx = off_k + tl.arange(0, BLOCK_K)
        # Compute mask for rows and cols
        mask_row = row_idx < N
        mask_col = col_idx < N

        a_ptrs = a_ptr + pid_batch * stride_ab + row_idx[:, None] * stride_a_row + col_idx[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_row[:, None] & mask_col[None, :], other=0.0)

        # For B, we need tile (K,N)
        # row index = off_k + tl.arange(0, BLOCK_K) (row because we treat K as row index)
        # col index = off_n + tl.arange(0, BLOCK_N)
        row_idx_b = off_k + tl.arange(0, BLOCK_K)
        col_idx_b = off_n + tl.arange(0, BLOCK_N)
        mask_row_b = row_idx_b < N
        mask_col_b = col_idx_b < N

        b_ptrs = b_ptr + pid_batch * stride_bb + row_idx_b[:, None] * stride_b_row + col_idx_b[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=mask_row_b[:, None] & mask_col_b[None, :], other=0.0)

        # b is shape (BLOCK_K, BLOCK_N), a is (BLOCK_M, BLOCK_K)
        # Multiply a (M x K) with b (K x N) => (M,N)
        acc += tl.dot(a, b)

    # Store result
    c_row_idx = off_m + tl.arange(0, BLOCK_M)
    c_col_idx = off_n + tl.arange(0, BLOCK_N)
    mask_c_row = c_row_idx < N
    mask_c_col = c_col_idx < N
    c_ptrs = c_ptr + pid_batch * stride_cb + c_row_idx[:, None] * stride_c_row + c_col_idx[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_c_row[:, None] & mask_c_col[None, :])

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr,
                          batch, N,
                          stride_a_batch, stride_a_row, stride_a_col,
                          stride_b_batch, stride_b_row, stride_b_col,
                          stride_c_batch, stride_c_row, stride_c_col,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # start offsets
    off_m = pid_m * BLOCK_M
    off_n = pid_n * BLOCK_N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute number of K blocks
    num_blocks_k = tl.cdiv(N, BLOCK_K)

    # loop over K blocks
    for k in range(num_blocks_k):
        off_k = k * BLOCK_K

        # row indices for A (M dimension)
        a_row = off_m + tl.arange(0, BLOCK_M)
        a_col = off_k + tl.arange(0, BLOCK_K)
        mask_a_row = a_row < N
        mask_a_col = a_col < N
        a_ptrs = a_ptr + pid_batch * stride_a_batch + a_row[:, None] * stride_a_row + a_col[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_a_row[:, None] & mask_a_col[None, :], other=0.0)

        # for B we need K x N block (since we multiply with transpose)
        b_row = off_k + tl.arange(0, BLOCK_K)  # K dimension as row index
        b_col = off_n + tl.arange(0, BLOCK_N)
        mask_b_row = b_row < N
        mask_b_col = b_col < N
        b_ptrs = b_ptr + pid_batch * stride_b_batch + b_row[:, None] * stride_b_row + b_col[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=mask_b_row[:, None] & mask_b_col[None, :], other=0.0)

        # compute dot product
        acc += tl.dot(a, b)

    # store into C
    c_row = off_m + tl.arange(0, BLOCK_M)
    c_col = off_n + tl.arange(0, BLOCK_N)
    mask_c_row = c_row < N
    mask_c_col = c_col < N
    c_ptrs = c_ptr + pid_batch * stride_c_batch + c_row[:, None] * stride_c_row + c_col[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_c_row[:, None] & mask_c_col[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # layer norm
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5
    )
    # cast to fp16
    x_h = x.to(torch.float16)

    # fused left projection & gate
    left_combined_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16).to(device)
    right_combined_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16).to(device)

    left_combined = F.linear(x_h, left_combined_w)  # [B,N,N,2*H]
    right_combined = F.linear(x_h, right_combined_w)

    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    # optional mask
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # output gate
    out_gate_weight = weights["out_gate.weight"].to(torch.float16).to(device)
    out_gate_pre = F.linear(x_h, out_gate_weight)  # [B,N,N,H]
    out_gate = torch.sigmoid(out_gate_pre)

    # core multiplication via Triton
    B, N, _, H = left.shape
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch_hidden = B * H
    # reshape to 3D tensors [B*H, N, N]
    left_flat = left_perm.view(batch_hidden, N, N)
    right_flat = right_perm.view(batch_hidden, N, N)

    # allocate output
    out_flat = torch.empty((batch_hidden, N, N), device=device, dtype=torch.float32)

    # get strides in elements
    stride_a_batch, stride_a_row, stride_a_col = left_flat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_flat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_flat.stride()

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (batch_hidden,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_hidden,
        N,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # now fp32

    # final LayerNorm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # apply output gate (convert to fp32)
    out = out * out_gate.to(torch.float32)

    # final linear projection
    out = F.linear(out, weights["to_out.weight"])

    return out
```

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy N³ reduction is implemented as a custom Triton batched matrix
multiplication kernel.  The input is layer‑normed and projected to a hidden
dimension (fp16).  After optional masking and gating the core operation

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

is performed by a tiled fp16 → fp32 matmul (left × rightᵀ) for every
(b, h) pair.  The result is brought back to fp32, layer‑normed,
gated and projected to the original dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr,                     # fp16 left  [B*H, N, N]
    b_ptr,                     # fp16 right [B*H, N, N]
    c_ptr,                     # fp32 out   [B*H, N, N]
    N,                         # sequence length
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matmul C = A @ Bᵀ for many (batch*head) matrices.

    A, B have shape (batch, N, N) in fp16.
    C has shape (batch, N, N) in fp32.
    """
    pid_batch = tl.program_id(0)      # which (batch*head) we process
    pid_m = tl.program_id(1)          # block row index
    pid_n = tl.program_id(2)          # block col index

    off_m = pid_m * BLOCK_M
    off_n = pid_n * BLOCK_N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_blocks_k = tl.cdiv(N, BLOCK_K)

    for k in range(num_blocks_k):
        off_k = k * BLOCK_K

        # ---- load A block (M × K) ------------------------------------------------
        a_row = off_m + tl.arange(0, BLOCK_M)
        a_col = off_k + tl.arange(0, BLOCK_K)

        mask_a_row = a_row < N
        mask_a_col = a_col < N
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + a_row[:, None] * stride_a_row
            + a_col[None, :] * stride_a_col
        )
        a = tl.load(
            a_ptrs,
            mask=mask_a_row[:, None] & mask_a_col[None, :],
            other=0.0,
        )  # fp16

        # ---- load Bᵀ block (K × N) -----------------------------------------------
        b_row = off_k + tl.arange(0, BLOCK_K)   # K dimension (rows of B)
        b_col = off_n + tl.arange(0, BLOCK_N)   # N dimension (cols of B)

        mask_b_row = b_row < N
        mask_b_col = b_col < N
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + b_row[:, None] * stride_b_row
            + b_col[None, :] * stride_b_col
        )
        b = tl.load(
            b_ptrs,
            mask=mask_b_row[:, None] & mask_b_col[None, :],
            other=0.0,
        )  # fp16

        # ---- accumulate ---------------------------------------------------------
        acc += tl.dot(a, b)   # fp16⋅fp16 -> fp32 accumulation

    # ---- store result ---------------------------------------------------------
    c_row = off_m + tl.arange(0, BLOCK_M)
    c_col = off_n + tl.arange(0, BLOCK_N)

    mask_c_row = c_row < N
    mask_c_col = c_col < N
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + c_row[:, None] * stride_c_row
        + c_col[None, :] * stride_c_col
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_c_row[:, None] & mask_c_col[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim], float32
            - mask: torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights: dict with the model parameters
            - config: dict containing at least "dim", "hidden_dim", and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim], float32
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------
    # 2) Cast to fp16 for the expensive part
    # ------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------
    # 3) Fuse projection + gate linear layers (fp16)
    # ------------------------------------------------------------
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16).to(device)
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16).to(device)

    left_combined = F.linear(x_h, left_combined_w)   # [B, N, N, 2*H] fp16
    right_combined = F.linear(x_h, right_combined_w)  # same shape

    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    # ------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------
    # 5) Gating (sigmoid, fp16)
    # ------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate      # [B, N, N, H] fp16
    right = right_proj * right_gate   # [B, N, N, H] fp16

    # ------------------------------------------------------------
    # 6) Output gate (fp16)
    # ------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16).to(device)
    out_gate_pre = F.linear(x_h, out_gate_w)           # [B, N, N, H] fp16
    out_gate = torch.sigmoid(out_gate_pre)             # fp16

    # ------------------------------------------------------------
    # 7) Core N³ reduction via Triton (batched matmul)
    #    out = left @ rightᵀ   per (batch, head)
    # ------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, _, hidden_dim

    # reshape to (B*H, N, N) for the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch_hidden = B * H

    left_flat = left_perm.view(batch_hidden, N, N)      # (B*H, N, N) fp16
    right_flat = right_perm.view(batch_hidden, N, N)    # same dtype

    out_flat = torch.empty(
        (batch_hidden, N, N), device=device, dtype=torch.float32
    )  # (B*H, N, N) fp32

    # strides in elements
    stride_a_b, stride_a_r, stride_a_c = left_flat.stride()
    stride_b_b, stride_b_r, stride_b_c = right_flat.stride()
    stride_c_b, stride_c_r, stride_c_c = out_flat.stride()

    # launch parameters (tuned for Ampere B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        batch_hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_a_b,
        stride_a_r,
        stride_a_c,
        stride_b_b,
        stride_b_r,
        stride_b_c,
        stride_c_b,
        stride_c_r,
        stride_c_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B, N, N, H] (float32)
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H] fp32

    # ------------------------------------------------------------
    # 8) Final LayerNorm, output gating and projection
    # ------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.to(torch.float32)                     # apply out gate
    out = F.linear(out, weights["to_out.weight"])             # back to dim

    return out
