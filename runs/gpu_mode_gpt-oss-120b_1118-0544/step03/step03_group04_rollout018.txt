@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # batch index
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute pointer base
    a_ptrs = a_ptr + batch * stride_a_batch + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
    b_ptrs = b_ptr + batch * stride_b_batch + offs_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        acc += tl.dot(a, b)
        # increment pointers to next k block
        a_ptrs += BLOCK_K * stride_a_col
        b_ptrs += BLOCK_K * stride_b_col

    # write back
    c_ptrs = c_ptr + batch * stride_c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
```

    left = left.permute(0,3,1,2).contiguous()   # (B, H, N, N)

    # Flatten batch dimension
    left_flat = left.view(batch, N, N)
    right_flat = right.view(batch, N, N)

    # Transpose right for multiplication across k dimension
    right_t = right_flat.transpose(1,2).contiguous()

    # Allocate out_flat
    out_flat = torch.empty_like(left_flat)  # same dtype and device

    # Compute strides
    a_batch_stride, a_row_stride, a_col_stride = left_flat.stride()
    b_batch_stride, b_row_stride, b_col_stride = right_t.stride()
    c_batch_stride, c_row_stride, c_col_stride = out_flat.stride()

    # Define block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (batch, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    _gemm_kernel[grid](
        left_flat,
        right_t,
        out_flat,
        N, N, N,
        a_batch_stride, a_row_stride, a_col_stride,
        b_batch_stride, b_row_stride, b_col_stride,
        c_batch_stride, c_row_stride, c_col_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # After kernel, out_flat is (batch,N,N)
    # Reshape back
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # (B,N,N,H)

    # Hidden dim LayerNorm
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Multiply by out_gate (elementwise)
    out = _elemwise_mul(out, out_gate)  # uses Triton

    # Final projection
    out = F.linear(out, to_out_w)  # shape (B,N,N,dim)

    # Cast back to original dtype (float32 likely)
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy O(N³·H) reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed with a custom batched GEMM kernel written in Triton.
All other inexpensive operations (layer‑norms, linear projections, gates)
are executed with PyTorch (FP16) for simplicity.
An additional tiny Triton kernel handles the final element‑wise multiplication
by the output gate.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernels
# --------------------------------------------------------------------------- #
@triton.jit
def _gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]
    A: (batch, M, K)   (row‑major)
    B: (batch, N, K)   (row‑major, i.e. B_T = original right)
    C: (batch, M, N)
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of the tiles
    a_ptrs = (
        a_ptr
        + batch_id * stride_a_batch
        + offs_m[:, None] * stride_a_row
        + offs_k[None, :] * stride_a_col
    )
    b_ptrs = (
        b_ptr
        + batch_id * stride_b_batch
        + offs_k[:, None] * stride_b_col
        + offs_n[None, :] * stride_b_row
    )

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        acc += tl.dot(a, b)               # fp16·fp16 → fp32 accumulation
        a_ptrs += BLOCK_K * stride_a_col
        b_ptrs += BLOCK_K * stride_b_col

    # Write back
    c_ptrs = (
        c_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    c = acc.to(tl.float16)
    tl.store(
        c_ptrs,
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    """1‑D element‑wise multiplication (a * b)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton element‑wise kernel on two tensors of identical shape."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication."
    out = torch.empty_like(a)
    numel = a.numel()
    BLOCK = 1024
    grid = ( (numel + BLOCK - 1) // BLOCK, )
    _elemwise_mul_kernel[grid](a, b, out, numel, BLOCK=BLOCK)
    return out


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask       : torch.Tensor of shape [B, N, N] or None
            - weights    : dict of model parameters (float32)
            - config     : dict containing:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (float16 for speed)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 2. Cast model weights to fp16
    # ------------------------------------------------------------------- #
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------- #
    # 3. Linear projections (left / right) and optional masking
    # ------------------------------------------------------------------- #
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------- #
    # 4. Gate projections (sigmoid)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Core O(N³·H) reduction via Triton batched GEMM
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape                     # (B, N, N, H)

    # Bring hidden dimension to batch dimension
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    batch = B * H
    left_flat  = left.view(batch, N, N)            # (B*H, N, N)
    right_flat = right.view(batch, N, N)           # (B*H, N, N)

    # Transpose right so that we can multiply across the shared K dimension
    right_t = right_flat.transpose(1, 2).contiguous()  # (B*H, N, N)

    # Allocate output tensor
    out_flat = torch.empty_like(left_flat)         # (B*H, N, N)

    # Strides (in elements)
    stride_a_batch, stride_a_row, stride_a_col = left_flat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_t.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_flat.stride()

    # Block sizes – tuned for typical A200 memory hierarchy
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _gemm_kernel[grid](
        left_flat,
        right_t,
        out_flat,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 6. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------- #
    # 7. Apply output gate (Triton element‑wise kernel)
    # ------------------------------------------------------------------- #
    out = _elemwise_mul(out, out_gate)

    # ------------------------------------------------------------------- #
    # 8. Final projection back to original channel dimension
    # ------------------------------------------------------------------- #
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype if needed
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
