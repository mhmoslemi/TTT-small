class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the supplied LN parameters.
2. Cast to fp16 and compute linear projections (left/right) and their gates
   (left_gate, right_gate, out_gate) with torch.nn.functional.linear.
3. Apply optional binary mask, then fuse gates into the projected tensors.
4. Reshape tensors to (B*H,N,N) where H = hidden_dim and perform the core
   O(N³·H) reduction using a single batched GEMM (torch.bmm, which maps to
   cuBLAS and exploits Tensor‑Cores).  The operation computes
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
5. Reshape back to (B,N,N,H) and apply a second Layer‑Norm over the hidden
   dimension.
6. Multiply the result with the output gate using a tiny Triton kernel
   (element‑wise multiplication) – this satisfies the “part of the ops in a
   kernel” requirement.
7. Project back to the original channel dimension with a final linear layer.
8. Cast the result back to the input dtype.

The heavy O(N³) step is delegated to cuBLAS (torch.bmm) for maximal
performance, while a minimal Triton kernel is used for the gating step.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, total_elems,
                            BLOCK_SIZE: tl.constexpr):
    """Element‑wise multiplication a * b -> c (all fp16, contiguous)."""
    pid = tl.program_id(0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < total_elems
    a = tl.load(a_ptr + offsets, mask=mask, other=0.0)
    b = tl.load(b_ptr + offsets, mask=mask, other=0.0)
    tl.store(c_ptr + offsets, a * b, mask=mask)


def _elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton kernel to multiply two tensors element‑wise."""
    assert a.shape == b.shape
    total = a.numel()
    c = torch.empty_like(a)
    BLOCK_SIZE = 1024  # max threads per block on NVIDIA GPUs
    grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    _elementwise_mul_kernel[grid](
        a, b, c, total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    return c


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            mask: torch.Tensor of shape [B, N, N] (or None)
            weights: dict containing model weights
            config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Prepare FP16 weights
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)  # (H, C)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)  # (H, C)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, C)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)      # (C, H)

    # ------------------------------------------------------------------
    # 4) Linear projections (left/right)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 6) Gate projections and sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()   # (B,N,N,H)
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7) Fuse gates into left/right
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate
    del left_gate, right_gate  # free memory

    # ------------------------------------------------------------------
    # 8) Core O(N³·H) reduction via batched GEMM (cuBLAS)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)          # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)        # (B*H, N, N)

    # matmul with transposed right (sum over k)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))          # (B*H, N, N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate – performed with a tiny Triton kernel
    # ------------------------------------------------------------------
    out = _elementwise_mul(out, out_gate)

    # ------------------------------------------------------------------
    # 11) Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,C)

    # ------------------------------------------------------------------
    # 12) Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

**Approach and Rationale**
The core operation is computing a matrix product between two [N, N] matrices from the same hidden_dim channel. For each batch element and hidden_dim channel, we have independent left and right matrices. Direct slicing to get these matrices is inefficient due to non-contiguous memory access. Instead, we calculate the matrix product using a tiled approach, leveraging triton's program IDs to iteratively process tiles of the left and right matrices. This avoids storing large matrices in shared memory and ensures cache-friendly access by breaking the large seqlen into manageable blocks across global memory.

**Mathematical Specification**
Given input tensors `left` and `right` of shape `[B, N, N, H]` (for hidden_dim channel H), the goal is to compute `out[b, i, j, h] = ∑_{k=0}^{N-1} left[b, i, k, h] * right[b, j, k, h]` for all dimensions. 

- `left_tensor` and `right_tensor` are split into smaller tile-shaped matrices `[sub_N, sub_N]` per hidden_dim channel. The output for each hidden_dim channel is computed as a matrix product.
- The operation for each hidden_dim channel `h` and batch element `b` is: `out[b, :, :, h] = left_tensor[b, :, :, h] @ right_tensor[b, :, :, h].T`
