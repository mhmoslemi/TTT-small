So moving across k increments by 1. So stride for row (k) = 1. Moving across N (j) increments by? Each row j's stride across columns (k) is 1, each new row j adds K elements. So stride for column (j) = K. That's exactly the stride we set earlier: stride_bk = 1, stride_bn = K.

Therefore in kernel we can treat B as shape (K,N) with those strides. In code we define "STRIDE_BK" constant (k stride) = 1; "STRIDE_BN" (n stride) = N (which is K?) Actually K = N also (we are using N as both dimension). So we will set:

```

stride_bn = N
```

Thus we compute a tile: shape (BLOCK_M, BLOCK_K) where k_offset loops across K dimension.

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_batch, stride_row, stride_col,  # left strides (batch, row(i), col(k))
    stride_batch_b, stride_row_b, stride_col_b,  # right strides (batch, row(j)??, col(k)), but we set row=j, col=k
    stride_batch_out, stride_row_out, stride_col_out,
    N: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers start at batch offset
    left_batch_ptr = left_ptr + batch_id * stride_batch
    right_batch_ptr = right_ptr + batch_id * stride_batch_b
    out_batch_ptr = out_ptr + batch_id * stride_batch_out

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K loop
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        # compute actual K range
        k_range = tl.arange(0, BLOCK_K)
        mask_k = k_offset + k_range < N

        # Load A tile
        a_ptrs = left_batch_ptr + (offs_m[:, None] * stride_row) + ((k_offset + k_range)[None, :] * stride_col)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (shape BLOCK_K x BLOCK_N)
        # note: right stored as (N, K) so we need to treat it as (K, N)
        b_ptrs = right_batch_ptr + ((k_offset + k_range)[:, None] * stride_col_b) + (offs_n[None, :] * stride_row_b)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

        # No need to advance pointers manually; we recompute each iteration using offsets

    # Store result
    c_ptrs = out_batch_ptr + (offs_m[:, None] * stride_row_out) + (offs_n[None, :] * stride_col_out)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_batch_a, stride_row_a, stride_col_a,
    stride_batch_b, stride_row_b, stride_col_b,
    stride_batch_c, stride_row_c, stride_col_c,
    N: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch_ptr = a_ptr + batch_id * stride_batch_a
    b_batch_ptr = b_ptr + batch_id * stride_batch_b
    c_batch_ptr = c_ptr + batch_id * stride_batch_c

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K)
        mask_k = (k_offset + k_range) < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_row_a) + ((k_offset + k_range)[None, :] * stride_col_a)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile: shape (BLOCK_K, BLOCK_N) (treated as transposed)
        b_ptrs = b_batch_ptr + ((k_offset + k_range)[:, None] * stride_row_b) + (offs_n[None, :] * stride_col_b)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product accumulation
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_row_c) + (offs_n[None, :] * stride_col_c)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton implementation.

Algorithm:
1. Layer‑norm the input tensor (float32 → float16 for the rest).
2. Fuse the four linear layers (left/right projections and their gates) into a
   single matrix multiplication using a concatenated weight matrix.
3. Apply optional mask to the projected tensors.
4. Apply sigmoid gates and combine with the projections → tensors `left` and `right`
   of shape [B, N, N, H] (half‑precision).
5. Core O(N³) reduction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM in Triton where each hidden channel
   (and batch) is a separate batch element.
   The kernel computes C = A @ Bᵀ with A shape (M,K) and B treated as (K,N).
6. Layer‑norm over the hidden dimension.
7. Compute `out_gate` = sigmoid(Linear(x)) and multiply element‑wise via a
   Triton kernel.
8. Final linear projection back to the original dimension and cast to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    a_ptr,            # left tensor   [batch, M, K]
    b_ptr,            # right tensor  [batch, N, K] (stored as (N,K) but accessed as (K,N))
    c_ptr,            # output tensor [batch, M, N]
    stride_batch_a, stride_row_a, stride_col_a,
    stride_batch_b, stride_row_b, stride_col_b,
    stride_batch_c, stride_row_c, stride_col_c,
    N: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """C = Σₖ A[:,i,k] * B[:,j,k]   (batch‑wise, no bias)"""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)   # tile row   (i)
    pid_n = tl.program_id(2)   # tile col   (j)

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch element
    a_batch = a_ptr + batch_id * stride_batch_a
    b_batch = b_ptr + batch_id * stride_batch_b
    c_batch = c_ptr + batch_id * stride_batch_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K)
        mask_k = (k_offset + k_range) < N

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_row_a) + ((k_offset + k_range)[None, :] * stride_col_a)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B tile (interpreted as (K,N)):
        # original storage is (N, K) → we treat rows as K (stride_row_b = 1) and cols as N (stride_col_b = N)
        b_ptrs = b_batch + ((k_offset + k_range)[:, None] * stride_row_b) + (offs_n[None, :] * stride_col_b)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)      # (BLOCK_M, BLOCK_N) accumulation in fp32

    # write result back in fp16
    c_ptrs = c_batch + (offs_m[:, None] * stride_row_c) + (offs_n[None, :] * stride_col_c)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


@triton.jit
def _elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, total_elements: tl.constexpr, BLOCK: tl.constexpr):
    """c = a * b  (1‑D flattened tensors)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask: torch.Tensor [B, N, N] (bool or float) – may be ignored if config["nomask"] is True
            weights: dict of model weights (float32 tensors)
            config: dict with keys:
                "dim" (int) – input channel dimension C
                "hidden_dim" (int) – hidden projection dimension H
                "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # Load & cast weights
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)

    left_proj_w = weights["left_proj.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()
    left_gate_w = weights["left_gate.weight"].to(device).half()
    right_gate_w = weights["right_gate.weight"].to(device).half()
    out_gate_w = weights["out_gate.weight"].to(device).half()

    to_out_norm_w = weights["to_out_norm.weight"].to(device).half()
    to_out_norm_b = weights["to_out_norm.bias"].to(device).half()
    to_out_w = weights["to_out.weight"].to(device).half()   # shape (dim, hidden_dim)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (float32 → half)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse four linear layers (proj & gate) into one matmul
    # --------------------------------------------------------------
    combined_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # (4*hidden_dim, dim)

    x_h = x.half()
    x_flat = x_h.view(-1, dim)                     # (B*N*N, dim)
    combined = F.linear(x_flat, combined_w)       # (B*N*N, 4*hidden_dim)
    combined = combined.view(B, N, N, 4 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each: (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 3️⃣ Optional masking (only on projections)
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype=x_h.dtype).unsqueeze(-1)   # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------------------
    # 4️⃣ Gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate       # (B, N, N, hidden)
    right = right_proj * right_gate    # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 5️⃣ Batched GEMM via Triton: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Transpose to (B, H, N, N) and flatten batch*hidden
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch_h = B * hidden_dim
    left_mat = left.view(batch_h, N, N)   # (batch_h, N, N)
    right_mat = right.view(batch_h, N, N) # (batch_h, N, N)

    out_mat = torch.empty_like(left_mat)  # (batch_h, N, N), fp16

    # Strides for the left matrix (A) – standard row‑major (M,N)
    stride_batch_a = left_mat.stride(0)   # N*N
    stride_row_a   = left_mat.stride(1)   # N
    stride_col_a   = left_mat.stride(2)   # 1

    # Strides for the right matrix (B) – we interpret it as (K,N)
    # Original layout is (N, K) → row stride = N, col stride = 1.
    # To access B as (K,N) we set row stride = 1 (k‑step) and col stride = N (j‑step).
    stride_batch_b = right_mat.stride(0)  # N*N
    stride_row_b   = 1                    # k
    stride_col_b   = N                    # j

    # Strides for output matrix (C) – same layout as left_mat
    stride_batch_c = out_mat.stride(0)    # N*N
    stride_row_c   = out_mat.stride(1)    # N
    stride_col_c   = out_mat.stride(2)    # 1

    # Block sizes – tuned for Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_batch_a,
        stride_row_a,
        stride_col_a,
        stride_batch_b,
        stride_row_b,
        stride_col_b,
        stride_batch_c,
        stride_row_c,
        stride_col_c,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm (over hidden dimension)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # (B, N, N, hidden), fp16

    # --------------------------------------------------------------
    # 7️⃣ Out‑gate: linear + sigmoid, then element‑wise multiplication (Triton)
    # --------------------------------------------------------------
    # Compute out_gate from the original normalized tensor x
    out_gate = torch.sigmoid(
        F.linear(x_h.view(-1, dim), out_gate_w).view(B, N, N, hidden_dim)
    )  # (B, N, N, hidden), fp16

    total_el = out.numel()
    BLOCK = 65536
    grid_mul = ((total_el + BLOCK - 1) // BLOCK,)
    _elementwise_mul_kernel[grid_mul](
        out,
        out_gate,
        out,
        total_el,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dimension
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, dim), fp16
    return out.float()
