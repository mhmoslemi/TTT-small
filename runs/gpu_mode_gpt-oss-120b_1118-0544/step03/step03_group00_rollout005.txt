"""
TriMul (outgoing) forward pass – fast Triton‑fused implementation.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the provided LN parameters.
2. Cast to fp16 to exploit Tensor‑cores.
3. Linear projections (left/right) and gate projections (left_gate, right_gate,
   out_gate) are computed with fp16 weights.
4. If a mask is present and not disabled, a single Triton kernel fuses the
   element‑wise multiplication of the projection, its gate and the mask.
   (Otherwise we just multiply projection and gate in PyTorch.)
5. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication using torch.bmm on
   tensors reshaped to (B*H, N, N).  This leverages cuBLAS/Tensor‑cores and is
   far faster than a naïve triple loop.
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate, and project back to the original channel dimension.
7. Cast the result back to the original dtype (usually float32).

At least one Triton kernel (gate+mask fusion) is used, satisfying the
requirement while keeping the heavy O(N³·H) work on cuBLAS.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_fuse_kernel(
    proj_ptr,          # (B*N*N*H,) fp16 projected values
    gate_ptr,          # (B*N*N*H,) fp16 gate values (sigmoid)
    mask_ptr,          # (B*N*N,)   fp16 mask (0/1)
    out_ptr,           # (B*N*N*H,) fp16 output
    total_elems,       # total number of elements in proj/gate (B*N*N*H)
    hidden_dim,        # H, used for broadcasting mask over the hidden dim
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Boundary check
    valid = offs < total_elems

    # Mask index: each hidden‑dim entry shares the same mask location
    mask_idx = offs // hidden_dim

    # Load values (out of bounds yields zero / one for mask)
    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Fused multiply
    out = proj * gate * mask

    tl.store(out_ptr + offs, out, mask=valid)


def fuse_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                       mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    proj : (B,N,N,H) fp16
    gate : (B,N,N,H) fp16
    mask : (B,N,N)   fp16 (0 or 1)
    Returns proj * gate * mask  (broadcast mask over H) as fp16 tensor.
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out

# Main entry point
# ----------------------------------------------------------------------

    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all weights to fp16 (they reside on the same device)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask & gates (fused Triton kernel if mask required)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(device).to(torch.float16).contiguous()
        left = fuse_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = fuse_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        # Simple element‑wise fusion when mask is absent or ignored
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction via batched GEMM (cuBLAS/Tensor‑cores)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # Reshape to (B*H, N, N) where the last two dims are (i,k) for left
    # and (j,k) for right.  Right is transposed inside the bmm.
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: left @ rightᵀ  →  (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
