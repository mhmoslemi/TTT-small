def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # input dtype
    dtype = input_tensor.dtype
    device = input_tensor.device
    
    # 1. Layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # cast to fp16
    x = x.to(torch.float16)
    
    # 2. Load weights as fp16
    # We may need to reshape left_proj weight shape: (hidden_dim, dim)
    # torch.nn.functional.linear expects weight shape (out_features, in_features). already.
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    
    # 3. Compute projections
    # shape: (B,N,N,dim) -> (B*N*N, dim) for linear
    # use torch.nn.functional.linear directly on x (shape B,N,N,dim)
    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_w)
    
    # 4. Apply mask if needed
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f
    
    # 5. Gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()
    
    # 6. Fuse gates
    left = left * left_gate
    right = right * right_gate
    
    # free some mem
    del left_gate, right_gate
    
    # 7. Permute for Triton
    left_h = left.permute(0,3,1,2).contiguous()
    right_h = right.permute(0,3,1,2).contiguous()
    
    B, H, N, _ = left_h.shape   # shape (B, H, N, N)
    # Ensure contiguous for right as well
    # 8. Compute output via Triton kernel
    out_h = trimul_outgoing_kernel_launcher(left_h, right_h, B, N, H)
    
    # out_h shape (B, H, N, N)
    out = out_h.permute(0,2,3,1).contiguous()
    
    # 9. Output layer_norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    
    # 10. Multiply by out_gate
    out = out * out_gate
    
    # 11. Final linear projection
    out = torch.nn.functional.linear(out, to_out_w)
    
    # 12. Cast back to original dtype
    if out.dtype != dtype:
        out = out.to(dtype)
    return out

def trimul_outgoing_kernel_launcher(left, right, B, N, H):
    # left,right: torch.Tensor shape (B, H, N, N), dtype torch.float16, contiguous.
    # Output shape (B, H, N, N)
    # Compute strides
    stride_b, stride_h, stride_i, stride_j = left.stride()
    # For left, stride_i = stride for the N dimension after permute (i dimension)
    # stride_k also same as stride_j because last two dims are N,N.
    # Actually after permute, shape is (B, H, N, N), where the third dim is i, fourth is k.
    # So stride_i = stride for dimension 2 (index i), stride_k = stride for dimension 3 (index k).
    stride_left_b, stride_left_h, stride_left_i, stride_left_k = left.stride()
    stride_right_b, stride_right_h, stride_right_i, stride_right_k = right.stride()
    # Output strides similar to left.
    out = torch.empty_like(left)   # same shape and dtype
    stride_out_b, stride_out_h, stride_out_i, stride_out_j = out.stride()
    
    # Determine block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64  # tile size for k dimension
    
    grid = (B * H, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    
    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left_b,
        stride_left_h=stride_left_h,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_right_b=stride_right_b,
        stride_right_h=stride_right_h,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_out_b=stride_out_b,
        stride_out_h=stride_out_h,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑Norm the input tensor.
2. FP16 linear projections for left/right and their gates (left_gate, right_gate,
   out_gate) using the provided weights.
3. Apply the optional pairwise mask to left/right.
4. Fuse gates into left/right (element‑wise multiplication).
5. Permute tensors to shape (B, H, N, N) and launch a Triton kernel that
   computes, for every batch b and head h:
        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
   (i.e. a batched matrix multiplication with the second operand transposed).
   The kernel tiles over the i/j dimensions (BLOCK_M × BLOCK_N) and iterates
   over k in tiles (BLOCK_K) using fp16 loads and a float32 accumulator.
6. Permute the result back to (B, N, N, H) and apply a second Layer‑Norm.
7. Multiply by the output gate.
8. Final linear projection back to the original channel dimension.
9. Cast the result to the input dtype and return.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,        # pointers
    B, N, H,                             # dimensions (runtime ints)
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = left @ right^T for each (batch, head) pair."""
    pid_batch_head = tl.program_id(0)   # combined batch * head index
    pid_m = tl.program_id(1)           # i‑tile index
    pid_n = tl.program_id(2)           # j‑tile index

    # --------------------------------------------------------------
    # Decompose combined index into batch and head
    # --------------------------------------------------------------
    b = pid_batch_head // H
    h = pid_batch_head % H

    # --------------------------------------------------------------
    # Offsets for the tile
    # --------------------------------------------------------------
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    i_mask = i_offset < N
    j_mask = j_offset < N

    # Base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over k dimension in tiles
    # --------------------------------------------------------------
    num_tiles_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_tiles_k):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base \
                 + i_offset[:, None] * stride_left_i \
                 + k_offset[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=i_mask[:, None] & k_mask[None, :],
                    other=0.0)

        # Load right tile transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base \
                 + k_offset[:, None] * stride_right_k \
                 + j_offset[None, :] * stride_right_i   # swapped strides
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & j_mask[None, :],
                    other=0.0)

        # Accumulate product (fp16 -> fp32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back the result (cast to fp16)
    # --------------------------------------------------------------
    out = acc.to(tl.float16)
    out_ptrs = out_base \
               + i_offset[:, None] * stride_out_i \
               + j_offset[None, :] * stride_out_j
    tl.store(out_ptrs, out, mask=i_mask[:, None] & j_mask[None, :])


def _tri_mul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton driver for the outgoing TriMul contraction.
    Expects left/right tensors of shape (B, H, N, N) and dtype float16.
    Returns a tensor of the same shape and dtype.
    """
    assert left.shape == right.shape
    B, H, N, _ = left.shape

    # Strides (in elements)
    stride_left_b, stride_left_h, stride_left_i, stride_left_k = left.stride()
    stride_right_b, stride_right_h, stride_right_i, stride_right_k = right.stride()

    out = torch.empty_like(left)  # (B, H, N, N), fp16
    stride_out_b, stride_out_h, stride_out_i, stride_out_j = out.stride()

    # Tunable block sizes – 64 works well for typical seq lengths up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left_b,
        stride_left_h=stride_left_h,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_right_b=stride_right_b,
        stride_right_h=stride_right_h,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_out_b=stride_out_b,
        stride_out_h=stride_out_h,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            mask: torch.Tensor of shape [B, N, N] (or None)
            weights: dict containing model weights
            config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Prepare FP16 weights
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)    # (hidden_dim, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)    # (hidden_dim, dim)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)   # (hidden_dim, dim)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)     # (hidden_dim, dim)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16) # (hidden_dim,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)   # (hidden_dim,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)      # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 4) Linear projections (left/right)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_w)  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 6) Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()   # (B, N, N, hidden_dim)
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()    # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7) Fuse gates into left/right
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate
    del left_gate, right_gate  # free intermediate tensors

    # ------------------------------------------------------------------
    # 8) Core O(N³·H) reduction via Triton kernel
    #    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # ------------------------------------------------------------------
    # Bring the hidden dimension to the second axis for easy tiling.
    left_h  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_h = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    out_h = _tri_mul_outgoing_triton(left_h, right_h)   # (B, H, N, N)

    # Back to original layout (B, N, N, H)
    out = out_h.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 12) Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
