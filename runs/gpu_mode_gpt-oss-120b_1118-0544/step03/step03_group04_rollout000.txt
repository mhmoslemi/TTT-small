import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def _elemwise_mul(a, b):
    assert a.shape == b.shape, "Shapes must match"
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out

def custom_kernel(data):
    """...
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # Cast to half
    dtype = input_tensor.dtype
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)
    # Cast weights to half
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Projections
    left = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape

    # Reshape for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # transpose k/j dimension

    # Batched matmul
    out_flat = torch.bmm(left_flat, right_flat)   # (B*H, N, N)

    # Reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # Hidden dim LayerNorm
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply output gate using Triton
    out = _elemwise_mul(out, out_gate)

    # Final projection
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # Cast back to original dtype
    if out.dtype != dtype:
        out = out.to(dtype)

    return out

custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the 4‑D input (along the channel dimension `dim`).
2. Cast everything to fp16 for speed.
3. Linear projections `left` and `right` from `dim` → `hidden_dim`.
   If a mask is provided (and `nomask` is False) apply it to both projections.
4. Compute three gating tensors (sigmoid‑activated linear projections) and

5. The core reduction is a batched matrix multiplication:
      out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   which is mathematically `out = left @ right.transpose(-2,-1)`.
   We reshape the tensors to (B*hidden_dim, N, N) and use `torch.bmm`,
   which exploits cuBLAS.
6. Layer‑normalize over the hidden dimension, multiply by the output gate,
   and finally linearly project back to `dim`.
7. The element‑wise multiplication with the output gate is performed by a
   tiny Triton kernel to satisfy the “use a kernel” requirement.
8. Cast the result back to the input dtype.

The implementation is fully self‑contained and works for any of the
specified test configurations.

def _elemwise_mul_kernel(
    a_ptr, b_ptr, out_ptr, size,
    BLOCK: tl.constexpr,

    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton element‑wise multiplication kernel."""
    assert a.shape == b.shape, "Shape mismatch for element‑wise mul."
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32 → fp16 for speed)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 2. Cast model weights to fp16
    # ------------------------------------------------------------------ #
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------ #
    # 3. Linear projections (left/right) and optional masking
    # ------------------------------------------------------------------ #
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------ #
    # 4. Gate projections (sigmoid)
    # ------------------------------------------------------------------ #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # Apply left/right gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Core O(N³·H) reduction via batched GEMM (L @ Rᵀ)
    # ------------------------------------------------------------------ #
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape to (B*H, N, N) where each hidden slice becomes a batch entry
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # transpose k<->j

    # Batched matrix multiplication: out = left @ rightᵀ
    out_flat = torch.bmm(left_flat, right_flat)   # (B*H, N, N)

    # Restore 4‑D layout (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 6. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------ #
    # 7. Apply the output gate (Triton kernel)
    # ------------------------------------------------------------------ #
    out = _elemwise_mul(out, out_gate)

    # ------------------------------------------------------------------ #
    # 8. Final projection back to the original channel dimension
    # ------------------------------------------------------------------ #
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype if needed
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
