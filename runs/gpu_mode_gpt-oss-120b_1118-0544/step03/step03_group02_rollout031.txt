import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, total_n, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    off = pid * BLOCK + tl.arange(0, BLOCK)
    mask = off < total_n
    a = tl.load(a_ptr + off, mask=mask)
    b = tl.load(b_ptr + off, mask=mask)
    tl.store(c_ptr + off, a * b, mask=mask)

def custom_kernel(data):
    """
    Fast forward implementation of the outgoing TriMul operator.
    The heavy linear projections and the N³ reduction are performed with
    highly‑optimized PyTorch GEMM kernels; a tiny Triton kernel handles the
    final element‑wise gating multiplication. All heavy arithmetic is done
    in float16 to exploit tensor cores; the result is returned as float32.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # Cast weights to appropriate dtype (fp16) on target device
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias = weights["norm.bias"].to(device).to(torch.float16)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # 1) LayerNorm on input (fp16)
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] fp16

    B, N, _, _ = x.shape

    # 2) Combined linear projection for left/right and their gates
    # Build a single weight matrix of shape (4*hidden_dim, dim)
    combined_weight = torch.cat(
        [left_proj_weight, left_gate_weight, right_proj_weight, right_gate_weight],
        dim=0,
    )  # (4*hidden, dim)

    x_flat = x.view(-1, dim)  # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 4*hidden)
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each: (B, N, N, hidden)

    # 3) Optional masking (applied before gating)
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # 4) Gating
    # sigmoid
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # Apply gates element‑wise (in‑place to save memory)
    left = left_proj
    left.mul_(left_gate)   # now left holds the gated projection
    right = right_proj
    right.mul_(right_gate)

    # 5) N³ reduction via batched GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # Transform to (B*hidden, N, N) matrices
    left = left.transpose(1, 3).contiguous()   # (B, hidden, N, N)
    right = right.transpose(1, 3).contiguous() # (B, hidden, N, N)

    left_mat = left.view(B * hidden_dim, N, N)      # (B*hidden, N, N)
    right_mat = right.view(B * hidden_dim, N, N)    # (B*hidden, N, N)

    # Batched matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    # Reshape back to (B, N, N, hidden) fp16
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 6) to_out_norm (LayerNorm over hidden dim)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B,N,N,hidden) fp16

    # 7) out_gate computation (linear + sigmoid)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # (B,N,N,hidden) fp16

    # 8) Apply out_gate via a tiny Triton kernel (element‑wise multiplication)
    total_el = out.numel()
    BLOCK = 65536  # tune for the GPU
    grid = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        out,
        out_gate,
        out,
        total_el,
        BLOCK=BLOCK,
    )

    # 9) Final projection back to original dimension
    out = F.linear(out, to_out_weight)  # (B,N,N,dim) fp16
    out = out.to(torch.float32)
    return out

    """c = a * b  (1‑D, fp16)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_n
    a = tl.load(a_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float)
            - weights: dict of model weight tensors (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # --------------------------------------------------------------
    # Cast weights to fp16 on the target device
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias = weights["norm.bias"].to(device).to(torch.float16)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (fp16)
    # --------------------------------------------------------------
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  fp16

    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 2️⃣ Combined linear projection for left/right and their gates
    # --------------------------------------------------------------
    combined_weight = torch.cat(
        [left_proj_weight, left_gate_weight, right_proj_weight, right_gate_weight],
        dim=0,
    )  # (4*hidden_dim, dim)

    x_flat = x.view(-1, dim)                     # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 4*hidden_dim)
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each: (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 3️⃣ Optional masking (applied before gating)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(x.dtype).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and combine with projections (in‑place)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj
    left.mul_(left_gate)        # left = left_proj * left_gate
    right = right_proj
    right.mul_(right_gate)      # right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5️⃣ N³ reduction via batched GEMM:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # reshape to (B*hidden, N, N) matrices
    left = left.transpose(1, 3).contiguous()   # (B, hidden, N, N)
    right = right.transpose(1, 3).contiguous() # (B, hidden, N, N)

    left_mat = left.view(B * hidden_dim, N, N)   # (B*hidden, N, N)
    right_mat = right.view(B * hidden_dim, N, N) # (B*hidden, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # out : (B, N, N, hidden_dim)  fp16

    # --------------------------------------------------------------
    # 6️⃣ to_out_norm (LayerNorm over hidden dimension)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # fp16

    # --------------------------------------------------------------
    # 7️⃣ out_gate linear + sigmoid
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # (B, N, N, hidden) fp16

    # --------------------------------------------------------------
    # 8️⃣ Apply out_gate via a tiny Triton kernel (element‑wise mul)
    # --------------------------------------------------------------
    total_el = out.numel()
    BLOCK = 65536
    grid = ((total_el + BLOCK - 1) // BLOCK,)
    _elementwise_mul_kernel[grid](
        out,
        out_gate,
        out,
        total_el,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 9️⃣ Final projection back to original dimension
    # --------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # (B, N, N, dim)  fp16
    out = out.to(torch.float32)         # return float32 as required
    return out
