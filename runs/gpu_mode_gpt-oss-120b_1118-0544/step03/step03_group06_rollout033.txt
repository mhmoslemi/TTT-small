"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the reference PyTorch TriMul but:
1. Uses fp16 for the compute‑heavy path.
2. Fuses mask application and gating with a lightweight Triton elementwise kernel.
3. Performs the O(N³·H) reduction with a batched torch.bmm (cuBLAS) which is
   highly tuned on NVIDIA GPUs.
4. All remaining linear / LayerNorm operations use torch.nn.functional.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels for elementwise fusion (mask * gate * tensor)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mul_kernel(
    a_ptr,          # tensor to be gated (flattened)
    b_ptr,          # gate tensor (flattened)
    out_ptr,        # output (can be same as a_ptr for in‑place)
    size,           # total number of elements
    stride_a: tl.constexpr,
    stride_b: tl.constexpr,
    stride_out: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs * stride_a, mask=mask)
    b = tl.load(b_ptr + offs * stride_b, mask=mask)
    out = a * b
    tl.store(out_ptr + offs * stride_out, out, mask=mask)

    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs * stride_a, mask=mask)
    b = tl.load(b_ptr + offs * stride_b, mask=mask)

    # each H consecutive entries share the same mask index
    mask_index = offs // H
    m = tl.load(mask_ptr + mask_index * stride_mask, mask=mask)

    out = a * b * m
    tl.store(out_ptr + offs * stride_out, out, mask=mask)


# ----------------------------------------------------------------------
# Helper for applying gates (and optional mask) using Triton
# ----------------------------------------------------------------------
def _apply_gate(left, gate, mask, hidden_dim, block_size=256):
    """
    left, gate : tensors of shape (B, N, N, H)  (contiguous)
    mask      : None or tensor of shape (B, N, N)  (contiguous, dtype=fp16)
    hidden_dim: int (H)
    """
    # flatten to 1‑D contiguous view
    left_flat = left.view(-1)
    gate_flat = gate.view(-1)
    size = left_flat.numel()

    # all strides are 1 after flattening
    stride = 1

    grid = ( (size + block_size - 1) // block_size, )

    if mask is None:
        _gate_mul_kernel[grid](
            left_flat, gate_flat, left_flat,
            size,
            stride, stride, stride,
            BLOCK_SIZE=block_size,
        )
    else:
        mask_flat = mask.view(-1)  # shape (B*N*N,)
        _gate_mul_mask_kernel[grid](
            left_flat, gate_flat, mask_flat, left_flat,
            size,
            hidden_dim,
            stride, stride, stride, stride,
            BLOCK_SIZE=block_size,
        )
    # in‑place modification, no need to return


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor of shape [B, N, N] (binary) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on input (dim = C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # Cast to fp16 for the compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Cast all linear weights to fp16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (shape [B, N, N, H])
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Gate projections (shape [B, N, N, H])
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # used later

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if any) and gates with a fused Triton kernel
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is fp16 and contiguous
        mask_f = mask.to(torch.float16).contiguous()
    else:
        mask_f = None

    _apply_gate(left, left_gate, mask_f, hidden_dim)
    _apply_gate(right, right_gate, mask_f, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³·H) reduction via batched matmul (cuBLAS)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) for batched matmul
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched GEMM: left @ rightᵀ   (shape (B*H, N, N))
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to the original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
