```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])
    # Convert to half
    x = x.to(torch.float16)

    # Cast weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Projections
    left = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply gates (and mask) using Triton kernel
    # Allocate output tensors
    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    # Triton kernel constants
    total_elements = left.numel()  # B*N*N*H
    BLOCK_SIZE = 65536  # 64K elements per block (customizable)

    grid = lambda meta: (triton.cdiv(total_elements, BLOCK_SIZE),)

    _apply_gates_kernel[grid](
        left, left_gate, right, right_gate,
        left_gated, right_gated,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # apply mask if needed
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_gated = left_gated * mask_f
        right_gated = right_gated * mask_f

    # batched matmul: left_gated @ right_gated^T over last dim
    B, N, _, H = left_gated.shape
    # reshape to (B*H, N, N)
    left_mat = left_gated.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_gated.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # compute out = left_mat @ right_mat.transpose(1,2)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape to original dims
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm over hidden_dim
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # Apply out_gate
    out = out * out_gate

    # Final projection
    out = F.linear(out, to_out_w)

    # Cast back to original dtype
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
```

```
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _apply_gates_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    total_elements: tl.int32,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements
    l = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0))
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0))
    r = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0))
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0))
    tl.store(out_left_ptr + offs, l * lg, mask=mask)
    tl.store(out_right_ptr + offs, r * rg, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    The implementation follows the reference PyTorch module but replaces the
    O(N³·hidden) reduction with a highly‑optimized batched GEMM (torch.bmm
    which leverages Tensor Cores) and fuses the element‑wise gating step with
    a tiny Triton kernel.  All heavy linear algebra is performed in FP16,
    while layer‐norms and the final projection are kept in the original dtype.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Work in FP16 for the heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast linear weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Fuse left/right with their gates via a tiny Triton kernel
    # ------------------------------------------------------------------
    left_gated  = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    total_elems = left.numel()
    BLOCK_SIZE = 65536  # 64K elements ≈ 128KB for fp16
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

    _apply_gates_kernel[grid](
        left, left_gate, right, right_gate,
        left_gated, right_gated,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 6) Optional binary mask
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_gated  = left_gated  * mask_fp
        right_gated = right_gated * mask_fp

    # ------------------------------------------------------------------
    # 7) Batched matrix multiplication: left @ rightᵀ
    # ------------------------------------------------------------------
    B, N, _, H = left_gated.shape
    # reshape to (B*H, N, N)
    left_mat  = left_gated.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_gated.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # torch.bmm uses Tensor Cores for FP16 and is heavily optimized
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9) Apply output gate and final projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
```

"""
TriMul – outgoing Triangle Multiplicative Update (AlphaFold3 style)

Algorithm
---------
1. Layer‑norm the input (float32) → fp16.
2. Linear projections (left/right) and three gate projections, all in fp16.
3. Apply the binary mask (if any) to the projected tensors via a tiny
   Triton kernel (element‑wise multiplication).
4. Fuse the element‑wise gating (left*left_gate, right*right_gate) with a
   second Triton kernel – performed in‑place to save memory.
5. Reshape to (B·H, N, N) and compute the core O(N³·H) reduction with
   torch.bmm (leverages cuBLAS Tensor‑cores).
6. Reshape back to (B, N, N, H), apply a second LayerNorm over the hidden
   dimension, multiply by the output gate and a final linear projection
   back to the original channel dimension.
7. Cast the result back to the input dtype.

Only the element‑wise mask and gate fusion steps are written in Triton,
while the dominant batched matrix multiplication uses the highly‑optimized
cuBLAS implementation.  All heavy arithmetic is in FP16, giving a fast,
memory‑efficient forward pass.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_kernel(
    left_ptr, right_ptr,            # input tensors [B,N,N,H] (flattened)
    mask_ptr,                       # mask tensor [B,N,N] (flattened)
    out_left_ptr, out_right_ptr,    # destinations (can be same as inputs)
    total_elements: tl.int32,       # total number of elements in left/right
    hidden_dim: tl.int32,           # size of the hidden dimension H
    BLOCK_SIZE: tl.constexpr,       # number of elements per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # Load left / right values
    l = tl.load(left_ptr + offs, mask=valid, other=tl.float16(0))
    r = tl.load(right_ptr + offs, mask=valid, other=tl.float16(0))

    # Compute mask index (broadcast H times)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(1.0))

    # Write masked results (in‑place write‑back is safe)
    tl.store(out_left_ptr + offs,  l * m, mask=valid)
    tl.store(out_right_ptr + offs, r * m, mask=valid)


@triton.jit
def _apply_gates_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    total_elements: tl.int32,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    l  = tl.load(left_ptr + offs,        mask=valid, other=tl.float16(0))
    lg = tl.load(left_gate_ptr + offs,  mask=valid, other=tl.float16(0))
    r  = tl.load(right_ptr + offs,       mask=valid, other=tl.float16(0))
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0))

    tl.store(out_left_ptr + offs,  l * lg, mask=valid)
    tl.store(out_right_ptr + offs, r * rg, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[bs, seq_len, seq_len, dim] (float32 or float16)
            - mask: torch.Tensor[bs, seq_len, seq_len] (bool/float) or None
            - weights: dict of model parameters (float32)
            - config: dict containing "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor[bs, seq_len, seq_len, dim] (same dtype as input)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32) → FP16 for the heavy path
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast linear weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Optional binary mask (applied before gating)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Flatten mask to (B * N * N) and make it fp16
        mask_flat = mask.to(torch.float16).contiguous().view(-1)
        total_elems = left.numel()
        BLOCK_SIZE = 65536  # ~128 KB per block (fp16)

        grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

        # In‑place mask application via Triton
        _apply_mask_kernel[grid](
            left, right,
            mask_flat,
            left, right,                 # outputs overwrite inputs
            total_elems,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
        )

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid) – computed from the original normalized input
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Fuse gating (left*left_gate, right*right_gate) – in‑place
    # ------------------------------------------------------------------
    total_elems = left.numel()
    BLOCK_SIZE = 65536
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

    _apply_gates_kernel[grid](
        left, left_gate,
        right, right_gate,
        left, right,                 # overwrite inputs with gated values
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 7) Batched matrix multiplication: left @ rightᵀ  (FP16, Tensor cores)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape      # (bs, seq_len, seq_len, hidden_dim)
    # reshape to (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9) Apply output gate and final projection to original channel dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 10) Cast back to the input dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
