def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # Prepare weights
    # Norm weight/bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Linear weights in half
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # 1. LayerNorm over channel dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    ) # shape [B, N, N, dim], dtype float32

    # Cast to FP16 for subsequent ops
    x_half = x.to(torch.float16)

    # 2. Compute left/right projections and gates in a fused linear
    # combine weights [left_proj; left_gate; right_proj; right_gate] shape (4*hidden, dim)
    combined_weight = torch.cat([left_proj_weight, left_gate_weight, right_proj_weight, right_gate_weight], dim=0) # shape (4*hidden, dim)

    # flatten x to (B*N*N, dim)
    B, N, _, _ = x.shape
    x_flat = x_half.view(-1, dim) # shape (B*N*N, dim)

    # compute all linear outputs
    combined_out = torch.nn.functional.linear(x_flat, combined_weight) # shape (B*N*N, 4*hidden)

    # reshape to [B, N, N, 4*hidden]
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)

    # split
    left_proj, left_gate, right_proj, right_gate = torch.split(combined_out, hidden_dim, dim=-1) # each shape (B, N, N, hidden)

    # mask
    if not nomask:
        mask_f = mask.to(x_half.dtype).unsqueeze(-1) # shape (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # gating
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    left = left_proj * left_gate # (B,N,N,H)
    right = right_proj * right_gate

    # 3. Batched GEMM for out: out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]

    # Rearrange to (B, H, N, N) for each batch/h
    left_permuted = left.permute(0, 3, 1, 2).contiguous() # shape B, H, N,N
    right_permuted = right.permute(0, 3, 1, 2).contiguous()

    # merge batch and hidden
    BH = B * hidden_dim
    left_mat = left_permuted.view(BH, N, N)
    right_mat = right_permuted.view(BH, N, N)

    # allocate output tensor
    out_mat = torch.empty_like(left_mat) # same dtype and device

    # call Triton batched GEMM
    # compute strides
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)
    stride_b_batch = right_mat.stride(0)
    stride_b_k = right_mat.stride(1)
    stride_b_n = right_mat.stride(2)
    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    M = N
    N_dim = N
    K = N
    # Choose block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (BH, 
            (M + BLOCK_M - 1) // BLOCK_M,
            (N_dim + BLOCK_N - 1) // BLOCK_N)

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        M, N_dim, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back to [B,N,N,hidden]
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # 4. to_out_norm (layernorm over hidden dim)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # 5. compute out_gate (sigmoid of linear)
    out_gate = torch.nn.functional.linear(x_half, out_gate_weight) # shape [B,N,N,hidden]
    out_gate = torch.sigmoid(out_gate)

    # 6. Apply out_gate elementwise using Triton
    # Could use elementwise kernel as previously
    total_el = out.numel()
    BLOCK = 1024
    grid_mul = (total_el + BLOCK - 1) // BLOCK
    elementwise_mul_kernel[grid_mul](out, out_gate, out, total_el, BLOCK=BLOCK)

    # 7. Final linear to dim
    # Ouptu shape [B, N, N, dim] using to_out.weight shape (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_weight)

    # cast to float32 as required
    out = out.to(torch.float32)
    return out
```

"""
Outgoing Triangle Multiplicative Update (TriMul) – optimized forward pass.

The implementation follows the reference PyTorch module but
* all linear projections and gating are fused into a single half‑precision
  linear layer,
* the cubic reduction `out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]`
  is performed by a custom Triton batched GEMM kernel (B·hidden independent
  matrix multiplications of size N×N),
* the final element‑wise multiplication with the output gate is also a
  Triton kernel.

All heavy arithmetic is done in float16 (with float32 accumulation in the
GEMM) and the result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_a_batch, stride_am, stride_ak,
    stride_b_batch, stride_bk, stride_bn,
    stride_c_batch, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ Bᵀ  where A,B shape (batch, M, K) and (batch, K, N)."""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # pointers for the first tile of the K dimension
    a_ptrs = A_ptr + pid_batch * stride_a_batch \
        + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_batch * stride_b_batch \
        + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, dtype=tl.float16)
        b = tl.load(b_ptrs, dtype=tl.float16)
        acc += tl.dot(a, b)            # (BLOCK_M, BLOCK_N) accumulation in fp32
        a_ptrs += BLOCK_K * stride_ak   # advance K dimension
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c_ptrs = C_ptr + pid_batch * stride_c_batch \
        + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, size: tl.constexpr, BLOCK: tl.constexpr):
    """c = a * b  for flat tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) – optional
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # --------------------------------------------------------------
    # Load and cast weights
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias   = weights["norm.bias"].to(device).to(torch.float16)

    left_proj_weight  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)

    left_gate_weight  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight   = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over channel dimension (half precision)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim], float16 after cast below
    x = x.to(torch.float16)

    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates
    #    Combined weight shape: (4*hidden, dim)
    # --------------------------------------------------------------
    combined_weight = torch.cat(
        [left_proj_weight, left_gate_weight, right_proj_weight, right_gate_weight],
        dim=0,
    )  # (4*hidden, dim)

    x_flat = x.view(-1, dim)                     # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 4*hidden)

    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each (B, N, N, hidden)

    # --------------------------------------------------------------
    # 3️⃣ Optional masking (applied before gating)
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and combine with projections
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate    # (B, N, N, hidden)
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Batched GEMM: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a Triton batched GEMM kernel.
    # --------------------------------------------------------------
    # reshape to (B, hidden, N, N) and merge batch & hidden
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, hidden, N, N)

    BH = B * hidden_dim
    left_mat = left.view(BH, N, N)
    right_mat = right.view(BH, N, N)
    out_mat = torch.empty_like(left_mat)  # (BH, N, N), fp16

    # strides for the kernel
    stride_a_batch = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_a_batch,
        stride_am,
        stride_ak,
        stride_b_batch,
        stride_bk,
        stride_bn,
        stride_c_batch,
        stride_cm,
        stride_cn,
        N,    # M
        N,    # N
        N,    # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # still half precision

    # --------------------------------------------------------------
    # 7️⃣ Output gate (linear + sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------
    out_gate = F.linear(x, out_gate_weight)  # (B, N, N, hidden)
    out_gate = torch.sigmoid(out_gate)

    total_el = out.numel()
    BLOCK = 1024
    grid_mul = (total_el + BLOCK - 1) // BLOCK
    elementwise_mul_kernel[grid_mul](
        out,
        out_gate,
        out,
        total_el,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 8️⃣ Final projection back to original dimension
    # --------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # (B, N, N, dim), fp16
    out = out.to(torch.float32)         # return float32
    return out
