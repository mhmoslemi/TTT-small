"""
TriMul (outgoing) forward pass accelerated with Triton.
----------------------------------------------------
The operator works on a tensor x ∈ ℝ^{B×N×N×C}:

1. Layer‑norm over the channel dimension C.
2. Linear projections to a hidden dimension H (left/right) and three
   gating projections (left_gate, right_gate, out_gate).  All linear
   layers are bias‑free and performed in FP16 for speed.
3. Optional binary mask (broadcast on the hidden dim) applied before the gates.
4. Gated left/right tensors are reshaped to (B·H, N, N) and the core
   O(N³·H) reduction is performed as a batched matrix multiplication
   C = A @ Bᵀ using a custom Triton kernel.  The kernel tiles the
   matrices (M,N,K) = (N,N,N) with block sizes tuned for the
   sequence length and uses fp16 loads with fp32 accumulation.
5. The result is reshaped back to (B,N,N,H), normalized (LayerNorm
   over H), multiplied by the output gate and finally projected back
   to the original channel dimension C.
6. The output is returned in the same dtype as the input (float32).

Only the heavy reduction (step 4) is implemented in Triton; all
other tensor operations use PyTorch which already provides fast
vectorized kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (computes C = A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dims (all == seqlen)
    stride_ah, stride_am, stride_ak,   # A: batch, row, col
    stride_bh, stride_bk, stride_bn,   # B (as transposed): batch, col(K), row(N)
    stride_ch, stride_cm, stride_cn,   # C: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for A @ Bᵀ.
    A, B, C have shape (batch, M, K) and (batch, N, K) respectively.
    The kernel computes C[b, i, j] = Σ_k A[b,i,k] * B[b,j,k].
    """
    pid_batch = tl.program_id(2)          # which batch (including hidden dim)
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block col

    # ------------------------------------------------------------------
    # Offsets within the tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Compute base pointers for this batch slice
    # ------------------------------------------------------------------
    a_batch = pid_batch * stride_ah
    b_batch = pid_batch * stride_bh
    c_batch = pid_batch * stride_ch

    # Pointers for the current tile (K dimension will be iterated)
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + offs_k[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (offs_k[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        k_mask = (k * BLOCK_K + offs_k) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_mask[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)    # (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store result
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[:, None])

def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the Triton batched GEMM kernel.
    A, B : FP16 tensors of shape (batch, N, N)
    Returns C of the same shape and dtype.
    """
    assert A.shape == B.shape, "A and B must have the same shape"
    batch, N, K = A.shape
    assert N == K, "TriMul requires square sequence dimension"

    # Choose block sizes heuristically
    if N <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
    else:  # N up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over M
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over N
        batch,                          # batch dimension (including hidden)
    )

    C = torch.empty_like(A)

    _batched_gemm_kernel[grid](
        A, B, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),   # A: batch, row, col
        B.stride(0), B.stride(2), B.stride(1),   # B as transposed (batch, K, N)
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )
    return C

# ----------------------------------------------------------------------
# Entrypoint: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Work in FP16 for the heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast linear weights to FP16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4. Optional mask (binary) – broadcast over hidden dim
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5. Gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Reshape for the batched GEMM: (B, H, N, N) → (B·H, N, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat  = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction via Triton
    # ------------------------------------------------------------------
    out_flat = _batched_gemm_triton(left_flat, right_flat)   # (B·H, N, N)

    # ------------------------------------------------------------------
    # 8. Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9. Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 10. Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11. Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 12. Cast back to original dtype if needed
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)

    return out
```
