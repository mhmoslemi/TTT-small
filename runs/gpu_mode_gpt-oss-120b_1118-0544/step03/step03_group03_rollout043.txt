def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    # Compute start offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    # offsets within block
    i_offset = i_base + tl.arange(0, BLOCK_M)
    j_offset = j_base + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Initialize accumulator for hidden chunk
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # loop over hidden dimension
    for d_off in range(0, H, BLOCK_D):
        # initialize accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # loop over K dimension
        for k_base in range(0, N, BLOCK_K):
            k = k_base + tl.arange(0, BLOCK_K)
            mask_k = k < N

            # pointers
            left_ptrs = left_ptr + \
                pid_b * stride_left_b + \
                i[:, None, None] * stride_left_i + \
                k[None, :, None] * stride_left_k + \
                (d_off + tl.arange(0, BLOCK_D))[None, None, :] * stride_left_d

            right_ptrs = right_ptr + \
                pid_b * stride_right_b + \
                j[:, None, None] * stride_right_j + \
                k[None, :, None] * stride_right_k + \
                (d_off + tl.arange(0, BLOCK_D))[None, None, :] * stride_right_d

            left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0).to(tl.float32)
            right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0).to(tl.float32)

            # accumulate over k within block
            for kk in range(0, BLOCK_K):
                # broadcasting left[:, kk, :] (M,D) and right[:, kk, :] (N,D)
                l = left[:, kk, :]   # (M, D)
                r = right[:, kk, :]  # (N, D)
                # broadcast and multiply
                acc += l[:, None, :] * r[None, :, :]

        # store result
        out_ptrs = out_ptr + \
            pid_b * stride_out_b + \
            i[:, None, None] * stride_out_i + \
            j[None, :, None] * stride_out_j + \
            (d_off + tl.arange(0, BLOCK_D))[None, None, :] * stride_out_d

        mask_d = (d_off + tl.arange(0, BLOCK_D)) < H
        mask_store = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        tl.store(out_ptrs, acc.to(tl.float16), mask=mask_store)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr):
    '''
    Compute out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    left/right: [B, N, N, H] (fp16)
    out: same shape (fp16)
    '''
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # coordinates of rows and columns of the output tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # loop over hidden dimension blocks
    for d_off in range(0, H, BLOCK_D):
        # allocate per‑d accumulator that will be written out piece‑wise
        # we compute each d in a separate inner loop to let tl.dot use Tensor Cores
        for d in range(0, BLOCK_D):
            # If we are beyond the actual hidden size, break out (mask later)
            # The mask will delete writes for out‑of‑range d.
            acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

            # loop over K (the third axis)
            for k_base in range(0, N, BLOCK_K):
                k = k_base + tl.arange(0, BLOCK_K)
                mask_k = k < N

                # pointers for left tile (M,K)
                left_ptrs = left_ptr + \
                    pid_b * stride_left_b + \
                    i[:, None] * stride_left_i + \
                    k[None, :] * stride_left_k + \
                    (d_off + d) * stride_left_d

                left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

                # pointers for right tile (K,N) – note the transposed layout needed for dot
                right_ptrs = right_ptr + \
                    pid_b * stride_right_b + \
                    j[None, :] * stride_right_j + \
                    k[:, None] * stride_right_k + \
                    (d_off + d) * stride_right_d

                right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

                # accumulate using Tensor‑Core‑enabled dot product
                acc += tl.dot(left_tile, right_tile)

            # store the accumulated matrix for this hidden slice
            out_ptrs = out_ptr + \
                pid_b * stride_out_b + \
                i[:, None] * stride_out_i + \
                j[None, :] * stride_out_j + \
                (d_off + d) * stride_out_d

            mask_store = mask_i[:, None] & mask_j[None, :]
            tl.store(out_ptrs, acc.to(tl.float16), mask=mask_store)

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy N³ reduction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is fused into a single Triton kernel.  The kernel works on tiles of the
(i,j) plane and iterates over the hidden dimension in blocks.  All
arithmetics are performed in fp16 with fp32 accumulation; the final
output is cast back to fp32 for the subsequent LayerNorm and output
projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr):
    """
    Compute
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    for a tile (BLOCK_M × BLOCK_N) of (i,j) and a block of hidden
    channels (BLOCK_D).  All tensors are stored as [B, N, N, H] with
    standard contiguous strides.
    """
    pid_b = tl.program_id(0)          # batch
    pid_i = tl.program_id(1)          # output i‑tile
    pid_j = tl.program_id(2)          # output j‑tile

    # i and j coordinates of the current tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)   # (M,)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)   # (N,)

    mask_i = i < N
    mask_j = j < N

    # iterate over hidden dimension blocks
    for d_off in range(0, H, BLOCK_D):
        # --------- loop over hidden sub‑block ---------------------------------
        for d in range(0, BLOCK_D):
            # stop when we exceed the actual hidden size
            valid_d = (d_off + d) < H

            # accumulator for this (i,j,d) slice
            acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

            # --------- loop over the k dimension -------------------------------
            for k_base in range(0, N, BLOCK_K):
                k = k_base + tl.arange(0, BLOCK_K)          # (K,)
                mask_k = k < N

                # ----- left tile  (M,K)  --------------------------------------
                left_ptrs = left_ptr + \
                    pid_b * stride_left_b + \
                    i[:, None] * stride_left_i + \
                    k[None, :] * stride_left_k + \
                    (d_off + d) * stride_left_d
                mask_l = mask_i[:, None] & mask_k[None, :] & valid_d
                left_tile = tl.load(left_ptrs, mask=mask_l, other=0.0)

                # ----- right tile (K,N) – note the transposed layout ---------- 
                right_ptrs = right_ptr + \
                    pid_b * stride_right_b + \
                    j[None, :] * stride_right_j + \
                    k[:, None] * stride_right_k + \
                    (d_off + d) * stride_right_d
                mask_r = mask_k[:, None] & mask_j[None, :] & valid_d
                right_tile = tl.load(right_ptrs, mask=mask_r, other=0.0)

                # ----- accumulate with Tensor‑Core‑enabled dot -----------------
                acc += tl.dot(left_tile, right_tile)   # (M,N) fp32

            # ----- write the result for this hidden channel --------------------
            out_ptrs = out_ptr + \
                pid_b * stride_out_b + \
                i[:, None] * stride_out_i + \
                j[None, :] * stride_out_j + \
                (d_off + d) * stride_out_d
            mask_store = mask_i[:, None] & mask_j[None, :] & valid_d
            tl.store(out_ptrs, acc.to(tl.float16), mask=mask_store)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module, fully fused except for
    the trivial LayerNorm/Linear parts.  Returns a float32 tensor of
    shape [bs, seq_len, seq_len, dim].
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ----------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ----------------------------------------------------
    # 2) Cast to fp16 for the heavy parts
    # ----------------------------------------------------
    x_h = x.to(torch.float16)

    # ----------------------------------------------------
    # 3) Fuse projection + gate linear layers (fp16)
    # ----------------------------------------------------
    left_combined_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    right_combined_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    left_combined = F.linear(x_h, left_combined_weight)    # [B,N,N,2*H] fp16
    right_combined = F.linear(x_h, right_combined_weight)

    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    # ----------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # ----------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ----------------------------------------------------
    # 5) Gating (sigmoid, fp16)
    # ----------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate      # [B,N,N,H] fp16
    right = right_proj * right_gate   # [B,N,N,H] fp16

    # 6) Output gate (fp16)
    # ----------------------------------------------------

    out_gate_pre = F.linear(x_h, out_gate_weight)   # [B,N,N,H] fp16
    out_gate = torch.sigmoid(out_gate_pre)          # fp16

    # ----------------------------------------------------
    # 7) Triangular multiplicative update via a Triton kernel
    # ----------------------------------------------------
    B, N, _, H = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out_fp16 = torch.empty_like(left, dtype=torch.float16)

    # strides of the tensors (contiguous layout)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out_fp16.stride()

    # Tile sizes – tuned for the B200 GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 32                     # hidden‑dim block

    grid = (
        B,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_fp16,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_l_b,
        stride_left_i=stride_l_i,
        stride_left_k=stride_l_k,
        stride_left_d=stride_l_d,
        stride_right_b=stride_r_b,
        stride_right_j=stride_r_i,      # second dim of right = j
        stride_right_k=stride_r_k,
        stride_right_d=stride_r_d,
        stride_out_b=stride_o_b,
        stride_out_i=stride_o_i,
        stride_out_j=stride_o_j,
        stride_out_d=stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ----------------------------------------------------
    # 8) Cast back to fp32 for the final LayerNorm / projection
    # ----------------------------------------------------
    out = out_fp16.to(torch.float32)

    # ----------------------------------------------------
    # 9) Final LayerNorm, output gating and linear projection
    # ----------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.to(torch.float32)            # apply output gate (fp32)
    out = F.linear(out, weights["to_out.weight"])    # final projection, fp32

    return out
