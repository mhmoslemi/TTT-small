def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Get device
    dev = input_tensor.device
    dtype = torch.float32 # keep input in float32

    # LayerNorm
    norm_weight = weights["norm.weight"].to(dev)
    norm_bias = weights["norm.bias"].to(dev)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5
    )  # shape [B,N,N,dim] float32

    # Cast to half for linear projections
    x_fp16 = x.to(torch.float16)

    B, N, _, _ = x.shape

    # Combine linear weights: left_proj, left_gate, right_proj, right_gate, out_gate
    # All shape (hidden, dim)
    left_proj_weight = weights["left_proj.weight"].to(dev).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(dev).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(dev).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(dev).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(dev).to(torch.float16)

    combined_weight = torch.cat([left_proj_weight,
                                 left_gate_weight,
                                 right_proj_weight,
                                 right_gate_weight,
                                 out_gate_weight], dim=0)  # (5*hidden, dim)

    # Linear projection: flatten first three dims
    x_flat = x_fp16.view(-1, dim)  # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 5*hidden)
    # reshape back
    combined_out = combined_out.view(B, N, N, 5 * hidden)

    # Split
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(combined_out, hidden, dim=-1)  # each: (B,N,N,hidden)

    # Apply mask if necessary
    if not nomask:
        # mask: [B,N,N], could be bool or float; convert to float16
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Gates(sigmoid)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Apply gates to projections
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Ensure half
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Allocate output tensor
    out = torch.empty_like(left, dtype=torch.float16)  # (B,N,N,hidden)

    # Triton kernel for batched GEMM (out = left @ right^T across k)
    # Compute pointers and strides
    a_ptr = left.data_ptr()
    b_ptr = right.data_ptr()
    c_ptr = out.data_ptr()
    # strides in elements
    stride_a_batch = left.stride(0)
    stride_ai = left.stride(1)   # i
    stride_ak = left.stride(2)   # k
    stride_ad = left.stride(3)   # hidden

    stride_b_batch = right.stride(0)
    stride_bk = right.stride(2)   # k (row for B)
    stride_bj = right.stride(1)   # j (col)
    stride_bd = right.stride(3)   # hidden

    stride_c_batch = out.stride(0)
    stride_ci = out.stride(1)
    stride_cj = out.stride(2)
    stride_cd = out.stride(3)

    # Determine block sizes based on N
    # Simple heuristic: set BLOCK_M = BLOCK_N = 128 if N >= 256 else 64; BLOCK_K = 32
    if N >= 256:

        BLOCK_N = 128
        BLOCK_K = 32

        BLOCK_N = 64
        BLOCK_K = 16

    grid = (batch_hidden, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

        a_ptr,
        b_ptr,
        c_ptr,
        stride_a_batch, stride_ai, stride_ak, stride_ad,
        stride_b_batch, stride_bk, stride_bj, stride_bd,
        stride_c_batch, stride_ci, stride_cj, stride_cd,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN=hidden,
        BATCH=B,
        num_warps=4,
    )

    # Now out is computed in half precision

    # Apply to_out_norm (LayerNorm over hidden dim)
    to_out_norm_weight = weights["to_out_norm.weight"].to(dev).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(dev).to(torch.float16)

    out = out.to(torch.float32)
    return out

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm:
1. Layer‑norm the input (float32) over the channel dimension.
2. Fuse the five linear projections (left/right and their three gates) into a
   single matmul (shape (B·N·N, 5·hidden)).
3. Split the result → left_proj, left_gate, right_proj, right_gate, out_gate.
4. Optional mask is applied to left/right projections.
5. Apply sigmoid gates and multiply with the corresponding projections.
6. Compute the core O(N³) reduction
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   as a batched GEMM using a custom Triton kernel (no data transposes or
   extra copies).
7. Layer‑norm over the hidden dimension, multiply by out_gate and project back
   to the original channel dimension.
All heavy arithmetic is performed in FP16 (with FP32 accumulation) and the
final result is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    # Pointers
    a_ptr, b_ptr, c_ptr,
    # Strides (in elements)
    stride_a_batch, stride_ai, stride_ak, stride_ad,
    stride_b_batch, stride_bk, stride_bj, stride_bd,
    stride_c_batch, stride_ci, stride_cj, stride_cd,
    # Matrix sizes
    M, N, K,
    # Compile‑time constants
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HIDDEN: tl.constexpr, BATCH: tl.constexpr,
):
    """Batched GEMM for TriMul:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
       left/right have shape (B, N, N, hidden) and are interpreted as
       (batch = B·hidden, M=N, K=N) and (batch = B·hidden, K=N, N=N) respectively.
    """
    pid_z = tl.program_id(0)          # combined (batch, hidden) dimension
    pid_m = tl.program_id(1)          # block row (i)
    pid_n = tl.program_id(2)          # block col (j)

    # Recover batch and hidden indices
    b = pid_z // HIDDEN
    d = pid_z % HIDDEN

    # Base offsets for this (b,d) pair
    a_batch_off = b * stride_a_batch + d * stride_ad
    b_batch_off = b * stride_b_batch + d * stride_bd
    c_batch_off = b * stride_c_batch + d * stride_cd

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds elements
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the current tile
    a_ptrs = a_ptr + a_batch_off + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + b_batch_off + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = offs_k + k_iter * BLOCK_K
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (cur_k[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(cur_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)           # FP16×FP16 → FP32 accumulation

        # Advance pointers by BLOCK_K elements in the K‑direction
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write result
    c_ptrs = c_ptr + c_batch_off + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ---------- 2. Fused linear projections (5 * hidden) ----------
    # Convert to half for the heavy matmuls
    x_fp16 = x.to(torch.float16)

    # Gather weight tensors, cast to fp16
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    # Combine into a single weight matrix of shape (5*hidden, dim)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # (5*hidden, dim)

    B, N, _, _ = x.shape
    x_flat = x_fp16.view(-1, dim)               # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 5*hidden)
    combined_out = combined_out.view(B, N, N, 5 * hidden)

    # Split back into the five tensors
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        combined_out, hidden, dim=-1
    )  # each: (B, N, N, hidden)

    # ---------- 3. Optional mask ----------
    if not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ---------- 4. Gates ----------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ---------- 5. Apply gates ----------
    left = left_proj * left_gate      # (B, N, N, hidden) fp16
    right = right_proj * right_gate   # (B, N, N, hidden) fp16

    # ---------- 6. Batched GEMM via Triton ----------
    out = torch.empty_like(left, dtype=torch.float16)  # (B,N,N,hidden)

    # Pointer and stride information (in element units)
    a_ptr = left.data_ptr()
    b_ptr = right.data_ptr()
    c_ptr = out.data_ptr()

    # Strides for A (left)
    stride_a_batch = left.stride(0)   # B‑stride
    stride_ai = left.stride(1)        # i‑stride
    stride_ak = left.stride(2)        # k‑stride
    stride_ad = left.stride(3)        # hidden‑stride (==1)

    # Strides for B (right). Note: we need right transposed (k‑j)
    stride_b_batch = right.stride(0)  # B‑stride
    stride_bk = right.stride(2)       # k‑stride (row of Bᵀ)
    stride_bj = right.stride(1)       # j‑stride (col of Bᵀ)
    stride_bd = right.stride(3)       # hidden‑stride (==1)

    # Strides for C (output)
    stride_c_batch = out.stride(0)
    stride_ci = out.stride(1)
    stride_cj = out.stride(2)
    stride_cd = out.stride(3)

    # Heuristic block sizes
    if N >= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 16

    batch_hidden = B * hidden
    grid = (
        batch_hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_gemm_kernel[grid](
        a_ptr,
        b_ptr,
        c_ptr,
        stride_a_batch,
        stride_ai,
        stride_ak,
        stride_ad,
        stride_b_batch,
        stride_bk,
        stride_bj,
        stride_bd,
        stride_c_batch,
        stride_ci,
        stride_cj,
        stride_cd,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN=hidden,
        BATCH=B,
        num_warps=4,
    )

    # ---------- 7. Post‑processing ----------
    # LayerNorm over the hidden dimension
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )

    # Apply out_gate (element‑wise)
    out = out * out_gate

    # Final projection back to the original dimension
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, dim) fp16

    # Return as float32
    return out.to(torch.float32)
