def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Load weights
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # 1. LayerNorm on input
    x = torch.nn.functional.layer_norm(input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5)  # shape [B,N,N,dim], float32

    # Cast to half for rest
    x_fp16 = x.to(torch.float16)

    B, N, _, _ = x_fp16.shape

    # 2. Combined linear: left_proj, left_gate, right_proj, right_gate, out_gate
    # combined weight shape (5*hidden_dim, dim)
    combined_weight = torch.cat([left_proj_weight,
                                 left_gate_weight,
                                 right_proj_weight,
                                 right_gate_weight,
                                 out_gate_weight], dim=0) # shape (5*hidden, dim)

    x_flat = x_fp16.view(-1, dim)  # shape (B*N*N, dim)
    combined_out = torch.nn.functional.linear(x_flat, combined_weight)  # (B*N*N, 5*hidden)

    # Reshape back
    combined_out = combined_out.view(B, N, N, 5*hidden_dim)

    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(combined_out, hidden_dim, dim=3)

    # Mask left and right
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # (B,N,N,1) float32
        mask_f = mask_f.to(torch.float16)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Gating (sigmoid)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # 3. Batched GEMM using bmm
    # permute to (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # reshape to (B*hidden, N, N)
    left_mat = left.view(B * hidden_dim, N, N)  # (batch*hidden, N, N)
    right_mat = right.view(B * hidden_dim, N, N)  # (batch*hidden, N, N)

    # bmm left_mat @ right_mat^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    # reshape to (B,N,N,hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # 4. to_out_norm (layernorm over hidden)
    out = torch.nn.functional.layer_norm(out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5)

    # 5. out_gate multiplication: apply sigmoid already? Wait out_gate is from combined linear raw output. Need to apply sigmoid.
    out_gate = torch.sigmoid(out_gate)  # (B,N,N, hidden)

    # Multiply out * out_gate using Triton kernel
    total_el = out.numel()
    # Use Triton kernel defined below

    # 6. final linear to dim
    out = torch.nn.functional.linear(out, to_out_weight)  # (B,N,N,dim)

    # cast back to float32
    out = out.to(torch.float32)

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the reference PyTorch module but fuses many
operations:
  * One combined linear layer produces left/right projections, their gates,
    and the out‑gate.
  * Optional masking is applied only to left/right projections.
  * Left/right are gated, then a batched GEMM (B*hidden_dim, N, N) @
    (B*hidden_dim, N, N).T computes the `einsum('... i k d, ... j k d -> ... i j d')`
    in essentially one cuBLAS call.
  * Layer‑norm on the hidden dimension and the element‑wise out‑gate
    multiplication are performed, the latter by a tiny Triton kernel.
  * A final linear projects back to the original channel dimension.

All heavy arithmetic runs in float16 (FP16) for speed; the final result is
returned as float32.
"""
import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def _elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, total_el, BLOCK: tl.constexpr):
    """c = a * b   (1‑D flattened tensors)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_el
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float); may be ignored
                    if config["nomask"] is True.
            - weights: dict of model weights (float32 tensors) with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with keys:
                "dim" (int), "hidden_dim" (int), optional "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # Load and cast weights (keep them in FP16 for the heavy compute)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)   # (dim, hidden)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the input channel dimension (float32 for precision)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  float32

    # Cast to FP16 for the remaining ops
    x_fp16 = x.to(torch.float16)
    B, N, _, _ = x_fp16.shape

    # ------------------------------------------------------------------
    # 2) Single fused linear layer produces left/right proj + gates + out_gate
    # ------------------------------------------------------------------
    # Concatenate weights (shape: (5*hidden_dim, dim))
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
            out_gate_weight,
        ],
        dim=0,
    )  # (5*hidden_dim, dim)

    # Linear on the flattened tensor
    x_flat = x_fp16.view(-1, dim)                     # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 5*hidden_dim)

    # Reshape back to 4‑D and split
    combined_out = combined_out.view(B, N, N, 5 * hidden_dim)
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        combined_out, hidden_dim, dim=3
    )  # each (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3) Optional masking (only on the projections)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: (B, N, N) -> (B, N, N, 1) -> fp16
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 4) Apply gates (sigmoid) and combine
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate          # (B, N, N, hidden)
    right = right_proj * right_gate       # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 5) Batched GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a single torch.bmm over (B*hidden) batches.
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) -> (B*hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left.view(B * hidden_dim, N, N)   # (B*hidden, N, N)
    right_mat = right.view(B * hidden_dim, N, N) # (B*hidden, N, N)

    # Batched matmul with transposed right operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    # Restore shape (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Hidden‑dim LayerNorm
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden)  fp16

    # ------------------------------------------------------------------
    # 7) Out‑gate (sigmoid) and element‑wise multiplication via Triton
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate)  # (B, N, N, hidden) fp16

    total_el = out.numel()
    BLOCK = 1024
    grid = ((total_el + BLOCK - 1) // BLOCK,)

    # Triton kernel: out = out * out_gate
    _elementwise_mul_kernel[grid](
        out, out_gate, out, total_el, BLOCK=BLOCK
    )

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # (B, N, N, dim) fp16

    # Return in float32 as required by the spec
    return out.to(torch.float32)
