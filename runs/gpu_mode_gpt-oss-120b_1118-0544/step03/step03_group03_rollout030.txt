Triton kernel signature:

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    PD, N,
    stride_left_pd, stride_left_i, stride_left_k,
    stride_right_pd, stride_right_i, stride_right_k,
    stride_out_pd, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    ):
   pid = tl.program_id(0) # maybe for output tile indices (?) Actually we need two dimensions: program_id(0) for i tile, 1 for j tile, 2 for pd.

   # dimension calculations:
   pid_i = tl.program_id(0)
   pid_j = tl.program_id(1)
   pid_pd = tl.program_id(2)

   # Compute i start, j start
   i_start = pid_i * BLOCK_M
   j_start = pid_j * BLOCK_N
   pd = pid_pd

   # Mask for out-of-range
   # Create range for BLOCK_M and BLOCK_N
   offs_i = i_start + tl.arange(0, BLOCK_M)
   offs_j = j_start + tl.arange(0, BLOCK_N)

   # Offsets for k block loop
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) # accumulate in fp32 for better precision

   for k in range(0, tl.cdiv(N, BLOCK_K)):
       k_start = k * BLOCK_K
       offs_k = k_start + tl.arange(0, BLOCK_K)

       # mask for out-of-range for i and k and j and k
       mask_i = offs_i < N
       mask_j = offs_j < N
       mask_k = offs_k < N

       # Load left: shape (BLOCK_M, BLOCK_K)
       left_ptrs = left_ptr + pd * stride_left_pd + offs_i[:,None] * stride_left_i + offs_k[None,:] * stride_left_k
       left = tl.load(left_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0).to(tl.float32) # upcast

       # Load right: shape (BLOCK_N, BLOCK_K)
       right_ptrs = right_ptr + pd * stride_right_pd + offs_j[:,None] * stride_right_i + offs_k[None,:] * stride_right_k
       right = tl.load(right_ptrs, mask=mask_j[:,None] & mask_k[None,:], other=0.0).to(tl.float32)

       # Multiply-accumulate: left (M,K) * right (N,K) transposed (K,N)
       # Use tl.dot with trans_b=True, ensure shapes (M,K) and (N,K)
       acc += tl.dot(left, right, trans_b=True)

   # Store results
   out_ptrs = out_ptr + pd * stride_out_pd + offs_i[:,None] * stride_out_i + offs_j[None,:] * stride_out_j
   mask_out = mask_i[:,None] & mask_j[None,:]
   tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)

"""
Outgoing Triangle Multiplicative Update (TriMul) implemented with a fused
Triton kernel.

The heavy N³ reduction:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

is performed as a batched matrix multiplication L @ Rᵀ where each
(b,h) pair is treated as an independent batch.  The kernel tiles the
(i,j) output plane (BLOCK_M × BLOCK_N) and accumulates over k in
BLOCK_K‑sized chunks, using Tensor Cores (fp16) and fp32 accumulation.
All other linear projections, gating and LayerNorms are done in
PyTorch (fp16 where possible).  The final result is returned as
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,               # fp16, shape [PD, N, N]
    right_ptr,              # fp16, shape [PD, N, N]
    out_ptr,                # fp16, shape [PD, N, N]
    PD,                     # total number of (batch * hidden) slices
    N,                      # sequence length
    stride_left_pd, stride_left_i, stride_left_k,
    stride_right_pd, stride_right_i, stride_right_k,
    stride_out_pd, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # tile row (i)
    pid_j = tl.program_id(1)          # tile col (j)
    pid_pd = tl.program_id(2)         # (batch * hidden) slice

    # start offsets for this tile
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # range of indices within the tile
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in blocks
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---------- load left[i, k] ----------
        # shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + pid_pd * stride_left_pd
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # ---------- load right[j, k] ----------
        # shape (BLOCK_N, BLOCK_K)
        right_ptrs = (
            right_ptr
            + pid_pd * stride_right_pd
            + offs_j[:, None] * stride_right_i
            + offs_k[None, :] * stride_right_k
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # matrix multiply: left (M,K) * rightᵀ (K,N) → (M,N)
        # trans_b=True treats `right` as (K,N)
        partial = tl.dot(left, right, trans_b=True)   # fp16 result
        acc += partial.to(tl.float32)

    # ---------- store result ----------
    out_ptrs = (
        out_ptr
        + pid_pd * stride_out_pd
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict with all model parameters
            - config: dict with at least "dim" and "hidden_dim" and optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ----------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ----------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ----------------------------------------------------
    # 2) Cast to fp16 for the heavy part
    # ----------------------------------------------------
    x_h = x.to(torch.float16)

    # ----------------------------------------------------
    # 3) Fuse projection + gate linear layers (fp16)
    # ----------------------------------------------------
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    left_comb = F.linear(x_h, left_comb_weight)   # [..., 2*H]
    right_comb = F.linear(x_h, right_comb_weight)  # [..., 2*H]

    left_proj, left_gate_pre = torch.split(left_comb, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_comb, hidden_dim, dim=-1)

    # ----------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # ----------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ----------------------------------------------------
    # 5) Gating (sigmoid, fp16)
    # ----------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate      # [B, N, N, H] fp16
    right = right_proj * right_gate   # [B, N, N, H] fp16

    # ----------------------------------------------------
    # 6) Output gate (fp16)
    # ----------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate_pre = F.linear(x_h, out_gate_weight)   # [B, N, N, H] fp16
    out_gate = torch.sigmoid(out_gate_pre)          # fp16

    # ----------------------------------------------------
    # 7) Batched triangle multiplication via Triton
    # ----------------------------------------------------
    # reshape to (PD, N, N) where PD = B * hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left.shape
    PD = B * H

    left_flat = left.view(PD, N, N)
    right_flat = right.view(PD, N, N)

    # strides for the flattened tensors
    stride_pd = left_flat.stride(0)
    stride_i = left_flat.stride(1)
    stride_k = left_flat.stride(2)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # block sizes (tuned for the B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        PD,
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        PD,
        N,
        stride_left_pd=stride_pd,
        stride_left_i=stride_i,
        stride_left_k=stride_k,
        stride_right_pd=stride_pd,
        stride_right_i=stride_i,
        stride_right_k=stride_k,
        stride_out_pd=stride_pd,
        stride_out_i=stride_i,
        stride_out_j=stride_k,   # last dimension stride is 1
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ----------------------------------------------------
    # 8) Final LayerNorm, gating and projection back to original dim
    # ----------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate.to(torch.float32)   # apply output gate
    output = F.linear(out_norm, weights["to_out.weight"])  # final projection, fp32

    return output
