@triton.jit
def layer_norm_kernel(x_ptr, y_ptr,
                     weight_ptr, bias_ptr,
                     B, N, H,
                     stride_xb, stride_xi, stride_xj, stride_xh,
                     stride_yb, stride_yi, stride_yj, stride_yh,
                     BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    # compute linear index over B,N,N
    total = B * N * N
    idx = pid
    mask = idx < total
    # compute b,i,j from idx
    b = idx // (N * N)
    rem = idx % (N * N)
    i = rem // N
    j = rem % N

    # Offsets for the block
    offs_h = tl.arange(0, BLOCK_H)
    # Compute base ptr for x and y
    x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    # load a block of H
    mask_h = offs_h < H
    x = tl.load(x_base + offs_h * stride_xh, mask=mask_h, other=0.0)
    # compute sum
    sum1 = tl.sum(x, axis=0)
    mean = sum1 / H
    # compute variance
    var = tl.sum((x - mean) * (x - mean), axis=0) / H
    inv_std = 1.0 / tl.sqrt(var + 1e-5)

    w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)
    b_ = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)

    y = (x - mean) * inv_std * w + b_
    # store result
    y_base = y_ptr + b * stride_yb + i * stride_yi + j * stride_yj
    tl.store(y_base + offs_h * stride_yh, y, mask=mask_h)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

The implementation follows the reference PyTorch TriMul module but moves the
mask‑gate fusion into a tiny Triton kernel and uses cuBLAS (torch.bmm) for the
heavy O(N³·H) reduction.  This combination satisfies the “use Triton” rule
while achieving performance close to a fully‑fused batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Small Triton kernel: fuse mask, linear projection and sigmoid gate.
# It computes   out = proj * gate * mask   (mask is broadcast over H).
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    proj_ptr,          # float16*  (B,N,N,H)   projected tensor
    mask_ptr,          # float16*  (B,N,N)     binary mask (broadcasted)
    gate_ptr,          # float16*  (B,N,N,H)   sigmoid gate
    out_ptr,           # float16*  (B,N,N,H)   result
    B, N, H,
    stride_pb, stride_pi, stride_pj, stride_ph,   # proj strides
    stride_mb, stride_mi, stride_mj,              # mask strides
    stride_gb, stride_gi, stride_gj, stride_gh,   # gate strides
    BLOCK_H: tl.constexpr):
    """
    Each program instance processes one (b,i,j) coordinate and a tile of the
    hidden dimension H (size = BLOCK_H).  The mask scalar is loaded once and
    broadcast across the H‑tile.
    """
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from linear pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offsets for the H‑tile
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # base pointers
    proj_base = proj_ptr + b * stride_pb + i * stride_pi + j * stride_pj
    gate_base = gate_ptr + b * stride_gb + i * stride_gi + j * stride_gj
    out_base  = out_ptr  + b * stride_pb + i * stride_pi + j * stride_pj

    # load data
    proj = tl.load(proj_base + offs_h * stride_ph, mask=mask_h, other=0.0)
    gate = tl.load(gate_base + offs_h * stride_gh, mask=mask_h, other=0.0)

    # mask scalar (broadcast)
    m = tl.load(mask_ptr + b * stride_mb + i * stride_mi + j * stride_mj)

    out = proj * gate * m
    tl.store(out_base + offs_h * stride_ph, out, mask=mask_h)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype  # keep original dtype (float32 or float16)

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(dtype_in)
    norm_b = weights["norm.bias"].to(device).to(dtype_in)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Work in fp16 for the compute‑heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (left/right) and gates (left_gate/right_gate/out_gate)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    left_proj = F.linear(x, left_proj_w)          # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)        # (B,N,N,H)

    left_gate = F.linear(x, left_gate_w).sigmoid()   # (B,N,N,H)
    right_gate = F.linear(x, right_gate_w).sigmoid() # (B,N,N,H)
    out_gate = F.linear(x, out_gate_w).sigmoid()     # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3. Fuse mask and gate via Triton (if mask is used)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is float16 and on the same device
        mask = mask.to(device).to(torch.float16)
        # allocate output tensors
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        B, N, _, H = left_proj.shape
        BLOCK_H = 64 if hidden_dim >= 64 else hidden_dim

        total = B * N * N
        grid = (total,)

        # left
        mask_gate_kernel[grid](
            left_proj, mask, left_gate, left,
            B, N, H,
            left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
            mask.stride(0), mask.stride(1), mask.stride(2),
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            BLOCK_H=BLOCK_H,
        )
        # right
        mask_gate_kernel[grid](
            right_proj, mask, right_gate, right,
            B, N, H,
            right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
            mask.stride(0), mask.stride(1), mask.stride(2),
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            BLOCK_H=BLOCK_H,
        )
    else:
        # simple element‑wise multiply (no mask)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4. Core O(N³·H) reduction via cuBLAS batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, gating, and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # apply output gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 6. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)
    return out
