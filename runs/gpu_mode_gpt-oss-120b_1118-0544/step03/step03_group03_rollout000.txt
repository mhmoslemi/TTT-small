"""
Outgoing Triangle Multiplicative Update (TriMul) – fast forward pass.

Algorithm
---------
1. Layer‑norm the input (float32 → fp16).
2. Fuse each pair of projection+gate linear layers:
     left_proj / left_gate   = Linear(x, [W_left_proj ; W_left_gate])
     right_proj / right_gate = Linear(x, [W_right_proj ; W_right_gate])
3. Apply sigmoid to the gate pre‑activations.
4. If a mask is present, fuse “mask * projection * gate” for left and right
   using a tiny Triton kernel (element‑wise, no synchronization).
   When no mask is given the element‑wise product is performed by PyTorch.
5. Compute the outgoing triangle update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply: for each (b,h) we compute
        L(b,h) @ R(b,h).T   (L,R ∈ ℝ^{N×N}).
   The batch size is B·hidden_dim, so we use a single torch.bmm call
   (cuBLAS/Tensor‑Cores).
6. Layer‑norm the result, apply the output gate and a final linear projection.
7. Cast the final tensor back to float32 (single cheap conversion).

Only the mask‑gate fusion is performed in Triton, satisfying the requirement
to contain a custom kernel while keeping the heavy N³ reduction in highly
optimized cuBLAS.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,            # fp16 * (B*N*N*H)
    right_ptr,           # fp16 * (B*N*N*H)
    left_gate_ptr,       # fp16 * (B*N*N*H)
    right_gate_ptr,      # fp16 * (B*N*N*H)
    mask_ptr,            # fp16 * (B*N*N)
    H: tl.constexpr,     # hidden dimension (known at compile time)
    N_ELEM: tl.constexpr,# total number of elements = B*N*N*H
    BLOCK: tl.constexpr, # elements per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM

    # Load data (fp16)
    left       = tl.load(left_ptr + offs,       mask=mask)
    right      = tl.load(right_ptr + offs,      mask=mask)
    left_gate  = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs,mask=mask)

    # broadcast mask: each H values share the same mask element
    mask_idx = offs // H               # integer division
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask)

    # fused computation
    left_out  = left  * mask_val * left_gate
    right_out = right * mask_val * right_gate

    # Store results in‑place
    tl.store(left_ptr + offs,  left_out,  mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model weights (see skeleton)
            - config: dict with at least "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------
    # 1) Input LayerNorm (float32 → fp16)
    # ----------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32
    x = x.to(dtype)  # fp16 for the heavy path

    # ----------------------------------------------------
    # 2) Fused projection + gate (fp16)
    # ----------------------------------------------------
    # concatenate projection and gate weights for left/right streams
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(dtype).to(device)  # (2*H, dim)

    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(dtype).to(device)  # (2*H, dim)

    # Linear -> [B, N, N, 2*H]
    left_comb = F.linear(x, left_comb_weight)   # fp16
    right_comb = F.linear(x, right_comb_weight) # fp16

    # split into projection and gate pre‑activations
    left_proj, left_gate_pre = torch.split(left_comb, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_comb, hidden_dim, dim=-1)

    # sigmoid gates
    left_gate = torch.sigmoid(left_gate_pre)   # fp16
    right_gate = torch.sigmoid(right_gate_pre) # fp16

    # ----------------------------------------------------
    # 3) Apply mask & gate (fused Triton kernel if mask present)
    # ----------------------------------------------------
    if not nomask and mask is not None:
        # ensure contiguity & correct dtype
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask.to(dtype).contiguous()

        total_elems = left_proj.numel()  # B * N * N * H
        BLOCK = 65536
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        _mask_gate_fuse_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            H=hidden_dim,
            N_ELEM=total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # After the kernel, left_proj/right_proj already contain the
        # masked & gated values.  Gates are no longer needed.
        left = left_proj
        right = right_proj
    else:
        # No mask: simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ----------------------------------------------------
    # 4) Output gate (fp16)
    # ----------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(dtype).to(device)
    out_gate_pre = F.linear(x, out_gate_weight)  # fp16
    out_gate = torch.sigmoid(out_gate_pre)       # fp16

    # ----------------------------------------------------
    # 5) Batched matrix multiplication: out = left @ rightᵀ
    #    Shapes: left/right  [B, N, N, H]  →  (B*H, N, N)
    # ----------------------------------------------------
    B, N, _, H = left.shape
    # bring H to the batch dimension
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # (B*H, N, N)

    # Batched GEMM – uses cuBLAS/Tensor Cores (fp16)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ----------------------------------------------------
    # 6) Final LayerNorm, gating and output projection (fp16)
    # ----------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype).to(device),
        bias=weights["to_out_norm.bias"].to(dtype).to(device),
        eps=1e-5,
    )

    out = out * out_gate  # broadcasted element‑wise multiply (fp16)

    out = F.linear(
        out,
        weights["to_out.weight"].to(dtype).to(device),
        bias=None,
    )  # fp16, shape [B, N, N, dim]

    # ----------------------------------------------------
    # 7) Cast to float32 for the final output
    # ----------------------------------------------------
    return out.to(torch.float32)
