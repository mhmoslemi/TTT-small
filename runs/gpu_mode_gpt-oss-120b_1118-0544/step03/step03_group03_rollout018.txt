```python
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # compute pointer offsets
        a_ptrs = a_ptr + batch_id * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_id * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load a and b with masking
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        acc += tl.dot(a, b).to(tl.float32)

    # store result
    c_ptrs = c_ptr + batch_id * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    # mask for store
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)
```

```python
def batched_matmul_triton(left, right):
    # left,right shape [B, N, N, H] fp16
    B, N, _, H = left.shape
    # permute to B, H, N, N -> shape (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # output
    out_flat = torch.empty_like(left_flat)  # fp16
    M = N
    K = N
    N_ = N

    # define strides for row-major matrices
    stride_am = K
    stride_ak = 1
    stride_bk = N_
    stride_bn = 1
    stride_cm = N_
    stride_cn = 1

    batch_stride_a = M * K
    batch_stride_b = K * N_
    batch_stride_c = M * N_

    # grid dims
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # compute grid size
    num_batch = B * H
    grid = (num_batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N_ + BLOCK_N - 1) // BLOCK_N)

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M, N_, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
```

```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', False)

    # layernorm
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights['norm.weight'],
                     bias=weights['norm.bias'],
                     eps=1e-5)

    # cast to half
    x_h = x.to(torch.float16)

    # linear projections
    left_proj = F.linear(x_h, weights['left_proj.weight'].to(torch.float16))
    right_proj = F.linear(x_h, weights['right_proj.weight'].to(torch.float16))

    # gates pre linear
    left_gate_pre = F.linear(x_h, weights['left_gate.weight'].to(torch.float16))
    right_gate_pre = F.linear(x_h, weights['right_gate.weight'].to(torch.float16))
    out_gate_pre = F.linear(x_h, weights['out_gate.weight'].to(torch.float16))

    # mask if needed
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # activations
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # batched matmul via Triton
    out_h = batched_matmul_triton(left, right)  # shape B,N,N,hidden_dim

    # convert to fp32
    out_fp32 = out_h.to(torch.float32)

    # final layernorm and out_gate
    out = F.layer_norm(out_fp32,
                       normalized_shape=(hidden_dim,),
                       weight=weights['to_out_norm.weight'],
                       bias=weights['to_out_norm.bias'],
                       eps=1e-5)

    out = out * out_gate.to(torch.float32)

    # final projection
    out = F.linear(out, weights['to_out.weight'])

    return out
```

```python
def custom_kernel(data):
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # Cast to half precision for heavy ops
    x_h = x.to(torch.float16)

    # Linear projections (fp16)
    left_proj = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Gates pre-activations
    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].to(torch.float16))
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].to(torch.float16))
    out_gate_pre = F.linear(x_h, weights["out_gate.weight"].to(torch.float16))

    # Apply mask if present
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Activations
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Batched matmul via Triton (left @ right^T)
    out_h = _batched_matmul_triton(left, right)

    # Convert to fp32 for subsequent ops
    out_fp32 = out_h.to(torch.float32)

    # Final LayerNorm over hidden dimension
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply output gate
    out = out * out_gate.to(torch.float32)

    # Final projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out
```

```python
def _batched_matmul_triton(left, right):
    # left,right shape [B,N,N,H] fp16
    B, N, _, H = left.shape
    # reshape
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # right need transpose on the inner dim
    right_T = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    M = N
    K = N
    N_ = N

    stride_am = left_flat.stride(1)  # = K
    stride_ak = left_flat.stride(2)  # = 1
    stride_bk = right_T.stride(1)    # = N
    stride_bn = right_T.stride(2)    # = 1
    stride_cm = out_flat.stride(1)   # = N
    stride_cn = out_flat.stride(2)   # = 1

    batch_stride_a = left_flat.stride(0)  # M*K
    batch_stride_b = right_T.stride(0)    # K*N
    batch_stride_c = out_flat.stride(0)   # M*N

    # block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N_ + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_T,
        c_ptr=out_flat,
        M=M,
        N=N_,
        K=K,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        batch_stride_a=batch_stride_a,
        batch_stride_b=batch_stride_b,
        batch_stride_c=batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
```

Now define kernel:

```python
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start offsets for the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator with float32 for higher precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Offset pointers for batch
    a_batch_ptr = a_ptr + batch_id * batch_stride_a
    b_batch_ptr = b_ptr + batch_id * batch_stride_b
    c_batch_ptr = c_ptr + batch_id * batch_stride_c

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A and B tiles
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        # FP16 dot; accumulate in FP32
        acc += tl.dot(a, b).to(tl.float32)

    # Write back results as FP16
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)
```

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton accelerated core.

The heavy part of TriMul is the N³ reduction:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
for every batch b and hidden head h.  This is equivalent to a batched
matrix multiplication per head:
    out[b,h] = left_flat[b,h] @ right_flat[b,h].T
where left_flat and right_flat are (N×N) matrices.  We execute this
batched GEMM with a custom Triton kernel in FP16 (Tensor‑Core friendly)
and accumulate in FP32 for numerical stability.  All other operations
(LayerNorm, linear projections, sigmoid gates) are done in PyTorch
(half‑precision where possible).  The final output is cast back to FP32
and passed through the final LayerNorm, output‑gate and projection.

This implementation satisfies the required API:
    custom_kernel(data) -> torch.Tensor [B, N, N, dim] (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ B   (A: M×K, B: K×N) for many independent batches."""
    batch_id = tl.program_id(0)          # batch index = B * hidden_dim
    pid_m = tl.program_id(1)             # tile row
    pid_n = tl.program_id(2)             # tile column

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the beginning of this batch
    a_batch = a_ptr + batch_id * batch_stride_a
    b_batch = b_ptr + batch_id * batch_stride_b
    c_batch = c_ptr + batch_id * batch_stride_c

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K sized chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Compute pointer offsets
        a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        # Load tiles (masked for edges)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        # FP16 multiply‑accumulate into FP32
        acc += tl.dot(a, b).to(tl.float32)

    # Write back result as FP16
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the core TriMul reduction using a Triton batched GEMM.

    Args:
        left:  Tensor of shape [B, N, N, H] (FP16)
        right: Tensor of shape [B, N, N, H] (FP16)

    Returns:
        out: Tensor of shape [B, N, N, H] (FP16)
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # right needs to be transposed on the inner dimension (k)
    right_T = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    M = N
    K = N
    N_ = N

    # Row‑major strides (contiguous tensors)
    stride_am = left_flat.stride(1)          # = K
    stride_ak = left_flat.stride(2)          # = 1
    stride_bk = right_T.stride(1)            # = N
    stride_bn = right_T.stride(2)            # = 1
    stride_cm = out_flat.stride(1)           # = N
    stride_cn = out_flat.stride(2)           # = 1

    batch_stride_a = left_flat.stride(0)      # M*K
    batch_stride_b = right_T.stride(0)        # K*N
    batch_stride_c = out_flat.stride(0)       # M*N

    # Block sizes – tuned for FP16 Tensor Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N_ + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_T,
        c_ptr=out_flat,
        M=M,
        N=N_,
        K=K,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        batch_stride_a=batch_stride_a,
        batch_stride_b=batch_stride_b,
        batch_stride_c=batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(data):
    """
    Outgoing TriMul forward pass with Triton‑accelerated N³ reduction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model weights (all tensors on the same device)
            - config: dict with at least "dim" and "hidden_dim";
                      optional "nomask" boolean.

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # -----------------------------------------------------------------
    # 2) Cast to FP16 for the heavy part
    # -----------------------------------------------------------------
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Linear projections + gates (FP16)
    # -----------------------------------------------------------------
    left_proj = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].to(torch.float16))
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].to(torch.float16))
    out_gate_pre = F.linear(x_h, weights["out_gate.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # 4) Optional masking (broadcast over hidden dimension)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # -----------------------------------------------------------------
    # 5) Gating (sigmoid, FP16)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    left = left_proj * left_gate          # [B, N, N, H] FP16
    right = right_proj * right_gate       # [B, N, N, H] FP16

    # -----------------------------------------------------------------
    # 6) Core N³ reduction via Triton batched GEMM
    # -----------------------------------------------------------------
    out_h = _batched_matmul_triton(left, right)   # [B, N, N, H] FP16

    # -----------------------------------------------------------------
    # 7) Cast back to FP32 for the final LayerNorm / projection
    # -----------------------------------------------------------------
    out_fp32 = out_h.to(torch.float32)

    # -----------------------------------------------------------------
    # 8) Final LayerNorm over hidden dimension
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 9) Apply output gate and final projection (float32)
    # -----------------------------------------------------------------
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
