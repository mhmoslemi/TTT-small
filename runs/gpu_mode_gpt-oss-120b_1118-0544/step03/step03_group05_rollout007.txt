Implementation details for kernel:

@triton.jit
def tri_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    # Compute indices
    pid = tl.program_id(0)
    num_block_m = tl.cdiv(N, BLOCK_M)
    num_block_n = tl.cdiv(N, BLOCK_N)
    batch_mn = num_block_m * num_block_n
    batch_id = pid // batch_mn
    block_id = pid % batch_mn
    block_m = block_id // num_block_n
    block_n = block_id % num_block_n

    # Offsets for rows, cols
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Manage masks
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointer offsets
    a_ptrs = a_ptr + batch_id * stride_a_batch + offs_m[:, None] * stride_a_row + (tl.arange(0, BLOCK_K))[None, :] * stride_a_col
    # We will compute k loop later.

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load A tile
        a = tl.load(
            a_ptr + batch_id * stride_a_batch + offs_m[:, None] * stride_a_row + k_offset[None, :] * stride_a_col,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load B^T tile (B: batch, K, N) where K = original col dimension, N = original row dimension
        b = tl.load(
            b_ptr + batch_id * stride_b_batch + k_offset[:, None] * stride_b_k + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # Multiply accumulate
        acc += tl.dot(a, b)

    # Cast and store
    c = acc.to(tl.float16)

    # Store
    tl.store(
        c_ptr + batch_id * stride_c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col,
        c,
        mask=mask_m[:, None] & mask_n[None, :]
    )

import torch
import triton
import triton.language as tl

@triton.jit
def tri_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # number of tiles in m and n dimensions
    num_block_m = (N + BLOCK_M - 1) // BLOCK_M
    num_block_n = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_batch = num_block_m * num_block_n
    batch_id = pid // tiles_per_batch
    tile_id = pid % tiles_per_batch
    block_m = tile_id // num_block_n
    block_n = tile_id % num_block_n

    # offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iteration over k dimension
    num_block_k = (N + BLOCK_K - 1) // BLOCK_K
    for k in range(num_block_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load a tile
        a = tl.load(
            a_ptr + batch_id * stride_a_batch
                  + offs_m[:, None] * stride_a_row
                  + k_offset[None, :] * stride_a_col,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # load b tile (transposed view)
        b = tl.load(
            b_ptr + batch_id * stride_b_batch
                  + k_offset[:, None] * stride_b_k
                  + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # accumulate
        acc += tl.dot(a, b)

    # write out
    c = acc.to(tl.float16)
    tl.store(
        c_ptr + batch_id * stride_c_batch
               + offs_m[:, None] * stride_c_row
               + offs_n[None, :] * stride_c_col,
        c,
        mask=mask_m[:, None] & mask_n[None, :]

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Layer norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for compute
    x = x.to(torch.float16)

    # Load weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # projections
    left = torch.nn.functional.linear(x, left_proj_w)  # B,N,N,H
    right = torch.nn.functional.linear(x, right_proj_w)

    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # B,N,N,1
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # free some temps
    del left_gate, right_gate, x

    # Prepare for Triton GEMM
    B, N, _, H = left.shape
    # permute to (B, H, N, N) and flatten batch*head
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat)

    # stride parameters
    a_batch_stride = left_mat.stride(0)
    a_row_stride = left_mat.stride(1)
    a_col_stride = left_mat.stride(2)

    b_batch_stride = right_mat.stride(0)
    # transposed view: treat K as original col, N as original row
    b_k_stride = right_mat.stride(2)
    b_n_stride = right_mat.stride(1)

    c_batch_stride = out_mat.stride(0)
    c_row_stride = out_mat.stride(1)
    c_col_stride = out_mat.stride(2)

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_block_m = (N + BLOCK_M - 1) // BLOCK_M
    num_block_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * H * num_block_m * num_block_n,)

    tri_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        a_batch_stride, a_row_stride, a_col_stride,
        b_batch_stride, b_k_stride, b_n_stride,
        c_batch_stride, c_row_stride, c_col_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2
    )

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # free left/right
    del left, right, left_mat, right_mat, out_mat

    # Output layer norm
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # apply output gate (elementwise mult) - can use torch
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, to_out_w)

    # cast back to original dtype
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the 4‑D input (B, N, N, C) using provided LN parameters.
2. Cast to fp16 and compute linear projections for the left/right paths and
   their additive gates (left_gate, right_gate, out_gate).
3. Apply optional binary mask, then fuse gates into the projected tensors.
4. Fuse the O(N³·H) reduction using a custom Triton kernel that performs a
   batched matrix multiply:   out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   The kernel processes each (batch, head) pair independently with a tiled
   GEMM (BLOCK_M×BLOCK_N tiles, BLOCK_K inner reduction), accumulating in
   float32 and storing the result in fp16.
5. Reshape back to (B, N, N, H), apply a second LayerNorm and the output gate.
6. Final linear projection back to the original channel dimension C.
7. Cast the result back to the input dtype (usually float32).

The core O(N³·H) work is performed entirely in Triton, satisfying the
requirement to implement a non‑trivial part of the operator in a kernel.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM for TriMul: out = A @ Bᵀ where A,B∈(B*H,N,N)."""
    pid = tl.program_id(0)

    # number of tiles per matrix dimension
    num_block_m = (N + BLOCK_M - 1) // BLOCK_M
    num_block_n = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_batch = num_block_m * num_block_n

    batch_id = pid // tiles_per_batch
    tile_id = pid % tiles_per_batch
    block_m = tile_id // num_block_n
    block_n = tile_id % num_block_n

    # compute offsets for the current tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the K dimension
    num_block_k = (N + BLOCK_K - 1) // BLOCK_K
    for k in range(num_block_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_row
            + k_offset[None, :] * stride_a_col,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Bᵀ tile: shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + batch_id * stride_b_batch
            + k_offset[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            mask: torch.Tensor of shape [B, N, N] (or None)
            weights: dict containing model weights
            config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Prepare fp16 weights
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)  # (H, C)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)  # (H, C)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, C)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)      # (C, H)

    # ------------------------------------------------------------------
    # 4) Linear projections (left/right)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Optional binary mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 6) Gate projections and sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()   # (B,N,N,H)
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7) Fuse gates into left/right
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # free unused temporaries early
    del left_gate, right_gate, x

    # ------------------------------------------------------------------
    # 8) Core O(N³·H) reduction via Triton batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) – contiguous for Triton
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # output buffer
    out_mat = torch.empty_like(left_mat)

    # strides for the kernel (in elements, not bytes)
    a_batch_stride = left_mat.stride(0)
    a_row_stride   = left_mat.stride(1)
    a_col_stride   = left_mat.stride(2)

    b_batch_stride = right_mat.stride(0)
    # For Bᵀ we swap the last two strides
    b_k_stride = right_mat.stride(2)   # original column stride → K dimension
    b_n_stride = right_mat.stride(1)   # original row stride    → N dimension

    c_batch_stride = out_mat.stride(0)
    c_row_stride   = out_mat.stride(1)
    c_col_stride   = out_mat.stride(2)

    # tile sizes – these work well for fp16 on RTX/A100‑class GPUs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_block_m = (N + BLOCK_M - 1) // BLOCK_M
    num_block_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * H * num_block_m * num_block_n,)

    _tri_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        a_batch_stride,
        a_row_stride,
        a_col_stride,
        b_batch_stride,
        b_k_stride,
        b_n_stride,
        c_batch_stride,
        c_row_stride,
        c_col_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # free large intermediates
    del left, right, left_mat, right_mat, out_mat

    # ------------------------------------------------------------------
    # 9) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,C)

    # ------------------------------------------------------------------
    # 12) Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
