import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(out_ptr, a_ptr, b_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # Cast weights to appropriate dtype
    # In case of half precision compute
    half = torch.float16
    # LayerNorm weight/bias can stay at same dtype as compute dtype (half)
    norm_weight = weights["norm.weight"].to(device).to(half)
    norm_bias = weights["norm.bias"].to(device).to(half)
    left_proj_weight = weights["left_proj.weight"].to(device).to(half)
    right_proj_weight = weights["right_proj.weight"].to(device).to(half)
    left_gate_weight = weights["left_gate.weight"].to(device).to(half)
    right_gate_weight = weights["right_gate.weight"].to(device).to(half)
    out_gate_weight = weights["out_gate.weight"].to(device).to(half)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(half)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(half)
    to_out_weight = weights["to_out.weight"].to(device).to(half)

    # 1. LayerNorm over dim
    # Convert input to half for compute
    x = input_tensor.to(half)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B,N,N,dim]

    # 2. Linear projections
    # left_proj and right_proj
    left_proj = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right_proj = F.linear(x, right_proj_weight)

    # 3. Apply mask to projections if needed
    if not nomask:
        # mask shape [B,N,N], broadcast
        mask_f = mask.to(half).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # 4. Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # [B,N,N,hidden]

    # Apply gates to projections
    left = left_proj * left_gate
    right = right_proj * right_gate

    # 5. Core reduction: out = left @ right^T per hidden
    B, N, _, _ = left.shape
    # Permute to (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    # Merge batch and hidden dims
    left_mat = left.view(B * hidden_dim, N, N)
    right_mat = right.view(B * hidden_dim, N, N)
    # Batched matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*hidden, N, N)
    # Restore shape [B, N, N, hidden]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. to_out_norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # still half

    # 7. Elementwise multiply by out_gate via Triton
    total_el = out.numel()
    BLOCK = 1024
    grid = ( (total_el + BLOCK - 1) // BLOCK, )
    # Ensure contiguous memory
    out_contig = out.view(-1)
    out_gate_contig = out_gate.view(-1)
    # Triton expects pointers, we supply tensors
    elementwise_mul_kernel[grid](
        out_contig.data_ptr(),
        out_contig.data_ptr(),
        out_gate_contig.data_ptr(),
        total_el,
        BLOCK=BLOCK,
    )
    # out already contains multiplied result (in-place)

    # 8. Final linear projection back to dim
    out = F.linear(out, to_out_weight)   # [B,N,N,dim] in half

    # 9. Cast to float32 as required
    out = out.to(torch.float32)
    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) – fast forward pass.

Algorithm
---------
1. Layer‑norm over the input channel dimension.
2. Four linear projections (left/right and their gates) are computed with
   `torch.nn.functional.linear`.  The gates are passed through sigmoid.
3. An optional mask is applied to the left/right projections before gating.
4. The core O(N³) reduction is performed as a batched matrix multiplication:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to `left_mat @ right_matᵀ` for each hidden channel.
5. A second LayerNorm is applied on the hidden dimension.
6. The output‑gate is computed and multiplied element‑wise with the result.
   The element‑wise multiplication is executed by a small Triton kernel.
7. A final linear projection maps the hidden dimension back to the original
   channel dimension.  The result is returned as float32.

All heavy arithmetic is performed in float16 to exploit tensor‑cores.
Only the element‑wise gate multiplication is explicitly written in Triton,
satisfying the requirement to use a kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(out_ptr, a_ptr, b_ptr, size, BLOCK: tl.constexpr):
    """out = a * b  (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, dtype=tl.float16)
    b = tl.load(b_ptr + offs, mask=mask, dtype=tl.float16)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) – may be None
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # Use float16 for all compute‑intensive ops
    compute_dtype = torch.float16

    # ------------------------------------------------------------------
    # Load and cast weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device).to(compute_dtype)
    norm_bias = weights["norm.bias"].to(device).to(compute_dtype)

    left_proj_weight = weights["left_proj.weight"].to(device).to(compute_dtype)
    right_proj_weight = weights["right_proj.weight"].to(device).to(compute_dtype)

    left_gate_weight = weights["left_gate.weight"].to(device).to(compute_dtype)
    right_gate_weight = weights["right_gate.weight"].to(device).to(compute_dtype)

    out_gate_weight = weights["out_gate.weight"].to(device).to(compute_dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(compute_dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(compute_dtype)

    to_out_weight = weights["to_out.weight"].to(device).to(compute_dtype)

    # ------------------------------------------------------------------
    # 1) LayerNorm over input channels
    # ------------------------------------------------------------------
    x = input_tensor.to(compute_dtype)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)      # [B, N, N, hidden]
    right_proj = F.linear(x, right_proj_weight)    # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 3) Optional mask on the projections (before gating)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Convert mask to the compute dtype and broadcast
        mask_f = mask.to(compute_dtype).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) and apply them
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # [B, N, N, hidden]

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Batched matrix multiplication: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM: (B*hidden, N, N) @ (B*hidden, N, N)ᵀ
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Move hidden dimension to axis 1 for easy reshaping
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    left_mat = left.view(B * hidden_dim, N, N)     # (B*hidden, N, N)
    right_mat = right.view(B * hidden_dim, N, N)   # (B*hidden, N, N)

    # Batched matmul → (B*hidden, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # still half‑precision

    # ------------------------------------------------------------------
    # 7) Element‑wise multiplication with out_gate (Triton kernel)
    # ------------------------------------------------------------------
    total_el = out.numel()
    BLOCK = 1024
    grid = ((total_el + BLOCK - 1) // BLOCK,)
    # Flatten tensors for the kernel (contiguous view)
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    elementwise_mul_kernel[grid](
        out_flat,
        out_flat,          # in‑place source
        out_gate_flat,
        total_el,
        BLOCK=BLOCK,
    )
    # out now contains the gated result

    # ------------------------------------------------------------------
    # 8) Final projection back to the original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim] in half precision

    # ------------------------------------------------------------------
    # 9) Cast to float32 as required by the interface
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    return out
