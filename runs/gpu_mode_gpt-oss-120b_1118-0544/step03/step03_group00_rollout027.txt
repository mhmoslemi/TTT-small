import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for mask and gates:
@triton.jit
def apply_gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                           total_elems, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # mask index = offs // hidden_dim (since mask doesn't have hidden_dim)
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def apply_gate_and_mask(proj, gate, mask, block_size=1024):
    # proj, gate: (B,N,N,H); mask: (B,N,N)
    B,N,_ ,H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = ( (total + block_size - 1) // block_size, )
    apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out

# Triton kernel for batched GEMM (A @ B^T)
@triton.jit
def batched_gemm_kernel(
        a_ptr, b_ptr, c_ptr,
        batch, M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(2)  # batch index
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = k_offset + offs_k
        k_valid = cur_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & k_valid[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B tile (N x K)  (we will transpose later)
        b_ptrs = b_batch + (offs_n[:, None] * stride_b_n) + (cur_k[None, :] * stride_b_k)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & k_valid[None, :], other=0.0)
        b = b.to(tl.float32)

        # transpose B tile to (K x N)
        b_T = b.t()

        acc += tl.dot(a, b_T)

    # write back to C
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Implements the AlphaFold3 TriMul operation using fused Triton kernels
    for mask/gate application and a batched GEMM (A @ Bᵀ) over the
    sequence dimension.  FP16 arithmetic is used for the heavy compute
    path; finally the result is cast back to the input dtype.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------
    # 1. LayerNorm on input
    # ------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to FP16 for compute
    x = x.to(torch.float16)

    # ------------------------------------------------------------
    # 2. Linear projections (FP16 weights)
    # ------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    left_proj = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------
    # 3. Apply gate and optional mask (fused via Triton)
    # ------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------
    # 4. Core reduction: out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM: (B*H, N, N) @ (B*H, N, N)^T
    # ------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) with contiguous memory per batch*head
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        N,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------
    # 5. Output LayerNorm, out_gate and final projection
    # ------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # shape (B,N,N,dim)

    # Cast back to original dtype if needed
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

    out_ptr,           # (total_elems,)   FP16
    total_elems,       # int
    hidden_dim,        # int (size of channel dimension)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # mask index = element // hidden_dim   (broadcast over channel)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """proj, gate: (B,N,N,H)   mask: (B,N,N)   →   proj*gate*mask (broadcast H)"""
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = ((total + block_size - 1) // block_size,)

    apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# 2) Batched GEMM kernel: C = A @ Bᵀ   where A,B : (batch, M, K)
#    Here M = N, K = N, and we treat each (batch, head) pair as one batch.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)          # which batch*head slice
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # start pointers for this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for cur_k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = cur_k * BLOCK_K
        k = k_offset + offs_k
        k_valid = k < K

        # A tile (M x K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m) + (k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & k_valid[None, :], other=0.0).to(tl.float32)

        # B tile (N x K) – will be transposed
        b_ptrs = b_batch + (offs_n[:, None] * stride_b_n) + (k[None, :] * stride_b_k)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & k_valid[None, :], other=0.0).to(tl.float32)

        # transpose B to (K x N) to perform dot(A, Bᵀ)
        b_T = b.t()
        acc += tl.dot(a, b_T)

    # write out C tile
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Input tuple: (input_tensor, mask, weights, config)
    Returns tensor of shape [B, seq_len, seq_len, dim] with the same dtype as input.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    ##################################################################
    # 1. Input LayerNorm (float32 for numerical stability)
    ##################################################################
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to FP16 for the heavy compute path
    x = x.to(torch.float16)

    ##################################################################
    # 2. Linear projections (FP16 weights)
    ##################################################################
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    left_proj = F.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    ##################################################################
    # 3. Fuse mask (if any) and gates
    ##################################################################
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    ##################################################################
    # 4. Core reduction: batched GEMM (A @ Bᵀ)
    #    Shape after reduction: (B, N, N, hidden_dim)
    ##################################################################
    B, N, _, H = left.shape
    # Flatten batch & head dimensions → (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    # Blocking parameters (tuned for typical B200 GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        N,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    ##################################################################
    # 5. Second LayerNorm, output gate and final projection
    ##################################################################
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # → (B, N, N, dim)

    # Cast back to the original dtype if needed
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
