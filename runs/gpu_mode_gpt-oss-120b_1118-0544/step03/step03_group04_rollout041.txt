def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Cast input to half
    x = input_tensor
    dtype = x.dtype
    # if dtype is torch.float64? Expect float32 or half. We'll cast to torch.float16.
    x = x.to(torch.float16)

    # LayerNorm
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"].to(torch.float16),
                     bias=weights["norm.bias"].to(torch.float16))

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # Mask
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # Move hidden dim to batch for GEMM
    B, N, _, H = left.shape
    # Permute to [B, H, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    batch = B * H

    left = left.view(batch, N, N)
    right = right.view(batch, N, N)

    # Allocate output
    out = torch.empty_like(left)

    # Triton kernel call
    _batched_gemm_kernel(left, right, out,
                         batch, N, N, N,
                         left.stride(0), left.stride(1), left.stride(2),
                         right.stride(0), right.stride(2), right.stride(1),   # swapped strides for transposed view
                         out.stride(0), out.stride(1), out.stride(2),
                         BLOCK_M=128, BLOCK_N=128, BLOCK_K=32)

    # Reshape output to [B, H, N, N] then permute back to [B,N,N,H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm layernorm (over hidden_dim)
    out = F.layer_norm(out, (H,), weight=weights["to_out_norm.weight"].to(torch.float16),
                       bias=weights["to_out_norm.bias"].to(torch.float16))

    # Apply out_gate
    out = out * out_gate

    # Final projection back to dim
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to original dtype
    out = out.to(dtype)

    return out

```
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # A tile
        a_ptrs = a_ptr + batch_id * stride_a_batch + (offs_m[:, None] * stride_a_row) + ((k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_a_col)
        mask_a = (offs_m[:, None] < M) & ((k_offset + tl.arange(0, BLOCK_K))[None, :] < K)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # B tile (transposed view)
        b_ptrs = b_ptr + batch_id * stride_b_batch + ((k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_b_row) + (offs_n[None, :] * stride_b_col)
        mask_b = ((k_offset + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # multiply-accumulate
        acc += tl.dot(a, b)
    # store result
    c_ptrs = c_ptr + batch_id * stride_c_batch + (offs_m[:, None] * stride_c_row) + (offs_n[None, :] * stride_c_col)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        off_k = k * BLOCK_K
        # A tile
        a_ptrs = a_ptr + batch_id * stride_a_batch \
                 + (offs_m[:, None] * stride_a_row) \
                 + ((off_k + tl.arange(0, BLOCK_K))[None, :] * stride_a_col)
        mask_a = (offs_m[:, None] < M) & ((off_k + tl.arange(0, BLOCK_K))[None, :] < K)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # B tile (treated as K x N by swapping strides)
        b_ptrs = b_ptr + batch_id * stride_b_batch \
                 + ((off_k + tl.arange(0, BLOCK_K))[:, None] * stride_b_row) \
                 + (offs_n[None, :] * stride_b_col)
        mask_b = ((off_k + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptr + batch_id * stride_c_batch \
             + (offs_m[:, None] * stride_c_row) \
             + (offs_n[None, :] * stride_c_col)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

def _batched_gemm_triton(A, B):
    """
    Batched half‑precision matrix multiplication C = A @ Bᵀ
    A, B: torch tensors of shape (batch, M, K) and (batch, K, N)
    Returns C of shape (batch, M, N) in torch.float16.
    """
    assert A.is_cuda and B.is_cuda
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # block sizes - chosen for good occupancy on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(2), B.stride(1),  # note swapped strides for transposed view
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C

def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float16/float32)
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys 'dim', 'hidden_dim', optionally 'nomask'

    Returns:
        torch.Tensor [B, N, N, C] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm on channel dimension (C)
    # ------------------------------------------------------------------
    x = input_tensor
    orig_dtype = x.dtype
    x = x.to(torch.float16)
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·H) reduction using a Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)
    # Rearrange to (B*H, N, N) to treat hidden dimension as batch
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    batch = B * H
    left = left.view(batch, N, N)
    right = right.view(batch, N, N)

    # Batched matmul: out = left @ rightᵀ
    out = _batched_gemm_triton(left, right)   # (batch, N, N) in fp16

    # Restore original 4‑D layout (B, N, N, H)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 7. Output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to original dtype
    out = out.to(orig_dtype)
    return out

custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy O(N³·H) reduction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32

        k_off = k * BLOCK_K

        a_ptrs = a_ptr + batch_id * stride_a_batch \
                 + (offs_m[:, None] * stride_a_row) \

    """
    Batched GEMM C = A @ Bᵀ with half‑precision inputs.
    A: (batch, M, K), B: (batch, K, N)   (here K == N)
    Returns C of shape (batch, M, N) in fp16.
    """
    assert A.is_cuda and B.is_cuda
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Tile sizes – empirically good for H100 (B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),          # A strides
        B.stride(0), B.stride(2), B.stride(1),          # B strides (row/col swapped for Bᵀ)
        C.stride(0), C.stride(1), C.stride(2),          # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, C] with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over channel dim C)
    # ------------------------------------------------------------------
    orig_dtype = input_tensor.dtype
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Apply optional mask
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid of linear projections of the normalized x)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·H) reduction via a Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # (B, N, N, H)
    # Merge hidden dim into the batch dimension
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)
    batch = B * H
    left = left.view(batch, N, N)                 # (batch, N, N)
    right = right.view(batch, N, N)               # (batch, N, N)

    # Batched GEMM: out = left @ rightᵀ
    out = _batched_gemm_triton(left, right)      # (batch, N, N), fp16

    # Restore original 4‑D layout (B, N, N, H)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 7. Output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to the original dtype
    out = out.to(orig_dtype)
    return out
