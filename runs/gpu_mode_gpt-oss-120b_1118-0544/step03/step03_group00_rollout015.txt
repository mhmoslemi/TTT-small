@triton.jit
def apply_gate_mask_two_kernel(
        left_proj_ptr, left_gate_ptr,
        right_proj_ptr, right_gate_ptr,
        mask_ptr,
        left_out_ptr, right_out_ptr,
        total_elems, hidden_dim,
        BLOCK_SIZE: tl.constexpr
    ):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    # mask for bounds
    mask_valid = offs < total_elems

    # compute mask index
    mask_idx = offs // hidden_dim

    # load
    lp = tl.load(left_proj_ptr + offs, mask=mask_valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask_valid, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask_valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask_valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    left_out = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs, left_out, mask=mask_valid)
    tl.store(right_out_ptr + offs, right_out, mask=mask_valid)
```

We will call this kernel when mask present. If mask not present or config.nomask, use simple multiplication: left = left_proj * left_gate; right = right_proj * right_gate (torch.mul). That's likely fast.

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    x = torch.nn.functional.layer_norm(input_tensor,
        (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast to fp16
    x = x.to(torch.float16)

    # Flatten x
    B, N, _, _ = x.shape  # Actually shape [B,N,N,dim]
    x_flat = x.view(-1, dim)  # shape (B*N*N, dim)

    # Build combined weights
    left_comb_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_comb_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    # Compute left and right combined linear
    left_comb = torch.nn.functional.linear(x_flat, left_comb_weight)  # (B*N*N, 2*hidden_dim)
    right_comb = torch.nn.functional.linear(x_flat, right_comb_weight)
    out_gate = torch.nn.functional.linear(x_flat, out_gate_weight)  # (B*N*N, hidden_dim)

    # Reshape
    left_comb = left_comb.view(B, N, N, 2*hidden_dim)
    right_comb = right_comb.view(B, N, N, 2*hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim).sigmoid()

    # split
    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # Apply mask and gating via Triton or simple mul
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        total_elems = left_proj.numel()
        BLOCK_SIZE = 1024  # can be 8192 as needed
        grid = (triton.cdiv(total_elems, BLOCK_SIZE),)
        apply_gate_mask_two_kernel[grid](
            left_proj, left_gate,
            right_proj, right_gate,
            mask_fp16,
            total_elems,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE
        )
    else:
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # Core reduction via batched matmul
    # left_proj, right_proj shape: (B,N,N,hidden_dim)
    # Permute for batch matmul
    left_mat = left_proj.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right_proj.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    # Multiply left * right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden_dim, N, N)

        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16)
    )
    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection
    # flatten to (B*N*N, hidden_dim)
    out_flat = out.view(-1, hidden_dim)
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out_flat, to_out_weight)  # (B*N*N, dim)
    out = out.view(B, N, N, dim)

    # Cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
TriMul (outgoing) forward pass optimized with Triton.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Cast to FP16 for all heavy compute.
3. Fuse the projection and gate linear layers:
   * left   = [proj, gate]  via a single linear (2·H × C) weight,
   * right  = [proj, gate]  via a single linear (2·H × C) weight,
   * out_gate       via a linear (H × C) weight.
   The results are split and the gate tensors are passed through sigmoid.
4. If a mask is provided (and not disabled) a Triton kernel fuses
   mask‑application and gating for both left and right tensors in‑place.
5. Core O(N³·H) reduction is a batched GEMM:
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   Implemented with torch.bmm on (B·H, N, N) matrices – this leverages cuBLAS
   Tensor‑cores and is faster than a pure‑Triton GEMM.
6. Apply a second LayerNorm on the hidden dimension, multiply by the out‑gate,
   and project back to the original channel dimension.
7. Cast the result back to the input dtype.

The implementation reduces the number of linear passes from five to three and
uses a custom Triton kernel to fuse mask and gate multiplications, resulting
in a fast forward pass suitable for the test configurations.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _apply_gate_mask_two_kernel(
    left_proj_ptr,   # fp16 * (B*N*N*H)
    left_gate_ptr,   # fp16 * (B*N*N*H)
    right_proj_ptr,  # fp16 * (B*N*N*H)
    right_gate_ptr,  # fp16 * (B*N*N*H)
    mask_ptr,        # fp16 * (B*N*N)
    total_elems,     # int32 scalar: B*N*N*H
    hidden_dim,      # int32 scalar: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # index into mask (broadcast over hidden dim)
    mask_idx = offs // hidden_dim

    # load values (out‑of‑bounds entries get neutral values)
    lp = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs,  mask=valid, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m  = tl.load(mask_ptr + mask_idx,    mask=valid, other=1.0)

    # fused element‑wise multiply
    lo = lp * lg * m
    ro = rp * rg * m

    # write back in‑place
    tl.store(left_proj_ptr + offs,  lo, mask=valid)
    tl.store(right_proj_ptr + offs, ro, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    B, N, _, _ = x.shape
    # flatten to (B·N·N, C) for linear layers
    x_flat = x.view(-1, dim)  # (B*N*N, C)

    # ------------------------------------------------------------------
    # 3. Fuse projection + gate linear layers
    # ------------------------------------------------------------------
    # left: [proj | gate]  -> (2·H, C)
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    # right: [proj | gate] -> (2·H, C)
    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)
    # out_gate weight -> (H, C)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    # compute in FP16
    left_comb = torch.nn.functional.linear(x_flat, left_comb_weight)   # (B*N*N, 2·H)
    right_comb = torch.nn.functional.linear(x_flat, right_comb_weight) # (B*N*N, 2·H)
    out_gate = torch.nn.functional.linear(x_flat, out_gate_weight)    # (B*N*N, H)

    # reshape
    left_comb = left_comb.view(B, N, N, 2 * hidden_dim)
    right_comb = right_comb.view(B, N, N, 2 * hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim).sigmoid()

    # split projection and gate, apply sigmoid on gates
    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask + gates (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        total = left_proj.numel()  # B·N·N·H
        BLOCK_SIZE = 1024
        grid = (triton.cdiv(total, BLOCK_SIZE),)
        _apply_gate_mask_two_kernel[grid](
            left_proj,
            left_gate,
            right_proj,
            right_gate,
            mask_fp16,
            total,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # after kernel left_proj/right_proj contain the gated & masked values
    else:
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Core reduction: batched GEMM (out = left @ rightᵀ) in FP16
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) for torch.bmm
    left_mat = left_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # matrix multiplication (each batch element corresponds to a hidden channel)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N)
    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    # final linear: (B·N·N, H) -> (B·N·N, C)
    out_flat = out.view(-1, hidden_dim)
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out_flat, to_out_weight)
    out = out.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 7. Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
