def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # local variables
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # Load weights
    norm_weight = weights["norm.weight"].to(device)
    norm_bias   = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # 1. LayerNorm over last dim (dim)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)   # keep dtype (float32)

    # Cast to fp16 for heavy compute
    x_fp16 = x.to(torch.float16)

    B, N, _, _ = x.shape

    # 2. Combined linear projection for left/right and gates
    combined_weight = torch.cat([left_proj_weight,
                                 left_gate_weight,
                                 right_proj_weight,
                                 right_gate_weight], dim=0)  # (4*hidden, dim)
    # Flatten spatial dims
    x_flat = x_fp16.view(-1, dim)  # (B*N*N, dim)
    combined = F.linear(x_flat, combined_weight)  # (B*N*N, 4*hidden)
    combined = combined.view(B, N, N, 4 * hidden_dim)
    left_proj, left_gate, right_proj, right_gate = torch.split(combined, hidden_dim, dim=-1)

    # 3. Apply mask (if needed) before gating (mask broadcasted)
    if not nomask:
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # 4. Gating
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate   # (B,N,N,hidden)
    right = right_proj * right_gate

    # 5. Batched matrix multiplication using torch.bmm
    # reshape to (B*hidden, N, N) where left matrix = left[i,k], right = right[j,k]
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    # compute out_mat = left_mat @ right_mat^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape to (B,N,N,hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden)

    # 6. Final LayerNorm over hidden channel
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    # 7. Compute out_gate and apply (elementwise) via Triton
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # (B,N,N,hidden)
    # Triton elementwise multiplication: out *= out_gate
    total = out.numel()
    BLOCK = 65536
    grid = lambda meta: ( (total + BLOCK - 1) // BLOCK, )
    elementwise_mul_kernel[grid](
        out,
        out_gate,
        out,
        total,
        BLOCK=BLOCK,
    )

    # 8. Final linear projection back to dim
    out = F.linear(out, to_out_weight)  # (B,N,N,dim) fp16
    out = out.to(torch.float32)
    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the reference PyTorch module while aggressively
fusing operations:
  * A single fused linear projection computes left/right values and their
    corresponding gates.
  * Optional masking is applied before gating.
  * The O(N³) reduction `out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]`
    is performed with a batched half‑precision GEMM (torch.bmm) after
    reshaping the tensors.
  * A small Triton kernel carries out the final element‑wise multiplication
    with the output gate.
All heavy arithmetic is carried out in fp16; the returned tensor is fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N, BLOCK: tl.constexpr):
    """c = a * b  (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # --------------------------------------------------------------
    # Load weights (cast to appropriate dtype)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias   = weights["norm.bias"].to(device)

    left_proj_weight  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight   = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the input channel dimension (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # Cast to fp16 for the heavy compute path
    x_fp16 = x.to(torch.float16)

    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections and their gates into one linear op
    # --------------------------------------------------------------
    # Combined weight shape: (4*hidden_dim, dim)
    combined_weight = torch.cat([
        left_proj_weight,
        left_gate_weight,
        right_proj_weight,
        right_gate_weight,
    ], dim=0)

    # Linear transform over the flattened pairwise dimension
    x_flat = x_fp16.view(-1, dim)                     # (B*N*N, dim)
    combined = F.linear(x_flat, combined_weight)     # (B*N*N, 4*hidden_dim)
    combined = combined.view(B, N, N, 4 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each: (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 3️⃣ Optional masking (applied to projections before gating)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: (B, N, N) -> (B, N, N, 1) and cast to fp16
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and combine with projections
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left  = left_proj * left_gate   # (B, N, N, hidden_dim)
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Batched GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a half‑precision batched matmul.
    # --------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N) where the inner dim is the summed axis (k)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # (B*hidden, N, N) @ (B*hidden, N, N)^T  ->  (B*hidden, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden_dim)  fp16

    # --------------------------------------------------------------
    # 7️⃣ Compute out_gate and apply it via a Triton kernel
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # (B, N, N, hidden_dim) fp16

    total_el = out.numel()
    BLOCK = 65536
    grid = ( (total_el + BLOCK - 1) // BLOCK, )
    # Triton expects flat contiguous tensors
    elementwise_mul_kernel[grid](
        out.view(-1),
        out_gate.view(-1),
        out.view(-1),
        total_el,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 8️⃣ Final projection back to the original dimension
    # --------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # (B, N, N, dim) fp16
    out = out.to(torch.float32)          # return float32 as required
    return out
