### Triton LayerNorm kernel

Goal: compute normalized version y = (x - mean) / sqrt(var + eps) * weight + bias.

Mean per row: sum_i x_i / C.

Variance per row: sum_i (x_i - mean)^2 / C.

Implementation pattern:

- Each program id corresponds to one row (i.e., the vector index). But we can map multiple rows to one block, i.e., block's threads load multiple entries.

Simplify: Use Triton's `tl.reduce` pattern. Usually recommended is to load a block of values for each row in parallel across threads using `tl.arange`. Since C <= 384, we can set BLOCK_SIZE = 64 or 128.

Better: let each program compute a row; each thread loads a subset of columns using loops.

Pseudo:

```
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    M, C, eps,
    stride_xm, stride_xc,
    stride_ym, stride_yc,
    stride_w, stride_b,
    BLOCK_C: tl.constexpr,
):
    row = tl.program_id(0)
    # compute pointer offsets
    x_ptrs = x_ptr + row * stride_xm + tl.arange(0, BLOCK_C) * stride_xc
    # compute mean
    sum_val = tl.zeros([1], dtype=tl.float32)
    for i in range(0, C, BLOCK_C):
        mask = i + tl.arange(0, BLOCK_C) < C
        x = tl.load(x_ptrs + i * stride_xc, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
    mean = sum_val / C

    # compute variance
    var_sum = tl.zeros([1], dtype=tl.float32)
    for i in range(0, C, BLOCK_C):
        mask = i + tl.arange(0, BLOCK_C) < C
        x = tl.load(x_ptrs + i * stride_xc, mask=mask, other=0.0).to(tl.float32)
        diff = x - mean
        var_sum += tl.sum(diff * diff, axis=0)
    var = var_sum / C
    inv_std = 1.0 / tl.sqrt(var + eps)

    # apply weight and bias
    for i in range(0, C, BLOCK_C):
        mask = i + tl.arange(0, BLOCK_C) < C
        x = tl.load(x_ptrs + i * stride_xc, mask=mask, other=0.0)
        w = tl.load(weight_ptr + i, mask=mask, other=0.0)
        b = tl.load(bias_ptr + i, mask=mask, other=0.0)
        y = (x - mean) * inv_std * w + b
        tl.store(y_ptr + row * stride_ym + i * stride_yc, y, mask=mask)
```

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using a handcrafted Triton kernel.
2. Cast to fp16 and apply the three linear projections (left, right, gates) with
   torch.nn.functional.linear (cuBLAS‑accelerated).
3. Optionally apply a binary mask to the left/right projections (element‑wise
   Triton kernel).
4. Fuse the left/right gates (sigmoid) into the projections (element‑wise
   Triton kernel).
5. Core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM using torch.bmm on tensors of shape
   (B·H, N, N).  This maps to cuBLAS Tensor‑Core kernels and is the main
   performance driver.
6. Layer‑norm over the hidden dimension H (Triton kernel) and apply the
   output gate (element‑wise Triton kernel).
7. Final linear projection back to the original channel dimension (cuBLAS).
8. Cast the result back to the input dtype.

All heavy memory‑bound element‑wise ops are fused into Triton kernels, while
the O(N³) matmul leverages cuBLAS for maximal throughput.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def _elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, total_elems,
                            BLOCK_SIZE: tl.constexpr):
    """Element‑wise multiply a * b → c (fp16, contiguous)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0))
    tl.store(c_ptr + offs, a * b, mask=mask)


def _elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton kernel for element‑wise multiplication."""
    assert a.shape == b.shape, "shapes must match"
    total = a.numel()
    c = torch.empty_like(a)
    BLOCK_SIZE = 1024
    grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    _elementwise_mul_kernel[grid](
        a, b, c, total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    return c


@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # input & output pointers
    weight_ptr, bias_ptr,       # LN parameters (fp16)
    M, C, eps,                  # M rows, C cols, epsilon
    stride_xm, stride_xc,       # strides for x (row, col)
    stride_ym, stride_yc,       # strides for y (row, col)
    stride_w, stride_b,         # strides for weight & bias (should be 1)
    BLOCK_C: tl.constexpr,      # tile size along C
):
    """Layer‑norm over the last dimension (C) for a 2‑D view (M × C)."""
    row = tl.program_id(0)                     # each program handles one row
    x_row_ptr = x_ptr + row * stride_xm
    y_row_ptr = y_ptr + row * stride_ym

    # ---------- mean ----------
    mean = tl.float32(0.0)
    for off in range(0, C, BLOCK_C):
        col = tl.arange(0, BLOCK_C)
        mask = (off + col) < C
        x = tl.load(x_row_ptr + (off + col) * stride_xc,
                    mask=mask, other=tl.float16(0.0))
        mean += tl.sum(x.to(tl.float32), axis=0)
    mean = mean / C

    # ---------- variance ----------
    var = tl.float32(0.0)
    for off in range(0, C, BLOCK_C):
        col = tl.arange(0, BLOCK_C)
        mask = (off + col) < C
        x = tl.load(x_row_ptr + (off + col) * stride_xc,
                    mask=mask, other=tl.float16(0.0))
        x_f = x.to(tl.float32)
        diff = x_f - mean
        var += tl.sum(diff * diff, axis=0)
    var = var / C
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ---------- affine ----------
    for off in range(0, C, BLOCK_C):
        col = tl.arange(0, BLOCK_C)
        mask = (off + col) < C
        x = tl.load(x_row_ptr + (off + col) * stride_xc,
                    mask=mask, other=tl.float16(0.0))
        w = tl.load(weight_ptr + (off + col) * stride_w,
                    mask=mask, other=tl.float16(0.0))
        b = tl.load(bias_ptr + (off + col) * stride_b,
                    mask=mask, other=tl.float16(0.0))

        x_f = x.to(tl.float32)
        w_f = w.to(tl.float32)
        b_f = b.to(tl.float32)

        y_f = (x_f - mean) * inv_std * w_f + b_f
        tl.store(y_row_ptr + (off + col) * stride_yc,
                 y_f.to(tl.float16), mask=mask)


def layernorm_triton(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Apply LayerNorm on the last dimension of a 2‑D tensor (M×C) using Triton.
    Input ``x`` is expected to be contiguous with shape (M, C) and dtype fp16.
    """
    assert x.dim() == 2 and weight.dim() == 1 and bias.dim() == 1
    M, C = x.shape
    y = torch.empty_like(x)

    stride_xm, stride_xc = x.stride()
    stride_ym, stride_yc = y.stride()
    stride_w = weight.stride(0)
    stride_b = bias.stride(0)

    # Heuristic: pick a tile size that divides typical C (<=384)
    BLOCK_C = 64

    grid = (M,)
    layernorm_kernel[grid](
        x, y,
        weight, bias,
        M, C, eps,
        stride_xm, stride_xc,
        stride_ym, stride_yc,
        stride_w, stride_b,
        BLOCK_C=BLOCK_C,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor, shape [B, N, N, C] (fp32 or fp16)
            mask: torch.Tensor, shape [B, N, N] (or None)
            weights: dict of model parameters
            config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as ``input_tensor``.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0) Device / dtype handling
    # ------------------------------------------------------------------
    device = input_tensor.device
    orig_dtype = input_tensor.dtype
    x = input_tensor.to(torch.float16)

    B, N, _, C = x.shape

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (Triton)
    # ------------------------------------------------------------------
    x_flat = x.view(-1, C)                # (B·N·N, C)
    norm_w = weights["norm.weight"].to(device).to(torch.float16)
    norm_b = weights["norm.bias"].to(device).to(torch.float16)
    x_norm = layernorm_triton(x_flat, norm_w, norm_b, eps=1e-5)
    x_norm = x_norm.view(B, N, N, C)      # (B, N, N, C)

    # ------------------------------------------------------------------
    # 2) Linear projections & gates (cuBLAS)
    # ------------------------------------------------------------------
    # prepare fp16 weight tensors
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)   # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16) # (H, C)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)   # (H, C)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16) # (H, C)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)    # (H, C)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)    # (H,)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)      # (C, H)

    # flatten for linear
    x_flat = x_norm.view(-1, C)           # (M, C), M = B·N·N

    left = torch.nn.functional.linear(x_flat, left_proj_w)      # (M, H)
    right = torch.nn.functional.linear(x_flat, right_proj_w)    # (M, H)

    left_gate = torch.nn.functional.linear(x_flat, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, out_gate_w).sigmoid()

    # reshape back
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3) Optional binary mask (element‑wise Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16, device=device).unsqueeze(-1)         # (B,N,N,1)
        mask_f = mask_f.expand(-1, -1, -1, hidden_dim)                       # broadcast
        left = _elementwise_mul(left, mask_f)
        right = _elementwise_mul(right, mask_f)

    # ------------------------------------------------------------------
    # 4) Fuse gates into projections (element‑wise Triton)
    # ------------------------------------------------------------------
    left = _elementwise_mul(left, left_gate)
    right = _elementwise_mul(right, right_gate)

    # ------------------------------------------------------------------
    # 5) Core O(N³·H) reduction via batched GEMM (cuBLAS)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)                # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)              # (B·H, N, N)

    # batched matmul: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N)

    # restore shape (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over hidden dimension (Triton)
    # ------------------------------------------------------------------
    out_flat = out.view(-1, hidden_dim)
    out_norm = layernorm_triton(out_flat, to_out_norm_w, to_out_norm_b, eps=1e-5)
    out_norm = out_norm.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7) Apply output gate (element‑wise Triton)
    # ------------------------------------------------------------------
    out = _elementwise_mul(out_norm, out_gate)

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original dim (cuBLAS)
    # ------------------------------------------------------------------
    out_flat = out.view(-1, hidden_dim)
    out = torch.nn.functional.linear(out_flat, to_out_w)   # (M, C)
    out = out.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
