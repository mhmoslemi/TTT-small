def triton_batched_matmul(A, B):
    # A, B shapes (batch, M, N)
    # assume both are contiguous in row-major.
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)
    # compute strides using .stride() (size in elements)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()
    # grid
    # (batch, ceil_div(M, BLOCK_M), ceil_div(N, BLOCK_N))
    # Use int divisions.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)
    batched_matmul_kernel[grid](
        A, B, C,

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy O(N³·H) reduction ∑ₖ left[i,k,h]·right[j,k,h] is performed as a
batched matrix multiplication A @ Bᵀ where A and B have shape (B·H, N, N).
A custom Triton kernel computes C = A @ B efficiently by tiling the
matrices (BLOCK_M×BLOCK_N) and accumulating over K‑tiles in FP32.
All other lightweight operations (LayerNorm, linear projections, gates)
are executed with PyTorch in half‑precision. The final result is cast back
to the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched matmul C = A @ B   (A: (B,M,K), B: (B,K,N))
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,      # pointers
    M, N, K,                  # sizes
    stride_az, stride_am, stride_ak,   # A: batch, row, col strides
    stride_bz, stride_bk, stride_bn,   # B: batch, row, col strides
    stride_cz, stride_cm, stride_cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute start pointers for this batch
    a_batch_ptr = a_ptr + batch_id * stride_az
    b_batch_ptr = b_ptr + batch_id * stride_bz
    c_batch_ptr = c_ptr + batch_id * stride_cz

    # mask for out‑of‑bounds rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers to the tiles of A and B
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        # load with masking
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # dot‑product accumulation
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor,
                         BLOCK_M=128, BLOCK_N=128, BLOCK_K=32) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A: (batch, M, K)   fp16, contiguous
    B: (batch, K, N)   fp16, contiguous
    Returns C: (batch, M, N)   fp16
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # strides in number of elements
    stride_az, stride_am, stride_ak = A.stride()
    stride_bz, stride_bk, stride_bn = B.stride()
    stride_cz, stride_cm, stride_cn = C.stride()

    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_az, stride_am, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C

# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] or None (binary 0/1)
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor [B, N, N, C] with same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------
    # 1. Input LayerNorm (float16 for speed)
    # --------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------
    # 2. Cast weight tensors to half‑precision
    # --------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------
    # 3. Linear projections (no bias) and optional masking
    # --------------------------------------------------
    # shape: (B, N, N, H)
    left  = F.linear(x, left_proj_w)
    right = F.linear(x, right_proj_w)

    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------
    # 4. Gate projections (sigmoid)
    # --------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # Apply gates to the projected tensors
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------
    # 5. Prepare tensors for batched matmul: left @ rightᵀ
    # --------------------------------------------------
    # Move hidden dimension to the batch axis
    # left: (B, N, N, H) -> (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    # rightᵀ: (B, N, N, H) -> (B, H, N, N) with the last two axes swapped
    right_t = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left.shape

    # Collapse batch and hidden dimensions: (B*H, N, N)
    left_flat = left.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)

    # --------------------------------------------------
    # 6. Core O(N³·H) reduction via Triton batched GEMM
    # --------------------------------------------------
    out_flat = triton_batched_matmul(left_flat, right_flat)  # (B*H, N, N), fp16

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------
    # 7. Hidden‑dim LayerNorm + output gate
    # --------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # --------------------------------------------------
    # 8. Final linear projection back to input dim
    # --------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to original dtype if needed
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile : (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B tile : (BLOCK_K, BLOCK_N)  (B is already transposed)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Update accumulator
        acc += tl.dot(a, b)

    # --------------------------------------------------
    # 5️⃣  Write back the result (cast to fp16)
    # --------------------------------------------------
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def triton_batched_matmul(A: torch.Tensor,
                         B: torch.Tensor,
                         BLOCK_M=128, BLOCK_N=128, BLOCK_K=32) -> torch.Tensor:
    """
    Batched matmul using the Triton kernel above.
    A: (batch, M, K) – fp16, contiguous
    B: (batch, K, N) – fp16, contiguous
    Returns C: (batch, M, N) – fp16
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # strides measured in element counts (not bytes)
    stride_az, stride_am, stride_ak = A.stride()
    stride_bz, stride_bk, stride_bn = B.stride()
    stride_cz, stride_cm, stride_cn = C.stride()

    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_az, stride_am, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C

# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] or None (binary)
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor [B, N, N, C] with the same dtype as ``input_tensor``.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------
    # 1️⃣  Input LayerNorm (float16 for speed)
    # --------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------
    # 2️⃣  Cast model weights to fp16
    # --------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------
    # 3️⃣  Linear projections (no bias) and optional masking
    # --------------------------------------------------
    # Shapes: (B, N, N, H)
    left  = F.linear(x, left_proj_w)
    right = F.linear(x, right_proj_w)

    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------
    # 4️⃣  Gate projections (sigmoid)
    # --------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------
    # 5️⃣  Prepare tensors for the batched matmul:
    #     compute left @ rightᵀ for each hidden slice
    # --------------------------------------------------
    # Move hidden dim to second axis and make them contiguous
    # left:  (B, N, N, H) -> (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    # rightᵀ: swap the two N axes before flattening
    right_t = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left.shape

    # Collapse batch & hidden dimensions → (B*H, N, N)
    left_flat  = left.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)

    # --------------------------------------------------
    # 6️⃣  Core O(N³·H) reduction via Triton batched GEMM
    # --------------------------------------------------
    out_flat = triton_batched_matmul(left_flat, right_flat)  # (B*H, N, N), fp16

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------
    # 7️⃣  Hidden‑dim LayerNorm + output gate
    # --------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # --------------------------------------------------
    # 8️⃣  Final linear projection back to the input dimension
    # --------------------------------------------------
    out = F.linear(out, to_out_w)       # (B, N, N, C)

    # Cast back to original dtype if necessary
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
