"""
TriMul (outgoing) forward pass with Triton‑accelerated elementwise scaling.
Algorithm
---------
1. Layer‑norm the 4‑D input tensor (B,N,N,C).
2. Cast to float16 for compute‑heavy operations.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are performed with torch.nn.functional.linear.
4. Optional binary mask is applied to left/right before gating.
5. Gates (sigmoid) are multiplied into the projections.
6. Core cubic reduction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is realized as a batch of matrix multiplications:
      left   → (B,H,N,N)  [i,k]
      right  → (B,H,N,N)  [k,j] (transpose k↔j)
      out    → (B,H,N,N)  = left @ right   (batched torch.bmm)
7. Apply a second LayerNorm over the hidden dimension.
8. Scale the result by out_gate using a tiny Triton kernel (in‑place).
9. Final linear projection back to the original channel dimension.
10. Cast the output back to the input dtype (float32).

Only the elementwise scaling (step 8) is implemented in Triton,
while the heavy matrix‑multiplication uses cuBLAS via torch.bmm,
which is considerably faster than a pure‑Triton GEMM for the
tested sizes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: in‑place elementwise multiplication (scale)
# ----------------------------------------------------------------------
@triton.jit
def _scale_kernel(x_ptr, scale_ptr, size, BLOCK: tl.constexpr):
    """x_ptr[x] = x_ptr[x] * scale_ptr[x]   (in‑place)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    s = tl.load(scale_ptr + offs, mask=mask, other=0.0)
    tl.store(x_ptr + offs, x * s, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (torch tensors, float32)
            - config: dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # ----------------------- unpack -----------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------- LayerNorm on input -----------------------
    norm_weight = weights["norm.weight"].to(device).to(input_tensor.dtype)
    norm_bias = weights["norm.bias"].to(device).to(input_tensor.dtype)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Cast to float16 for the compute‑heavy part
    x_fp16 = x.to(torch.float16) if x.dtype != torch.float16 else x

    # ----------------------- load weights (fp16) -----------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # ----------------------- linear projections -----------------------
    # (B, N, N, H)
    left = F.linear(x_fp16, left_proj_w)      # left_proj
    right = F.linear(x_fp16, right_proj_w)    # right_proj

    # ----------------------- optional mask -----------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(x_fp16.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp
        right = right * mask_fp

    # ----------------------- gate computations -----------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ----------------------- core reduction (batched GEMM) -----------------------
    B, N, _, _ = left.shape  # (_, N, _, H) - we will re‑use N
    # reshape to (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  -> (i,k)
    right = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  -> (k,j)

    # flatten batch & head dimensions for torch.bmm
    left_flat = left.view(B * hidden_dim, N, N)    # (B*H, N, K)
    right_flat = right.view(B * hidden_dim, N, N)  # (B*H, K, N)

    out_flat = torch.bmm(left_flat, right_flat)    # (B*H, N, N)
    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------- second LayerNorm -----------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ----------------------- scale by out_gate (Triton) -----------------------
    # both tensors are contiguous and have the same number of elements
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    total_elems = out.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)
    _scale_kernel[grid](out.data_ptr(),
                        out_gate.data_ptr(),
                        total_elems,
                        BLOCK=BLOCK)

    # ----------------------- final projection -----------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to original dtype if needed
    out = out.to(orig_dtype) if out.dtype != orig_dtype else out
    return out
