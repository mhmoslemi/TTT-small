@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr, num_warps: tl.constexpr
):
   # Compute program IDs
   pid = tl.program_id(0)
   # total tiles per batch = ceil(M/BM) * ceil(N/BN)
   # compute batch index and tile coordinates
   batch_id = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
   tile_index = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
   block_m = (tile_index // tl.cdiv(N, BLOCK_N)) * BLOCK_M
   block_n = (tile_index % tl.cdiv(N, BLOCK_N)) * BLOCK_N

   # then offsets for a and b & c
   # offsets for A: (block_m + tl.arange(0, BLOCK_M), tl.arange(0, BLOCK_K))
   # actual pointer: a_ptr + batch_id * batch_stride_a + (block_m + tl.arange(...)) * stride_am + (k_offset) * stride_ak
   # For B: k_offset + tl.arange(0, BLOCK_K) , block_n + tl.arange(0, BLOCK_N)
   # pointer: b_ptr + batch_id * batch_stride_b + (k_offset) * stride_bk + (block_n + tl.arange(...)) * stride_bn
   ...

   # accumulator in fp32 because half precision may produce overflow; but final cast to fp16.
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
   for k in range(0, K, BLOCK_K):
       # load A and B tiles
       a = tl.load(a_ptrs, mask=... , other=0.0) # shape (BLOCK_M, BLOCK_K)
       b = tl.load(b_ptrs, mask=..., other=0.0) # shape (BLOCK_K, BLOCK_N)
       # Possibly cast to float32
       a = a.to(tl.float32)
       b = b.to(tl.float32)
       acc += tl.dot(a, b)
       # update offsets
   # after loop, store result
   c = acc.to(tl.float16)
   tl.store(c_ptrs, c, mask=mask)

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr, num_warps: tl.constexpr):
    pid = tl.program_id(0)
    # compute block indices
    n_blocks_m = tl.cdiv(M, BLOCK_M)
    n_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_id = pid // (n_blocks_m * n_blocks_n)
    tile_id = pid % (n_blocks_m * n_blocks_n)
    block_m = (tile_id // n_blocks_n) * BLOCK_M
    block_n = (tile_id % n_blocks_n) * BLOCK_N

    # Load pointers for this batch
    a_ptrs = a_ptr + batch_id * batch_stride_a
    b_ptrs = b_ptr + batch_id * batch_stride_b
    c_ptrs = c_ptr + batch_id * batch_stride_c

    # offsets
    offs_m = block_m + tl.arange(0, BLOCK_M)
    offs_n = block_n + tl.arange(0, BLOCK_N)
    # create mask for S within M/N bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k in blocks
    for k in range(0, K, BLOCK_K):
        k_offset = k + tl.arange(0, BLOCK_K)
        # mask for K dimension
        mask_k = k_offset < K

        # Compute a and b pointers
        a = tl.load(
            a_ptrs + (offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        b = tl.load(
            b_ptrs + (k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # Cast to float32 for compute
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # Dot product accumulation
        acc += tl.dot(a, b)

    # Write back
    # Cast to fp16
    c = acc.to(tl.float16)
    # apply mask
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(
        c_ptrs + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask
    )

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑Norm the input tensor (B,N,N,C) using the provided LN parameters.
2. Cast to fp16 and apply four linear projections (left/right and their gates).
   The mask (if present) is applied before the gates.
3. Fuse the gate values into the projected tensors.
4. Core O(N³·H) reduction:
   For each head h we need   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   This is a batched matrix multiplication of shape (B*H,N,N) × (B*H,N,N)ᵀ.
   We implement this batched GEMM in pure Triton (kernel *batched_gemm_kernel*),
   keeping the data in fp16 and accumulating in fp32.
5. Reshape back to (B,N,N,H) and apply a second Layer‑Norm over H.
6. Multiply with the output gate (element‑wise, fused via a tiny Triton kernel).
7. Final linear projection back to the original channel dimension.
8. Cast the result back to the original input dtype.

Only the heavy O(N³·H) reduction and the final element‑wise gate multiplication
are performed in Triton; all other ops use PyTorch which already runs efficiently
on the GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B) where A: (batch, M, K), B: (batch, K, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr, num_warps: tl.constexpr
):
    pid = tl.program_id(0)

    # number of tiles per matrix
    n_blocks_m = tl.cdiv(M, BLOCK_M)
    n_blocks_n = tl.cdiv(N, BLOCK_N)

    # derive batch and tile indices from pid
    batch_id = pid // (n_blocks_m * n_blocks_n)
    tile_id = pid % (n_blocks_m * n_blocks_n)
    block_m = (tile_id // n_blocks_n) * BLOCK_M
    block_n = (tile_id % n_blocks_n) * BLOCK_N

    # base pointers for this batch
    a_base = a_ptr + batch_id * batch_stride_a
    b_base = b_ptr + batch_id * batch_stride_b
    c_base = c_ptr + batch_id * batch_stride_c

    # offsets within the tile
    offs_m = block_m + tl.arange(0, BLOCK_M)
    offs_n = block_n + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the K dimension in blocks
    for k in range(0, K, BLOCK_K):
        k_range = k + tl.arange(0, BLOCK_K)
        mask_k = k_range < K

        # load A block (M x K)
        a = tl.load(
            a_base + (offs_m[:, None] * stride_am + k_range[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # load B block (K x N)
        b = tl.load(
            b_base + (k_range[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # compute dot product; cast to fp32 for accumulation
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # store result C (M x N)
    c = acc.to(tl.float16)
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(
        c_base + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=c_mask
    )


def batched_gemm(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Compute batched matrix multiplication C = A @ B where A:(B,M,K), B:(B,K,N).
    All tensors must be contiguous, fp16 and on the same CUDA device.
    """
    assert a.is_cuda and b.is_cuda
    assert a.dtype == torch.float16 and b.dtype == torch.float16
    batch, M, K = a.shape
    batch_b, K_b, N = b.shape
    assert batch == batch_b and K == K_b

    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    # strides (in elements)
    batch_stride_a = a.stride(0)
    stride_am = a.stride(1)
    stride_ak = a.stride(2)

    batch_stride_b = b.stride(0)
    stride_bk = b.stride(1)
    stride_bn = b.stride(2)

    batch_stride_c = c.stride(0)
    stride_cm = c.stride(1)
    stride_cn = c.stride(2)

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid_m = (M + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch * grid_m * grid_n

    batched_gemm_kernel[total_blocks](
        a,
        b,
        c,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        M,
        N,
        K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    return c


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication (used for out‑gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, total_elems,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


def elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    assert a.shape == b.shape
    total = a.numel()
    c = torch.empty_like(a)
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        a,
        b,
        c,
        total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            mask: torch.Tensor of shape [B, N, N] (or None)
            weights: dict containing model weights (fp32)
            config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (performed in fp16 but accumulates in fp32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2) Cast all weights to fp16 for the compute‑heavy part
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)     # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 3) Linear projections (left/right)
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)  # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 4) Optional mask
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 5) Gates (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, hidden_dim)
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()    # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 6) Fuse gates into projections
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 7) Prepare tensors for the O(N³·H) reduction
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N) – make the head dimension contiguous
    left_ = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right needs to be transposed on the last two axes: (B, H, N, N) with K dimension first
    right_ = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    # merge batch and head dimensions for a single batched GEMM
    left_mat  = left_.view(B * H, N, N)   # (B*H, N, K)
    right_mat = right_.view(B * H, N, N)  # (B*H, K, N)  (already transposed)

    # --------------------------------------------------------------
    # 8) Core O(N³·H) reduction via Triton batched GEMM
    # --------------------------------------------------------------
    out_mat = batched_gemm(left_mat, right_mat)   # (B*H, N, N)

    # --------------------------------------------------------------
    # 9) Reshape back to (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 10) Output LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 11) Apply output gate (element‑wise, Triton)
    # --------------------------------------------------------------
    out = elementwise_mul(out, out_gate)

    # --------------------------------------------------------------
    # 12) Final linear projection back to original channel dimension
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 13) Cast back to the original dtype if needed
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
