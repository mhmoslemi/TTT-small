"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (B,N,N,C) using the provided LN parameters.
2. Cast to float16 for the compute‑heavy sub‑graph.
3. Linear projections:
   * left_proj = x @ left_proj_weight.T
   * right_proj = x @ right_proj_weight.T
   * left_gate  = sigmoid(x @ left_gate_weight.T)
   * right_gate = sigmoid(x @ right_gate_weight.T)
   * out_gate   = sigmoid(x @ out_gate_weight.T)      # shape (B,N,N,H)
4. Fuse mask (if present) and gate multiplication with a tiny Triton
   kernel:
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
   The mask has shape (B,N,N) and is broadcast over the hidden dimension H.
5. Core reduction:
        out_h = left_h @ right_hᵀ   for each hidden dim h.
   This is implemented as a batched GEMM using `torch.bmm` on tensors
   reshaped to (B*H, N, N).  cuBLAS tensor‑core kernels handle the
   O(N³·H) work efficiently.
6. Apply a second layer‑norm on the hidden dimension, multiply by the
   previously computed `out_gate`, and project back to the original
   channel dimension C with a linear layer.
7. Cast the result back to the input dtype (usually float32).

The only Triton component is the element‑wise mask‑gate fusion kernel,
which reduces memory traffic compared to a naïve PyTorch implementation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: elementwise (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # *float16   projected tensor (flattened)
    gate_ptr,          # *float16   gate tensor (flattened)
    mask_ptr,          # *float16   mask tensor (flattened, shape B*N*N)
    out_ptr,           # *float16   output tensor (flattened)
    total_elems,       # i64        total number of elements in proj/gate
    hidden_dim,        # i64        hidden dimension H (for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard against OOB
    mask = offs < total_elems

    # Compute the index of the mask element (each mask entry is reused H times)
    mask_idx = offs // hidden_dim

    # Load data
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    proj : (B,N,N,H)  float16
    gate : (B,N,N,H)  float16
    mask : (B,N,N)    float16 (0/1)
    Returns proj * gate * mask (broadcast over H) as float16.
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor of shape [B, N, N, C]
        mask         : torch.Tensor of shape [B, N, N] (or None)
        weights      : dict of model parameters (float32)
        config       : dict with keys:
                       - "dim"         : input channel dimension C
                       - "hidden_dim" : hidden dimension H
                       - "nomask"     : bool (optional, default False)

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (same dtype as input)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast weights once to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,C) -> (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (same shape), apply sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Apply mask (if any) and fuse with gates via Triton kernel
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj,  left_gate,  mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        # No mask – simple elementwise multiplication
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (left @ rightᵀ) over (B*H) batches.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
