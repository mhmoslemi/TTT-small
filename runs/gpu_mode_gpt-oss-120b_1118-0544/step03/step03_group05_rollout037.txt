@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_group = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    b = pid_group // H
    h = pid_group % H

    # Compute start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Base pointers
    left_base = left_ptr + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base = out_ptr + b * stride_bo + h * stride_ho

    # Offsets for i and j
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    # Compute valid masks for i and j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Create broadcasted masks for load
    # We'll later apply to store.

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k0 in range(0, N, BLOCK_K):
        # Compute k offsets
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + (offs_i[:, None] * stride_il) + (offs_k[None, :] * stride_kl)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile: shape (BLOCK_K, BLOCK_N) - note swapped order
        b_ptrs = right_base + (offs_k[:, None] * stride_kr) + (offs_j[None, :] * stride_jr)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Compute multiplication-accumulation
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

    # Apply mask for output (i,j out of bounds)
    # Convert acc to fp16 for storing
    out = acc.to(tl.float16)

    # Store
    c_ptrs = out_base + (offs_i[:, None] * stride_io) + (offs_j[None, :] * stride_jo)
    tl.store(c_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_group = tl.program_id(0)  # group index (b, h)
    pid_i = tl.program_id(1)  # block row i
    pid_j = tl.program_id(2)  # block col j

    # decode b and h
    b = pid_group // H
    h = pid_group % H

    # compute start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # base pointers for this group
    # left_base = left_ptr + b*stride_bl + h*stride_hl
    # right_base = right_ptr + b*stride_br + h*stride_hr
    # out_base = out_ptr + b*stride_bo + h*stride_ho

    left_base = left_ptr + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base = out_ptr + b * stride_bo + h * stride_ho

    # offsets for i, j
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k0 in range(0, N, BLOCK_K):
        # compute k offsets
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile
        a_ptrs = left_base + offs_i[:, None] * stride_il + offs_k[None, :] * stride_kl
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (k rows, j cols)
        b_ptrs = right_base + offs_k[:, None] * stride_kr + offs_j[None, :] * stride_jr
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Compute outer product accumulation
        acc += tl.dot(a, b)

    # Cast result to fp16
    out = acc.to(tl.float16)

    # Store to output
    c_ptrs = out_base + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo
    tl.store(c_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input tensor (B, N, N, C) using the provided LN parameters.
2. Cast to fp16 and apply the four linear projections (left, right, left_gate,
   right_gate, out_gate) with torch.nn.functional.linear (bias‑free linear).
3. Optional binary mask (broadcast over the hidden dim) is applied to the
   left/right projections.
4. Gating: left ← left * sigmoid(left_gate),  right ← right * sigmoid(right_gate).
5. Core O(N³·H) reduction:
   For each batch element *b* and hidden channel *h* we compute
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiply where the left matrix is (N×N) and the
   right matrix is interpreted as (N×N)ᵀ.  The operation is performed by a
   custom Triton kernel that tiles the output in (BLOCK_M, BLOCK_N) and loops
   over the K dimension in blocks of size BLOCK_K, using `tl.dot` for fast
   Tensor‑Core accumulation (fp32 accumulator → fp16 result).
6. Layer‑norm over the hidden dimension, multiply by the output gate.
7. Final linear projection back to the original channel dimension.
8. Cast the result back to the original dtype (usually float32).

The Triton kernel fuses the massive N³ reduction and avoids the large
overhead of launching many separate cuBLAS batched GEMMs.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes for a single (batch, hidden) pair:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    """
    pid_group = tl.program_id(0)      # linear index for (b, h)
    pid_i = tl.program_id(1)          # tile row index
    pid_j = tl.program_id(2)          # tile column index

    # decode batch and hidden indices
    b = pid_group // H
    h = pid_group % H

    # start coordinates of this tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base   = out_ptr   + b * stride_bo + h * stride_ho

    # thread‑level row / column offsets
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # pointers to the first K‑block
    a_ptrs = left_base + (offs_i[:, None] * stride_il) + (tl.arange(0, BLOCK_K)[None, :] * stride_kl)
    b_ptrs = right_base + (tl.arange(0, BLOCK_K)[:, None] * stride_kr) + (offs_j[None, :] * stride_jr)

    # iterate over K dimension
    for k in range(0, N, BLOCK_K):
        # compute masks for the current K‑block
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load tiles, applying the correct masks
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)          # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)          # (BLOCK_K, BLOCK_N)

        # accumulate the product
        acc += tl.dot(a, b)                           # (BLOCK_M, BLOCK_N)

        # advance K‑pointers
        a_ptrs += BLOCK_K * stride_kl
        b_ptrs += BLOCK_K * stride_kr

    # write back FP16 result
    out = acc.to(tl.float16)
    c_ptrs = out_base + (offs_i[:, None] * stride_io) + (offs_j[None, :] * stride_jo)
    tl.store(c_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (torch.Tensors)
            - config: dict containing "dim", "hidden_dim" and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Prepare fp16 weights
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)  # (H, C)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)  # (H, C)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, C)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)      # (C, H)

    # ------------------------------------------------------------------
    # 4) Linear projections (left/right) and gates
    # ------------------------------------------------------------------
    left      = F.linear(x, left_proj_w)      # (B, N, N, H)
    right     = F.linear(x, right_proj_w)     # (B, N, N, H)

    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    left_gate  = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse gates into left/right
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate
    # free memory of gate tensors that are no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core O(N³·H) reduction – Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # (B, N, N, H), fp16

    # strides (in elements) for the tensors
    stride_bl, stride_il, stride_kl, stride_hl = left.stride()
    stride_br, stride_jr, stride_kr, stride_hr = right.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out.stride()

    BLOCK_M = 128   # rows per thread block (also number of threads)
    BLOCK_N = 128   # columns per thread block
    BLOCK_K = 32    # reduction block size

    grid = (
        B * H,                                 # groups (batch * hidden)
        (N + BLOCK_M - 1) // BLOCK_M,          # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,          # tiles over j
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_jr, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 7) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype:
        out = out.to(dtype)

    return out
