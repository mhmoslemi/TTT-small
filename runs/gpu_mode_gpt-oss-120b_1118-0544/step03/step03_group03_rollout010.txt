    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device
    dtype = input_tensor.dtype
    # Normalization
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    x_fp16 = x.half()
    # linear projections and gates
    # (we will compute left_proj, right_proj, left_gate, right_gate, out_gate)
    left_proj = torch.nn.functional.linear(x_fp16, weights["left_proj.weight"].half())
    right_proj = torch.nn.functional.linear(x_fp16, weights["right_proj.weight"].half())
    left_gate_pre = torch.nn.functional.linear(x_fp16, weights["left_gate.weight"].half())
    right_gate_pre = torch.nn.functional.linear(x_fp16, weights["right_gate.weight"].half())
    out_gate_pre = torch.nn.functional.linear(x_fp16, weights["out_gate.weight"].half())
    # mask
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)
    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate
    # shape
    B, N, _, _ = left.shape
    # reshape to batch*hidden
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)  # shape (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)  # shape (B*H, N, N)
    # allocate output
    out_mat = torch.empty_like(left_mat)  # fp16
    # compute strides
    stride_batch = N * N
    stride_am = N  # stride m (row) in A
    stride_ak = 1  # stride k (col) in A
    stride_bk = 1  # stride row (k) in B (since we treat B transposed)

        b_ptr=right_mat,

        stride_batch_b=stride_batch,
        stride_batch_c=stride_batch,
        stride_am=stride_am,

        stride_bn=stride_bn,

    stride_am, stride_ak,

    batch_head = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_id // num_blocks_n
    pid_n = block_id % num_blocks_n

    # Coordinate offsets

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy O(N³) reduction is implemented as a fused batched matrix multiplication
using a custom Triton kernel.  All linear projections, gates and the final
LayerNorm/linear projection are performed with PyTorch.  The kernel works on
fp16 tensors (TensorCore friendly) and supports optional masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_matmul_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # problem size
    M, N, K,
    # strides for batch dimension
    stride_batch_a, stride_batch_b, stride_batch_c,
    # matrix A layout (M x K)
    stride_am, stride_ak,
    # matrix B layout (K x N) – B is accessed as transposed (original shape N x K)
    stride_bk, stride_bn,
    # matrix C layout (M x N)
    stride_cm, stride_cn,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of tiles per matrix
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_mat = num_blocks_m * num_blocks_n

    # which matrix (batch*head) and which tile inside that matrix
    batch_head = pid // blocks_per_mat
    tile_id = pid % blocks_per_mat
    pid_m = tile_id // num_blocks_n
    pid_n = tile_id % num_blocks_n

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this matrix
    a_base = a_ptr + batch_head * stride_batch_a
    b_base = b_ptr + batch_head * stride_batch_b
    c_base = c_ptr + batch_head * stride_batch_c

    # pointers to the current tile
    a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)
        # tl.dot performs the inner‑product (fp16 * fp16 -> fp32)
        acc += tl.dot(a, b)
        # advance K‑tiles
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c = acc.to(tl.float16)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, c, mask=c_mask)


def custom_kernel(data):
    """
    Outgoing TriMul forward pass with a Triton kernel for the core N³ reduction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model weights (see skeleton)
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ----------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ----------------------------------------------------
    # 2) Cast to fp16 for the heavy part
    # ----------------------------------------------------
    x_h = x.half()  # [B, N, N, dim] fp16

    # ----------------------------------------------------
    # 3) Linear projections (fp16) + gates
    # ----------------------------------------------------
    left_proj = F.linear(x_h, weights["left_proj.weight"].half())          # [B, N, N, H]
    right_proj = F.linear(x_h, weights["right_proj.weight"].half())        # [B, N, N, H]
    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].half())      # [B, N, N, H]
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].half())    # [B, N, N, H]
    out_gate_pre = F.linear(x_h, weights["out_gate.weight"].half())        # [B, N, N, H]

    # ----------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # ----------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ----------------------------------------------------
    # 5) Gating (sigmoid, fp16)
    # ----------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)  # [B, N, N, H] fp16

    left = left_proj * left_gate   # [B, N, N, H] fp16
    right = right_proj * right_gate # [B, N, N, H] fp16

    # ----------------------------------------------------
    # 6) Batched matrix multiplication via Triton
    #    out[b, i, j, h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    # ----------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N) – row‑major contiguous layout
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)   # [B*H, N, N]
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N) # [B*H, N, N]
    out_mat = torch.empty_like(left_mat)  # fp16 output

    # strides (in elements)
    stride_batch_a = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_batch_b = right_mat.stride(0)
    stride_bk = right_mat.stride(2)   # row of transposed view (k)
    stride_bn = right_mat.stride(1)   # col of transposed view (j)

    stride_batch_c = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # problem size
    M = N
    K = N

    # block sizes – tuned for typical B200 GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = left_mat.shape[0] * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    tri_mul_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=M,
        N=N,
        K=K,
        stride_batch_a=stride_batch_a,
        stride_batch_b=stride_batch_b,
        stride_batch_c=stride_batch_c,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ----------------------------------------------------
    # 7) Reshape back to [B, N, N, H] (fp16)
    # ----------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ----------------------------------------------------
    # 8) Final LayerNorm, gating and projection (float32)
    # ----------------------------------------------------
    out_fp32 = out.float()
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate.float()          # apply output gate
    out_final = F.linear(out_norm, weights["to_out.weight"])  # [B, N, N, dim] float32

    return out_final
