    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy N³ reduction is expressed as a batched matrix multiplication:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

All linear projections, gating and optional masking are performed in fp16.
The N³ reduction is fused into a single Triton kernel that processes every
(batch, hidden) pair as a separate GEMM, eliminating the per‑channel kernel
launches of torch.bmm and reducing memory traffic by using tiled loads and
fp32 accumulation.  The final LayerNorm and output projection are done in
PyTorch (fp32) to retain numerical stability.

The kernel works for any batch size B∈{1,2}, sequence length N up to 1024 and
hidden dimension hidden_dim (c_z) up to 384.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,            # fp16, [B, N, N, H]
    right_ptr,           # fp16, [B, N, N, H]
    out_ptr,             # fp16, [B, N, N, H]
    B, N, H,             # runtime scalars
    # strides for left
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right (same layout as left)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for out
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)               # i‑tile
    pid_j = tl.program_id(1)               # j‑tile
    pid_bh = tl.program_id(2)              # combined (batch, hidden)

    # decode batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # tile origins
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # accumulator in fp32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K

        # offsets
        offs_i = i0 + tl.arange(0, BLOCK_M)
        offs_j = j0 + tl.arange(0, BLOCK_N)
        offs_k = k0 + tl.arange(0, BLOCK_K)

        # bounds
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # ----- load left tile (M x K) -----
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + h * stride_left_h
        )
        left_tile = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)      # (BLOCK_M, BLOCK_K)

        # ----- load right tile (N x K) -----
        # layout: (j, k); we later transpose to (k, j)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_j[:, None] * stride_right_j
            + offs_k[None, :] * stride_right_k
            + h * stride_right_h
        )
        right_tile = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)      # (BLOCK_N, BLOCK_K)

        # transpose to (K, N) so we can use tl.dot
        right_tile_T = tl.trans(right_tile)   # (BLOCK_K, BLOCK_N)

        # ----- GEMM accumulation -----
        acc += tl.dot(left_tile, right_tile_T)   # (BLOCK_M, BLOCK_N)

    # ----- write result (convert back to fp16) -----
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h * stride_out_h
    )
    out_tile = acc.to(tl.float16)
    tl.store(
        out_ptrs,
        out_tile,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model weights (see docstring)
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy linear / gate work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections and gates (fp16)
    # ------------------------------------------------------------------
    left_proj = F.linear(
        x_h,
        weights["left_proj.weight"].to(torch.float16).to(device),
        bias=None,
    )  # [B, N, N, hidden_dim] fp16
    right_proj = F.linear(
        x_h,
        weights["right_proj.weight"].to(torch.float16).to(device),
        bias=None,
    )

    left_gate_pre = F.linear(
        x_h,
        weights["left_gate.weight"].to(torch.float16).to(device),
        bias=None,
    )
    right_gate_pre = F.linear(
        x_h,
        weights["right_gate.weight"].to(torch.float16).to(device),
        bias=None,
    )
    out_gate_pre = F.linear(
        x_h,
        weights["out_gate.weight"].to(torch.float16).to(device),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 5) Gating (sigmoid, fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    left = left_proj * left_gate   # [B, N, N, hidden_dim] fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Batched GEMM using Triton (N³ reduction)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left, dtype=torch.float16)  # [B, N, N, hidden_dim]

    # strides (batch, i, k, hidden)
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_j, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # Tile sizes – chosen to map well to Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    tri_mul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        hidden_dim,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_h,
        s_r_b,
        s_r_j,
        s_r_k,
        s_r_h,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7) Final LayerNorm, gating and projection (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate.to(torch.float32)

    final = F.linear(out_norm, weights["to_out.weight"])

    return final
