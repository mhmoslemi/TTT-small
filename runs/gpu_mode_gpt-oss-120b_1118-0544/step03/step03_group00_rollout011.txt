@triton.jit
def apply_gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
        total_elems, hidden_dim,
        BLOCK_SIZE: tl.constexpr):
    # using pointer arithmetic to process loops
    offsets = tl.arange(0, BLOCK_SIZE)
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + offsets

    mask = offs < total_elems
    # compute mask index: integer division offs // hidden_dim
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask)
```

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Layer Norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Cast to float16
    x = x.to(torch.float16)

    # Cast all weights to float16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # shape B,N,N,hidden_dim
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # Gate projections + sigmoid
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)

        right = right_proj * right_gate

    left_flat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)

"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) with the supplied LN parameters.
2. Cast to float16 for all heavy compute.
3. Perform six linear projections (left/right, three gates, output) using
   FP16 weights.
4. Fuse element‑wise mask, projection and gate with a tiny Triton kernel
   (proj * gate * mask) when a mask is present.
5. Core O(N³·H) update:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM: for each hidden channel `d` we compute
   a matrix multiply of size (N,N). The tensors are reshaped to (B·H,N,N)
   and `torch.bmm` (cuBLAS/Tensor‑cores) does the work.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate, and a final linear projection back to the original dimension.
7. Cast the result back to the input dtype (float32).

At least one operation (mask + gate fusion) is executed in Triton.
"""
import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused projection * gate * mask (broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,          # (B*N*N*H,) FP16
    gate_ptr,          # (B*N*N*H,) FP16
    mask_ptr,          # (B*N*N,)    FP16
    out_ptr,           # (B*N*N*H,) FP16
    total_elems,       # int: total number of elements in proj/gate
    hidden_dim,        # int: H (used for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard against OOB
    mask = offs < total_elems

    # Index into the mask (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def apply_gate_and_mask(proj: torch.Tensor,
                        gate: torch.Tensor,
                        mask: torch.Tensor,
                        block_size: int = 8192) -> torch.Tensor:
    """
    Fuse element‑wise multiplication of a projection tensor with its gate
    and an optional binary mask.

    Arguments:
        proj : (B,N,N,H) FP16 tensor
        gate : (B,N,N,H) FP16 tensor
        mask : (B,N,N)   FP16 tensor (values 0 or 1)
        block_size : launch granularity

    Returns:
        Tensor of shape (B,N,N,H) FP16 = proj * gate * mask
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim"          – input channel dimension C
            * "hidden_dim"   – hidden dimension H
            * "nomask" (opt) – if True, ignore the mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (dtype matches input_tensor.dtype)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> fp16 later)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy sections
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all weights to fp16 (biases are None for the Linear layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections: (B,N,N,C) -> (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (same shape) + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask + gates (fused via Triton when a mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous (broadcasted over hidden_dim)
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the third dimension (k)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # restore original layout (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype (typically float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
