```
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ag, stride_am, stride_ak,
    stride_bg, stride_bk, stride_bn,
    stride_cg, stride_cm, stride_cn,
    GROUP: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    num_blocks = num_blocks_m * num_blocks_n
    group_id = pid // num_blocks
    block_id = pid % num_blocks
    pid_m = block_id // num_blocks_n
    pid_n = block_id % num_blocks_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this group
    a_ptrs = a_ptr + group_id * stride_ag
    b_ptrs = b_ptr + group_id * stride_bg
    c_ptrs = c_ptr + group_id * stride_cg

    # compute pointers for each tile
    # for each k block
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(
            a_ptrs + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        b = tl.load(
            b_ptrs + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # a: [BM, BK]; b: [BK, BN]
        acc += tl.dot(a, b)  # uses fp16 multiply, accumulate fp32
    # write result
    c = acc.to(tl.float16)
    tl.store(c_ptrs + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn), c,
        mask=mask_m[:, None] & mask_n[None, :])
```

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Step 1: Input LayerNorm
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=1e-5)
    # cast to fp16
    x_h = x.to(torch.float16)

    # Step 2: Linear projections (fp16)
    # combine proj+gate for left and right as earlier for efficiency
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    # Use torch.nn.functional.linear directly
    left_proj = F.linear(x_h, left_proj_weight)
    left_gate_pre = F.linear(x_h, left_gate_weight)
    right_proj = F.linear(x_h, right_proj_weight)
    right_gate_pre = F.linear(x_h, right_gate_weight)

    # Apply mask if needed (broadcast)
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Compute out_gate
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate_pre = F.linear(x_h, out_gate_weight)
    out_gate = torch.sigmoid(out_gate_pre)

    # Step 3: Batched matmul with Triton kernel
    B, N, _, _ = left.shape
    # reshape to groups
    left_perm = left.permute(0,3,1,2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0,3,1,2).contiguous()
    # flatten groups
    G = B * hidden_dim
    left_flat = left_perm.view(G, N, N)
    right_flat = right_perm.view(G, N, N)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # define block sizes based on N: maybe 64
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # compute grid
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = G * num_blocks_m * num_blocks_n
    grid = (total_programs,)

    # compute strides in elements
    stride_ag = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bg = right_flat.stride(0)
    stride_bk = right_flat.stride(2)   # row stride = original col
    stride_bn = right_flat.stride(1)   # col stride = original row

    stride_cg = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    batched_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        M=N, N=N, K=N,
        stride_ag, stride_am, stride_ak,
        stride_bg, stride_bk, stride_bn,
        stride_cg, stride_cm, stride_cn,
        GROUP=G,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # Cast to fp32
    out_fp32 = out.to(torch.float32)

    # Final layernorm on hidden dim
    out_norm = F.layer_norm(out_fp32,
            normalized_shape=(hidden_dim,),
            weight=weights["to_out_norm.weight"],
            bias=weights["to_out_norm.bias"],
            eps=1e-5)

    out_norm = out_norm * out_gate.to(torch.float32)
    out = F.linear(out_norm, weights["to_out.weight"])

    return out

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ag, stride_am, stride_ak,
    stride_bg, stride_bk, stride_bn,
    stride_cg, stride_cm, stride_cn,
    GROUP: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_group = num_blocks_m * num_blocks_n
    group = pid // blocks_per_group
    block = pid % blocks_per_group
    pid_m = block // num_blocks_n
    pid_n = block % num_blocks_n

    # bounds
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers
    a_ptrs = a_ptr + group * stride_ag
    b_ptrs = b_ptr + group * stride_bg
    c_ptrs = c_ptr + group * stride_cg

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a = tl.load(
            a_ptrs + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (transposed)
        b = tl.load(
            b_ptrs + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b) # a,b are fp16, acc is fp32

    # Store result
    c = acc.to(tl.float16)
    tl.store(
        c_ptrs + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )

"""
TriMul outgoing update optimized with a custom Triton batched matrix multiplication kernel.
The kernel computes, for each batch and hidden channel, the matrix product:
    out[b, h] = left[b, h] @ right[b, h].T
using FP16 arithmetic and block tiling, avoiding the costly torch.bmm and extra
memory copies. The surrounding linear projections, gating, layer‑norms and the
final linear projection are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ag, stride_am, stride_ak,
    stride_bg, stride_bk, stride_bn,
    stride_cg, stride_cm, stride_cn,
    GROUP: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # Number of tiles per matrix
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_group = num_tiles_m * num_tiles_n

    # Identify the group (batch*head) and the tile within this group
    group_id = pid // tiles_per_group
    tile_id = pid % tiles_per_group
    pid_m = tile_id // num_tiles_n
    pid_n = tile_id % num_tiles_n

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this group
    a_base = a_ptr + group_id * stride_ag
    b_base = b_ptr + group_id * stride_bg
    c_base = c_ptr + group_id * stride_cg

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (shape BLOCK_M x BLOCK_K)
        a = tl.load(
            a_base + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B tile (shape BLOCK_K x BLOCK_N).  B is logically transposed,
        # so we stride on the original column dimension.
        b = tl.load(
            b_base + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # fp16 * fp16 -> fp32 accumulation
        acc += tl.dot(a, b)

    # Write result back in fp16
    c = acc.to(tl.float16)
    tl.store(
        c_base + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.
    Arguments:
        data: tuple (input_tensor, mask, weights, config)
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ----------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ----------------------------------------------------
    # 2) Cast to fp16 for the heavy work
    # ----------------------------------------------------
    x_h = x.to(torch.float16)

    # ----------------------------------------------------
    # 3) Linear projections + gating (fp16)
    # ----------------------------------------------------
    left_proj = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))
    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].to(torch.float16))
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].to(torch.float16))

    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate      # [B, N, N, hidden_dim] fp16
    right = right_proj * right_gate   # [B, N, N, hidden_dim] fp16

    # ----------------------------------------------------
    # 4) Output gate (fp16)
    # ----------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(torch.float16))
    )  # [B, N, N, hidden_dim] fp16

    # ----------------------------------------------------
    # 5) Batched matrix multiplication via Triton
    #    out[b, h] = left[b, h] @ right[b, h].T
    # ----------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim
    # Move hidden dimension to second axis and make tensors contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Collapse batch and heads into a single dimension
    G = B * H
    left_flat = left_perm.view(G, N, N)
    right_flat = right_perm.view(G, N, N)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # ----------------------------------------------------
    # 6) Launch Triton kernel
    # ----------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = G * num_tiles_m * num_tiles_n
    grid = (total_programs,)

    # Strides (in elements)
    stride_ag = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bg = right_flat.stride(0)
    stride_bk = right_flat.stride(2)   # row stride after logical transpose
    stride_bn = right_flat.stride(1)   # column stride after logical transpose

    stride_cg = out_flat.stride(0)
    stride_cm = out_flat.stride(1)

"""
TriMul outgoing update optimized with a custom Triton batched matrix‑multiplication
kernel.  The kernel computes, for each batch and hidden channel, the product

    out[b, h] = left[b, h] @ right[b, h].T

using FP16 arithmetic and block tiling, thus avoiding the expensive torch.bmm
and extra memory copies.  All surrounding linear projections, gating, layer‑norms
and the final projection are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ag, stride_am, stride_ak,
    stride_bg, stride_bk, stride_bn,
    stride_cg, stride_cm, stride_cn,
    GROUP: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute, for every group g ∈ [0, GROUP), the matrix product
        C_g = A_g @ B_g.T
    where A_g and B_g have shape (M, K) and (N, K) respectively.
    The kernel works on FP16 data and accumulates in FP32.
    """
    pid = tl.program_id(0)

    # Tiling layout
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_group = num_tiles_m * num_tiles_n

    # Resolve identifiers
    group_id = pid // tiles_per_group
    tile_id = pid % tiles_per_group
    pid_m = tile_id // num_tiles_n
    pid_n = tile_id % num_tiles_n

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current group
    a_base = a_ptr + group_id * stride_ag
    b_base = b_ptr + group_id * stride_bg
    c_base = c_ptr + group_id * stride_cg

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Reduce over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_base + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile (logically transposed): (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_base + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp16 * fp16 -> fp32 accumulation
        acc += tl.dot(a, b)

    # Store the result (FP16)
    c = acc.to(tl.float16)
    tl.store(
        c_base + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model weights (see skeleton)
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] FP32

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections + gating (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))
    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].to(torch.float16))
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].to(torch.float16))

    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate      # [B, N, N, hidden_dim] FP16
    right = right_proj * right_gate   # [B, N, N, hidden_dim] FP16

    # ------------------------------------------------------------------
    # 4) Output gate (FP16)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(torch.float16))
    )  # [B, N, N, hidden_dim] FP16

    # ------------------------------------------------------------------
    # 5) Batched matrix multiplication via Triton:
    #    out[b, h] = left[b, h] @ right[b, h].T
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Move hidden dimension to the second axis and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    G = B * H
    left_flat = left_perm.view(G, N, N)
    right_flat = right_perm.view(G, N, N)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Tiling parameters (adjustable)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid configuration
    tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_progs = G * tiles_m * tiles_n
    grid = (total_progs,)

    # Strides in elements
    stride_ag = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bg = right_flat.stride(0)
    # For the transposed view we stride over the original column (k) as rows
    stride_bk = right_flat.stride(2)  # row stride after logical transpose
    stride_bn = right_flat.stride(1)  # column stride after logical transpose

    stride_cg = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # Launch the Triton kernel
    batched_matmul_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        M=N,
        N=N,
        K=N,
        stride_ag=stride_ag,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bg=stride_bg,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cg=stride_cg,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        GROUP=G,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6) Reshape back to [B, N, N, hidden_dim] and cast to FP32
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out_fp32 = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 7) Final LayerNorm, output gating and projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate.to(torch.float32)   # apply output gate
    out = F.linear(out_norm, weights["to_out.weight"])  # final projection

    return out
