"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Project to a hidden dimension (left/right) and apply optional mask.
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Multiply left/right with their respective gates.
5. Perform the core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (left @ rightᵀ) where each
   (batch * hidden) pair is an independent matmul.  The reduction is
   implemented in a fused Triton kernel (batched GEMM) that works on
   fp16 inputs and accumulates in fp32.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate, and project back to the original channel dimension.
All heavy tensor work stays on the GPU; only light Python glue is used.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to A, B, C
    M, N, K,                           # matrix dimensions
    stride_am, stride_ak,               # A: (M,K) strides
    stride_bk, stride_bn,               # B: (K,N) strides (B is accessed transposed)
    stride_cm, stride_cn,               # C: (M,N) strides
    batch_stride_a, batch_stride_b, batch_stride_c,   # strides for the leading batch dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: for each batch idx p, compute C[p] = A[p] @ B[p].T."""
    pid_m = tl.program_id(axis=0)   # tile row
    pid_n = tl.program_id(axis=1)   # tile col
    pid_b = tl.program_id(axis=2)   # batch index

    # ---- tile offsets ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K dimension ----
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers to the current K‑tile of A and B (B is accessed transposed)
        a_ptrs = a_ptr + pid_b * batch_stride_a + \
                 (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_ptr + pid_b * batch_stride_b + \
                 (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        # masked loads
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # block‑level matrix multiplication
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # cast back to fp16 (or whatever the input dtype is)
    c = acc.to(tl.float16)

    # store result
    c_ptrs = c_ptr + pid_b * batch_stride_c + \
             (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------- #
    # 1. Input LayerNorm
    # --------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Work in fp16 for the heavy path
    x = x.to(torch.float16)

    # --------------------------------------------------- #
    # 2. Cast model weights to fp16
    # --------------------------------------------------- #
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------- #
    # 3. Linear projections (left/right) and optional mask
    # --------------------------------------------------- #
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    if (mask is not None) and (not nomask):
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------- #
    # 4. Gates (sigmoid)
    # --------------------------------------------------- #
    left_gate  = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()    # (B, N, N, H)

    # Apply gates to projections
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------- #
    # 5. Core O(N³·H) reduction – batched GEMM via Triton
    # --------------------------------------------------- #
    # Rearrange tensors to (B, H, N, N) and then flatten batch*H
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch = B * H

    left_flat  = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides for the batched matmul
    stride_am = left_flat.stride(1)      # stride for M dimension
    stride_ak = left_flat.stride(2)      # stride for K dimension
    batch_stride_a = left_flat.stride(0)

    # For B we need transposed view: we access right as (K,N) => (k, j)
    stride_bk = right_flat.stride(2)     # stride for K (original second N)
    stride_bn = right_flat.stride(1)     # stride for N (original first N)
    batch_stride_b = right_flat.stride(0)

    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    batch_stride_c = out_flat.stride(0)

    M = N
    K = N
    N_ = N

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N_ + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N_,
        K,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------- #
    # 6. Hidden‑dim LayerNorm
    # --------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------- #
    # 7. Apply output gate (element‑wise)
    # --------------------------------------------------- #
    out = out * out_gate

    # --------------------------------------------------- #
    # 8. Final projection back to original channel dimension
    # --------------------------------------------------- #
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the input dtype if needed
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
