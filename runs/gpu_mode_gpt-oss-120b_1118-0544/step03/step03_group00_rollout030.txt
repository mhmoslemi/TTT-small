def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # Get batch id
    pid_batch = tl.program_id(0)  # batch dimension
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # base pointers for this batch
    a_ptrs = A_ptr + pid_batch * batch_stride_a
    b_ptrs = B_ptr + pid_batch * batch_stride_b
    c_ptrs = C_ptr + pid_batch * batch_stride_c

    # compute pointers for each tile
    a_ptrs = a_ptrs + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptrs + offs_n[None, :] * stride_bn + tl.arange(0, BLOCK_K)[:, None] * stride_bk  # Actually if B shape is (batch, K, N) we need stride for K then N? We can have B stored as (batch, K, N). With stride_bk for K dimension and stride_bn for N dimension.

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (tl.arange(0, BLOCK_K)[None, :] + k*BLOCK_K < K), other=0.0)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # config includes dim, hidden_dim, perhaps nomask

    # Extract parameters
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]
    dim = config["dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on dim
    x = torch.nn.functional.layer_norm(input_tensor,
            (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast to float16
    x = x.to(torch.float16)

    # Cast weights to float16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections: left_proj, right_proj
    left_proj = torch.nn.functional.linear(x, left_proj_w)  # shape (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # Gate projections
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Apply mask + gate
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core reduction: batched matmul
    Bsize, Ndim, _, Hdim = left.shape
    # left shape (B,N,N,H), right shape same.

    # Permute and reshape
    left_perm = left.permute(0, 3, 1, 2).contiguous() # (B,H,N,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B,H,N,N)

    batch = Bsize * Hdim
    left_mat = left_perm.view(batch, Ndim, Ndim)   # (batch, M=N, K=N)
    right_mat = right_perm.view(batch, Ndim, Ndim) # (batch, K=N, N)

    # Allocate output
    out_mat = torch.empty((batch, Ndim, Ndim), dtype=torch.float16, device=input_tensor.device)

    # Compute strides
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    batch_stride_a = left_mat.stride(0)

"""
TriMul (outgoing) forward pass – Triton accelerated implementation.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the provided LN parameters.
2. Cast to fp16 for the heavy compute path.
3. Linear projections to the hidden dimension for the left/right streams,
   and three gate projections (left, right, out).  All linear ops are
   performed with fp16 weights.
4. If a mask is supplied (and `nomask` is False) fuse mask application,
   gating and element‑wise multiplication with a small Triton kernel.
5. Core reduction: for each batch * hidden‑dim we need
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   This is a batched matrix multiplication A @ B where
       A: (B*H, N, N)   = left.permute(0,3,1,2).contiguous()
       B: (B*H, N, N)   = right.permute(0,3,2,1).contiguous()
   The multiplication is performed by a custom Triton kernel that
   accumulates in fp32 and writes fp16.
6. Reshape the result back to (B,N,N,H), apply a second LayerNorm,
   multiply by the out‑gate and a final linear projection back to C.
7. Cast the output back to the original input dtype.

The implementation uses two Triton kernels:
* `apply_gate_mask_kernel`  – element‑wise (proj * gate * mask).
* `batched_matmul_kernel`   – batched fp16 matrix multiplication with
  fp32 accumulation.
Both kernels are launched from `custom_kernel`.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Kernel 1: fuse projection, gate and optional mask
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,          # (B*N*N*H,) fp16
    gate_ptr,          # (B*N*N*H,) fp16
    mask_ptr,          # (B*N*N,)   fp16 (values 0 or 1)
    out_ptr,           # (B*N*N*H,) fp16
    total_elems,       # int – total number of elements in proj/gate
    hidden_dim,        # int – H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard against out‑of‑bounds
    valid = offs < total_elems

    # mask index (broadcast over the hidden dimension)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(out_ptr + offs, proj * gate * mask, mask=valid)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor,
                        block_size: int = 1024) -> torch.Tensor:
    """
    proj : (B,N,N,H) fp16
    gate : (B,N,N,H) fp16
    mask : (B,N,N)   fp16 (0/1)
    Returns proj * gate * mask  (broadcast over H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: batched matrix multiplication C = A @ B
#   A: (batch, M, K)   fp16
#   B: (batch, K, N)   fp16
#   C: (batch, M, N)   fp16
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # program ids
    # --------------------------------------------------------------
    batch_id = tl.program_id(0)      # batch dimension (B*H)
    pid_m = tl.program_id(1)         # row tile
    pid_n = tl.program_id(2)         # column tile

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_ptrs = A_ptr + batch_id * batch_stride_a
    b_ptrs = B_ptr + batch_id * batch_stride_b
    c_ptrs = C_ptr + batch_id * batch_stride_c

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # main loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # masks for loading
        mask_a = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        mask_b = (cur_k[:, None] < K) & (offs_n[None, :] < N)

        # load A tile (BLOCK_M x BLOCK_K)
        a = tl.load(a_ptrs + offs_m[:, None] * stride_am
                              + cur_k[None, :] * stride_ak,
                    mask=mask_a, other=0.0)

        # load B tile (BLOCK_K x BLOCK_N)
        b = tl.load(b_ptrs + cur_k[:, None] * stride_bk
                              + offs_n[None, :] * stride_bn,
                    mask=mask_b, other=0.0)

        # dot product over K dimension
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # write back
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs + offs_m[:, None] * stride_cm
                     + offs_n[None, :] * stride_cn,
             c, mask=mask_c)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (FP32)
            - config: dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm on the last dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all weights to fp16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,C) -> (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (same shape)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Apply mask + gates (fused via Triton if a mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = apply_gate_and_mask(left_proj,  left_gate,  mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM (out = left @ rightᵀ)
    #    We reshape to (B*H, N, N) and use a custom Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                      # B,N,N,H
    # bring the hidden dimension to the second axis and make tensors contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N)  (k,j) order

    batch = B * H
    left_mat  = left_perm.view(batch, N, N)   # (batch, M=N, K=N)
    right_mat = right_perm.view(batch, N, N)  # (batch, K=N, N)

    out_mat = torch.empty((batch, N, N), dtype=torch.float16, device=device)

    # strides (in elements, not bytes)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    batch_stride_a = left_mat.stride(0)
    batch_stride_b = right_mat.stride(0)
    batch_stride_c = out_mat.stride(0)

    # tile sizes – tuned for B200 (fp16 tensor cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                     # M, N, K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
