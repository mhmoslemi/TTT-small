    pid_j = tl.program_id(1) # block col index (j tile)
    pid_hb = tl.program_id(2) # combined batch*head index

@triton.jit
def triu_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)  # i tile
    pid_j = tl.program_id(1)  # j tile
    pid_hb = tl.program_id(2) # batch*head tile

    # derived indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    b = pid_hb // H
    h = pid_hb % H

    # offsets for output
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # mask i,j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base offsets for left/right per batch and head
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for kk in range(0, N, BLOCK_K):
        # k offsets
        offs_k = kk + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block (M x K)
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)

        # Load right block (K x N)
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0, dtype=tl.float16)

        # dot product
        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton accelerated implementation.

The core operation is:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

where `left` and `right` are the gated, linearly projected tensors
of shape [B, N, N, H] (H = hidden_dim).  This is a batched matrix‑multiply
with the inner‑dimension K = N.  A custom Triton kernel computes the
N³ reduction directly on the original (B,N,N,H) layout, thus avoiding the
expensive `permute/contiguous` copies required by a naïve `torch.bmm`
approach.  All heavy arithmetic is performed in FP16 with FP32
accumulation, while the final result is written as FP32 so that the
subsequent LayerNorm and output projection can stay in FP32.

The surrounding operations (LayerNorm, linear projections, gates,
output‑gate, final LayerNorm & projection) are kept in PyTorch for
readability and to leverage its highly‑optimized kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: out = Σ_k left[..., i, k, h] * right[..., j, k, h]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # FP16 pointer, shape [B, N, N, H]
    right_ptr,              # FP16 pointer, shape [B, N, N, H]
    out_ptr,                # FP32 pointer, shape [B, N, N, H]
    B, N, H,                # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,  # tile rows  (i dimension)
    BLOCK_N: tl.constexpr,  # tile cols  (j dimension)
    BLOCK_K: tl.constexpr   # reduction tile (k dimension)
):
    pid_i = tl.program_id(0)          # tile along i (rows)
    pid_j = tl.program_id(1)          # tile along j (cols)
    pid_hb = tl.program_id(2)         # combined batch*head index

    # ------------------------------------------------------------------
    # Compute actual indices and bounds
    # ------------------------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    b = pid_hb // H                  # batch index
    h = pid_hb % H                   # head index

    # offsets for the i/j dimensions
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the current (b, h) slice
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduction over k
    # ------------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left block [M, K]
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16)

        # load right block [K, N] (note the transposed view)
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0,
                        dtype=tl.float16)

        # FP16 dot → FP32 accumulation
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # Write result (FP32)
    # ------------------------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (see skeleton)
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------
    # 2) Cast to FP16 for the heavy part
    # ------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------
    # 3) Fuse projection + gate linear layers (FP16)
    # ------------------------------------------------------------
    # left side: projection + gate
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16).to(device)  # (2*H, dim)
    left_combined = F.linear(x_h, left_combined_w)                # [B, N, N, 2*H]

    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)

    # right side: projection + gate
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16).to(device)  # (2*H, dim)
    right_combined = F.linear(x_h, right_combined_w)                # [B, N, N, 2*H]

    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    # ------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------
    # 5) Gating (sigmoid, FP16)
    # ------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate   # [B, N, N, H] FP16
    right = right_proj * right_gate  # [B, N, N, H] FP16

    # ------------------------------------------------------------
    # 6) Output gate (FP16) – applied after the final LayerNorm
    # ------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16).to(device)
    out_gate_pre = F.linear(x_h, out_gate_w)          # [B, N, N, H] FP16
    out_gate = torch.sigmoid(out_gate_pre)            # FP16, will be cast later

    # ------------------------------------------------------------
    # 7) TriMul: batched matrix multiplication via Triton
    #    out[b, i, j, h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------
    # Allocate output buffer in FP32 (so we can feed it directly to LayerNorm)
    out_fp32 = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides for the kernels (all tensors are contiguous)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_fp32.stride()

    # Tile sizes – tuned for Ampere GPUs (B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # i dimension
        (N + BLOCK_N - 1) // BLOCK_N,          # j dimension
        B * hidden_dim                         # combined batch*head dimension
    )

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_fp32,
        B=B,
        N=N,
        H=hidden_dim,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_h=stride_l_h,
        stride_r_b=stride_r_b,
        stride_r_j=stride_r_j,
        stride_r_k=stride_r_k,
        stride_r_h=stride_r_h,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_h=stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------
    # 8) Final LayerNorm, output gating and projection (FP32)
    # ------------------------------------------------------------
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.to(torch.float32)          # Apply output gate
    out = F.linear(out, weights["to_out.weight"])   # Project back to dim

    return out
