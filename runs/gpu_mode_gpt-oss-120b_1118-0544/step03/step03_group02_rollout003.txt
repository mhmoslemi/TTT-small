"""
Outgoing Triangle Multiplicative Update (TriMul) – optimized forward.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (float32).
2. Cast to float16 and fuse the four linear projections (left/right + their
   gates) into a single matrix multiply.  The combined weight has shape
   (4·hidden_dim, dim) and the result is split back into the four tensors.
3. Optional mask is applied to the left/right projections **before** gating.
4. Apply sigmoid gates and multiply with the projected values.
5. For each batch and hidden channel compute
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication:
        out_d = left_d @ right_dᵀ.
   This is done with a single `torch.bmm` on half‑precision tensors.
6. Apply a second LayerNorm over the hidden dimension.
7. Compute the out‑gate (linear → sigmoid) and fuse the element‑wise
   multiplication using a tiny Triton kernel.
8. Final linear projection back to the original channel dimension and cast
   to float32.

All heavy arithmetic is performed in float16 to exploit tensor cores; the
final output matches the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N, BLOCK: tl.constexpr):
    """
    Element‑wise multiplication: c = a * b   (1‑D flattened tensors).
    """
    pid = tl.program_id(0)
    off = pid * BLOCK + tl.arange(0, BLOCK)
    mask = off < N
    a = tl.load(a_ptr + off, mask=mask)
    b = tl.load(b_ptr + off, mask=mask)
    tl.store(c_ptr + off, a * b, mask=mask)

            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:

    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # Cast to half for the heavy work
    x_fp16 = x.to(torch.float16)

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Fuse the four linear layers (proj + gate for left/right)
    # ------------------------------------------------------------------
    # combined_weight shape: (4*hidden_dim, dim)
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    )  # (4*hidden_dim, dim)

    # Flatten spatial dims and apply the combined linear mapping
    x_flat = x_fp16.view(-1, dim)                     # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 4*hidden_dim)

    # Reshape back to [B, N, N, 4*hidden_dim] and split
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each: (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Optional masking (applied only to the projections)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and combine with projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate   # (B, N, N, hidden_dim)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Batched GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a half‑precision batched matmul.
    # ------------------------------------------------------------------
    # Permute to (B, hidden_dim, N, N) and merge batch & hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    left_mat = left_perm.view(B * hidden_dim, N, N)   # (B*hidden, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N) # (B*hidden, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, hidden_dim)  half

    # ------------------------------------------------------------------
    # 6. to_out_norm (LayerNorm over hidden dimension)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden_dim) half

    # ------------------------------------------------------------------
    # 7. Compute out_gate and apply it (Triton kernel)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # (B, N, N, hidden_dim) half

    total_el = out.numel()
    BLOCK = 1024
    grid = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        out,
        out_gate,
        out,
        total_el,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 8. Final projection back to original dimension
    # ------------------------------------------------------------------
    out_flat = out.view(-1, hidden_dim)               # (B*N*N, hidden_dim)
    out = F.linear(out_flat, to_out_weight)           # (B*N*N, dim) half
    out = out.view(B, N, N, dim)                      # (B, N, N, dim)

    # Return float32 as required
    return out.to(torch.float32)
