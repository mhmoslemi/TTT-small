    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # loop over K
    for k in range(0, K, BLOCK_K):

        offs_k = k + tl.arange(0, BLOCK_K)

"""
TriMul (outgoing) – Triton implementation of the core O(N³·H) update.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the supplied LN parameters.
2. Cast to float16 for the compute‑heavy path.
3. Linear projections (left/right) and their gates (left_gate, right_gate,
   out_gate) are performed with torch.nn.functional.linear.
4. Optional binary mask is applied (broadcast over the hidden dimension).
5. Gates are fused into the projected tensors (elementwise multiplication).
6. The contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix multiplication of size (N×N) @ (N×N)^T for each
   hidden channel.  This is executed with a custom Triton kernel that
   computes C = A @ Bᵀ for every (batch * hidden) matrix.
7. The result is reshaped back to (B,N,N,H) and a second Layer‑Norm
   (with its own weights) is applied.
8. The output gate (out_gate) modulates the tensor element‑wise.
9. A final linear projection brings the hidden dimension back to the
   original channel dimension C.
10. The output is cast back to the input dtype (usually float32).

At least the O(N³·H) reduction (step 6) is performed on the GPU via Triton,
meeting the requirement to have a non‑trivial kernel.
"""

import torch
import triton
import triton.language as tl

# Triton kernel: batched GEMM computing C = A @ Bᵀ
#   A : (B, M, K)   – row‑major
#   B : (B, N, K)   – row‑major (will be accessed as Bᵀ via swapped strides)
#   C : (B, M, N)   – row‑major result
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Compute which batch (combined batch*hidden) and tile we are in
    # --------------------------------------------------------------
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    batch_idx = pid // (num_tiles_m * num_tiles_n)
    tile_id   = pid % (num_tiles_m * num_tiles_n)
    tile_m    = tile_id // num_tiles_n
    tile_n    = tile_id % num_tiles_n

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Pointers to the start of the tile in A and Bᵀ
    #   A: (M, K)   – row‑major
    #   Bᵀ: (K, N) – column‑major (real B is row‑major, we just swap strides)
    # --------------------------------------------------------------
    a_base = a_ptr + batch_idx * batch_stride_a \
           + offs_m[:, None] * stride_am \
           + 0 * stride_ak          # k‑offset added inside the loop

    b_base = b_ptr + batch_idx * batch_stride_b \
           + 0 * stride_bk          # k‑offset added inside the loop
           + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in BLOCK_K sized steps
    # ------------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)          # (BLOCK_K,)

        # Load A tile
        a = tl.load(
            a_base + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & (cur_k < K)[None, :],
            other=0.0,
        )
        # Load Bᵀ tile (actually accesses original B with swapped strides)
        b = tl.load(
            b_base + cur_k[:, None] * stride_bk,
            mask=(cur_k < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back the tile to C
    # --------------------------------------------------------------
    c_ptrs = c_ptr + batch_idx * batch_stride_c \
           + offs_m[:, None] * stride_cm \
           + offs_n[None, :] * stride_cn

    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _tri_mul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = left @ rightᵀ for tensors of shape (B*H, N, N).

    Arguments
    ---------
    left  : torch.Tensor of shape (B*H, N, N), dtype=float16, device=cuda
    right : torch.Tensor of shape (B*H, N, N), dtype=float16, device=cuda

    Returns
    -------
    out   : torch.Tensor of shape (B*H, N, N), dtype=float16, device=cuda
    """
    assert left.shape == right.shape
    B_H, N, K = left.shape  # K == N
    assert K == N

    # Output allocation
    out = torch.empty_like(left)

    # Strides (as Python integers)
    stride_a_batch = left.stride(0)
    stride_am = left.stride(1)
    stride_ak = left.stride(2)

    stride_b_batch = right.stride(0)
    stride_bk = right.stride(2)   # column stride (k) → 1 for contiguous tensors
    stride_bn = right.stride(1)   # row (n)   → N for contiguous tensors

    stride_c_batch = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    # Tunable tile sizes – work well for the given ranges
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: one program per (batch_h, tile_m, tile_n)
    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B_H * num_tiles_m * num_tiles_n
    grid = (total_programs,)

    _tri_mul_gemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        stride_a_batch,
        stride_am,
        stride_ak,
        stride_b_batch,
        stride_bk,
        stride_bn,
        stride_c_batch,
        stride_cm,
        stride_cn,
        N,          # M
        N,          # N
        N,          # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative
    Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, S, S, C]
        - mask         : torch.Tensor of shape [B, S, S] (or None)
        - weights      : dict of torch.Tensors with the model parameters
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, S, S, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2) Cast to float16 for the compute‑heavy path
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast weights to float16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)  # (H, C)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)  # (H, C)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, C)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)

    to_out_w      = weights["to_out.weight"].to(torch.float16)      # (C, H)

    # --------------------------------------------------------------
    # 4) Linear projections
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # (B, S, S, H)
    right = torch.nn.functional.linear(x, right_proj_w)  # (B, S, S, H)

    # --------------------------------------------------------------
    # 5) Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, S, S, 1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 6) Gates (sigmoid) and fusion
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()   # (B, S, S, H)
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()    # (B, S, S, H)

    left = left * left_gate
    right = right * right_gate
    # free gates we no longer need (helps memory)
    del left_gate, right_gate

    # --------------------------------------------------------------
    # 7) Core O(N³·H) reduction using Triton GEMM kernel
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, S, _, H = left.shape  # H == hidden_dim

    # reshape to (B*H, S, S) with contiguous layout
    left_view = left.permute(0, 3, 1, 2).contiguous()   # (B, H, S, S)
    right_view = right.permute(0, 3, 1, 2).contiguous()  # (B, H, S, S)

    batch_h = B * H
    left_mat = left_view.view(batch_h, S, S)
    right_mat = right_view.view(batch_h, S, S)

    # Triton batched GEMM (left @ rightᵀ)
    out_mat = _tri_mul_gemm(left_mat, right_mat)   # (B*H, S, S)

    # reshape back to (B, S, S, H)
    out = out_mat.view(B, H, S, S).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8) Output LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 9) Output gate (element‑wise)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 10) Final linear projection back to original channel dimension
    # --------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B, S, S, C)

    # --------------------------------------------------------------
    # 11) Cast back to the original dtype (usually float32)
    # --------------------------------------------------------------
    if out.dtype != dtype:
        out = out.to(dtype)

    return out
```

This implementation follows the reference PyTorch module closely while off‑loading the dominant O(N³·H) contraction to a custom Triton kernel. The kernel processes each (batch × hidden) matrix in tiles of 64 × 64 (with a K‑tile of 32), accumulates in float32 for numerical stability, and writes the result back in float16. The remaining linear, LayerNorm, and gating operations are handled with PyTorch's highly‑optimized primitives, yielding a fast end‑to‑end forward pass.
