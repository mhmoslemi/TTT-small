"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor.
2. Fuse the left/right projection and gating linear layers:
   - Concatenate projection and gate weight matrices.
   - One linear call yields both projection and gate pre‑activations;
     split and apply sigmoid to the gate half.
3. Apply optional binary mask after gating (order does not affect the result).
4. Fuse left and right tensors (shape B,N,N,H) into a batch dimension
   (B*H,N,N) and compute the core O(N³·H) reduction with a custom
   Triton batched GEMM kernel that evaluates
        out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
   (equivalent to a batched matmul A @ Bᵀ).
5. Reshape the result back to (B,N,N,H), apply the output LayerNorm,
   multiply by the output gate and project back to the original channel
   dimension.
6. Return the result in float32.

Only the heavy O(N³·H) reduction is performed in Triton; all
Linear/LayerNorm/Gates use PyTorch (highly optimized) and the two
projections+gates are fused to halve the number of GEMM calls.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,            # pointers
    M, N, K,                        # matrix dimensions
    stride_ah, stride_am, stride_ak,  # A: batch, row, col
    stride_bh, stride_bk, stride_bn,  # B: batch, k, n  (B is accessed as K×N)
    stride_ch, stride_cm, stride_cn,  # C: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel: C[b,i,j] = Σ_k A[b,i,k] * B[b,k,j] (i.e. A @ Bᵀ)."""
    pid_b = tl.program_id(2)          # batch index
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column

    # ---- block offsets ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- batch base pointers ----
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # ---- pointer arithmetic for tiles ----
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K dimension ----
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)                     # (M,K) × (K,N) → (M,N)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ---- write back ----
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Wrapper for the batched GEMM Triton kernel.
    A: (batch, M, K)   fp16
    B: (batch, N, K)   fp16  (the K‑dimension will be treated as the reduction axis)
    Returns C of shape (batch, M, N) (fp16).
    """
    assert A.shape[0] == B.shape[0], "Batch size must match"
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2, "Inner dimensions must match"

    # Heuristic block sizes
    if M <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif M <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif M <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    _batched_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(2), B.stride(1),   # note: treat B as (batch, K, N)
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return C

            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict containing model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Configuration & shortcuts
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Cast everything to fp16 for the heavy path
    # ------------------------------------------------------------------

        input_tensor,

        bias=weights["norm.bias"].to(target_dtype),
    )
    x = x.to(target_dtype)

    # ------------------------------------------------------------------
    # Fuse projection + gate for left and right paths
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]

    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(target_dtype)   # (2*H, C)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(target_dtype)

    left_combined = F.linear(x, left_combined_w)   # (B,N,N,2H)
    right_combined = F.linear(x, right_combined_w) # (B,N,N,2H)

    # split
    left_proj = left_combined[..., :hidden_dim]
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])

    right_proj = right_combined[..., :hidden_dim]
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:])

    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Optional mask (applied after gating – order does not matter)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(target_dtype).unsqueeze(-1)          # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Output gate
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(target_dtype))
    )  # (B,N,N,H)

    # ------------------------------------------------------------------
    # Prepare tensors for batched GEMM: fold hidden dimension into batch
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B,H,N,N)

    left_flat = left.view(B * H, N, N)    # (B*H, N, K) where K == N
    right_flat = right.view(B * H, N, N)  # (B*H, N, K)

    # ------------------------------------------------------------------
    # Core O(N³·H) reduction via Triton (batched GEMM A @ Bᵀ)
    # ------------------------------------------------------------------
    # B is accessed as (batch, K, N) inside the kernel, so we keep it
    # in the original (batch, N, K) layout and simply swap the stride
    # arguments when launching the kernel.
    out_flat = _batched_gemm_triton(left_flat, right_flat)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(target_dtype),
        bias=weights["to_out_norm.bias"].to(target_dtype),
    )

    # ------------------------------------------------------------------
    # Apply output gate and final projection back to original channel dimension
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(
        out,
        weights["to_out.weight"].to(target_dtype)
    )  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # Cast to float32 for the final output
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
