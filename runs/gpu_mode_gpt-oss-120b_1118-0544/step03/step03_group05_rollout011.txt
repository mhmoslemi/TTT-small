def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Convert to fp16 for compute
    x = x.to(torch.float16)

    # 2. Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # mask
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # gate
    left = left * left_gate
    right = right * right_gate

    # Core matmul via Triton
    B, N, _, H = left.shape

    # Allocate result tensor
    out = torch.empty_like(left, dtype=torch.float16)  # same shape (B,N,N,H)

    # Get strides
    left_strides = left.stride()
    right_strides = right.stride()
    out_strides = out.stride()

    # Launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B * H,  # batch*hidden
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_matmul_kernel[grid](
        left, right, out,

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the supplied LN parameters.
2. Cast to fp16 for the compute‑heavy path.
3. Linear projections (left/right) and their gates (left_gate, right_gate, out_gate)
   are computed with torch.nn.functional.linear + sigmoid.
4. Optional binary mask is applied, then gates are fused into the projected tensors.
5. Core O(N³·H) reduction is performed by a custom Triton kernel that computes,
   for each batch element b and hidden channel h,
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   The kernel tiles the (i,j) dimensions (BLOCK_M×BLOCK_N) and iterates over k
   (BLOCK_K), accumulating in FP32 and storing the result in FP16.
6. The hidden‑dim output is LayerNorm‑ed, multiplied by the output gate,
   and finally projected back to the original channel dimension.
7. The result is cast back to the original dtype (usually float32).

Only the O(N³·H) reduction is implemented in Triton; all other steps use
high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the batched per‑channel matrix multiplication:
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
    num_stages: tl.constexpr,
):
    pid = tl.program_id(0)          # batch * hidden index
    pid_i = tl.program_id(1)        # tile row
    pid_j = tl.program_id(2)        # tile col

    # --------------------------------------------------------------
    # Compute batch (b) and hidden channel (h) from pid
    # --------------------------------------------------------------
    b = pid // H
    h = pid % H

    # --------------------------------------------------------------
    # Offsets for the i and j dimensions within their tiles
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------
    # Base pointers for this (b, h) pair
    # --------------------------------------------------------------
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # --------------------------------------------------------------
    # Accumulator (FP32) for the tile
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds accesses
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_base
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float32)

        # Load right tile: shape (BLOCK_N, BLOCK_K)
        right_ptrs = (
            right_base
            + offs_j[:, None] * stride_r_i   # stride for the row dimension (j)
            + offs_k[None, :] * stride_r_k
        )
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        right = right.to(tl.float32)

        # Accumulate: Σ_k left[i,k] * right[j,k]
        # right is transposed inside tl.dot via trans_b=True
        acc += tl.dot(left, right, trans_b=True)

    # --------------------------------------------------------------
    # Store the output tile (cast back to fp16)
    # --------------------------------------------------------------
    mask_store = mask_i[:, None] & mask_j[None, :]
    out_ptrs = (
        out_base
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_store)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict containing model weights (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (left/right) and gating projections
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"].to(torch.float16)
    )  # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"].to(torch.float16)
    )
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4) Optional binary mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Fuse gates into the projected tensors
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core O(N³·H) reduction via custom Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    assert H == hidden_dim, "Hidden dim mismatch"

    # Allocate output tensor (fp16)
    out = torch.empty_like(left, dtype=torch.float16)

    # Strides (in elements, not bytes)
    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out.stride()

    # Tiling parameters – chosen for good occupancy on Hopper GPUs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                         # batch * hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,  # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,  # tiles over j
    )

    tri_mul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 7) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate (element‑wise multiplication)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 10) Cast back to the input dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
