@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0) # combined batch and head flatten index
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch index and head index
    # flatten = pid_bh
    # we can compute batch = pid_bh // H, head = pid_bh % H
    b = pid_bh // H
    h = pid_bh % H

    # compute i and j block start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # offsets for i and j
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    # mask 
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (M x K)
        left = tl.load(
            left_ptr + b*stride_left_b + h*stride_left_h
                + offs_i[:, None] * stride_left_i
                + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )
        # load right tile transposed (K x N)
        right = tl.load(
            right_ptr + b*stride_right_b + h*stride_right_h
                + offs_k[:, None] * stride_right_k
                + offs_j[None, :] * stride_right_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )
        # dot
        acc += tl.dot(left, right, out_dtype=tl.float32)   # left (M,K), right (K,N) -> (M,N)
        # move k pointer increments? Not needed because we recompute offsets each iteration.

    # store result
    out = acc.to(tl.float16)  # or keep as float32; final out may be float16 then cast later.
    # store with final mask
    tl.store(
        out_ptr + b*stride_out_b + h*stride_out_h
            + offs_i[:, None] * stride_out_i
            + offs_j[None, :] * stride_out_j,
        out,
        mask=mask_i[:, None] & mask_j[None, :]
    )
```

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The kernel implements the O(N³·H) reduction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
directly on the GPU, avoiding the costly
permute‑view copy required for a batched GEMM.
All linear projections, layer‑norms and gates are performed with
torch.nn.functional for simplicity; the heavy reduction is fused
into a single Triton kernel.  The computation runs in fp16 with
float32 accumulation and the final result is cast back to the input
dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute a batched matrix multiplication over the third dimension of the
    pairwise tensors (the "k" dimension).  The inputs have shape
    [B, N, N, H] in memory order (batch, i, k, head) for both left and right.
    The output has the same shape.
    """
    pid_bh = tl.program_id(0)          # combined batch & head index
    pid_i  = tl.program_id(1)          # i‑tile index
    pid_j  = tl.program_id(2)          # j‑tile index

    # De‑compose batch and head
    b = pid_bh // H
    h = pid_bh % H

    # Starting coordinates of the tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets within the tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_idx in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_idx * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right tile transposed: shape (BLOCK_K, BLOCK_N)
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_i,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # Accumulate (FP16·FP16 → FP32)
        acc += tl.dot(left, right)   # left (M×K), right (K×N) → (M×N)

    # Write result back in fp16
    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j,
        out,
        mask=mask_i[:, None] & mask_j[None, :],
    )

    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Cast to fp16 for the heavy path
    # ------------------------------------------------------------------ #
    x = input_tensor.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 2. Input LayerNorm
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
    )

    # 3. Cast model weights to fp16
    # ------------------------------------------------------------------ #
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------ #
    # 4. Linear projections and optional masking
    # ------------------------------------------------------------------ #
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    left = left.contiguous()
    right = right.contiguous()
    B, N, _, H = left.shape  # shape is (B, N, N, H)

    out = torch.empty((B, N, N, H), device=left.device, dtype=torch.float16)

    # Strides (in elements, not bytes)
    lsb, lsi, lsk, lsh = left.stride()
    rsb, rsi, rsk, rsh = right.stride()
    osb, osi, osj, osh = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        lsb,
        lsi,
        lsk,
        lsh,
        rsb,
        rsi,
        rsk,
        rsh,
        osb,
        osi,
        osj,
        osh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------ #
    # 7. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------ #
    # 8. Apply the output gate
    # ------------------------------------------------------------------ #
    out = out * out_gate

    # ------------------------------------------------------------------ #
    # 9. Final projection back to the original channel dimension
    # ------------------------------------------------------------------ #
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype if necessary
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
