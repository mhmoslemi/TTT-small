"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Cast to FP16 for the heavy path.
3. Compute left/right projections and three gating signals (left, right, out)
   using linear layers and sigmoid activations.
4. Apply optional binary mask to the left/right tensors.
5. Fuse left/right gates by elementwise multiplication.
6. Fold the hidden dimension into the batch dimension and run a batched
   matrix multiplication kernel:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a GEMM C = A @ Bᵀ with A,B shaped (B*H, N, N).
7. Reshape back to (B, N, N, H), apply a second LayerNorm over H,
   multiply by the output gate, and project back to the original channel
   dimension C.
8. Return the result in the original dtype (float32).

Only the O(N³·H) reduction (step 6) is performed in Triton; the rest uses
high‑performance PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (A,B,C are FP16, acc in FP32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,   # A: batch, row, col
    stride_bh, stride_bk, stride_bn,   # B: batch, k (row), n (col)  (B is accessed as K×N)
    stride_ch, stride_cm, stride_cn,   # C: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch index (B*H)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column

    # Offsets within the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # Pointers to the current K‑tile (will be incremented inside the loop)
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper for the batched GEMM
# ----------------------------------------------------------------------
def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for FP16 tensors using a Triton kernel.
    A, B: (batch, N, N)  FP16
    Returns C of same shape/dtype.
    """
    assert A.shape == B.shape
    batch, N, K = A.shape
    assert N == K, "TriMul requires square sequence dimension."

    # Choose block sizes heuristically
    if N <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:  # up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows
        (N + BLOCK_N - 1) // BLOCK_N,   # cols
        batch,                          # batch * hidden
    )

    C = torch.empty_like(A)

    _batched_matmul_kernel[grid](
        A, B, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(2), B.stride(1),  # note the swap for K,N
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    return C

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Work in FP16 for the heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Optional mask (binary)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply gates to projections
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Fold hidden dimension into batch for the GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left = left.view(B * H, N, N)                 # (B*H, N, N)
    right = right.view(B * H, N, N)

    # ------------------------------------------------------------------
    # 7) Core reduction: C = left @ rightᵀ   (batched GEMM)
    # ------------------------------------------------------------------
    out_flat = _batched_matmul_triton(left, right)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 9) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 12) Cast to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
