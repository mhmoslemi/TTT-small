    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    eps = 1e-5

    # 1. layernorm
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)

    # Cast to fp16
    x_h = x.to(torch.float16)

    # 2. Combined projections
    left_comb_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16).to(device)
    right_comb_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16).to(device)

    left_comb = F.linear(x_h, left_comb_weight)  # [B,N,N,2*H]
    right_comb = F.linear(x_h, right_comb_weight)

    left_proj, left_gate_pre = torch.split(left_comb, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_comb, hidden_dim, dim=-1)

    # mask
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # out_gate
    out_gate_weight = weights["out_gate.weight"].to(torch.float16).to(device)
    out_gate_pre = F.linear(x_h, out_gate_weight)  # [B,N,N,H]
    out_gate = torch.sigmoid(out_gate_pre)

    # reshape for batched matmul
    # left: [B,N,N,H] -> [B,H,N,N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape  # note N repeated

    # flatten batch
    left_flat = left_perm.view(B*H, N, N)
    right_flat = right_perm.view(B*H, N, N)

    # prepare output fp32
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # strides
    a_batch_stride, a_stride_m, a_stride_k = left_flat.stride()
    b_batch_stride, b_stride_k, b_stride_n = right_flat.stride()
    c_batch_stride, c_stride_m, c_stride_n = out_flat.stride()

    # kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B*H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        a_batch_stride, a_stride_m, a_stride_k,
        b_batch_stride, b_stride_k, b_stride_n,
        c_batch_stride, c_stride_m, c_stride_n,
        M=N, N=N, K=N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B,N,N,H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # final layernorm (hidden_dim)
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    # gating
    out = out * out_gate.to(torch.float32)

    # final projection
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy N³ reduction is performed by a custom Triton kernel that computes,
for each batch and hidden dimension, the matrix product
    out[b, h] = left[b, h] @ right[b, h].T
where left and right have shape (N, N).  All other operations (layer‑norms,
linear projections, gating) are done with PyTorch (using fp16 where possible)
and the kernel writes the result directly in fp32, avoiding an extra cast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    a_batch_stride, a_stride_m, a_stride_k,   # strides for A (B*H, M, K)
    b_batch_stride, b_stride_k, b_stride_n,   # strides for B (B*H, K, N)
    c_batch_stride, c_stride_m, c_stride_n,   # strides for C (B*H, M, N)
    M, N, K,                                 # matrix dimensions
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B.T   (A,B: fp16, C: fp32)"""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointer to the start of the batch
    a_batch_offset = pid_batch * a_batch_stride
    b_batch_offset = pid_batch * b_batch_stride
    c_batch_offset = pid_batch * c_batch_stride

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m < M) & (offs_k < K)
        b_mask = (offs_k < K) & (offs_n < N)

        # Load A tile (M x K) and B tile (K x N)
        a_ptrs = a_ptr + a_batch_offset \
                 + offs_m[:, None] * a_stride_m \
                 + offs_k[None, :] * a_stride_k
        b_ptrs = b_ptr + b_batch_offset \
                 + offs_k[:, None] * b_stride_k \
                 + offs_n[None, :] * b_stride_n

        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # Accumulate the dot product
        acc += tl.dot(a, b)

    # Write result back in FP32
    c_ptrs = c_ptr + c_batch_offset \
             + offs_m[:, None] * c_stride_m \
             + offs_n[None, :] * c_stride_n
    c_mask = (offs_m < M) & (offs_n < N)
    tl.store(c_ptrs, acc, mask=c_mask)

    Returns:

    """

    left = left_proj * left_gate          # [B, N, N, H] fp16
    right = right_proj * right_gate       # [B, N, N, H] fp16

    # ----------------------------------------------------
    # 6) Output gate (fp16)
    # ----------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(torch.float16).to(device)
    out_gate_pre = F.linear(x_h, out_gate_weight)  # [B, N, N, H] fp16
    out_gate = torch.sigmoid(out_gate_pre)        # fp16

    # ----------------------------------------------------
    # 7) Batched matrix multiplication via Triton:
    #    out[b, h] = left[b, h] @ right[b, h].T
    # ----------------------------------------------------
    # Rearrange to (B, H, N, N) and flatten batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Output buffer (FP32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # Strides (in elements)
    a_batch_stride, a_stride_m, a_stride_k = left_flat.stride()
    b_batch_stride, b_stride_k, b_stride_n = right_flat.stride()
    c_batch_stride, c_stride_m, c_stride_n = out_flat.stride()

    # Tile sizes – tuned for B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        a_batch_stride,
        a_stride_m,
        a_stride_k,
        b_batch_stride,
        b_stride_k,
        b_stride_n,
        c_batch_stride,
        c_stride_m,
        c_stride_n,
        M=N,
        N=N,
        K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------
    # 8) Final LayerNorm, gating and projection (float32)
    # ----------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)          # apply output gate
    out = F.linear(out, weights["to_out.weight"])   # final projection (dim)

    return out
