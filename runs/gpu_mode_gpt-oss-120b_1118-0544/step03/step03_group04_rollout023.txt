@triton.jit
def matmul_kernel(A_ptr, B_ptr, C_ptr,
                  M, N, K,
                  stride_am, stride_ak, stride_ab,
                  stride_bk, stride_bn, stride_bb,
                  stride_cm, stride_cn, stride_cb,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
                  BLOCK_K: tl.constexpr):
   pid_b = tl.program_id(0) # batch
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # Compute offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   # mask for bounds
   offs_m_mask = offs_m < M
   offs_n_mask = offs_n < N

   # Load accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # Loop over k
   for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        # mask for k dimension
        k_mask = offs_k < K

        # Pointers
        a_ptrs = A_ptr + pid_b * stride_ab + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = B_ptr + pid_b * stride_bb + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        # load
        a = tl.load(a_ptrs, mask=offs_m_mask[:, None] & k_mask[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & offs_n_mask[None, :], other=0.0)
        # dot
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)
   # write back
   c_ptrs = C_ptr + pid_b * stride_cb + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
   tl.store(c_ptrs, acc.to(tl.float16), mask=offs_m_mask[:, None] & offs_n_mask[None, :])

@triton.jit
def matmul_outgoing_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
   pid_b = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # Compute offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   # mask for out-of-bounds
   mask_m = offs_m < M
   mask_n = offs_n < N

   # Initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # Loop over k
   for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # compute pointer offsets
        a_ptrs = A + pid_b * stride_ab + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = B + pid_b * stride_bb + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        # Load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # compute dot product (matrix multiplication)
        acc += tl.dot(a, b)

   # Write result
   c_ptrs = C + pid_b * stride_cb + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
   tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy O(N³·H) reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is rewritten as a batched matrix multiplication:
    left[b,h]  : (N, N)
    right[b,h] : (N, N)   → we multiply left with rightᵀ.
A custom Triton kernel performs this batched GEMM on half‑precision
tensors (accumulating in fp32) and therefore replaces the slow
torch.bmm used in the reference implementation.  All other operations
(LayerNorm, Linear, gating, final projection) stay in PyTorch but are
executed in fp16 for bandwidth reasons.  The kernel is launched with
a 3‑D grid (batch, block‑row, block‑col) and uses tile sizes
BLOCK_M = BLOCK_N = 128, BLOCK_K = 32 which give good occupancy on a
B200 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_outgoing_kernel(
    A, B, C,                     # pointers
    M, N, K,                     # matrix dimensions
    stride_am, stride_ak, stride_ab,   # A strides: (row, col, batch)
    stride_bk, stride_bn, stride_bb,   # B strides (col, row, batch) – transposed view
    stride_cm, stride_cn, stride_cb,   # C strides (row, col, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C[batch, i, j] = Σ_k A[batch, i, k] * B[batch, j, k]   (B is used transposed)."""
    pid_b = tl.program_id(0)          # batch index (B * H)
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # -------------------------------------------------------------------------
    # 1) Compute tile offsets
    # -------------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------------------------------------------------------------------------
    # 2) Accumulator
    # -------------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------------------
    # 3) Loop over K dimension
    # -------------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers with proper strides (including batch offset)
        a_ptrs = A + pid_b * stride_ab + (offs_m[:, None] * stride_am
                                          + offs_k[None, :] * stride_ak)
        b_ptrs = B + pid_b * stride_bb + (offs_k[:, None] * stride_bk
                                          + offs_n[None, :] * stride_bn)

        # Load tiles (out‑of‑bounds positions are masked with zero)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Tile‑wise matrix product
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N) in fp32

    # -------------------------------------------------------------------------
    # 4) Write result (store as fp16)
    # -------------------------------------------------------------------------
    c_ptrs = C + pid_b * stride_cb + (offs_m[:, None] * stride_cm
                                      + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_outgoing(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the outgoing TriMul reduction.
    A, B : tensors of shape (batch, N, N), dtype = torch.float16, contiguous.
    Returns C with same shape and dtype.
    """
    assert A.shape == B.shape
    batch, N, _ = A.shape
    M = N
    K = N

    # output tensor
    C = torch.empty_like(A)

    # Strides in elements (not bytes)
    stride_am, stride_ak, stride_ab = A.stride(1), A.stride(2), A.stride(0)
    # B is used transposed: original stride (batch, row=j, col=k)
    # we need stride over k (col) and over j (row) after transpose
    stride_bk = B.stride(2)   # stride of k (original col)
    stride_bn = B.stride(1)   # stride of j (original row)
    stride_bb = B.stride(0)

    stride_cm, stride_cn, stride_cb = C.stride(1), C.stride(2), C.stride(0)

    # Tile size – compile‑time constants
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid dimensions
    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _batched_matmul_outgoing_kernel[grid](
        A, B, C,
        M, N, K,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (binary) or None
            - weights: dict of model weights (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, C] with the same dtype as the input.
    """
    torch.set_grad_enabled(False)

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------------------------------
    # 1) Input LayerNorm
    # -------------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )
    # Work in fp16 for the heavy path
    x = x.to(torch.float16)

    # -------------------------------------------------------------------------
    # 2) Cast model weights to fp16
    # -------------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, torch.float16)

    # -------------------------------------------------------------------------
    # 3) Linear projections and optional masking
    # -------------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w) # (B, N, N, H)

    if mask is not None and not nomask:
        mask_f = mask.to(device, dtype=torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------------------------------
    # 4) Gate projections (sigmoid)
    # -------------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 5) Core O(N³·H) reduction via Triton batched GEMM
    #    left, right : (B, N, N, H)   →  (B, H, N, N) → (B*H, N, N)
    # -------------------------------------------------------------------------
    B, N, _, H = left.shape
    # rearrange to (B, H, N, N) and make contiguous
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch * hidden_dim
    left_flat = left.view(-1, N, N)   # (B*H, N, N)
    right_flat = right.view(-1, N, N) # (B*H, N, N)

    # batched matrix multiplication (left @ rightᵀ)
    out_flat = _batched_matmul_outgoing(left_flat, right_flat)  # (B*H, N, N), fp16

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------------------
    # 6) Hidden‑dim LayerNorm
    # -------------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------------------------------
    # 7) Apply output gate (element‑wise)
    # -------------------------------------------------------------------------
    out = out * out_gate

    # -------------------------------------------------------------------------
    # 8) Final projection back to original channel dimension
    # -------------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
