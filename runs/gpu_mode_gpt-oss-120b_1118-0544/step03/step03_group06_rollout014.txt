"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1️⃣ Layer‑norm the input tensor (B,N,N,C) with the supplied LN parameters.
2️⃣ Cast to FP16 for the compute‑heavy region.
3️⃣ Linear projections (left/right) and three gating projections (left_gate,
   right_gate, out_gate) are performed with the supplied weights (bias‑free).
4️⃣ An optional binary mask (if provided and not disabled) is applied before
   the gating operation.
5️⃣ Gates (sigmoid) are multiplied into the projected tensors.
6️⃣ Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (left @ rightᵀ) where the hidden dimension
   `h` (hidden_dim) is fused into the batch dimension.
   The heavy O(N³·H) work is done by a custom Triton kernel.
7️⃣ Reshape back to (B,N,N,hidden_dim) and apply a second LayerNorm.
8️⃣ Multiply by the output gate, project back to the original channel dimension
   (dim) and cast to the input dtype.

Both the kernel and the surrounding PyTorch code use mixed precision
(FP16 for compute, FP32 for the final output) to obtain high throughput on
modern GPUs (e.g. NVIDIA B200).
"""

# ----------------------------------------------------------------------

    M, N, K,
    stride_ah, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bh, stride_bk, stride_bn,   # B: (batch, K, N)  (originally (batch,N,K))
    stride_ch, stride_cm, stride_cn,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)        # batch * hidden
    pid_m = tl.program_id(0)        # tile row (i)
    pid_n = tl.program_id(1)        # tile col (j)

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for this batch
    # ------------------------------------------------------------------
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # Pointers to the start of the tile (K dimension will be stepped inside loop)
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # tl.dot performs the MAC; accumulation in FP32
        acc += tl.dot(a, b)

        # Increment pointers for next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write result back (cast to FP16)
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Performs batched matrix multiplication C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]
    using the Triton kernel defined above.

    A, B : fp16 tensors of shape (batch, N, N)
    Returns C of the same shape and dtype.
    """
    assert A.shape == B.shape, "A and B must have identical shapes"
    batch, N, K = A.shape
    assert N == K, "TriMul expects square sequence dimension"

    # Choose tile sizes – they work well up to N=1024
    BLOCK_M = BLOCK_N = 64
    BLOCK_K = 32
    # For very small N we can increase tile size to reduce launch overhead
    if N <= 128:
        BLOCK_M = BLOCK_N = 128

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    C = torch.empty_like(A)  # fp16 output

    _batched_matmul_kernel[grid](
        A, B, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),
        # B is interpreted as (batch, K, N) → swap last two strides
        B.stride(0), B.stride(2), B.stride(1),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask
    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # ---------------------------- unpack ---------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------- LayerNorm --------------------------------
    # Input LN over channel dimension C
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------- precision --------------------------------
    # Use FP16 for the heavy part
    x = x.to(torch.float16)

    # --------------------------- weight casting ----------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------- linear projections -------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------ mask ---------------------------------
    if mask is not None and not nomask:
        # mask shape (B, N, N) -> (B, N, N, 1) broadcast over hidden dim
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------- gating ----------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -------------------------- reshape for GEMM -------------------------
    B, N, _, H = left.shape                # H == hidden_dim
    # (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    # Collapse batch and hidden -> (B*H, N, N)
    left = left.view(B * H, N, N)
    right = right.view(B * H, N, N)

    # ----------------------- batched GEMM (Trident) ----------------------
    out_flat = _batched_matmul_triton(left, right)   # (B*H, N, N)

    # ---------------------------- reshape back ---------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # -------------------------- to_out_norm -----------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------- output gate -----------------------------
    out = out * out_gate

    # -------------------------- final projection ------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ---------------------- cast back to original dtype -----------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
