Pseudo for kernel:

```
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bm, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    BATCH, M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute batch index, block row and column indices
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n
    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    block_m = block_idx // num_blocks_n
    block_n = block_idx % num_blocks_n

    # Compute offsets for each tile
    # Starting offsets for this block
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulator 
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute batch offset
    a_batch_offset = batch_idx * batch_stride_a
    b_batch_offset = batch_idx * batch_stride_b
    c_batch_offset = batch_idx * batch_stride_c

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # offsets for k dimension within block
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Compute loading pointers with masks
        a_ptrs = a_ptr + a_batch_offset + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + b_batch_offset + offs_k[:, None] * stride_bm + offs_n[None, :] * stride_bn

        # Load with masking for out-of-bound tiles
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0).to(tl.float32)
        # Dot product
        acc += tl.dot(a, b)

    # Store results
    c_ptrs = c_ptr + c_batch_offset + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device

    # 1. Input LayerNorm (float32)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=1e-5)
    # 2. Cast to fp16 for heavy ops
    x_h = x.half()

    # 3. Linear projections (fp16)
    left_proj = F.linear(x_h, weights["left_proj.weight"].half())
    right_proj = F.linear(x_h, weights["right_proj.weight"].half())
    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].half())
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].half())
    out_gate_pre = F.linear(x_h, weights["out_gate.weight"].half())

    # 4. Apply mask if present
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # 5. Apply gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    left = left_proj * left_gate
    right = right_proj * right_gate

    out_gate = torch.sigmoid(out_gate_pre)

    # 6. Permute tensors to batch over hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_perm.shape

    # 7. Allocate output buffer in fp16
    out_fp16 = torch.empty((B, H, N, N), dtype=torch.float16, device=device)

    # 8. Compute strides
    a_batch_stride = left_perm.stride(0)
    stride_am = left_perm.stride(2)   # row stride (i)
    stride_ak = left_perm.stride(3)   # col stride (k)

    b_batch_stride = right_perm.stride(0)
    # want B_T: row stride = stride for original col (k) = right_perm.stride(3)
    # col stride = stride for original row (j) = right_perm.stride(2)
    stride_bm = right_perm.stride(3)  # row (k)
    stride_bn = right_perm.stride(2)  # col (j)

    c_batch_stride = out_fp16.stride(0)
    stride_cm = out_fp16.stride(2)   # row (i)
    stride_cn = out_fp16.stride(3)   # col (j)

    # 9. Launch Triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    # number of blocks in M,N
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * H * num_blocks_m * num_blocks_n
    grid = (total_programs,)

    batched_matmul_kernel[grid](
        a_ptr=left_perm,
        b_ptr=right_perm,
        c_ptr=out_fp16,
        batch_stride_a=a_batch_stride,
        stride_am=stride_am,
        stride_ak=stride_ak,
        batch_stride_b=b_batch_stride,
        stride_bm=stride_bm,
        stride_bn=stride_bn,
        batch_stride_c=c_batch_stride,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BATCH=B*H,
        M=N,
        N=N,
        K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        GROUP_SIZE_M=8,
        num_warps=4,
    )

    # 10. Rearrange output to original shape
    out = out_fp16.permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] fp16

    # 11. Convert to fp32 for final processing
    out = out.float()

    # 12. Final LayerNorm (hidden_dim)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # 13. Apply out_gate (already fp16, cast to fp32)
    out = out * out_gate.float()

    # 14. Output projection (float32)
    out = F.linear(out, weights["to_out.weight"])

    return out

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bm, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_SIZE_M: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute number of blocks
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch

    # reorder to reduce warp divergence following the grouping pattern
    # this part can be optional, but has better caching
    # Use group size to reorder block indices
    # Compute block_m and block_n linear
    # For simplicity, we use:
    block_m = block_idx // num_blocks_n
    block_n = block_idx % num_blocks_n

    # compute offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch_ptr = a_ptr + batch_idx * batch_stride_a
    b_batch_ptr = b_ptr + batch_idx * batch_stride_b
    c_batch_ptr = c_ptr + batch_idx * batch_stride_c

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # compute pointers for tile of A and B
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bm + offs_n[None, :] * stride_bn

        # load blocks
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        # convert to fp16 if not already? Already half.
        # do dot
        acc += tl.dot(a, b)  # default accumulation in fp32

    # store result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

"""
Outgoing Triangle Multiplicative (TriMul) update.

The heavy O(N³) part (L @ Rᵀ for each hidden channel) is implemented as a
batched matrix‑multiply Triton kernel.  Input tensors are layer‑normed and
projected in fp16, then the custom kernel computes the per‑channel matrix
multiplication using Tensor Cores.  The result is cast back to fp32 for the
final layer‑norm, gating and output projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    batch_stride_a, stride_am, stride_ak,    # A strides
    batch_stride_b, stride_bk, stride_bn,    # B (transposed) strides
    batch_stride_c, stride_cm, stride_cn,    # C strides
    M, N, K,                                 # matrix sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for many independent (M×K)·(K×N) matmuls."""
    # program ids
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch (here batch = B*hidden_dim)

    # compute start offsets of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute batch base pointers
    a_batch = a_ptr + pid_b * batch_stride_a
    b_batch = b_ptr + pid_b * batch_stride_b
    c_batch = c_ptr + pid_b * batch_stride_c

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )

        # Bᵀ tile: we treat B as already transposed, so rows = k, cols = n.
        # B tile shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )

        # dot product (tensor‑core friendly)
        acc += tl.dot(a, b)

    # store result, cast back to fp16
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass using a Triton kernel for the
    O(N³) matrix multiplication.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float) or None
            - weights      : dict of model parameters
            - config       : dict containing 'dim' and 'hidden_dim' and optional 'nomask'

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ----------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ----------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim], float32

    # ----------------------------------------------------
    # 2) Cast to fp16 for the heavy part
    # ----------------------------------------------------
    x_h = x.half()  # [B, N, N, dim], fp16

    # ----------------------------------------------------
    # 3) Linear projections + gates (fp16)
    # ----------------------------------------------------
    left_proj = F.linear(x_h, weights["left_proj.weight"].half())
    right_proj = F.linear(x_h, weights["right_proj.weight"].half())
    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].half())
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].half())
    out_gate_pre = F.linear(x_h, weights["out_gate.weight"].half())

    # ----------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # ----------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ----------------------------------------------------
    # 5) Apply gating (sigmoid, fp16)
    # ----------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate   # [B, N, N, H], fp16
    right = right_proj * right_gate  # [B, N, N, H], fp16

    out_gate = torch.sigmoid(out_gate_pre)  # [B, N, N, H], fp16

    # ----------------------------------------------------
    # 6) Rearrange to (batch, hidden, N, N) and make contiguous
    # ----------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    total_batches = B * H

    # flatten the first two dimensions for the kernel
    left_flat = left_perm.view(total_batches, N, N)
    right_flat = right_perm.view(total_batches, N, N)

    # ----------------------------------------------------
    # 7) Allocate output buffer (fp16)
    # ----------------------------------------------------
    out_fp16 = torch.empty((total_batches, N, N), dtype=torch.float16, device=device)

    # ----------------------------------------------------
    # 8) Strides for the kernel (in elements, not bytes)
    # ----------------------------------------------------
    a_batch_stride = left_flat.stride(0)
    stride_am = left_flat.stride(1)   # row stride (i)
    stride_ak = left_flat.stride(2)   # col stride (k)

    b_batch_stride = right_flat.stride(0)
    # B is used transposed inside the kernel: rows = k, cols = j
    stride_bk = right_flat.stride(2)   # row stride for k
    stride_bn = right_flat.stride(1)   # col stride for j

    c_batch_stride = out_fp16.stride(0)
    stride_cm = out_fp16.stride(1)   # row stride (i)
    stride_cn = out_fp16.stride(2)   # col stride (j)

    # ----------------------------------------------------
    # 9) Launch the Triton kernel (batched matmul)
    # ----------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (num_blocks_m, num_blocks_n, total_batches)
    batched_matmul_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_fp16,
        batch_stride_a=a_batch_stride,
        stride_am=stride_am,
        stride_ak=stride_ak,
        batch_stride_b=b_batch_stride,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        batch_stride_c=c_batch_stride,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        M=N,
        N=N,
        K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ----------------------------------------------------
    # 10) Reshape back to [B, N, N, H] (still fp16)
    # ----------------------------------------------------
    out = out_fp16.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ----------------------------------------------------
    # 11) Convert to fp32 for the final LayerNorm and projection
    # ----------------------------------------------------
    out = out.float()  # [B, N, N, H], fp32

    # ----------------------------------------------------
    # 12) Final LayerNorm over the hidden dimension
    # ----------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ----------------------------------------------------
    # 13) Apply output gate (cast to fp32) and final linear projection
    # ----------------------------------------------------
    out = out * out_gate.float()
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim], fp32

    return out
```
