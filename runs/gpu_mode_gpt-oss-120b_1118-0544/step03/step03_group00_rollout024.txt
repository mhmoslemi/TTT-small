"""
TriMul (outgoing) forward pass implemented with Triton.
Algorithm:
1. LayerNorm over the channel dimension.
2. Linear projections (left/right) and gate projections (left_gate, right_gate,
   out_gate) using FP16 weights.
3. Fuse element‑wise mask, gate and projection via a small Triton kernel.
4. Core O(N³·H) reduction: for each batch and hidden channel compute
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]  (= left @ rightᵀ).
   Implemented as a batched GEMM kernel in Triton that processes all
   (batch, hidden) slices together with block‑wise tiling.
5. LayerNorm on the hidden dimension, multiply by out_gate, final linear
   projection back to the original channel dimension.
All heavy compute is in FP16 (accumulated in FP32 for numerical stability);
the final result is returned in the input dtype.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel: elementwise proj * gate * mask (mask optional)
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,     # (B*N*N*H,) fp16
    gate_ptr,     # (B*N*N*H,) fp16
    mask_ptr,     # (B*N*N,)   fp16 (0/1) or empty if unused
    out_ptr,      # (B*N*N*H,) fp16
    total_elems,  # int = B*N*N*H
    H,            # hidden dim (for broadcasting mask)
    use_mask: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # compute mask index (broadcast over H)
    mask_idx = offs // H

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid & use_mask, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def apply_gate_and_mask(proj, gate, mask, block_size=1024):
    """Fuse proj*gate*mask across the hidden dimension."""
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        use_mask=True,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel: batched matrix multiplication A @ Bᵀ for many (batch,hidden) slices.
#   A, B: shape (BH, N, N) where BH = B * H, stored in row‑major order.
#   C:   shape (BH, N, N), accumulation in FP32, result stored as FP32.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    N: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)          # BH index
    pid_m = tl.program_id(1)              # block row index
    pid_n = tl.program_id(2)              # block col index

    # starting offsets of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A block: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + batch_idx * stride_a_batch
            + offs_m[:, None] * stride_a_row
            + offs_k[None, :] * stride_a_col
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Bᵀ block: (BLOCK_K, BLOCK_N)
        # Bᵀ[k, j] = B[j, k] – we fetch B with swapped indices.
        b_ptrs = (
            b_ptr
            + batch_idx * stride_b_batch
            + offs_n[None, :] * stride_b_row   # j * row_stride
            + offs_k[:, None] * stride_b_col   # k * col_stride
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = (
        c_ptr
        + batch_idx * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: Tensor[bs, seq_len, seq_len, dim]
            - mask: Tensor[bs, seq_len, seq_len] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor[bs, seq_len, seq_len, dim]  (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all weight tensors to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)    # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (same shape) + sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Fuse mask (if present) and gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    We treat each (batch, hidden) pair as a separate matrix
    #    multiplication: A @ Bᵀ where A,B ∈ ℝ^{N×N}
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B, H, N, N) -> contiguous -> (BH, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H

    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # strides for the kernel (in elements)
    stride_a_batch = left_perm.stride(0)   # N*N
    stride_a_row   = left_perm.stride(2)   # N
    stride_a_col   = left_perm.stride(3)   # 1

    stride_b_batch = right_perm.stride(0)  # N*N
    stride_b_row   = right_perm.stride(2)  # N   (original row stride, used for j)
    stride_b_col   = right_perm.stride(3)  # 1   (original col stride, used for k)

    stride_c_batch = out_flat.stride(0)    # N*N
    stride_c_row   = out_flat.stride(1)    # N
    stride_c_col   = out_flat.stride(2)    # 1

    # Tunable block sizes (feel free to adjust)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (BH, num_blocks_m, num_blocks_n)

    tri_mul_matmul_kernel[grid](
        left_perm,
        right_perm,
        out_flat,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) float32

    # ------------------------------------------------------------------
    # 8. Output LayerNorm (over hidden dimension), out_gate and final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # out_gate is fp16; promote to fp32 for multiplication
    out = out * out_gate.to(torch.float32)

    # final linear projection back to original dim
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim), fp16
    out = out.to(dtype_in)  # restore original dtype

    return out
