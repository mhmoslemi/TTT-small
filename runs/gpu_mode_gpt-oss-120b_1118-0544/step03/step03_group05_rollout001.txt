def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)
    B,N,Nt, C = input_tensor.shape; confirm N==Nt.

    # layer norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,),
        weight=weights["norm.weight"], bias=weights["norm.bias"])

    # cast to half
    x = x.to(torch.float16)

    # combine weights
    # all weights have shape (hidden_dim, dim) except out_gate weight also (hidden_dim, dim)
    # order: left_proj, right_proj, left_gate, right_gate, out_gate
    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)

    # linear projection - shape (B,N,N,5*hidden_dim)
    proj = torch.nn.functional.linear(x, combined_weight)  # no bias

    # Split
    split_tensors = torch.split(proj, hidden_dim, dim=-1)
    left, right, left_gate_raw, right_gate_raw, out_gate_raw = split_tensors

    # mask
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # gating: apply sigmoid
    left_gate = left_gate_raw.sigmoid()
    right_gate = right_gate_raw.sigmoid()
    out_gate = out_gate_raw.sigmoid()

    left = left * left_gate
    right = right * right_gate

    # core contraction
    B, N, _, H = left.shape

    # reshape for batched matmul
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)

    # matmul: left_mat @ right_mat^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H,N,N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # output layernorm
    out = torch.nn.functional.layer_norm(out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"])

    # multiply with out_gate (elementwise)
    # Using Triton kernel
    out = _elementwise_mul(out, out_gate)  # ensure shape matches

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # cast back
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor using the provided LN parameters.
2. Cast to fp16 for compute‑heavy operations.
3. Fuse the five Linear projections (left, right, left_gate, right_gate,
   out_gate) into a single matmul by concatenating their weight matrices.
4. Split the concatenated result, apply optional binary mask, and fuse the
   sigmoid gates into the projected tensors.
5. Core O(N³·H) contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a single batched GEMM (torch.bmm) on tensors of shape
   (B·H, N, N).
6. Apply a second Layer‑Norm over the hidden dimension.
7. Multiply the result by the output gate using a tiny Triton kernel
   (element‑wise multiplication).
8. Final linear projection back to the original channel dimension.
9. Cast back to the input dtype and return.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for element‑wise multiplication (used for the out‑gate)
# ----------------------------------------------------------------------
@triton.jit
def _elem_mul_kernel(a_ptr, b_ptr, c_ptr, total_elems,
                     BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < total_elems
    a = tl.load(a_ptr + offsets, mask=mask, other=0.0)
    b = tl.load(b_ptr + offsets, mask=mask, other=0.0)
    tl.store(c_ptr + offsets, a * b, mask=mask)

def _elem_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton kernel to compute a * b (both fp16, same shape)."""
    assert a.shape == b.shape
    total = a.numel()
    c = torch.empty_like(a)
    BLOCK_SIZE = 1024  # fits 1024 threads per block
    grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    _elem_mul_kernel[grid](
        a, b, c, total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    return c


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            mask: torch.Tensor [B, N, N] (or None)
            weights: dict of model parameters (already on CUDA)
            config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Fuse five Linear projections into one matmul
    # ------------------------------------------------------------------
    # Concatenate weight matrices along the output dimension:
    # order = [left_proj, right_proj, left_gate, right_gate, out_gate]
    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)   # shape (5*hidden_dim, dim)

    # Linear projection (no bias)
    proj = torch.nn.functional.linear(x, combined_weight)  # (B, N, N, 5*H)

    # ------------------------------------------------------------------
    # 4) Split into individual tensors
    # ------------------------------------------------------------------
    left, right, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=-1
    )  # each (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 6) Gating (sigmoid) and fuse into left/right
    # ------------------------------------------------------------------
    left_gate = left_gate_raw.sigmoid()
    right_gate = right_gate_raw.sigmoid()
    out_gate = out_gate_raw.sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Core O(N³·H) contraction via a single batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matrix multiplication (left_mat @ right_mat^T)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 9) Apply output gate using Triton kernel
    # ------------------------------------------------------------------
    out = _elem_mul(out, out_gate)

    # ------------------------------------------------------------------
    # 10) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float16)
    )  # (B, N, N, C)

    # ------------------------------------------------------------------
    # 11) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
