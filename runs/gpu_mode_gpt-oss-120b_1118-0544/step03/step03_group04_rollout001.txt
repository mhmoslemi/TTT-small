    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    device = input_tensor.device

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast to half for compute
    x = x.to(torch.float16)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # Reshape for batched GEMM
    B, N, _, H = left.shape  # left shape (B,N,N,H)
    # Permute to (B,H,N,N)
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()

    left_flat = left_perm.view(B*H, N, N)
    right_flat = right_perm.view(B*H, N, N)

    # GPU compute matmul using Triton
    # allocate output
    out_flat = torch.empty_like(left_flat)  # half

    # Compute strides
    stride_am = left_flat.stride(1)  # stride between rows (M dimension)

    stride_bk = right_flat.stride(1) # stride for K dimension? We'll treat B as transposed.

    batch_stride = left_flat.stride(0)

    M = N
    K = N

    pid_m = tl.program_id(1)

    mask_n = offs_n < N

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy O(N³·H) reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a batched matrix multiplication (left @ rightᵀ) with a
custom Triton kernel.  All other operations (LayerNorm, linear projections,
gates, final projection) are expressed with native PyTorch kernels in
half‑precision for speed, while the final result is cast back to the input
dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ Bᵀ  where A,B have shape (batch, M, K) and (batch, N, K)."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block   [M, K]
        a = tl.load(
            a_ptr
            + batch_id * batch_stride_a
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load Bᵀ block  [K, N] (B is accessed transposed)
        b = tl.load(
            b_ptr
            + batch_id * batch_stride_b
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate FP16 dot product into FP32 accumulator
        acc += tl.dot(a, b)

    # Write result back (cast to FP16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + batch_id * batch_stride_c
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int)          – input channel dimension C
                * "hidden_dim" (int)   – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------ #
    # Unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Work in FP16 for the compute‑heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 2. Cast model weights to FP16 once
    # ------------------------------------------------------------------ #
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------ #
    # 3. Linear projections + optional mask
    # ------------------------------------------------------------------ #
    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w) # (B, N, N, H)

    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------ #
    # 4. Gating
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Core O(N³·H) reduction via Triton batched GEMM
    # ------------------------------------------------------------------ #
    B, N, _, H = left.shape  # left: (B, N, N, H)

    # Prepare tensors of shape (B*H, N, N) for the batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(B * H, N, N)             # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat)               # FP16

    # Strides for the kernel (row‑major layout)
    stride_am = left_flat.stride(1)   # = N
    stride_ak = left_flat.stride(2)   # = 1
    stride_bk = right_flat.stride(2)  # = 1  (column stride, used as K‑dim)
    stride_bn = right_flat.stride(1)  # = N  (row stride, used as N‑dim)
    stride_cm = out_flat.stride(1)    # = N
    stride_cn = out_flat.stride(2)    # = 1
    batch_stride = left_flat.stride(0)  # = N*N (same for a, b, c)

    M = N
    K = N
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                              # batch * hidden
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        batch_stride,
        batch_stride,
        batch_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 6. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------ #
    # 7. Apply output gate (element‑wise)
    # ------------------------------------------------------------------ #
    out = out * out_gate

    # ------------------------------------------------------------------ #
    # 8. Final linear projection back to original dimension
    # ------------------------------------------------------------------ #
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype if needed
    if out.dtype != dtype:
        out = out.to(dtype)

    return out
