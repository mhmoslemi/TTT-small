    M, N, K,

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.
---------------------------------------------------------

The heavy N³ reduction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is implemented as a batched half‑precision GEMM:
    left  : (B,H,N,N)  rows = i, cols = k
    right : (B,H,N,N)  rows = k, cols = j   (pre‑transposed)
    out   : (B,H,N,N) = left @ right
A custom Triton kernel performs the batched GEMM with fp16 inputs,
fp32 accumulation and stores the result back as fp16.  All other
operations (LayerNorms, linear projections, sigmoids, gating) are
handled in PyTorch (mostly in fp16) to keep the kernel focussed on the
compute‑bound N³ part.

The final output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                        # matrix dimensions (all equal to seq_len)
    stride_az, stride_am, stride_ak,  # strides for A (batch, row, col)
    stride_bz, stride_bk, stride_bn,  # strides for B (batch, row, col) – B is already transposed
    stride_cz, stride_cm, stride_cn,  # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ B   where A: (batch, M, K), B: (batch, K, N)
    All tensors are fp16, accumulation in fp32.
    """
    pid_z = tl.program_id(2)        # batch * hidden index
    pid_m = tl.program_id(0)        # block row
    pid_n = tl.program_id(1)        # block col

    # -------------------------------------------------
    # Offsets within the tile
    # -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch*hidden slice
    a_base = a_ptr + pid_z * stride_az
    b_base = b_ptr + pid_z * stride_bz
    c_base = c_ptr + pid_z * stride_cz

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Loop over K dimension
    # -------------------------------------------------
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K

        # Pointers with row/col offsets
        a_ptrs = a_base + (offs_m[:, None] * stride_am) + ((k_offset + offs_k)[None, :] * stride_ak)
        b_ptrs = b_base + ((k_offset + offs_k)[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        # Load with masks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & ((k_offset + offs_k) < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=((k_offset + offs_k)[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # -------------------------------------------------
    # Store result back to C
    # -------------------------------------------------
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) – may be None
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict containing at least "dim" and "hidden_dim"
                     optional "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------
    # 1) Input LayerNorm (float32)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # -------------------------------------------------
    # 2) Cast to fp16 for the heavy work
    # -------------------------------------------------
    x_h = x.to(torch.float16)

    # -------------------------------------------------
    # 3) Linear projections (fp16)
    # -------------------------------------------------
    left_proj = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Optional mask (broadcast over hidden dim)
    if (not nomask) and (mask is not None):
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # -------------------------------------------------
    # 4) Gating (sigmoid, fp16)
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].to(torch.float16)))
    left = left_proj * left_gate      # [B, N, N, H] fp16
    right = right_proj * right_gate   # [B, N, N, H] fp16

    # -------------------------------------------------
    # 5) Output gate (fp16)
    # -------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))  # [B,N,N,H]

    # -------------------------------------------------
    # 6) Prepare tensors for batched GEMM
    # -------------------------------------------------
    # left: rows=i, cols=k   -> shape [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: rows=k, cols=j (transpose for matmul) -> shape [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    batch_hidden = B * H

    # Collapse batch*head dimension
    left_flat = left_perm.view(batch_hidden, N, N)   # [BH, M=N, K=N]
    right_flat = right_perm.view(batch_hidden, N, N) # [BH, K=N, N]

    # Output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # -------------------------------------------------
    # 7) Launch Triton batched GEMM kernel
    # -------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch_hidden,
    )

    # Strides (in elements) for the flattened tensors
    stride_az, stride_am, stride_ak = left_flat.stride()
    stride_bz, stride_bk, stride_bn = right_flat.stride()
    stride_cz, stride_cm, stride_cn = out_flat.stride()

    batched_gemm_fp16_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_az, stride_am, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # -------------------------------------------------
    # 8) Reshape back to [B, N, N, H] and cast to fp32
    # -------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out_fp32 = out.to(torch.float32)

    # -------------------------------------------------
    # 9) Final LayerNorm, gating, and projection (fp32)
    # -------------------------------------------------
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate.to(torch.float32)        # apply output gate
    output = F.linear(out_norm, weights["to_out.weight"])  # final projection

    return output
```
