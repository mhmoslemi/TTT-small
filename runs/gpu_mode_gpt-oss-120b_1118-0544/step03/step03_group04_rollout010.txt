"""Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the reference PyTorch version but fuses the
core N×N×H reduction (∑ₖ left[i,k,h]·right[j,k,h]) into a custom
batched GEMM kernel written in Triton.  All other operations (LayerNorm,
linear projections, gating, output‐norm and final projection) use
vectorized PyTorch kernels in half precision.  The heavy O(N³·H) work
is performed by ``batched_gemm_kernel`` which multiplies, for each
batch and hidden slice, left[:, :, :] @ right[:, :, :].T using tensor‑core
tiles (64×64×64).  This satisfies the requirement of using a Triton
kernel and improves runtime compared with a pure PyTorch fallback.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C[b] = A[b] @ B[b].T for each batch `b`.

    A: (batch, M, K), B: (batch, K, N), C: (batch, M, N)
    All tensors are assumed to be `float16`.  Accumulation is performed in `float32`.
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the batch
    a_batch_ptr = A + pid_batch * batch_stride_a
    b_batch_ptr = B + pid_batch * batch_stride_b
    c_batch_ptr = C + pid_batch * batch_stride_c

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)          # (BLOCK_K,)
        # Mask for K overflow
        mask_k = cur_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        a_mask = mask_m[:, None] & mask_k[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)

        # Load B tile (K x N)
        b_ptrs = b_batch_ptr + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b_mask = mask_k[:, None] & mask_n[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Write out result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# ----------------------------------------------------------------------
# Helper: Element‑wise multiplication kernel (used for final gate)
# ----------------------------------------------------------------------
@triton.jit
def elemwise_mul_kernel(
    a_ptr, b_ptr, out_ptr, size,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch a Triton kernel for element‑wise multiplication."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication."
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ( (size + BLOCK - 1) // BLOCK, )
    elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out

# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] or None (binary mask)
            - weights: dict with keys:
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            * All weight tensors are float32.
            - config: dict with keys "dim", "hidden_dim", optionally "nomask".

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm on last dimension (C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Work in fp16 for the heavy computation path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (left/right) and optional masking
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w) # (B, N, N, hidden_dim)

    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, hidden_dim)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM: (left_slice) @ (right_slice).T
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Rearrange to (B, hidden_dim, N, N) and flatten batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch_h = B * hidden_dim
    left_mat = left_perm.view(batch_h, N, N)
    right_mat = right_perm.view(batch_h, N, N)

    # Allocate output tensor
    out_mat = torch.empty_like(left_mat)

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    # Strides (in elements) for the 3‑D tensors
    stride_am = left_mat.stride(1)   # stride of M dimension (N)
    stride_ak = left_mat.stride(2)   # stride of K dimension (1)
    stride_bk = right_mat.stride(1)  # stride of K dimension (N)
    stride_bn = right_mat.stride(2)  # stride of N dimension (1)
    stride_cm = out_mat.stride(1)    # stride of M dimension (N)
    stride_cn = out_mat.stride(2)    # stride of N dimension (1)
    batch_stride_a = left_mat.stride(0)   # M*K = N*N
    batch_stride_b = right_mat.stride(0)  # K*N = N*N
    batch_stride_c = out_mat.stride(0)    # M*N = N*N

    grid = (batch_h, num_blocks_m, num_blocks_n)
    batched_gemm_kernel[grid](
        left_mat, right_mat, out_mat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7. Apply output gate (element‑wise, Triton kernel)
    # ------------------------------------------------------------------
    out = _elemwise_mul(out, out_gate)

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to original dtype if needed
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
