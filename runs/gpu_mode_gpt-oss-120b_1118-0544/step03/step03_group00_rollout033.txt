@triton.jit
def batched_matmul_transpose_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bi, stride_bk,
    stride_cm, stride_ci, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
   # Get program ids: pid_m for batch index, pid_i for block row, pid_j for block col? We need three axes.

   pid_batch = tl.program_id(0)   # batch dimension (M)
   pid_i = tl.program_id(1)       # row tile index
   pid_j = tl.program_id(2)       # col tile index

   # Compute offsets
   block_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
   block_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

   # Initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # accumulate in fp32

   # Loop over K
   for k in range(0, tl.cdiv(K, BLOCK_K)):
       # Compute k offsets
       k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

       # bounds check
       k_valid = k_offset < K
       i_valid = block_i < N
       j_valid = block_j < N

       # Compute pointers for A and B
       # A[m, i, k]
       a_ptrs = A_ptr + pid_batch * stride_am + block_i[:, None] * stride_ai + k_offset[None, :] * stride_ak
       # B[m, j, k] -> we need to load as (k, j) to compute dot: So B_tile shape (BLOCK_K, BLOCK_N)
       b_ptrs = B_ptr + pid_batch * stride_bm + block_j[None, :] * stride_bi + k_offset[:, None] * stride_bk
       # Because we need B_tile[T][N] where row k, col j.

       # Load tiles
       a = tl.load(a_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0).to(tl.float32)
       b = tl.load(b_ptrs, mask=j_valid[None, :] & k_valid[:, None], other=0.0).to(tl.float32)

       # Perform dot
       # For each pair (i,j) we need sum over k: a[i,k]*b[j,k] = dot of a row i (size K) and b row j (size K) (but we have b transposed shape)
       # Actually we have a shape (BLOCK_M, BLOCK_K) and b shape (BLOCK_K, BLOCK_N) => dot produces (BLOCK_M, BLOCK_N)
       acc += tl.dot(a, b)

       # Advance pointers not needed because we recompute each loop.

   # Store result
   c_ptrs = C_ptr + pid_batch * stride_cm + block_i[:, None] * stride_ci + block_j[None, :] * stride_ck
   mask_store = (block_i[:, None] < N) & (block_j[None, :] < N)
   tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)

def batched_matmul_transpose(A: torch.Tensor, B: torch.Tensor):
    # A and B shape (M,N,N), dtype fp16, contiguous.
    # Returns C same shape
    M,N,K = A.shape[0], A.shape[1], A.shape[2] (K==N)
    # get strides
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bi, stride_bk = B.stride()
    # allocate output C
    C = torch.empty_like(A)
    stride_cm, stride_ci, stride_ck = C.stride()
    # grid
    grid = (M, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_matmul_transpose_kernel[grid](
        A, B, C,
        M, N, K,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bi, stride_bk,
        stride_cm, stride_ci, stride_ck,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C

@triton.jit
def batched_matmul_transpose_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bi, stride_bk,
    stride_cm, stride_ci, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    a_ptrs = A_ptr + pid_batch * stride_am
    b_ptrs = B_ptr + pid_batch * stride_bm
    c_ptrs = C_ptr + pid_batch * stride_cm

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # mask valid indexes
        i_valid = offs_i < N
        j_valid = offs_j < N
        k_valid = k_offset < K

        # pointers for A tile (i,k)
        a_tile_ptrs = a_ptrs + (offs_i[:, None] * stride_ai) + (k_offset[None, :] * stride_ak)
        # pointers for B tile (j,k) but we need (k,j) for dot, i.e., rows = k, cols = j
        b_tile_ptrs = b_ptrs + (k_offset[:, None] * stride_bk) + (offs_j[None, :] * stride_bi)

        a = tl.load(a_tile_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_tile_ptrs, mask=k_valid[:, None] & j_valid[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # store to C
    c_tile_ptrs = c_ptrs + (offs_i[:, None] * stride_ci) + (offs_j[None, :] * stride_ck)
    mask_store = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(c_tile_ptrs, acc.to(tl.float16), mask=mask_store)

"""
TriMul (outgoing) – fast forward only.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the provided LN parameters.
2. Cast to float16 for compute‑heavy stages.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are performed in FP16.
4. If a mask is present, a Triton kernel fuses `proj * gate * mask`
   to avoid extra memory passes.
5. Core reduction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   i.e. a batched matrix multiplication C = A @ Bᵀ where A and B have
   shape (M,N,N) with M = B * hidden_dim.  The reduction is performed by
   a custom Triton kernel that tiles over i, j and k, accumulating in
   float32 and writing back in float16.
6. Output Layer‑norm, multiplication by `out_gate`, and final linear
   projection back to the original channel dimension.
7. Cast back to the input dtype (usually float32).

The implementation fuses the mask‑gate multiplication (if needed) and
the batched matrix multiplication into Triton kernels, while keeping
the remaining inexpensive operations (layer‑norm, linear layers) in
PyTorch.  This yields a large speed‑up over a pure‑PyTorch reference.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Kernel: element‑wise (proj * gate * mask) with mask broadcast over H
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,          # (B*N*N*H,) fp16
    gate_ptr,          # (B*N*N*H,) fp16
    mask_ptr,          # (B*N*N,)   fp16 (0/1)
    out_ptr,           # (B*N*N*H,) fp16
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def apply_gate_and_mask(proj: torch.Tensor,
                        gate: torch.Tensor,
                        mask: torch.Tensor,
                        block_size: int = 1024) -> torch.Tensor:
    """
    Fuse element‑wise multiplication of projection, gate and optional mask.
    All tensors are expected to be contiguous FP16.
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel: batched matrix multiplication C = A @ Bᵀ
#   A, B: (M, N, K)   FP16, where K == N
#   C:    (M, N, N)   FP16
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_transpose_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bi, stride_bk,
    stride_cm, stride_ci, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_batch = tl.program_id(0)      # batch (M)
    pid_i = tl.program_id(1)          # tile row
    pid_j = tl.program_id(2)          # tile col

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch_ptr = A_ptr + pid_batch * stride_am
    b_batch_ptr = B_ptr + pid_batch * stride_bm
    c_batch_ptr = C_ptr + pid_batch * stride_cm

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction axis k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        i_valid = offs_i < N
        j_valid = offs_j < N
        k_valid = k_offset < K

        # A tile: (i, k)
        a_ptrs = a_batch_ptr + (offs_i[:, None] * stride_ai) + (k_offset[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=i_valid[:, None] & k_valid[None, :],
                    other=0.0).to(tl.float32)

        # B tile: we need (k, j) for dot product, i.e. rows = k, cols = j
        b_ptrs = b_batch_ptr + (k_offset[:, None] * stride_bk) + (offs_j[None, :] * stride_bi)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & j_valid[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # store result
    c_ptrs = c_batch_ptr + (offs_i[:, None] * stride_ci) + (offs_j[None, :] * stride_ck)
    mask_store = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)


def batched_matmul_transpose(A: torch.Tensor,
                             B: torch.Tensor,
                             BLOCK_M: int = 64,
                             BLOCK_N: int = 64,
                             BLOCK_K: int = 32) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for tensors of shape (M, N, K) with K == N.
    All tensors must be FP16 and contiguous.
    """
    assert A.shape == B.shape, "A and B must have the same shape"
    M, N, K = A.shape
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bi, stride_bk = B.stride()

    C = torch.empty_like(A)
    stride_cm, stride_ci, stride_ck = C.stride()

    grid = (M,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))
    batched_matmul_transpose_kernel[grid](
        A, B, C,
        M, N, K,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bi, stride_bk,
        stride_cm, stride_ci, stride_ck,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float) or None
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict containing:
                "dim" (C), "hidden_dim" (H), optional "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    dtype_in = input_tensor.dtype
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,dim) -> (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gates (same shape) + optional mask fusion
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # (B,N,N,hidden_dim)

    if mask is not None and not nomask:
        # ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core reduction: batched GEMM C = A @ Bᵀ
    #    Shapes after permute: (B, hidden_dim, N, N) -> (B*hidden_dim, N, N)
    # ------------------------------------------------------------------
    B_batch, N_seq, _, H = left.shape
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B_batch * H, N_seq, N_seq)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B_batch * H, N_seq, N_seq)

    out_flat = batched_matmul_transpose(left_flat, right_flat)

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B_batch, H, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8. Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
