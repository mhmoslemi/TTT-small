"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C).
2. Project to a hidden dimension H with two linear layers (left/right).
3. Apply an optional binary mask (broadcast over H) and three sigmoid gates
   (left, right, out).
4. Multiply left/right by their respective gates.
5. Fold the hidden dimension into the batch axis -> tensors of shape
   (B*H, N, N).  The core O(N³·H) reduction is performed with a custom
   Triton kernel that computes, for each batch slice,
       out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
   i.e. a batched matrix multiplication C = A @ Bᵀ where B is accessed
   transposed without an explicit memory transpose.
6. Reshape back to (B,N,N,H), apply a second LayerNorm over H, multiply by
   the output gate and project back to the original channel dimension C.
7. Return the result in the original dtype (float32).

All heavy arithmetic (the O(N³·H) reduction) is done in fp16 with fp32
accumulation, leveraging Tensor‑cores on the B200.  The rest of the
operations use PyTorch's highly optimized kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the triangular multiplication.
# Computes  C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]   (i.e. A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all = seq_len)
    stride_ah, stride_am, stride_ak,   # A: batch, row, col
    stride_bh, stride_bk, stride_bn,   # B: batch, col(k), row(j)  (accessed as K×N)
    stride_ch, stride_cm, stride_cn,   # C: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)   # batch index (B * H)
    pid_m = tl.program_id(0)       # block row (i)
    pid_n = tl.program_id(1)       # block column (j)

    # ----------------------- block offsets ----------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----------------------- base pointers ----------------------------
    a_batch = pid_batch * stride_ah
    b_batch = pid_batch * stride_bh
    c_batch = pid_batch * stride_ch

    # pointers for the current tile (K dimension will be iterated)
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        # k‑dimension indices for this tile
        k_base = k * BLOCK_K
        k_idx = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        # load A[i, k] and B[j, k] with boundary checks
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write result (convert back to fp16)
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_tri_mul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.
    A, B: fp16 tensors of shape (batch, N, N) where batch = B * hidden_dim.
    Returns C of the same shape and dtype.
    """
    assert A.shape == B.shape
    batch, N, K = A.shape
    assert N == K, "TriMul expects square sequence dimension."

    # Heuristic block sizes
    if N <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    C = torch.empty_like(A)

    _tri_mul_kernel[grid](
        A, B, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),       # A: batch, row, col
        B.stride(0), B.stride(2), B.stride(1),       # B accessed as (batch, K, N)
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor [B, N, N, C] with the same dtype as the input.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Use fp16 for the heavy path
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2) Cast linear weights to fp16 (no bias)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 3) Linear projections
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w)  # (B,N,N,H)

    # --------------------------------------------------------------
    # 4) Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 5) Gates
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6) Fold hidden dimension into the batch axis
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left = left.view(B * H, N, N)   # (B*H, N, N)
    right = right.view(B * H, N, N)  # (B*H, N, N)

    # --------------------------------------------------------------
    # 7) Core O(N³·H) reduction (batched GEMM via Triton)
    # --------------------------------------------------------------
    out_flat = _batched_tri_mul_triton(left, right)   # (B*H, N, N)

    # --------------------------------------------------------------
    # 8) Reshape back to (B,N,N,H)
    # --------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 9) Output LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 10) Apply output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 11) Final linear projection back to original channel dimension
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 12) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
