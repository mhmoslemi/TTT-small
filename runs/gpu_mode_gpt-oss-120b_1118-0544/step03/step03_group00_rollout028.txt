@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, N,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
   pid = tl.program_id(0)
   # Compute blocks per batch
   num_m_blocks = tl.cdiv(N, BLOCK_M)
   num_n_blocks = tl.cdiv(N, BLOCK_N)
   blocks_per_batch = num_m_blocks * num_n_blocks

   batch_idx = pid // blocks_per_batch
   block_idx = pid % blocks_per_batch
   pid_m = block_idx // num_n_blocks
   pid_n = block_idx % num_n_blocks

   # ranges
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

   mask_m = offs_m < N
   mask_n = offs_n < N

   # pointer base offsets for this batch
   a_base = a_ptr + batch_idx * stride_a_batch
   b_base = b_ptr + batch_idx * stride_b_batch
   c_base = c_ptr + batch_idx * stride_c_batch

   # initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # Loop over K dimension
   for k in range(0, tl.cdiv(N, BLOCK_K)):
       k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
       # mask for k range
       mask_k = k_offset < N

       # load A tile
       a_ptrs = a_base + offs_m[:, None] * stride_a_m + k_offset[None, :] * stride_a_k
       # mask for A tile: m valid and k valid
       mask_a = mask_m[:, None] & mask_k[None, :]
       a = tl.load(a_ptrs, mask=mask_a, other=0.0)

       # load B tile (B is used as transposed: shape (K,N))
       b_ptrs = b_base + k_offset[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
       # mask for B tile: k valid and n valid
       mask_b = mask_k[:, None] & mask_n[None, :]
       b = tl.load(b_ptrs, mask=mask_b, other=0.0)

       acc += tl.dot(a, b)

   # write output
   c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
   c = acc.to(tl.float16)
   mask_c = mask_m[:, None] & mask_n[None, :]
   tl.store(c_ptrs, c, mask=mask_c)

        BLOCK_K=block_k,

1. Layer‑norm the input tensor.
2. Cast to FP16 for compute‑heavy ops.
3. Linear projections and sigmoid gates (no bias) on the last dimension.
4. Optional mask is applied together with the gates (element‑wise, broadcasted).
5. Core reduction:
       out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
   implemented as a batched GEMM (left @ rightᵀ) via a custom Triton kernel.
6. Second Layer‑norm, output‑gate multiplication and final linear projection.
7. Cast back to the input dtype.

The Triton kernel works on tensors of shape (B·H, N, N) in FP16, using a
persistent‑grid strategy that iterates over the K dimension in blocks.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the outgoing TriMul
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,  # left  : [batch, N, N]  (FP16)
    b_ptr,  # right : [batch, N, N]  (FP16) – will be read as (K, N) i.e. transposed
    c_ptr,  # out   : [batch, N, N]  (FP16)
    batch: tl.int32,
    N: tl.int32,
    # strides for a (batch, M=N, K=N)
    stride_a_batch, stride_a_m, stride_a_k,
    # strides for b (batch, K=N, N), note that we treat b as transposed
    stride_b_batch, stride_b_k, stride_b_n,
    # strides for c (batch, M=N, N)
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1) decode block indices (batch, block_m, block_n)
    # --------------------------------------------------------------
    num_m_blocks = tl.cdiv(N, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_m_blocks * num_n_blocks

    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    pid_m = block_idx // num_n_blocks
    pid_n = block_idx % num_n_blocks

    # --------------------------------------------------------------
    # 2) compute tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for the current batch
    a_batch_ptr = a_ptr + batch_idx * stride_a_batch
    b_batch_ptr = b_ptr + batch_idx * stride_b_batch
    c_batch_ptr = c_ptr + batch_idx * stride_c_batch

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) loop over K dimension (the summation axis)
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)          # (BLOCK_K,)
        mask_k = k_offset < N

        # ---- load a tile -------------------------------------------------
        a_ptrs = (
            a_batch_ptr
            + offs_m[:, None] * stride_a_m
            + k_offset[None, :] * stride_a_k
        )
        mask_a = mask_m[:, None] & mask_k[None, :]
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # ---- load b tile (b is logically transposed) --------------------
        b_ptrs = (
            b_batch_ptr
            + k_offset[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )
        mask_b = mask_k[:, None] & mask_n[None, :]
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # ---- accumulate dot product ---------------------------------------
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # 4) write result (cast back to FP16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, c, mask=mask_c)


def batched_matmul_triton(left: torch.Tensor, right: torch.Tensor,
                         block_m: int = 128, block_n: int = 128, block_k: int = 32) -> torch.Tensor:
    """
    Batched matrix multiplication for the TriMul core:
        out = left @ rightᵀ
    left / right : (batch, N, N) FP16
    Returns out  : (batch, N, N) FP16
    """
    assert left.shape == right.shape, "left and right must have the same shape"
    batch, N, _ = left.shape
    out = torch.empty_like(left)

    # strides (in elements) for the three tensors
    stride_a_batch, stride_a_m, stride_a_k = left.stride()
    stride_b_batch, stride_b_n, stride_b_k = right.stride()  # note: original (batch, N, N)
    # we need b as (batch, K=N, N) -> k stride = stride over columns, n stride = stride over rows
    # in the original layout: stride over columns = 1, rows = N
    # thus:
    stride_b_k = right.stride(2)   # column stride (1)
    stride_b_n = right.stride(1)   # row stride (N)

    stride_c_batch, stride_c_m, stride_c_n = out.stride()

    num_m_blocks = (N + block_m - 1) // block_m
    num_n_blocks = (N + block_n - 1) // block_n
    total_blocks = batch * num_m_blocks * num_n_blocks
    grid = (total_blocks,)

    tri_mul_outgoing_kernel[grid](
        left, right, out,
        batch, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Entry point – forward pass of the outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (B,N,N,C) -> (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid) – also (B,N,N,H)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6) Apply mask (if any) and fuse with gates (element‑wise)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Reshape to (B*H, N, N) for the batched GEMM
    # ------------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N) -> (B*H,N,N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(-1, left.shape[1], left.shape[2])
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(-1, right.shape[1], right.shape[2])

    # ------------------------------------------------------------------
    # 8) Core O(N³·H) reduction using Triton
    # ------------------------------------------------------------------
    out_mat = batched_matmul_triton(left_mat, right_mat,
                                    block_m=128, block_n=128, block_k=32)   # (B*H,N,N)

    # ------------------------------------------------------------------
    # 9) Restore original shape (B,N,N,H)
    # ------------------------------------------------------------------
    B = input_tensor.shape[0]
    N = input_tensor.shape[1]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 10) Output LayerNorm + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to dim channels
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 12) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
