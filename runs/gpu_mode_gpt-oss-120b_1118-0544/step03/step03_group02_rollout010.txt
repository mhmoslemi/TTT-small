import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, size: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offset < size
    a = tl.load(a_ptr + offset, mask=mask)
    b = tl.load(b_ptr + offset, mask=mask)
    tl.store(c_ptr + offset, a * b, mask=mask)

def custom_kernel(data):
    """Outgoing TriMul forward pass (no gradients).

    Implements the TriMul operation described in AlphaFold3 with heavy
    fusion: two fused linear+gate projections, a batched matrix multiplication
    for the O(N³) reduction, and a Triton kernel for the final out‑gate
    element‑wise multiplication. All heavy compute is performed in float16
    and the final output is returned in float32.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ---- 1. LayerNorm over channel dimension (float32) ----
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # Cast to fp16 for the rest of the pipeline
    x_fp16 = x.to(torch.float16)

    B, N, _, _ = x_fp16.shape
    # flatten spatial dimensions for linear ops
    x_flat = x_fp16.view(-1, dim)  # (B*N*N, dim)

    # ---- 2. Fuse left projection + left gate ----
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    left_fused_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*hidden, dim)
    left_fused = F.linear(x_flat, left_fused_w)  # (B*N*N, 2*hidden)
    left_fused = left_fused.view(B, N, N, 2 * hidden_dim)
    left_proj, left_gate = torch.split(left_fused, hidden_dim, dim=-1)

    # ---- 3. Fuse right projection + right gate ----
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    right_fused_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*hidden, dim)
    right_fused = F.linear(x_flat, right_fused_w)
    right_fused = right_fused.view(B, N, N, 2 * hidden_dim)
    right_proj, right_gate = torch.split(right_fused, hidden_dim, dim=-1)

    # ---- 4. Optional mask (applied after projection, before gating) ----
    if not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ---- 5. Gating (sigmoid) and combine ----
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    left = left_proj * left_gate   # [B,N,N,hidden] fp16
    right = right_proj * right_gate

    # ---- 6. Batched matrix multiplication: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] ----
    # reshape to (B*hidden, N, N) for batched gemm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # ---- 7. to_out_norm (LayerNorm over hidden) ----
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # fp16

    # ---- 8. Compute out_gate (linear + sigmoid) ----
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(B, N, N, hidden_dim)

    # ---- 9. Triton element‑wise multiplication out *= out_gate ----
    total = out.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK, )
    # Ensure contiguity
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    elementwise_mul_kernel[grid](
        out, out_gate, out, total, BLOCK=BLOCK
    )

    # ---- 10. Final linear projection back to dim ----
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out_flat = out.view(-1, hidden_dim)
    out_flat = F.linear(out_flat, to_out_w)  # (B*N*N, dim) fp16
    out = out_flat.view(B, N, N, dim).to(torch.float32)

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, size: tl.constexpr, BLOCK: tl.constexpr):
    """
    Triton kernel: c = a * b  (1‑D flattened tensors).
    """
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offset < size
    a = tl.load(a_ptr + offset, mask=mask)
    b = tl.load(b_ptr + offset, mask=mask)
    tl.store(c_ptr + offset, a * b, mask=mask)


def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – may be ignored if config["nomask"] is True
            - weights: dict of model weights (float32 tensors) with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with entries:
                "dim" (int), "hidden_dim" (int), optional "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. LayerNorm over channel dimension (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the remaining ops
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)
    B, N, _, _ = x_fp16.shape
    x_flat = x_fp16.reshape(-1, dim)  # (B*N*N, dim)

    # ------------------------------------------------------------------
    # 3. Fuse left projection + left gate (2 * hidden -> dim)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    left_fused_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*hidden, dim)

    left_fused = F.linear(x_flat, left_fused_w)                # (B*N*N, 2*hidden)
    left_fused = left_fused.view(B, N, N, 2 * hidden_dim)
    left_proj, left_gate = torch.split(left_fused, hidden_dim, dim=-1)

    # ------------------------------------------------------------------
    # 4. Fuse right projection + right gate
    # ------------------------------------------------------------------
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    right_fused_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*hidden, dim)

    right_fused = F.linear(x_flat, right_fused_w)
    right_fused = right_fused.view(B, N, N, 2 * hidden_dim)
    right_proj, right_gate = torch.split(right_fused, hidden_dim, dim=-1)

    # ------------------------------------------------------------------
    # 5. Optional masking (applied after projection, before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 6. Gating (sigmoid) and combine
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    left = left_proj * left_gate   # (B, N, N, hidden) fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Batched GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Equivalent to (B*hidden) independent N×N matrix multiplications.
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 8. LayerNorm on hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # fp16

    # ------------------------------------------------------------------
    # 9. Compute out_gate (linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 10. Element‑wise multiplication with out_gate via Triton
    # ------------------------------------------------------------------
    total = out.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK, )
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    elementwise_mul_kernel[grid](
        out, out_gate, out, total, BLOCK=BLOCK
    )

    # ------------------------------------------------------------------
    # 11. Final projection back to original dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out_flat = out.view(-1, hidden_dim)
    out_flat = F.linear(out_flat, to_out_w)               # (B*N*N, dim) fp16
    out = out_flat.view(B, N, N, dim).to(torch.float32)

    return out
