@triton.jit
def matmul_triple_kernel(
    left_ptr, right_ptr, out_ptr,
    BH, N,
    stride_left_bh, stride_left_i, stride_left_k,
    stride_right_bh, stride_right_j, stride_right_k,
    stride_out_bh, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    bh = tl.program_id(0)  # batch hidden index
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # K offset iteration
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # load left tile
        left_ptrs = left_ptr + bh * stride_left_bh + offs_m[:, None] * stride_left_i + k_offset[None, :] * stride_left_k
        # load right tile (transposed)
        # right has shape (BH,N,N). We need right[j,k] but in multiplication we need right[bh, j, k] . The dot product uses k as inner dimension, so we need right tile transposed (k,N) -> we load right[bh, k_offset, offs_n]
        right_ptrs = right_ptr + bh * stride_right_bh + k_offset[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        # shape loaded: left: (BLOCK_M, BLOCK_K), right: (BLOCK_K, BLOCK_N)
        left = tl.load(left_ptrs, mask=(offs_m < N)[:, None] & (k_offset < N)[None, :], other=0.0).to(tl.float32)
        right = tl.load(right_ptrs, mask=(k_offset < N)[:, None] & (offs_n < N)[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + bh * stride_out_bh + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    mask = (offs_m < N)[:, None] & (offs_n < N)[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask)

@triton.jit
def batched_gemm_trimul(
    left_ptr, right_ptr, out_ptr,
    BH, N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # block indices for bh, i, j
    bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for rows and columns
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for rows and columns within N (avoid out-of-range)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + bh * stride_left_batch + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right tile (transposed): shape (BLOCK_K, BLOCK_N) reading right[bh, k, j]
        right_ptrs = right_ptr + bh * stride_right_batch + offs_k[:, None] * stride_right_i + offs_n[None, :] * stride_right_k
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # accumulate dot product across K block
        # Use tl.dot if supported for float32.
        acc += tl.dot(left, right)

    # write out
    out_ptrs = out_ptr + bh * stride_out_batch + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_k
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # 1. LayerNorm on input (C dim)
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Cast to half
    x = x.to(torch.float16)

    # 2. cast weights to half
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # 3. Projections
    left = F.linear(x, left_proj_w)  # shape (B,N,N,H)
    right = F.linear(x, right_proj_w)

    # 4. Apply mask if needed
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # 5. Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # 6. Core TriMul matrix multiplication using Triton
    B, N, _, H = left.shape  # (B, N, N, H)
    # permute and view to (BH, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)  # same dtype float16

    # get strides (int)
    stride_left_batch, stride_left_i, stride_left_k = left_flat.stride()
    # right stride same as left (contiguous) but we treat transposed reading
    stride_right_batch, stride_right_i, stride_right_k = right_flat.stride()
    # output stride
    stride_out_batch, stride_out_i, stride_out_k = out_flat.stride()

    # define block sizes (tune)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # compute grid dims
    grid = (BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # launch kernel
    batched_gemm_trimul[grid](
        left_flat,
        right_flat,
        out_flat,
        BH, N,
        stride_left_batch, stride_left_i, stride_left_k,
        stride_right_batch, stride_right_i, stride_right_k,
        stride_out_batch, stride_out_i, stride_out_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape output back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 7. to_out_norm layernorm on hidden dim
    out = F.layer_norm(out,
                       (H,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # 8. multiply by out_gate
    out = out * out_gate

    # 9. final linear projection to dim
    out = F.linear(out, to_out_w)

    # cast back to original dtype
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def triton_tri_mul_matmul(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets for the tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: (i, k)
        left_ptrs = left_ptr + pid_bh * stride_left_batch + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16).to(tl.float32)

        # right transposed: (k, j) where we read right[bh, k, j]
        right_ptrs = right_ptr + pid_bh * stride_right_batch + offs_k[:, None] * stride_right_i + offs_j[None, :] * stride_right_k
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0, dtype=tl.float16).to(tl.float32)

        acc += tl.dot(left, right)

    # write back result
    out_ptrs = out_ptr + pid_bh * stride_out_batch + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_k
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    The implementation follows the reference PyTorch code but replaces the
    O(N³·H) reduction with a fused Triton matmul that processes each
    (batch, hidden) slice as an independent matrix‑multiply.  All other
    operations (LayerNorm, linear projections, gating) are performed in
    half‑precision with PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm over channel dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Work in fp16 for the heavy path
    x = x.to(torch.float16)

    # 2. Cast model weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # 3. Linear projections (left/right)
    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w) # (B, N, N, H)

    # 4. Optional masking (applied before gating)
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # 5. Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # 6. Core O(N³·H) reduction using Triton
    B, N, _, H = left.shape   # shapes are (B, N, N, H)

    # Rearrange to (B, H, N, N) and flatten batch*head dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    out_flat = torch.empty_like(left_flat)

    # Strides (in elements) for the contiguous (BH, N, N) tensors
    stride_l_batch, stride_l_i, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_i, stride_r_k = right_flat.stride()
    stride_o_batch, stride_o_i, stride_o_k = out_flat.stride()

    # Triton launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    triton_tri_mul_matmul[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_l_batch,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_i,
        stride_r_k,
        stride_o_batch,
        stride_o_i,
        stride_o_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 7. Hidden‑dim LayerNorm
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 8. Output gate
    out = out * out_gate

    # 9. Final linear projection back to original channel dimension
    out = F.linear(out, to_out_w)

    # 10. Cast back to input dtype if needed
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the reference PyTorch version but replaces the
O(N³·H) reduction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
with a fused Triton batched matrix‑multiply.  Each (batch, hidden) slice is
treated as an independent (N×N) matrix multiplication, which is executed by a
single persistent Triton kernel.  All other steps (LayerNorm, linear projections,
gates) are performed in half‑precision with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def triton_tri_mul_matmul(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul core.

    left_ptr , right_ptr : (BH, N, N) in FP16
    out_ptr               : (BH, N, N) in FP16
    BH                    : batch * hidden_dim  (encoded via grid‐dim 0)
    """
    pid_bh = tl.program_id(0)          # batch‑hidden index
    pid_i  = tl.program_id(1)          # block row index
    pid_j  = tl.program_id(2)          # block column index

    # Tile coordinates
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:   left[bh, i, k]
        left_ptrs = left_ptr + pid_bh * stride_left_batch \
                    + offs_i[:, None] * stride_left_i \
                    + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16).to(tl.float32)

        # right tile (transposed): we need right[bh, k, j]
        right_ptrs = right_ptr + pid_bh * stride_right_batch \
                    + offs_k[:, None] * stride_right_i \
                    + offs_j[None, :] * stride_right_k
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0,
                        dtype=tl.float16).to(tl.float32)

        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N) accumulation

    # Write result back as FP16
    out_ptrs = out_ptr + pid_bh * stride_out_batch \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_k
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])
