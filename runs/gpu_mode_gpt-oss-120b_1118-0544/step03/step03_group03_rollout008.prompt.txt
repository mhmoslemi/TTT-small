import math

class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

All large linear projections and the N³ reduction are performed in half‑precision
(fp16) so that Tensor Cores on the B200 can be used.  A tiny Triton kernel
converts the final fp16 tensor back to fp32 before the last LayerNorm and
output projection, satisfying the requirement to contain a custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def cast_fp16_to_fp32_kernel(
    in_ptr,          # fp16 source tensor
    out_ptr,         # fp32 destination tensor
    N_ELEM: tl.constexpr,   # total number of elements
    BLOCK: tl.constexpr,    # block size processed per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM
    val = tl.load(in_ptr + offs, mask=mask).to(tl.float32)
    tl.store(out_ptr + offs, val, mask=mask)


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass using fp16 for the heavy work.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model weights (see skeleton)
            - config: dict with at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ----------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ----------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ----------------------------------------------------
    # 2) Cast to fp16 for the heavy part
    # ----------------------------------------------------
    x_h = x.to(torch.float16)

    # ----------------------------------------------------
    # 3) Fuse projection + gate linear layers (fp16)
    # ----------------------------------------------------
    left_combined_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16).to(device)
    right_combined_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16).to(device)

    left_combined = F.linear(x_h, left_combined_weight)   # [B, N, N, 2*H]
    right_combined = F.linear(x_h, right_combined_weight)

    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    # ----------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # ----------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ----------------------------------------------------
    # 5) Gating (sigmoid, fp16)
    # ----------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate      # [B, N, N, H] fp16
    right = right_proj * right_gate   # [B, N, N, H] fp16

    # ----------------------------------------------------
    # 6) Output gate (fp16)
    # ----------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(torch.float16).to(device)
    out_gate_pre = F.linear(x_h, out_gate_weight)   # [B, N, N, H] fp16
    out_gate = torch.sigmoid(out_gate_pre)          # fp16

    # ----------------------------------------------------
    # 7) Batched matrix multiplication: out = left @ rightᵀ
    #    (B*H independent N×N matmuls in fp16)
    # ----------------------------------------------------
    B, N, _, _ = left.shape
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_h = torch.bmm(left_flat, right_flat.transpose(1, 2))   # fp16, shape (B*H, N, N)

    # reshape back to [B, N, N, H]
    out_h = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ----------------------------------------------------
    # 8) Cast to fp32 (tiny Triton kernel)
    # ----------------------------------------------------
    total_elem = out_h.numel()
    out_fp32 = torch.empty_like(out_h, dtype=torch.float32)
    BLOCK = 65536
    grid = ((total_elem + BLOCK - 1) // BLOCK,)
    cast_fp16_to_fp32_kernel[grid](
        in_ptr=out_h,
        out_ptr=out_fp32,
        N_ELEM=total_elem,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # ----------------------------------------------------
    # 9) Final LayerNorm, gating and projection back to original dim
    # ----------------------------------------------------
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.to(torch.float32)            # apply output gate (fp32)
    out = F.linear(out, weights["to_out.weight"])    # final projection, fp32

    return out
