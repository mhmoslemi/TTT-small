class TriMul(nn.Module):
    def __init__(self, dim, hidden_dim):
        self.norm = nn.LayerNorm(dim)
        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)
    def forward(self, x, mask):
        x = self.norm(x)
        left = self.left_proj(x)
        right = self.right_proj(x)
        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask
        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()
        left = left * left_gate
        right = right * right_gate
        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    out_stride_m, out_stride_n, out_stride_h? Not needed.
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0) # which batch (m)
    pid_i = tl.program_id(1) # block i
    pid_j = tl.program_id(2) # block j

    # compute offsets for block i
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # Load a and b tiles
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offsets in K
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # compute pointer offsets
        a_ptrs = a_ptr + pid_m * stride_am * 0 + offs_i[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_m * stride_bk * 0 + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn
        # load tiles
        a = tl.load(a_ptrs, mask=(offs_i < M) & (offs_k < K), other=0.0) # shape (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=(offs_k < K) & (offs_j < N), other=0.0) # shape (BLOCK_K, BLOCK_N)
        # matmul
        acc += tl.dot(a, b)
    # store
    c_ptrs = c_ptr + pid_m * stride_cm * 0 + offs_i[:, None] * stride_cm + offs_j[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=(offs_i[:, None] < M) & (offs_j[None, :] < N))

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The kernel fuses the heavy N³ reduction:
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
by performing a batched fp16 matrix multiplication (left @ rightᵀ) using
Triton. All other operations (LayerNorms, linear projections, gates) are
performed in PyTorch (mostly in fp16). The result of the Triton kernel is
produced directly in fp32, avoiding an extra cast kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # dimensions: M = batch*hidden, N = seq_len, K = seq_len
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched fp16 matmul C = A @ B (A: (M,N,K), B: (M,K,N) -> C: (M,N,N)).
       A and B are fp16, accumulation is fp32, and C is stored as fp32.
    """
    pid_m = tl.program_id(0)   # which (batch * hidden) matrix
    pid_i = tl.program_id(1)   # block row index
    pid_j = tl.program_id(2)   # block col index

    # Compute start offsets for this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    i_offset = i_start + tl.arange(0, BLOCK_M)
    j_offset = j_start + tl.arange(0, BLOCK_N)

    # Pointer to the beginning of the matrix for this batch
    a_base = a_ptr + pid_m * stride_a_batch
    b_base = b_ptr + pid_m * stride_b_batch
    c_base = c_ptr + pid_m * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_start = k * BLOCK_K
        k_offset = k_start + tl.arange(0, BLOCK_K)

        # Load A tile (M,N,K) -> (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + i_offset[:, None] * stride_a_row + k_offset[None, :] * stride_a_col
        a_mask = (i_offset[:, None] < N) & (k_offset[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # Load B tile (M,K,N) -> (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + k_offset[:, None] * stride_b_row + j_offset[None, :] * stride_b_col
        b_mask = (k_offset[:, None] < K) & (j_offset[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Compute dot product; accumulation is fp32
        acc += tl.dot(a, b)

    # Store result to C (fp32)
    c_ptrs = c_base + i_offset[:, None] * stride_c_row + j_offset[None, :] * stride_c_col
    c_mask = (i_offset[:, None] < N) & (j_offset[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Executes the outgoing TriMul forward pass.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict containing model weights, see skeleton in the prompt
            - config: dict containing at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ----------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ----------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ----------------------------------------------------
    # 2) Cast to fp16 for the heavy part
    # ----------------------------------------------------
    x_h = x.to(torch.float16)

    # ----------------------------------------------------
    # 3) Fuse projection + gate linear layers (fp16)
    # ----------------------------------------------------
    # left projection + left gate
    left_combined_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    left_combined = F.linear(x_h, left_combined_weight)   # [B, N, N, 2*hidden_dim]

    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)

    # right projection + right gate
    right_combined_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)
    right_combined = F.linear(x_h, right_combined_weight)  # [B, N, N, 2*hidden_dim]

    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    # ----------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # ----------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ----------------------------------------------------
    # 5) Gating (sigmoid, fp16)
    # ----------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate          # [B, N, N, hidden_dim] fp16
    right = right_proj * right_gate       # [B, N, N, hidden_dim] fp16

    # ----------------------------------------------------
    # 6) Output gate (fp16)
    # ----------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate_pre = F.linear(x_h, out_gate_weight)      # [B, N, N, hidden_dim] fp16
    out_gate = torch.sigmoid(out_gate_pre)             # fp16

    # ----------------------------------------------------
    # 7) Prepare tensors for batched matmul: out = left @ rightᵀ
    # ----------------------------------------------------
    B, N, _, _ = left.shape
    M = B * hidden_dim

    # left: (B, N, N, H) -> (M, N, N) where rows are i, cols are k
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(M, N, N)

    # rightᵀ: (B, N, N, H) -> (M, N, N) where rows are k, cols are j
    right_T_flat = right.permute(0, 3, 2, 1).contiguous().view(M, N, N)

    # Output tensor (fp32) to store the result of the reduction
    out_flat = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # ----------------------------------------------------
    # 8) Triton batched GEMM kernel
    # ----------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (M,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_T_flat,
        c_ptr=out_flat,
        M=M,
        N=N,
        K=N,
        stride_a_batch=left_flat.stride(0),
        stride_a_row=left_flat.stride(1),
        stride_a_col=left_flat.stride(2),
        stride_b_batch=right_T_flat.stride(0),
        stride_b_row=right_T_flat.stride(1),
        stride_b_col=right_T_flat.stride(2),
        stride_c_batch=out_flat.stride(0),
        stride_c_row=out_flat.stride(1),
        stride_c_col=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ----------------------------------------------------
    # 9) Reshape back to [B, N, N, hidden_dim]
    # ----------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # ----------------------------------------------------
    # 10) Final LayerNorm, gating and output projection (float32)
    # ----------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.to(torch.float32)                # apply output gate
    out = F.linear(out, weights["to_out.weight"])         # final projection to dim

    return out
