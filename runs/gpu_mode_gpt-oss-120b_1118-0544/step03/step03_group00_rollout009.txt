import torch
import triton
import triton.language as tl

@triton.jit
def gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                     total_elems, hidden_dim,
                     BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # broadcast mask over hidden_dim
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out_val = proj * gate * mask_val
    tl.store(out_ptr + offs, out_val, mask=mask)

def apply_gate_and_mask(proj, gate, mask, block_size=1024):
    # proj, gate: (B,N,N,H) fp16 contiguous
    # mask: (B,N,N) fp16 contiguous
    B,N,N2,H = proj.shape
    assert N == N2
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out

def custom_kernel(data):
    """
    Fast forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Implements:
        1. Layer‑norm on the input.
        2. Linear projections (left/right) and gated projections.
        3. Optional mask application fused with gating in a Triton kernel.
        4. Core O(N³·H) reduction as a batched FP16 matrix multiplication
           (left @ rightᵀ) for each hidden channel.
        5. Output layer‑norm, output‑gate, and final linear projection.
    Returns a tensor of shape [B, N, N, C] in the same dtype as the input.
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm in FP16
    # ------------------------------------------------------------------
    # Convert LN parameters to FP16 once
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    ).to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast weights to FP16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections
    # ------------------------------------------------------------------
    # proj: (B,N,N,H)  FP16
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # no bias
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Gate values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5. Apply mask + gate (fusion via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure contiguous FP16 tensors
        mask_fp16 = mask.to(torch.float16).contiguous()
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core O(N³·H) reduction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # transpose right on the two N axes before flattening
    right_T = right.permute(0, 2, 1, 3)          # (B, N, N, H) -> (B, N(k), N(j), H)
    right_flat = right_T.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # batched matrix multiplication
    out_flat = torch.bmm(left_flat, right_flat)   # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8. Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
Fast forward pass for the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the provided LN parameters.
2. Cast to float16 for the compute‑heavy path.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) using FP16 weights.
4. If a mask is present, fuse mask‑application with gating via a small
   Triton kernel (`apply_gate_and_mask`).  This replaces two separate
   element‑wise passes.
5. Core O(N³·H) reduction: for each hidden channel h we need
       out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h]
   which is exactly a batched matrix multiplication
       left[h] @ right[h].T .
   We reshape to (B*H, N, N) and perform a single `torch.bmm`,
   letting cuBLAS (Tensor‑cores) handle the massive work.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate, and project back to the original channel dimension.
7. Cast the result back to the input dtype (usually float32).

Only the mask‑gate fusion is performed in Triton; the rest relies on
high‑performance PyTorch kernels.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: elementwise (proj * gate * mask) on a flattened tensor
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # Tensor[(B*N*N*H,), fp16]   – projected values
    gate_ptr,          # Tensor[(B*N*N*H,), fp16]   – gate values (sigmoid)
    mask_ptr,          # Tensor[(B*N*N,),    fp16] – binary mask (broadcast)
    out_ptr,           # Tensor[(B*N*N*H,), fp16]   – output
    total_elems,       # int – total number of elements in proj/gate
    hidden_dim,        # int – H (used for broadcast)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each mask element corresponds to BLOCK_SIZE // hidden_dim consecutive proj entries
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` where `mask` is broadcast over the hidden dim.
    All tensors must be contiguous and of dtype torch.float16.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],            # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict containing
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP16)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    ).to(torch.float16)   # now FP16

    # ------------------------------------------------------------------
    # 2. Cast all linear weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections: (B,N,N,C) -> (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Gate values (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5. Apply mask + gate (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj.contiguous(),
                                    left_gate.contiguous(),
                                    mask_fp16)
        right = _apply_gate_and_mask(right_proj.contiguous(),
                                     right_gate.contiguous(),
                                     mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core reduction: batch GEMM (left @ rightᵀ) for each hidden channel
    #    Shapes after re‑ordering:
    #        left_flat   : (B*H, N, N)   where each slice is left[..., i, k, h]
    #        right_flat_T: (B*H, N, N)   where each slice is right[..., k, j, h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # left: (B,N,N,H) -> (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # rightᵀ: first swap the two N dimensions, then bring H forward
    right_T = right.permute(0, 2, 1, 3)                      # (B, N(k), N(j), H)
    right_flat = right_T.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (Tensor‑core friendly)
    out_flat = torch.bmm(left_flat, right_flat)   # (B*H, N, N)

    # Restore original layout: (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection back to C
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8. Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
