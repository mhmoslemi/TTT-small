import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(out_ptr + offs, mask=mask)
    b = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul forward pass (no backward).
    Performs
        1. LayerNorm on input.
        2. Linear projections (left/right) and gates (left/right) fused.
        3. Optional mask applied to left/right projections.
        4. Gating (sigmoid) and element‑wise multiplication.
        5. Batched GEMM across the sequence dimension for each hidden channel.
        6. LayerNorm on hidden dimension.
        7. Out‑gate (sigmoid) applied via a small Triton kernel.
        8. Final linear projection back to original dimension.
    All heavy arithmetic is done in float16; final output is float32.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # --------------------------------------------------------------
    # Load weights
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device)  # keep float32 for layernorm
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # --------------------------------------------------------------
    # 1) LayerNorm on the input (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] float32

    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 2) Fuse four linear layers (proj & gate for left/right)
    #    Using a single matmul on the flattened tensor.
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)  # [B, N, N, dim] half
    x_flat = x_fp16.view(-1, dim)  # [(B*N*N), dim]

    # Concatenate weights: shape (4*hidden, dim)
    combined_weight = torch.cat(
        [left_proj_weight,
         left_gate_weight,
         right_proj_weight,
         right_gate_weight],
        dim=0,
    )  # (4*hidden, dim)

    # Linear transformation
    combined_out = F.linear(x_flat, combined_weight)  # [(B*N*N), 4*hidden]

    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each: (B, N, N, hidden)

    # --------------------------------------------------------------
    # 3) Apply optional mask to the projections (not to gates)
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------------------
    # 4) Gating (sigmoid) and combine with projections
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)   # (B, N, N, hidden)
    right_gate = torch.sigmoid(right_gate) # (B, N, N, hidden)

    left = left_proj * left_gate   # (B, N, N, hidden)
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5) Batched GEMM: out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a half‑precision batched matmul.
    # --------------------------------------------------------------
    # Prepare for batched matmul: reshape to (B*hidden, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Compute matmul: left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    # Reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6) LayerNorm on hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden) half

    # --------------------------------------------------------------
    # 7) Out‑gate (sigmoid) applied element‑wise via Triton kernel
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # (B, N, N, hidden) half

    # Triton kernel expects 1‑D contiguous tensors
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    total_el = out_flat.numel()
    BLOCK = 1024
    grid = ( (total_el + BLOCK - 1) // BLOCK, )
    elementwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        total_el,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 8) Final projection back to original dimension
    # --------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # (B, N, N, dim) half
    out = out.to(torch.float32)
    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, N, BLOCK: tl.constexpr):
    """
    In‑place element‑wise multiplication: out *= gate
    Arguments:
        out_ptr  – pointer to the output tensor (float16)
        gate_ptr – pointer to the gate tensor (float16)
        N        – total number of elements (int)
        BLOCK    – compile‑time block size
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(out_ptr + offs, mask=mask)
    b = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Steps:
        1. LayerNorm on the input tensor.
        2. Fuse left/right projections and their gates into a single
           half‑precision matmul (4 * hidden_dim rows).
        3. Optional mask applied to the left/right projections.
        4. Apply sigmoid gates and combine with projections.
        5. Batched half‑precision GEMM:
           out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].
        6. LayerNorm on the hidden dimension.
        7. Compute out‑gate (sigmoid) and apply it element‑wise via a
           small Triton kernel.
        8. Final linear projection back to the original dimension.

    All heavy arithmetic is performed in float16; the final result
    is returned as float32.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] float32

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2) Fuse four linear layers (proj & gate for left/right)
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)                     # [B, N, N, dim] half
    x_flat = x_fp16.view(-1, dim)                    # [(B·N·N), dim]

    # Concatenate the four weight matrices: (4*hidden, dim)
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    )  # (4*hidden, dim)

    # Single half‑precision linear transformation
    combined_out = F.linear(x_flat, combined_weight)  # [(B·N·N), 4*hidden]
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each: (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 3) Apply mask to projections (if required)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and combine with projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)   # (B, N, N, hidden) half
    right_gate = torch.sigmoid(right_gate) # (B, N, N, hidden) half

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Batched GEMM across the sequence dimension per hidden channel
    # ------------------------------------------------------------------
    # Reshape to (B*hidden, N, N) for batched matmul
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_mat[b*h, i, j] = Σₖ left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    # Restore original layout: (B, N, N, hidden)
    out = (
        out_mat.view(B, hidden_dim, N, N)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # (B, N, N, hidden) half

    # Triton kernel works on 1‑D contiguous tensors
    out_flat = out.view(-1)
    out_gate_flat = out_gate.view(-1)
    total_el = out_flat.numel()
    BLOCK = 1024

        out_flat,
        out_gate_flat,
        total_el,
        BLOCK=BLOCK,
    )

    # 8) Final projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # (B, N, N, dim) half
    out = out.to(torch.float32)          # return float32 as required
    return out
