"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1️⃣  Layer‑norm the input tensor (B,N,N,dim) using the supplied LN parameters.
2️⃣  Cast to fp16 for all compute‑heavy operations.
3️⃣  Fuse linear projections:
    - Concatenate left_proj, left_gate and out_gate weights → one matmul.
    - Concatenate right_proj and right_gate weights → one matmul.
   This reduces the number of GEMMs from five to three.
4️⃣  Slice the concatenated results into left, right, left_gate, right_gate,
    and out_gate tensors.
5️⃣  Apply optional binary mask (broadcast over the hidden dimension) to
    left/right.
6️⃣  Apply sigmoid to the gate tensors and fuse them into left/right.
7️⃣  Core O(N³·H) contraction:
    for each (b,h) compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    which is a batched matrix multiplication
        out[b,:, :,h] = left[b,:, :,h] @ right[b,:, :,h].transpose(-2,-1)
    Implemented via torch.bmm on the reshaped tensors.
8️⃣  Layer‑norm over the hidden dimension.
9️⃣  Multiply by the output gate – performed with a tiny Triton kernel.
🔟  Final linear projection back to the original channel dimension.
💎  Cast the result back to the input dtype and return.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, total_elems,
                            BLOCK_SIZE: tl.constexpr):
    """Element‑wise multiplication a * b → c (fp16, contiguous)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


def _elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton kernel to multiply two tensors element‑wise."""
    assert a.shape == b.shape
    total = a.numel()
    c = torch.empty_like(a)
    BLOCK_SIZE = 1024
    grid = ( (total + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    _elementwise_mul_kernel[grid](
        a, b, c, total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    return c


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            mask: torch.Tensor of shape [B, N, N] (or None)
            weights: dict containing model weights
            config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------------------------------------------------------
    # 1) Input LayerNorm (fp16)
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
    ).to(torch.float16)

    # ---------------------------------------------------------
    # 2) Fuse linear projections:
    #    left_proj + left_gate + out_gate
    #    right_proj + right_gate
    # ---------------------------------------------------------
    # Combined weight matrices (out_features, in_features)
    left_comb_weight = torch.cat([
        weights["left_proj.weight"].to(torch.float16),
        weights["left_gate.weight"].to(torch.float16),
        weights["out_gate.weight"].to(torch.float16),
    ], dim=0)                            # (3*H, dim)

    right_comb_weight = torch.cat([
        weights["right_proj.weight"].to(torch.float16),
        weights["right_gate.weight"].to(torch.float16),
    ], dim=0)                            # (2*H, dim)

    # Single matmul for each combined weight
    left_comb = F.linear(x, left_comb_weight)   # (B, N, N, 3*H)
    right_comb = F.linear(x, right_comb_weight) # (B, N, N, 2*H)

    # ---------------------------------------------------------
    # 3) Slice tensors
    # ---------------------------------------------------------
    H = hidden_dim
    left = left_comb[..., :H]                    # (B, N, N, H)
    left_gate = left_comb[..., H:2*H]            # (B, N, N, H)
    out_gate = left_comb[..., 2*H:]              # (B, N, N, H)

    right = right_comb[..., :H]                  # (B, N, N, H)
    right_gate = right_comb[..., H:]             # (B, N, N, H)

    # ---------------------------------------------------------
    # 4) Apply mask (if present and not nomask)
    # ---------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ---------------------------------------------------------
    # 5) Gates (sigmoid) and fuse
    # ---------------------------------------------------------
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 6) Core O(N³·H) reduction via batched GEMM (cuBLAS/Tensor‑Cores)
    #    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # ---------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()

    # left_mat @ right_mat^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------------------------------------------
    # 7) Output LayerNorm over hidden dimension
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ---------------------------------------------------------
    # 8) Apply output gate via Triton element‑wise multiplication
    # ---------------------------------------------------------
    out = _elementwise_mul(out, out_gate)

    # ---------------------------------------------------------
    # 9) Final linear projection back to original channel dimension
    # ---------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16)  # (dim, H)
    )

    # ---------------------------------------------------------
    # 10) Cast back to original dtype
    # ---------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
