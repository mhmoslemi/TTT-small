"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑Norm the input tensor (B,N,N,C) with supplied LN parameters.
2. Cast to fp16 and apply the four linear projections (left, right, gates) via
   torch.nn.functional.linear (small cost compared to the O(N³·H) contraction).
3. Fuse gating scalars into the projected tensors.
4. Core contraction:
     out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden slice.
   Implemented in a custom Triton kernel that tiles the (i,j) plane (8×8)
   and processes the hidden dimension in chunks of 32.  The kernel loads
   tiles of left/right for a given hidden chunk, computes the outer‑product
   across the reduction dimension k (blocked by 32), accumulates in FP32,
   and writes the result in FP16.
5. Apply a second Layer‑Norm over the hidden dimension and multiply by the
   output‑gate (element‑wise, performed with vanilla PyTorch).
6. Final linear projection back to the original channel dimension.
7. Cast back to the input dtype.

The heavy O(N³·H) work is fully fused inside the Triton kernel, avoiding the
overhead of many small cuBLAS calls while keeping memory traffic modest.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel for the core TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def _triu_mul_kernel(
    left_ptr,            # *fp16   [B,N,N,H]
    right_ptr,           # *fp16   [B,N,N,H]
    out_ptr,             # *fp16   [B,N,N,H]
    stride_batch,        # int64   stride of batch dimension (N*N*H)
    stride_i,            # int64   stride of first sequence dimension (N*H)
    stride_k,            # int64   stride of second sequence dimension (H)
    stride_h,            # int64   stride of hidden dimension (1)
    B, N, H,             # runtime ints
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Tile indices (batch, i‑tile, j‑tile)
    # --------------------------------------------------------------
    total_i_tiles = tl.cdiv(N, BLOCK_I)
    total_j_tiles = tl.cdiv(N, BLOCK_J)
    batch = pid // (total_i_tiles * total_j_tiles)
    pid_tile = pid % (total_i_tiles * total_j_tiles)
    tile_i = pid_tile // total_j_tiles
    tile_j = pid_tile % total_j_tiles

    i_start = tile_i * BLOCK_I
    j_start = tile_j * BLOCK_J

    i_off = i_start + tl.arange(0, BLOCK_I)
    j_off = j_start + tl.arange(0, BLOCK_J)

    i_mask = i_off < N
    j_mask = j_off < N

    # --------------------------------------------------------------
    # Loop over hidden dimension in chunks of BLOCK_H
    # --------------------------------------------------------------
    h_start = 0
    while h_start < H:
        h_off = h_start + tl.arange(0, BLOCK_H)
        h_mask = h_off < H

        # accumulator in FP32
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # ----------------------------------------------------------
        # Reduction over the k dimension (blocked)
        # ----------------------------------------------------------
        k_start = 0
        while k_start < N:
            k_off = k_start + tl.arange(0, BLOCK_K)
            k_mask = k_off < N

            # Pointers for the current tile
            left_ptrs = (left_ptr
                         + batch * stride_batch
                         + i_off[:, None, None] * stride_i
                         + k_off[None, :, None] * stride_k
                         + h_off[None, None, :] * stride_h)
            right_ptrs = (right_ptr
                          + batch * stride_batch
                          + j_off[:, None, None] * stride_i
                          + k_off[None, :, None] * stride_k
                          + h_off[None, None, :] * stride_h)

            # Load tiles, zero‑pad out‑of‑bounds entries
            left_tile = tl.load(
                left_ptrs,
                mask=i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :],
                other=0.0).to(tl.float32)          # (B_I, B_K, B_H)
            right_tile = tl.load(
                right_ptrs,
                mask=j_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :],
                other=0.0).to(tl.float32)         # (B_J, B_K, B_H)

            # Broadcast across i/j and sum over k:
            # left_tile: (B_I, B_K, B_H) -> (B_I, B_K, 1, B_H)
            # right_tile: (B_J, B_K, B_H) -> (1, B_K, B_J, B_H)
            prod = left_tile[:, :, None, :] * right_tile[None, :, :, :]   # (B_I, B_K, B_J, B_H)
            acc += tl.sum(prod, axis=1)                                 # (B_I, B_J, B_H)

            k_start += BLOCK_K

        # ----------------------------------------------------------
        # Write the computed block to global memory
        # ----------------------------------------------------------
        out_ptrs = (out_ptr
                    + batch * stride_batch
                    + i_off[:, None, None] * stride_i
                    + j_off[None, :, None] * stride_k   # stride_k == stride for the j‑dim (H)
                    + h_off[None, None, :] * stride_h)

        tl.store(
            out_ptrs,
            acc.to(tl.float16),
            mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :]
        )

        h_start += BLOCK_H


# ----------------------------------------------------------------------
# Helper for element‑wise multiplication (optional, kept for completeness)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, c_ptr, total, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    assert a.shape == b.shape
    total = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ( (total + BLOCK - 1) // BLOCK, )
    _elemwise_mul_kernel[grid](
        a, b, out, total,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            mask: torch.Tensor of shape [B, N, N] (or None)
            weights: dict of model weights (float32)
            config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2) Cast to fp16 for the compute‑heavy path
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Prepare fp16 weights (store once, reuse)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16) # (H, C)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)  # (H, C)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16) # (H, C)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)   # (H, C)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)       # (C, H)

    # --------------------------------------------------------------
    # 4) Linear projections (left/right) and gates
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # (B,N,N,H)
    right = F.linear(x, right_proj_w)  # (B,N,N,H)

    # --------------------------------------------------------------
    # 5) Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 6) Gates (sigmoid) and fusion
    # --------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()   # (B,N,N,H)
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()     # (B,N,N,H)

    left = left * left_gate
    right = right * right_gate
    # free memory early
    del left_gate, right_gate, x

    # --------------------------------------------------------------
    # 7) Core O(N³·H) contraction via Triton
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float16, device=left.device)

    # Strides for [B, N, N, H] tensor (contiguous in PyTorch)
    stride_batch, stride_i, stride_k, stride_h = left.stride()

    # Kernel launch configuration
    BLOCK_I, BLOCK_J, BLOCK_K, BLOCK_H = 8, 8, 32, 32
    total_i_tiles = (N + BLOCK_I - 1) // BLOCK_I
    total_j_tiles = (N + BLOCK_J - 1) // BLOCK_J
    grid = (B * total_i_tiles * total_j_tiles, )

    _triu_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        stride_batch,
        stride_i,
        stride_k,
        stride_h,
        B, N, H,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 8) Output layer‑norm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 9) Apply output gate (element‑wise)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 10) Final linear projection back to original channel dimension
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 11) Cast back to original dtype if needed
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
