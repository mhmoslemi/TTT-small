"""
Outgoing Triangle Multiplicative Update (TriMul) – optimized forward pass.

Algorithm:
1. Layer‑norm the input tensor (float16).
2. Fuse the four linear projections (left_proj, left_gate, right_proj, right_gate)
   into a single matrix multiplication and split the result.
3. Apply optional mask to the left/right projections.
4. Apply sigmoid gates and combine with projections.
5. Compute the N‑wise reduction as a batched matrix multiplication:
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   which is realized as a batch of GEMMs (B·hidden) of size N×N.
6. Layer‑norm over the hidden dimension.
7. Compute the out‑gate from the normalized input and apply it element‑wise
   using a tiny Triton kernel.
8. Final linear projection back to the original channel dimension and cast to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel for element‑wise multiplication (out *= out_gate)
# --------------------------------------------------------------------------- #
@triton.jit
def _elem_mul_kernel(out_ptr, gate_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    out_val = tl.load(out_ptr + offs, mask=mask)
    gate_val = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out_val * gate_val, mask=mask)

# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config["nomask"] is True
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # Cast weights to half precision (fp16) on the target device
    # --------------------------------------------------------------
    dtype = torch.float16
    norm_weight = weights["norm.weight"].to(device, dtype)
    norm_bias   = weights["norm.bias"].to(device, dtype)

    left_proj_weight  = weights["left_proj.weight"].to(device, dtype)
    right_proj_weight = weights["right_proj.weight"].to(device, dtype)
    left_gate_weight  = weights["left_gate.weight"].to(device, dtype)
    right_gate_weight = weights["right_gate.weight"].to(device, dtype)
    out_gate_weight   = weights["out_gate.weight"].to(device, dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device, dtype)

    to_out_weight = weights["to_out.weight"].to(device, dtype)   # (dim, hidden)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (float16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    ).to(dtype)                     # [B, N, N, dim] fp16

    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 2️⃣ Fuse four linear layers into a single matmul
    # --------------------------------------------------------------
    # Combined weight: (4*hidden_dim, dim)
    combined_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
        ],
        dim=0,
    )  # (4*hidden_dim, dim)

    # Flatten spatial dims and apply linear map
    x_flat = x.view(-1, dim)                         # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 4*hidden_dim)

    # Restore 4‑D layout and split
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each: (B, N, N, hidden)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied to projections only)
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)   # (B, N, N, 1)
        left_proj  = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and combine with projections
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left  = left_proj * left_gate   # (B, N, N, hidden)
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Batched GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) for batched matmul
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    # Merge batch and hidden dimensions
    left_mat  = left.view(B * hidden_dim, N, N)   # (B*hidden, N, N)
    right_mat = right.view(B * hidden_dim, N, N)  # (B*hidden, N, N)

    # Batched matmul: (B*hidden, N, N) @ (B*hidden, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    # Reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 6️⃣ LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden) fp16

    # --------------------------------------------------------------
    # 7️⃣ Out‑gate computation & element‑wise multiplication via Triton
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # (B, N, N, hidden) fp16

    # Triton kernel launch
    total_el = out.numel()
    BLOCK = 1024
    grid = ((total_el + BLOCK - 1) // BLOCK,)
    _elem_mul_kernel[grid](
        out,
        out_gate,
        total_el,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 8️⃣ Final projection back to original dimension
    # --------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # (B, N, N, dim) fp16
    out = out.to(torch.float32)          # cast to required output dtype

    return out
