@triton.jit
def trimul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Ai, stride_Ak,
    stride_Bb, stride_Bk, stride_Bj,
    stride_Cb, stride_Ci, stride_Cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)
    # compute offsets for i and j
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        # compute pointers
        a_ptrs = A_ptr + batch_id * stride_Ab + offs_i[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        b_ptrs = B_ptr + batch_id * stride_Bb + offs_k[:, None] * stride_Bk + offs_j[None, :] * stride_Bj
        # load tiles
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)
        acc += tl.dot(a, b)
    # store to C
    c_ptrs = C_ptr + batch_id * stride_Cb + offs_i[:, None] * stride_Ci + offs_j[None, :] * stride_Cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) forward pass optimized with a custom Triton kernel.

The kernel computes the N³ reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
as a batched matrix multiplication left @ rightᵀ for each (batch, head) pair.
All heavy work (linear projections, gating and the N³ reduction) is performed
in fp16 to leverage Tensor Cores.  The final LayerNorms and output projection
are kept in fp32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------
# Triton kernel: batched_matmul (left @ rightᵀ)
# ---------------------------------------------------------
@triton.jit
def trimul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    BATCH, N, K,                       # dimensions (BATCH = B*H, N=seq_len, K=seq_len)
    stride_A_batch, stride_A_i, stride_A_k,
    stride_B_batch, stride_B_k, stride_B_j,
    stride_C_batch, stride_C_i, stride_C_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)                # which (batch, head) matrix
    pid_i = tl.program_id(1)                   # block row
    pid_j = tl.program_id(2)                   # block column

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A: [BLOCK_M, BLOCK_K]   (row i, col k)
        a_ptrs = (A_ptr + batch_id * stride_A_batch
                  + offs_i[:, None] * stride_A_i
                  + offs_k[None, :] * stride_A_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Bᵀ: we need rows = k, cols = j
        # B layout is (batch, row=j, col=k). To read Bᵀ we index [k, j].
        b_ptrs = (B_ptr + batch_id * stride_B_batch
                  + offs_k[:, None] * stride_B_k
                  + offs_j[None, :] * stride_B_j)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # fp16 dot -> fp32 accumulation
        acc += tl.dot(a, b)

    # store result as fp16
    c_ptrs = (C_ptr + batch_id * stride_C_batch
              + offs_i[:, None] * stride_C_i
              + offs_j[None, :] * stride_C_j)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

# ---------------------------------------------------------
# Entry point
# ---------------------------------------------------------
def custom_kernel(data):
    """
    Perform the outgoing TriMul forward pass.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model weight tensors
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # -------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # -------------------------------------------------
    # 2. Cast to fp16 for the heavy part
    # -------------------------------------------------
    x_h = x.to(torch.float16)

    # -------------------------------------------------
    # 3. Linear projections (fp16)
    # -------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # -------------------------------------------------
    # 4. Gates (pre‑activation)
    # -------------------------------------------------
    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].to(torch.float16))
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].to(torch.float16))
    out_gate_pre = F.linear(x_h, weights["out_gate.weight"].to(torch.float16))

    # -------------------------------------------------
    # 5. Optional mask (broadcast on hidden dimension)
    # -------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # -------------------------------------------------
    # 6. Apply sigmoid gates (fp16)
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 7. Batched matmul left @ rightᵀ via Triton
    # -------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim
    # Move hidden_dim to its own batch dimension for the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    M = B * H                      # total number of independent matrices
    left_flat = left_perm.view(M, N, N)   # [M, N, N]
    right_flat = right_perm.view(M, N, N) # [M, N, N]

    out_flat = torch.empty_like(left_flat)   # fp16 output

    # Strides (batch, row, col)
    stride_A_batch, stride_A_i, stride_A_k = left_flat.stride()
    stride_B_batch, stride_B_i, stride_B_k = right_flat.stride()
    stride_C_batch, stride_C_i, stride_C_j = out_flat.stride()

    # Triton block sizes (tuned for Tensor Cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        M,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N,
        N,
        stride_A_batch,
        stride_A_i,
        stride_A_k,
        stride_B_batch,
        stride_B_k,   # k‑dimension stride (original column)
        stride_B_i,   # j‑dimension stride (original row)
        stride_C_batch,
        stride_C_i,
        stride_C_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [B, N, N, H]
    out_h = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # -------------------------------------------------
    # 8. Cast back to fp32 for the final layers
    # -------------------------------------------------
    out = out_h.to(torch.float32)

    # -------------------------------------------------
    # 9. Output LayerNorm, gating and final projection
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate  # apply output gate (broadcasted)
    out = F.linear(out, weights["to_out.weight"])  # final projection to dim

    return out
