"""
TriMul (outgoing) forward pass – Triton‑accelerated batched GEMM.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the provided LN parameters.
2. Cast to fp16 for compute‑heavy ops.
3. Linear projections (left/right) to the hidden dimension H.
4. Compute gate vectors (left_gate, right_gate, out_gate) via linear+sigmoid.
5. Apply optional mask (broadcast over H) and multiply by the gates:
       left  = left_proj  * mask * left_gate
       right = right_proj * mask * right_gate
   (If `nomask` is True the mask is ignored.)
6. Reshape to (B*H, N, N) and perform a batched matrix multiplication:
       out[b*h,i,j] = Σ_k left[b*h,i,k] * right[b*h,j,k]
   This is implemented as a fused Triton GEMM kernel.
7. Reshape back to (B,N,N,H), apply a second LayerNorm over H,
   multiply by out_gate and a final linear projection back to C.
8. Cast the result to the original input dtype.

The heavy O(N³·H) reduction is the only part executed on Triton;
all other operations use native PyTorch (which is already fast for the
remaining O(B·N²·H) work).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (C = A @ Bᵀ) for tensors of shape
#   A : (batch, M, K)   (fp16)
#   B : (batch, K, N)   (fp16)
#   C : (batch, M, N)   (fp16)
#   where M = N = seqlen, K = seqlen.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,          # fp16*  (batch, M, K)
    b_ptr,          # fp16*  (batch, K, N)
    c_ptr,          # fp16*  (batch, M, N)
    batch,          # int: total number of batches (B*H)
    M, N, K,        # matrix dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the output tile
    c_ptrs = (
        c_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = (
            a_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B tile (K x N)
        b_ptrs = (
            b_ptr
            + batch_id * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        # Accumulate the block product
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Write back the result in fp16
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def batched_gemm(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 64,
    BLOCK_N: int = 64,
    BLOCK_K: int = 32,
    num_warps: int = 4,
) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ for 3‑D tensors (batch, N, N) in fp16.
    A: (batch, M, K)  fp16
    B: (batch, K, N)  fp16
    Returns C: (batch, M, N) fp16
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, Kb, N = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    grid = (batch, (M + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    # Strides (in elements) for the three tensors
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    batched_gemm_kernel[grid](
        A,
        B,
        C,
        batch,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        num_warps=num_warps,
    )
    return C

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections to hidden_dim
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5. Compute gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Apply optional mask and gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) -> broadcast over H
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left_proj * mask_fp16
        right = right_proj * mask_fp16
    else:
        left = left_proj
        right = right_proj

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7. Reshape for batched GEMM: (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # For right we need the transpose on the last two axes before reshaping
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # ------------------------------------------------------------------
    # 8. Batched matrix multiplication via Triton
    # ------------------------------------------------------------------
    out_mat = batched_gemm(left_mat, right_mat)  # (B*H, N, N) fp16

    # ------------------------------------------------------------------
    # 9. Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 10. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, C)

    # ------------------------------------------------------------------
    # 11. Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
