    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # Load weights
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # normalized input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # Cast to fp16 for rest of computation
    x_fp16 = x.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x_fp16, left_proj_weight)  # (B,N,N,hidden)
    right = torch.nn.functional.linear(x_fp16, right_proj_weight)

    # Optional mask
    if not nomask:
        mask_f = mask.to(x_fp16.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Batched GEMM: out = left @ right^T over k dimension
    B, N, _, _ = left.shape

    # Permute to (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden,N,N)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # Multiply by out_gate elementwise using Triton
    total_elems = out.numel()
    BLOCK = 1024
    grid = ( (total_elems + BLOCK - 1) // BLOCK, )
    elementwise_mul_kernel[grid](
        out,
        out_gate,
        out,
        total_elems,
        BLOCK=BLOCK,
    )

    # Final projection back to dim
    out = torch.nn.functional.linear(out, to_out_weight)  # (B,N,N,dim) fp16
    out = out.to(torch.float32)

"""
Outgoing Triangle Multiplicative Update (TriMul) – forward pass.

The implementation follows the PyTorch reference but fuses several steps
and moves the final element‑wise out‑gate multiplication to a tiny Triton
kernel. The core O(N³) reduction is expressed as a batched half‑precision
matrix multiplication:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batch of GEMMs of shape (hidden, N, N).
All heavy arithmetic is performed in float16 (fp16) to leverage tensor
cores; the result is cast back to float32 before returning.

The required operations are:
1. LayerNorm over the channel dimension.
2. Linear projections for left/right and their gates.
3. Optional masking of left/right projections.
4. Sigmoid gating and element‑wise multiplication.
5. Batched GEMM to compute the triangle multiplicative update.
6. LayerNorm over the hidden dimension.
7. Element‑wise multiplication with the out‑gate (performed by Triton).
8. Final linear projection back to the original dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """c = a * b  (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float); may be ignored.
            - weights: dict containing the model's weight tensors.
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ---------- Load and cast weights ----------
    # LayerNorm on input
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    # Linear projection / gate weights (fp16 for speed)
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    # Output LayerNorm and projection
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # ---------- 1️⃣ Input LayerNorm ----------
    # Normalize over the channel dimension (dim)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape: [B, N, N, dim], dtype: float32

    # Cast to fp16 for the remaining heavy work
    x_fp16 = x.to(torch.float16)

    # ---------- 2️⃣ Linear projections ----------
    left = F.linear(x_fp16, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x_fp16, right_proj_weight)  # [B, N, N, hidden]

    # ---------- 3️⃣ Optional masking ----------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] and cast to fp16
        mask_f = mask.to(x_fp16.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ---------- 4️⃣ Gates ----------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))   # [B, N, N, hidden]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))  # [B, N, N, hidden]
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))   # [B, N, N, hidden]

    left = left * left_gate
    right = right * right_gate

    # ---------- 5️⃣ Triangle multiplicative reduction ----------
    B, N, _, _ = left.shape

    # Rearrange to (B, hidden, N, N) and view as (B*hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * hidden_dim, N, N)    # (B*hidden, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)  # (B*hidden, N, N)

    # Batched matrix multiplication: left @ right^T over the k‑dimension
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    # Reshape back to [B, N, N, hidden]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # ---------- 6️⃣ Output LayerNorm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # fp16

    # ---------- 7️⃣ Apply out‑gate (Triton element‑wise multiplication) ----------
    total_elements = out.numel()
    BLOCK = 1024
    grid = ((total_elements + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        out,
        out_gate,
        out,
        total_elements,
        BLOCK=BLOCK,
    )

    # ---------- 8️⃣ Final linear projection ----------
    out = F.linear(out, to_out_weight)  # [B, N, N, dim] in fp16
    out = out.to(torch.float32)        # required output dtype

    return out
