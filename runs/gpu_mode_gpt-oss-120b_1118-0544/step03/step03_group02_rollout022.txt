"""
Fast Triton implementation of the outgoing Triangle Multiplicative Update (TriMul).

The forward pass performs:
  1️⃣ LayerNorm over the channel dimension.
  2️⃣ Four linear projections (left/right & their gates) and a sigmoid
     gate on the input.
  3️⃣ Optional masking of the left/right projections.
  4️⃣ Element‑wise gating of the projected tensors.
  5️⃣ Batched GEMM:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
     implemented as a fused Triton kernel (massive O(N³) work).
  6️⃣ LayerNorm over the hidden dimension, out‑gate multiplication,
     and the final linear projection back to the original dimension.

Only the heavy matrix multiplication is done in Triton; all other
operations use PyTorch (mixed‑precision fp16) for simplicity.
The kernel works for any batch size (≤2), sequence length (≤1024) and
hidden dimension (≤384) supported by the test matrix.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ Bᵀ  (A,B ∈ (batch, M, K), C ∈ (batch, M, N)).
       A and B are stored row‑major; B is interpreted as transposed via
       swapped strides (k ↔ n).  Accumulation is done in FP32, result stored
       in FP16.
    """
    pid_b = tl.program_id(0)   # batch index
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block column

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to the start of this batch
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M x K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_a_m) + (cur_k[None, :] * stride_a_k),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load B tile (K x N) – B is interpreted as transposed
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Write back FP16 result
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config["nomask"] is True
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # Load and cast model weights
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)

    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the input channel dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # Switch to fp16 for the remaining heavy work
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections & gate projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_h, left_proj_w)          # [B, N, N, hidden]
    right_proj = F.linear(x_h, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (applied only to the projections)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x_h.dtype).unsqueeze(-1)               # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Apply gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate       # [B, N, N, hidden]  fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM via Triton:
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Bring hidden dimension next to batch and make tensors contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]

    batch_hidden = B * hidden_dim
    left_mat = left_perm.view(batch_hidden, N, N)   # (BH, M, K)
    right_mat = right_perm.view(batch_hidden, N, N) # (BH, K, N) after transposition in kernel

    out_mat = torch.empty_like(left_mat)            # (BH, M, N) fp16

    # Strides for A (left_mat) – row‑major
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    # Strides for B (right_mat) interpreted as transposed
    stride_b_batch = right_mat.stride(0)
    stride_b_k = right_mat.stride(2)   # original column stride => becomes K stride
    stride_b_n = right_mat.stride(1)   # original row stride => becomes N stride

    # Strides for C (output)
    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    # Block sizes (tuned for fp16 tensor‑cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch_hidden,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _batched_matmul_kernel[grid](
        left_mat.data_ptr(),
        right_mat.data_ptr(),
        out_mat.data_ptr(),
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Restore the original 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (over hidden dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Apply out‑gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)          # [B, N, N, dim]  fp16
    out = out.to(torch.float32)            # cast to required dtype

    return out
```
