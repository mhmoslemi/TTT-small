"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The core O(N³·H) reduction:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed with a custom batched GEMM kernel written in Triton.
All other operations (LayerNorm, Linear, gating) are executed in PyTorch
with mixed‑precision (fp16) to keep the kernel simple and fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    batch, M, N, K,                     # dimensions
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication: C = A @ Bᵀ  (A,B shape = [batch,M,K])"""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = (a_ptr + pid_batch * stride_a_batch
                  + offs_m[:, None] * stride_a_row
                  + offs_k[None, :] * stride_a_col)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Bᵀ tile: (BLOCK_K, BLOCK_N)  (load transposed)
        b_ptrs = (b_ptr + pid_batch * stride_b_batch
                  + offs_k[:, None] * stride_b_row
                  + offs_n[None, :] * stride_b_col)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot product over K
        acc += tl.dot(a, b)   # a:(M,K), b:(K,N) -> (M,N)

    # write back C tile
    c_ptrs = (c_ptr + pid_batch * stride_c_batch
              + offs_m[:, None] * stride_c_row
              + offs_n[None, :] * stride_c_col)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size,
                         BLOCK: tl.constexpr):
    """1‑D element‑wise multiplication a * b."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton element‑wise kernel on two tensors of identical shape."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication."
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ( (size + BLOCK - 1) // BLOCK, )
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype          # original dtype (usually float32)
    half = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Cast model weights to fp16 once
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(half).contiguous()

    norm_weight = to_half(weights["norm.weight"])
    norm_bias = to_half(weights["norm.bias"])

    left_proj_weight = to_half(weights["left_proj.weight"])
    right_proj_weight = to_half(weights["right_proj.weight"])

    left_gate_weight = to_half(weights["left_gate.weight"])
    right_gate_weight = to_half(weights["right_gate.weight"])
    out_gate_weight = to_half(weights["out_gate.weight"])

    to_out_norm_weight = to_half(weights["to_out_norm.weight"])
    to_out_norm_bias = to_half(weights["to_out_norm.bias"])
    to_out_weight = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (fp16)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    x = x.to(half)

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight) # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3. Optional masking
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(half).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and gate application
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·H) reduction via Triton batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    batch = B * hidden_dim               # treat each (b,h) as independent batch

    # reshape to (batch, N, N) where batch = B*H
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(batch, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(batch, N, N)

    # output buffer
    out_flat = torch.empty_like(left_flat)

    # strides for pointer arithmetic
    stride_a_batch, stride_a_row, stride_a_col = left_flat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_flat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_flat.stride()

    # tiling choices (tuned for fp16 tensor cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch, N, N, N,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # restore shape to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # ------------------------------------------------------------------
    # 7. Apply output gate (elementwise multiplication via Triton)
    # ------------------------------------------------------------------
    out = _elemwise_mul(out, out_gate)

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != dtype:
        out = out.to(dtype)

    return out
