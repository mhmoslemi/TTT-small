def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # 1) Input LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=1e-5) # float32

    # 2) Cast to fp16
    x_h = x.to(torch.float16)

    # 3) Project left+gate, right+gate (combined) to hidden_dim using FP16 linear
    left_combined_weight = torch.cat([weights['left_proj.weight'], weights['left_gate.weight']], dim=0).to(torch.float16) # shape (2*hidden_dim, dim)
    right_combined_weight = torch.cat([weights['right_proj.weight'], weights['right_gate.weight']], dim=0).to(torch.float16)
    # Do linear
    left_combined = torch.nn.functional.linear(x_h, left_combined_weight)  # shape (B,N,N,2*hidden_dim)
    right_combined = torch.nn.functional.linear(x_h, right_combined_weight)
    # split
    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    # 4) Apply mask to projections if needed
    if not config.get('nomask', False) and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1) # [B,N,N,1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # 5) Gate(sigmoid) and combine
    left_gate = torch.sigmoid(left_gate_pre)

def tri_mul_matmul_kernel(A, B, C,
                          M, N, K,
                          stride_am, stride_ak,
                          stride_bk, stride_bn,
                          stride_cm, stride_cn,
                          batch_stride_a, batch_stride_b, batch_stride_c,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # compute pointers with mask
        a_ptrs = A + batch_id * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = B + batch_id * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # mask for bounds
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

    # Write out
    c_ptrs = C + batch_id * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=c_mask)

"""
Fast outgoing Triangle Multiplicative Update (TriMul) implementation.

The heavy N³ reduction is performed by a custom Triton kernel that computes,
for each batch element *hidden* slice, the matrix product
    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
which is exactly the einsum used in AlphaFold3.
All large linear projections and the reduction are carried out in fp16 to
exploit Tensor Core acceleration, while the final LayerNorm and output
projection are done in fp32 for numerical stability.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_matmul_kernel(
    A, B, C,               # [batch, M, K], [batch, K, N] (as transposed), [batch, M, N]
    M, N, K,               # problem sizes (int)
    stride_am, stride_ak,  # strides for A (row, col)
    stride_bk, stride_bn,  # strides for B (row=k, col=n) -> B is treated as K×N
    stride_cm, stride_cn,  # strides for C (row, col)
    batch_stride_a, batch_stride_b, batch_stride_c,  # batch strides (elements)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul C = A @ Bᵀ  (A: [M,K], Bᵀ: [K,N]) for many batches."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Pointers with batch offset
        a_ptrs = A + batch_id * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = B + batch_id * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # masks for edges
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N) += (BLOCK_M, BLOCK_K)@(BLOCK_K, BLOCK_N)

    # write out
    c_ptrs = C + batch_id * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model weights (see skeleton)
            - config: dict with at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # -----------------------------------------------------------------
    # 2) Cast to fp16 for the heavy part
    # -----------------------------------------------------------------
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Project + gate (fused linear for proj & gate)
    # -----------------------------------------------------------------
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, device=device)               # (2*H, dim)
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, device=device)               # (2*H, dim)

    left_combined = F.linear(x_h, left_combined_w)   # [B, N, N, 2*H]
    right_combined = F.linear(x_h, right_combined_w)

    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    # -----------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # -----------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_h = mask.to(torch.float16, device=device).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # -----------------------------------------------------------------
    # 5) Gating (sigmoid) and combine
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate          # [B, N, N, H] fp16
    right = right_proj * right_gate       # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 6) Output gate (fp16)
    # -----------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16, device=device)
    out_gate_pre = F.linear(x_h, out_gate_w)                     # [B, N, N, H] fp16
    out_gate = torch.sigmoid(out_gate_pre)                       # fp16

    # -----------------------------------------------------------------
    # 7) Batched matrix multiplication via Triton: out = left @ rightᵀ per hidden slice
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (batch*H, N, N) where the last two dims are (M=row, K=col)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # output buffer (fp16)
    out_fp16 = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # block sizes (tuned for Tensor Cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides (elements, not bytes)
    stride_am = left_flat.stride(1)          # = N
    stride_ak = left_flat.stride(2)          # = 1
    stride_bk = right_flat.stride(2)         # = 1  (K dimension of Bᵀ)
    stride_bn = right_flat.stride(1)         # = N  (N dimension of Bᵀ)
    stride_cm = out_fp16.stride(1)           # = N
    stride_cn = out_fp16.stride(2)           # = 1

    batch_stride_a = left_flat.stride(0)     # = N*N
    batch_stride_b = right_flat.stride(0)    # = N*N
    batch_stride_c = out_fp16.stride(0)      # = N*N

    # grid: (batch, blocks_m, blocks_n)
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_fp16,
        N,  # M
        N,  # N
        N,  # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to [B, N, N, H] (fp16)
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 8) Final LayerNorm, gating and projection (float32)
    # -----------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate.to(torch.float32)  # apply output gate

    # final linear projection back to original dimension
    final = F.linear(out_norm, weights["to_out.weight"])

    return final
