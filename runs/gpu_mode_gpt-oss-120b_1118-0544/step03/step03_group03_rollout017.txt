def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_az, stride_ai, stride_ak,
    stride_bz, stride_bj, stride_bk,
    stride_cz, stride_ci, stride_cj,

    BLOCK_N: tl.constexpr,

computes, for each head `h` (flattened batch‑head dimension), the
matrix product:

    out[h, i, j] = sum_k left[h, i, k] * right[h, j, k]

which is equivalent to the einsum used in the reference PyTorch
implementation.  All other operations (LayerNorm, linear projections,
gating, and final LayerNorm + projection) are carried out with PyTorch
in FP32/FP16.  The kernel runs in FP16 with FP32 accumulation and
produces an FP16 result that is cast back to FP32 for the final layers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,                     # all equal to sequence length
    stride_az, stride_ai, stride_ak,   # A stride: (z, i, k)
    stride_bz, stride_bj, stride_bk,   # B stride: (z, j, k) – B is accessed as (k, j)
    stride_cz, stride_ci, stride_cj,   # C stride: (z, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # -----------------------------------------------------------
    # Compute 3‑D grid coordinates: (z, block_i, block_j)
    # -----------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_z = num_blocks_m * num_blocks_n

    z = pid // blocks_per_z
    pid_in_z = pid % blocks_per_z
    block_i = pid_in_z // num_blocks_n
    block_j = pid_in_z % num_blocks_n

    # -----------------------------------------------------------
    # Offsets within the current block
    # -----------------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # -----------------------------------------------------------
    # Pointers to the current slices of A, B and C
    # -----------------------------------------------------------
    a_ptrs = A_ptr + z * stride_az + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + z * stride_bz + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
    c_ptrs = C_ptr + z * stride_cz + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj

    # -----------------------------------------------------------
    # Masks for out‑of‑bounds loads / stores
    # -----------------------------------------------------------
    mask_a = (offs_i[:, None] < M) & (offs_k[None, :] < K)
    mask_b = (offs_k[:, None] < K) & (offs_j[None, :] < N)
    mask_c = (offs_i[:, None] < M) & (offs_j[None, :] < N)

    # -----------------------------------------------------------
    # Accumulator in FP32
    # -----------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------
    # Loop over the K dimension
    # -----------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float32)
        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

        a_ptrs += BLOCK_K * stride_ak           # move to next K‑block in A
        b_ptrs += BLOCK_K * stride_bk           # move to next K‑block in B

    # -----------------------------------------------------------
    # Write the result back in FP16
    # -----------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_c)


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model weights
            - config: dict with at least "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # -----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # -----------------------------------------------------------
    # 2) Cast to FP16 for the heavy part
    # -----------------------------------------------------------
    x_h = x.to(torch.float16)

    # -----------------------------------------------------------
    # 3) Linear projections and gates (FP16)
    # -----------------------------------------------------------
    left_proj = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))          # [B, N, N, H]
    right_proj = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))        # [B, N, N, H]

    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].to(torch.float16))
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].to(torch.float16))
    out_gate_pre = F.linear(x_h, weights["out_gate.weight"].to(torch.float16))

    # -----------------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # -----------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # -----------------------------------------------------------
    # 5) Gating (sigmoid, FP16)
    # -----------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)   # will be used later (FP16)

    left = left_proj * left_gate   # [B, N, N, H]
    right = right_proj * right_gate

    # -----------------------------------------------------------
    # 6) Batched matrix multiplication: out = left @ rightᵀ
    #    (implemented in Triton)
    # -----------------------------------------------------------
    B, N, _, _ = left.shape
    Z = B * hidden_dim  # flattened batch‑head dimension

    # Rearrange to (Z, N, N) for the kernel
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(Z, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(Z, N, N)

    out_fp16 = torch.empty((Z, N, N), dtype=torch.float16, device=device)

    # Strides (in elements)
    stride_lz, stride_li, stride_lk = left_flat.stride()
    stride_rz, stride_rj, stride_rk = right_flat.stride()
    stride_oz, stride_oi, stride_oj = out_fp16.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (Z * num_blocks_m * num_blocks_n,)

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_fp16,
        N, N, N,
        stride_lz, stride_li, stride_lk,
        stride_rz, stride_rj, stride_rk,
        stride_oz, stride_oi, stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # -----------------------------------------------------------
    # 7) Reshape back to [B, N, N, H] and cast to FP32
    # -----------------------------------------------------------
    out_h = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_h.to(torch.float32)   # [B, N, N, H]

    # -----------------------------------------------------------
    # 8) Final LayerNorm, gating and projection back to original dim (FP32)
    # -----------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.to(torch.float32)                     # apply output gate
    out = F.linear(out, weights["to_out.weight"])              # final projection

    return out
