Kernel code details:

@triton.jit
def batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak, stride_a_batch,
    stride_bk, stride_bn, stride_b_batch,
    stride_cm, stride_cn, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    # Compute sizes
    n_blocks_m = tl.cdiv(M, BLOCK_M)
    # batch index and block row:
    batch_id = pid0 // n_blocks_m
    block_m = pid0 % n_blocks_m

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid1 * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute offsets for A (A is left)
    # A shape (batch, M, K)
    # B shape (batch, K, N)
    # C shape (batch, M, N)
    # create accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)
        # A block: rows offs_m, cols k_range
        a_ptrs = A + batch_id * stride_a_batch + offs_m[:, None] * stride_am + k_range[None, :] * stride_ak
        # B block: rows k_range, cols offs_n
        b_ptrs = B + batch_id * stride_b_batch + k_range[:, None] * stride_bk + offs_n[None, :] * stride_bn
        a = tl.load(a_ptrs, mask = (offs_m < M) & (k_range < K), other = 0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask = (k_range < K) & (offs_n < N), other = 0.0).to(tl.float32)
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = C + batch_id * stride_c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=(offs_m < M) & (offs_n < N))

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy N³ reduction is rewritten as a batched matrix multiplication:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
For each hidden dimension h this is a GEMM (N×N)·(N×N)ᵀ.
We fuse the batched GEMM into a custom Triton kernel (fp16 inputs, fp32
accumulation) and subsequently apply the remaining LayerNorms, gates and
output projection in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A,                     # left   [B*H, M, K]  fp16
    B,                     # rightᵀ [B*H, K, N]  fp16
    C,                     # out    [B*H, M, N]  fp16
    M, N, K,
    stride_am, stride_ak, stride_a_batch,
    stride_bk, stride_bn, stride_b_batch,
    stride_cm, stride_cn, stride_c_batch,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B  (each batch independent)"""
    pid0 = tl.program_id(0)          # encodes batch_id * n_blocks_m + block_m
    pid1 = tl.program_id(1)          # block column index

    n_blocks_m = tl.cdiv(M, BLOCK_M)

    batch_id = pid0 // n_blocks_m
    block_m = pid0 % n_blocks_m

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid1 * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in BLOCK_K sized tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)

        # pointers for the current tile of A and B
        a_ptrs = A + batch_id * stride_a_batch \
                   + offs_m[:, None] * stride_am \
                   + k_range[None, :] * stride_ak
        b_ptrs = B + batch_id * stride_b_batch \
                   + k_range[:, None] * stride_bk \
                   + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs,
                    mask=(offs_m < M) & (k_range < K),
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=(k_range < K) & (offs_n < N),
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N)

    # write back the result as fp16
    c_ptrs = C + batch_id * stride_c_batch \
                + offs_m[:, None] * stride_cm \
                + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(offs_m < M) & (offs_n < N))


@triton.jit
def cast_fp16_to_fp32_kernel(
    in_ptr, out_ptr,
    N_ELEM: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """Simple cast kernel used to convert the fp16 result back to fp32."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEM
    val = tl.load(in_ptr + offs, mask=mask).to(tl.float32)
    tl.store(out_ptr + offs, val, mask=mask)


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model weights (see skeleton)
            - config: dict with at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ----------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ----------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ----------------------------------------------------
    # 2) Cast to fp16 for the heavy part
    # ----------------------------------------------------
    x_h = x.to(torch.float16)

    # ----------------------------------------------------
    # 3) Fuse projection + gate linear layers (fp16)
    # ----------------------------------------------------
    # left_proj + left_gate
    left_combined_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, device=device)
    left_combined = F.linear(x_h, left_combined_weight)   # [B, N, N, 2*H]

    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate_pre)

    # right_proj + right_gate
    right_combined_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, device=device)
    right_combined = F.linear(x_h, right_combined_weight)

    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate_pre)

    # ----------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # ----------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16, device=device).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ----------------------------------------------------
    # 5) Gating (fp16)
    # ----------------------------------------------------
    left = left_proj * left_gate          # [B, N, N, H] fp16
    right = right_proj * right_gate       # [B, N, N, H] fp16

    # ----------------------------------------------------
    # 6) Output gate (fp16)
    # ----------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(torch.float16, device=device)
    out_gate_pre = F.linear(x_h, out_gate_weight)   # [B, N, N, H] fp16
    out_gate = torch.sigmoid(out_gate_pre)          # fp16

    # ----------------------------------------------------
    # 7) Batched GEMM using Triton (left @ rightᵀ)
    # ----------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_T = right.permute(0, 3, 2, 1).contiguous().view(-1, N, N)

    batch_hidden = B * hidden_dim
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    n_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    n_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    out_fp16 = torch.empty_like(left_flat, dtype=torch.float16)

    # launch kernel
    grid = (batch_hidden * n_blocks_m, n_blocks_n)
    batched_matmul_kernel[grid](
        left_flat,
        right_T,
        out_fp16,
        M=N,
        N=N,
        K=N,
        stride_am=left_flat.stride(1),
        stride_ak=left_flat.stride(2),
        stride_a_batch=left_flat.stride(0),
        stride_bk=right_T.stride(1),
        stride_bn=right_T.stride(2),
        stride_b_batch=right_T.stride(0),
        stride_cm=out_fp16.stride(1),
        stride_cn=out_fp16.stride(2),
        stride_c_batch=out_fp16.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ----------------------------------------------------
    # 8) Reshape back to [B, N, N, hidden_dim] and cast to fp32
    # ----------------------------------------------------
    out_fp16 = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # Cast to fp32 (tiny Triton kernel)
    total_elem = out_fp16.numel()
    out_fp32 = torch.empty_like(out_fp16, dtype=torch.float32)
    BLOCK = 65536
    grid_cast = ((total_elem + BLOCK - 1) // BLOCK,)
    cast_fp16_to_fp32_kernel[grid_cast](
        out_fp16,
        out_fp32,
        N_ELEM=total_elem,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # ----------------------------------------------------
    # 9) Final LayerNorm, gating and projection back to original dim
    # ----------------------------------------------------
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out_norm * out_gate.to(torch.float32)          # apply output gate (fp32)
    out = F.linear(out, weights["to_out.weight"])        # final projection, fp32

    return out
