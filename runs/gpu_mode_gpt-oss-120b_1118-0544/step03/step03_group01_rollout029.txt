def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # Determine dtype
    orig_dtype = input_tensor.dtype

    # 1. LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for heavy compute
    x_fp16 = x.to(torch.float16)

    # Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Compute projections
    left = torch.nn.functional.linear(x_fp16, left_proj_w)  # shape (B,N,N,H)
    right = torch.nn.functional.linear(x_fp16, right_proj_w)

    # Apply mask if provided and not nomask
    if mask is not None and not nomask:
        # ensure mask dtype matches left/right
        mask_exp = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # Compute gates
    left_gate = torch.nn.functional.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()

    # Apply left/right gates
    left = left * left_gate
    right = right * right_gate

    # Reshape for batched GEMM
    B, N, _, H = left.shape
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Collapse batch and hidden
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Compute batched matrix multiplication using triton
    out_flat = _batched_matmul_triton(left_flat, right_flat)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Apply to_out_norm (layernorm over hidden_dim axis)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # Apply out_gate (shape (B,N,N,H))
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_w)  # shape (B,N,N,dim)
    # Cast back to original dtype if needed
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
```

@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)  # batch index (B*H)
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col

    # Compute actual offsets
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    
    # Masks for bounds
    mask_m = off_m < M
    mask_n = off_n < N

    # pointers for the start of the batch
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # Pointers to A and B for this tile (K dimension iterates)
    # A: shape (M, K)
    a_ptrs = A_ptr + a_batch + off_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    # B: shape (N, K) but we need B_T: shape (K, N) indexing as (k, n)
    b_ptrs = B_ptr + b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_bk + off_n[None, :] * stride_bn

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        # compute masks for K
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

        # Advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + off_m[:, None] * stride_cm + off_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) kernel with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Project to a hidden dimension `H` with two linear layers (`left_proj`,
   `right_proj`).  Apply an optional binary mask.
3. Compute three gating tensors (`left_gate`, `right_gate`, `out_gate`) from the
   normalized input and apply sigmoid.
4. Multiply the projected tensors by their respective gates.
5. Fold the hidden dimension into the batch dimension and perform the core O(N³·H)
   reduction with a custom Triton batched GEMM:
       C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]   (i.e. A @ Bᵀ)
   where A and B are the gated left/right tensors.
6. Unfold the hidden dimension, apply a second layer‑norm over `H`,
   multiply by `out_gate`, and project back to the original channel dimension.
7. Return the result in the original dtype of the input tensor.

Only the heavy matrix‑multiplication (step 5) is performed in Triton; all
other operations use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix dimensions
    stride_am, stride_ak, stride_ah,    # A: row stride, col stride, batch stride
    stride_bk, stride_bn, stride_bh,    # B: row (K) stride, col (N) stride, batch stride
    stride_cm, stride_cn, stride_ch,    # C: row stride, col stride, batch stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel: C = A @ Bᵀ  (A,B shape (batch,M,K) and (batch,N,K))."""
    pid_b = tl.program_id(2)   # batch index (B * H)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch offsets
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # pointers for the current tile (K dimension will be iterated)
    a_ptrs = A_ptr + a_batch + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)               # (BLOCK_M,BLOCK_K) × (BLOCK_K,BLOCK_N) → (BLOCK_M,BLOCK_N)

        a_ptrs += BLOCK_K * stride_ak      # advance A pointers
        b_ptrs += BLOCK_K * stride_bk      # advance B pointers

    # write back result
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Wrapper for the Triton batched GEMM kernel.
    A, B: FP16 tensors of shape (batch, N, N)
    Returns C of the same shape and dtype.
    """
    assert A.shape == B.shape, "A and B must have the same shape"
    batch, N, K = A.shape
    assert N == K, "TriMul requires square matrices"

    # Choose block sizes heuristically based on N
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
    elif N <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 64
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:  # up to 1024 in our tests
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along M
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along N
        batch,                          # batch dimension (B * H)
    )

    C = torch.empty_like(A)

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        N,
        N,
        N,
        A.stride(1),  # stride_am (row)
        A.stride(2),  # stride_ak (col)
        A.stride(0),  # stride_ah (batch)
        B.stride(1),  # stride_bk (row / K)
        B.stride(2),  # stride_bn (col / N)
        B.stride(0),  # stride_bh (batch)
        C.stride(1),  # stride_cm (row)
        C.stride(2),  # stride_cn (col)
        C.stride(0),  # stride_ch (batch)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict containing the model parameters
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    orig_dtype = input_tensor.dtype
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for the heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Optional mask (binary) – broadcast over hidden dim
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid activations)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid()  # (B, N, N, H)
    out_gate   = F.linear(x, out_gate_w).sigmoid()    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Apply gates to projected tensors
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Prepare for Triton GEMM: fold hidden dim into batch
    # ------------------------------------------------------------------
    B, N, _, H = left.shape        # B, seq_len, seq_len, hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right.view(B * H, N, N) # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8) Core O(N³·H) reduction using Triton (batched GEMM)
    # ------------------------------------------------------------------
    out_flat = _batched_matmul_triton(left_flat, right_flat)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 9) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 10) Output LayerNorm over hidden dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 11) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 13) Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
```
