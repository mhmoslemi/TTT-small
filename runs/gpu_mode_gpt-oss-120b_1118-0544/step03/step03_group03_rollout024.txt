@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_abl, stride_aik, stride_ak,
    stride_bbl, stride_bnk, stride_bk,
    stride_cbl, stride_cmi, stride_cnj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr=4,
):
    pid = tl.program_id(0)
    batch = BATCH

    # number of blocks per batch
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_pid_m * num_pid_n

    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    pid_m = block_idx // num_pid_n
    pid_n = block_idx % num_pid_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # load a pointers
    a_ptrs = a_ptr + batch_idx * stride_abl + offs_m[:, None] * stride_aik + offs_k[None, :] * stride_ak
    # for b (which is transposed B), shape (batch, N, K)
    b_ptrs = b_ptr + batch_idx * stride_bbl + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bnk

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # loop over K dim
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)  # shape (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)  # shape (BLOCK_K, BLOCK_N)
        # compute dot product and accumulate
        # dot expects (M, K) and (K, N)
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Combine projections and gates for left and right to reduce kernel launches?
    left_proj = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].to(torch.float16))
    right_proj = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].to(torch.float16))

    right = right_proj * right_gate

    B, N, _, _ = left.shape  # B,N,N,H

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr = 4,
):
    pid = tl.program_id(0)
    # Compute block indices
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_pid_m * num_pid_n

    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    pid_m = block_idx // num_pid_n
    pid_n = block_idx % num_pid_n

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute masks for out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers for A and B
    a_ptrs = a_ptr + batch_idx * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + batch_idx * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Write back
    c_ptrs = c_ptr + batch_idx * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask)

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The core of TriMul is the O(N³·H) reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which can be expressed as a batch of H independent matrix multiplications:
    out_{b,h} = left_{b,h} @ right_{b,h}ᵀ.
We fuse this reduction in a custom Triton kernel that performs a batched
matmul in fp16 using Tensor Cores, then finish the remaining linear
layers, gating and LayerNorms in PyTorch (fp32).  This yields a large
speed‑up over a pure‑PyTorch implementation while keeping the final
output in float32.
"""

import torch.nn.functional as F

def batched_matmul_kernel(
    a_ptr,                     # fp16 left  tensor   (batch, M, K)
    b_ptr,                     # fp16 right tensor   (batch, N, K)  (already transposed)
    c_ptr,                     # fp16 output tensor  (batch, M, N)
    BATCH, M, N, K,            # problem sizes
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr = 4,
):
    pid = tl.program_id(0)

    # ---------------------------------------------------------
    # 1) Decompose program ID into batch, block_m, block_n
    # ---------------------------------------------------------
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_pid_m * num_pid_n

    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    pid_m = block_idx // num_pid_n
    pid_n = block_idx % num_pid_n

    # ---------------------------------------------------------
    # 2) Compute offsets for this tile
    # ---------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    offs_k = tl.arange(0, BLOCK_K)                    # [BLOCK_K]

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers for the current tile of A and B
    a_ptrs = (
        a_ptr
        + batch_idx * stride_a_batch
        + offs_m[:, None] * stride_a_m
        + offs_k[None, :] * stride_a_k
    )
    b_ptrs = (
        b_ptr
        + batch_idx * stride_b_batch
        + offs_k[:, None] * stride_b_k
        + offs_n[None, :] * stride_b_n
    )

    # ---------------------------------------------------------
    # 3) Accumulate over the K dimension
    # ---------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # ---------------------------------------------------------
    # 4) Write result
    # ---------------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch_idx * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_store)


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Arguments:
        data = (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict with all Linear/LayerNorm parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ---------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ---------------------------------------------------------
    # 2) Cast to fp16 for the heavy work
    # ---------------------------------------------------------
    x_h = x.to(torch.float16)

    # ---------------------------------------------------------
    # 3) Linear projections + gates (all fp16)
    # ---------------------------------------------------------
    left_proj = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].to(torch.float16))
    right_proj = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].to(torch.float16))
    out_gate_pre = F.linear(x_h, weights["out_gate.weight"].to(torch.float16))

    # ---------------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # ---------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)          # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ---------------------------------------------------------
    # 5) Gating (sigmoid, fp16)
    # ---------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate          # [B, N, N, H] fp16
    right = right_proj * right_gate       # [B, N, N, H] fp16

    out_gate = torch.sigmoid(out_gate_pre)  # [B, N, N, H] fp16

    # ---------------------------------------------------------
    # 6) Prepare tensors for the batched matmul kernel
    #    left  : (B, H, N, N)   rows = i, cols = k
    #    right : (B, H, N, N)   rows = j, cols = k  -> we need its transpose (k, j)
    # ---------------------------------------------------------
    B, N, _, _ = left.shape

    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)  (transposed)

    left_flat = left_perm.view(-1, N, N)    # (B*H, N, N)
    right_flat = right_perm.view(-1, N, N)  # (B*H, N, N)

    batch = left_flat.shape[0]  # B * H
    out_flat = torch.empty_like(left_flat)  # fp16 output

    # ---------------------------------------------------------
    # 7) Launch Triton kernel for the batched matmul
    # ---------------------------------------------------------
    a_ptr = left_flat
    b_ptr = right_flat
    c_ptr = out_flat

    stride_a_batch, stride_a_m, stride_a_k = a_ptr.stride()
    stride_b_batch, stride_b_n, stride_b_k = b_ptr.stride()
    stride_c_batch, stride_c_m, stride_c_n = c_ptr.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch
        * ((N + BLOCK_M - 1) // BLOCK_M)
        * ((N + BLOCK_N - 1) // BLOCK_N),
    )

    batched_matmul_kernel[grid](
        a_ptr,
        b_ptr,
        c_ptr,
        batch,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ---------------------------------------------------------
    # 8) Reshape back to [B, N, N, H] (still fp16)
    # ---------------------------------------------------------
    out_h = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H] fp16

    # ---------------------------------------------------------
    # 9) Final LayerNorm, gating and projection (float32)
    # ---------------------------------------------------------
    out_fp32 = out_h.to(torch.float32)

    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.to(torch.float32)            # apply output gate
    out = F.linear(out, weights["to_out.weight"])    # final projection to dim

    return out
