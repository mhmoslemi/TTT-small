### Triton kernel design:

Define:

```
@triton.jit
def batched_gemm_fp16_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_Ab, stride_Ai, stride_Ak,
    stride_Bb, stride_Bk, stride_Bj,
    stride_Cb, stride_Ci, stride_Cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Pointers for this batch
    a_batch_ptr = A_ptr + pid_batch * stride_Ab
    b_batch_ptr = B_ptr + pid_batch * stride_Bb
    c_batch_ptr = C_ptr + pid_batch * stride_Cb

    # iterate over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    left_proj_weight = weights["left_proj.weight"].to(torch.float16)

    right_gate_weight = weights["right_gate.weight"].to(torch.float16)

@triton.jit
def batched_gemm_fp16_kernel(
    A_ptr, B_ptr, C_ptr,
    total_batch, hidden_dim,
    M, N, K,
    stride_Ab, stride_Ai, stride_Ak, stride_Ah,
    stride_Bb, stride_Bk, stride_Bj, stride_Bh,
    stride_Cb, stride_Ci, stride_Cj, stride_Ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch and hidden idx
    b = pid // hidden_dim
    h = pid % hidden_dim

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this (b, h)
    a_ptr = A_ptr + b * stride_Ab + h * stride_Ah
    b_ptr = B_ptr + b * stride_Bb + h * stride_Bh
    c_ptr = C_ptr + b * stride_Cb + h * stride_Ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a_tile_ptrs = a_ptr + offs_m[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        a = tl.load(a_tile_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # Load B tile (B is transposed: rows=k, cols=j)
        b_tile_ptrs = b_ptr + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bj
        b = tl.load(b_tile_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        # dot product
        acc += tl.dot(a, b)

    # Store result
    c_tile_ptrs = c_ptr + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cj
    tl.store(c_tile_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
Outgoing Triangle Multiplicative Update (TriMul) – fast Triton implementation.

Algorithm:
1. Layer‑norm the input pair representation (float32).
2. Cast to fp16 and compute left / right projections and their gates in a
   fused fashion (two linear layers per side, concatenated weights).
3. Apply optional mask and element‑wise sigmoid gates.
4. Compute the core N³ reduction:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (left @ rightᵀ) for each hidden channel.
   A custom Triton kernel performs the GEMM directly on the
   [B, N, N, H] layout, avoiding the costly permute+contiguous copies
   required by the pure‑PyTorch version.
5. Cast the result back to float32, apply the output LayerNorm,
   the output gate, and the final linear projection to the original
   feature dimension.
The kernel works for arbitrary batch size (1‑2), sequence length
(≤1024) and hidden dimension (≤384) and respects optional masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_fp16_kernel(
    A_ptr, B_ptr, C_ptr,                      # data pointers (fp16)
    total_batch, hidden_dim,                  # total batches = B*H, hidden dimension H
    M, N, K,                                  # matrix sizes (all = seq_len)
    stride_Ab, stride_Ai, stride_Ak, stride_Ah,  # strides for A (left)
    stride_Bb, stride_Bk, stride_Bj, stride_Bh,  # strides for B (right, transposed)
    stride_Cb, stride_Ci, stride_Cj, stride_Ch,  # strides for C (output)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_batch = tl.program_id(0)   # batch * hidden
    pid_m = tl.program_id(1)       # block row (i)
    pid_n = tl.program_id(2)       # block col (j)

    # decode batch index and hidden index
    b_idx = pid_batch // hidden_dim
    h_idx = pid_batch % hidden_dim

    # offsets within the M/N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this (b, h)
    a_base = A_ptr + b_idx * stride_Ab + h_idx * stride_Ah
    b_base = B_ptr + b_idx * stride_Bb + h_idx * stride_Bh
    c_base = C_ptr + b_idx * stride_Cb + h_idx * stride_Ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # tiled K‑loop
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # B tile (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        # fused multiply‑accumulate
        acc += tl.dot(a, b)

    # store result (fp16)
    c_ptrs = c_base + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing triangle multiplicative update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model weights
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] fp32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections + gates (fused via concatenated weights)
    # ------------------------------------------------------------------
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, device=device)
    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, device=device)

    left_combined = F.linear(x_h, left_comb_weight)   # [B, N, N, 2*H]
    right_combined = F.linear(x_h, right_comb_weight)

    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 5) Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate      # [B, N, N, H] fp16
    right = right_proj * right_gate   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 6) Output gate (fp16)
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(torch.float16, device=device)
    out_gate_pre = F.linear(x_h, out_gate_weight)   # [B, N, N, H] fp16
    out_gate = torch.sigmoid(out_gate_pre)          # fp16

    # ------------------------------------------------------------------
    # 7) Core N³ reduction via Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # fp16, same layout [B, N, N, H]

    # strides for the tensors (contiguous layout)
    stride_A_batch = left.stride(0)
    stride_A_i = left.stride(1)
    stride_A_k = left.stride(2)
    stride_A_h = left.stride(3)

    stride_B_batch = right.stride(0)
    stride_B_k = right.stride(2)   # row = k
    stride_B_j = right.stride(1)   # col = j
    stride_B_h = right.stride(3)

    stride_C_batch = out.stride(0)
    stride_C_i = out.stride(1)
    stride_C_j = out.stride(2)
    stride_C_h = out.stride(3)

    total_batch = B * H
    M = N
    N_ = N
    K = N

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N_ + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_fp16_kernel[grid](
        A_ptr=left,
        B_ptr=right,
        C_ptr=out,
        total_batch=total_batch,
        hidden_dim=H,
        M=M,
        N=N_,
        K=K,
        stride_Ab=stride_A_batch,
        stride_Ai=stride_A_i,
        stride_Ak=stride_A_k,
        stride_Ah=stride_A_h,
        stride_Bb=stride_B_batch,
        stride_Bk=stride_B_k,
        stride_Bj=stride_B_j,
        stride_Bh=stride_B_h,
        stride_Cb=stride_C_batch,
        stride_Ci=stride_C_i,
        stride_Cj=stride_C_j,
        stride_Ch=stride_C_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 8) Cast back to fp32 for the post‑processing layers
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 9) Output LayerNorm + output gate + final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate.to(torch.float32)   # apply output gate
    out_final = F.linear(out_norm, weights["to_out.weight"])  # back to dim, fp32

    return out_final
