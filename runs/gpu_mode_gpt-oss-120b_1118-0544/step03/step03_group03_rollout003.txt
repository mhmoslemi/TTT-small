@triton.jit
def batched_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute block indices
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch_idx * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0).to(tl.float32)

        # Load B_T tile: shape (BLOCK_K, BLOCK_N). We load original B but with swapped strides.
        b_ptrs = b_ptr + batch_idx * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0).to(tl.float32)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write back to C
    c_ptrs = c_ptr + batch_idx * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy N³ reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a batched matrix multiplication C = A @ Bᵀ for each
(b, h) pair using a custom Triton kernel. All linear projections,
gates and the final LayerNorm / projection are kept in torch for
convenience, while the core GEMM runs entirely in fp16 on Tensor Cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_transpose_kernel(
    a_ptr,          # left  [B*H, M, K]   fp16
    b_ptr,          # right [B*H, N, K]   fp16 (will be read transposed)
    c_ptr,          # out   [B*H, M, N]   fp16
    M, N, K,        # matrix dimensions (int)
    stride_a_batch, stride_a_m, stride_a_k,   # strides for A
    stride_b_batch, stride_b_k, stride_b_n,   # strides for Bᵀ (k,n swapped)
    stride_c_batch, stride_c_m, stride_c_n,   # strides for C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # Number of tiles per matrix
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # Decode the 3‑D launch grid: (batch * tile_m * tile_n)
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    tile_idx = pid % (num_blocks_m * num_blocks_n)
    pid_m = tile_idx // num_blocks_n
    pid_n = tile_idx % num_blocks_n

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile: [BLOCK_M, BLOCK_K]
        a = tl.load(
            a_ptr
            + batch_idx * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        ).to(tl.float32)

        # Load Bᵀ tile: [BLOCK_K, BLOCK_N]
        # (original B is read with swapped strides to achieve the transpose)
        b = tl.load(
            b_ptr
            + batch_idx * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        ).to(tl.float32)

        # Dot product
        acc += tl.dot(a, b)

    # Store result
    c = acc.to(tl.float16)
    c_ptrs = (
        c_ptr
        + batch_idx * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, c, mask=mask)


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model weights
            - config: dict with at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5
    device = input_tensor.device

    # ----------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ----------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] fp32

    # ----------------------------------------------------
    # 2) Cast to fp16 for the heavy part
    # ----------------------------------------------------
    x_h = x.to(torch.float16)

    # ----------------------------------------------------
    # 3) Linear projections and gates (all in fp16)
    # ----------------------------------------------------
    left_proj = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))
    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].to(torch.float16))
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].to(torch.float16))

    # Optional mask (broadcast on hidden_dim)
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Gating
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    left = left_proj * left_gate   # [B, N, N, H] fp16
    right = right_proj * right_gate

    # Output gate (computed from the original normalized input)
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(torch.float16))
    )  # [B, N, N, H] fp16

    # ----------------------------------------------------
    # 4) Core N³ reduction via Triton batched GEMM (A @ Bᵀ)
    # ----------------------------------------------------
    B, N, _, _ = left.shape  # N is sequence length
    # Re‑order to (B, H, N, N) and flatten the first two dims
    left_view = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    out_view = torch.empty_like(left_view, dtype=torch.float16)

    # Strides for the packed tensors
    stride_a_batch, stride_a_m, stride_a_k = left_view.stride()   # (N*N, N, 1)
    stride_b_batch, stride_b_row, stride_b_col = right_view.stride()  # (N*N, N, 1)
    # For Bᵀ we swap the row/col strides
    stride_b_k = stride_b_col   # 1
    stride_b_n = stride_b_row   # N
    stride_c_batch, stride_c_m, stride_c_n = out_view.stride()   # (N*N, N, 1)

    # Tile sizes (tuned for B200 Tensor Cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_batch = B * hidden_dim
    grid = (total_batch * num_blocks_m * num_blocks_n,)

    batched_matmul_transpose_kernel[grid](
        a_ptr=left_view,
        b_ptr=right_view,
        c_ptr=out_view,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out_fp16 = (
        out_view.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H] fp16

    # ----------------------------------------------------
    # 5) Final LayerNorm, output gating and projection (float32)
    # ----------------------------------------------------
    out_fp32 = out_fp16.to(torch.float32)

    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply the output gate (convert to fp32)
    out = out * out_gate.to(torch.float32)
    # Final linear projection back to the original dimensionality
    out = F.linear(out, weights["to_out.weight"])

    return out
