def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
        eps=1e-5,
    )
    # Cast to fp16
    x_fp16 = x.to(torch.float16)

    # 2. Compute proj and gate in fused way
    # Combine left_proj, left_gate, right_proj, right_gate
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)

    combined_weight = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0)  # (4*hidden, dim)

    # Flatten input and linear
    B, N, _, _ = x_fp16.shape
    x_flat = x_fp16.view(-1, dim)  # (B*N*N, dim)
    combined_out = torch.nn.functional.linear(x_flat, combined_weight)  # (B*N*N, 4*hidden)
    # reshape
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)
    left_proj, left_gate, right_proj, right_gate = torch.split(combined_out, hidden_dim, dim=-1)

    # 3. Mask
    if not nomask:
        # mask may be bool or float. Convert to float16.
        mask_fp = mask_tensor.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # 4. Gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate  # (B,N,N,hidden)
    right = right_proj * right_gate

    # 5. Batched GEMM
    # permute b,d,i,k etc.
    # left: (B,N,N,hidden) -> (B*hidden,N,N) after permute (0,3,1,2)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose k and j
    left_batch = left_perm.view(B * hidden_dim, N, N)
    right_batch = right_perm.view(B * hidden_dim, N, N)
    out_batch = torch.bmm(left_batch, right_batch)  # (B*hidden, N, N)
    # reshape back
    out = out_batch.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # 6. to_out_norm
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )

    # 7. out_gate linear and sigmoid then elementwise multiply with out

"""
TriMul forward (outgoing) – fused implementation.

The module follows the reference PyTorch code but aggressively fuses the four
linear projections (left/right + their gates) into a single matmul, keeps the
intermediate tensors in fp16 to exploit tensor‑core performance and uses a
tiny Triton kernel for the final element‑wise multiplication with the output
gate. The N³ reduction is carried out as a batched half‑precision GEMM via
torch.bmm, which is highly optimized on modern NVIDIA GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N, BLOCK: tl.constexpr):
    """c = a * b  (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config["nomask"] is True
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ----------------------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32 → fp16)
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  float32
    x_fp16 = x.to(torch.float16)  # fp16 for the rest of the pipeline

    B, N, _, _ = x_fp16.shape

    # ----------------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (four linear layers in one)
    # ----------------------------------------------------------------------
    # Weights are stored as (out_features, in_features)
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)

    # Concatenate to obtain a single weight matrix of shape (4*hidden, dim)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # (4*hidden, dim)

    # Linear over the flattened spatial dimensions
    x_flat = x_fp16.view(-1, dim)                     # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 4*hidden)
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each: (B, N, N, hidden)

    # ----------------------------------------------------------------------
    # 3️⃣ Optional mask (applied only to projections)
    # ----------------------------------------------------------------------
    if not nomask:
        mask_fp = mask_tensor.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # ----------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and combine with projections
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate   # (B, N, N, hidden)  fp16
    right = right_proj * right_gate

    # ----------------------------------------------------------------------
    # 5️⃣ Batched GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched half‑precision matmul.
    # ----------------------------------------------------------------------
    # Permute to (B, hidden, N, N) and reshape batch dimension.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, N, N) – transpose k/j

    left_batch = left_perm.view(B * hidden_dim, N, N)   # (B*hidden, N, N)
    right_batch = right_perm.view(B * hidden_dim, N, N)

    out_batch = torch.bmm(left_batch, right_batch)     # (B*hidden, N, N)

    out = (
        out_batch.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, hidden)  fp16

    # ----------------------------------------------------------------------
    # 6️⃣ to_out_norm (LayerNorm over hidden dimension)
    # ----------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # (B, N, N, hidden)  fp16

    # ----------------------------------------------------------------------
    # 7️⃣ Output gate: linear + sigmoid, then element‑wise multiplication
    # ----------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))  # (B, N, N, hidden) fp16

    # Triton element‑wise multiplication (in‑place)
    total_el = out.numel()
    BLOCK = 8192  # tuneable; must be a compile‑time constant
    grid = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        out,
        out_gate,
        out,
        total_el,
        BLOCK=BLOCK,
    )

    # ----------------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dimension
    # ----------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, dim) fp16

    # Return in float32 as required by the interface
    return out.to(torch.float32)
