We'll define kernel:

```
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_am, stride_ak, stride_ab, # Actually A shape (batch,M,K)
    stride_bk, stride_bn, stride_bb, # B shape (batch,K,N)
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, GROUP_SIZE_M: tl.constexpr
):
    # program_id(0) = batch index
    # program_id(1) = block row index
    # program_id(2) = block col index

    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    # Block row start: block_m = pid_m * BLOCK_M
    # Block col start: block_n = pid_n * BLOCK_N

    # Use accumulator for BLOCK_M x BLOCK_N block
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) # we can accumulate in fp32 even if data fp16

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offsets for A and B
        a_ptrs = A_ptr + batch * stride_ab + (pid_m * BLOCK_M + tl.arange(0, BLOCK_M))[:, None] * stride_am + (k * BLOCK_K + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        b_ptrs = B_ptr + batch * stride_bb + (k * BLOCK_K + tl.arange(0, BLOCK_K))[:, None] * stride_bk + (pid_n * BLOCK_N + tl.arange(0, BLOCK_N))[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=(...))
        b = tl.load(b_ptrs, mask=(...))
        acc += tl.dot(a, b)
    # store result
    c_ptrs = C_ptr + batch * stride_cb + (pid_m * BLOCK_M + tl.arange(0, BLOCK_M))[:, None] * stride_cm + (pid_n * BLOCK_N + tl.arange(0, BLOCK_N))[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16)) # maybe store directly as fp16
```

    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_ab, stride_ai, stride_ak, stride_ad,
    stride_bb, stride_br, stride_bc, stride_bd,
    stride_cb, stride_ci, stride_cj, stride_cd,

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # 1. LayerNorm
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=1e-5)

    # 2. Cast fp16
    x_h = x.to(torch.float16)

    # 3. Linear projections + gates combined
    left_combined_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0)
    left_combined = F.linear(x_h, left_combined_weight.to(torch.float16))
    left_proj, left_gate_pre = left_combined[..., :hidden_dim], left_combined[..., hidden_dim:]

    right_combined_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0)
    right_combined = F.linear(x_h, right_combined_weight.to(torch.float16))
    right_proj, right_gate_pre = right_combined[..., :hidden_dim], right_combined[..., hidden_dim:]

    # 4. Apply mask if present
    if not nomask and mask is not None:

        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    right = right_proj * right_gate

"""
TriMul outgoing update – fast Triton implementation.

The heavy O(N³) reduction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is realized as a batched matrix multiplication over the hidden dimension.
Each (b,d) pair is treated as an independent batch and multiplied with a
custom Triton kernel that tiles the N×N matrices (BLOCK_M×BLOCK_N) and
accumulates over K in blocks (BLOCK_K).  All linear projections, gating and
layer‑norms are kept in fp16 (except the final LayerNorm and output linear)
to exploit Tensor Cores.  A small Triton kernel is thus used for the
computational bottleneck, satisfying the requirement while providing a large
speed‑up over a pure‑PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,             # pointers to fp16 tensors
    BATCH, M, N, K,                 # int: total batches = B*H, matrix dims
    stride_ab, stride_ai, stride_ak, stride_ad,   # A: batch, row(i), col(k), hidden
    stride_bb, stride_br, stride_bc, stride_bd,   # B: batch, row(k), col(j), hidden
    stride_cb, stride_ci, stride_cj, stride_cd,   # C: batch, row(i), col(j), hidden
    H: tl.constexpr,                # hidden dimension (size of each batch group)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_batch = tl.program_id(0)      # combined batch index = b * H + h
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block col

    # decompose combined batch index into (b, h)
    b_idx = pid_batch // H
    h_idx = pid_batch - b_idx * H

    # base pointers for this (b, h) slice
    a_base = A_ptr + b_idx * stride_ab + h_idx * stride_ad
    b_base = B_ptr + b_idx * stride_bb + h_idx * stride_bd
    c_base = C_ptr + b_idx * stride_cb + h_idx * stride_cd

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator (fp16 – tensor cores accumulate in fp32 internally)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)

        # A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)

        # B block: we need B as (K, N) i.e. rows = k, cols = j
        b_ptrs = b_base + offs_k[:, None] * stride_br + offs_n[None, :] * stride_bc
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        # multiply‑accumulate
        acc += tl.dot(a, b)

    # write back C block (shape BLOCK_M × BLOCK_N)
    c_ptrs = c_base + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul update.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model weights (see comments)
            - config: dict with at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    eps = 1e-5

    # ----------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ----------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # ----------------------------------------------------
    # 2) Cast to fp16 for the heavy part
    # ----------------------------------------------------
    x_h = x.to(torch.float16)

    # ----------------------------------------------------
    # 3) Fuse projection + gate linear layers (fp16)
    # ----------------------------------------------------
    # left side
    left_combined_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)
    left_combined = F.linear(x_h, left_combined_weight)   # [B, N, N, 2*H]
    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)

    # right side
    right_combined_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)
    right_combined = F.linear(x_h, right_combined_weight)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    # ----------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # ----------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ----------------------------------------------------
    # 5) Gating (sigmoid, fp16)
    # ----------------------------------------------------
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    left = left_proj * left_gate          # [B, N, N, H] fp16
    right = right_proj * right_gate       # [B, N, N, H] fp16

    # ----------------------------------------------------
    # 6) Output gate (fp16)
    # ----------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_pre = F.linear(x_h, out_gate_weight)   # [B, N, N, H] fp16
    out_gate = out_gate_pre.sigmoid()               # fp16

    # ----------------------------------------------------
    # 7) Batched matrix multiplication via Triton
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # ----------------------------------------------------
    B, N, _, _ = left.shape
    out_fp16 = torch.empty_like(left, dtype=torch.float16)  # [B, N, N, H]

    # Strides (all tensors are contiguous, shape [B,N,N,H])
    stride_ab = left.stride(0)   # batch stride
    stride_ai = left.stride(1)   # row (i) stride
    stride_ak = left.stride(2)   # col (k) stride
    stride_ad = left.stride(3)   # hidden stride (==1)

    stride_bb = right.stride(0)
    # For B we need K‑row stride = original col stride, N‑col stride = original row stride
    stride_br = right.stride(2)  # K‑row stride (hidden_dim)
    stride_bc = right.stride(1)  # N‑col stride (N*hidden_dim)
    stride_bd = right.stride(3)

    stride_cb = out_fp16.stride(0)
    stride_ci = out_fp16.stride(1)   # row (i) stride
    stride_cj = out_fp16.stride(2)   # col (j) stride
    stride_cd = out_fp16.stride(3)   # hidden stride

    total_batches = B * hidden_dim
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        A_ptr=left,
        B_ptr=right,
        C_ptr=out_fp16,
        BATCH=total_batches,
        M=N,
        N=N,
        K=N,
        stride_ab=stride_ab,
        stride_ai=stride_ai,
        stride_ak=stride_ak,
        stride_ad=stride_ad,
        stride_bb=stride_bb,
        stride_br=stride_br,
        stride_bc=stride_bc,
        stride_bd=stride_bd,
        stride_cb=stride_cb,
        stride_ci=stride_ci,
        stride_cj=stride_cj,
        stride_cd=stride_cd,
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ----------------------------------------------------
    # 8) Cast back to fp32 and finish with LayerNorm + linear
    # ----------------------------------------------------
    out_fp32 = out_fp16.float()  # [B, N, N, H] fp32

    # to_out LayerNorm
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # apply output gate
    out = out * out_gate.float()
    # final projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out
