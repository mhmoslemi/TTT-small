@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab, # Actually stride batch (some)
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)  # each batch (B*H)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # valid masks for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # k offset
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_ptr + batch_idx * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_idx * stride_ab + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # Write back c
    c_ptrs = c_ptr + batch_idx * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The core operation is a batched matrix multiplication:
    out[b,h] = left[b,h] @ right[b,h].T
where left/right have shape (B, N, N, H).  This is implemented by a
custom Triton kernel that processes many small N×N matrices in parallel.
All other (LayerNorm, Linear, Gate, Mask) steps are performed with
PyTorch in half precision.  The final result is cast back to the input
dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A : (batch,M,K), B : (batch,K,N))"""
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for the current block of A and B
        a_ptrs = (
            a_ptr
            + batch_idx * stride_batch_a
            + offs_m[:, None] * stride_m_a
            + offs_k[None, :] * stride_k_a
        )
        b_ptrs = (
            b_ptr
            + batch_idx * stride_batch_b
            + offs_k[:, None] * stride_k_b
            + offs_n[None, :] * stride_n_b
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # FP16 dot, accumulation in FP32
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = (
        c_ptr
        + batch_idx * stride_batch_c
        + offs_m[:, None] * stride_m_c
        + offs_n[None, :] * stride_n_c
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of outgoing TriMul using a Triton batched GEMM.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of parameters (float32)
            - config: dict with keys:
                * "dim" (int)          – input channel dimension C
                * "hidden_dim" (int)   – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor [B, N, N, C] with the same dtype as `input_tensor`.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (FP32 for stability)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------ #
    # 2. Cast to half precision for the heavy path
    # ------------------------------------------------------------------ #
    x = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 3. Cast model weights to half precision
    # ------------------------------------------------------------------ #
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------ #
    # 4. Linear projections & optional masking
    # ------------------------------------------------------------------ #
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------ #
    # 5. Gate projections
    # ------------------------------------------------------------------ #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 6. Core O(N³·H) reduction via a Triton batched GEMM
    #    left:  (B, N, N, H) -> (B*H, N, N)
    #    right: (B, N, N, H) -> (B*H, N, N) and transposed
    # ------------------------------------------------------------------ #
    B, N, _, _ = left.shape
    # (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_flat = B * hidden_dim
    left_flat = left_perm.view(batch_flat, N, N)
    # right needs to be transposed for the matmul
    right_t = right_perm.view(batch_flat, N, N).transpose(-2, -1)

    out_flat = torch.empty_like(left_flat)  # (batch_flat, N, N), half precision

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_flat,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_t,
        out_flat,
        N,
        N,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_t.stride(0),
        right_t.stride(1),
        right_t.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 7. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------ #
    # 8. Apply output gate and final linear projection
    # ------------------------------------------------------------------ #
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, C)

    # Cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out
