@triton.jit
def matmul_batch_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n
    batch = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute pointer base for this batch
    a_batch_ptr = a_ptr + batch * stride_ab
    b_batch_ptr = b_ptr + batch * stride_bb
    c_batch_ptr = c_ptr + batch * stride_cb

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # offsets in k dimension
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # compute masks for k boundaries
        mask_k = offs_k < K

        # A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B block: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product accumulate
        acc += tl.dot(a, b)

    # convert to output dtype (float16)
    c = acc.to(tl.float16)

    # store
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – Triton‑accelerated implementation.

The core of the module is the 3‑D reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is a batched matrix multiplication of shape
    (B·H, N, N) × (B·H, N, N)ᵀ  →  (B·H, N, N).

Only this O(N³·H) reduction is written in Triton; all other
operations (LayerNorms, linear projections, sigmoids, masking,
final projection) are performed with PyTorch kernels in half‑precision.
The kernel works on contiguous half‑precision tensors, uses float‑32
accumulators, and is tiled with 64×64×64 blocks.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B shape: (batch, N, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dims (M=N, K=N)
    stride_am, stride_ak, stride_ab,   # A strides (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # B strides (batch, K, N)
    stride_cm, stride_cn, stride_cb,   # C strides (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # tiles per batch
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    # batch index and tile indices
    batch_idx = pid // blocks_per_batch
    tile_idx = pid % blocks_per_batch
    pid_m = tile_idx // num_blocks_n
    pid_n = tile_idx % num_blocks_n

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch_idx * stride_ab
    b_batch = b_ptr + batch_idx * stride_bb
    c_batch = c_ptr + batch_idx * stride_cb

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A block: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B block (already transposed): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)            # float32 accumulation

    # write result (float16)
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    A: (batch, M, K)  half
    B: (batch, K, N)  half   (here K == N == M)
    Returns C: (batch, M, N) half
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2 and M == N, "TriMul expects square matrices per batch."

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # strides in elements
    stride_am = A.stride(1)
    stride_ak = A.stride(2)
    stride_ab = A.stride(0)

    stride_bk = B.stride(1)
    stride_bn = B.stride(2)
    stride_bb = B.stride(0)

    stride_cm = C.stride(1)
    stride_cn = C.stride(2)
    stride_cb = C.stride(0)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4, num_stages=2
    )
    return C


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → keep precision)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast everything to half for the heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # cast model weights to half precision
    w = {}
    for name in [
        "left_proj.weight", "right_proj.weight",
        "left_gate.weight", "right_gate.weight", "out_gate.weight",
        "to_out_norm.weight", "to_out_norm.bias",
        "to_out.weight",
    ]:
        w[name] = weights[name].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections + optional masking
    # ------------------------------------------------------------------
    left = F.linear(x, w["left_proj.weight"])   # (B, N, N, H)
    right = F.linear(x, w["right_proj.weight"]) # (B, N, N, H)

    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gates
    # ------------------------------------------------------------------
    left_gate = F.linear(x, w["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, w["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, w["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·H) reduction via a fused Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # (B, N, N, H)

    # reshape to (B·H, N, N) for the batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Triton kernel computes: out_flat = left_flat @ right_flatᵀ
    out_flat = _batched_matmul_triton(left_flat, right_flat)

    # restore original layout (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 7. Apply the output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, w["to_out.weight"])

    # Cast back to the original dtype if necessary
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)
    return out
