@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute block indices
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    blocks_per_mat = num_pid_m * num_pid_n
    mat_id = pid // blocks_per_mat
    block_id = pid % blocks_per_mat
    pid_m = block_id // num_pid_n
    pid_n = block_id % num_pid_n

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Compute pointers with batch offset
        a_ptrs = a_ptr + mat_id * batch_stride_a + (offs_m[:, None] * stride_am + (k_offset + offs_k)[None, :] * stride_ak)
        b_ptrs = b_ptr + mat_id * batch_stride_b + ((k_offset + offs_k)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        # Load blocks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & ((k_offset + offs_k) < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=((k_offset + offs_k) < K)[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute dot product
        acc += tl.dot(a, b)
    # Cast acc to fp16
    c = acc.to(tl.float16)

    # Store
    c_ptrs = c_ptr + mat_id * batch_stride_c + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The heavy N³ reduction is performed with a custom Triton kernel that computes
batched matrix multiplications of shape (N, N) × (N, N)ᵀ for each hidden
dimension.  Inputs and linear projections are kept in fp16 to exploit Tensor
Cores, while the final LayerNorm and output projection run in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # matrix pointers
    M, N, K,                                 # dimensions (all = seq_len)
    stride_am, stride_ak,                    # A strides: (row, col)
    stride_bk, stride_bn,                    # B strides: (row, col) – B is transposed
    stride_cm, stride_cn,                    # C strides: (row, col)
    batch_stride_a, batch_stride_b, batch_stride_c,  # stride between matrices in the batch
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B   where
        A : (batch, M, K)
        B : (batch, K, N)   (already transposed from the original right tensor)
        C : (batch, M, N)
    All tensors are fp16; accumulation is in fp32.
    """
    pid = tl.program_id(0)

    # Number of tiles per matrix
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    blocks_per_mat = num_pid_m * num_pid_n

    # Identify which matrix (batch*hidden) and which tile within that matrix
    mat_id   = pid // blocks_per_mat
    tile_id  = pid % blocks_per_mat
    pid_m = tile_id // num_pid_n
    pid_n = tile_id % num_pid_n

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = k_offset + offs_k
        mask_k = cur_k < K

        # Pointers with proper batch offset
        a_ptrs = a_ptr + mat_id * batch_stride_a + (
            offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        )
        b_ptrs = b_ptr + mat_id * batch_stride_b + (
            cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        )

        # Load tiles (apply masks for out‑of‑bounds elements)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Tensor‑core friendly dot
        acc += tl.dot(a, b)

    # Cast back to fp16 and store
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + mat_id * batch_stride_c + (
        offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    )
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, c, mask=store_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of torch.Tensor parameters
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ----------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ----------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ----------------------------------------------------
    # 2) Cast to fp16 for the heavy work
    # ----------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ----------------------------------------------------
    # 3) Linear projections + gates (fp16)
    # ----------------------------------------------------
    left_proj = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    left_gate_pre = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16))
    right_gate_pre = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16))
    out_gate_pre = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16))

    # ----------------------------------------------------
    # 4) Optional mask (broadcast on hidden_dim)
    # ----------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ----------------------------------------------------
    # 5) Gating (sigmoid, fp16)
    # ----------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate   # [B, N, N, H] fp16
    right = right_proj * right_gate

    # ----------------------------------------------------
    # 6) Output gate (fp16)
    # ----------------------------------------------------
    out_gate = torch.sigmoid(out_gate_pre)   # [B, N, N, H] fp16

    # ----------------------------------------------------
    # 7) Batched matrix multiplication via Triton
    #    out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]
    # ----------------------------------------------------
    B, N, _, H = left.shape
    batch_hidden = B * H

    # Prepare tensors for the kernel (contiguous layout)
    left_t = left.permute(0, 3, 1, 2).contiguous().view(batch_hidden, N, N)
    right_t = right.permute(0, 3, 2, 1).contiguous().view(batch_hidden, N, N)
    out_t = torch.empty_like(left_t)  # fp16 output

    # Strides (all tensors are contiguous)
    stride_am, stride_ak = left_t.stride(1), left_t.stride(2)
    stride_bk, stride_bn = right_t.stride(1), right_t.stride(2)
    stride_cm, stride_cn = out_t.stride(1), out_t.stride(2)
    batch_stride_a = left_t.stride(0)
    batch_stride_b = right_t.stride(0)
    batch_stride_c = out_t.stride(0)

    # Block sizes (tuned for Tensor Cores)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    blocks_per_mat = num_pid_m * num_pid_n
    grid = (batch_hidden * blocks_per_mat,)

    batched_matmul_kernel[grid](
        a_ptr=left_t,
        b_ptr=right_t,
        c_ptr=out_t,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        batch_stride_a=batch_stride_a,
        batch_stride_b=batch_stride_b,
        batch_stride_c=batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to [B, N, N, H]
    out = out_t.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ----------------------------------------------------
    # 8) Cast to float32 for the final layers
    # ----------------------------------------------------
    out_fp32 = out.to(torch.float32)

    # ----------------------------------------------------
    # 9) Final LayerNorm, output gate and projection (float32)
    # ----------------------------------------------------
    out_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_fp32 = out_fp32 * out_gate.to(torch.float32)
    out_fp32 = F.linear(out_fp32, weights["to_out.weight"])  # shape [B, N, N, dim]

    return out_fp32
