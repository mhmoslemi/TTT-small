@triton.jit
def triton_tri_mul(
    left_ptr, right_ptr, out_ptr,
    BH, N,
    stride_lh, stride_li, stride_lk,
    stride_rh, stride_ri, stride_rk,
    stride_oh, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for M and N in bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for left[batch,i,k]
        left_ptrs = left_ptr + batch * stride_lh + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk
        # load left with mask
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # pointers for right transposed: right_T[batch, k, j] where j is column index (offs_n)
        right_ptrs = right_ptr + batch * stride_rh + offs_k[:, None] * stride_ri + offs_n[None, :] * stride_rk
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # dot product: (M,K) x (K,N) => (M,N)
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + batch * stride_oh + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The core O(N³·H) reduction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is performed with a custom Triton kernel that treats each (batch, hidden) slice
as an independent matrix multiplication:   left_{i,k} @ right_{j,k}ᵀ.
All other lightweight operations (LayerNorm, linear projections, gates) use
torch kernels, while the reduction is fully fused in Triton to avoid
intermediate copies and extra transposes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched matrix multiplication of (B*H, N, N) tensors
# --------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    BH, N,
    stride_lh, stride_li, stride_lk,
    stride_rh, stride_ri, stride_rk,
    stride_oh, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)               # index over (B*H)
    pid_m   = tl.program_id(1)                # tile row
    pid_n   = tl.program_id(2)                # tile col

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[batch, i, k]
        left_ptrs = (left_ptr +
                     batch_id * stride_lh +
                     offs_m[:, None] * stride_li +
                     offs_k[None, :] * stride_lk)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right_T[batch, k, j]  (i.e. right[batch, j, k]ᵀ)
        right_ptrs = (right_ptr +
                      batch_id * stride_rh +
                      offs_k[:, None] * stride_ri +
                      offs_n[None, :] * stride_rk)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate dot product (M×K)·(K×N) → (M×N)
        acc += tl.dot(left, right)

    # Write result back as float16
    out_ptrs = (out_ptr +
                batch_id * stride_oh +
                offs_m[:, None] * stride_oi +
                offs_n[None, :] * stride_oj)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------
# Helper: element‑wise multiplication (used for the output gate)
# --------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    assert a.shape == b.shape, "Shape mismatch for element‑wise multiply"
    numel = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((numel + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, numel, BLOCK=BLOCK)
    return out

# --------------------------------------------------------------
# Main entrypoint
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor [B, N, N, C] (same dtype as input_tensor)
    """
    # ----------------------------------------------------------
    # Unpack arguments
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ----------------------------------------------------------
    # 1) Input LayerNorm
    # ----------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)                     # work in fp16 from here

    # ----------------------------------------------------------
    # 2) Cast model weights to fp16
    # ----------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, dtype=torch.float16)

    # ----------------------------------------------------------
    # 3) Linear projections (left/right) and optional masking
    # ----------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------
    # 4) Gate projections
    # ----------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()      # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()       # (B, N, N, H)

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 5) Core O(N³·H) reduction using a fused Triton kernel
    # ----------------------------------------------------------
    B, N, _, H = left.shape   # (B, N, N, H)

    # Rearrange to (B*H, N, N) for the kernel
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)        # (B*H, N, N)

    # Strides in elements
    stride_lh, stride_li, stride_lk = left_flat.stride()
    stride_rh, stride_ri, stride_rk = right_flat.stride()
    stride_oh, stride_oi, stride_oj = out_flat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat, right_flat, out_flat,
        B * H, N,
        stride_lh, stride_li, stride_lk,
        stride_rh, stride_ri, stride_rk,
        stride_oh, stride_oi, stride_oj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------
    # 6) Hidden‑dim LayerNorm
    # ----------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ----------------------------------------------------------
    # 7) Apply the output gate (element‑wise multiplication)
    # ----------------------------------------------------------
    out = _elemwise_mul(out, out_gate)

    # ----------------------------------------------------------
    # 8) Final projection back to original channel dimension
    # ----------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to original dtype if needed
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
