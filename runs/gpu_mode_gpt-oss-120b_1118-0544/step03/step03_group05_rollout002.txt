    # a: (batch, M, K)
    # b: (batch, K, N) (could be transposed view)

    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

"""
TriMul (outgoing) forward pass with Triton‑accelerated core contraction.

Algorithm
---------
1. Layer‑norm over the last channel dimension (dim).
2. Cast to float16 and apply linear projections (left/right) and gating
   projections (left_gate, right_gate, out_gate) using the provided weights.
3. Optional binary mask is applied to the projected tensors.
4. Fuse the gate sigmoids into the projected tensors.
5. Compute the O(N³·H) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication of shape (B·H, N, N) where the
   second operand is transposed.  The contraction is performed by a custom
   Triton kernel `batched_matmul_kernel`.
6. Apply a second Layer‑norm over the hidden dimension, then multiply by the
   output gate (element‑wise).
7. Final linear projection back to the original channel dimension.
8. Cast the result back to the input dtype.

The kernel uses 64×64 tiles with a 32‑element inner reduction and accumulates
in float32 for numerical stability.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ B
#   A: (batch, M, K)   B: (batch, K, N)   C: (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,           # pointers
    M, N, K,                       # matrix sizes
    stride_ah, stride_am, stride_ak,   # A strides: batch, row, col
    stride_bh, stride_bk, stride_bn,   # B strides: batch, row, col
    stride_ch, stride_cm, stride_cn,   # C strides: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program identifiers
    # ------------------------------------------------------------------
    pid = tl.program_id(0)            # tile id (combined over M and N)
    batch_id = tl.program_id(1)       # which matrix in the batch

    # Number of tiles in the M and N dimensions
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # Compute tile coordinates
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # Compute start offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create mask for out‑of‑bounds in M and N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        # K‑dimension offsets for this block
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current block of A and B
        a_ptrs = (a_ptr
                  + batch_id * stride_ah
                  + offs_m[:, None] * stride_am
                  + offs_k[None, :] * stride_ak)
        b_ptrs = (b_ptr
                  + batch_id * stride_bh
                  + offs_k[:, None] * stride_bk
                  + offs_n[None, :] * stride_bn)

        # Load with masking (out‑of‑bounds values are zero)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Fused dot product, accumulation in fp32
        acc += tl.dot(a, b)

    # Cast accumulator back to fp16
    c = acc.to(tl.float16)

    # Store the result
    c_ptrs = (c_ptr
              + batch_id * stride_ch
              + offs_m[:, None] * stride_cm
              + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B using the Triton kernel above.
    A: (batch, M, K)   B: (batch, K, N)   -> C: (batch, M, N)
    All inputs must be contiguous and of dtype float16.
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Inner dimensions must match"
    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N)),   # tiles over M×N
        batch                                          # batch dimension
    )

    batched_matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        a.stride(0), a.stride(1), a.stride(2),
        b.stride(0), b.stride(1), b.stride(2),
        c.stride(0), c.stride(1), c.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return c

# ----------------------------------------------------------------------
# Custom Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask:        torch.Tensor of shape [B, N, N] (or None)
            - weights:     dict with model parameters (see code)
            - config:      dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # ---------------------------- unpack ----------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------- 1) Input LayerNorm -----------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------- 2) Cast to fp16 -------------------------
    x = x.to(torch.float16)

    # ------------------- 3) Prepare FP16 weights -----------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16) # (H, C)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)  # (H, C)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, C)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)      # (C, H)

    # ------------------- 4) Linear projections -------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)  # (B,N,N,H)

    # ------------------- 5) Optional mask -------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------- 6) Gate projections --------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()   # (B,N,N,H)
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()    # (B,N,N,H)

    # ------------------- 7) Fuse gates ---------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------- 8) Core O(N³·H) contraction ------------
    # Reshape to (B*H, N, N) for batched matmul
    B, N, _, H = left.shape
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_mat  = left_t.view(B * H, N, N).contiguous()        # (B*H, N, N)
    right_mat = right_t.view(B * H, N, N).contiguous()       # (B*H, N, N)

    # Transpose right for the matmul: (B*H, N, N) -> (B*H, N, N) with swapped strides
    right_mat_T = right_mat.transpose(-2, -1)  # view with swapped strides

    # Batched matrix multiplication using Triton
    out_mat = batched_matmul(left_mat, right_mat_T)  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------- 9) Output LayerNorm --------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------- 10) Apply output gate ------------------
    out = out * out_gate

    # ------------------- 11) Final linear projection ------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,C)

    # ------------------- 12) Cast back to input dtype ------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)
    return out
