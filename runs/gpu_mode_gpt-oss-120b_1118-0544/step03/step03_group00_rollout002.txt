import torch
import triton
import triton.language as tl

@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,
    gate_ptr,
    mask_ptr,
    out_ptr,
    total_elems,
    hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # compute mask index (broadcast mask over hidden dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor, block_size: int = 1024):
    """
    Multiply projection and gate tensors elementwise and optionally a mask,
    using a fused Triton kernel.
    proj, gate: (B,N,N,H) half
    mask: (B,N,N) half (broadcast over H)
    Returns tensor of same shape as proj/gate.
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Implements:
    1. LayerNorm on input.
    2. Linear projections to hidden dimension.
    3. Gating (sigmoid) and optional mask, fused via a Triton kernel.
    4. Core reduction: out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
       performed as batched matrix multiplication.
    5. Output LayerNorm, output gating and final projection back to input dim.
    The computation uses FP16 internally for speed and returns FP32.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm (FP32)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to FP16 for the heavy path
    x = x.to(torch.float16)

    # 2. Cast all weights to FP16 once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # 3. Linear projections to hidden dim
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # 4. Compute gates (sigmoid)
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # 5. Apply mask + gates
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 6. Core reduction: batched GEMM over hidden dimension
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Compute left @ right^T for each (B*H) slice
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 7. Output LayerNorm (half)
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 8. Multiply by output gate and final linear proj to dim
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # Cast back to input dtype (usually float32)
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

"""
TriMul (outgoing) forward pass.

The algorithm follows the reference PyTorch implementation:
1. Layer‑norm the input tensor.
2. Cast to FP16 for compute‑heavy paths.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are performed with FP16 weights.
4. If a mask is provided (and not disabled) a fused Triton kernel
   multiplies projection * gate * mask in one pass, saving memory traffic.
5. Core reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is expressed as a batched GEMM: for each hidden channel d we
   compute left @ rightᵀ.  The tensors are reshaped to (B*H, N, N) and
   torch.bmm (cuBLAS) performs the large matrix multiplication efficiently.
6. Apply a second LayerNorm, multiply by the output gate and project back
   to the original dimension.
All heavy arithmetic is in FP16; the final result is cast back to the
input dtype (typically FP32).
"""

import torch
import triton
import triton.language as tl


@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,      # (B*N*N*H,)   fp16 projected values
    gate_ptr,      # (B*N*N*H,)   fp16 gate values
    mask_ptr,      # (B*N*N,)     fp16 mask (broadcast over H)
    out_ptr,       # (B*N*N*H,)   fp16 output
    total_elems,   # scalar, total number of elements in proj/gate
    hidden_dim,    # scalar, H (used for broadcast)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # broadcast mask over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=valid)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Elementwise multiplication of projection and gate tensors,
    optionally fused with a binary mask, using a Triton kernel.

    Arguments
    ---------
    proj : torch.Tensor (B, N, N, H), dtype=torch.float16
    gate : torch.Tensor (B, N, N, H), dtype=torch.float16
    mask : torch.Tensor (B, N, N),   dtype=torch.float16 (0/1)

    Returns
    -------
    torch.Tensor (B, N, N, H) with the same dtype as `proj`.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

    # ------------------------------------------------------------------
    # 4. Linear projections from C -> H
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask + gates (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction using batched GEMM:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H should equal hidden_dim
    # reshape to (B*H, N, N) for torch.bmm
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication (left @ rightᵀ)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final projection back to C
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,C)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
