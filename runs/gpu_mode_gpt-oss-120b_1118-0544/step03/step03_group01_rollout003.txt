"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,C]).
2. Linear projections (left/right) and three gates (left, right, out) are computed
   with bias‑less linear layers and a sigmoid activation.  All heavy tensors are
   cast to FP16 for speed.
3. Optional binary mask (if present and nomask=False) is applied before gating.
4. left/right are multiplied by their gates.
5. The hidden dimension H is folded into the batch dimension, producing tensors
   of shape (B*H, N, N).  The core O(N³·H) reduction is expressed as a batched
   matrix multiply:
        out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]   (i.e. A @ Bᵀ)
   This is performed by a custom Triton kernel that tiles the matrices,
   uses FP16 loads, FP32 accumulation and tl.dot (Tensor‑core friendly).
6. The result is reshaped back to (B,N,N,H), layer‑normed over the hidden dim,
   multiplied by the output gate and finally projected back to the original
   channel dimension C.
7. The output is cast back to the dtype of the input tensor.

Only the O(N³·H) reduction is implemented in Triton; all other operations use
high‑performance PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM where each batch corresponds to a (B, H) pair
# Computes: C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]   (i.e. A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix dimensions
    stride_ah, stride_am, stride_ak,    # A: batch, row, col strides
    stride_bh, stride_bk, stride_bn,    # B: batch, k (col), n (row) strides
    stride_ch, stride_cm, stride_cn,    # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch index (B * H)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Batch offsets
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # Pointers to the start of the current tile
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N)

        # Advance pointers to next k‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write out (convert back to FP16)
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton batched GEMM kernel.
    A, B: FP16 tensors of shape (batch, N, N)
    Returns C with the same shape and dtype.
    """
    assert A.shape == B.shape
    batch, N, K = A.shape
    assert N == K, "TriMul expects square sequence dimension."

    # Choose block sizes heuristically based on N
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks in M dimension
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks in N dimension
        batch,                          # batch dimension (B * H)
    )

    C = torch.empty_like(A)

    _batched_matmul_kernel[grid](
        A, B, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),   # A: batch, row, col
        B.stride(0), B.stride(2), B.stride(1),   # B: treat as (batch, K, N)
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Work in FP16 for the heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast model weights to FP16 (all linear layers are bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # fp16
    right = F.linear(x, right_proj_w)  # fp16

    # ------------------------------------------------------------------
    # 4) Optional binary mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Fold hidden dim into batch for batched GEMM
    #    left/right: (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left = left.view(B * H, N, N)
    right = right.view(B * H, N, N)

    # ------------------------------------------------------------------
    # 7) Core O(N³·H) reduction via Triton kernel
    # ------------------------------------------------------------------
    out_flat = _batched_matmul_triton(left, right)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate (broadcasted)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 12) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
