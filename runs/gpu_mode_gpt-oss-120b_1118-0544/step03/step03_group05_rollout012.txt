@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_batch_hidden = tl.program_id(2)  # combined batch*hidden index
    
    # compute batch and hidden index
    bh = pid_batch_hidden
    batch_idx = bh // H
    hidden_idx = bh % H

    # compute row/col offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # create mask for out-of-bound
    mask_m = offs_m < N
    mask_n = offs_n < N

    # compute pointers for output
    # out_ptr + batch_idx*stride_out_b + hidden_idx*stride_out_h + offs_m*stride_out_i + offs_n*stride_out_j
    out_ptrs = out_ptr + batch_idx * stride_out_b + hidden_idx * stride_out_h + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # offsets for left tile
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for K
        mask_k = k_offset < N

        # load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + batch_idx * stride_left_b + hidden_idx * stride_left_h + offs_m[:, None] * stride_left_i + k_offset[None, :] * stride_left_j

        # but left tensor indexing: left[b, i, k, h] has i dimension (stride_left_i), k dimension (stride_left_j).
        # So left_ptrs = left_ptr + batch_idx * stride_left_b + hidden_idx * stride_left_h + offs_m[:, None] * stride_left_i + k_offset[None, :] * stride_left_j

        # apply mask for i,k? Actually we need mask for i, k - but we can load with mask combined.
        left_tile = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right tile: we need B_T shape (BLOCK_K, BLOCK_N) from right[b, k, j, h]
        # Right indices: k dimension is at stride_right_i? Wait original right shape: [B, N, N, H]; indexing: (b, i, j, h). i is first sequence index, j second, h hidden. For B_T we need right[b, k, j, h] where k corresponds to i dimension, j corresponds to j. So we will treat 'i' as k and 'j' as j.
        # So offsets: i offset = k_offset (for i index), j offset = offs_n (for j index)
        right_ptrs = right_ptr + batch_idx * stride_right_b + hidden_idx * stride_right_h + k_offset[:, None] * stride_right_i + offs_n[None, :] * stride_right_j

        # shape (BLOCK_K, BLOCK_N)
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # compute dot
        acc += tl.dot(left_tile, right_tile)  # left: (M,K) dot (K,N) = (M,N)

        # increment pointers (not needed if using offsets each iteration)
    # after loop, convert acc to fp16 and store
    out = acc.to(tl.float16)
    tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # Layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Cast to fp16
    x = x.to(torch.float16)

    # Linear projections: left_proj, right_proj
    left = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))

    # mask (if any)
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Fuse gates
    left = left * left_gate
    right = right * right_gate

    # Compute core contraction using Triton kernel:
    # allocate out tensor
    out = torch.empty_like(left, dtype=torch.float16)   # shape [B,N,N,hidden_dim]

    # Determine strides
    stride_left_b, stride_left_i, stride_left_j, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_j, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor.
2. Cast to fp16 and apply linear projections (left/right) and gates.
3. Optional binary mask is applied before gating.
4. Fuse gates into the projected tensors.
5. Core O(N³·H) contraction is performed as a batched GEMM:
   For each batch b and hidden index h we compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a matrix multiplication A @ Bᵀ with A = left[...,h] and B = right[...,h].
   The GEMM is implemented in a custom Triton kernel that tiles the matrices
   and uses fp16 arithmetic with fp32 accumulation.
6. Apply the second LayerNorm, multiply by the output gate and project back
   to the original channel dimension.
7. Cast the result back to the input dtype.

Only the heavy contraction (step 5) is written in Triton; all other
operations use PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul contraction.
    Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] (i.e. A @ Bᵀ)
    where A = left[...,h] and B = right[...,h].
    """
    pid_m = tl.program_id(0)          # tile row of the output (i)
    pid_n = tl.program_id(1)          # tile column of the output (j)
    pid_bh = tl.program_id(2)         # combined batch*hidden index

    # decode batch and hidden indices
    batch_idx = pid_bh // H
    hidden_idx = pid_bh % H

    # global row/col offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # output pointer for this tile
    out_ptrs = (
        out_ptr
        + batch_idx * stride_out_b
        + hidden_idx * stride_out_h
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j
    )

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_tiles):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load a tile of left: shape (BLOCK_M, BLOCK_K)
        left_tile_ptrs = (
            left_ptr
            + batch_idx * stride_left_b
            + hidden_idx * stride_left_h
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_j
        )
        left_tile = tl.load(
            left_tile_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # load a tile of right^T (i.e. right with i↔k swapped): shape (BLOCK_K, BLOCK_N)
        right_tile_ptrs = (
            right_ptr
            + batch_idx * stride_right_b
            + hidden_idx * stride_right_h
            + offs_k[:, None] * stride_right_i   # k dimension becomes the row
            + offs_n[None, :] * stride_right_j   # j dimension becomes the column
        )
        right_tile = tl.load(
            right_tile_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # multiply‑accumulate
        acc += tl.dot(left_tile, right_tile)

    # write result (convert to fp16)
    out_tile = acc.to(tl.float16)
    tl.store(out_ptrs, out_tile, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor, shape [B, N, N, C] (float32 or fp16)
            mask: torch.Tensor, shape [B, N, N] or None
            weights: dict of model weights (all tensors on CUDA)
            config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """

    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Prepare fp16 weights
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16) # (H, C)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16) # (H, C)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, C)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)      # (C, H)

    # ------------------------------------------------------------------
    # 4) Linear projections (left/right)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w) # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 6) Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()   # (B,N,N,H)
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()  # (B,N,N,H)
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7) Fuse gates into left/right
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate
    # free gate tensors that are no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 8) Core contraction via custom Triton GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left, dtype=torch.float16)   # (B,N,N,H)

    # strides in elements
    s_lb, s_li, s_lj, s_lh = left.stride()
    s_rb, s_ri, s_rj, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # tiling parameters (tuned for typical B200 memory hierarchy)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m, num_blocks_n, B * H)

    _tri_mul_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=s_lb,
        stride_left_i=s_li,
        stride_left_j=s_lj,
        stride_left_h=s_lh,
        stride_right_b=s_rb,
        stride_right_i=s_ri,
        stride_right_j=s_rj,
        stride_right_h=s_rh,
        stride_out_b=s_ob,
        stride_out_i=s_oi,
        stride_out_j=s_oj,
        stride_out_h=s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    # 9) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 12) Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)

    return out
