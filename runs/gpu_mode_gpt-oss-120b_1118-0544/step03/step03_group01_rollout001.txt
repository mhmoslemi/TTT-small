"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy O(N³·H) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a batched matrix multiplication C = A @ Bᵀ, where the hidden
dimension H is folded into the batch dimension (B·H).  The kernel uses
block‑wise tiling, fp16 loads, fp32 accumulation and tl.dot for Tensor‑Core
efficiency.  All lightweight operations (LayerNorms, linear projections,
sigmoid gates, masking) are done in PyTorch.  The final result is returned in
the original input dtype.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,           # pointers
    M, N, K,                       # matrix dimensions (M=N, K=N)
    stride_ah, stride_am, stride_ak,  # A: batch, row, col
    stride_bh, stride_bk, stride_bn,  # B (interpreted as K×N): batch, K, N
    stride_ch, stride_cm, stride_cn,  # C: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM kernel:
        C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]   (i.e. A @ Bᵀ)
    A, B, C are stored as fp16; accumulator is fp32.
    """
    pid_b = tl.program_id(2)   # batch index (B * H)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column

    # ---- tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- base pointers for this batch
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # ---- pointers for the current tile (K dimension iterated)
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- iterate over K tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)  # (BLOCK_M,BLOCK_K) × (BLOCK_K,BLOCK_N) → (BLOCK_M,BLOCK_N)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ---- write out
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Wrapper for the Triton batched GEMM kernel.
    A, B: fp16 tensors of shape (batch, N, N)
    Returns C of same shape and dtype.
    """
    assert A.shape == B.shape, "A and B must have identical shapes"
    batch, N, K = A.shape
    assert N == K, "TriMul expects square matrices per batch"

    # Choose block sizes (heuristic)
    if N <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:  # N up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks in M direction
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks in N direction
        batch,                          # batch dimension (B * H)
    )

    C = torch.empty_like(A)

    _batched_matmul_kernel[grid](
        A, B, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),   # A: batch, row, col
        B.stride(0), B.stride(2), B.stride(1),   # B as K×N (transposed view)
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool, uint8 or float) or None
            - weights: dict containing all linear / layer‑norm parameters
            - config: dict with keys:
                * "dim" (int): input channel dimension C
                * "hidden_dim" (int): hidden dimension H
                * "nomask" (bool, optional): if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (over the channel dimension C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Convert to fp16 for the compute‑heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Optional binary mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Fold hidden dimension into batch for the batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim
    # (B, H, N, N) layout -> contiguous for view
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left = left.view(B * H, N, N)   # (B*H, N, N)
    right = right.view(B * H, N, N)  # (B*H, N, N)

    # ------------------------------------------------------------------
    # 7) Core O(N³·H) reduction using Triton (A @ Bᵀ)
    # ------------------------------------------------------------------
    out_flat = _batched_matmul_triton(left, right)  # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8) Restore original 4‑D layout
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 9) Output LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 12) Cast back to the input dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
