"""
TriMul (outgoing) forward pass – optimized Triton‑fused implementation.

Algorithm
---------
1. LayerNorm the input (float32) and cast to FP16 for the heavy compute path.
2. Fuse the two linear projections for each side (proj + gate) into a single
   matrix multiplication by concatenating the weight matrices.
3. Apply the sigmoid gate and (optional) mask in a single Triton kernel:
       out = proj * gate * mask
4. Core O(N³·H) reduction:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM via torch.bmm on tensors reshaped to
   (B·H, N, N). This leverages cuBLAS/Tensor‑cores.
5. Apply a second LayerNorm, the output gate, and the final linear projection
   back to the original channel dimension.
6. Cast back to the input dtype.

The only Triton kernel used is `apply_gate_mask_kernel`, which fuses
element‑wise multiplication of the projection, gate, and mask.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,          # (B*N*N*H,)       projected values   (fp16)
    gate_ptr,          # (B*N*N*H,)       gate values        (fp16)
    mask_ptr,          # (B*N*N,)         mask values        (fp16)
    out_ptr,           # (B*N*N*H,)       output             (fp16)
    total_elems,       # int  – B*N*N*H
    hidden_dim,        # int  – H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # index into the mask (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Element‑wise fused multiplication of projection, sigmoid‑gate and mask.
    All tensors must be fp16 and contiguous.
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (dtype matches input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_orig = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability) and FP16 cast
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast all needed weights to FP16 (once)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Fuse projection + gate linear layers (weight concatenation)
    # ------------------------------------------------------------------
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, C)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, C)

    # Linear: (B,N,N,C) -> (B,N,N,2*H)
    left_combined = torch.nn.functional.linear(x, left_combined_w)   # no bias
    right_combined = torch.nn.functional.linear(x, right_combined_w)

    # Split projection and gate
    left_proj = left_combined[..., :hidden_dim]
    left_gate_lin = left_combined[..., hidden_dim:]
    left_gate = left_gate_lin.sigmoid()

    right_proj = right_combined[..., :hidden_dim]
    right_gate_lin = right_combined[..., hidden_dim:]
    right_gate = right_gate_lin.sigmoid()

    # ------------------------------------------------------------------
    # 4. Output gate (C -> H)
    # ------------------------------------------------------------------
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5. Apply mask and gates (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core reduction: batch GEMM (left @ right^T) over the sequence dim
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim
    # reshape to (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # batched matrix multiplication
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8. Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != dtype_orig:
        out = out.to(dtype_orig)

    return out
