@triton.jit
def gate_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, H,
    stride_left_b0, stride_left_b1, stride_left_b2, stride_left_b3,
    stride_right_b0, stride_right_b1, stride_right_b2, stride_right_b3,
    stride_lgate_b0, stride_lgate_b1, stride_lgate_b2, stride_lgate_b3,
    stride_rgate_b0, stride_rgate_b1, stride_rgate_b2, stride_rgate_b3,
    stride_mask_b0, stride_mask_b1, stride_mask_b2,
    stride_out_b0, stride_out_b1, stride_out_b2, stride_out_b3,
    BLOCK_T: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_h = tl.program_id(1)

    # compute flat index for (b,i,j)
    offs = pid * BLOCK_T + tl.arange(0, BLOCK_T)
    mask_t = offs < B * N * N

    # compute b,i,j from offs
    # use integer division
    b = offs // (N * N)
    rem = offs % (N * N)
    i = rem // N
    j = rem % N

    # h offset
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # compute pointers
    left_ptrs = left_ptr + b[:, None] * stride_left_b0 + i[:, None] * stride_left_b1 + j[:, None] * stride_left_b2 + offs_h[None, :] * stride_left_b3
    right_ptrs = right_ptr + b[:, None] * stride_right_b0 + i[:, None] * stride_right_b1 + j[:, None] * stride_right_b2 + offs_h[None, :] * stride_right_b3
    left_gate_ptrs = left_gate_ptr + b[:, None] * stride_lgate_b0 + i[:, None] * stride_lgate_b1 + j[:, None] * stride_lgate_b2 + offs_h[None, :] * stride_lgate_b3
    right_gate_ptrs = right_gate_ptr + b[:, None] * stride_rgate_b0 + i[:, None] * stride_rgate_b1 + j[:, None] * stride_rgate_b2 + offs_h[None, :] * stride_rgate_b3

    # mask pointer (if mask present)
    mask_ptrs = mask_ptr + b[:, None] * stride_mask_b0 + i[:, None] * stride_mask_b1 + j[:, None] * stride_mask_b2

    # load
    left = tl.load(left_ptrs, mask=mask_t[:, None] & mask_h[None, :], other=0.0)
    right = tl.load(right_ptrs, mask=mask_t[:, None] & mask_h[None, :], other=0.0)
    left_gate = tl.load(left_gate_ptrs, mask=mask_t[:, None] & mask_h[None, :], other=0.0)
    right_gate = tl.load(right_gate_ptrs, mask=mask_t[:, None] & mask_h[None, :], other=0.0)
    if mask_ptr is not None:
        mask_val = tl.load(mask_ptrs, mask=mask_t, other=1.0)  # shape (BLOCK_T)
        mask_val = mask_val[:, None]  # broadcast over H
    else:
        mask_val = tl.full((BLOCK_T, 1), 1.0, dtype=tl.float16)

    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    # store
    left_out_ptrs = left_out_ptr + b[:, None] * stride_out_b0 + i[:, None] * stride_out_b1 + j[:, None] * stride_out_b2 + offs_h[None, :] * stride_out_b3
    right_out_ptrs = right_out_ptr + b[:, None] * stride_out_b0 + i[:, None] * stride_out_b1 + j[:, None] * stride_out_b2 + offs_h[None, :] * stride_out_b3
    tl.store(left_out_ptrs, left_out, mask=mask_t[:, None] & mask_h[None, :])
    tl.store(right_out_ptrs, right_out, mask=mask_t[:, None] & mask_h[None, :])

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl
    from torch.nn import functional as F

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to fp16
    x = x.to(torch.float16)

    # Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Projections
    left = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # Gates
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # Mask handling
    if mask is None or nomask:
        # mask of ones
        mask_f = torch.ones_like(left[..., :1], dtype=torch.float16, device=device)
    else:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)

    # Prepare output tensors
    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    # Triton kernel for elementwise gating
    total_elements = left.numel()
    BLOCK_SIZE = 256

    grid = ( (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    @triton.jit
    def gate_kernel(
        left_ptr, right_ptr,
        left_gate_ptr, right_gate_ptr,
        mask_ptr,
        left_out_ptr, right_out_ptr,
        total_elements,
        BLOCK_SIZE: tl.constexpr
    ):
        pid = tl.program_id(0)
        offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
        mask = offset < total_elements

        left = tl.load(left_ptr + offset, mask=mask, other=0.0)
        right = tl.load(right_ptr + offset, mask=mask, other=0.0)
        left_gate = tl.load(left_gate_ptr + offset, mask=mask, other=0.0)
        right_gate = tl.load(right_gate_ptr + offset, mask=mask, other=0.0)
        mask_val = tl.load(mask_ptr + offset, mask=mask, other=1.0)

        left_out = left * left_gate * mask_val
        right_out = right * right_gate * mask_val

        tl.store(left_out_ptr + offset, left_out, mask=mask)
        tl.store(right_out_ptr + offset, right_out, mask=mask)

    gate_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_f,
        left_gated,
        right_gated,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # Now compute batched matmul using torch.bmm (fast cuBLAS)
    B = left_gated.shape[0]
    N = left_gated.shape[1]
    H = hidden_dim

    # reshape for batched matmul: (B*H,N,N)
    left_bmm = left_gated.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_bmm = right_gated.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Compute out = left @ right^T
    out_flat = torch.bmm(left_bmm, right_bmm.transpose(1, 2))

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm over hidden dim
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b
    )

    # Apply out_gate
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, to_out_w)

    # Cast back to original dtype
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)
    return out

"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1. Layer‑norm the input tensor on the channel axis (dim C).
2. Cast to fp16 and apply the four linear projections (left, right, left‑gate,
   right‑gate, out‑gate) with the given weights (no bias).
3. A tiny Triton kernel fuses the element‑wise mask (if any) and the left/right
   gates:   left' = left * left_gate * mask,   right' = right * right_gate * mask.
4. Reshape the gated tensors to (B*H, N, N) and perform a batched matrix
   multiplication C = A @ Bᵀ using torch.bmm (cuBLAS – highly tuned on B200).
5. Reshape back to (B, N, N, H), apply a second LayerNorm over the hidden dim,
   multiply by the out‑gate and linearly project back to the original channel dim.
6. Cast the result back to the input dtype.

Only the element‑wise mask/gate step is implemented in Triton; the heavy O(N³·H)
reduction leverages the highly‑optimized cuBLAS GEMM.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elements,
    BLOCK_SIZE: tl.constexpr,
):
    """Element‑wise: left_out = left * left_gate * mask,
                    right_out = right * right_gate * mask."""
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < total_elements

    left      = tl.load(left_ptr      + offset, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offset, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offset, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offset, mask=mask, other=0.0)
    mask_val  = tl.load(mask_ptr      + offset, mask=mask, other=1.0)

    left_out  = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    tl.store(left_out_ptr  + offset, left_out,  mask=mask)
    tl.store(right_out_ptr + offset, right_out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, mask is ignored

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over channel dim C
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the compute‑heavy portion
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (B, N, N, H)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Gate projections (B, N, N, H)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6️⃣ Mask handling – broadcast to (B,N,N,1) and cast to fp16
    # ------------------------------------------------------------------
    if mask is None or nomask:
        mask_f = torch.ones_like(left[..., :1], dtype=torch.float16, device=device)
    else:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)

    # ------------------------------------------------------------------
    # 7️⃣ Fuse mask & gates with a tiny Triton kernel
    # ------------------------------------------------------------------
    left_gated  = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    total_el = left.numel()
    BLOCK = 256
    grid = ( (total_el + BLOCK - 1) // BLOCK, )

    _gate_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_f,
        left_gated,
        right_gated,
        total_el,
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Batched GEMM: C = A @ Bᵀ   (use cuBLAS via torch.bmm)
    # ------------------------------------------------------------------
    B, N, _, H = left_gated.shape  # H == hidden_dim
    # reshape to (B*H, N, N)
    left_mat  = left_gated.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_gated.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # core O(N³·H) reduction
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9️⃣ Output LayerNorm over hidden dim H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 🔟 Apply out_gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Final linear projection back to original channel dim C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 1️⃣2️⃣ Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
