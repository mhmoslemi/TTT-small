    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left_proj = torch.nn.functional.linear(x_fp16, left_proj_w)  # shape [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x_fp16, right_proj_w)

    # Gates
    left_gate = torch.nn.functional.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()

    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16)
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Prepare for batched matmul: reshape to (batch*hidden_dim, N, N)
    B, N, _, H = left.shape
    batch_h = B * H
    # Permute to batch dimension first
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(batch_h, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(batch_h, N, N)

    # Transpose right for multiplication
    right_T = right_flat.transpose(1, 2).contiguous()

    # Output tensor
    out_flat = torch.empty_like(left_flat)

    # Call Triton matmul kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Strides
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_T.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    grid = (batch_h, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LN
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C).
2. Cast to fp16 and compute linear projections (left/right) and gating
   (left_gate, right_gate, out_gate).  The gates are sigmoid‑activated.
3. Fuse mask (if present) with gate and projection using a tiny Triton
   kernel `apply_gate_mask_kernel`.  This saves two memory passes.
4. Core O(N³·H) reduction is performed as a batched matrix multiplication:
   for each hidden dimension h:
        out_h[b,i,j] = Σ_k left_h[b,i,k] * right_h[b,j,k]
   Implemented by a custom Triton kernel `batched_matmul_kernel`,
   which computes C = A @ Bᵀ for all batches (B·H) in fp16 with
   fp32 accumulation.
5. Apply a second LayerNorm on the hidden dimension, multiply by the
   output gate and a final linear projection back to the original channel
   dimension.
6. Cast the result back to the input dtype (usually float32).

Only the elementwise mask‑gate fusion and the batched matrix
multiplication are written in Triton; the surrounding linear layers and
layer‑norms use PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse projection, gate and optional mask (elementwise)
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,          # (B*N*N*H,) fp16
    gate_ptr,          # (B*N*N*H,) fp16
    mask_ptr,          # (B*N*N,)   fp16, broadcast over H
    out_ptr,           # (B*N*N*H,) fp16
    total_elems,       # int: B*N*N*H
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    mask_idx = offs // hidden_dim  # broadcast mask over hidden dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """Elementwise `proj * gate * mask` (mask broadcast over hidden dim)."""
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out

# Kernel 2: batched matrix multiplication C = A @ Bᵀ  (FP16, FP32 acc)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_aba, stride_abr, stride_abc,   # A: (batch, M, K)
    stride_bba, stride_bbr, stride_bbc,   # B: (batch, K, N)  (already transposed)
    stride_cca, stride_ccm, stride_ccn,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = A_ptr + batch_id * stride_aba
    b_base = B_ptr + batch_id * stride_bba
    c_base = C_ptr + batch_id * stride_cca

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = (k_offset + offs_k) < K

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_abr + (k_offset + offs_k)[None, :] * stride_abc
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (K x N) – B is already transposed (K,N)
        b_ptrs = b_base + (k_offset + offs_k)[:, None] * stride_bbr + offs_n[None, :] * stride_bbc
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)   # fp16->fp32 accumulation

    # Write back result
    c = acc.to(tl.float16)
    c_ptrs = c_base + offs_m[:, None] * stride_ccm + offs_n[None, :] * stride_ccn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device
    dtype_in = x.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy ops
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast weights to fp16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections
    # --------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x_fp16, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask & gates (fused via Triton if mask is present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction: batched GEMM (C = left @ rightᵀ)
    #    Shapes after view: (B*H, N, N) @ (B*H, N, N)ᵀ
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    batch_h = B * H

    left_flat = left.permute(0, 3, 1, 2).contiguous().view(batch_h, N, N)   # (B*H, N, K)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(batch_h, N, N) # (B*H, N, K)

    # Transpose right for the multiplication (now (B*H, K, N))
    right_T = right_flat.transpose(1, 2).contiguous()

    out_flat = torch.empty_like(left_flat)

    # Kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_T.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    grid = (
        batch_h,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        N, N, N,                     # M, N, K
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8. Reshape back to (B,N,N,H)
    # --------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 9. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 10. Cast back to original dtype if required
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
