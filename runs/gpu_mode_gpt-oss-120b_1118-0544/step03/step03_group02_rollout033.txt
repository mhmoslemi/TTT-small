def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm (float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5
    )

    # Combined projection and gates linear
    # careful about weight shapes:
    # left_proj.weight: (hidden_dim, dim)
    # left_gate.weight: (hidden_dim, dim)
    # right_proj.weight: (hidden_dim, dim)
    # right_gate.weight: (hidden_dim, dim)

    # Concatenate weights rows
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    # maybe convert to half
    dev = input_tensor.device
    dtype = torch.float16
    # flatten x to (B*N*N, dim)
    x_fp16 = x.to(dtype)

    # combine weights: [left_proj; left_gate; right_proj; right_gate]
    # shape (4*hidden, dim)
    combined_weight = torch.cat([left_proj_weight, left_gate_weight, right_proj_weight, right_gate_weight], dim=0).to(dev).to(dtype)

    # linear
    x_flat = x_fp16.view(-1, dim)  # (B*N*N, dim)
    combined_out = torch.nn.functional.linear(x_flat, combined_weight)  # (B*N*N, 4*hidden)

    # reshape to (B,N,N,4*hidden)
    combined_out = combined_out.view(B, N, N, 4*hidden_dim)

    left_proj, left_gate, right_proj, right_gate = torch.split(combined_out, hidden_dim, dim=-1)

    # mask
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1) # shape (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # gating
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # apply gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    # now compute out via triton matmul per hidden and batch

    # allocate out tensor
    out = torch.empty_like(left, dtype=dtype, device=dev)  # shape (B,N,N,hidden)

    # compute strides
    # left shape: (B,N,N,hidden) (b,i,k,h) order is (b,i,k,h)
    # get strides in elements (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()

    left_ptr, right_ptr, out_ptr,

    stride_r_b, stride_r_j, stride_r_k, stride_r_h,

    pid_j = tl.program_id(2)  # column block

      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   using a custom Triton kernel that performs a batched GEMM for each
   (batch, hidden) pair.  The kernel tiles the K‑dimension to exploit
   shared memory and Tensor‑cores.
6. Layer‑norm over the hidden dimension.
7. Compute the out‑gate (sigmoid of a linear projection) and apply it
   with a small Triton element‑wise kernel.
8. Final linear projection back to the original channel dimension and
   cast the result to float32.

The implementation mixes PyTorch for light‑weight operations and two
Triton kernels (the batched matmul and the element‑wise multiplication).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul for TriMul: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]"""
    pid_group = tl.program_id(0)       # encodes (batch, hidden)
    pid_i = tl.program_id(1)           # block row
    pid_j = tl.program_id(2)           # block column

    # decode batch and hidden index
    b = pid_group // H
    h = pid_group % H

    # base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # start offsets of the current tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k0 in range(0, N, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:  (M, K)
        left_ptrs = left_base + \
                    offs_i[:, None] * stride_l_i + \
                    offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile transposed: (K, N)  →  we load as (K, N) directly
        right_ptrs = right_base + \
                     offs_k[:, None] * stride_r_k + \
                     offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # accumulate dot product (M,N) += (M,K) @ (K,N)
        acc += tl.dot(left, right)

    # write the result
    out_ptrs = out_base + \
               offs_i[:, None] * stride_o_i + \
               offs_j[None, :] * stride_o_j
    tl.store(out_ptrs,
             tl.cast(acc, tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


@triton.jit
def _elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """c = a * b   (1‑D flattened tensors)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float) – optional
            - weights: dict of model weight tensors (float32)
            - config: dict with keys:
                * "dim" (int) – channel dimension C
                * "hidden_dim" (int) – hidden projection size
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_fp16 = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------
    # 1. Input layer‑norm (float32)
    # --------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  float32

    # --------------------------------------------------
    # 2. Fused linear projections (+ gates)
    # --------------------------------------------------
    # collect and cast weights
    left_proj_w  = weights["left_proj.weight"].to(device).to(dtype_fp16)
    right_proj_w = weights["right_proj.weight"].to(device).to(dtype_fp16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(dtype_fp16)
    right_gate_w = weights["right_gate.weight"].to(device).to(dtype_fp16)

    # concatenate for a single matmul: (4*hidden, dim)
    combined_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # (4*hidden, dim)

    # linear projection on flattened input
    x_fp16 = x.to(dtype_fp16)                     # (B, N, N, dim)
    B, N, _, _ = x_fp16.shape
    x_flat = x_fp16.view(-1, dim)                # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_w)  # (B*N*N, 4*hidden)

    # reshape to 4‑D tensor and split
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each: (B, N, N, hidden)

    # --------------------------------------------------
    # 3. Optional mask on the projections (not on gates)
    # --------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype_fp16).unsqueeze(-1)   # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------
    # 4. Apply sigmoid gates
    # --------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------
    # 5. TriMul core: batched GEMM via Triton
    # --------------------------------------------------
    # Allocate output tensor
    out = torch.empty_like(left, dtype=dtype_fp16, device=device)  # (B, N, N, hidden)

    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tiling configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    total_groups = B * hidden_dim  # each (batch, hidden) gets its own program

    grid = (total_groups, num_i, num_j)

    _trimul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------
    # 6. Output LayerNorm (hidden)
    # --------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(dtype_fp16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )

    # --------------------------------------------------
    # 7. Out‑gate computation and element‑wise multiplication
    # --------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device).to(dtype_fp16)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))  # (B, N, N, hidden)

    # Triton element‑wise multiplication (in‑place)
    total_elements = out.numel()
    BLOCK = 1024
    grid_el = ((total_elements + BLOCK - 1) // BLOCK,)
    _elementwise_mul_kernel[grid_el](
        out, out_gate, out, total_elements, BLOCK=BLOCK
    )

    # --------------------------------------------------
    # 8. Final projection back to original dimension
    # --------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(dtype_fp16)
    out = F.linear(out, to_out_w)              # (B, N, N, dim)  fp16
    out = out.to(torch.float32)                # return float32 as required
    return out
```
