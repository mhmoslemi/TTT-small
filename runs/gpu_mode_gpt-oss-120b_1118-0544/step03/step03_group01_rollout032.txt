"""
TriMul (outgoing) forward pass – Triton‑accelerated core multiplication.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) (weights are stored in FP16).
2. Fuse the two projection linear layers (proj & gate) for the left and right
   tensors by concatenating the weight matrices.  Compute both in a single
   matmul, split the result and apply the sigmoid gate.
3. Apply the optional mask *before* the gates (as in the reference code).
4. Reshape left/right from (B,N,N,H) → (B*H,N,N) by folding the hidden dim
   into the batch dimension.
5. Compute the O(N³·H) reduction with a custom Triton batched‑GEMM kernel:
   out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]  (i.e. left @ rightᵀ)
   The kernel uses block‑wise tiling, fp16 loads, fp32 accumulation and
   tensor‑core‑friendly tl.dot.
6. Reshape the result back to (B,N,N,H) and apply a second LayerNorm over H.
7. Multiply by the output gate (sigmoid of a linear on the original input).
8. Final linear projection back to the original channel dimension C.
9. Cast the result to the dtype of the original input tensor.

The heavy O(N³·H) work is performed by the Triton kernel; all other
operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM: C = A @ Bᵀ   (A,B : [batch, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                      # matrix dimensions (M=N, K=N)
    stride_ah, stride_am, stride_ak,   # A: batch, row, col
    stride_bh, stride_bk, stride_bn,   # B: batch, col (k), row (n) – accessed as (k,n)
    stride_ch, stride_cm, stride_cn,   # C: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k] (i.e. A @ Bᵀ)."""
    pid_b = tl.program_id(2)      # batch index (B * hidden_dim)
    pid_m = tl.program_id(0)      # block row
    pid_n = tl.program_id(1)      # block column

    # --------------------------------------------------------------
    # Compute tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch offsets
    a_batch_offset = pid_b * stride_ah
    b_batch_offset = pid_b * stride_bh
    c_batch_offset = pid_b * stride_ch

    # pointers for the current tile (K‑dimension will be traversed)
    a_ptrs = A_ptr + a_batch_offset + (
        offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = B_ptr + b_batch_offset + (
        tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K‑tiles
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)          # (BLOCK_M,BLOCK_K) × (BLOCK_K,BLOCK_N)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Write back
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch_offset + (
        offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Wrapper for the Triton batched GEMM kernel.
    A, B: fp16 tensors of shape (batch, N, N); returns C of the same shape.
    """
    assert A.shape == B.shape
    batch, N, K = A.shape
    assert N == K, "TriMul requires square sequence dimension."

    # Heuristic block sizes
    if N <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along M
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along N
        batch,                          # batch dimension (B * hidden_dim)
    )

    C = torch.empty_like(A)

    # Choose number of warps: 4 for smaller blocks, 8 for large blocks
    num_warps = 4 if BLOCK_M <= 64 else 8

    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        N,
        N,
        N,
        A.stride(0),  # batch stride
        A.stride(1),  # row stride (M)
        A.stride(2),  # col stride (K)
        B.stride(0),
        B.stride(2),  # col stride (k) – we access B as (k,n)
        B.stride(1),  # row stride (n)
        C.stride(0),
        C.stride(1),
        C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )
    return C


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys
            * "dim"         – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask"      – optional bool, default False

    Returns
    -------
    torch.Tensor [B, N, N, C] (same dtype as the input)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # Cast weights to half‑precision (the heavy path works in fp16)
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # LayerNorm (first)
    norm_weight = to_half(weights["norm.weight"])
    norm_bias   = to_half(weights["norm.bias"])

    # Projection + gate weights (concatenated)
    left_proj_w  = to_half(weights["left_proj.weight"])
    left_gate_w  = to_half(weights["left_gate.weight"])
    right_proj_w = to_half(weights["right_proj.weight"])
    right_gate_w = to_half(weights["right_gate.weight"])
    out_gate_w   = to_half(weights["out_gate.weight"])

    # Output side
    to_out_norm_weight = to_half(weights["to_out_norm.weight"])
    to_out_norm_bias   = to_half(weights["to_out_norm.bias"])
    to_out_w           = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp16 computation, fp32 reduction internally)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    ).to(torch.float16)   # cast once for the rest of the pipeline

    # ------------------------------------------------------------------
    # 2) Fuse left projection + left gate, and right projection + right gate
    # ------------------------------------------------------------------
    # Concatenate weights along the output dimension
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, C)
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # Combined linear passes
    left_cat = F.linear(x, left_cat_w)   # (B,N,N,2*H)
    right_cat = F.linear(x, right_cat_w)

    # Split back
    left_proj, left_gate = left_cat.split([hidden_dim, hidden_dim], dim=-1)
    right_proj, right_gate = right_cat.split([hidden_dim, hidden_dim], dim=-1)

    # ------------------------------------------------------------------
    # 3) Optional mask (applied before gates, exactly as reference)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4) Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Output gate (computed on the original normalized input)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6) Prepare tensors for the batched GEMM: fold hidden dim into batch
    # ------------------------------------------------------------------
    Bsz, N, _, H = left.shape   # Bsz == batch size
    # (B, N, N, H) → (B, H, N, N) → view as (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left.view(Bsz * H, N, N)
    right_flat = right.view(Bsz * H, N, N)

    # ------------------------------------------------------------------
    # 7) Core O(N³·H) reduction via Triton
    # ------------------------------------------------------------------
    out_flat = _batched_gemm_triton(left_flat, right_flat)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8) Reshape back to (B,N,N,H)
    # ------------------------------------------------------------------
    out = out_flat.view(Bsz, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    # 10) Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 12) Cast back to the dtype of the original input
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
