"""
TriMul (outgoing) – fast Triton implementation.
------------------------------------------------
1. LayerNorm on the input tensor (B,N,N,C).
2. Linear projections to hidden dimension H (no bias) and three sigmoid gates.
3. Optional binary mask is applied before gating.
4. Elementwise gating of the projections.
5. Core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM C = A @ Bᵀ where A and B have shape (B*H,N,N).
   The kernel fuses the batch and hidden dimensions using a single
   program_id, eliminates any data copies, and uses fp16 loads with fp32
   accumulation (Tensor‑core friendly).
6. LayerNorm over the hidden dimension, output‑gate multiplication,
   and final linear projection back to the original channel dimension.
7. The result is returned in the original dtype of the input.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    # Pointers
    left_ptr, right_ptr, out_ptr,
    # Matrix sizes
    M, N, K,
    # Strides for left tensor (B,N,N,H) → (batch,i,k,hidden)
    stride_left_batch, stride_left_hidden, stride_left_i, stride_left_k,
    # Strides for right tensor (B,N,N,H) → (batch,j,k,hidden)
    stride_right_batch, stride_right_hidden, stride_right_k, stride_right_j,
    # Strides for output tensor (B,N,N,H) → (batch,i,j,hidden)
    stride_out_batch, stride_out_hidden, stride_out_i, stride_out_j,
    # Hidden dimension (used to split program_id into batch & hidden)
    hidden_dim,
    # Compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_bh = tl.program_id(2)  # combined batch * hidden index

    # ------------------------------------------------------------------
    # Decompose pid_bh into batch and hidden indices
    # ------------------------------------------------------------------
    batch_idx = pid_bh // hidden_dim
    hidden_idx = pid_bh - batch_idx * hidden_dim

    # ------------------------------------------------------------------
    # Base pointers for this (batch, hidden) slice
    # ------------------------------------------------------------------
    a_base = left_ptr + batch_idx * stride_left_batch + hidden_idx * stride_left_hidden
    b_base = right_ptr + batch_idx * stride_right_batch + hidden_idx * stride_right_hidden
    c_base = out_ptr + batch_idx * stride_out_batch + hidden_idx * stride_out_hidden

    # ------------------------------------------------------------------
    # Block offsets within the N×N matrix
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the current tile (k‑dimension will be iterated)
    # ------------------------------------------------------------------
    a_ptrs = a_base + (offs_m[:, None] * stride_left_i
                       + tl.arange(0, BLOCK_K)[None, :] * stride_left_k)
    b_ptrs = b_base + (tl.arange(0, BLOCK_K)[:, None] * stride_right_k
                       + offs_n[None, :] * stride_right_j)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K‑tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        # k‑mask for the current tile
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # ------------------------------------------------------------------
    # Write back the result (fp32 → fp16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_base + (offs_m[:, None] * stride_out_i
                       + offs_n[None, :] * stride_out_j)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated core of the outgoing TriMul.
    left, right: tensors of shape (B, N, N, H) in fp16.
    Returns: tensor of the same shape and dtype.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape

    out = torch.empty_like(left)

    # ------------------------------------------------------------------
    # Choose block sizes (heuristic, works for all test configs)
    # ------------------------------------------------------------------
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
    elif N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 128

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        N,
        N,
        N,
        # strides for left (batch, hidden, i, k)
        left.stride(0),
        left.stride(3),
        left.stride(1),
        left.stride(2),
        # strides for right (batch, hidden, k, j) – note the transposed view
        right.stride(0),
        right.stride(3),
        right.stride(2),
        right.stride(1),
        # strides for output (batch, hidden, i, j)
        out.stride(0),
        out.stride(3),
        out.stride(1),
        out.stride(2),
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul)
    with a Triton‑accelerated O(N³·H) core.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Work in fp16 for the heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast linear weights to fp16 (no bias terms)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4) Optional mask (binary) – broadcast over hidden dim
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Gating (sigmoid) and apply to projections
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core triangular multiplication (batched GEMM)
    # ------------------------------------------------------------------
    # left/right are already (B,N,N,H) in fp16; no extra copies.
    out = _tri_mul_triton(left, right)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 10) Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
