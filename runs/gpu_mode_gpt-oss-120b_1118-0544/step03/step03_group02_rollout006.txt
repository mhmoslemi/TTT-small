"""
Outgoing Triangle Multiplicative Update (TriMul) – fast forward pass.

Algorithm:
1. Layer‑norm the 4‑D input tensor over the channel dimension (dim).
2. Fuse the four linear projections (left/right + their gates) into a single
   matrix multiplication using a concatenated weight matrix.
3. Apply the optional pairwise mask to the left/right projections only.
4. Apply sigmoid to the gate tensors and combine with the projections.
5. Compute the O(N³) reduction as a batched half‑precision GEMM:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to (B·hidden) independent matrix multiplies
   L @ Rᵀ using torch.bmm.
6. Layer‑norm over the hidden dimension.
7. Apply the output gate (sigmoid of a linear projection) element‑wise.
   This multiplication is performed by a tiny Triton kernel.
8. Final linear projection back to the original channel dimension.
9. Return the result in float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: element‑wise multiplication (out *= gate)
# -------------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, c_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device

    # --------------------------------------------------------------
    # Cast and cache weights (half‑precision where possible)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] (float32)

    B, N, _, _ = x.shape

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right proj & gate linear layers
    # --------------------------------------------------------------
    # Combined weight of shape (4*hidden_dim, dim)
    combined_weight = torch.cat(
        [left_proj_weight, left_gate_weight, right_proj_weight, right_gate_weight],
        dim=0,
    )  # (4*hidden, dim)

    # Half‑precision forward through combined linear
    x_fp16 = x.to(torch.float16)
    x_flat = x_fp16.view(-1, dim)                     # (B*N*N, dim)
    combined = F.linear(x_flat, combined_weight)     # (B*N*N, 4*hidden)
    combined = combined.view(B, N, N, 4 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )  # each: (B, N, N, hidden)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask on left/right projections only
    # --------------------------------------------------------------
    if not nomask:
        # mask: (B, N, N) -> (B, N, N, 1) and cast to half
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and combine with projections
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate     # (B, N, N, hidden) half
    right = right_proj * right_gate  # (B, N, N, hidden) half

    # --------------------------------------------------------------
    # 5️⃣ Batched GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched matmul over (B*hidden) batches.
    # --------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) for contiguous view
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    left_mat = left.view(B * hidden_dim, N, N)      # (B*hidden, N, N)
    right_mat = right.view(B * hidden_dim, N, N)    # (B*hidden, N, N)

    # Batched matrix multiplication: (B*hidden, N, N) @ (B*hidden, N, N).T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    # Reshape back to (B, N, N, hidden)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, hidden) half

    # --------------------------------------------------------------
    # 6️⃣ LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden) half

    # --------------------------------------------------------------
    # 7️⃣ Output gate (sigmoid) and element‑wise multiplication via Triton
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # (B, N, N, hidden) half

    # Triton element‑wise multiplication: out *= out_gate (in‑place)
    total_elements = out.numel()
    BLOCK = 1024
    grid = ((total_elements + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](
        out,
        out_gate,
        out,
        total_elements,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # (B, N, N, dim) half
    out = out.to(torch.float32)         # Cast to float32 as required

    return out
