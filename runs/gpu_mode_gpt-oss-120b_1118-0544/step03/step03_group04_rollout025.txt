def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0) # batch*hidden index
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch b and hidden h
    b = pid // H
    h = pid % H

    # compute start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # allocate accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute masks for i and j range (if out-of-bound)
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    mask_j = j_range < N

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast to fp16
    x = x.to(torch.float16)

    # Cast weight tensors to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_w) # shape (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_w)

    # Apply mask if needed
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1) # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # Gating
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Core reduction using Triton
    B, N, _, H = left.shape

    # Prepare output tensor
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=left.device)

    # Get strides
    left_strides = left.stride()
    right_strides = right.stride()
    out_strides = out.stride()

    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_strides
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_strides
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_strides

    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

"""
TriMul (outgoing) forward pass using a fused Triton kernel.

Algorithm
---------
1. LayerNorm over the channel dimension `dim`.
2. Linear projections → `left`, `right` (shape [B,N,N,H]).
3. Optional binary mask (broadcasted on the hidden dim).
4. Gating: `left_gate`, `right_gate`, `out_gate` (sigmoid of linear maps).
   Apply `left_gate` and `right_gate`.
5. Core O(N³·H) reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (left @ rightᵀ) with a Triton
   matmul kernel that treats the hidden index `h` as part of the batch.
6. Hidden‑dim LayerNorm, element‑wise multiplication with `out_gate`,
   and final linear projection back to `dim`.
All heavy arithmetic is performed in float16 with fp32 accumulation;
the final result is cast back to the input dtype (float32).

Only the reduction (step 5) is written in Triton – the rest uses
high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    B, N, H,
    # strides (in elements) for the 4‑D tensors [B, N, N, H]
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # -------------------------------------------------------------
    # program identifiers
    # -------------------------------------------------------------
    pid_batch = tl.program_id(0)   # encodes batch * hidden
    pid_i = tl.program_id(1)       # tile along output dim i
    pid_j = tl.program_id(2)       # tile along output dim j

    # Decode batch index b and hidden index h
    b = pid_batch // H
    h = pid_batch % H

    # Starting offsets for the i‑ and j‑tiles
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Create the actual i/j indices for this tile
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    # Guard against out‑of‑bounds tiles
    mask_i = i < N
    mask_j = j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # loop over the reduction axis k
    # -------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ----- load left[b, i, k, h] -----
        a_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + i[:, None] * stride_left_i
            + k[None, :] * stride_left_k
        )
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # ----- load right[b, j, k, h] and treat it as (k, j) -----
        # we need right[b, j, k, h] but we want a tile shaped (k, j)
        b_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + k[:, None] * stride_right_k
            + j[None, :] * stride_right_i
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # ----- dot product (FP16×FP16 → FP32) -----
        acc += tl.dot(a, b)

    # -------------------------------------------------------------
    # write back the result
    # -------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + i[:, None] * stride_out_i
        + j[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _launch_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Launch the Triton kernel that computes left @ rightᵀ for every hidden slice."""
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in element space
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_i, sR_k, sR_h = right.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    # block sizes – chosen to map well to Tensor Cores on recent GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        sL_b,
        sL_i,
        sL_k,
        sL_h,
        sR_b,
        sR_i,
        sR_k,
        sR_h,
        sO_b,
        sO_i,
        sO_j,
        sO_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------------------
    # 1. Input LayerNorm
    # -------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Work in half precision for the heavy part
    x = x.to(torch.float16)

    # -------------------------------------------------------------
    # 2. Cast model weights to half precision
    # -------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------------------
    # 3. Linear projections (left/right)
    # -------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # -------------------------------------------------------------
    # 4. Optional masking
    # -------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------------------
    # 5. Gating
    # -------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()    # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()     # (B, N, N, H)

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # 6. Core O(N³·H) reduction via Triton
    # -------------------------------------------------------------
    out = _launch_tri_mul(left, right)   # (B, N, N, H)

    # -------------------------------------------------------------
    # 7. Hidden‑dim LayerNorm
    # -------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------------------
    # 8. Apply output gate (element‑wise)
    # -------------------------------------------------------------
    out = out * out_gate

    # -------------------------------------------------------------
    # 9. Final projection back to the original channel dimension
    # -------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the input dtype if needed
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
