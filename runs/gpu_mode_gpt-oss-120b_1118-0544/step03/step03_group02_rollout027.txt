@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_abatch, stride_bbatch, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch = pid // (num_blocks_m * num_blocks_n)
    block_id_in_batch = pid % (num_blocks_m * num_blocks_n)
    block_m = block_id_in_batch // num_blocks_n
    block_n = block_id_in_batch % num_blocks_n

    # offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks within matrix bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers with appropriate batch offsets
    a_batch_offset = batch * stride_abatch
    b_batch_offset = batch * stride_bbatch
    c_batch_offset = batch * stride_cbatch

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute actual k offset within K dimension
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # a_ptrs
        a_ptrs = A_ptr + a_batch_offset + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        # b_ptrs
        b_ptrs = B_ptr + b_batch_offset + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load with masks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # store result
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch_offset + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) kernel.

Algorithm:
1. Layer‑norm the input tensor (float32 → fp16 afterwards).
2. Fuse the four linear projections (left/right + their gates) into a single
   matmul, then split the result to obtain left_proj, left_gate, right_proj,
   right_gate (all fp16).
3. Optionally apply the pair‑wise mask to left/right projections.
4. Apply sigmoid gates and compute left = left_proj * left_gate,
   right = right_proj * right_gate.
5. Compute the O(N³) reduction
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   as a batched FP16 GEMM (left @ rightᵀ) using a custom Triton kernel.
6. Layer‑norm over the hidden dimension.
7. Compute out_gate = sigmoid(x_norm @ out_gate_weight) and multiply it
   element‑wise with the tensor from step 5 using a tiny Triton kernel.
8. Final linear projection back to the original dimension and cast to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_abatch, stride_bbatch, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched FP16 GEMM: C[b] = A[b] @ B[b]  (A: M×K, B: K×N)."""
    pid = tl.program_id(0)

    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch = pid // (num_blocks_m * num_blocks_n)
    block_in_batch = pid % (num_blocks_m * num_blocks_n)
    block_m = block_in_batch // num_blocks_n
    block_n = block_in_batch % num_blocks_n

    # Offsets for the tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch_off = batch * stride_abatch
    b_batch_off = batch * stride_bbatch
    c_batch_off = batch * stride_cbatch

    # Accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = A_ptr + a_batch_off + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = B_ptr + b_batch_off + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch_off + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, size: tl.constexpr, BLOCK: tl.constexpr):
    """c = a * b  (flat 1‑D view)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)

# Main entry point
# ----------------------------------------------------------------------


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config["nomask"]
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dim (float32 → fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] float32
    x = x_norm.to(torch.float16)  # work in fp16 from now on

    # ------------------------------------------------------------------
    # 2. Fuse left/right projections and their gates
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)

    # (4*hidden, dim) weight matrix
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # shape (4*hidden, dim)

    # Linear projection (single matmul)
    proj = F.linear(x.view(-1, dim), combined_weight)  # (B*N*N, 4*hidden)
    proj = proj.view(B, N, N, 4 * hidden)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        proj, hidden, dim=-1
    )  # each: (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 3. Apply mask (if any) to the *projections* only
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate   # (B, N, N, hidden)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Batched GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as (B*hidden) independent matmuls: (N,N) @ (N,N)^T
    # ------------------------------------------------------------------
    # Permute to (B, hidden, N, N) and merge batch*hidden
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, hidden, N, N)

    bh = B * hidden
    left_mat = left.view(bh, N, N)                     # (B*hidden, M=N, K=N)
    right_mat = right.view(bh, N, N).permute(0, 2, 1).contiguous()  # (B*hidden, K=N, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # strides (int)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    stride_abatch = left_mat.stride(0)
    stride_bbatch = right_mat.stride(0)
    stride_cbatch = out_mat.stride(0)

    # Tiling sizes – tuned for fp16 tensor cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    nb_m = (N + BLOCK_M - 1) // BLOCK_M
    nb_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (bh * nb_m * nb_n,)

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_abatch,
        stride_bbatch,
        stride_cbatch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. to_out_norm (LayerNorm over hidden dim)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 7. out_gate computation and element‑wise multiplication (Triton)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B,N,N,hidden)

    total = out.numel()
    BLOCK = 65536
    grid = ((total + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        out,
        out_gate,
        out,
        total,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 8. Final projection back to original dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, dim) fp16
    out = out.to(torch.float32)     # required output dtype
    return out
