@triton.jit
def batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_m, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute starting offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # offset pointer to the start of this batch
    left_ptrs = left_ptr + batch_idx * stride_left_batch
    right_ptrs = right_ptr + batch_idx * stride_right_batch
    out_ptrs = out_ptr + batch_idx * stride_out_batch

    # initialize accumulation (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offset
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(left_ptrs + (offs_m[:, None] * stride_left_m) + (offs_k[None, :] * stride_left_k), mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B tile: shape (BLOCK_K, BLOCK_N)
        b = tl.load(right_ptrs + (offs_k[:, None] * stride_right_m) + (offs_n[None, :] * stride_right_k), mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        # cast to float32 for accumulation
        acc += tl.dot(a, b)

    # store result
    tl.store(out_ptrs + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

```
"""
custom_kernel for outgoing TriMul.
...
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for batched GEMM (left @ right^T)
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, out_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_m, stride_b_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # allocate accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile [M, K]
        a = tl.load(
            a_ptr
            + batch_idx * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile [K, N] (note transposed access)
        b = tl.load(
            b_ptr
            + batch_idx * stride_b_batch
            + offs_k[:, None] * stride_b_m
            + offs_n[None, :] * stride_b_k,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # Write back result to out (cast to fp16)
    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + batch_idx * stride_out_batch
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n,
        out,
        mask=mask_m[:, None] & mask_n[None, :],
    )

def batched_matmul_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using Triton.
    Computes out = a @ b.transpose(-1, -2) for each batch.
    a, b: tensors of shape [B, N, N] (half precision).
    Returns tensor of same shape and dtype.
    """
    assert a.shape == b.shape, "Shapes must match"
    batch, M, K = a.shape
    N = K  # square

    out = torch.empty_like(a)

    # Tunable block sizes (good for typical N up to 1024)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        a,
        b,
        out,
        batch,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

def custom_kernel(data):
    """Outgoing TriMul forward pass with Triton‑accelerated core reduction."""
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on input (dim)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Work in fp16 for the heavy path
    x = x.to(torch.float16)

    # Cast model weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w) # (B, N, N, H)

    # Optional mask
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # Prepare for batched GEMM (left @ right^T per hidden slice)
    B, N, _, H = left.shape

    # Move hidden dimension to batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Core O(N³·H) reduction using Triton
    out_flat = batched_matmul_triton(left_flat, right_flat)  # (B*H, N, N)

    # Restore shape (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Hidden‑dim LayerNorm
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Apply output gate (element‑wise)
    out = out * out_gate

    # Final projection back to original channel dimension
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # Cast back to original dtype
    if out.dtype != input_tensor.dtype:

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The core reduction 
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h] 
is a batched matrix multiplication (left @ rightᵀ) over the hidden dimension.
We implement this reduction with a fused Triton kernel that processes every
(hidden‑slice) batch in parallel, accumulating in float32 while keeping all
other operations (layernorm, linear projections, gating) in PyTorch for
convenience.  The final output is cast back to the input dtype (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM (A @ Bᵀ) for many small square matrices.
# Each program processes a tile of the output (BLOCK_M × BLOCK_N) for a single
# batch element (i.e. a particular hidden slice).  Accumulation is done in
# float32 for accuracy.
# ----------------------------------------------------------------------------- #
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, out_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_m, stride_b_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # -------------------------------------------------
    # program IDs (batch, row‑tile, col‑tile)
    # -------------------------------------------------
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile column

    # -------------------------------------------------
    # tile offsets
    # -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------------------------------------------------
    # accumulator in fp32
    # -------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # loop over K dimension in BLOCK_K sized chunks
    # -------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + batch_idx * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile (transposed): shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr
            + batch_idx * stride_b_batch
            + offs_k[:, None] * stride_b_m
            + offs_n[None, :] * stride_b_k,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot‑product accumulation (fp32)
        acc += tl.dot(a, b)

    # -------------------------------------------------
    # write back: cast to fp16
    # -------------------------------------------------
    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + batch_idx * stride_out_batch
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n,
        out,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def batched_matmul_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Compute out = a @ bᵀ for each batch using the Triton kernel above.
    a, b: tensors of shape [B, N, N] (dtype torch.float16).
    Returns tensor of same shape and dtype.
    """
    assert a.shape == b.shape, "Shapes of a and b must match"
    batch, M, K = a.shape
    N = K  # square matrices

    out = torch.empty_like(a)

    # Tunable tile sizes – work well for N up to 1024 on Hopper GPUs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        a,
        b,
        out,
        batch,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------------- #
# Main entry point – matches the required signature.
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C]   (float32 or float16)
            - mask:        torch.Tensor [B, N, N] or None (binary)
            - weights:     dict of model parameters (float32)
            - config:      dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------------- #
    # 1. Input LayerNorm (along channel dimension C)
    # ----------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Compute heavy path in half precision
    x = x.to(torch.float16)

    # ----------------------------------------------------------------- #
    # 2. Cast model weights to half precision
    # ----------------------------------------------------------------- #
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ----------------------------------------------------------------- #
    # 3. Linear projections (left / right)
    # ----------------------------------------------------------------- #
    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w) # (B, N, N, H)

    # ----------------------------------------------------------------- #
    # 4. Optional masking (applied before gates)
    # ----------------------------------------------------------------- #
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------------- #
    # 5. Gates (sigmoid of linear projections)
    # ----------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------- #
    # 6. Core O(N³·H) reduction using Triton batched GEMM
    #    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # ----------------------------------------------------------------- #
    B, N, _, H = left.shape  # (B, N, N, H)

    # Move hidden dimension into the batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Batched GEMM: left @ rightᵀ for each hidden slice
    out_flat = batched_matmul_triton(left_flat, right_flat)  # (B*H, N, N)

    # Restore original layout (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------- #
    # 7. Hidden‑dim LayerNorm
    # ----------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------------------------------------------------- #
    # 8. Output gate (element‑wise)
    # ----------------------------------------------------------------- #
    out = out * out_gate

    # ----------------------------------------------------------------- #
    # 9. Final linear projection back to original channel dimension C
    # ----------------------------------------------------------------- #
    out = F.linear(out, to_out_w)  # (B, N, N, C)

    # ----------------------------------------------------------------- #
    # 10. Cast back to original dtype if necessary
    # ----------------------------------------------------------------- #
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
