def matmul_kernel(

    a = tl.load(a_ptrs)
    b = tl.load(b_ptrs)
    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        a_next = tl.load(a_ptrs)
        b_next = tl.load(b_ptrs)

        # Move pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
        # Advance
        a = a_next
        b = b_next

    # Write back

"""
custom_kernel: Outgoing Triangle Multiplicative Update (TriMul) with Triton + PyTorch.

Algorithm
---------
1. LayerNorm the input tensor (B,N,N,C).
2. Cast to fp16 for the compute‑heavy path.
3. Linear projections (left/right) and three sigmoid gates (left, right, out)
   are computed with fp16 weights.
4. A tiny Triton kernel fuses mask, gate and elementwise multiplication:
       out = proj * gate * mask
   (mask is all‑ones if no mask is provided).
5. Collapse the hidden dimension H into the batch dimension:
       (B,N,N,H) → (B*H,N,N)
6. Core O(N³·H) reduction is performed by a batched matrix‑multiply
   (C = A @ Bᵀ) using Torch's highly‑optimized cuBLAS implementation.
7. Reshape back to (B,N,N,H), apply a second LayerNorm,
   multiply by the output gate and project back to the original channel
   dimension C.
8. Cast the result back to the original dtype.

Only the element‑wise mask‑gate fusion is implemented in Triton,
satisfying the requirement of using a custom kernel while leveraging
cuBLAS for the expensive O(N³·H) contraction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    inp_ptr, gate_ptr, mask_ptr, out_ptr,    # pointers
    size,                                    # total number of elements
    stride_inp: tl.constexpr,
    stride_gate: tl.constexpr,
    stride_mask: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """Element‑wise fusion: out = inp * gate * mask."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    m = offs < size

    a = tl.load(inp_ptr + offs * stride_inp, mask=m, other=0.0)
    g = tl.load(gate_ptr + offs * stride_gate, mask=m, other=0.0)
    mk = tl.load(mask_ptr + offs * stride_mask, mask=m, other=0.0)

    out = a * g * mk
    tl.store(out_ptr + offs * stride_inp, out, mask=m)


def _apply_mask_gate(inp, gate, mask):
    """Apply the Triton fusion kernel to (inp, gate, mask)."""
    # All tensors are assumed contiguous and of the same shape.
    total = inp.numel()
    BLOCK = 1024
    grid = ( (total + BLOCK - 1) // BLOCK, )
    _mask_gate_fuse_kernel[grid](
        inp, gate, mask, inp,                 # in‑place write back into `inp`
        total,
        inp.stride(-1), gate.stride(-1), mask.stride(-1),
        BLOCK_SIZE=BLOCK,
    )
    return inp


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict with model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 -> fp16 after)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)        # (B,N,N,H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 4) Prepare mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)            # (B,N,N,1)
    else:
        # all‑ones mask
        mask_f = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2], 1),
            dtype=torch.float16,
            device=device,
        )

    # Ensure contiguity for Triton kernel
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_f = mask_f.contiguous()

    # ------------------------------------------------------------------
    # 5) Fuse mask and gates (Triton)
    # ------------------------------------------------------------------
    left = _apply_mask_gate(left_proj, left_gate, mask_f)   # in‑place result
    right = _apply_mask_gate(right_proj, right_gate, mask_f)

    # ------------------------------------------------------------------
    # 6) Collapse hidden dim into batch for the O(N³·H) contraction
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                      # (B,N,N,H)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left = left.view(B * H, N, N)                  # (B*H,N,N)
    right = right.view(B * H, N, N)

    # ------------------------------------------------------------------
    # 7) Core reduction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # out_flat[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]  (i.e. left @ rightᵀ)
    out_flat = torch.bmm(left, right.transpose(1, 2))

    # ------------------------------------------------------------------
    # 8) Reshape back to (B,N,N,H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 12) Cast back to the original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != dtype:
        out = out.to(dtype)

    return out
