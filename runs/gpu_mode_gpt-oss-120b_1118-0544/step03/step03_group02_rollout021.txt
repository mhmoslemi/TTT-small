@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ah, stride_ak, stride_ab, # a: (batch, M, K) stride per dim
    stride_bk, stride_bn, stride_bb, # b: (batch, K, N)
    stride_ch, stride_cn, stride_cb, # c: (batch, M, N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr, num_warps: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # create accumulators
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # pointers to start of batch
    a_ptrs = a_ptr + batch_id * stride_ab + offs_m[:, None] * stride_ah + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (k*BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k*BLOCK_K + tl.arange(0, BLOCK_K) < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
    # store
    c_ptrs = c_ptr + batch_id * stride_cb + offs_m[:, None] * stride_ch + offs_n[None, :] * stride_cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ah, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_ch, stride_cn, stride_cb,

"""
Outgoing Triangle Multiplicative Update (TriMul) – fast forward implementation.

Algorithm
---------
1. Layer‑norm over the channel dimension (`dim`).
2. Fuse the four linear layers (left/right projections + their gates) into a
   single matrix multiplication using a concatenated weight matrix.
3. Optional masking of the projections.
4. Apply sigmoid gates and element‑wise multiplication.
5. Core O(N³) step: for each batch and hidden channel compute
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM (left @ rightᵀ) implemented in Triton.
6. Layer‑norm over the hidden dimension, apply the output gate, and final
   linear projection back to `dim`.
All heavy arithmetic is performed in float16; the final result is returned in
float32.  The Triton kernel in step 5 provides the required GPU‑level fusion.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ Bᵀ) where A and B have shape
# (batch, M, K) and (batch, K, N).  Here M = N = K = seq_len.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ah, stride_ak, stride_ab,   # A: (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # B: (batch, K, N)
    stride_ch, stride_cn, stride_cb,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr,
    num_warps: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in fp16 (tensor‑core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # Pointers to the start of the current block (no K offset yet)
    a_ptrs = a_ptr + batch_id * stride_ab + offs_m[:, None] * stride_ah
    b_ptrs = b_ptr + batch_id * stride_bb + offs_n[None, :] * stride_bn

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        k_offs = cur_k + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        mask_k = k_offs < K
        mask_a = (offs_m[:, None] < M) & mask_k[None, :]
        mask_b = mask_k[:, None] & (offs_n[None, :] < N)

        a = tl.load(a_ptrs + k_offs[None, :] * stride_ak, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs + k_offs[:, None] * stride_bk, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

        # advance K‑dimension pointers for the next iteration
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store result
    c_ptrs = c_ptr + batch_id * stride_cb + offs_m[:, None] * stride_ch + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config["nomask"] is True
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

        torch.Tensor of shape [B, N, N, dim] (float32)
    """

    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Load and cast weights
    # ------------------------------------------------------------------

    norm_bias = weights["norm.bias"].to(device)

    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    # ------------------------------------------------------------------

        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # 2️⃣ Fuse four linear layers (proj & gate) into one matmul
    # ------------------------------------------------------------------
    combined_weight = torch.cat(
        [left_proj_weight, left_gate_weight, right_proj_weight, right_gate_weight],
        dim=0,
    )  # (4*hidden, dim)

    # 4️⃣ Gating

    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate          # (B, N, N, hidden)
    right = right_proj * right_gate

    # ------------------------------------------------------------------

    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    right_t = right.transpose(2, 3).contiguous()  # (B, hidden, N, N)

    P = B * hidden_dim
    left_flat = left.view(P, N, N)
    right_flat = right_t.view(P, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    stride_ab = left_flat.stride(0)
    stride_ah = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cb = out_flat.stride(0)
    stride_ch = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # Launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        P,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        a_ptr=left_flat.data_ptr(),
        b_ptr=right_flat.data_ptr(),
        c_ptr=out_flat.data_ptr(),
        M=N,
        N=N,
        K=N,
        stride_ah=stride_ah,
        stride_ak=stride_ak,
        stride_ab=stride_ab,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_bb=stride_bb,
        stride_ch=stride_ch,
        stride_cn=stride_cn,
        stride_cb=stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # Reshape back to (B, N, N, hidden)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ to_out_norm (LayerNorm over hidden dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden)  fp16

    # ------------------------------------------------------------------
    # 7️⃣ Output gate and final projection
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # (B, N, N, hidden) fp16
    out = out * out_gate

    out = F.linear(out, to_out_weight)   # (B, N, N, dim) fp16
    out = out.to(torch.float32)          # return float32 as required
    return out
