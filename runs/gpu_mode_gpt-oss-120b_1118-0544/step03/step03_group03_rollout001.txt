Implementation of Triton kernel:

```
@triton.jit
def triu_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_j, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)  # combined batch*head

    # compute batch index and head index
    b = pid_hb // H
    h = pid_hb % H

    # compute block start indices
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # offsets within block
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    # mask for out-of-bound
    mask_i = offs_i < N
    mask_j = offs_j < N

    # allocate accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N   # not used for load mask but for to limit load

        # compute pointers
        a_ptrs = left_ptr + b*stride_l_b + h*stride_l_h + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        b_ptrs = right_ptr + b*stride_r_b + h*stride_r_h + (offs_j[:, None] * stride_r_j) + (offs_k[None, :] * stride_r_k)

        # load with mask
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # matrix multiplication: (M,K) x (N,K)^T -> (M,N)
        acc += tl.dot(a, b, trans_b=True)
    # store result
    out_ptrs = out_ptr + b*stride_o_b + h*stride_o_h + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy O(B * hidden_dim * N³) reduction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is implemented as a batched matrix multiplication using a custom Triton kernel.
All linear projections, gating and the final LayerNorm / projection are done in
PyTorch (FP16 where possible). The kernel computes the N×N products for each
(batch, head) pair in FP16 with FP32 accumulation for numerical stability.
The output of the kernel is FP32 and is then processed by the remaining
operations to produce the final tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes for each (batch, head) pair the matrix
#   C = A @ Bᵀ  with  A[i, k] = left[b, h, i, k],  B[j, k] = right[b, h, j, k]
#   (i, j are the output row/col indices, k is summed over)
# ----------------------------------------------------------------------
@triton.jit
def triu_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_i = tl.program_id(0)          # block row
    pid_j = tl.program_id(1)          # block col
    pid_hb = tl.program_id(2)         # combined batch*head

    # decode batch and head
    b = pid_hb // H
    h = pid_hb % H

    # block start offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # offsets within the block
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    # boundary masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for the current tile
        a_ptrs = (left_ptr
                  + b * stride_l_b
                  + h * stride_l_h
                  + (offs_i[:, None] * stride_l_i)
                  + (offs_k[None, :] * stride_l_k))
        b_ptrs = (right_ptr
                  + b * stride_r_b
                  + h * stride_r_h
                  + (offs_j[:, None] * stride_r_i)
                  + (offs_k[None, :] * stride_r_k))

        # load tiles (FP16)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=mask_j[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # matrix multiply A (M×K) * Bᵀ (K×N) -> (M×N)
        # result is FP16, cast to FP32 before accumulation
        acc += tl.dot(a, b, trans_b=True).to(tl.float32)

    # store result (FP32)
    out_ptrs = (out_ptr
                + b * stride_o_b
                + h * stride_o_h
                + (offs_i[:, None] * stride_o_i)
                + (offs_j[None, :] * stride_o_j))
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be None
            - weights: dict of model weights (see skeleton)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    eps = 1e-5

    # ----------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # ----------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # ----------------------------------------------------
    # 2) Cast to FP16 for the heavy part
    # ----------------------------------------------------
    x_h = x.to(torch.float16)

    # ----------------------------------------------------
    # 3) Linear projections + gates (FP16, fused)
    # ----------------------------------------------------
    # concatenate left_proj & left_gate weights
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device).to(torch.float16)
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device).to(torch.float16)

    left_combined = F.linear(x_h, left_combined_w)   # [B, N, N, 2*H] fp16
    right_combined = F.linear(x_h, right_combined_w)  # [B, N, N, 2*H] fp16

    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)

    # ----------------------------------------------------
    # 4) Optional mask (broadcast on hidden dim)
    # ----------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ----------------------------------------------------
    # 5) Gating (sigmoid, FP16)
    # ----------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate    # [B, N, N, H] fp16
    right = right_proj * right_gate  # [B, N, N, H] fp16

    # ----------------------------------------------------
    # 6) Output gate (FP16)
    # ----------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    out_gate_pre = F.linear(x_h, out_gate_w)   # [B, N, N, H] fp16
    out_gate = torch.sigmoid(out_gate_pre)     # fp16

    # ----------------------------------------------------
    # 7) Triangular multiplicative update via Triton
    # ----------------------------------------------------
    B, N, _, _ = left.shape

    # Rearrange to (B, H, N, N) for the kernel
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] fp16
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N] fp16

    # Output tensor (FP32) in the same layout
    out_t = torch.empty((B, hidden_dim, N, N), dtype=torch.float32, device=device)

    # Strides (in elements)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left_t.stride()
    stride_r_b, stride_r_h, stride_r_i, stride_r_k = right_t.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out_t.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m, num_blocks_n, B * hidden_dim)

    triu_mul_kernel[grid](
        left_ptr=left_t,
        right_ptr=right_t,
        out_ptr=out_t,
        B=int(B),
        H=int(hidden_dim),
        N=int(N),
        stride_l_b=int(stride_l_b),
        stride_l_h=int(stride_l_h),
        stride_l_i=int(stride_l_i),
        stride_l_k=int(stride_l_k),
        stride_r_b=int(stride_r_b),
        stride_r_h=int(stride_r_h),
        stride_r_i=int(stride_r_i),
        stride_r_k=int(stride_r_k),
        stride_o_b=int(stride_o_b),
        stride_o_h=int(stride_o_h),
        stride_o_i=int(stride_o_i),
        stride_o_j=int(stride_o_j),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ----------------------------------------------------
    # 8) Reshape back to [B, N, N, H] (FP32)
    # ----------------------------------------------------
    out = out_t.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] FP32

    # ----------------------------------------------------
    # 9) Final LayerNorm, output gating and projection back to dim
    # ----------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)          # apply output gate
    out = F.linear(out, weights["to_out.weight"])   # final projection to dim

    return out
