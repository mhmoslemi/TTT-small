"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton‑accelerated forward.
The heavy O(N³·hidden) reduction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is expressed as a batched matrix multiplication (left @ rightᵀ) and executed
with a custom Triton kernel. All other linear / gating / layer‑norm steps are
performed in mixed‑precision (FP16) with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_ab, stride_am, stride_ak,
    stride_bb, stride_bk, stride_bn,
    stride_cb, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C[b, i, j] = Σₖ A[b, i, k] * B[b, j, k] for many batches.
    A : (batch, M, K)
    B : (batch, N, K)   (note the K‑dimension is the inner one, we multiply A with Bᵀ)
    C : (batch, M, N)
    """
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    # Coordinate offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a_ptrs = a_ptr + batch_id * stride_ab \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (K × N) – B is stored as (N, K) so we treat K as rows
        b_ptrs = b_ptr + batch_id * stride_bb \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate the dot product
        acc += tl.dot(a, b)

    # Write back C tile (M × N)
    c_ptrs = c_ptr + batch_id * stride_cb \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config["nomask"] is True
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # cast / cache weights (FP16 where possible)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)            # float32
    norm_bias   = weights["norm.bias"].to(device)              # float32

    left_proj_weight  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight   = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the input channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # still float32
    x = x.to(torch.float16)   # work in FP16 from here

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections + gates
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)          # (B,N,N,hidden)
    right_proj = F.linear(x, right_proj_weight)

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))

    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask on the projections (mask is broadcasted)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Apply gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate          # (B,N,N,hidden)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented with the Triton kernel above.
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N) for the kernel
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(-1, N, N)   # (B*hidden, N, N)
    right_mat = right_perm.view(-1, N, N)

    out_mat = torch.empty_like(left_mat)   # FP16 output

    # block sizes – tuned for B200 (tensor‑core friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        left_mat.shape[0],                     # batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        a_ptr = left_mat,
        b_ptr = right_mat,
        c_ptr = out_mat,
        stride_ab = left_mat.stride(0),
        stride_am = left_mat.stride(1),
        stride_ak = left_mat.stride(2),
        stride_bb = right_mat.stride(0),
        stride_bk = right_mat.stride(2),   # K‑row stride (inner)
        stride_bn = right_mat.stride(1),   # N‑col stride (outer)
        stride_cb = out_mat.stride(0),
        stride_cm = out_mat.stride(1),
        stride_cn = out_mat.stride(2),
        M = N,
        N = N,
        K = N,
        BLOCK_M = BLOCK_M,
        BLOCK_N = BLOCK_N,
        BLOCK_K = BLOCK_K,
    )

    # reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply out‑gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)  FP16
    out = out.to(torch.float32)          # return as FP32
    return out
