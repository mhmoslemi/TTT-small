def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    
    # LayerNorm
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias = weights["norm.bias"].to(dtype)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    
    # Cast to half for heavy ops
    x = x.to(torch.float16)
    
    # Load linear weights to half
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    
    # Linear projections
    left = F.linear(x, left_proj_w)   # shape (B,N,N,H)
    right = F.linear(x, right_proj_w)
    
    # Mask
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # shape (B,N,N,1)
        left = left * mask_f
        right = right * mask_f
    
    # Gates
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()
    
    left = left * left_gate
    right = right * right_gate
    
    # Core TriMul
    out_flat = _batched_matmul_triton(left, right, hidden_dim)
    
    # Reshape back to (B,N,N,H)
    bs, N, _, H = left.shape
    out = out_flat.view(bs, H, N, N).permute(0, 2, 3, 1).contiguous()
    
    # to_out_norm
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    
    # Apply output gate
    out = out * out_gate
    
    # Final linear projection
    out = F.linear(out, to_out_w)
    
    # Cast back to original dtype
    if out.dtype != dtype:
        out = out.to(dtype)
    return out
```

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1️⃣ Layer‑norm the (B, N, N, C) input.
2️⃣ Cast to float16 and apply four linear projections:
   left/right (to hidden_dim) and three gates (sigmoid).
3️⃣ Optional binary mask is applied before gating.
4️⃣ Gates are fused into left/right.
5️⃣ Core O(N³·H) reduction:
   For each hidden channel d we need
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   This is a batched matrix‑multiply C = A @ Bᵀ where
   A = left   (shape (B, N, N, H) → (B·H, N, N)),
   B = right  (same shape, transposed on the last two N dimensions).
   A single Triton kernel computes all batches simultaneously,
   exposing the hidden dimension as an extra program‑id dimension
   to avoid any data copy or permute.
6️⃣ The result is reshaped back to (B, N, N, H), layer‑normed,
   multiplied by the output gate and linearly projected back to C.
7️⃣ Output is returned in the original dtype (float32).

The heavy O(N³·H) work is done in Triton; all other ops use
torch.nn.functional for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,                           # matrix sizes (all = seq_len)
    stride_ah, stride_am, stride_ak, stride_ad,   # A: batch, i, k, hidden
    stride_bh, stride_bk, stride_bn, stride_bd,   # B: batch, i (as k), k (as n), hidden
    stride_ch, stride_cm, stride_cn, stride_cd,   # C: batch, i, n, hidden
    HIDDEN_DIM: tl.constexpr,          # hidden dimension size
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  with hidden dim folded into the batch."""
    pid_m = tl.program_id(0)            # tile over output rows   (i)
    pid_n = tl.program_id(1)            # tile over output cols   (j)
    pid   = tl.program_id(2)            # flattened (batch, hidden)

    # ----- decode batch & hidden indices -----
    batch = pid // HIDDEN_DIM
    hid   = pid % HIDDEN_DIM

    # ----- offsets inside the tile -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----- base pointers for this (batch, hidden) -----
    a_batch = batch * stride_ah + hid * stride_ad
    b_batch = batch * stride_bh + hid * stride_bd
    c_batch = batch * stride_ch + hid * stride_cd

    # pointers to the start of the tile
    a_ptrs = A_ptr + a_batch + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over K dimension -----
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        k_start = k * BLOCK_K
        k_idx   = k_start + tl.arange(0, BLOCK_K)
        mask_k  = k_idx < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)               # fp16·fp16 → fp32

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ----- write back -----
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor, hidden_dim: int) -> torch.Tensor:
    """
    Perform C = left @ rightᵀ for each hidden channel using Triton.

    left, right: tensors of shape (B, N, N, H) and dtype float16.
    Returns tensor of shape (B*H, N, N) (float16) – the hidden dimension is folded into batch.
    """
    bs, N, _, H = left.shape
    assert N == left.shape[2] and N == right.shape[2] and H == hidden_dim
    # Strides in element units
    stride_ah, stride_am, stride_ak, stride_ad = left.stride()
    stride_bh, stride_bi, stride_bk, stride_bd = right.stride()
    # For the transposed view we swap the i/k strides:
    #   Bᵀ is accessed as (k, n) where k maps to original i dimension,
    #   n maps to original k dimension.
    stride_bk = stride_bi            # original i stride becomes K stride (after transpose)
    stride_bn = stride_bk            # original k stride becomes N stride (after transpose)
    # Allocate output
    out = torch.empty((bs * H, N, N), dtype=left.dtype, device=left.device)

    # Output strides
    stride_ch, stride_cm, stride_cn = out.stride()

    # Heuristic tile sizes
    if N <= 128:
        BLOCK_M = BLOCK_N = 32
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128
    BLOCK_K = 32  # good for fp16 Tensor Cores

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * H,
    )
    _tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        N, N, N,
        stride_ah, stride_am, stride_ak, stride_ad,
        stride_bh, stride_bk, stride_bn, stride_bd,
        stride_ch, stride_cm, stride_cn, 1,          # stride_cd = 1 (hidden dim stride)
        HIDDEN_DIM=H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- 1️⃣ LayerNorm ----------
    norm_w = weights["norm.weight"].to(orig_dtype)
    norm_b = weights["norm.bias"].to(orig_dtype)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for the compute‑heavy path
    x = x.to(torch.float16)

    # ---------- 2️⃣ Load linear weights (fp16) ----------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ---------- 3️⃣ Linear projections ----------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ---------- 4️⃣ Optional mask ----------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ---------- 5️⃣ Gates ----------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- 6️⃣ Core TriMul via Triton ----------
    # left/right shape: (B, N, N, H)
    out_flat = _batched_matmul_triton(left, right, hidden_dim)   # (B*H, N, N)

    # ---------- 7️⃣ Reshape back to (B, N, N, H) ----------
    bs, N, _, _ = left.shape
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ---------- 8️⃣ Output LayerNorm ----------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ---------- 9️⃣ Apply output gate ----------
    out = out * out_gate

    # ---------- 🔟 Final linear projection ----------
    out = F.linear(out, to_out_w)    # (B, N, N, C)

    # Cast back to original dtype if needed
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)
    return out
