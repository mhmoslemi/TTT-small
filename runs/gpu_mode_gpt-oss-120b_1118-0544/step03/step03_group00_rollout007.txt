"""
TriMul (outgoing) forward pass with Triton fusion.

Algorithm
---------
1. Layer‑norm the 4‑D input (B,N,N,C) using provided LN parameters.
2. Cast to fp16 for compute‑heavy ops.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are performed with fp16 weights (torch.nn.functional.linear).
4. Fuse mask‑application, gate‑multiplication and optional broadcasting over the
   hidden dimension with a Triton kernel (apply_gate_mask_kernel).  This avoids
   extra memory passes compared to separate PyTorch elementwise ops.
5. Core O(N³·H) reduction is a batched matrix multiplication:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented by reshaping to (B*H, N, N) and calling torch.bmm, which uses
   cuBLAS Tensor‑cores.
6. Apply a second LayerNorm, multiply by out_gate and project back to the
   original channel dimension with a final linear.
7. Cast the result back to the input dtype.

The kernel only handles the elementwise mask‑gate fusion; the heavy matrix
multiplication is delegated to cuBLAS for optimal performance.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: (proj * gate * mask) – mask broadcast over hidden_dim
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,          # [total_elems] proj values (fp16)
    gate_ptr,          # [total_elems] gate values (fp16)
    mask_ptr,          # [mask_elems] mask values (fp16) – broadcast over hidden_dim
    out_ptr,           # [total_elems] output (fp16)
    total_elems,       # int: total number of elements in proj/gate (B*N*N*H)
    hidden_dim,        # int: H (used to map to mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds
    mask = offs < total_elems

    # Index into the mask (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=mask)


def apply_gate_and_mask(proj: torch.Tensor,
                        gate: torch.Tensor,
                        mask: torch.Tensor | None,
                        block_size: int = 8192) -> torch.Tensor:
    """
    Elementwise multiplication of projection, gate and (optional) mask.

    proj, gate : (B, N, N, H)  – fp16
    mask       : (B, N, N)     – fp16 (0/1) or None
    Returns    : (B, N, N, H)  – fp16
    """
    if mask is None:
        # No mask → just elementwise multiply
        return proj * gate

    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    # flatten mask to 1‑D (contiguous)
    mask_flat = mask.contiguous().view(-1)

    grid = (triton.cdiv(total, block_size),)

    apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C]
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (fp32)
            - config       : dict with keys
                * "dim" (int)          – input channel dimension C
                * "hidden_dim" (int)   – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all relevant weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, C) -> (B, N, N, H)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (same shape)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Fuse mask and gate (Triton kernel) if mask is present
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM (out = left @ right^T)
    #    Shapes: left/right -> (B, N, N, H)
    #    Reshape to (B*H, N, N) and use torch.bmm.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N) without extra copy – permute only changes strides
    left_perm = left.permute(0, 3, 1, 2)
    right_perm = right.permute(0, 3, 1, 2)

    # Make them contiguous for the batched matmul
    left_flat = left_perm.contiguous().view(B * H, N, N)
    right_flat = right_perm.contiguous().view(B * H, N, N)

    # Batched matrix multiplication (Tensor‑core friendly)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # (B, N, N, C)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
