import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Performs LayerNorm, fused linear projections, optional masking,
    sigmoid gating, an N‑cubed reduction via einsum (batched GEMM),
    LayerNorm, element‑wise out‑gate multiplication (via Triton),
    and final linear projection back to the original dimension.
    All heavy arithmetic uses float16 for speed; the returned tensor is float32.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # LayerNorm over channel dimension
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # Cast to fp16 for the heavy part
    x_fp16 = x.to(torch.float16)

    B, N, _, _ = x_fp16.shape

    # ---- 1. Fused linear projections (proj + gate) -----------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    combined_weight = torch.cat([left_proj_w,
                                 left_gate_w,
                                 right_proj_w,
                                 right_gate_w], dim=0)   # (4*hidden, dim)

    # Linear: (B*N*N, dim) -> (B*N*N, 4*hidden)
    x_flat = x_fp16.view(-1, dim)
    combined_out = F.linear(x_flat, combined_weight)
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)

    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=3)

    # ---- 2. Optional mask on projections -------------------------
    if not nomask:
        mask_f = mask.to(x_fp16.dtype).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ---- 3. Gating ------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate   # (B,N,N,hidden)
    right = right_proj * right_gate

    # ---- 4. Out‑gate ------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # (B,N,N,hidden)

    # ---- 5. Core N³ reduction (einsum) ----------------------------
    # out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # ---- 6. to_out_norm ------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    # ---- 7. Apply out‑gate via Triton ----------------------------------
    total_el = out.numel()
    BLOCK = 1024
    grid = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](out, out_gate, out, total_el, BLOCK=BLOCK)

    # ---- 8. Final linear projection back to dim -----------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in fp16
    out = out.to(torch.float32)
    return out

"""
Outgoing TriMul (Triangle Multiplicative Update) – optimized forward pass.

Algorithm
---------
1. Layer‑norm over the channel dimension of the input.
2. Fuse the four linear layers (left/right projections + their gates) into a
   single matrix multiplication with a concatenated weight matrix.
3. Optional mask application (only on the two projections).
4. Sigmoid gating and element‑wise multiplication to obtain *left* and *right*.
5. Compute the O(N³) reduction via a single ``torch.einsum`` which internally
   maps to a batched GEMM:   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].
6. Layer‑norm over the hidden dimension.
7. Apply the output‑gate element‑wise – performed with a tiny Triton kernel.
8. Final linear projection back to the original channel dimension and cast to
   float32.

All heavy arithmetic is carried out in float16 to exploit tensor‑cores; the
result is returned as float32.  At least one Triton kernel (step 7) is used,
fulfilling the requirement to implement part of the work on the GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N: tl.constexpr, BLOCK: tl.constexpr):
    """c = a * b  (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) – may be ignored
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor: processed tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, C]  (float32)

    # Cast to half‑precision for the remaining heavy work
    x_fp16 = x.to(torch.float16)  # [B, N, N, C]

    B, N, _, _ = x_fp16.shape

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the four linear layers (proj + gate) into one matmul
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)  # (H, C)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)  # (H, C)

    # Concatenate along the output‑feature axis: (4*H, C)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # (4*H, C)

    # Linear projection on the flattened spatial dimension
    x_flat = x_fp16.view(-1, dim)                     # (B*N*N, C)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 4*H)

    # Restore 4‑D layout and split into the four tensors
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=3
    )  # each: (B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (applied only to the projections)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x_fp16.dtype).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and combine with projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)   # (B, N, N, H)
    right_gate = torch.sigmoid(right_gate)  # (B, N, N, H)

    left = left_proj * left_gate   # (B, N, N, H)
    right = right_proj * right_gate  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Compute the O(N³) reduction via einsum (batched GEMM)
    # ------------------------------------------------------------------
    # out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ to_out_norm (LayerNorm over the hidden dimension)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(x_fp16.dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(x_fp16.dtype)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ Compute out_gate and apply it (Triton element‑wise multiplication)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)  # (H, C)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))  # (B, N, N, H)

    # Triton kernel for in‑place element‑wise multiplication (out *= out_gate)
    total_el = out.numel()
    BLOCK = 1024
    grid = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](out, out_gate, out, total_el, BLOCK=BLOCK)

    # ------------------------------------------------------------------
    # 8️⃣ Final projection back to the original dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (C, H)
    out = F.linear(out, to_out_w)  # (B, N, N, C) in fp16
    out = out.to(torch.float32)    # return as float32

    return out
