"""
TriMul (outgoing) forward pass – fused implementation.

The kernel follows the reference PyTorch module but aggressively fuses
operations to keep the Python overhead minimal:

1. LayerNorm on the input channels (float32 → fp16).
2. A single linear projection produces the four tensors needed for the
   left/right projections and their gates.
3. Optional masking is applied only to the left/right projections.
4. Gating (sigmoid) and element‑wise multiplication are done in fp16.
5. The O(N³) reduction is a batched half‑precision GEMM:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is exactly a batched matrix multiplication
   (left @ rightᵀ) for each hidden channel.
6. A second LayerNorm on the hidden dimension.
7. The out‑gate is computed and applied with a tiny Triton kernel.
8. Final linear projection back to the original dimension (fp16 → fp32).

Only the element‑wise out‑gate multiplication is performed with Triton,
fulfilling the requirement to include a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def eltwise_mul_kernel(
    a_ptr,   # out * out_gate input (fp16)
    b_ptr,   # out_gate (fp16)
    c_ptr,   # destination (fp16)
    N: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """c = a * b  (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float), may be None
            - weights: dict of model weights (float32 tensors)
            - config: dict containing:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # Load weights (convert to half for the heavy compute part)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias   = weights["norm.bias"].to(device)

    # Combine four linear layers (proj & gate for left/right) into one
    left_proj_w  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)

    out_gate_w   = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the input channel dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  float32

    # Cast to fp16 for the remainder
    x_fp16 = x.to(torch.float16)

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gating (four linear layers in one matmul)
    # ------------------------------------------------------------------
    # Build a single weight matrix of shape (4*H, C)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w],
        dim=0,
    )  # (4*hidden_dim, dim)

    # Linear mapping on the flattened tensor
    x_flat = x_fp16.view(-1, dim)                     # (B*N*N, dim)
    combined_out = F.linear(x_flat, combined_weight)  # (B*N*N, 4*H)

    # Restore shape and split
    combined_out = combined_out.view(B, N, N, 4 * hidden_dim)
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined_out, hidden_dim, dim=-1
    )  # each: (B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (applied only to projections)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N]; broadcast to hidden dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)       # (B,N,N,1)
        mask_f = mask_f.to(torch.float16)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and combine with projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)   # (B,N,N,H) fp16
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate           # (B,N,N,H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched half‑precision matmul.
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for easy view
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Fuse batch and hidden dimensions
    left_mat = left.view(B * hidden_dim, N, N)      # (B*H, N, N)
    right_mat = right.view(B * hidden_dim, N, N)    # (B*H, N, N)

    # Matmul with transposed right (rightᵀ) -> use .transpose
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 6️⃣ LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 7️⃣ Compute out_gate and apply it (Triton kernel)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))  # (B,N,N,H) fp16

    # Triton element‑wise multiplication: out *= out_gate
    total_elements = out.numel()
    BLOCK = 1024
    grid = ( (total_elements + BLOCK - 1) // BLOCK , )
    eltwise_mul_kernel[grid](
        out, out_gate, out,
        total_elements,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Final projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim) fp16
    return out.to(torch.float32)
