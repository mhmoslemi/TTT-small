def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_WARPS: tl.constexpr, NUM_STAGES: tl.constexpr
):
    pid0 = tl.program_id(0)  # batch*H
    pid1 = tl.program_id(1)  # i block
    pid2 = tl.program_id(2)  # j block

    b = pid0 // H
    h = pid0 % H

    # offsets start
    i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask
    mask_i = i < N
    mask_j = j < N

    # pointers
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # allocate accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loops over k
    for k in range(0, N, BLOCK_K):
        k_end = tl.minimum(N - k, BLOCK_K)
        # offsets for k
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile: But need right[j,k] where k is column index, j is row index.
        # Since we want right transposed: we load right tile with shape (BLOCK_N, BLOCK_K) where we treat k as inner dimension.
        right_ptrs = right_base + j[:, None] * stride_r_i + offs_k[None, :] * stride_r_k
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Compute dot product: left (M, K) dot right (N, K). We need sum_k left[i,k] * right[j,k].
        # This is like (left) * (right). To get that, we can use tl.dot with left and right transposed?
        # We can compute component-wise multiplication and sum across k:
        # For each (i,j): acc[i,j] += sum_k left[i,k] * right[j,k].
        # We can compute using broadcasting: left[:, None, :] * right[None, :, :] then sum over k.
        # However that's expensive.

        # But we can use matrix multiplication by transposing right: need right_T with shape (BLOCK_K, BLOCK_N).
        # So we want right_T = tl.load(... ) where offsets for k dimension first.
        # Alternative: use tl.dot with left (M,K) and right_T (K,N). Compute right_T using tile shape (BLOCK_K, BLOCK_N) where row index is k, column is j.

        # So compute pointer for right_T: base + k_idx * stride_r_k + j_idx * stride_r_i? For right_T element (k, j), we need coordinate k (row) and j (col) with stride of row = stride_r_k, col = stride_r_i.
        # So we can compute right_T_ptrs = right_base + offs_k[:, None] * stride_r_k + j[None, :] * stride_r_i.
        # Then we load shape (BLOCK_K, BLOCK_N).
        right_T_ptrs = right_base + offs_k[:, None] * stride_r_k + j[None, :] * stride_r_i

        # Now we can compute acc += tl.dot(left, right_T) (M,K) x (K,N) -> (M,N)
        acc += tl.dot(left, right_T)

    # Store result
    out_ptrs = out_base + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to fp16 for compute
    x_fp16 = x.to(torch.float16)

    # 2. Compute left_proj+left_gate via concatenated weight
    left_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0)  # shape (2*H, C)
    left_comb = torch.nn.functional.linear(x_fp16, left_weight.to(torch.float16))  # shape (B,N,N,2*H)
    left_proj = left_comb[..., :hidden_dim]   # (B,N,N,H)
    left_gate = left_comb[..., hidden_dim:].sigmoid()

    right_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0)

    left = left_proj * left_gate
    right = right_proj * right_gate

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the supplied LN parameters.
2. Cast to fp16 and compute linear projections + gating:
   - left = (x @ left_proj) * sigmoid(x @ left_gate)
   - right = (x @ right_proj) * sigmoid(x @ right_gate)
   - out_gate = sigmoid(x @ out_gate_weight)
3. (Optional) apply the binary mask to left/right.
4. Core O(N³·H) reduction: for each batch element * hidden channel we compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is implemented as a fused batched GEMM in Triton (each (b,h) pair is a
   separate program).  Tiling with BLOCK_M/N/K = 64/64/64 and using tf32/FP16
   arithmetic yields high throughput.
5. Layer‑norm the hidden result (per‑channel) and multiply by out_gate
   (tiny Triton element‑wise kernel).
6. Final linear projection back to the original C dimension.
7. Cast back to the input dtype.

All heavy arithmetic (step 4) is performed in Triton; the remaining
operations use PyTorch but stay on the GPU to minimise data movement.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,               # *float16
    right_ptr,              # *float16
    out_ptr,                # *float16
    B, N, H,                # scalars
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid0 = tl.program_id(0)   # batch*hidden index
    pid1 = tl.program_id(1)   # i block
    pid2 = tl.program_id(2)   # j block

    # decode batch and hidden dimension from pid0
    b = pid0 // H
    h = pid0 % H

    # base offset for this (b,h) slice (adds batch and hidden strides)
    base = b * stride_l_b + h * stride_l_h   # same for left/right/out because layout is identical

    # block start indices
    i_start = pid1 * BLOCK_M
    j_start = pid2 * BLOCK_N

    # generate tile indices
    i_idx = i_start + tl.arange(0, BLOCK_M)
    j_idx = j_start + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / columns
    mask_i = i_idx < N
    mask_j = j_idx < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop (tiled)
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        k_idx = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # --------------------- load left tile (M×K) -------------------
        # left[b,i,k,h]   -> strides: i * stride_l_i  +  k * stride_l_k
        a_ptrs = left_ptr + base \
                 + i_idx[:, None] * stride_l_i \
                 + k_idx[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ------------------- load right tile transposed (K×N) -----------
        # we need right_T[k,j] = right[b,j,k,h]
        # strides for right_T: k * stride_r_k + j * stride_r_i
        b_ptrs = right_ptr + base \
                 + k_idx[:, None] * stride_r_k \
                 + j_idx[None, :] * stride_r_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # ----------------------- dot product -------------------------
        acc += tl.dot(a, b)   # (M,K)·(K,N) → (M,N)

    # ------------------------------------------------------------------
    # Store the result (convert back to fp16)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + base \
               + i_idx[:, None] * stride_o_i \
               + j_idx[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


@triton.jit
def _elemwise_mul_kernel(
    a_ptr, b_ptr, c_ptr, total,
    BLOCK_SIZE: tl.constexpr,
):
    """c = a * b  (all fp16, contiguous)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch Triton element‑wise multiplication."""
    assert a.shape == b.shape
    total = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](
        a, b, out, total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Front‑end entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            mask: torch.Tensor of shape [B, N, N] (or None)
            weights: dict containing model weights
            config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections + gates (concatenated for fewer GEMMs)
    # ------------------------------------------------------------------
    # left_proj + left_gate
    left_w_cat = torch.cat([weights["left_proj.weight"],
                            weights["left_gate.weight"]], dim=0)   # (2*H, C)
    left_cat = torch.nn.functional.linear(x, left_w_cat.to(torch.float16))
    left_proj = left_cat[..., :hidden_dim]
    left_gate = left_cat[..., hidden_dim:].sigmoid()

    # right_proj + right_gate
    right_w_cat = torch.cat([weights["right_proj.weight"],
                             weights["right_gate.weight"]], dim=0)   # (2*H, C)
    right_cat = torch.nn.functional.linear(x, right_w_cat.to(torch.float16))
    right_proj = right_cat[..., :hidden_dim]
    right_gate = right_cat[..., hidden_dim:].sigmoid()

    # out_gate (scalar per channel)
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # ------------------------------------------------------------------
    # 4) Optional binary mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 5) Fuse gates into projected tensors
    # ------------------------------------------------------------------
    left = left_proj * left_gate    # (B,N,N,H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core O(N³·H) reduction via a Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=left.device)

    # Strides for the three tensors (all are contiguous in the last dim)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tiling parameters – chosen to give good occupancy on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                     # batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,              # i blocks
        (N + BLOCK_N - 1) // BLOCK_N,              # j blocks
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 7) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate – tiny Triton element‑wise kernel
    # ------------------------------------------------------------------
    out = _elemwise_mul(out, out_gate)

    # ------------------------------------------------------------------
    # 9) Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 10) Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
