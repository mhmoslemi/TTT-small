"""
custom_kernel: Outgoing Triangle Multiplicative Update (TriMul) accelerated with Triton.

Algorithm
---------
1. LayerNorm over the channel dimension C.
2. Linear projections (left, right) and three gating linear layers (left_gate,
   right_gate, out_gate) are computed in FP32.  Mask (if provided) is applied
   to the projections before gating.
3. Projections are gated element‑wise and then cast to FP16.
4. The core O(N³·H) reduction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM C = A @ Bᵀ using a custom Triton kernel.
   The hidden dimension H is folded into the batch dimension, i.e. the kernel
   sees tensors of shape (B*H, N, N) and computes C = A @ Bᵀ for each batch.
5. The result is reshaped back to (B,N,N,H) and cast to FP32.
6. A second LayerNorm over the hidden dimension H, multiplication by the
   output gate, and a final linear projection back to the original channel
   dimension C are applied (all in FP32).
7. The output is cast to the dtype of the input tensor.

The Triton kernel uses block‑wise tiling with fp16 loads, fp32 accumulation,
and `tl.dot` (Tensor‑core friendly).  Block sizes are chosen heuristically
based on the sequence length N, and the kernel launch uses an appropriate
number of warps.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dims (all == seq_len)
    stride_ah, stride_am, stride_ak,   # A: batch, row, col
    stride_bh, stride_bk, stride_bn,   # B: batch, k (inner dim), n (output col)
    stride_ch, stride_cm, stride_cn,   # C: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]   (i.e. A @ Bᵀ)
    All tensors are stored in fp16; accumulation uses fp32.
    """
    pid_b = tl.program_id(2)      # batch index (B * H)
    pid_m = tl.program_id(0)      # block row index
    pid_n = tl.program_id(1)      # block col index

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # Pointers for the current tile (K dimension will be tiled)
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        # K‑mask for the current tile
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Wrapper for the Triton batched matmul kernel.
    A, B: fp16 tensors of shape (batch, N, N)
    Returns C of same shape and dtype.
    """
    assert A.shape == B.shape, "A and B must have identical shapes"
    batch, N, K = A.shape
    assert N == K, "TriMul requires square sequence dimension"

    # Heuristic block sizes based on N
    if N <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
        num_warps = 2
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:  # up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    C = torch.empty_like(A)

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(2), B.stride(1),  # note: B is accessed as (batch, K, N)
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # LayerNorm over channel dimension C (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor.to(torch.float32),
        (C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, C)  float32

    # --------------------------------------------------------------
    # Linear projections (FP32)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # (B, N, N, H)
    right = F.linear(x, weights["right_proj.weight"])  # (B, N, N, H)

    # --------------------------------------------------------------
    # Optional binary mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float32).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # Gates (FP32)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # (B, N, N, H)

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Cast projections to FP16 for the heavy O(N³·H) reduction
    # --------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # --------------------------------------------------------------
    # Fold hidden dimension into batch and launch Triton GEMM kernel
    # --------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matmul: out_flat = left @ right.T   (shape B*H, N, N)
    out_flat = _batched_matmul_triton(left, right)  # fp16

    # --------------------------------------------------------------
    # Reshape back to (B, N, N, H) and cast to FP32 for subsequent ops
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # (B, N, N, H)

    # --------------------------------------------------------------
    # Second LayerNorm over hidden dimension H (FP32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # Apply output gate and final linear projection back to C (FP32)
    # --------------------------------------------------------------
    out = out * out_gate  # out_gate is still FP32
    out = F.linear(out, weights["to_out.weight"])  # (B, N, N, C)

    # --------------------------------------------------------------
    # Cast back to original dtype (if needed)
    # --------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
