"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Fuse the left/right linear projections with their gates:
   * Compute (proj, gate) pairs together using a single `linear` call
     for left and right respectively.
3. (Optional) apply the pairwise mask to the projected values.
4. Apply sigmoid gates and multiply the gates with the projected tensors.
5. Compute the O(N³) reduction:
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication of shape
   (B·hidden, N, N) @ (B·hidden, N, N)ᵀ which is performed by a
   custom Triton kernel (`batched_matmul_kernel`).  The kernel uses
   FP16 inputs, a FP32 accumulator and 64×64×64 tiling, fully exploiting
   tensor‑cores.
6. Layer‑norm the intermediate result over the hidden dimension.
7. Apply the output gate (element‑wise multiplication).
8. Final linear projection back to the original channel dimension.
9. Return the result in `float32`.

Only the heavy O(N³) reduction is implemented in Triton; all other
operations use PyTorch (high‑throughput cuBLAS / CUDA kernels).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ B  (A: (B,M,K), B: (B,K,N))
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # batch index

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers for this block (k = 0)
    a_ptrs = (
        a_ptr
        + pid_b * stride_a_batch
        + offs_m[:, None] * stride_a_m
        + offs_k[None, :] * stride_a_k
    )
    b_ptrs = (
        b_ptr
        + pid_b * stride_b_batch
        + offs_k[:, None] * stride_b_k
        + offs_n[None, :] * stride_b_n
    )
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        # Masks for out‑of‑bounds loads
        mask_a = (offs_m < M)[:, None] & (k + offs_k < K)[None, :]
        mask_b = (k + offs_k < K)[:, None] & (offs_n < N)[None, :]

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)  # FP16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)  # FP16

        acc += tl.dot(a, b)  # FP16·FP16 → FP32

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Write back
    c = acc.to(tl.float16)
    mask_c = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, c, mask=mask_c)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) – may be ignored
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # Cast weights to the appropriate dtype / device
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    # Linear weights – use FP16 for compute
    left_proj_weight = weights["left_proj.weight"].to(device, dtype=torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device, dtype=torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device, dtype=torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device, dtype=torch.float16)

    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate linear layers (half‑precision)
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left_combined = F.linear(x_fp16, left_combined_weight)  # [B, N, N, 2*H]

    right_combined_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0)  # (2*H, C)
    right_combined = F.linear(x_fp16, right_combined_weight)  # [B, N, N, 2*H]
    right_proj, right_gate = torch.split(right_combined, hidden_dim, dim=-1)

    # out gate (single linear)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (applied only to the projections)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype=x_fp16.dtype).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate   # (B, N, N, H)  fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a Triton batched matrix multiply.
    # ------------------------------------------------------------------
    # Transpose the second N‑axis of `right` to match the matmul contract.
    right_t = right.permute(0, 2, 1, 3).contiguous()  # (B, N, N, H) with (k,i) swapped

    # Bring hidden dimension to the batch axis for a contiguous layout.
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_mat = right_t.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_mat.shape
    batch = B * H

    # Reshape to (batch, N, N) where each batch element is one (M×K) matrix.
    left_mat = left_mat.reshape(batch, N, N)
    right_mat = right_mat.reshape(batch, N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Strides in element units
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_k = right_mat.stride(1)
    stride_b_n = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), batch),
    )
    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,   # M
        N,   # N
        N,   # K
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to (B, H, N, N) and permute to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B,N,N,H)  fp16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate  # fp16 * fp16 → fp16

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # (B,N,N,dim)  fp16

    # Return float32 as required
    return out.to(torch.float32)
```
