Goal: Avoid reading X multiple times (twice) for mean and var. Use a two-pass approach: first compute sum (for mean) across C, then compute sum of squares (or compute variance with second pass). Since C is small, we can compute both in one pass by storing values in registers in a loop.

Pseudo:

```
@triton.jit
def layer_norm_kernel(
    X_ptr, out_ptr, weight_ptr, bias_ptr,
    B, N, C,
    stride_xb, stride_xn, stride_xc,
    stride_ob, stride_on, stride_oc,
    BLOCK_C: tl.constexpr,
    EPS: tl.constexpr = 1e-5,
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total: return
    # convert pid to coordinates (b,i,j)
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # initialize sum and sumsq
    sum = 0.0
    sumsq = 0.0

    # iterate over C dimension
    for offset in range(0, tl.cdiv(C, BLOCK_C)):
        offs = offset * BLOCK_C + tl.arange(0, BLOCK_C)
        mask = offs < C
        x = tl.load(X_ptr + b*stride_xb + i*stride_xn + j*stride_xc + offs, mask=mask, other=0.0)
        sum += tl.sum(x, axis=0)
        sumsq += tl.sum(x * x, axis=0)

    mean = sum / C
    var = sumsq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # second pass: normalize and apply weight & bias
    for offset in range(0, tl.cdiv(C, BLOCK_C)):
        offs = offset * BLOCK_C + tl.arange(0, BLOCK_C)
        mask = offs < C
        x = tl.load(X_ptr + b*stride_xb + i*stride_xn + j*stride_xc + offs, mask=mask, other=0.0)
        w = tl.load(weight_ptr + offs, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr + offs, mask=mask, other=0.0)
        y = (x - mean) * inv_std * w + b_
        tl.store(out_ptr + b*stride_ob + i*stride_on + j*stride_oc + offs, y, mask=mask)
```

@triton.jit
def layer_norm_kernel(
    X_ptr, Out_ptr, weight_ptr, bias_ptr,
    B, N, C,
    stride_xb, stride_xn, stride_xc,
    stride_ob, stride_on, stride_oc,
    EPS: tl.constexpr = 1e-5,
    BLOCK_C: tl.constexpr = 128,
):
    pid = tl.program_id(0)
    # total number of (b,i,j) entries
    total = B * N * N
    if pid >= total:
        return

    # decode pid to (b,i,j)
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    base_x = X_ptr + b * stride_xb + i * stride_xn + j * stride_xc
    # first pass: compute mean and variance
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    # loop over channel blocks
    for c_off in range(0, tl.cdiv(C, BLOCK_C)):
        offs = c_off * BLOCK_C + tl.arange(0, BLOCK_C)
        mask = offs < C
        x = tl.load(base_x + offs, mask=mask, other=0.0)
        # cast to float32 for reduction
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=0)
        sum_sq += tl.sum(x_f * x_f, axis=0)

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # second pass: write normalized values
    for c_off in range(0, tl.cdiv(C, BLOCK_C)):
        offs = c_off * BLOCK_C + tl.arange(0, BLOCK_C)
        mask = offs < C
        x = tl.load(base_x + offs, mask=mask, other=0.0)
        w = tl.load(weight_ptr + offs, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr + offs, mask=mask, other=0.0)
        y = ((x - mean) * inv_std) * w + b_
        tl.store(Out_ptr + b * stride_ob + i * stride_on + j * stride_oc + offs, y, mask=mask)
```

"""
TriMul (outgoing) forward pass with Triton‑fused LayerNorm and mask‑gate.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using a custom Triton kernel.
   The kernel reduces over the channel dimension (C) in FP32 for numeric stability.
2. Cast to FP16 for the compute‑heavy path.
3. Fuse the linear projection and its gate for the left & right branches:
   - Concatenate the projection and gate weight matrices (2*H × C) and call a
     single torch.nn.functional.linear.
   - Split the result into projection and gate tensors and apply sigmoid to the gate.
4. Fuse mask multiplication, projection and gate using a Triton elementwise kernel.
5. Core O(N³·H) reduction: compute
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   via a vectorised torch.einsum (equivalent to a batched GEMM per hidden channel).
6. Apply a second LayerNorm (FP16) using the same Triton kernel (hidden dim H).
7. Multiply by the out‑gate (sigmoid) and apply the final linear projection back
   to the original channel dimension.
8. Cast the result back to the original dtype (normally float32).

Only the LayerNorm and mask‑gate steps are implemented in Triton; the heavy
matrix‑multiply is delegated to torch which uses highly‑optimized cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm over the last dimension (C)
# ----------------------------------------------------------------------
@triton.jit
def layer_norm_kernel(
    X_ptr,               # input    [(B,N,N,C), fp16]
    Out_ptr,             # output   [(B,N,N,C), fp16]
    Weight_ptr,          # gamma    [(C,), fp16]
    Bias_ptr,            # beta     [(C,), fp16]
    B, N, C,             # static sizes
    stride_xb, stride_xn, stride_xc,   # strides of X
    stride_ob, stride_on, stride_oc,   # strides of Out
    EPS: tl.constexpr,   # epsilon for variance
    BLOCK_C: tl.constexpr,  # tile size over channels
):
    pid = tl.program_id(0)                      # one program per (b,i,j)
    total = B * N * N
    if pid >= total:
        return

    # decode pid -> (b,i,j)
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base pointers for this (b,i,j)
    x_base = X_ptr + b * stride_xb + i * stride_xn + j * stride_xc
    out_base = Out_ptr + b * stride_ob + i * stride_on + j * stride_oc

    # ---------- first pass: compute mean & variance ----------
    sum_f = tl.float32(0.0)
    sum_sq_f = tl.float32(0.0)
    offset = 0
    while offset < C:
        offs = offset + tl.arange(0, BLOCK_C)          # (BLOCK_C,)
        mask = offs < C
        x = tl.load(x_base + offs, mask=mask, other=0.0)  # fp16
        x_f = x.to(tl.float32)
        sum_f += tl.sum(x_f, axis=0)
        sum_sq_f += tl.sum(x_f * x_f, axis=0)
        offset += BLOCK_C

    mean = sum_f / C
    var = sum_sq_f / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # ---------- second pass: write normalized values ----------
    offset = 0
    while offset < C:
        offs = offset + tl.arange(0, BLOCK_C)
        mask = offs < C
        x = tl.load(x_base + offs, mask=mask, other=0.0)
        w = tl.load(Weight_ptr + offs, mask=mask, other=0.0)
        b_ = tl.load(Bias_ptr + offs, mask=mask, other=0.0)
        y = ((x - mean) * inv_std) * w + b_
        tl.store(out_base + offs, y, mask=mask)
        offset += BLOCK_C


def layer_norm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5) -> torch.Tensor:
    """Layer‑norm over the last dimension using Triton."""
    B, N, N2, C = x.shape
    assert N == N2, "input must be square in the two sequence dimensions"
    out = torch.empty_like(x)

    grid = (B * N * N,)
    layer_norm_kernel[grid](
        x,
        out,
        weight,
        bias,
        B, N, C,
        x.stride(0), x.stride(1), x.stride(3),
        out.stride(0), out.stride(1), out.stride(3),
        EPS=eps,
        BLOCK_C=128,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: elementwise (proj * gate * mask) on a flattened tensor
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,          # [(B,N,N,H), fp16]
    gate_ptr,          # [(B,N,N,H), fp16]
    mask_ptr,          # [(B,N,N),    fp16] – broadcast over H
    out_ptr,           # [(B,N,N,H), fp16]
    total_elems,       # int = B*N*N*H
    hidden_dim,        # int = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load values (out‑of‑bounds entries get zero / 1.0)
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    # broadcast mask over the hidden dimension
    mask_idx = offs // hidden_dim
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * mval
    tl.store(out_ptr + offs, out, mask=mask)


def apply_gate_and_mask(proj: torch.Tensor,
                        gate: torch.Tensor,
                        mask: torch.Tensor,
                        block_size: int = 1024) -> torch.Tensor:
    """
    Fuse elementwise multiplication: proj * gate * mask.
    proj, gate : (B,N,N,H) fp16
    mask       : (B,N,N)  fp16 (0/1)
    Returns a tensor of shape (B,N,N,H) fp16.
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)

    apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
        num_warps=4,
    )
    return out

# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """

    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # cast weights to the compute dtype (fp16) and move to device if needed
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    norm_weight = to_fp16(weights["norm.weight"])
    norm_bias   = to_fp16(weights["norm.bias"])

    left_proj_weight  = to_fp16(weights["left_proj.weight"])
    right_proj_weight = to_fp16(weights["right_proj.weight"])
    left_gate_weight  = to_fp16(weights["left_gate.weight"])
    right_gate_weight = to_fp16(weights["right_gate.weight"])
    out_gate_weight   = to_fp16(weights["out_gate.weight"])

    to_out_norm_weight = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_bias   = to_fp16(weights["to_out_norm.bias"])
    to_out_weight      = to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (Triton)
    # ------------------------------------------------------------------
    # work in fp16 for the rest of the pipeline
    x_fp16 = input_tensor.to(torch.float16)
    x = layer_norm_triton(x_fp16, norm_weight, norm_bias)  # (B,N,N,C) fp16

    # ------------------------------------------------------------------
    # 2) Fuse projection + gate for left and right branches
    # ------------------------------------------------------------------
    # concatenate weights: [proj ; gate]  ⇒ shape (2*H, C)
    left_comb_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0)   # (2*H, C)
    right_comb_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0)  # (2*H, C)

    # Linear projection (single matmul per side)
    left_comb = torch.nn.functional.linear(x, left_comb_weight)   # (B,N,N,2*H)
    right_comb = torch.nn.functional.linear(x, right_comb_weight) # (B,N,N,2*H)

    # split into projection and gate streams
    left_proj = left_comb[..., :hidden_dim]                      # (B,N,N,H)
    left_gate = left_comb[..., hidden_dim:].sigmoid()           # (B,N,N,H)

    right_proj = right_comb[..., :hidden_dim]                    # (B,N,N,H)
    right_gate = right_comb[..., hidden_dim:].sigmoid()         # (B,N,N,H)

    # out‑gate (separate linear)
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3) Apply mask and gates (Triton) – if mask is present and not disabled
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(
            left_proj.contiguous(),
            left_gate.contiguous(),
            mask_fp16,
            block_size=4096,
        )
        right = apply_gate_and_mask(
            right_proj.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
            block_size=4096,
        )
    else:
        # simple element‑wise multiplication when no mask is used
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4) Core reduction: Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Using torch.einsum – it maps to a series of batched GEMMs internally.
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Output LayerNorm (Triton) – on the hidden dimension
    # ------------------------------------------------------------------
    out = layer_norm_triton(
        out,
        to_out_norm_weight,
        to_out_norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 6) Multiply by out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)          # (B,N,N,C)

    # ------------------------------------------------------------------
    # 7) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
