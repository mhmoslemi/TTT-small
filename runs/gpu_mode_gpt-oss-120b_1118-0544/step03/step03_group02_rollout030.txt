def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    B, M, N, K,
    stride_am, stride_ak, stride_ab,   # for a: batch, row, col?
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    m_mask = offs_m < M
    n_mask = offs_n < N

    # Pointers offsets into a and b:
    a_ptrs = a_ptr + batch_id * stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=m_mask[:, None] & (tl.arange(0, BLOCK_K) < K - k*BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=n_mask[None, :] & (tl.arange(0, BLOCK_K) < K - k*BLOCK_K), other=0.0)

        acc += tl.dot(a, b) # dot performs sum over K axis

        # increment pointers for next k tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    c = acc.to(tl.float16)
    c_ptrs = c_ptr + batch_id * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start indices

```
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_abl, stride_aim, stride_aik,
    stride_bbl, stride_bik, stride_bjn,
    stride_cbl, stride_cim, stride_cjn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for within-range
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to start of batch
    a_batch_ptr = a_ptr + batch_id * stride_abl
    b_batch_ptr = b_ptr + batch_id * stride_bbl
    c_batch_ptr = c_ptr + batch_id * stride_cbl

    # tile pointers
    a_ptrs = a_batch_ptr + offs_m[:, None] * stride_aim + tl.arange(0, BLOCK_K)[None, :] * stride_aik
    b_ptrs = b_batch_ptr + tl.arange(0, BLOCK_K)[:, None] * stride_bik + offs_n[None, :] * stride_bjn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute which K indices are valid
        cur_k = k * BLOCK_K
        k_valid = (cur_k + tl.arange(0, BLOCK_K)) < K

        # load a and b
        a = tl.load(a_ptrs, mask=mask_m[:, None] & k_valid[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=k_valid[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_aik
        b_ptrs += BLOCK_K * stride_bik

    # write back
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cim + offs_n[None, :] * stride_cjn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton accelerated version.

Algorithm
---------
1. Layer‑norm the input (float32 → float16) and apply four linear
   projections (left/right) together with their gates.
2. Optional mask is applied to the projected tensors.
3. Multiply each projection by its gate (sigmoid) → tensors L and R.
4. Compute the core O(N³) reduction:
        out[b,i,j,d] = Σₖ L[b,i,k,d] * R[b,j,k,d]
   as a batched matrix multiplication using a custom Triton GEMM
   kernel (batch dimension = batch * hidden_dim).
5. Layer‑norm over the hidden dimension, apply the out‑gate, and a final
   linear projection back to the original feature dimension.
All heavy arithmetic (the batched GEMM) runs in fp16 on Tensor‑cores.
The final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,      # a: (batch, M, K)
    stride_bb, stride_bk, stride_bn,      # b: (batch, K, N)
    stride_cb, stride_cm, stride_cn,      # c: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B (A: M×K, B: K×N) for each batch."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ----- tile offsets -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----- compute base pointers for this batch -----
    a_batch = a_ptr + batch_id * stride_ab
    b_batch = b_ptr + batch_id * stride_bb
    c_batch = c_ptr + batch_id * stride_cb

    # ----- pointers to the current tile -----
    a_ptrs = a_batch + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over K dimension in tiles -----
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        k_valid = (cur_k + tl.arange(0, BLOCK_K)) < K

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & k_valid[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=k_valid[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)               # (BLOCK_M, BLOCK_N) ← fp32 accumulation

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ----- store result -----
    c = acc.to(tl.float16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config["nomask"] is True
            - weights: dict of model weight tensors (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension (float32 → fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias = weights["norm.bias"].to(device).to(torch.float16)

    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    x = x.to(torch.float16)                     # [B, N, N, C] in fp16

    # ------------------------------------------------------------------
    # 2) Linear projections + gates (four separate Linear layers)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    # projections
    left_proj = F.linear(x, left_proj_weight)          # [B, N, N, hidden]
    right_proj = F.linear(x, right_proj_weight)

    # gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # optional mask on the projected values (mask shape B,N,N → B,N,N,1)
    if not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # broadcastable
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # combine projection with its gate
    left = left_proj * left_gate          # [B,N,N,hidden]   fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 3) Core O(N³) reduction via batched GEMM (Triton)
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # reshape to (B, hidden, N, N) and create the transposed RHS
    left = left.permute(0, 3, 1, 2).contiguous()           # (B, hidden, N, N)
    right_T = right.permute(0, 3, 2, 1).contiguous()      # (B, hidden, N, N) where K dimension is first

    # merge batch and hidden into a single batch dimension for the kernel
    batch_hidden = B * hidden_dim
    left_mat = left.view(batch_hidden, N, N)
    right_mat = right_T.view(batch_hidden, N, N)

    out_mat = torch.empty_like(left_mat)   # fp16, shape (B*hidden, N, N)

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    num_m_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_n_blocks = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch_hidden, num_m_blocks, num_n_blocks)

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_ab=left_mat.stride(0),
        stride_am=left_mat.stride(1),
        stride_ak=left_mat.stride(2),
        stride_bb=right_mat.stride(0),
        stride_bk=right_mat.stride(1),
        stride_bn=right_mat.stride(2),
        stride_cb=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # 4) Reshape back to (B, N, N, hidden)
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 5) Post‑processing: LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # fp16

    # apply out‑gate (element‑wise)
    out = out * out_gate

    # final projection back to original dimension
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)  # (dim, hidden)
    out = F.linear(out, to_out_weight)   # fp16, shape (B, N, N, dim)

    # ------------------------------------------------------------------
    # 6) Return in float32 as required
    # ------------------------------------------------------------------
    return out.to(torch.float32)
