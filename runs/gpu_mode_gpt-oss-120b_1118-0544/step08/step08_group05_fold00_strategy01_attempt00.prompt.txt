```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

```python
"""
custom_kernel: Optimized outgoing Triangle Multiplicative Update (TriMul)
-----------------------------------------------------------------------
Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 (mixed precision) for compute‑intensive parts.
3. Linear projections (no bias) → `left_proj`, `right_proj`.
4. Gating projections → `left_gate`, `right_gate`, `out_gate` (sigmoid).
5. Fuse mask, projection and gate for both left/right tensors with a small
   Triton kernel (only when a mask is present and not ignored).  When no
   mask is used the element‑wise product is done in PyTorch.
6. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden slice `h`.
   We reshape to (B*H, N, N) and use `torch.bmm` (cuBLAS).
7. Apply output LayerNorm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
8. Cast the result back to the original dtype (usually float32).

The kernel in step 5 is mandatory to satisfy the “custom kernel” requirement.
All other heavy‑weight operations rely on highly‑optimized PyTorch (cuBLAS)
routines.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B,N,N,H)
    right_proj_ptr,  # fp16 * (B,N,N,H)
    left_gate_ptr,   # fp16 * (B,N,N,H)
    right_gate_ptr,  # fp16 * (B,N,N,H)
    mask_ptr,        # fp16 * (B,N,N)
    left_out_ptr,    # fp16 * (B,N,N,H)
    right_out_ptr,   # fp16 * (B,N,N,H)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard
    valid = offs < total_elems

    # each mask element repeats `hidden_dim` times in the flattened tensor
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    left_out = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gate and mask for both left and right tensors using Triton.

    All tensors must be fp16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)

    Returns:
        left, right : (B, N, N, H)  = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor of shape [B, N, N] (0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    # Cast LayerNorm parameters to compute dtype (we will use fp16 later)
    norm_w = weights["norm.weight"].to(dtype_in)
    norm_b = weights["norm.bias"].to(dtype_in)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # ------------------------------------------------------------------
    compute_dtype = torch.float16
    x = x.to(compute_dtype)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(compute_dtype)
    right_proj_w  = weights["right_proj.weight"].to(compute_dtype)
    left_gate_w   = weights["left_gate.weight"].to(compute_dtype)
    right_gate_w  = weights["right_gate.weight"].to(compute_dtype)
    out_gate_w    = weights["out_gate.weight"].to(compute_dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype)
    to_out_w      = weights["to_out.weight"].to(compute_dtype)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fusion via Triton if mask is used)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(compute_dtype).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
        )
    else:
        # No mask (or ignored) → simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    = left_h @ right_hᵀ for each hidden slice h
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
```
