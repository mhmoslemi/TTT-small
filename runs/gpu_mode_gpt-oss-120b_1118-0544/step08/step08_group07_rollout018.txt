def _proj_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems,
    hidden_dim,
    BLOCK_SIZE: tl.constexpr
):

    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # compute mask index
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask)

    right_out = rp * rg * m

"""
TriMul outgoing (AlphaFold3) forward pass.
- LayerNorm over the input channel dimension.
- Linear projections + sigmoid gating.
- Optional mask & gate fusion performed by a small Triton kernel.
- Core O(N³) reduction computed as a batched GEMM (torch.bmm) after reshaping.
- Output LayerNorm, out‑gate, final linear projection.
All compute‑intensive parts run in FP16; the final result is cast back to the
input dtype (typically float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16*   [B, N, N, H]
    right_proj_ptr,  # fp16*   [B, N, N, H]
    left_gate_ptr,   # fp16*   [B, N, N, H]
    right_gate_ptr,  # fp16*   [B, N, N, H]
    mask_ptr,        # fp16*   [B, N, N]
    left_out_ptr,    # fp16*   [B, N, N, H]
    right_out_ptr,   # fp16*   [B, N, N, H]
    total_elems,     # i64     total number of elements in the flattened [B,N,N,H] tensor
    hidden_dim,      # i64     H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gating and optional mask for left/right tensors."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # guard
    mask = offs < total_elems

    # each mask element repeats `hidden_dim` times in the flattened tensor
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # 1.0 for unmasked positions

    left_out  = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs,  left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)
