"""
TriMul (outgoing) forward pass implemented in Triton.
The heavy N³ reduction `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
is performed by a custom batched GEMM kernel (left @ rightᵀ) in FP16.
All other ops (LayerNorm, Linear projections, gating, mask handling) use
PyTorch for simplicity. The final output is returned in the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   where
       A : (batch, M, K)
       B : (batch, K, N)   (already transposed)
       C : (batch, M, N)
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask = mask_m[:, None] & mask_n[None, :]

    # base pointers for this batch
    a_batch = a_ptr + pid_batch * stride_ah
    b_batch = b_ptr + pid_batch * stride_bh
    c_batch = c_ptr + pid_batch * stride_ch

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # pointers
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load with masks
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        # dot‑product over K‑axis
        acc += tl.dot(a, b)

    # store result (down‑cast to fp16)
    c = acc.to(tl.float16)
    tl.store(c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
             c, mask=mask)

    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(torch.float16))
    )

    # ------------------------------------------------------------------
    # 4) Apply mask and gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but caller asked to ignore it -> treat as ones
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp16
            right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (left @ rightᵀ) using Triton.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # transpose the right operand (so we can use a standard GEMM kernel)
    right_T = right_flat.transpose(1, 2).contiguous()   # (B*H, N, N) -> (B*H, K, N)

    out_flat = torch.empty_like(left_flat)  # fp16 result

    # strides
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    stride_ah = left_flat.stride(0)

    stride_bk = right_T.stride(1)
    stride_bn = right_T.stride(2)
    stride_bh = right_T.stride(0)

    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    stride_ch = out_flat.stride(0)

    # block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_gemm_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_ah,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_cm,
        stride_cn,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
