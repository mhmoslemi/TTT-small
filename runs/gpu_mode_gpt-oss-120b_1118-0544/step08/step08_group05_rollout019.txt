@triton.jit
def mask_gate_fuse_kernel(
    left_proj_ptr,   # * (total elements)
    right_proj_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,        # (B,N,N) broadcasted across hidden dim
    left_out_ptr,
    right_out_ptr,
    total,           # total number of elements (B*N*N*H)
    hidden_dim,      # H
    BLOCK_SIZE: tl.constexpr
):

    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total

    # Compute mask index: each mask element repeats hidden_dim times
    # mask_idx = offs // hidden_dim
    # That is integer division.

    right_out = rp * rg * m

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # cast to fp16
    compute_dtype = torch.float16
    x = x.to(compute_dtype)

    # combine left and left_gate weights, right and right_gate.
    left_comb = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=1).to(compute_dtype)  # shape (dim, 2*hidden_dim)
    right_comb = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=1).to(compute_dtype)

    # linear
    left_concat = F.linear(x, left_comb)  # B,N,N,2*H
    right_concat = F.linear(x, right_comb)
    # split

    left_gate = torch.sigmoid(left_concat[..., hidden_dim:])
    right_proj = right_concat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_concat[..., hidden_dim:])

    # out_gate linear
    out_gate_w = weights["out_gate.weight"].to(compute_dtype)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

        right = right_proj * right_gate
        if mask is not None and not nomask:

            left = left * mask_fp16
            right = right * mask_fp16
    # core reduction

    # flatten to (B*H, N, N)
    left_flat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_flat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()

    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    out = F.linear(out, to_out_w)

    # cast back to original dtype
    out = out.to(input_tensor.dtype)

"""
custom_kernel: Optimized outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for compute‑intensive parts.
3. Fuse the projection and gate linear layers:
   - Concatenate (proj, gate) weights for the left and right branches,
     perform a single matmul per branch and split the result.
4. Compute the output gate (sigmoid of a linear projection).
5. If a mask is present, fuse mask, projection and gate for both left/right
   tensors with a small Triton kernel (otherwise multiply element‑wise).
6. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM for each hidden slice; we reshape to
   (B*H, N, N) and use torch.bmm (cuBLAS).
7. Apply output LayerNorm, multiply by `out_gate`, and a final linear
   projection back to `dim`.
8. Cast the result back to the original dtype.

The Triton kernel is used for step 5 (mask + gate fusion), satisfying the
requirement of having a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _mask_gate_fuse_kernel(
    left_proj_ptr,   # fp16 * (B,N,N,H)
    right_proj_ptr,  # fp16 * (B,N,N,H)
    left_gate_ptr,   # fp16 * (B,N,N,H)
    right_gate_ptr,  # fp16 * (B,N,N,H)
    mask_ptr,        # fp16 * (B,N,N)
    left_out_ptr,    # fp16 * (B,N,N,H)
    right_out_ptr,   # fp16 * (B,N,N,H)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # each mask element repeats `hidden_dim` times in the flattened tensor
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(left_out_ptr + offs, lp * lg * m, mask=mask)
    tl.store(right_out_ptr + offs, rp * rg * m, mask=mask)


def _apply_mask_gate(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gate and mask for left/right branches using Triton.

    All tensors must be fp16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)

    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _mask_gate_fuse_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(dtype_in)
    norm_b = weights["norm.bias"].to(dtype_in)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # ------------------------------------------------------------------
    compute_dtype = torch.float16
    x = x.to(compute_dtype)

    # ------------------------------------------------------------------
    # 3) Fuse projection & gate linear layers (left & right)
    # ------------------------------------------------------------------
    # Concatenate projection and gate weights for each side
    left_comb = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(compute_dtype)  # (2*H, dim)
    right_comb = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(compute_dtype)  # (2*H, dim)

    # Single matmul per side
    left_concat = F.linear(x, left_comb)   # (B,N,N,2*H)
    right_concat = F.linear(x, right_comb) # (B,N,N,2*H)

    # Split into projection and gate parts
    left_proj = left_concat[..., :hidden_dim]
    left_gate = torch.sigmoid(left_concat[..., hidden_dim:])
    right_proj = right_concat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_concat[..., hidden_dim:])

    # Output gate (separate linear)
    out_gate_w = weights["out_gate.weight"].to(compute_dtype)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4) Apply mask & gating (fusion via Triton if mask is used)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(compute_dtype).contiguous()
        left, right = _apply_mask_gate(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
        )
    else:
        # Simple element‑wise fusion; if a mask exists but nomask=True we ignore it
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # Multiply by mask (broadcasted over hidden dim)
            mask_fp16 = mask.to(compute_dtype).unsqueeze(-1)
            left = left * mask_fp16
            right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype)
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(compute_dtype)  # (dim, hidden_dim)
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 7) Cast back to the original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)
    return out
```
