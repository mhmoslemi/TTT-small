@triton.jit
def batched_matmul_kernel(A, B, C, M, N, K,
                          stride_aa, stride_ak,
                          stride_bk, stride_bn,
                          stride_ca, stride_cn,
                          batch_stride_a, batch_stride_b, batch_stride_c,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        # Load A and B with bounds checks
        a_ptrs = A + batch_id * batch_stride_a + offs_m[:, None] * stride_aa + offs_k[None, :] * stride_ak
        b_ptrs = B + batch_id * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = C + batch_id * batch_stride_c + offs_m[:, None] * stride_ca + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

"""
TriMul outgoing forward pass (AlphaFold3) accelerated with Triton.
- Layer‑norm on the input channels.
- FP16 linear projections and sigmoid gates.
- Optional mask applied element‑wise.
- Core N‑by‑N‑by‑N reduction implemented as a batched
  matrix‑multiplication kernel in Triton (left @ rightᵀ) for each
  hidden slice.
- Output layer‑norm, out‑gate and final linear projection.
The kernel operates on tensors of shape (B*H, N, N) and computes
C[b,i,j] = Σ_k A[b,i,k] * B[b,k,j] using fp16 arithmetic with
float32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C = A @ B where A,B,C are (batch, M, K), (batch, K, N), (batch, M, N)."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # row and column offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        # mask for out‑of‑bounds
        mask_a = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        mask_b = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        a_ptrs = A + batch_id * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = B + batch_id * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

    # write‑back
    mask_c = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    c_ptrs = C + batch_id * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (0/1) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(input_tensor.dtype)
    norm_bias = weights["norm.bias"].to(input_tensor.dtype)

    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gates (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Apply mask (if present) and fuse with gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp
        right = right_proj * right_gate * mask_fp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched matmul using a Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, N, N, hidden_dim

    # reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # right needs to be transposed on the last two dims (k <-> j)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)  # fp16 output

    # strides
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    batch_stride_a = left_flat.stride(0)

    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)
    batch_stride_b = right_flat.stride(0)

    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    batch_stride_c = out_flat.stride(0)

    # block sizes – tuned for typical B200 memory hierarchy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
