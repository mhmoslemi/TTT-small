"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input (FP32 for stability) → cast to FP16.
2. Fuse linear projections:
   * Combine left_proj, right_proj and out_gate weights → one FP16 GEMM.
   * Split into left, right and out_gate (apply sigmoid to out_gate).
   * Combine left_gate and right_gate weights → one FP16 GEMM.
   * Split and apply sigmoid to obtain left_gate & right_gate.
3. Fuse mask, left‑gate and right‑gate multiplications in a single Triton kernel
   (in‑place on `left` and `right`).
4. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched FP16 matrix multiplication:
   (B, H, N, N) × (B, H, N, N)ᵀ.
5. Second LayerNorm over the hidden dimension H (FP16).
6. Apply the output gate (element‑wise sigmoid output from step 2).
7. Final linear projection H → C (FP16) and cast result back to FP32.

The kernel only handles the mask‑gate fusion; all heavy linear algebra is
performed with cuBLAS via `torch.nn.functional.linear` and `torch.matmul`,
which on modern GPUs provides Tensor‑Core speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16*   (B*N*N*H) left tensor (flattened)
    right_ptr,         # fp16*   (B*N*N*H) right tensor (flattened)
    mask_ptr,          # fp16*   (B*N*N)   mask tensor (flattened)
    left_gate_ptr,     # fp16*   (B*N*N*H) left gate tensor (flattened)
    right_gate_ptr,    # fp16*   (B*N*N*H) right gate tensor (flattened)
    H: tl.constexpr,   # hidden dimension
    TOTAL_ELEM: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place multiplication:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    Mask is broadcasted over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELEM

    # Load values
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Compute corresponding mask index (flattened B*N*N)
    idx = offs // H
    m = tl.load(mask_ptr + idx, mask=mask)

    # Apply fused multiplication and store back
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)

            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates are no longer needed
        del left_gate, right_gate
    else:
        # Simple element‑wise multiplication when no mask is provided
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5) Core reduction: batched matrix multiplication (B, H, N, N) × (B, H, N, N)ᵀ
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for efficient matmul
    left_b = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Batched matmul over the sequence dimension
    out_b = torch.matmul(left_b, right_b.transpose(-2, -1))  # (B, H, N, N)

    # Back to original layout
    out = out_b.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # (B, N, N, H), fp16

    # ------------------------------------------------------------------
    # 7) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection H → C
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, C), fp16

    # ------------------------------------------------------------------
    # 9) Cast back to float32 (required output dtype)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
```
