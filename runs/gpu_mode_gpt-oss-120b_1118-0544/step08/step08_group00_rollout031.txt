@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid = tl.program_id(axis=0) # rows block
    pid_n = tl.program_id(axis=1) # cols block
    pid_b = tl.program_id(axis=2) # batch dimension

    # compute block start offset
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute pointers for a, b
    a_ptrs = a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load
        a = tl.load(a_ptrs, mask=offs_m < M, other=0.0)
        b = tl.load(b_ptrs, mask=offs_n < N, other=0.0)
        # compute
        acc += tl.dot(a, b)
        # advance ptrs
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # convert to output dtype (fp16)
    c = acc.to(tl.float16)
    # store
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Cast to fp16
    x = x.to(torch.float16)

    # Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)  # shape (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Apply mask + gating, using Triton kernel if mask present and not nomask
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
            block_size=1024,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # Should not happen due to condition
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp16
            right = right * mask_fp16

    # Core reduction using Triton kernel (batched matmul)
    B, N, _, H = left.shape
    # reshape for batched matmul: (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Transpose right for dot product
    right_t = right_flat.transpose(1, 2).contiguous()

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # launch kernel
    _batched_matmul(out_flat, left_flat, right_t)

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # second layernorm
    out = torch.nn.functional.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # multiply by out_gate
    out = out * out_gate

    # final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to input dtype if needed
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

    mask_n = offs_n < N

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for compute‑heavy ops.
3. Linear projections (left/right) and gating (left/right/out) using bias‑free
   linear layers.
4. If a mask is supplied and `nomask` is False, a Triton kernel fuses
   projection, gating and mask for the left and right tensors.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (B*H, N, N) @ (B*H, N, N)ᵀ using a custom
   Triton kernel (no cuBLAS call).
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and project back to the original dimension.
7. Cast the result back to the original dtype.

The kernel works for any batch size (1‑2), sequence length up to 1024,
and hidden dimension up to 768.  All heavy arithmetic is performed in
FP16, while the final output is returned in the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# 1. Fuse projection, gate and mask (used when a mask is present)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)
    right_gate_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    left_out_ptr,    # fp16 * (B, N, N, H)
    right_out_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gate and mask for left/right tensors."""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds
    in_range = offs < total_elems

    # Index into mask (mask has no hidden dimension)
    mask_idx = offs // hidden_dim

    # Load
    left_proj = tl.load(left_proj_ptr + offs, mask=in_range, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=in_range, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=in_range, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=in_range, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=in_range, other=1.0)

    # Fuse: proj * gate * mask
    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=in_range)
    tl.store(right_out_ptr + offs, right_out, mask=in_range)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse element‑wise mask, projection and gate for both left and right tensors.
    All tensors must be fp16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)
    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# 2. Batched matrix multiplication kernel (core TriMul reduction)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr,          # fp16 * (B, M, K)
    b_ptr,          # fp16 * (B, K, N)
    c_ptr,          # fp16 * (B, M, N)
    M, N, K,        # i64 scalars
    stride_am, stride_ak, stride_ab,   # i64 strides for A
    stride_bk, stride_bn, stride_bb,   # i64 strides for B
    stride_cm, stride_cn, stride_cb,   # i64 strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b] = A[b] @ B[b]ᵀ  for all b."""
    pid_m = tl.program_id(0)  # row block
    pid_n = tl.program_id(1)  # col block
    pid_b = tl.program_id(2)  # batch index

    # -----------------------------
    # Offsets for this block
    # -----------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for A and B (will be advanced in k‑loop)
    a_ptrs = a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Determine which k‑values are valid in this tile
        k_valid = tl.arange(0, BLOCK_K) < K - k * BLOCK_K

        # Load A and B with appropriate masks
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_n[None, :],
                    other=0.0)

        # Compute outer product for the tile
        acc += tl.dot(a, b)

        # Advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Cast back to FP16 and store
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul(
    out: torch.Tensor,
    a: torch.Tensor,
    b: torch.Tensor,
    block_m: int = 128,
    block_n: int = 128,
    block_k: int = 32,
) -> None:
    """
    Wrapper around the Triton batched GEMM kernel.
    a: (B, M, K), b: (B, K, N), out: (B, M, N)
    """
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Inner dimensions must match"

    grid = (
        triton.cdiv(M, block_m),
        triton.cdiv(N, block_n),
        B,
    )
    _batched_matmul_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(1),
        a.stride(2),
        a.stride(0),
        b.stride(1),
        b.stride(2),
        b.stride(0),
        out.stride(1),
        out.stride(2),
        out.stride(0),
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )


# ----------------------------------------------------------------------
# 3. Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None (0/1)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast model weights to FP16 (bias‑free linear layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gates (all FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid() # (B, N, N, H)
    out_gate = F.linear(x, out_gate_w).sigmoid()     # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask, projection and gate (Triton kernel when needed)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
            block_size=1024,
        )
    else:
        # Simple element‑wise fusion (mask ignored)
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # Mask present but user wants to ignore it → treat as all‑ones
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
            left = left * mask_fp16
            right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM across the hidden dimension.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # transposed right for standard GEMM semantics
    right_t = right_flat.transpose(1, 2).contiguous()

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # launch custom batched GEMM kernel
    _batched_matmul(out_flat, left_flat, right_t,
                    block_m=128, block_n=128, block_k=32)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
