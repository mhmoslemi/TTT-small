"""
TriMul (outgoing) forward kernel – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑norm over the input channel dimension (`dim`).
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections (`left_proj`, `right_proj`) and three gates
   (`left_gate`, `right_gate`, `out_gate`) – all in FP16.
4. Fuse mask (if present) with the projections and gates.
5. Core reduction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM for every (batch, hidden) pair.
   Implemented with a custom Triton kernel that processes
   tiles of size (BLOCK_M, BLOCK_N, BLOCK_K) and accumulates
   in FP32 before casting back to FP16.
6. Output LayerNorm over the hidden dimension, multiplication by
   `out_gate`, and final linear projection back to `dim`.
7. Cast the result back to the input dtype (normally FP32).

The kernel operates on the flattened batch*hidden dimension, i.e.
`M = B * hidden_dim`.  Each program instance computes a
( BLOCK_M × BLOCK_N ) tile of the output matrix for a single
`m`‑index.
"""

import math
from typing import Tuple, Dict

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _triton_trimul_kernel(
    A_ptr,               # left  : fp16 * (M, N, K)
    B_ptr,               # right : fp16 * (M, N, K)
    C_ptr,               # out   : fp16 * (M, N, N)
    M, N, K,
    stride_am, stride_an, stride_ak,   # strides for A
    stride_bm, stride_bn, stride_bk,   # strides for B
    stride_cm, stride_cn, stride_ck,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)   # flattened batch * hidden
    pid_i = tl.program_id(1)   # row tile
    pid_j = tl.program_id(2)   # col tile

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers to the start of the tile
    a_ptrs = A_ptr + pid_m * stride_am + offs_i[:, None] * stride_an + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_m * stride_bm + offs_j[:, None] * stride_bn + offs_k[None, :] * stride_bk

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(tl.cdiv(K, BLOCK_K)):
        # load tiles (mask out‑of‑bounds)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & (offs_k[None, :] < K), other=0.0)

        # dot product across K
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # cast back to FP16
    c = acc.to(tl.float16)

    # store result
    c_ptrs = C_ptr + pid_m * stride_cm + offs_i[:, None] * stride_cn + offs_j[None, :] * stride_ck
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_trimul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the TriMul reduction using the Triton kernel.

    Args:
        left, right: tensors of shape (B, N, N, H), dtype=fp16, contiguous.

    Returns:
        Tensor of shape (B, N, N, H), dtype=fp16.
    """
    B, N, _, H = left.shape
    # reshape to (M, N, N) where M = B * H
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    # block sizes – chosen to fit typical B200 SM resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        left_flat.shape[0],                         # M
        (N + BLOCK_M - 1) // BLOCK_M,               # rows
        (N + BLOCK_N - 1) // BLOCK_N,               # cols
    )

    _triton_trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        left_flat.shape[0],
        N,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
