"""
TriMul (outgoing) forward pass implemented with Triton.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for the compute‑intensive part.
3. Linear projections without bias:
   * left  = x @ left_proj_weight
   * right = x @ right_proj_weight
4. Apply optional mask (element‑wise) before gating.
5. Gating via sigmoid (linear + sigmoid) and element‑wise multiplication.
6. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each head `h`.  A custom
   Triton kernel performs the multiplication `left @ rightᵀ` for every
   (batch, head) pair.
7. Output LayerNorm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
8. Cast back to the original input dtype.

The Triton kernel implements the O(N³) reduction efficiently by tiling
the three dimensions and accumulating in fp32 before storing as fp16.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left tensor
    stride_lb, stride_li, stride_lk, stride_lh,
    # strides for right tensor
    stride_rb, stride_rj, stride_rk, stride_rh,
    # strides for out tensor
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication for TriMul:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    """
    pid_i = tl.program_id(0)   # i‑block
    pid_j = tl.program_id(1)   # j‑block
    pid_bh = tl.program_id(2)  # combined batch‑head index

    # decode batch and head indices
    b = pid_bh // H
    h = pid_bh % H

    # compute i, j offsets within the block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, h) pair
    left_base = left_ptr + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base = out_ptr + b * stride_ob + h * stride_oh

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # load right tile transposed: shape (BLOCK_K, BLOCK_N)
        # we need right[b, j, k, h] => transposed to (k, j)
        b_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # matrix multiply‑accumulate
        acc += tl.dot(a, b)

    # write result back in fp16
    c_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor of shape [B, N, N] (0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension `dim`
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(dtype_in)
    norm_b = weights["norm.bias"].to(dtype_in)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy parts
    # ------------------------------------------------------------------
    compute_dtype = torch.float16
    x = x.to(compute_dtype)

    # ------------------------------------------------------------------
    # 3) Load linear weights (biases are None) and cast to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(compute_dtype)
    right_proj_w  = weights["right_proj.weight"].to(compute_dtype)
    left_gate_w   = weights["left_gate.weight"].to(compute_dtype)
    right_gate_w  = weights["right_gate.weight"].to(compute_dtype)
    out_gate_w    = weights["out_gate.weight"].to(compute_dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype)
    to_out_w      = weights["to_out.weight"].to(compute_dtype)

    # ------------------------------------------------------------------
    # 4) Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # [B, N, N, H]
    right_proj = F.linear(x, right_proj_w)        # [B, N, N, H]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Apply mask (if present) before gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape: [B, N, N] -> broadcast to hidden dim
        mask = mask.to(compute_dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left_proj * mask
        right = right_proj * mask
    else:
        left = left_proj
        right = right_proj

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction using Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # allocate output tensor
    out = torch.empty_like(left, dtype=compute_dtype)

    # Determine block sizes (tuned for typical B200 GPUs)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Compute grid dimensions
    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    grid_bh = B * H
    grid = (grid_i, grid_j, grid_bh)

    # Strides (in elements) for each tensor
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Launch Triton kernel
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
