"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The kernel fuses:
  * element‑wise mask application,
  * left/right gating (sigmoid linear layers),
  * the core reduction  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

All heavy work is performed in a single Triton kernel that computes a
batched matrix multiplication for each (batch, head) pair.  The remaining
operations (LayerNorm, final projection) are carried out with PyTorch
functional calls.  The final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _outgoing_trimul_kernel(
    # tensors
    left_ptr, right_ptr, out_ptr,               # fp16
    left_gate_ptr, right_gate_ptr,               # fp16
    mask_ptr,                                    # fp16 (optional)
    # shapes / strides
    B, N, H,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    stride_lgate_b, stride_lgate_h, stride_lgate_i, stride_lgate_k,
    stride_rgate_b, stride_rgate_h, stride_rgate_i, stride_rgate_k,
    stride_mask_b, stride_mask_i, stride_mask_k,
    # compile‑time constants
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """Batched matmul with fused mask and gating.
    Grid: (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    """
    pid0 = tl.program_id(0)      # combined batch * head
    pid1 = tl.program_id(1)      # block row
    pid2 = tl.program_id(2)      # block col

    bh = pid0
    b = bh // H                 # batch index
    h = bh % H                  # head index

    # ----- base pointers for this (b, h) -----
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    lgate_base = left_gate_ptr  + b * stride_lgate_b  + h * stride_lgate_h
    rgate_base = right_gate_ptr + b * stride_rgate_b + h * stride_rgate_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h
    mask_base  = mask_ptr  + b * stride_mask_b                     # mask has no head dim

    # ----- tile indices -----
    i = tl.arange(0, BLOCK_M) + pid1 * BLOCK_M
    j = tl.arange(0, BLOCK_N) + pid2 * BLOCK_N
    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # ----- main reduction over k -----
    for kk in range(0, N, BLOCK_K):
        k = tl.arange(0, BLOCK_K) + kk
        mask_k = k < N

        # left tile: shape (M, K)
        left_off = left_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        l = tl.load(left_off, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        lgate_off = lgate_base + i[:, None] * stride_lgate_i + k[None, :] * stride_lgate_k
        lg = tl.load(lgate_off, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        if HAS_MASK:
            mask_off = mask_base + i[:, None] * stride_mask_i + k[None, :] * stride_mask_k
            m = tl.load(mask_off, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
            l = l * lg * m
        else:
            l = l * lg

        # right tile: need shape (K, N) → rows are k, cols are j
        right_off = right_base + k[:, None] * stride_right_i + j[None, :] * stride_right_k
        r = tl.load(right_off, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        rgate_off = rgate_base + k[:, None] * stride_rgate_i + j[None, :] * stride_rgate_k
        rg = tl.load(rgate_off, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        if HAS_MASK:
            mask_off = mask_base + k[:, None] * stride_mask_i + j[None, :] * stride_mask_k
            m = tl.load(mask_off, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
            r = r * rg * m
        else:
            r = r * rg

        # accumulate dot product
        acc += tl.dot(l, r)

    # ----- write back -----
    out_off = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_off, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (binary) or None
            - weights: dict of parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    # -------------------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # -------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Work in fp16 from here on
    x = x.to(torch.float16)

    # -------------------------------------------------------------
    # 2) Cast weights to fp16
    # -------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # -------------------------------------------------------------
    # 3) Linear projections to hidden dimension
    # -------------------------------------------------------------
    left_raw  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_raw = F.linear(x, right_proj_w)   # (B, N, N, H)

    # -------------------------------------------------------------
    # 4) Compute sigmoid gates
    # -------------------------------------------------------------
    left_gate_raw  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate_raw = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate_raw   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # -------------------------------------------------------------
    # 5) Rearrange to (B, H, N, N) for the kernel
    # -------------------------------------------------------------
    left  = left_raw.permute(0, 3, 1, 2).contiguous()        # (B, H, N, N)
    right = right_raw.permute(0, 3, 1, 2).contiguous()
    left_gate  = left_gate_raw.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate_raw.permute(0, 3, 1, 2).contiguous()
    out_gate   = out_gate_raw.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape

    # -------------------------------------------------------------
    # 6) Prepare mask (optional)
    # -------------------------------------------------------------
    if mask is None or nomask:
        has_mask = 0
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
    else:
        has_mask = 1
        mask_fp16 = mask.to(device=device, dtype=torch.float16)

    # -------------------------------------------------------------
    # 7) Allocate output tensor
    # -------------------------------------------------------------
    out = torch.empty_like(left)  # (B, H, N, N), fp16

    # -------------------------------------------------------------
    # 8) Launch Triton kernel
    # -------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _outgoing_trimul_kernel[grid](
        # pointers
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_fp16,
        # shape / strides
        B=B, N=N, H=H,
        stride_left_b=left.stride(0),   stride_left_h=left.stride(1),
        stride_left_i=left.stride(2),   stride_left_k=left.stride(3),
        stride_right_b=right.stride(0), stride_right_h=right.stride(1),
        stride_right_i=right.stride(2), stride_right_k=right.stride(3),
        stride_out_b=out.stride(0),     stride_out_h=out.stride(1),
        stride_out_i=out.stride(2),     stride_out_j=out.stride(3),
        stride_lgate_b=left_gate.stride(0),   stride_lgate_h=left_gate.stride(1),
        stride_lgate_i=left_gate.stride(2),   stride_lgate_k=left_gate.stride(3),
        stride_rgate_b=right_gate.stride(0),  stride_rgate_h=right_gate.stride(1),
        stride_rgate_i=right_gate.stride(2),  stride_rgate_k=right_gate.stride(3),
        stride_mask_b=mask_fp16.stride(0) if has_mask else 0,
        stride_mask_i=mask_fp16.stride(1) if has_mask else 0,
        stride_mask_k=mask_fp16.stride(2) if has_mask else 0,
        # compile‑time constants
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        HAS_MASK=has_mask,
        num_warps=4,
    )

    # -------------------------------------------------------------
    # 9) Bring tensor back to (B, N, N, H) layout
    # -------------------------------------------------------------
    out = out.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # -------------------------------------------------------------
    # 10) Second LayerNorm over hidden dimension H
    # -------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------------------
    # 11) Apply output gate (element‑wise)
    # -------------------------------------------------------------
    out = out * out_gate_raw

    # -------------------------------------------------------------
    # 12) Final linear projection back to original channel dimension C
    # -------------------------------------------------------------
    out = F.linear(out, to_out_w)      # (B, N, N, C)

    # -------------------------------------------------------------
    # 13) Cast to float32 (required output dtype)
    # -------------------------------------------------------------
    out = out.to(torch.float32)

    return out
