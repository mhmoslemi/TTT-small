    b = tl.program_id(2)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # mask for within bounds

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LN input
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast to FP16
    x = x.to(torch.float16)

    # Get weight tensors in FP16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left_proj = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # Gate projections
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply mask if needed
    if mask is not None and not nomask:
        mask = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Core reduction via triton kernel
    B, N, _, H = left.shape
    batch = B * H
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(batch, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(batch, N, N)
    out_flat = torch.empty_like(left_flat)

    # Get strides
    stride_left_batch, stride_left_m, stride_left_n = left_flat.stride()
    stride_right_batch, stride_right_m, stride_right_n = right_flat.stride()
    stride_out_batch, stride_out_m, stride_out_n = out_flat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), batch)

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch,
        N,
        N,
        N,
        stride_left_batch,
        stride_left_m,
        stride_left_n,
        stride_right_batch,
        stride_right_m,
        stride_right_n,
        stride_out_batch,
        stride_out_m,
        stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)

    # Cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out

"""
TriMul outgoing (AlphaFold3) forward pass – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑norm over the input channel dimension `dim`.
2. Cast to FP16 for compute‑intensive parts.
3. Linear projections → `left_proj`, `right_proj` (no bias).
4. Gating projections → `left_gate`, `right_gate`, `out_gate` (sigmoid).
5. Optional mask multiplication (broadcasted over the hidden dimension).
6. Element‑wise fusion: `left = left_proj * left_gate * mask`,
   `right = right_proj * right_gate * mask`.
7. Core reduction (TriMul):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication using a custom Triton kernel
   (each hidden slice is treated as an independent batch).
8. Layer‑norm over the hidden dimension, multiplication by `out_gate`,
   final linear projection back to `dim`.
9. Cast back to the original dtype (usually float32).

Only the reduction step is written in Triton; all other ops use
torch.nn.functional for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    left_ptr,    # fp16 * (batch, M, K)
    right_ptr,   # fp16 * (batch, N, K)   (will be accessed transposed)
    out_ptr,     # fp16 * (batch, M, N)
    batch,       # i64 scalar = B * H
    M,           # i64 scalar = N (seq_len)
    N,           # i64 scalar = N (seq_len)
    K,           # i64 scalar = N (seq_len)
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_m, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication: out = left @ right.T  (per hidden slice)."""
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block column
    b = tl.program_id(2)      # batch index (B*H)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr
            + b * stride_left_batch
            + offs_m[:, None] * stride_left_m
            + offs_k[None, :] * stride_left_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load right tile transposed: we need right_T[k, n]
        # Original right shape (batch, N, K); we access (k, n) to get (K, N)
        right = tl.load(
            right_ptr
            + b * stride_right_batch
            + offs_k[:, None] * stride_right_m   # k as first dim -> stride of original row
            + offs_n[None, :] * stride_right_k,   # n as second dim -> stride of original col
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate dot product
        acc += tl.dot(left, right)  # (M,K) @ (K,N) -> (M,N)

    # Write back the result (cast to fp16)
    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + b * stride_out_batch
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n,
        out,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (0/1) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim" and "hidden_dim" (and optional "nomask")

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to FP16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply optional mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 6) Fuse projections with gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate   # (B, N, N, H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul) via Triton kernel
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    batch = B * H  # treat each hidden slice as a separate batch

    # reshape to (batch, N, N) where batch = B * H
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(batch, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(batch, N, N)
    out_flat = torch.empty_like(left_flat)

    # strides for the flattened tensors
    stride_left_batch, stride_left_m, stride_left_n = left_flat.stride()
    stride_right_batch, stride_right_m, stride_right_n = right_flat.stride()
    stride_out_batch, stride_out_m, stride_out_n = out_flat.stride()

    # Block sizes – tuned for typical B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks in M dimension
        triton.cdiv(N, BLOCK_N),   # blocks in N dimension
        batch,                     # one program per (B*H)
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch,
        N,
        N,
        N,
        stride_left_batch,
        stride_left_m,
        stride_left_n,
        stride_right_batch,
        stride_right_m,
        stride_right_n,
        stride_out_batch,
        stride_out_m,
        stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
