@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # Program id
    pid0 = tl.program_id(0)  # batch * head id
    pid1 = tl.program_id(1)  # tile_i
    pid2 = tl.program_id(2)  # tile_j

    # decode batch and head
    bh = pid0
    b = bh // H
    h = bh % H

    # start offsets
    i_start = pid1 * BLOCK_M
    j_start = pid2 * BLOCK_N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K

        # compute offsets for left tile
        i_idx = i_start + tl.arange(0, BLOCK_M)
        k_idx = k_start + tl.arange(0, BLOCK_K)

        # guard for i and k
        mask_i = i_idx < N
        mask_k = k_idx < N

        # left pointer offsets
        left_offset = (b * stride_left_b + 
                       i_idx[:, None] * stride_left_i + 
                       k_idx[None, :] * stride_left_k + 
                       h * stride_left_h)

        left_tile = tl.load(left_ptr + left_offset,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # right tile shape (K,N)
        j_idx = j_start + tl.arange(0, BLOCK_N)
        # guard j
        mask_j = j_idx < N

        right_offset = (b * stride_right_b + 
                        k_idx[:, None] * stride_right_k + 
                        j_idx[None, :] * stride_right_i + 
                        h * stride_right_h)

        right_tile = tl.load(right_ptr + right_offset,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # Multiply-accumulate
        # Convert to fp32 for accumulation
        acc += tl.dot(left_tile, right_tile).to(tl.float32)

    # Write out
    # output offsets
    i_idx = i_start + tl.arange(0, BLOCK_M)
    j_idx = j_start + tl.arange(0, BLOCK_N)
    mask_i = i_idx < N
    mask_j = j_idx < N
    out_offset = (b * stride_out_b + 
                  i_idx[:, None] * stride_out_i + 
                  j_idx[None, :] * stride_out_j + 
                  h * stride_out_h)
    tl.store(out_ptr + out_offset,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid0 = tl.program_id(0)  # batch_head
    pid1 = tl.program_id(1)  # tile_i
    pid2 = tl.program_id(2)  # tile_j

    bh = pid0
    b = bh // H
    h = bh % H

    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    # create range vectors
    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)

    # accumulate in fp32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K blocks
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_iter * BLOCK_K
        k_range = k0 + tl.arange(0, BLOCK_K)

        # masks for bounds
        mask_i = i_range < N
        mask_j = j_range < N
        mask_k = k_range < N

        # left tile (M,K)
        left_ptrs = left_ptr + b * stride_left_b + \
                    i_range[:, None] * stride_left_i + \
                    k_range[None, :] * stride_left_k + \
                    h * stride_left_h

                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

               i_range[:, None] * stride_out_i + \
               j_range[None, :] * stride_out_j + \
               h * stride_out_h

"""
custom_kernel: Outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for the compute‑intensive part.
3. Linear projections (left/right) and gating (left/right/out) are performed
   with `torch.nn.functional.linear` + sigmoid.
4. If a mask is provided and not ignored, it is broadcast and fused with
   the gating via simple element‑wise multiplication (the heavy work is still
   done in the Triton kernel below).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden slice `h`.  It is
   implemented as a custom Triton kernel that tiles the output (i,j) plane and
   accumulates over `k` using `tl.dot`.  The kernel works on fp16 inputs and
   accumulates in fp32 for numerical stability.
6. Output LayerNorm over the hidden dimension, element‑wise multiplication by
   `out_gate`, and a final linear projection back to the original `dim`.
7. Cast the result back to the input dtype.

Only the reduction (step 5) is performed with Triton; all other operations
use highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    left_ptr,           # fp16 * (B, N, N, H)
    right_ptr,          # fp16 * (B, N, N, H)
    out_ptr,            # fp16 * (B, N, N, H)
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """TriMul core reduction kernel."""
    # program ids
    pid0 = tl.program_id(0)   # batch * head dimension
    pid1 = tl.program_id(1)   # tile over i
    pid2 = tl.program_id(2)   # tile over j

    bh = pid0
    b = bh // H
    h = bh % H

    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    # range vectors
    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        k0 = kk * BLOCK_K
        k_range = k0 + tl.arange(0, BLOCK_K)

        # bounds
        mask_i = i_range < N
        mask_j = j_range < N
        mask_k = k_range < N

        # ----- left tile (M x K) -----
        left_offset = (
            b * stride_left_b
            + i_range[:, None] * stride_left_i
            + k_range[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(
            left_ptr + left_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # ----- right tile (K x N)  -----
        # note: we load as (K, N) where K is the reduction axis
        right_offset = (
            b * stride_right_b
            + k_range[:, None] * stride_right_k
            + j_range[None, :] * stride_right_i
            + h * stride_right_h
        )
        right = tl.load(
            right_ptr + right_offset,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # multiply‑accumulate
        acc += tl.dot(left, right).to(tl.float32)

    # ----- store result -----
    out_offset = (
        b * stride_out_b
        + i_range[:, None] * stride_out_i
        + j_range[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(
        out_ptr + out_offset,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (0/1) or None
            - weights       : dict of model parameters (float32)
            - config        : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype
    compute_dtype = torch.float16

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype_in).to(device)
    norm_bias = weights["norm.bias"].to(dtype_in).to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(compute_dtype)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(compute_dtype).to(device)
    right_proj_w  = weights["right_proj.weight"].to(compute_dtype).to(device)
    left_gate_w   = weights["left_gate.weight"].to(compute_dtype).to(device)
    right_gate_w  = weights["right_gate.weight"].to(compute_dtype).to(device)
    out_gate_w    = weights["out_gate.weight"].to(compute_dtype).to(device)
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype).to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype).to(device)
    to_out_w      = weights["to_out.weight"].to(compute_dtype).to(device)

    # ------------------------------------------------------------------
    # 4) Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)      # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Fuse gating (and optional mask) – simple elementwise ops
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(compute_dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguity for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 6) Core reduction – Triton implementation
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=compute_dtype, device=device)

    # strides (int)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tile sizes – chosen to fit typical B200 resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                                   # batch * hidden dimension
        triton.cdiv(N, BLOCK_M),                 # tiles over i
        triton.cdiv(N, BLOCK_N),                 # tiles over j
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
