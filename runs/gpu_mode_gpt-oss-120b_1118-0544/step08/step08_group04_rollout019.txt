"""
TriMul (outgoing) forward pass accelerated with Triton.

Algorithm
---------
1. Layer‑normalize the 4‑D input (float32 for numerical stability) and cast to fp16.
2. Linear projection of the normalized tensor to the hidden dimension `H`:
   * left  = x @ left_proj_weight
   * right = x @ right_proj_weight
3. Compute three sigmoid gates (left, right, out) from the same normalized input.
4. Fuse mask, left‑gate and right‑gate element‑wise multiplications in a single
   Triton kernel (in‑place, no extra temporaries).  When no mask is present we
   perform simple element‑wise multiplication.
5. Core O(N³·H) reduction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM on tensors of shape (B·H, N, N) using
   torch.matmul (fp16).
6. Layer‑normalize over the hidden dimension `H`.
7. Apply the output gate.
8. Final linear projection back to the original channel dimension `C`.
9. Cast the result to float32 (required output dtype).

All heavy arithmetic is performed in fp16, while layer‑norms use fp32 weights
for stability.  The only required Triton kernel is the mask‑gate fusion.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    TOTAL_ELEMS: tl.constexpr,  # total number of elements in left/right
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place fusion of mask and the two gate tensors:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELEMS

    # Load values
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Compute mask index (each (b,i,j) has H consecutive entries)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Fuse the multiplications
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict containing model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ----------------------- unpack inputs -----------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    nomask = config.get("nomask", False)

    # ------------------- 1) Input LayerNorm --------------------
    # LayerNorm in fp32 for stability, then cast to fp16
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32)
    )
    x = x.to(torch.float16)

    # ------------------- 2) Cast weights to fp16 ----------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------- 3) Linear projections -----------------
    # (B, N, N, H) in fp16
    left  = F.linear(x, left_proj_w)   # shape (B, N, N, H)
    right = F.linear(x, right_proj_w)  # shape (B, N, N, H)

    # ------------------- 4) Compute gates (sigmoid) ------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------- 5) Fuse mask & gates -----------------
    if mask is not None and not nomask:
        # Cast mask to fp16 and flatten
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()                     # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_kernel[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            H=hidden_dim,
            TOTAL_ELEMS=total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates have been merged; free memory
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------- 6) Core reduction (batched GEMM) -----
    # Reshape to (B*H, N, N) for batched matmul
    B, N, _, H = left.shape
    left_b  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H, N, N)

    # out_b shape: (B*H, N, N)
    out_b = torch.matmul(left_b, right_b.transpose(1, 2))

    # Restore original layout (B, N, N, H)
    out = out_b.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- 7) Second LayerNorm (over H) --------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------- 8) Apply output gate -----------------
    out = out * out_gate
    del out_gate  # free

    # ------------------- 9) Final linear projection -----------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------- 10) Cast to float32 -----------------
    out = out.to(torch.float32)

    return out
