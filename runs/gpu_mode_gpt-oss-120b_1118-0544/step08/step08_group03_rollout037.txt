import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_gate_fusion_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    H: tl.constexpr,
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Compute mask idx
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply
    tl.store(left_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    The forward pass performs:
      1. LayerNorm on input.
      2. Linear projections (left/right) to hidden dimension.
      3. Sigmoid gates (left, right, out).
      4. In‑place fusion of mask and gates via a Triton kernel.
      5. Batched matrix multiplication over the sequence dimension:
         out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
      6. LayerNorm on the hidden dimension.
      7. Apply output gate.
      8. Final linear projection back to original channel dimension.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm (float32 for stability)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32)
    )
    # Keep computation in float32 (can be changed to fp16 for speed)
    x = x.to(torch.float32)

    # 2. Linear projections to hidden dim (no bias)
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float32)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float32)
    left = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # 3. Compute sigmoid gates
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float32)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float32)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float32)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # 4. Fuse mask and gates
    if mask is not None and not nomask:
        # Ensure tensors are contiguous
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp = mask.to(device=device, dtype=torch.float32).contiguous()
        total_elements = left.numel()  # B * N * N * hidden_dim
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)
        mask_gate_fusion_kernel[grid](
            left,
            right,
            mask_fp.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates no longer needed
        del left_gate, right_gate
    else:
        # No mask: just multiply by gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # 5. Core reduction: batched matrix multiplication per hidden dim
    # reshape to (B, hidden_dim, N, N)
    left_b = left.permute(0, 3, 1, 2).contiguous()
    # right needs to be transposed on last two dims before matmul
    right_b = right.permute(0, 3, 2, 1).contiguous()
    out_b = torch.matmul(left_b, right_b)  # (B, hidden_dim, N, N)
    out = out_b.permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden_dim)

    # 6. Second LayerNorm over hidden_dim
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 7. Apply output gate
    out = out * out_gate

    # 8. Final linear projection back to dim
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float32)
    out = F.linear(out, to_out_w)

    # Cast back to original dtype if needed
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # *float32 left  (B,N,N,H) flattened
    right_ptr,         # *float32 right (B,N,N,H) flattened
    mask_ptr,          # *float32 mask  (B,N,N)   flattened
    left_gate_ptr,     # *float32 left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # *float32 right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and gates:
    left  = left  * mask * left_gate
    right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load values
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Broadcast mask (one scalar per hidden element)
    mask_idx = offs // H                     # index into mask (B,N,N)
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply fusion and store back
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    Forward pass steps:
      1. LayerNorm on the input tensor.
      2. Linear projections (left/right) from dim → hidden_dim.
      3. Compute sigmoid gates (left, right, out).
      4. Fuse mask and gates in‑place via a Triton kernel.
      5. Batched matrix multiplication over the sequence dimension:
         out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
      6. LayerNorm over the hidden dimension.
      7. Apply the output gate.
      8. Final linear projection back to dim.
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float32)  # keep everything in FP32 for numerical fidelity

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device=device, dtype=torch.float32)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float32)

    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Sigmoid gates
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device=device, dtype=torch.float32)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float32)
    out_gate_w   = weights["out_gate.weight"].to(device=device, dtype=torch.float32)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 4. Fuse mask and gates (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure contiguous layout
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp = mask.to(device=device, dtype=torch.float32).contiguous()

        total_elements = left.numel()                # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_kernel[grid](
            left,
            right,
            mask_fp.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        del left_gate, right_gate   # free memory
    else:
        # No mask: just apply the gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5. Core reduction: batched matrix multiplication per hidden dim
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N)
    left_b  = left.permute(0, 3, 1, 2).contiguous()
    # right needs to have (k,j) order for the matmul
    right_b = right.permute(0, 3, 2, 1).contiguous()
    out_b = torch.matmul(left_b, right_b)            # (B,H,N,N)
    out = out_b.permute(0, 2, 3, 1).contiguous()    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7. Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float32)
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
