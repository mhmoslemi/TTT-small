"""
TriMul outgoing (AlphaFold3) forward pass fused with Triton.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections (left/right) and gating projections (left_gate,
   right_gate, out_gate) – all bias‑free.
4. Fuse projection, gating and (optional) mask with a small Triton kernel.
   The kernel produces `left` and `right` tensors of shape (B,N,N,H):
        left  = (x @ left_proj_w ) * sigmoid(x @ left_gate_w ) * mask
        right = (x @ right_proj_w) * sigmoid(x @ right_gate_w) * mask
   If no mask is present the fusion is done with plain PyTorch.
5. Core reduction (tri‑multiplicative update):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is implemented as a batched GEMM per hidden slice:
        out_h = left_h @ right_hᵀ  for every h∈[0,H).
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and apply the final linear projection back to `dim`.
7. Cast the result back to the original dtype.

Only the projection‑gate‑mask fusion is performed in Triton; the rest
leverages highly‑optimized PyTorch kernels (torch.nn.functional.linear,
torch.bmm, etc.).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse projection, gating and optional mask:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
    The mask has shape (B, N, N) and is broadcast over the hidden
    dimension. `total_elems` is the number of elements in the flattened
    (B,N,N,H) tensors.
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Boundary check
    mask = offs < total_elems

    # Compute index of the corresponding mask element.
    # Each mask element repeats `hidden_dim` times in the flattened tensor.
    mask_idx = offs // hidden_dim

    # Load values (masked loads return 0.0 for out‑of‑bounds elements)
    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # default mask=1

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Launch the Triton kernel that fuses projection, gating and mask.
    All tensors must be contiguous and of dtype torch.float16.
    Returns a tuple (left, right) each of shape (B, N, N, H).
    """
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)
    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        left_proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: Tensor of shape [B, N, N, C] (float32)
            - mask: Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict with model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Fuse mask, projection and gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16, contiguous and has no extra dimension.
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask (or mask ignored): simple element‑wise fusion
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    This is a batched matmul per hidden slice.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
