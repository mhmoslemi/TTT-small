custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. LayerNorm on input
    x = torch.nn.functional.layer_norm(input_tensor,
                    (dim,),
                    weight=weights["norm.weight"],
                    bias=weights["norm.bias"])

    # 2. Cast to fp16
    x = x.to(torch.float16)

    # 3. Linear projections (no bias)
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))

    # 4. Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # 5. Apply mask and gating
    if mask is not None and not nomask:
        # broadcast mask: shape [B,N,N,1]
        mask_fp16 = mask.to(torch.float16)
        # Multiply left_proj,right_proj by mask
        left = left_proj * mask_fp16.unsqueeze(-1) * left_gate
        right = right_proj * mask_fp16.unsqueeze(-1) * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16)
            left = left * mask_fp16.unsqueeze(-1)
            right = right * mask_fp16.unsqueeze(-1)

    # 6. Core reduction using Triton kernel
    B, N, _, H = left.shape
    # reshape to (B*H,N,N) contiguous
    left_flat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_flat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)

    out_flat = torch.empty_like(left_flat)  # shape (B*H,N,N)

    # compute strides
    stride_l_batch = left_flat.stride(0)
    stride_l_i = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_i = right_flat.stride(1)
    stride_r_k = right_flat.stride(2)

    stride_o_batch = out_flat.stride(0)
    stride_o_i = out_flat.stride(1)
    stride_o_j = out_flat.stride(2)

    # compute grid dimensions
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_j = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = B * H * num_tiles_i * num_tiles_j

    grid = (total_tiles,)

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_i, stride_r_k,
        stride_o_batch, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,H]

    # 7. Output LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # multiply by out_gate
    out = out * out_gate
    # final projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_hidden, N,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # Compute indices
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_tiles_i * num_tiles_j

    bh_idx = pid // tiles_per_batch
    tile_id = pid % tiles_per_batch
    tile_i = tile_id // num_tiles_j
    tile_j = tile_id % num_tiles_j

    i_offset = tile_i * BLOCK_M
    j_offset = tile_j * BLOCK_N

    # offsets
    i_range = i_offset + tl.arange(0, BLOCK_M)
    j_range = j_offset + tl.arange(0, BLOCK_N)

    # mask for i and j within N
    i_mask = i_range < N
    j_mask = j_range < N

    # offsets for batch
    batch_offset = bh_idx * stride_l_batch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k_offset in range(0, N, BLOCK_K):
        k_range = k_offset + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # Load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + batch_offset + i_range[:, None] * stride_l_i + k_range[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask = i_mask[:, None] & k_mask[None, :], other=0.0)

        # Load B: shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + batch_offset + k_range[:, None] * stride_r_k + j_range[None, :] * stride_r_i
        b = tl.load(b_ptrs, mask = k_mask[:, None] & j_mask[None, :], other=0.0)

        # Compute dot product and accumulate in fp32
        # Cast to fp32 for dot if needed
        acc += tl.dot(a, b).to(tl.float32)

    # Write out
    out_ptrs = out_ptr + bh_idx * stride_o_batch + i_range[:, None] * stride_o_i + j_range[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask = i_mask[:, None] & j_mask[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The kernel implements the heavy N³ reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
as a tiled batched GEMM using Triton, while the surrounding
layer‑norms, linear projections and gating are performed with
torch.nn.functional for simplicity.  The computation is carried out
in fp16 (except the final layer‑norms which stay in fp32) and the
output is cast back to the input dtype.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 * (BH, N, N)
    right_ptr,         # fp16 * (BH, N, N)
    out_ptr,           # fp16 * (BH, N, N)
    batch_hidden,      # int64 scalar = B*H
    N,                 # int64 scalar = sequence length
    # strides for left
    stride_l_batch,    # stride for batch*hidden dimension
    stride_l_i,        # stride for i dimension
    stride_l_k,        # stride for k dimension
    # strides for right
    stride_r_batch,
    stride_r_i,
    stride_r_k,
    # strides for out
    stride_o_batch,
    stride_o_i,
    stride_o_j,
    # tile sizes
    BLOCK_M: tl.constexpr,  # rows of output tile (i dimension)
    BLOCK_N: tl.constexpr,  # cols of output tile (j dimension)
    BLOCK_K: tl.constexpr,  # reduction tile size (k dimension)
):
    pid = tl.program_id(0)

    # number of tiles per batch_hidden
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)
    tiles_per_bh = num_tiles_i * num_tiles_j

    # decode program id
    bh_idx = pid // tiles_per_bh
    tile_id = pid % tiles_per_bh
    tile_i = tile_id // num_tiles_j
    tile_j = tile_id % num_tiles_j

    # block offsets
    i0 = tile_i * BLOCK_M
    j0 = tile_j * BLOCK_N

    # ranges
    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)
    i_mask = i_range < N
    j_mask = j_range < N

    # base pointer for this batch*hidden slice
    batch_offset = bh_idx * stride_l_batch

    # accumulator in fp32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k0 in range(0, N, BLOCK_K):
        k_range = k0 + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + batch_offset \
                 + i_range[:, None] * stride_l_i \
                 + k_range[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=(i_mask[:, None] & k_mask[None, :]),
                    other=0.0)

        # right: shape (BLOCK_K, BLOCK_N)  (k as rows, j as cols)
        b_ptrs = right_ptr + batch_offset \
                 + k_range[:, None] * stride_r_k \
                 + j_range[None, :] * stride_r_i
        b = tl.load(b_ptrs,
                    mask=(k_mask[:, None] & j_mask[None, :]),
                    other=0.0)

        # dot product and accumulate (cast to fp32)
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # write out
    out_ptrs = out_ptr + bh_idx * stride_o_batch \
               + i_range[:, None] * stride_o_i \
               + j_range[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=(i_mask[:, None] & j_mask[None, :]))


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (bias‑free) and gates
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    left_gate = F.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4) Apply mask (if any) and combine projection + gate
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16)
        mask_fp16 = mask_fp16.unsqueeze(-1)          # [B, N, N, 1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but caller asked to ignore it → treat as all‑ones
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp16
            right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5) Core reduction using a Triton GEMM kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # reshape to (B*H, N, N) contiguous
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)       # (B*H, N, N)

    # strides (in elements) for the contiguous layout (batch, i, k)
    stride_l_batch = left_flat.stride(0)
    stride_l_i = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_i = right_flat.stride(1)
    stride_r_k = right_flat.stride(2)

    stride_o_batch = out_flat.stride(0)
    stride_o_i = out_flat.stride(1)
    stride_o_j = out_flat.stride(2)

    # tile sizes – chosen to give good occupancy on A100/B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_j = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = B * H * num_tiles_i * num_tiles_j
    grid = (total_tiles,)

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        stride_l_batch,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_i,
        stride_r_k,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 7) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
