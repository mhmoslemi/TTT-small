@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)  # combined batch (b*h)
    pid_m = tl.program_id(1)      # block row i
    pid_n = tl.program_id(2)      # block col j

    # Decompose pid_batch into b and h
    b = pid_batch // H
    h = pid_batch % H

    # Compute base pointer offsets for this batch-channel
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_left_b + h * stride_left_h  # same stride

    # Offsets for row and col
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out-of-bounds rows and cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Compute offsets for k
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (left): shape (BLOCK_M, BLOCK_K)
        a = tl.load(left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k,
                     mask=mask_i[:, None] & mask_k[None, :],
                     other=0.0)

        # Load B tile (right) transposed: shape (BLOCK_K, BLOCK_N)
        b_tile = tl.load(right_base + offs_k[:, None] * stride_left_k + offs_j[None, :] * stride_left_i,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0)

        # Compute dot product
        acc += tl.dot(a, b_tile)

    # Cast accumulator to fp16
    out_tile = acc.to(tl.float16)

    # Store result
    tl.store(out_ptr + b * stride_left_b + h * stride_left_h
             + offs_i[:, None] * stride_left_i + offs_j[None, :] * stride_left_i,
             out_tile,
             mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1.  Layer‑normalize the input (float32 for numerical stability) and cast to fp16.
2.  Linear projections to the hidden dimension (`left_proj`, `right_proj`).
3.  Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`).
4.  Fuse mask (if present) and the two element‑wise gate multiplications
    into a single Triton kernel (`_mask_gate_fusion_inplace`).
5.  Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is implemented as a batched GEMM in Triton (`_tri_mul_kernel`),
    avoiding the costly permute+contiguous copies of the reference.
6.  Layer‑normalize the hidden result.
7.  Apply the output gate.
8.  Final linear projection back to the original channel dimension.
9.  Cast the output back to float32.

Only the mask‑gate fusion and the O(N³) reduction are expressed in Triton;
all other operations use PyTorch’s highly‑optimized kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place fusion of mask and the two gate tensors.
    left  = left  * mask * left_gate
    right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load tensors
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # mask has no hidden dimension, broadcast it
    mask_idx = offs // H            # (b,i,j) index
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_j, stride_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for the TriMul core:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    The tensor layout is (B, N, N, H) with strides:
        stride_b : jump between batches
        stride_i : jump between rows   (first N)
        stride_j : jump between cols   (second N)
        stride_h : jump between channels (H)
    """
    pid_batch = tl.program_id(0)   # combined batch * head index
    pid_m = tl.program_id(1)       # block row   (i)
    pid_n = tl.program_id(2)       # block column(j)

    # Decompose combined index into batch and head
    b = pid_batch // H
    h = pid_batch % H

    # Base pointers for this (batch, head)
    left_base  = left_ptr  + b * stride_b + h * stride_h
    right_base = right_ptr + b * stride_b + h * stride_h
    out_base   = out_ptr   + b * stride_b + h * stride_h

    # Offsets for the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,i,k,h]   -> shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            left_base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_j,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right[b,j,k,h]  -> we need it transposed: (k,j)
        b_tile = tl.load(
            right_base + offs_k[:, None] * stride_j + offs_j[None, :] * stride_i,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b_tile)

    # Write the result (fp16) back
    out_tile = acc.to(tl.float16)
    tl.store(
        out_base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_j,
        out_tile,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for the compute‑heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w) # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask (if any) with left/right gate multiplications
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()                     # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates are now fused, we can free them
        del left_gate, right_gate
    else:
        # Simple element‑wise multiplication when no mask
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched GEMM implemented in Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # fp16 output buffer

    # Strides for the (B, N, N, H) layout
    stride_b = left.stride(0)
    stride_i = left.stride(1)   # stride for the first N (row)
    stride_j = left.stride(2)   # stride for the second N (col)
    stride_h = left.stride(3)   # stride for hidden dim (should be 1)

    # Block sizes – chosen to give good occupancy on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                         # batch * head
        triton.cdiv(N, BLOCK_M),       # rows
        triton.cdiv(N, BLOCK_N),       # cols
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_b,
        stride_i,
        stride_j,
        stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over the hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast to float32 (required output dtype)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
