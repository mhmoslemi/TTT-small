```python
@triton.jit
def matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_cn, stride_co,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask = mask_i[:, None] & mask_j[None, :]

    # Pointers
    a_ptrs = a_ptr + pid_m * stride_am + offs_i[:, None] * stride_an + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_m * stride_bm + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_i[:, None], other=0.0)  # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=mask_j[None, :], other=0.0)  # (BLOCK_K, BLOCK_N)
        # multiply-accumulate
        acc += tl.dot(a, b)

        # Increment pointers to next k tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Convert accumulator to fp16
    c = acc.to(tl.float16)

    # Store results
    c_ptrs = c_ptr + pid_m * stride_cm + offs_i[:, None] * stride_cn + offs_j[None, :] * stride_co
    tl.store(c_ptrs, c, mask=mask)
```

"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (dim).
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (left/right) and gating (left/right/out) – all fp16.
4. Apply mask (if present) and element‑wise gating.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication using a custom Triton
   kernel that computes C = A @ Bᵀ for each (batch × hidden) slice.
6. Output LayerNorm over the hidden dimension, multiply by out_gate,
   and a final linear projection back to the original dim.
7. Cast back to the input dtype (usually float32).

Only the reduction step is written in Triton; all other ops use
high‑performance PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # dimensions
    stride_am, stride_an, stride_ak,   # strides for A (M,N,K)
    stride_bm, stride_bk, stride_bj,   # strides for B (M,K,N) – B is read transposed
    stride_cm, stride_cn, stride_co,   # strides for C (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[m,i,j] = Σ_k A[m,i,k] * B[m,j,k]   (B is accessed transposed).
    Each program instance works on one (m, i_tile, j_tile) block.
    """
    pid_m = tl.program_id(0)      # batch * hidden index
    pid_i = tl.program_id(1)      # tile over rows i
    pid_j = tl.program_id(2)      # tile over cols j

    # Offsets within the tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Boundary masks for i and j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers for the current tile
    a_ptrs = a_ptr + pid_m * stride_am + offs_i[:, None] * stride_an + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_m * stride_bm + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        k_valid = (cur_k + offs_k) < K

        # Load tiles with appropriate masking
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & k_valid[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_j[None, :],
                    other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

        # Advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write result back in fp16
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_m * stride_cm + offs_i[:, None] * stride_cn + offs_j[None, :] * stride_co
    mask = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, c, mask=mask)


def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (bool/int/float) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to fp16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Apply mask (if any) and gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp16
            right = right * mask_fp16

    # ------------------------------------------------------------------
    # 7) Core reduction via Triton batched GEMM:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Merge batch and hidden dimensions for the batched matmul
    M = B * H
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(M, N, N)   # (M, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(M, N, N) # (M, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Block sizes – tuned for B200 (shared‑memory friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        M,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _matmul_transpose_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        M=M,
        N=N,
        K=N,
        stride_am=left_flat.stride(0),
        stride_an=left_flat.stride(1),
        stride_ak=left_flat.stride(2),
        stride_bm=right_flat.stride(0),
        stride_bk=right_flat.stride(1),   # second dim (original row) – acts as K stride after transpose
        stride_bj=right_flat.stride(2),   # third dim – column stride after transpose
        stride_cm=out_flat.stride(0),
        stride_cn=out_flat.stride(1),
        stride_co=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
