def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Perform LayerNorm and linear transformations using PyTorch
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"],
                          bias=weights["norm.bias"])
    x = x.to(torch.float16)
    left_proj = F.linear(x, weights["left_proj.weight"], None)
    right_proj = F.linear(x, weights["right_proj.weight"], None)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"], None))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"], None))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"], None))

    # Apply mask if provided and not in nomask mode
    if not nomask and mask is not None:
        mask = mask.to(torch.float16)
        left_proj = left_proj * mask.unsqueeze(-1)
        right_proj = right_proj * mask.unsqueeze(-1)

    # Combine projected and gated tensors
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Define mapping parameters and grid
    B = input_tensor.size(0)
    N = input_tensor.size(1)
    C = dim
    H = hidden_dim
    total = B * N * N * H
    grid = (1, 1)
    num_warps = 8
    num_stages = 3
    num_blocks = (total + BLOCK_SIZE - 1) // BLOCK_SIZE

    # Initialize output tensor
    left_out = torch.empty_like(input_tensor)
    right_out = torch.empty_like(input_tensor)
    init_left_out()
    init_right_out()

    # Launch kernel to compute the reduction and gating
    if mask is None or nomask:
        kernel_name = "tri_mul_kernel"
    else:
        kernel_name = "tri_mul_kernel_mask"

    # Compute the kernel execution configuration
    config = dict(
        meta = dict(
            grid=(num_blocks,),   # Number of blocks in x dimension
            num_warps=num_warps,  # Number of warps per block
            num_stages=num_stages,
        )
    )

    # Run kernel with experimental parser
    piolid_annKernelCompile(config, kernel_name, left, right, left_gate, right_gate, mask, nomask)
