import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)   # row block
    pid_n = tl.program_id(1)   # col block
    pid_bh = tl.program_id(2)  # batch*head index

    b = pid_bh // H
    h = pid_bh % H

    # start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block (M x K)
        a_ptr = left_ptr + \
                b * stride_left_b + \
                h * stride_left_h + \
                offs_m[:, None] * stride_left_i + \
                offs_k[None, :] * stride_left_k
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load right block (N x K)
        b_ptr = right_ptr + \
                b * stride_right_b + \
                h * stride_right_h + \
                offs_n[:, None] * stride_right_j + \
                offs_k[None, :] * stride_right_k
        b_val = tl.load(b_ptr, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # Compute dot product across K (a @ b^T)
        acc += tl.dot(a, b_val, trans_b=True)

    # Store result
    out_ptrs = out_ptr + \
               b * stride_out_b + \
               h * stride_out_h + \
               offs_m[:, None] * stride_out_i + \
               offs_n[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def _tri_mul_batched(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute batched triangular multiplication:
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    left, right: (B, N, N, H) FP16, contiguous
    returns out: same shape, FP16
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left)
    # strides
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_j = right.stride(1)
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # tuning constants
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (tl.cdiv(N, BLOCK_M),),
        (tl.cdiv(N, BLOCK_N),),
        (B * H,),
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,

    This implementation:
      1. Layer‑norms the input (PyTorch).
      2. Casts to FP16.
      3. Performs linear projections and gates.
      4. Optionally applies a mask.
      5. Executes the core O(N³·H) contraction with a custom Triton kernel.
      6. Applies a second layer‑norm, output‑gate and final linear projection.
      7. Returns the result in the original dtype.
    """

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Apply mask (if any) and fuse with gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape [B, N, N] -> broadcast to hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left_proj * mask_fp16 * left_gate
        right = right_proj * mask_fp16 * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core contraction via Triton kernel
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = _tri_mul_batched(left, right)  # (B, N, N, H), fp16

    # ------------------------------------------------------------------
    # 8) Output layer‑norm, output‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (PyTorch).
2. Cast to FP16 for the compute‑intensive part.
3. Linear projections (left/right) and three gates (left, right, out) are
   performed with bias‑free linear layers (FP16).
4. If a mask is provided and ``nomask`` is False, the mask is fused with the
   projections and gates (simple elementwise multiplication).
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (B·H independent NxN matrix multiplies) realised
   with a custom Triton kernel that tiles the output matrix and loops over
   the reduction dimension K.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and project back to the original channel dimension.
7. Cast the result back to the input dtype.

The kernel operates on FP16 tensors and accumulates in FP32 for numerical
stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched TriMul: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]."""
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile column
    pid_bh = tl.program_id(2)  # combined batch‑head index

    b = pid_bh // H            # batch index
    h = pid_bh % H             # hidden‑dim index

    # offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block (M × K)
        a_ptr = left_ptr + \
                b * stride_left_b + \
                h * stride_left_h + \
                offs_m[:, None] * stride_left_i + \
                offs_k[None, :] * stride_left_k
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right block (N × K)
        b_ptr = right_ptr + \
                b * stride_right_b + \
                h * stride_right_h + \
                offs_n[:, None] * stride_right_j + \
                offs_k[None, :] * stride_right_k
        b_val = tl.load(b_ptr, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # accumulate a @ b^T over K
        acc += tl.dot(a, b_val, trans_b=True).to(tl.float32)

    # store the result tile
    out_ptrs = out_ptr + \
               b * stride_out_b + \
               h * stride_out_h + \
               offs_m[:, None] * stride_out_i + \
               offs_n[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _tri_mul_batched(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched TriMul using the Triton kernel.
    Both inputs must be contiguous FP16 tensors of shape (B, N, N, H).
    Returns an FP16 tensor of the same shape.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides (in elements)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # tile sizes – chosen to give good occupancy on modern GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows
        (N + BLOCK_N - 1) // BLOCK_N,   # cols
        B * H,                          # batch * hidden
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]

    """
    Forward pass of the outgoing TriMul (AlphaFold3) using Triton.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] (float32 or float16)
            - mask: [B, N, N] or None
            - weights: dict of model parameters
            - config: {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns:
        Tensor of shape [B, N, N, dim] with the same dtype as ``input_tensor``.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Apply mask (if any) and fuse with gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left_proj * mask_fp16 * left_gate
        right = right_proj * mask_fp16 * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core TriMul contraction via Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = _tri_mul_batched(left, right)   # (B, N, N, H), FP16

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, output‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != input_tensor.dtype:
        out = out.to(input_tensor.dtype)

    return out
