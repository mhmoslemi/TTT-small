@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    P, N,
    stride_ap, stride_ai, stride_ak,
    stride_bp, stride_bi, stride_bk,
    stride_cp, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute p, block_i, block_j
    num_block_i = tl.cdiv(N, BLOCK_M)
    num_block_j = tl.cdiv(N, BLOCK_N)
    p = pid // (num_block_i * num_block_j)
    block_idx = pid % (num_block_i * num_block_j)
    block_i = block_idx // num_block_j
    block_j = block_idx % num_block_j

    # Compute global offsets
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A sub-block (M x K)
        a_ptrs = a_ptr + (p * stride_ap + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B sub-block (N x K) note that B is transposed. We need B_T which is (P,N,N) with j dimension first, k second (since we transposed). We need to load B_T's rows j (block_j) and columns k.
        b_ptrs = b_ptr + (p * stride_bp + offs_j[:, None] * stride_bi + offs_k[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # Accumulate using dot
        acc += tl.dot(a, b)

        # Move to next k tile automatically by loop increment
    # Write output
    c_ptrs = c_ptr + (p * stride_cp + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

    # F.linear expects input ( ..., dim ), weight (out_features, in_features). We have weight shapes: left_proj.weight shape (hidden_dim, dim). So F.linear(x, left_proj_w) returns shape (B,N,N,hidden_dim)
    left = F.linear(x, left_proj_w)        # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # step 4: gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # step 5: mask gate fusion (if mask exists)
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()
        BLOCK = 1024  # may tune else
        grid = (triton.cdiv(total_elements, BLOCK),)
        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # delete gate tensors after usage? Actually we used left_gate and right_gate now fused; we can delete.
        del left_gate, right_gate
    else:
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # step 6: batched matmul using Triton
    # Permute to (B, hidden_dim, N, N)
    left_b = left.permute(0, 3, 1, 2).contiguous()
    right_b = right.permute(0, 3, 1, 2).contiguous()
    # Transpose right for matmul
    right_t = right_b.transpose(-2, -1).contiguous()
    B, H, N, _ = left_b.shape
    P = B * H
    # reshape
    A = left_b.reshape(P, N, N)
    B_mat = right_t.reshape(P, N, N)

    # allocate output
    C = torch.empty_like(A)  # fp16
    # Determine block sizes; try 64x64
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_prog = P * num_blocks_m * num_blocks_n
    grid = (total_prog,)

    # compute strides
    stride_a_p, stride_a_i, stride_a_k = A.stride()
    stride_b_p, stride_b_i, stride_b_k = B_mat.stride()
    stride_c_p, stride_c_i, stride_c_j = C.stride()

    batched_matmul_kernel[grid](
        A,
        B_mat,
        C,
        P,
        N,
        stride_a_p,
        stride_a_i,
        stride_a_k,
        stride_b_p,
        stride_b_i,
        stride_b_k,
        stride_c_p,
        stride_c_i,
        stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # reshape back to (B, H, N, N)
    out = C.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,H)

    # step 7: second layernorm over hidden_dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # step 8: apply output gate
    out = out * out_gate

    # step 9: final linear projection
    out = F.linear(out, to_out_w)

    # step 10: cast to float32
    out = out.to(torch.float32)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor (float32) → cast to fp16.
2. Linear projections (left/right) from C → H using fp16 GEMM.
3. Compute three sigmoid gates (left, right, out) with fp16 linear layers.
4. Fuse mask (if present) and the left/right gate multiplications in‑place
   with a small Triton kernel.
5. Core O(N³·H) reduction:
   - Reshape left/right to (B·H, N, N) and transpose the right tensor.
   - Perform a batched matrix multiplication A @ Bᵀ using a custom
     Triton kernel that tiles the i, j, k dimensions (BLOCK_M×BLOCK_N×BLOCK_K).
6. Layer‑normalize over the hidden dimension H.
7. Apply the output gate (element‑wise).
8. Final linear projection H → C.
9. Cast the result back to float32.

All heavy compute (mask‑gate fusion and the batched GEMM) is done in Triton,
while lightweight linear layers and activations use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,            # fp16 * left  (B,N,N,H) flattened
    right_ptr,           # fp16 * right (B,N,N,H) flattened
    mask_ptr,            # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,       # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,      # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,     # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion: left = left * mask * left_gate,
                      right = right * mask * right_gate."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load tensors
    l  = tl.load(left_ptr  + offs, mask=mask)
    r  = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr  + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Compute mask index (broadcast across the hidden dimension)
    idx = offs // H               # (b,i,j) index
    m = tl.load(mask_ptr + idx, mask=mask)

    # Apply fused multiplication
    tl.store(left_ptr  + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs,  r * m * rg, mask=mask)


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    P, N,
    stride_ap, stride_ai, stride_ak,   # strides for A (P,N,N)
    stride_bp, stride_bk, stride_bj,   # strides for B (P,N,N) where dim1 = k, dim2 = j
    stride_cp, stride_ci, stride_cj,   # strides for C (P,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ B (both with shape (P,N,N)) using tiling."""
    pid = tl.program_id(0)

    # grid decomposition
    num_block_i = tl.cdiv(N, BLOCK_M)
    num_block_j = tl.cdiv(N, BLOCK_N)

    p = pid // (num_block_i * num_block_j)
    block_idx = pid % (num_block_i * num_block_j)
    block_i = block_idx // num_block_j
    block_j = block_idx % num_block_j

    # offsets within the tile
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block (M x K)
        a_ptrs = a_ptr + (
            p * stride_ap
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B block (K x N) – note B is stored as (P, k, j)
        b_ptrs = b_ptr + (
            p * stride_bp
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write the result
    c_ptrs = c_ptr + (
        p * stride_cp
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc.to(tl.float16),
              mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see keys in code)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask and gate multiplications (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()                     # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates have been fused; free memory
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication using Triton
    #    Compute out[p,i,j] = Σ_k left[p,i,k] * right[p,j,k]
    # ------------------------------------------------------------------
    # Rearrange tensors to (B, H, N, N)
    left_b  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Transpose the right tensor so that the multiplication becomes A @ B
    right_t = right_b.transpose(-2, -1).contiguous()  # (B, H, N, N) with dims (p, k, j)

    B_, H_, N, _ = left_b.shape
    P = B_ * H_

    A = left_b.reshape(P, N, N)          # (P, N, N)
    B_mat = right_t.reshape(P, N, N)     # (P, N, N)

    C = torch.empty_like(A)              # output buffer (fp16)

    # Tiling parameters (tuned for typical B200 GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = P * num_blocks_m * num_blocks_n
    grid = (total_programs,)

    # Strides for the flattened tensors
    stride_ap, stride_ai, stride_ak = A.stride()
    stride_bp, stride_bk, stride_bj = B_mat.stride()
    stride_cp, stride_ci, stride_cj = C.stride()

    _batched_matmul_kernel[grid](
        A,
        B_mat,
        C,
        P,
        N,
        stride_ap,
        stride_ai,
        stride_ak,
        stride_bp,
        stride_bk,
        stride_bj,
        stride_cp,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, N, N, H)
    out = C.reshape(B_, H_, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply the output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast to float32 (required output dtype)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
