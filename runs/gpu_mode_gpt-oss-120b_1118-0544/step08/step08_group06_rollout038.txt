@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bn, stride_bk,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0) # batch_hidden index
    pid_i = tl.program_id(1) # row tile index
    pid_j = tl.program_id(2) # col tile index

    # Offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # Load A block (M, i, k)
        a = tl.load(a_ptr + pid_m * stride_am + offs_i[:, None] * stride_an + offs_k[None, :] * stride_ak,

        # Compute dot

def triton_tri_mul_kernel_launcher(left, right):
    # left/right shape (B,N,N,hidden_dim)
    # permute and view to (B*hidden_dim, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous()
    right_flat = right.permute(0, 3, 1, 2).contiguous()
    B, N, _, H = left.shape
    M = B * H
    left_flat = left_flat.view(M, N, N)
    right_flat = right_flat.view(M, N, N)
    out_flat = torch.empty_like(left_flat)

    # get strides
    stride_am, stride_an, stride_ak = left_flat.stride()
    stride_bm, stride_bn, stride_bk = right_flat.stride()
    stride_cm, stride_cn, stride_ck = out_flat.stride()
    # compute grid dimensions
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (M, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)

    tri_mul_kernel[grid](
        left_flat, right_flat, out_flat,
        M, N, N,
        stride_am, stride_an, stride_ak,
        stride_bm, stride_bn, stride_bk,
        stride_cm, stride_cn, stride_ck,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()
    return out

"""
TriMul (outgoing) forward implementation with Triton fusion.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for the compute‑intensive parts.
3. Linear projections (left / right) and gating (left / right / out) using
   the supplied weights.
4. Fuse projection, gating and optional mask with a custom Triton kernel.
5. Core TriMul reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM A @ Bᵀ for each hidden slice.
   Implemented as a Triton kernel that processes all (batch × hidden) slices
   in parallel, accumulating in fp32 and storing fp16.
6. Apply output LayerNorm, multiply by `out_gate`, and a final linear
   projection back to the original dimension.
7. Cast the result back to the input dtype.

The kernel in step 4 and the kernel in step 5 are the parts written in
Triton; the rest uses PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Kernel 1: fuse projection, gate and mask (if present)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B,N,N,H)
    right_proj_ptr,  # fp16 * (B,N,N,H)
    left_gate_ptr,   # fp16 * (B,N,N,H)
    right_gate_ptr,  # fp16 * (B,N,N,H)
    mask_ptr,        # fp16 * (B,N,N)
    left_out_ptr,    # fp16 * (B,N,N,H)
    right_out_ptr,   # fp16 * (B,N,N,H)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # mask for out‑of‑bounds
    valid = offs < total_elems

    # index into the mask (broadcast over hidden dim)
    mask_idx = offs // hidden_dim

    left_proj = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gate and mask for both left and right tensors using a
    single Triton kernel. All tensors must be fp16, contiguous and have
    shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)
    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Kernel 2: core TriMul reduction (batched GEMM A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr,          # fp16 * (M, N, K)  (left)
    b_ptr,          # fp16 * (M, N, K)  (right)
    c_ptr,          # fp16 * (M, N, N)  (output)
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bn, stride_bk,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # index over M = batch * hidden
    pid_i = tl.program_id(1)          # row tile index
    pid_j = tl.program_id(2)          # col tile index

    # Offsets for the output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        mask_i = offs_i < N
        mask_k = offs_k < K
        mask_a = mask_i[:, None] & mask_k[None, :]

        # Load A block: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_m * stride_am
            + offs_i[:, None] * stride_an
            + offs_k[None, :] * stride_ak,
            mask=mask_a,
            other=0.0,
        )

        # Load B block transposed: we need Bᵀ, i.e. (K, N)
        mask_j = offs_j < N
        mask_b = mask_k[:, None] & mask_j[None, :]

        b = tl.load(
            b_ptr
            + pid_m * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bn,
            mask=mask_b,
            other=0.0,
        )

        # Convert to fp32 for accumulation
        a_fp32 = a.to(tl.float32)
        b_fp32 = b.to(tl.float32)

        # Dot product
        acc += tl.dot(a_fp32, b_fp32)

    # Write back the result (cast to fp16)
    c = acc.to(tl.float16)
    mask_out = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(
        c_ptr
        + pid_m * stride_cm
        + offs_i[:, None] * stride_cn
        + offs_j[None, :] * stride_ck,
        c,
        mask=mask_out,
    )


def _tri_mul_forward(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    left / right : (B, N, N, H)   fp16
    Returns       : (B, N, N, H)   fp16
    """
    B, N, _, H = left.shape
    M = B * H  # combine batch and hidden dimensions

    # Reshape to (M, N, N) → contiguous layout
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(M, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(M, N, N)
    out_flat = torch.empty_like(left_flat)

    # Strides
    stride_am, stride_an, stride_ak = left_flat.stride()
    stride_bm, stride_bn, stride_bk = right_flat.stride()
    stride_cm, stride_cn, stride_ck = out_flat.stride()

    # Block sizes (tuned for typical B200 GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        M,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N,
        N,
        stride_am,
        stride_an,
        stride_ak,
        stride_bm,
        stride_bn,
        stride_bk,
        stride_cm,
        stride_cn,
        stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (bool/int/float) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast weights to fp16 (biases are None for projections)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Apply mask & gating (fused in Triton when mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous fp16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        # No mask or mask ignored – simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask exists but we still ignore it (should not happen)
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 7) Core TriMul reduction (batched GEMM A @ Bᵀ)
    # ------------------------------------------------------------------
    out = _tri_mul_forward(left, right)  # (B, N, N, H), fp16

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
