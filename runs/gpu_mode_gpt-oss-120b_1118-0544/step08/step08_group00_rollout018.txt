"""
TriMul forward (outgoing) implemented with Triton.
Algorithm
---------
1. Layer‑norm over the channel dimension (dim) using provided weights.
2. Cast to FP16 for compute‑heavy ops.
3. Linear projections (left/right) and corresponding gates (sigmoid).
4. Optional element‑wise masking (broadcasted over the hidden dim).
5. Core reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (left @ rightᵀ) with a custom
   Triton kernel (tiling over M,N,K).
6. Second LayerNorm over the hidden dim, multiply by the output gate,
   and linear projection back to the original dimension.
7. Cast back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Optional


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,           # a: (batch, M, K)
    stride_bk, stride_bn,           # b: (batch, K, N)
    stride_cm, stride_cn,           # c: (batch, M, N)
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication:
        C[b, m, n] = Σ_k A[b, m, k] * B[b, k, n]
    All tensors are expected to be in fp16; accumulation is in fp32.
    """
    pid_m = tl.program_id(0)    # block row
    pid_n = tl.program_id(1)    # block col
    pid_b = tl.program_id(2)    # batch index (combined batch*head)

    # Block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile
        a_ptrs = a_ptr + \
                 pid_b * batch_stride_a + \
                 offs_m[:, None] * stride_am + \
                 offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)

        # Load B tile
        b_ptrs = b_ptr + \
                 pid_b * batch_stride_b + \
                 offs_k[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)

        # Cast to fp32 and accumulate
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # Write back
    c_ptrs = c_ptr + \
             pid_b * batch_stride_c + \
             offs_m[:, None] * stride_cm + \
             offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

    # ------------------------------------------------------------------
    # 1) LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to fp16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (fp16) and gates
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid() # (B, N, N, H)
    out_gate = F.linear(x, out_gate_w).sigmoid()     # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask (if present) and combine projection with gate
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape: (B, N, N) -> broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16)
        mask_fp16 = mask_fp16[..., None]  # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (left @ rightᵀ) with a Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # Reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Transpose right for the required multiplication
    right_T = right_flat.transpose(1, 2).contiguous()   # (B*H, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        N, N, N,
        left_flat.stride(1), left_flat.stride(2),   # stride_am, stride_ak
        right_T.stride(1), right_T.stride(2),       # stride_bk, stride_bn
        out_flat.stride(1), out_flat.stride(2),     # stride_cm, stride_cn
        left_flat.stride(0), right_T.stride(0), out_flat.stride(0),   # batch strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
