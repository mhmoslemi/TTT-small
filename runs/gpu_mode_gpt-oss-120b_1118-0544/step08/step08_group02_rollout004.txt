"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The kernel fuses the element‑wise mask, linear projection and gate
operations for the left and right branches of the TriMul module.
The heavy reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed with a highly‑optimized batched GEMM (torch.bmm) on
FP16 tensors.  The final LayerNorm, output‑gate multiplication and
output linear projection are applied in FP32/FP16 as appropriate.
The function returns a float32 tensor of shape [B, N, N, C].
"""

import torch
import torch.nn.functional as F
from typing import Tuple, Dict
import triton
import triton.language as tl


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,    # fp16 * (B, N, N, H)
    right_proj_ptr,   # fp16 * (B, N, N, H)
    left_gate_ptr,    # fp16 * (B, N, N, H)
    right_gate_ptr,   # fp16 * (B, N, N, H)
    mask_ptr,         # fp16 * (B, N, N)
    left_out_ptr,     # fp16 * (B, N, N, H)
    right_out_ptr,    # fp16 * (B, N, N, H)
    total_elems,      # i64 = B * N * N * H
    hidden_dim,       # i64 = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gate and mask:
    out = proj * gate * mask   (mask broadcast over hidden_dim)
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # mask for out‑of‑range threads
    valid = offs < total_elems

    # Corresponding mask index (one mask element per hidden_dim entries)
    mask_idx = offs // hidden_dim

    # Load operands
    lp = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Fuse
    left_out = lp * lg * m
    right_out = rp * rg * m

    # Store
    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Apply mask, projection and gate in a single Triton kernel.
    All tensors must be contiguous fp16 with shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)
    Returns left, right tensors of the same shape.
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict with model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive ops
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast all linear weights to FP16 (bias‑free)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections and gates (FP16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)         # (B, N, N, H)

    left_gate  = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid()  # (B, N, N, H)
    out_gate   = F.linear(x, out_gate_w).sigmoid()    # (B, N, N, H)

    # --------------------------------------------------------------
    # 5) Fuse mask, projection and gate (Triton when mask is used)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            block_size=1024,
        )
    else:
        # No mask or mask ignored
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # Mask present but configuration says ignore – still broadcast
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # --------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM (B*H, N, N) @ (B*H, N, N)^T
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 8) Cast back to original dtype (float32)
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
