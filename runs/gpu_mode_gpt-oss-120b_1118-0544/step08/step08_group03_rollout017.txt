"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input (float32 → fp16).
2. Fuse the five linear projections (left, left_gate, right, right_gate, out_gate)
   into a single GEMM by concatenating their weight matrices.
3. Split the concatenated result, apply sigmoid to the three gate tensors.
4. Fuse mask, left‑gate and right‑gate multiplications in‑place with a Triton kernel.
5. Perform the core reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched matrix multiplication (fp16 inputs, fp32 accumulation).
6. Layer‑normalize over the hidden dimension, apply the output gate.
7. Final linear projection back to the original channel dimension.
8. Cast the result to the original dtype.

The Triton kernel is only used for the mask‑gate fusion; all heavy
linear operations use PyTorch’s highly‑optimised cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    mask_ptr,          # fp16 *
    left_gate_ptr,     # fp16 *
    right_gate_ptr,    # fp16 *
    H: tl.constexpr,   # hidden dimension
    total_elems: tl.constexpr,   # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load tensors
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Broadcast mask (one mask value per hidden element)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Apply fused operations
    tl.store(left_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (binary) or None
            - weights: dict of model parameters (see keys below)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)  # fp16 for the rest of the pipeline

    # ------------------------------------------------------------------
    # 2) Fuse the five linear projections into a single GEMM
    # ------------------------------------------------------------------
    # Concatenate weight matrices (all are (hidden, dim) and bias=False)
    proj_weights = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )  # shape (5*hidden, dim)
    proj_weights = proj_weights.to(device=device, dtype=torch.float16)

    # One large linear projection
    combined = F.linear(x, proj_weights)  # (B, N, N, 5*hidden)

    # ------------------------------------------------------------------
    # 3) Split combined tensor and apply sigmoid to gates
    # ------------------------------------------------------------------
    h = hidden
    left = combined[..., 0:h]
    left_gate = combined[..., h:2 * h]
    right = combined[..., 2 * h:3 * h]
    right_gate = combined[..., 3 * h:4 * h]
    out_gate = combined[..., 4 * h:5 * h]

    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 4) Fuse mask and gate multiplications (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()  # B * N * N * hidden
        BLOCK = 65536  # threads per program; large enough for good occupancy
        grid = (triton.cdiv(total_elements, BLOCK),)

        # Ensure tensors are contiguous for flat indexing
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask_fp16.contiguous()

        _mask_gate_fusion_kernel[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates are no longer needed after fusion
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5) Core reduction: batched matrix multiplication over the sequence dim
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) for batched matmul
    left_b = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    # out_b[b, h, i, j] = Σ_k left_b[b, h, i, k] * right_b[b, h, j, k]
    out_b = torch.matmul(left_b, right_b.transpose(-2, -1))

    # Back to original layout (B, N, N, hidden)
    out = out_b.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
    )

    # ------------------------------------------------------------------
    # 7) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, C)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
