"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1.  Layer‑norm the input (float32 for stability) and cast to float16.
2.  Linear projections *left* and *right* from C → H.
3.  Compute three sigmoid gates (left, right, out) with linear layers.
4.  Fuse mask (if present) and gate multiplications in‑place via a Triton kernel:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
   (When no mask is supplied, the same computation is done with PyTorch.)
5.  Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication using torch.matmul.
6.  Layer‑norm over the hidden dimension H.
7.  Apply the output gate.
8.  Final linear projection H → C.
9.  Cast the result back to float32.

The kernel only handles the mask‑gate fusion; all other heavy linear
operations use torch’s optimized kernels (cuBLAS/torch‑nn). This design
keeps the implementation simple while still providing a non‑trivial
Trident acceleration.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,            # fp16 * left  [B,N,N,H] flattened
    right_ptr,           # fp16 * right [B,N,N,H] flattened
    mask_ptr,            # fp16 * mask  [B,N,N]   flattened
    left_gate_ptr,       # fp16 * left_gate  [B,N,N,H] flattened
    right_gate_ptr,      # fp16 * right_gate [B,N,N,H] flattened
    H: tl.constexpr,     # hidden dimension
    TOTAL_ELEMS: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and left/right gates.

    left  = left  * mask * left_gate
    right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELEMS

    # Load data
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Map each element to the corresponding mask entry (broadcast over H)
    mask_idx = offs // H          # integer division
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply multiplication and store back in‑place
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see code for expected keys)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32) → cast to float16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float32)

    x_fp32 = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x_fp32.half()   # fp16 for the rest of the pipeline

    # ------------------------------------------------------------------
    # 2) Cast linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    # (B, N, N, H)
    left = F.linear(x, left_proj_w)          # no bias
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Fuse mask and gates (in‑place) via Triton when mask is present
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Cast mask to fp16 for the kernel
        mask_fp16 = mask.to(device=device, dtype=torch.float16)

        total_elements = left.numel()                # B * N * N * H
        BLOCK = 1024                                  # threads per program
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_kernel[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates are no longer needed after fusion
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication over seq_len (k)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for torch.matmul
    left_b  = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)

    # Matrix multiplication with right transposed on the last two dims
    out_b = torch.matmul(left_b, right_b.transpose(-2, -1))  # (B, H, N, N)

    # Back to original layout (B, N, N, H)
    out = out_b.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to float32 (required output dtype)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
