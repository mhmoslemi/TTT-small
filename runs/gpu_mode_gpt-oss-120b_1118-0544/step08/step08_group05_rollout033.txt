"""
custom_kernel: Optimized outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to float16 for compute‑intensive parts.
3. Linear projections (left/right) and gating projections (left/right/out).
4. Fuse mask, projection and gate for left/right tensors with a small Triton
   kernel (only when a mask is present).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication where each hidden slice `h`
   performs `left_h @ right_hᵀ`. Implemented as a custom Triton kernel.
6. Output LayerNorm, multiply by `out_gate`, final linear projection back to `dim`.
7. Cast back to the original dtype (usually float32).

All heavy arithmetic is performed in float16 (accumulated in float32) and
only the mask‑gate fusion and the TriMul reduction are written in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Kernel 1: fuse mask, projection and gating (used only when a mask is present)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B,N,N,H)
    right_proj_ptr,  # fp16 * (B,N,N,H)
    left_gate_ptr,   # fp16 * (B,N,N,H)
    right_gate_ptr,  # fp16 * (B,N,N,H)
    mask_ptr,        # fp16 * (B,N,N)
    left_out_ptr,    # fp16 * (B,N,N,H)
    right_out_ptr,   # fp16 * (B,N,N,H)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds
    valid = offs < total_elems

    # Index of the corresponding mask element (each mask repeats `hidden_dim` times)
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    left_out = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """Fuse proj‑gate‑mask for both left and right tensors using Triton."""
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Kernel 2: core TriMul reduction (batched matmul with left * rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 * (B*H, N, N)
    right_ptr,         # fp16 * (B*H, N, N)
    out_ptr,           # fp16 * (B*H, N, N)
    N,                 # i64
    stride_l_batch,    # i64
    stride_l_row,      # i64
    stride_l_col,      # i64
    stride_r_batch,    # i64
    stride_r_row,      # i64
    stride_r_col,      # i64
    stride_o_batch,    # i64
    stride_o_row,      # i64
    stride_o_col,      # i64
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)               # batch * hidden slice
    pid_m = tl.program_id(0)               # row tile
    pid_n = tl.program_id(1)               # col tile

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the beginning of the current (batch, head) slice
    left_batch_ptr  = left_ptr  + pid_b * stride_l_batch
    right_batch_ptr = right_ptr + pid_b * stride_r_batch
    out_batch_ptr   = out_ptr   + pid_b * stride_o_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = left_batch_ptr + (offs_m[:, None] * stride_l_row
                                   + offs_k[None, :] * stride_l_col)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile: (BLOCK_N, BLOCK_K)  (rows of right)
        b_ptrs = right_batch_ptr + (offs_n[:, None] * stride_r_row
                                    + offs_k[None, :] * stride_r_col)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # Accumulate in FP32
        acc += tl.dot(a, b).to(tl.float32)

    # Write back
    c = acc.to(tl.float16)
    out_ptrs = out_batch_ptr + (offs_m[:, None] * stride_o_row
                                + offs_n[None, :] * stride_o_col)
    tl.store(out_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor of shape [B, N, N] (0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(dtype_in)
    norm_b = weights["norm.bias"].to(dtype_in)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # --------------------------------------------------------------
    compute_dtype = torch.float16
    x = x.to(compute_dtype)

    # --------------------------------------------------------------
    # 3) Cast all linear weights to FP16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(compute_dtype)
    right_proj_w  = weights["right_proj.weight"].to(compute_dtype)
    left_gate_w   = weights["left_gate.weight"].to(compute_dtype)
    right_gate_w  = weights["right_gate.weight"].to(compute_dtype)
    out_gate_w    = weights["out_gate.weight"].to(compute_dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype)
    to_out_w      = weights["to_out.weight"].to(compute_dtype)

    # --------------------------------------------------------------
    # 4) Linear projections and gates (FP16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)          # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 5) Fuse mask, projection and gate (Triton) if needed
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(compute_dtype).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6) Core TriMul reduction (batched matmul via Triton)
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Move hidden dimension to the batch axis for a simple 3‑D layout
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    # Strides for a contiguous (batch, N, N) tensor in row‑major order
    stride_batch = N * N
    stride_row   = N
    stride_col   = 1

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_batch, stride_row, stride_col,   # left strides
        stride_batch, stride_row, stride_col,   # right strides
        stride_batch, stride_row, stride_col,   # out strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 8) Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(dtype_in)
    return out
