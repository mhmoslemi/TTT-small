def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_head, M, N, K,
    stride_lbh, stride_lm, stride_lk,
    stride_rbh, stride_rk, stride_rn,
    stride_obh, stride_om, stride_on,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    bh = tl.program_id(0) # batch*head index
    # Compute offsets for block row and column
    pid_m = tl.program_id(1) # block row
    pid_n = tl.program_id(2) # block col

    # Compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension

            left_ptr + bh*stride_lbh + offs_m[:, None] * stride_lm + offs_k[None, :] * stride_lk,

@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B_H, M, N, K,
    stride_lbh, stride_lm, stride_lk,
    stride_rbh, stride_rk, stride_rn,
    stride_obh, stride_om, stride_on,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # guard mask for k dimension
        mask_k = offs_k < K

        # Load left block (M x K)
        left = tl.load(
            left_ptr + bh * stride_lbh + offs_m[:, None] * stride_lm + offs_k[None, :] * stride_lk,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load right block (K x N) using transposed access (since right shape (B_H, N, K))
        # We need right_T element at (k, n) = right[n, k]
        right = tl.load(
            right_ptr + bh * stride_rbh + offs_k[:, None] * stride_rk + offs_n[None, :] * stride_rn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )

        # Dot product accumulate
        acc += tl.dot(left, right)

    # Store result
    out = acc.to(tl.float16)
    tl.store(
        out_ptr + bh * stride_obh + offs_m[:, None] * stride_om + offs_n[None, :] * stride_on,
        out,
        mask=mask_m[:, None] & mask_n[None, :]
    )

"""
TriMul outgoing (AlphaFold3) forward pass – Triton‑accelerated.

Algorithm
---------
1. Layer‑norm over the input channel dimension `dim`.
2. Cast to FP16 for compute‑intensive ops.
3. Linear projections for left/right tensors and their gates (no bias).
4. Fuse projection, gate and optional mask with a small Triton kernel.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication (left @ rightᵀ) via a
   custom Triton kernel that tiles the `i` and `j` dimensions.
6. Output LayerNorm, multiply by `out_gate`, final linear projection back to
   `dim`.
7. Cast back to the original input dtype.

Both the mask‑gate fusion and the N³ reduction are executed on the GPU
using Triton kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B*N*N*H,)
    right_proj_ptr,  # fp16 * (B*N*N*H,)
    left_gate_ptr,   # fp16 * (B*N*N*H,)
    right_gate_ptr,  # fp16 * (B*N*N*H,)
    mask_ptr,        # fp16 * (B*N*N,)
    left_out_ptr,    # fp16 * (B*N*N*H,)
    right_out_ptr,   # fp16 * (B*N*N*H,)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gate and mask for both left and right tensors."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # Load tensors
    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Each mask element repeats `hidden_dim` times in the flattened tensor
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Fuse
    left_out = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


@triton.jit
def _trimul_matmul_kernel(
    left_ptr,    # fp16 * (B*H, N, N)
    right_ptr,   # fp16 * (B*H, N, N)   (accessed transposed)
    out_ptr,     # fp16 * (B*H, N, N)
    batch_head,  # i64 = B*H
    M, N, K,     # i64 = N (M=N=output rows, N=output cols, K=inner dim)
    stride_lbh, stride_lm, stride_lk,   # i64 strides for left
    stride_rbh, stride_rk, stride_rn,   # i64 strides for right
    stride_obh, stride_om, stride_on,   # i64 strides for out
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication: out = left @ rightᵀ  (per hidden slice)."""
    bh = tl.program_id(0)          # batch * hidden index
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block column

    # Compute global offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a block of left: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr
            + bh * stride_lbh
            + offs_m[:, None] * stride_lm
            + offs_k[None, :] * stride_lk,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a block of right transposed: shape (BLOCK_K, BLOCK_N)
        # right is stored as (B*H, N, K); we need rightᵀ => (K, N)
        right = tl.load(
            right_ptr
            + bh * stride_rbh
            + offs_k[:, None] * stride_rk
            + offs_n[None, :] * stride_rn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc += tl.dot(left, right)

    # Write back result
    out = acc.to(tl.float16)
    tl.store(
        out_ptr
        + bh * stride_obh
        + offs_m[:, None] * stride_om
        + offs_n[None, :] * stride_on,
        out,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (0/1) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast model weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Fuse projection, gate and optional mask (Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguously stored in FP16
        mask_fp16 = mask.to(torch.float16).contiguous()
        total = left_proj.numel()
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        BLOCK_SIZE = 2048
        grid = (triton.cdiv(total, BLOCK_SIZE),)

        _proj_gate_mask_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            left,
            right,
            total,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        # Simple element‑wise fusion when no mask is used
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched matrix multiplication via Triton.
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and then flatten batch*head dimension
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left.shape
    bh = B * H

    left_flat  = left.view(bh, N, N)   # (B*H, N, N)
    right_flat = right.view(bh, N, N)  # (B*H, N, N)
    out_flat   = torch.empty_like(left_flat)

    # Kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        bh,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _trimul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        bh,
        N,          # M
        N,          # N
        N,          # K (inner dimension)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype (normally float32)
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
