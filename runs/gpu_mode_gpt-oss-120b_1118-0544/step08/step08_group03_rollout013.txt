"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1.  Layer‑normalize the input (float32 for numeric stability).
2.  Cast to fp16 and apply two linear projections (left / right) to the hidden
    dimension.
3.  Compute three sigmoid gates (left, right, out) from the same normalized
    input.
4.  Fuse mask, left‑gate and right‑gate with a custom Triton kernel
    (in‑place element‑wise multiplication).  If no mask is supplied, the
    multiplication is done with a simple PyTorch expression.
5.  Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    This is realised as a batched matrix‑multiply using torch.matmul on tensors
    shaped (B, H, N, N).
6.  Layer‑normalize over the hidden dimension.
7.  Apply the output gate.
8.  Final linear projection back to the original channel dimension.
9.  Cast the result back to the input dtype (float32).

Only the mask‑gate fusion is performed in Triton; all other operations rely on
high‑performance PyTorch kernels (cuBLAS / cuDNN).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # *fp16  (B,N,N,H) flattened
    right_ptr,         # *fp16  (B,N,N,H) flattened
    mask_ptr,          # *fp16  (B,N,N)   flattened
    left_gate_ptr,     # *fp16  (B,N,N,H) flattened
    right_gate_ptr,    # *fp16  (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and left/right gates.

    left  = left  * mask * left_gate
    right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # load the data
    l  = tl.load(left_ptr + offs, mask=mask)
    r  = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # each mask element spans H hidden entries
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # apply and store back
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict containing all model parameters:
                * norm.weight, norm.bias
                * left_proj.weight, right_proj.weight
                * left_gate.weight, right_gate.weight, out_gate.weight
                * to_out_norm.weight, to_out_norm.bias
                * to_out.weight
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the remaining pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left = F.linear(
        x, weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    )
    right = F.linear(
        x, weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    )

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(device=device, dtype=torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(
            x, weights["right_gate.weight"].to(device=device, dtype=torch.float16)
        )
    )
    out_gate = torch.sigmoid(
        F.linear(
            x, weights["out_gate.weight"].to(device=device, dtype=torch.float16)
        )
    )

    # ------------------------------------------------------------------
    # 5) Fuse mask and gates (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure contiguous tensors for pointer arithmetic
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask.to(device=device, dtype=torch.float16).contiguous()

        total_elements = left.numel()                     # B * N * N * H
        BLOCK = 8192                                      # threads per program
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left_ptr=left,
            right_ptr=right,
            mask_ptr=mask_fp16,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            H=hidden_dim,
            total_elements=total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates are no longer needed after fusion
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication over the sequence dim
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Bring hidden dimension to the second axis for torch.matmul
    left_b = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Perform batched matmul: (B, H, N, N) @ (B, H, N, N)^T on last two dims
    out_b = torch.matmul(left_b, right_b.transpose(-2, -1))

    # Restore original layout (B, N, N, H)
    out = out_b.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"]
        .to(device=device, dtype=torch.float16)
        .contiguous(),
        bias=weights["to_out_norm.bias"]
        .to(device=device, dtype=torch.float16)
        .contiguous(),
    )

    # ------------------------------------------------------------------
    # 8) Apply the output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(
        out, weights["to_out.weight"].to(device=device, dtype=torch.float16)
    )

    # ------------------------------------------------------------------
    # 10) Cast back to the original dtype (float32)
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
