    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_lm, stride_lk,
    stride_rk, stride_rn,
    stride_om, stride_on,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_n = tl.program_id(1)
    pid_z = tl.program_id(2)  # batch*head index

    # Starting offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute valid mask
    mask_m = offs_m < M
    mask_n = offs_n < N

    right_batch_ptr = right_ptr + pid_z * stride_rk * K  # stride_rk * K? Actually right stride for batch dimension is stride_rk * K? need actual stride for batch dimension.

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input (float32) → fp16.
2. Linear projections (left/right) from C → H using fp16 GEMM.
3. Compute three sigmoid gates (left, right, out) with a linear layer.
4. Fuse mask, left‑gate and right‑gate element‑wise multiplications in a
   Triton kernel (in‑place, no temporaries).
5. Core reduction: for each batch and head compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM in Triton (no extra transposes).
6. Layer‑normalize over the hidden dimension H.
7. Apply the output gate (element‑wise).
8. Final linear projection H → C.
9. Cast the result back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,            # fp16 * left  (B,N,N,H) flattened
    right_ptr,           # fp16 * right (B,N,N,H) flattened
    mask_ptr,            # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,       # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,      # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,     # hidden dimension
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and the two gate tensors:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # Load tensors
    l = tl.load(left_ptr + offs, mask=valid, other=0.0)
    r = tl.load(right_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # Mask has no hidden dimension; broadcast it.
    mask_idx = offs // H                     # index in (B,N,N)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Apply fused multiplication
    tl.store(left_ptr + offs,  l * m * lg, mask=valid)
    tl.store(right_ptr + offs, r * m * rg, mask=valid)


@triton.jit
def _tri_mul_matmul_kernel(
    left_ptr,            # fp16 * left  (B*H, M, K) flattened
    right_ptr,           # fp16 * right (B*H, K, N) flattened (already transposed)
    out_ptr,             # fp16 * out   (B*H, M, N) flattened
    M: tl.constexpr,     # = N (sequence length)
    N: tl.constexpr,
    K: tl.constexpr,     # = N
    stride_l_batch: tl.constexpr,
    stride_l_m: tl.constexpr,
    stride_l_k: tl.constexpr,
    stride_r_batch: tl.constexpr,
    stride_r_k: tl.constexpr,
    stride_r_n: tl.constexpr,
    stride_o_batch: tl.constexpr,
    stride_o_m: tl.constexpr,
    stride_o_n: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul core:
       out[b] = left[b] @ right[b]   where left: (M,K), right: (K,N)
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_z = tl.program_id(2)   # batch * head index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the current batch matrix
    left_batch_ptr  = left_ptr  + pid_z * stride_l_batch
    right_batch_ptr = right_ptr + pid_z * stride_r_batch
    out_batch_ptr   = out_ptr   + pid_z * stride_o_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of left (M x K)
        a = tl.load(
            left_batch_ptr + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a tile of right (K x N)
        b = tl.load(
            right_batch_ptr + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c = acc.to(tl.float16)
    out_ptrs = out_batch_ptr + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    tl.store(out_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see code for expected keys)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for compute‑heavy parts
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask and gate multiplications (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask expected to be binary (0/1); cast to fp16
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()                     # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates are now fused; free memory
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication (TriMul)
    #    left: (B, N, N, H) -> (B, H, N, N)
    #    right: (B, N, N, H) -> (B, H, N, N) then transpose last two dims
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N)
    left_b = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2) # (B, H, N, N)

    # Transpose right for the GEMM: (B, H, N, N) -> (B, H, N, N) with swapped last axes
    right_t = right_b.permute(0, 1, 3, 2)  # (B, H, N, N) where dims are (b,h,k,i)

    B, H, N_seq, _ = left_b.shape
    # Flatten batch & head dimensions for the kernel
    left_mat = left_b.contiguous().view(-1, N_seq, N_seq)   # (B*H, M, K)
    right_mat = right_t.contiguous().view(-1, N_seq, N_seq) # (B*H, K, N)
    out_mat = torch.empty_like(left_mat)                    # (B*H, M, N)

    # Strides (in elements)
    stride_l_batch, stride_l_m, stride_l_k = left_mat.stride()
    stride_r_batch, stride_r_k, stride_r_n = right_mat.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_mat.stride()

    # Choose block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N_seq, BLOCK_M),   # blocks over M
        triton.cdiv(N_seq, BLOCK_N),   # blocks over N
        left_mat.shape[0],             # B*H
    )

    _tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N_seq,          # M
        N_seq,          # N
        N_seq,          # K
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_k,
        stride_r_n,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast to float32 (required output dtype)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
