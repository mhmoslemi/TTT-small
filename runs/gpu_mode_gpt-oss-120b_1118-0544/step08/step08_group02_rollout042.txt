"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to fp16 and apply linear projections (left/right) and gating linear layers.
3. Fuse mask, projection and gating with a small Triton kernel.
4. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (A @ Bᵀ) with a custom Triton kernel that
   processes each (batch * hidden_dim) matrix pair in blocks.
5. Apply a second LayerNorm, output‑gate multiplication and a final linear
   projection back to the original channel dimension.
6. Cast the result back to the input dtype.

Only the mask‑gate fusion and the batched GEMM are performed in Triton;
all other operations use PyTorch (highly‑optimized cuBLAS/cuDNN kernels).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Kernel 1: fuse mask, projection and gate (left & right branches)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)
    right_gate_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    left_out_ptr,    # fp16 * (B, N, N, H)
    right_out_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # int64  = B * N * N * H
    hidden_dim,      # int64  = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # bounds check
    valid = offs < total_elems

    # each element belongs to a (b,i,j) entry; mask is broadcast over H
    mask_idx = offs // hidden_dim

    # load tensors
    left_proj = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # fused element‑wise multiplication
    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    # store results
    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse mask, projection and gate for both left and right tensors using Triton.

    All tensors must be fp16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)

    Returns
    -------
    left, right : torch.Tensor
        Shape (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the core TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute, for each batch index `b`:
        C[b] = A[b] @ B[b].T    where A,B have shape (M, K) and (N, K)
    The tensors are provided as flattened 3‑D tensors (batch, M, K) and
    (batch, N, K).  Accumulation is performed in fp32 and the result is stored
    as fp16.
    """
    pid = tl.program_id(0)

    # --- decode program id -------------------------------------------------
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_tiles_m * num_tiles_n

    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    pid_m = tile_idx // num_tiles_n
    pid_n = tile_idx % num_tiles_n

    # pointers to the beginning of the current batch
    a_batch = a_ptr + batch_idx * batch_stride_a
    b_batch = b_ptr + batch_idx * batch_stride_b
    c_batch = c_ptr + batch_idx * batch_stride_c

    # --- block offsets -----------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --- loop over K dimension --------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # load a tile of A : (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )
        # load a tile of B : (BLOCK_K, BLOCK_N)   (note the transpose)
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # store result
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to fp16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gates (all fp16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)       # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)     # (B, N, N, hidden_dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))     # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5) Apply mask + gating (fused via Triton when mask is used)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16, block_size=1024
        )
    else:
        # simple element‑wise fusion (mask ignored or absent)
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (A @ Bᵀ) using Triton.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)  # fp16 output

    # kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    batch = left_flat.shape[0]
    tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch * tiles_m * tiles_n,)

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        N,                     # K
        left_flat.stride(1),   # stride_am (row stride)
        left_flat.stride(2),   # stride_ak (col stride)
        right_flat.stride(1),  # stride_bk (row stride of K dim)
        right_flat.stride(2),  # stride_bn (col stride of N dim)
        out_flat.stride(1),    # stride_cm
        out_flat.stride(2),    # stride_cn
        left_flat.stride(0),   # batch stride A
        right_flat.stride(0),  # batch stride B
        out_flat.stride(0),    # batch stride C
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, output‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
