@triton.jit
def _tri_mul_kernel(left_ptr, right_ptr, out_ptr, N, BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program ids
    batch_h = tl.program_id(0)
    i_tile = tl.program_id(1)
    j_tile = tl.program_id(2)

    # compute offsets
    i_start = i_tile * BLOCK_M
    j_start = j_tile * BLOCK_N

    # base pointer offset for this batch_h (multiple of N*N)
    offset_batch = batch_h * N * N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K

        # compute indices for left: rows i, cols k
        i_range = i_start + tl.arange(0, BLOCK_M)
        k_range = k_start + tl.arange(0, BLOCK_K)
        # mask for i,k within bounds
        i_mask = i_range < N
        k_mask = k_range < N
        # combine for load mask
        mask_i_k = i_mask[:, None] & k_mask[None, :]

        left_idx = offset_batch + i_range[:, None] * N + k_range[None, :]
        left = tl.load(left_ptr + left_idx, mask=mask_i_k, other=0.0)

        # compute indices for right: rows j, cols k
        j_range = j_start + tl.arange(0, BLOCK_N)
        # j_mask
        j_mask = j_range < N
        mask_j_k = j_mask[:, None] & k_mask[None, :]

        right_idx = offset_batch + j_range[:, None] * N + k_range[None, :]
        right = tl.load(right_ptr + right_idx, mask=mask_j_k, other=0.0)

        # compute block multiplication: left (M,K) * right (N,K)^T -> (M,N)
        # Use tl.dot with trans_b=True to treat right as (K,N)
        acc += tl.dot(left, right, trans_b=True)

    # store result
    # compute output index for each (i,j) in block
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)
    i_mask = i_range < N
    j_mask = j_range < N
    mask_out = i_mask[:, None] & j_mask[None, :]

    out_idx = offset_batch + i_range[:, None] * N + j_range[None, :]
    # convert acc to fp16 for storage
    tl.store(out_ptr + out_idx, acc.to(tl.float16), mask=mask_out)

def _proj_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):

    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # Compute mask index (same per hidden dim)
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)

    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = lp * lg * m

"""
TriMul outgoing (AlphaFold3) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm over the channel dimension `dim`.
2. Cast to fp16 for compute‑intensive path.
3. Linear projections (left/right) and gate projections (left/right/out).
4. Fuse projection, gate and optional mask with a small Triton kernel
   (produces `left` and `right` tensors of shape [B, N, N, H]).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom tiled GEMM kernel in Triton (fp16 × fp16 → fp32
   accumulation, fp16 store).
6. Output layer‑norm, out‑gate, final linear projection back to `dim`.
7. Cast back to the original dtype (usually fp32).

All heavy linear algebra (projections, final linear) uses PyTorch’s cuBLAS
kernels; the mask‑gate fusion and the O(N³) reduction are performed with
Triton for maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 *
    right_proj_ptr,  # fp16 *
    left_gate_ptr,   # fp16 *
    right_gate_ptr,  # fp16 *
    mask_ptr,        # fp16 *
    left_out_ptr,    # fp16 *
    right_out_ptr,   # fp16 *
    total_elems,     # i64
    hidden_dim,      # i64
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gate and mask for both left and right tensors."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds.
    in_bounds = offs < total_elems

    # Each mask element repeats `hidden_dim` times in the flattened tensor.
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=in_bounds, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=in_bounds, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=in_bounds, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=in_bounds, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=in_bounds, other=1.0)

    left_out  = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs,  left_out,  mask=in_bounds)
    tl.store(right_out_ptr + offs, right_out, mask=in_bounds)


@triton.jit
def _tri_mul_kernel(
    left_ptr,   # fp16 *
    right_ptr,  # fp16 *
    out_ptr,    # fp16 *
    N: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    The tensors are laid out as (B*H, N, N) in row‑major order.
    """
    batch_h = tl.program_id(0)      # index over B*H
    i_tile  = tl.program_id(1)      # tile along i dimension
    j_tile  = tl.program_id(2)      # tile along j dimension

    i_start = i_tile * BLOCK_M
    j_start = j_tile * BLOCK_N

    # base pointer for this batch_h
    batch_offset = batch_h * N * N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K

        # ----- load left tile (i, k) -----
        i_range = i_start + tl.arange(0, BLOCK_M)
        k_range = k_start + tl.arange(0, BLOCK_K)

        i_valid = i_range < N
        k_valid = k_range < N
        mask_left = i_valid[:, None] & k_valid[None, :]

        left_idx = batch_offset + i_range[:, None] * N + k_range[None, :]
        left = tl.load(left_ptr + left_idx, mask=mask_left, other=0.0)

        # ----- load right tile (j, k) -----
        j_range = j_start + tl.arange(0, BLOCK_N)
        j_valid = j_range < N
        mask_right = j_valid[:, None] & k_valid[None, :]

        right_idx = batch_offset + j_range[:, None] * N + k_range[None, :]
        right = tl.load(right_ptr + right_idx, mask=mask_right, other=0.0)

        # Multiply left (M,K) with right (N,K)^T  →  (M,N)
        acc += tl.dot(left, right, trans_b=True)

    # ----- store result -----
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)
    i_valid = i_range < N
    j_valid = j_range < N
    mask_out = i_valid[:, None] & j_valid[None, :]

    out_idx = batch_offset + i_range[:, None] * N + j_range[None, :]
    tl.store(out_ptr + out_idx, acc.to(tl.float16), mask=mask_out)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (0/1) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to fp16 (no bias)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Fuse projection, gate and optional mask
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous FP16 and broadcastable
        mask_fp16 = mask.to(torch.float16).contiguous()
        B, N, _, H = left_proj.shape
        total_elems = left_proj.numel()
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        BLOCK_SIZE = 1024
        grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

        _proj_gate_mask_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            left,
            right,
            total_elems,
            H,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction (TriMul) – custom Triton GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # ensure shapes are up‑to‑date
    # Move hidden dimension to second axis for easier flattening
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Combine batch and hidden dimensions
    BH = B * H
    left_flat = left.view(BH, N, N)
    right_flat = right.view(BH, N, N)

    out_flat = torch.empty_like(left_flat)

    # Tile sizes – chosen to give good occupancy on modern GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the input dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
