import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fusion_inplace(left_ptr, right_ptr, mask_ptr,
                              left_gate_ptr, right_gate_ptr,
                              H: tl.constexpr,
                              total_elements: tl.constexpr,
                              BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load values
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # compute mask index: each H consecutive elements belong to same mask entry
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply
    tl.store(left_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)

    nomask = config.get("nomask", False)

        weight=ln_weight,
        bias=ln_bias,

    # -------------------------------------------------------------

    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)

    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)

Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The kernel follows the PyTorch reference implementation:

1. Layer‑norm over the channel dimension.
2. Linear projections `left` and `right` from C → H (hidden_dim).
3. Three sigmoid gates (left, right, out) computed with linear layers.
4. In‑place fusion of the optional mask and the left/right gates using a
   tiny Triton kernel (no extra temporaries).
5. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication for each hidden channel.
6. Layer‑norm over the hidden dimension, element‑wise output‑gate,
   and final linear projection back to C.
7. Cast back to the original dtype.

Only the mask‑gate fusion is performed in Triton; the heavy linear layers
and the batched matmul are delegated to PyTorch/cuBLAS for maximal
throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and left/right gates.

    left  = left  * mask * left_gate
    right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load tensors
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # One mask value per hidden channel → integer division by H
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply and write back in‑place
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)

    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) for torch.bmm
    left_b = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_b = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    out_b = torch.bmm(left_b, right_b.transpose(1, 2))  # (B*H, N, N)

    # back to (B, N, N, H)
    out = out_b.reshape(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
```
