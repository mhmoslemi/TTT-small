"""
TriMul outgoing (AlphaFold3) – forward pass.

Algorithm
---------
1. Layer‑norm over the input channel dimension `dim`.
2. Cast to FP16 for compute‑heavy ops.
3. Linear projections (left/right) and gating (left/right/out) using FP16
   weights.  Gating uses a sigmoid.
4. Fuse projection, gate and optional mask with a tiny Triton kernel:
       left  = left_proj  * left_gate  * mask
       right = right_proj * right_gate * mask
   (mask is omitted when `nomask` is True or mask is None.)
5. Core TriMul reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM with torch.bmm on tensors of shape
   (B*H, N, N) → (B*H, N, N).
6. Output layer‑norm over the hidden dimension, apply the out‑gate, and
   final linear projection back to `dim`.
7. Cast the result back to the original dtype (typically float32).

The only non‑trivial operation executed on the GPU via Triton is the
mask‑gate fusion (step 4).  All other compute‑intensive parts use
highly‑optimized cuBLAS kernels through `torch.nn.functional.linear` and
`torch.bmm`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B*N*N*H,)
    right_proj_ptr,  # fp16 * (B*N*N*H,)
    left_gate_ptr,   # fp16 * (B*N*N*H,)
    right_gate_ptr,  # fp16 * (B*N*N*H,)
    mask_ptr,        # fp16 * (B*N*N,)
    left_out_ptr,    # fp16 * (B*N*N*H,)
    right_out_ptr,   # fp16 * (B*N*N*H,)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gate and mask for left and right tensors."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # each mask element repeats `hidden_dim` times in the flattened tensor
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(left_out_ptr + offs,  lp * lg * m, mask=mask)
    tl.store(right_out_ptr + offs, rp * rg * m, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask         : torch.Tensor of shape [B, N, N] (0/1) or None
            - weights      : dict containing model parameters (float32)
            - config       : dict containing:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast compute to FP16
    # ------------------------------------------------------------------
    compute_dtype = torch.float16
    x = x.to(compute_dtype)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(compute_dtype)
    right_proj_w  = weights["right_proj.weight"].to(compute_dtype)
    left_gate_w   = weights["left_gate.weight"].to(compute_dtype)
    right_gate_w  = weights["right_gate.weight"].to(compute_dtype)
    out_gate_w    = weights["out_gate.weight"].to(compute_dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype)
    to_out_w      = weights["to_out.weight"].to(compute_dtype)

    # ------------------------------------------------------------------
    # 4) Linear projections + gates (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Fuse mask, projection and gate (Triton) if needed
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure contiguous FP16 mask
        mask_fp16 = mask.to(compute_dtype).contiguous()
        B, N, _, H = left_proj.shape
        total = left_proj.numel()
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        BLOCK_SIZE = 1024
        grid = (triton.cdiv(total, BLOCK_SIZE),)

        _proj_gate_mask_kernel[grid](
            left_proj,    # input pointers
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            left,
            right,
            total,
            H,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N).T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)
    return out
