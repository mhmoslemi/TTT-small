"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input over the channel dimension (dim).
2. Cast to fp16 for the heavy compute.
3. Linear projections (left/right) and linear gates (left/right/out) are computed
   with torch.nn.functional.linear.  The gates are passed through a sigmoid.
4. If a mask is provided and not disabled, a Triton kernel fuses
   mask‑application, projection and gating:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
   otherwise simple element‑wise multiplication is performed.
5. Core reduction: for every head h,
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is implemented as a batched GEMM using torch.bmm on tensors of shape
   (B*H, N, N).  The operation is equivalent to an einsum over the k‑dimension.
6. Apply a second LayerNorm over the hidden dimension, multiply by the out‑gate,
   and a final linear projection back to the original dimension.
7. Cast the result back to the input dtype.

Only the mask‑gate‑fusion step is performed in Triton, satisfying the
requirement that a part of the computation be a custom kernel.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, projection and gate for left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)
    right_gate_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    left_out_ptr,    # fp16 * (B, N, N, H)
    right_out_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # i64 = B * N * N * H
    hidden_dim,      # i64 = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑range threads
    valid = offs < total_elems

    # Each element belongs to a (b,i,j) triplet; the mask index is
    # floor(element_index / H)
    mask_idx = offs // hidden_dim

    # Load values (masked loads)
    left_proj = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Fuse mask, projection and gate
    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    # Store results
    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse mask, projection and gate for both left and right tensors using Triton.

    All tensors must be FP16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)

    Returns
    -------
    left, right : torch.Tensor
        Shape (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

# Main entrypoint
# ----------------------------------------------------------------------

    """

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # 4) Linear projections and gates (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)            # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)          # (B, N, N, H)

    left_gate = F.linear(x, left_gate_w).sigmoid()  # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid()# (B, N, N, H)
    out_gate = F.linear(x, out_gate_w).sigmoid()    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask + gating (fused via Triton when mask is used)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            block_size=1024,
        )
    else:
        # simple element‑wise fusion (mask ignored or absent)
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but disabled → still broadcast if needed
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 6) Core reduction:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM across the hidden dimension.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matrix multiplication
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
