"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor (float32 → fp16 for the rest).
2. Linear projections `left` and `right` from C → H (hidden_dim) using fp16.
3. Compute three sigmoid gates (left, right, out) via linear layers.
4. Fuse mask (if present) and the left/right gates in‑place with a small
   Triton kernel to avoid extra temporaries and extra memory traffic.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication
   (B, H, N, N) × (B, H, N, N)ᵀ using torch.matmul (cuBLAS/Tensor Cores).
6. Layer‑normalize over the hidden dimension H.
7. Apply the output gate.
8. Final linear projection H → C.
9. Cast back to the original dtype.

The kernel only handles the mask‑gate fusion; all other heavy
operations rely on highly‑optimised cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,   # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and left/right gates:
           left  = left  * mask * left_gate
           right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # Load values
    l = tl.load(left_ptr + offs, mask=valid)
    r = tl.load(right_ptr + offs, mask=valid)
    lg = tl.load(left_gate_ptr + offs, mask=valid)
    rg = tl.load(right_gate_ptr + offs, mask=valid)

    # Broadcast mask: one mask value per hidden element
    mask_idx = offs // H            # integer division → index in mask
    m = tl.load(mask_ptr + mask_idx, mask=valid)

    # Apply and store back in‑place
    tl.store(left_ptr + offs,  l * m * lg, mask=valid)
    tl.store(right_ptr + offs, r * m * rg, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see keys used below)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    # Move layernorm parameters to the correct device/dtype
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float32)

    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    # Cast to FP16 for the rest of the pipeline
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    # Shapes: (B, N, N, H)
    left = F.linear(x, left_proj_w)      # left_proj: dim -> hidden_dim
    right = F.linear(x, right_proj_w)    # right_proj: dim -> hidden_dim

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Fuse mask and gates (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Cast mask to fp16 and flatten
        mask_fp16 = mask.to(device=device, dtype=torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)

        # Ensure tensors are contiguous for the kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_elements = left.numel()                     # B * N * N * H
        BLOCK_SIZE = 1024                                 # threads per program
        grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_flat,
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK_SIZE,
            num_warps=4,
        )
        # Gates are no longer needed after fusion
        del left_gate, right_gate
    else:
        # Simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication per hidden head
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for a standard batched matmul
    left_b = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Perform matmul: (B, H, N, N) @ (B, H, N, N)^T → (B, H, N, N)
    out_b = torch.matmul(left_b, right_b.transpose(-2, -1))

    # Back to original layout (B, N, N, H)
    out = out_b.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
