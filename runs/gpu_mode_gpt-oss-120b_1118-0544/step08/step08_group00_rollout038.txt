@triton.jit
def batched_matmul_kernel(
    A, B, C,
    batch, M, N, K,
    stride_Am, stride_Ak, stride_Ab,
    stride_Bk, stride_Bn, stride_Bb,
    stride_Cm, stride_Cn, stride_Cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    batch_idx = pid
    # block indices
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        # mask for out-of-bounds
        a_mask = (offs_m < M) & (offs_k < K)
        b_mask = (offs_k < K) & (offs_n < N)
        # load A block (M,K)
        a_ptrs = A + batch_idx * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)
        # load B block (K,N)
        b_ptrs = B + batch_idx * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)
        acc += tl.dot(a, b)
    # write output
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = C + batch_idx * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)
```

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Cast to FP16 for the compute‑intensive part.
3. Linear projections (left / right) and gating (left / right / out) are
   performed with bias‑free linear layers.  When a mask is present the
   projection, gating and masking are fused in a single Triton kernel.
4. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication where each hidden slice
   computes   (N,N) @ (N,N)ᵀ .  The implementation reshapes the tensors to
   (B*H, N, N) and uses a tiled Triton GEMM kernel (BLOCK_M/N/K = 64).
5. Apply a second LayerNorm, multiply by the output gate and linearly project
   back to the original dimension.
6. Cast the result back to the input dtype.

Both the mask‑gate‑projection fusion and the batched GEMM are written in
Triton, giving a large speed‑up over the pure‑PyTorch reference.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Optional


# ----------------------------------------------------------------------
# 1. Fuse projection, gate and optional mask for the left and right tensors
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)
    right_gate_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    left_out_ptr,    # fp16 * (B, N, N, H)
    right_out_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds
    in_bounds = offs < total_elems

    # Index of the element in the mask (mask has no hidden dim)
    mask_idx = offs // hidden_dim

    # Load tensors (out‑of‑bounds loads return 0.0)
    left_proj  = tl.load(left_proj_ptr + offs,   mask=in_bounds, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs,  mask=in_bounds, other=0.0)
    left_gate  = tl.load(left_gate_ptr + offs,   mask=in_bounds, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs,  mask=in_bounds, other=0.0)
    m          = tl.load(mask_ptr + mask_idx,   mask=in_bounds, other=1.0)

    # Fuse projection * gate * mask
    left_out  = left_proj  * left_gate  * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr  + offs, left_out,  mask=in_bounds)
    tl.store(right_out_ptr + offs, right_out, mask=in_bounds)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse element‑wise mask, projection and gate for both left and right tensors.
    All tensors must be FP16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)
    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# 2. Batched GEMM kernel for the core TriMul reduction:
#    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A,  # (batch, M, K)   fp16
    B,  # (batch, K, N)   fp16   (right transposed)
    C,  # (batch, M, N)   fp16   output
    batch, M, N, K,
    stride_Am, stride_Ak, stride_Ab,
    stride_Bk, stride_Bn, stride_Bb,
    stride_Cm, stride_Cn, stride_Cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Block start coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds
        a_mask = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        b_mask = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        # Load A block (M×K)
        a_ptrs = A + pid_batch * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)

        # Load B block (K×N)
        b_ptrs = B + pid_batch * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back result
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    c_ptrs = C + pid_batch * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def _batched_matmul(
    A: torch.Tensor,
    B: torch.Tensor,
    C: torch.Tensor,
    BLOCK_M: int = 64,
    BLOCK_N: int = 64,
    BLOCK_K: int = 64,
) -> None:
    """
    Perform C = A @ B where A : (batch, M, K), B : (batch, K, N).
    All tensors must be contiguous FP16.
    """
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match"

    # Strides (in elements)
    stride_Ab, stride_Am, stride_Ak = A.stride()
    stride_Bb, stride_Bk, stride_Bn = B.stride()
    stride_Cb, stride_Cm, stride_Cn = C.stride()

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        batch,
        M,
        N,
        K,
        stride_Am,
        stride_Ak,
        stride_Ab,
        stride_Bk,
        stride_Bn,
        stride_Bb,
        stride_Cm,
        stride_Cn,
        stride_Cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )


# ----------------------------------------------------------------------
# 3. Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, Optional[torch.Tensor], Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None (0/1)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # Cast to FP16 for compute‑heavy ops
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast linear weights to FP16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid() # (B, N, N, H)
    out_gate = F.linear(x, out_gate_w).sigmoid()     # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask, projection and gate (Triton kernel when needed)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
            block_size=1024,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but caller asked to ignore it → treat as 1
            left = left * mask.to(torch.float16)[..., None]
            right = right * mask.to(torch.float16)[..., None]

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a tiled batched GEMM in Triton.
    # ------------------------------------------------------------------
    # Reshape to (B, H, N, N) → (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)                     # (B*H, N, N)
    right_t = right_perm.transpose(2, 3).contiguous()           # (B, H, N, N) -> (B, H, N, N) with last dims swapped
    right_flat = right_t.view(B * H, N, N)                       # (B*H, N, N)  (now K dimension is second)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Batched matmul: (B*H, N, N) @ (B*H, N, N)^T  → (B*H, N, N)
    _batched_matmul(left_flat, right_flat, out_flat, BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
