"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the PyTorch reference:
1. Layer‑normalize the input (float32 → fp16).
2. Linear projections (left/right) to a hidden dimension H.
3. Compute three sigmoid gates (left, right, out).
4. Fuse mask, left‑gate and right‑gate element‑wise multiplications in a
   custom Triton kernel (in‑place, no temporaries).
5. Core reduction: for each batch and hidden head compute
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] via a batched
   matrix‑multiply (torch.matmul on (B·H, N, N) tensors).
6. Layer‑normalize over the hidden dimension.
7. Apply the output gate.
8. Final linear projection back to the original channel dimension C.
9. Cast the result to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place fusion of mask and the two gate tensors:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load data (masked loads return 0.0 for out‑of‑bounds)
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Compute the mask index (each H consecutive elements share the same mask)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # Fuse the three element‑wise multiplications
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see code for expected keys)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for the compute‑heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast linear weights to fp16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask & gate multiplications (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Convert mask to fp16 (0.0/1.0) and flatten
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()                     # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates are now fused; release memory
        del left_gate, right_gate
    else:
        # No mask → simple elementwise multiplication with gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    We reshape to (B*H, N, N) and use torch.matmul.
    # ------------------------------------------------------------------
    #   (B,N,N,H) -> (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(-1, left.shape[1], left.shape[2])
    right_mat = right.permute(0, 3, 1, 2).reshape(-1, right.shape[1], right.shape[2])

    # (B*H, N, N) = (B*H, N, N) @ (B*H, N, N)^T
    out_mat = torch.matmul(left_mat, right_mat.transpose(-2, -1))

    # Reshape back to (B, N, N, H)
    out = out_mat.reshape(left.shape[0], hidden_dim, left.shape[1], left.shape[2])
    out = out.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply the output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast to float32 (required output dtype)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
