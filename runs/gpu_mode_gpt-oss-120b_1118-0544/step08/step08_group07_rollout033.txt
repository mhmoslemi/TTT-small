"""
TriMul outgoing (AlphaFold3) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm over the input channel dimension `dim`.
2. Cast to FP16 for compute‑intensive parts.
3. Linear projections → left_proj, right_proj (no bias).
4. Gating projections → left_gate, right_gate, out_gate (sigmoid).
5. Fuse projection, gate and optional mask with a small Triton kernel.
6. Core reduction (TriMul) computed by a batched GEMM written in Triton:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   (equivalent to left @ rightᵀ per hidden slice h).
7. Output LayerNorm over the hidden dimension, multiply by `out_gate`,
   final linear projection back to `dim`.
8. Cast back to the original dtype (usually float32).

The kernel in step 5 and the custom batched matrix multiplication in step 6
are the parts written in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B*N*N*H,)
    right_proj_ptr,  # fp16 * (B*N*N*H,)
    left_gate_ptr,   # fp16 * (B*N*N*H,)
    right_gate_ptr,  # fp16 * (B*N*N*H,)
    mask_ptr,        # fp16 * (B*N*N,)
    left_out_ptr,    # fp16 * (B*N*N*H,)
    right_out_ptr,   # fp16 * (B*N*N*H,)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # guard
    mask = offs < total_elems

    # each mask element repeats `hidden_dim` times in the flat tensor
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(left_out_ptr  + offs, lp * lg * m, mask=mask)
    tl.store(right_out_ptr + offs, rp * rg * m, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel: batched TriMul (left @ rightᵀ) over hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,   # fp16 * (B, N, N, H)
    right_ptr,  # fp16 * (B, N, N, H)
    out_ptr,    # fp16 * (B, N, N, H)
    B, N, H,    # dimensions
    # strides for left
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    # strides for right
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    # strides for out
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(2)          # batch * hidden dimension
    pid_m  = tl.program_id(0)          # block row i
    pid_n  = tl.program_id(1)          # block col j

    b = pid_bh // H                   # batch index
    h = pid_bh % H                    # hidden index

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_base = kb * BLOCK_K
        cur_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left[i,k,h]
        a_ptr = (left_ptr
                 + b * stride_l_batch
                 + offs_i[:, None] * stride_l_i
                 + cur_k[None, :] * stride_l_k
                 + h * stride_l_h)
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right[j,k,h]  (we need (k,j) layout for dot)
        b_ptr = (right_ptr
                 + b * stride_r_batch
                 + offs_j[None, :] * stride_r_i
                 + cur_k[:, None] * stride_r_k
                 + h * stride_r_h)
        b = tl.load(b_ptr,
                    mask=mask_j[None, :] & mask_k[:, None],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # store result
    c_ptr = (out_ptr
             + b * stride_o_batch
             + offs_i[:, None] * stride_o_i
             + offs_j[None, :] * stride_o_j
             + h * stride_o_h)
    tl.store(c_ptr, acc.to(tl.float16),
              mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast weights to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (FP16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 5) Fuse projection, gate and mask (Triton)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        B, N, _, H = left_proj.shape
        total = left_proj.numel()
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        BLOCK_SIZE = 1024
        grid = (triton.cdiv(total, BLOCK_SIZE),)
        _proj_gate_mask_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            left,
            right,
            total,
            H,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6) Core reduction (TriMul) – custom batched GEMM in Triton
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # (B, N, N, H)

    # strides for the tensors (contiguous layout)
    stride_l = left.stride()   # (N*N*H, N*H, H, 1)
    stride_r = right.stride()
    stride_o = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        B * H,                     # batch * hidden
    )
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 8) Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(dtype_in)

    return out
