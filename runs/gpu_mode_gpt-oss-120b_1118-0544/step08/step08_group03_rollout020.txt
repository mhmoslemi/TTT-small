import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    H: tl.constexpr,
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Compute mask index (one mask per hidden dim)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(left_ptr + offs, l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    Implements the forward pass of the TriMul module described in AlphaFold3.
    The heavy linear transformations are performed with torch.linalg, while
    the element‑wise mask‑gate fusion is executed in a small Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast weights to fp16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5. Fuse mask and gates (in‑place) with Triton if needed
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(device=device, dtype=torch.float16).contiguous()
        total = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates have been fused, can drop them
        del left_gate, right_gate
    else:
        # Simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6. Core reduction: batched matrix multiplication per hidden dim
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    B, N, _, _ = left.shape
    left_b  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_b = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # out_b[b*h, i, j] = sum_k left_b[b*h, i, k] * right_b[b*h, j, k]
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_b.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8. Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)

    # ------------------------------------------------------------------
    # 10. Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and gates:
    left  = left  * mask * left_gate
    right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    l  = tl.load(left_ptr + offs,   mask=mask)
    r  = tl.load(right_ptr + offs,  mask=mask)
    lg = tl.load(left_gate_ptr + offs,  mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # One mask value per (b,i,j), repeated over H
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)

    Implements the forward pass of the TriMul module from AlphaFold3.

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,

    # 2) Cast all linear weights to fp16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
