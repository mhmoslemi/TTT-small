@triton.jit
def triton_tri_mul_out(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # compute batch and hidden id
    bh = pid_bh
    b = bh // H
    h = bh % H

    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # Mask for out-of-range
    i_mask = i < N
    j_mask = j < N

    # Compute base offsets
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        # Compute actual k indices
        k_idx = k_start + k

        # mask for k dimension
        k_mask = k_idx < N

        # Load A tile
        a_ptrs = left_base \
                 + (i[:, None] * stride_left_i) \
                 + (k_idx[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # Load B_T tile (k, j). Right memory layout is (b, j, k, h)
        # B_T tile expects rows = k, cols = j
        b_ptrs = right_base \
                 + (k_idx[:, None] * stride_right_k) \
                 + (j[None, :] * stride_right_j)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # Compute dot product
        # a shape: (M,K) ; b shape: (K,N)
        acc += tl.dot(a, b)

    # Write back to output
    out_ptrs = out_base \
               + (i[:, None] * stride_out_i) \
               + (j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input (float32 → fp16).
2. Linear projections (left/right) from C → H (hidden_dim) using fp16 GEMM.
3. Compute three sigmoid gates (left, right, out) with fp16 linear layers.
4. Fuse mask, left‑gate and right‑gate element‑wise multiplications in a
   Triton kernel (in‑place, no temporaries).
5. Core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (per (b,h) pair) in Triton, avoiding
   extra transposes and extra memory traffic.
6. Layer‑normalize over the hidden dimension H.
7. Apply the output gate.
8. Final linear projection H → C.
9. Cast result back to float32.

All heavy compute (mask‑gate fusion and the O(N³) contraction) is performed
with Triton kernels; the remaining light operations use PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask & gate fusion
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (flattened B*N*N*H)
    right_ptr,         # fp16 * right (flattened B*N*N*H)
    mask_ptr,          # fp16 * mask  (flattened B*N*N)
    left_gate_ptr,     # fp16 * left_gate  (flattened B*N*N*H)
    right_gate_ptr,    # fp16 * right_gate (flattened B*N*N*H)
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load tensors
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Compute index of the (b,i,j) triple to fetch mask value
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply fused multiplication
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the triangle multiplicative update
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left = (b,i,k,h)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right = (b,j,k,h)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for out = (b,i,j,h)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr
):
    pid_i = tl.program_id(0)  # tile row (i)
    pid_j = tl.program_id(1)  # tile column (j)
    pid_bh = tl.program_id(2) # combined batch & hidden index

    # --------------------------------------------------------------
    # Decode batch and hidden indices
    # --------------------------------------------------------------
    bh = pid_bh
    b = bh // H
    h = bh % H

    # --------------------------------------------------------------
    # Compute start coordinates of this tile
    # --------------------------------------------------------------
    i_base = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_base < N
    j_mask = j_base < N

    # Base pointers for the current (b, h) pair
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # --------------------------------------------------------------
    # Accumulator in float32
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_base < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base \
                 + (i_base[:, None] * stride_left_i) \
                 + (k_base[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=i_mask[:, None] & k_mask[None, :],
                    other=0.0)

        # Load B_T tile (k, j) directly from right which is stored as (b, j, k, h)
        b_ptrs = right_base \
                 + (k_base[:, None] * stride_right_k) \
                 + (j_base[None, :] * stride_right_j)
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & j_mask[None, :],
                    other=0.0)

        # Dot product (M,K)·(K,N) → (M,N); accumulate in FP32
        acc += tl.dot(a, b).to(tl.float32)

    # --------------------------------------------------------------
    # Write back the result (cast back to fp16)
    # --------------------------------------------------------------
    out_ptrs = out_base \
               + (i_base[:, None] * stride_out_i) \
               + (j_base[None, :] * stride_out_j)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=i_mask[:, None] & j_mask[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for the heavy part
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # --------------------------------------------------------------
    # 4) Compute sigmoid gates
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))    # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))   # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # --------------------------------------------------------------
    # 5) Fuse mask and gate multiplications (in‑place) via Triton
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) -> fp16 and flatten
        mask_fp16 = mask.to(device=device, dtype=torch.float16).contiguous()
        total_elements = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Release temporary gate tensors
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # Ensure contiguous layout for the GEMM kernel
    left = left.contiguous()
    right = right.contiguous()

    # --------------------------------------------------------------
    # 6) Core O(N³·H) reduction using Triton GEMM
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # Compute strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    _tri_mul_gemm[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 8) Apply output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 9) Final linear projection H → C
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # --------------------------------------------------------------
    # 10) Cast to float32 (required output dtype)
    # --------------------------------------------------------------
    out = out.to(torch.float32)

    return out
