"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

...
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_gate_fusion_inplace(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    H: tl.constexpr,
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < total_elements

    # Load the main tensors
    l = tl.load(left_ptr + offset, mask=mask)
    r = tl.load(right_ptr + offset, mask=mask)
    lg = tl.load(left_gate_ptr + offset, mask=mask)
    rg = tl.load(right_gate_ptr + offset, mask=mask)

    # Compute mask index (b,i,j) by integer division by H
    mask_idx = offset // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gates in place
    tl.store(left_ptr + offset, l * m * lg, mask=mask)
    tl.store(right_ptr + offset, r * m * rg, mask=mask)

def _apply_mask_and_gates(left, right, mask, left_gate, right_gate, hidden_dim):
    # All tensors are fp16 and contiguous
    total = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    mask_fp16 = mask.to(dtype=torch.float16)
    mask_flat = mask_fp16.view(-1)
    mask_gate_fusion_inplace[grid](
        left,
        right,
        mask_flat,
        left_gate,
        right_gate,
        hidden_dim,
        total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return left, right

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see keys below)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for compute‑heavy part
    x = x.to(torch.float16)

    # 2. Cast weights to fp16
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # 3. Linear projections (B,N,N,H)
    left  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w)  # (B,N,N,H)

    # 4. Compute sigmoid gates (B,N,N,H)
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # 5. Fuse mask and gate multiplication (in‑place) via Triton
    if mask is not None and not nomask:
        left, right = _apply_mask_and_gates(left, right, mask, left_gate, right_gate, hidden_dim)
        # Gates have been consumed; free memory
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # 6. Core reduction: batched matrix multiplication
    #   left, right : (B,N,N,H) -> (B*H,N,N)
    left_b  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_b = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    #   (B*H,N,N) x (B*H,N,N)^T  => (B*H,N,N)
    out_b = torch.matmul(left_b, right_b.transpose(-2, -1))

    #   reshape back to (B,N,N,H)
    out = out_b.permute(0, 2, 3, 1).contiguous()

    # 7. Second LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 8. Apply output gate
    out = out * out_gate

    # 9. Final linear projection back to original dimension C
    out = F.linear(out, to_out_w)

    # 10. Cast to float32
    return out.to(torch.float32)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input (float32 for stability) and cast to fp16.
2. Linear projections (left/right) from C → H using fp16 GEMM.
3. Compute three sigmoid gates (left, right, out) with a linear layer.
4. Fuse mask, left‑gate and right‑gate element‑wise multiplications in a
   single Triton kernel (in‑place, no temporaries).
5. Core reduction: batched matrix multiplication over the sequence dimension
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   Implemented via a single torch.matmul on tensors of shape (B, H, N, N)
   (cuBLAS / tensor‑cores).
6. Layer‑normalize over the hidden dimension H.
7. Apply the output gate (element‑wise).
8. Final linear projection H → C.
9. Cast the result back to float32.

The implementation uses mixed precision (fp16) for all compute‑heavy stages
and a custom Triton kernel for the mask‑gate fusion, which is the main
memory‑bound part of the operator.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_gate_fusion_inplace(
    left_ptr,            # fp16* left  (B,N,N,H) flattened
    right_ptr,           # fp16* right (B,N,N,H) flattened
    mask_ptr,            # fp16* mask  (B,N,N)   flattened
    left_gate_ptr,       # fp16* left_gate  (B,N,N,H) flattened
    right_gate_ptr,      # fp16* right_gate (B,N,N,H) flattened
    H: tl.constexpr,     # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,      # threads per program
):
    """
    In‑place fusion of mask and the two gate tensors.
    left  = left  * mask * left_gate
    right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load tensors
    l  = tl.load(left_ptr + offs,  mask=mask)
    r  = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs,  mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Derive mask index (b,i,j) from the flattened offset
    mask_idx = offs // H          # integer division
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply fused multiplication
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


def _apply_mask_and_gates(left, right, mask, left_gate, right_gate, hidden_dim):
    """Launch the Triton kernel that fuses mask and gate multiplications."""
    total = left.numel()
    BLOCK = 1024                      # threads per program
    grid = (triton.cdiv(total, BLOCK),)

    mask_fp16 = mask.to(dtype=torch.float16)
    mask_flat = mask_fp16.view(-1)

    mask_gate_fusion_inplace[grid](
        left,
        right,
        mask_flat,
        left_gate,
        right_gate,
        hidden_dim,
        total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return left, right


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see keys below)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)  # cast to fp16 for the heavy part

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask and gate multiplications (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        left, right = _apply_mask_and_gates(left, right, mask, left_gate, right_gate, hidden_dim)
        # Gates are no longer needed
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication
    #    left/right: (B,N,N,H) -> (B,H,N,N)
    # ------------------------------------------------------------------
    left_b  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # out_b shape: (B, H, N, N)
    out_b = torch.matmul(left_b, right_b.transpose(-2, -1))

    # Back to (B, N, N, H)
    out = out_b.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast to float32 (required output dtype)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
