def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16
    x = x.half()

    # Linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"].half())
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"].half())
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"].half()))

    # Apply mask and gating
    if mask is not None and not nomask:
        mask = mask.unsqueeze(-1).half()
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # TriMul kernel
    out = tri_mul_kernel(left, right, hidden_dim)

    # Output LayerNorm
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Multiply by out_gate (need broadcast to hidden_dim)
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"].half())

    # Cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out
```

```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(2)  # batch_h
    pid_x = tl.program_id(0)  # block_m
    pid_y = tl.program_id(1)  # block_n

    # map pid_z to batch b and hidden h
    b = pid_z // H
    h = pid_z % H

    # compute offsets
    # base pointers for each batch and channel
    left_batch = left_ptr + b * stride_left_b + h * stride_left_h
    right_batch = right_ptr + b * stride_right_b + h * stride_right_h
    out_batch = out_ptr + b * stride_out_b + h * stride_out_h

    # compute ranges
    offs_m = pid_x * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_y * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundary
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A tile: left[ b, i, k, h]
        a_ptrs = left_batch + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        # load B tile: right[ b, j, k, h] but we need transpose, so we use stride_right_k for k axis, stride_right_j for j axis
        b_ptrs = right_batch + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j

        # apply masks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot
        acc += tl.dot(a, b)

    # write output
    c_ptrs = out_batch + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

    B, N, _, H = left.shape
    assert H == hidden_dim
    out = torch.empty_like(left)

    stride_right = right.stride()
    stride_out = out.stride()

"""
custom_kernel: Triton‐accelerated outgoing Triangle Multiplicative Update (TriMul)

The implementation follows the reference PyTorch module but offloads the
O(N³·H) reduction to a custom Triton kernel. Steps
1. LayerNorm on the input.
2. FP16 casts for compute‑intensive tensors.
3. Linear projections (left/right) and gating (left/right/out) using
   torch.nn.functional.linear.
4. Optional mask fusion (element‑wise multiplication in FP16).
5. Core TriMul reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a blocked matrix‑multiply kernel where each hidden
   channel is treated as an independent batch.
6. Output LayerNorm, out‑gate, and final linear projection back to `dim`.
7. Cast back to the input dtype.

Only the reduction kernel is written in Triton; the rest uses PyTorch
(with mixed‑precision where beneficial).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel for the core TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 * (B, N, N, H)
    right_ptr,         # fp16 * (B, N, N, H)
    out_ptr,           # fp16 * (B, N, N, H)
    B, N, H,           # runtime sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program IDs
    pid_z = tl.program_id(2)          # batch * hidden index
    pid_x = tl.program_id(0)          # block row
    pid_y = tl.program_id(1)          # block col

    # Decode batch and hidden channel
    b = pid_z // H
    h = pid_z % H

    # Base pointers for this (b, h)
    left_batch  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_batch = right_ptr + b * stride_right_b + h * stride_right_h
    out_batch   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Offsets within the block
    offs_m = pid_x * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_y * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks for M and N dimensions
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile from left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_batch + \
                 offs_m[:, None] * stride_left_i + \
                 offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load a tile from right (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_batch + \
                 offs_k[:, None] * stride_right_k + \
                 offs_n[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # FP16 dot product, accumulate in FP32
        acc += tl.dot(a, b)

    # Write result back (cast to FP16)
    c_ptrs = out_batch + \
             offs_m[:, None] * stride_out_i + \
             offs_n[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _tri_mul_forward(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Launch the Triton kernel to compute the TriMul reduction.
    left, right: tensors of shape [B, N, N, H] (fp16, contiguous)
    Returns: out tensor of the same shape and dtype.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides (in elements) for each tensor
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    # Choose block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks in M (i)
        triton.cdiv(N, BLOCK_N),   # blocks in N (j)
        B * H                      # batch * hidden
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point as required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (0/1) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device=device, dtype=orig_dtype)
    norm_b = weights["norm.bias"].to(device=device, dtype=orig_dtype)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (no bias)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gates (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Fuse mask, projection and gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)  # (B,N,N,1)
        left = left_proj * left_gate * mask_fp
        right = right_proj * right_gate * mask_fp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguity for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction via Triton
    # ------------------------------------------------------------------
    out = _tri_mul_forward(left, right)   # (B,N,N,hidden_dim), fp16

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim), fp16

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
