"""
TriMul outgoing (AlphaFold3) – Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑norm over the input channel dimension `dim`.
2. Cast the normalized tensor to fp16 (all heavy compute in half‑precision).
3. Linear projections:
      left_proj = x @ W_left      (dim → hidden_dim, no bias)
      right_proj = x @ W_right
4. Gating projections + sigmoid:
      left_gate  = sigmoid(x @ W_left_gate)
      right_gate = sigmoid(x @ W_right_gate)
      out_gate   = sigmoid(x @ W_out_gate)
5. Fuse projection, gate and optional mask with a tiny Triton kernel:
      left  = left_proj  * left_gate  * mask
      right = right_proj * right_gate * mask
   (If `nomask` is True or mask is None the mask factor is omitted.)
6. Core reduction (TriMul):
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM: for each hidden slice h,
      out_h = left_h @ right_hᵀ .
7. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to the original `dim`.
8. Cast the result back to the input dtype (usually float32).

Only the mask‑gate fusion (step 5) is implemented in Triton; the rest
uses PyTorch’s highly‑optimized kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B*N*N*H,)
    right_proj_ptr,  # fp16 * (B*N*N*H,)
    left_gate_ptr,   # fp16 * (B*N*N*H,)
    right_gate_ptr,  # fp16 * (B*N*N*H,)
    mask_ptr,        # fp16 * (B*N*N,)
    left_out_ptr,    # fp16 * (B*N*N*H,)
    right_out_ptr,   # fp16 * (B*N*N*H,)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gate and mask for left and right tensors."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds accesses
    mask = offs < total_elems

    # Each mask entry repeats `hidden_dim` times in the flattened tensor
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Element‑wise fusion
    left_val  = lp * lg * m
    right_val = rp * rg * m

    tl.store(left_out_ptr + offs,  left_val,  mask=mask)
    tl.store(right_out_ptr + offs, right_val, mask=mask)

    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    x = F.layer_norm(

        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],

    # ------------------------------------------------------------------

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    # ------------------------------------------------------------------
