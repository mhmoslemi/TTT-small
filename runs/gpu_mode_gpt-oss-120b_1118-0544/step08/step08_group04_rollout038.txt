@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    pid_batch = tl.program_id(0) # batch index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        a_ptrs = a_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)

        # cast to fp32 if needed
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

The strides for a: shape (batch, M, K). So stride_a_batch: stride for batch dimension = stride over first dimension (maybe stride for batch). stride_a_m: stride for M dimension (second), stride_a_k: stride for K dimension (third). Similarly for b: shape (batch, K, N). So stride_b_batch: stride over batch; stride_b_k for K dimension, stride_b_n for N dimension.

Now we need to compute these strides in Python code after reshaping.

During PyTorch operations, we will have left_b shape (B, H, N, N) contiguous after permute. We can view it as left_flat = left_b.view(B*H, N, N). This yields a tensor of shape (B*H, N, N) contiguous in memory: stride[0] = N*N (since each batch is contiguous block of N*N), stride[1] = N (since moving along rows), stride[2] = 1 (since moving along columns). But because we call .view (not .reshape) after contiguous, it's likely to be contiguous.

Similarly right_T shape (B, H, N, N) after transpose and contiguous, then .view as (B*H, N, N). However note that if we do right_T = right_b.transpose(-2,-1).contiguous().view(B*H, N, N) then shape is same (batch, N, N) but memory order is contiguous.

Thus we can compute stride for a: a.stride() returns (N*N, N, 1). So stride_a_batch = a.stride(0), stride_a_m = a.stride(1), stride_a_k = a.stride(2).

Similarly for b: shape (batch, K, N) where K = N, N = N. So stride_b_batch = b.stride(0), stride_b_k = b.stride(1), stride_b_n = b.stride(2). So it's same as a's stride.

    device = input_tensor.device
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # layernorm (float32)
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # cast weights
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    left = F.linear(x, left_proj_w)   # (B,N,N,hidden)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # batch index, row block, col block
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # compute linear offsets
        a_ptrs = a_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        # load with bounds checking
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)

        # cast to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    The implementation follows the reference PyTorch module:
    1. Layer‑normalize the input.
    2. Linear projections to a hidden dimension.
    3. Compute three sigmoid gates.
    4. Fuse optional mask with the left/right gates.
    5. Core O(N³·H) reduction performed by a custom batched GEMM kernel
       (≈ einsum '... i k d, ... j k d -> ... i j d').
    6. Layer‑normalize the hidden result, apply the output gate and a final
       projection back to the original channel dimension.
    All heavy compute is carried out in fp16, with fp32 accumulation inside the
    Triton kernel. The final tensor is returned in float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm (float32 for stability)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Cast to fp16 for compute‑heavy stages
    x = x.to(torch.float16)

    # 2. Cast all linear weights to fp16
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # 3. Linear projections to hidden dimension
    left = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w) # (B, N, N, H)

    # 4. Compute sigmoid gates
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # 5. Apply mask and gates
    if mask is not None and not nomask:
        # mask: [B, N, N] -> [B, N, N, 1]
        mask_fp16 = mask.to(device=device, dtype=torch.float16).unsqueeze(-1)
        left = left * mask_fp16 * left_gate
        right = right * mask_fp16 * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # 6. Core reduction using a custom batched GEMM kernel
    #    reshape to (B, H, N, N)
    left_b = left.permute(0, 3, 1, 2).contiguous()
    right_b = right.permute(0, 3, 1, 2).contiguous()
    # transpose right for matmul
    right_t = right_b.permute(0, 1, 3, 2).contiguous()

    B, H, N, _ = left_b.shape
    batch = B * H

    # Flatten batch & hidden dimension
    left_flat = left_b.view(batch, N, N)
    right_flat = right_t.view(batch, N, N)

    # Allocate output
    out_flat = torch.empty((batch, N, N), device=device, dtype=torch.float16)

    # Strides
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # Tile sizes – choose based on N
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    # make sure tiles do not exceed dimensions
    if N < BLOCK_M:
        BLOCK_M = max(16, 1 << (N-1).bit_length() // 2)  # just a safe fallback
    if N < BLOCK_N:
        BLOCK_N = max(16, 1 << (N-1).bit_length() // 2)

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, H, N, N)
    out_b = out_flat.view(B, H, N, N)

    # Permute back to (B, N, N, H)
    out = out_b.permute(0, 2, 3, 1).contiguous()

    # 7. Second LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 8. Apply output gate
    out = out * out_gate

    # 9. Final linear projection back to original channels
    out = F.linear(out, to_out_w)

    # 10. Cast to float32 for the final result
    out = out.to(torch.float32)

    return out

custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input (float32 for numerical stability) and cast to fp16.
2. Linear projections `left` and `right` from C → hidden_dim (H) using fp16 GEMM.
3. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) with
   bias‑free linear layers.
4. Fuse optional mask with the left/right gates (element‑wise multiplication).
5. Core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   Implemented as a batched GEMM kernel in Triton
   (equivalent to `einsum('... i k d, ... j k d -> ... i j d')`).
6. Layer‑normalize over the hidden dimension, apply the output gate,
   and project back to the original channel dimension.
7. Return the result in float32.

All heavy compute is performed in fp16 with fp32 accumulation inside the
Triton kernel.
"""

import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication: C = A @ B  where
       A.shape = (batch, M, K), B.shape = (batch, K, N) and C.shape = (batch, M, N).

    Each program processes one (batch, block_m, block_n) tile.
    """

    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block column

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers with proper strides

                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        # Load with bounds checking

                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)

        b = b.to(tl.float32)

        acc += tl.dot(a, b)
