@triton.jit
def triu_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)  # batch*hidden index
    pid_m = tl.program_id(0)  # i block
    pid_n = tl.program_id(1)  # j block
    
    # compute batch and hidden index
    b = pid_z // H
    h = pid_z % H

    # compute offsets
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_i = i < N
    mask_j = j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        k_range = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # Load left tile A: shape (BLOCK_M, BLOCK_K)
        # pointer = left_ptr + b*stride_left_b + i*stride_left_i + k_range*stride_left_k + h*stride_left_h
        a_ptr = left_ptr + b * stride_left_b + i[:, None] * stride_left_i + k_range[None, :] * stride_left_k + h * stride_left_h
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile B_transposed shape (BLOCK_K, BLOCK_N) from right
        # We want B_T[k, j] = right[b, j, k, h]
        b_ptr = right_ptr + b * stride_right_b + j[None, :] * stride_right_j + k_range[:, None] * stride_right_k + h * stride_right_h
        # shape (BLOCK_K, BLOCK_N)
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # compute dot product
        acc += tl.dot(a, b)

    # store result
    # out_ptr + b*stride_out_b + i*stride_out_i + j*stride_out_j + h*stride_out_h
    c_ptr = out_ptr + b * stride_out_b + i[:, None] * stride_out_i + j[None, :] * stride_out_j + h * stride_out_h
    # cast to fp16
    acc_fp16 = acc.to(tl.float16)
    tl.store(c_ptr, acc_fp16, mask=mask_i[:, None] & mask_j[None, :])

    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device
    dtype = input_tensor.dtype

    # 1. LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # cast to fp16 for compute
    x = x.to(torch.float16)

    # Linear proj weight conversion to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # projections
    left_proj = F.linear(x, left_proj_w) # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # apply mask and gates
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # this case never occurs based on check, but just in case
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp16
            right = right * mask_fp16

    # core reduction using Triton kernel
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # shape (B,N,N,H), fp16

    # compute strides
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # grid dimensions

    BLOCK_N = 64
    BLOCK_K = 64

        left,
        right,
        out,

        stride_right[0], stride_right[1], stride_right[2], stride_right[3],

        BLOCK_N=BLOCK_N,

    out = out * out_gate

"""
TriMul (outgoing) forward kernel implemented with Triton.
-------------------------------------------------------
1. LayerNorm over the channel dimension (`dim`) – PyTorch.
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (left/right) and gating (left/right/out) – PyTorch FP16.
4. Optional mask multiplication (element‑wise).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a fused batched GEMM in Triton.  Each (b,h) pair computes
   an NxN matrix multiplication of `left[b,:,:,h]` with the transpose of
   `right[b,:,:,h]`.  The kernel tiles the M/N dimensions (64×64) and the
   reduction K dimension (64) and accumulates in fp32 before storing fp16.
6. Output LayerNorm, out‑gate, final linear projection – PyTorch.
7. Cast back to the input dtype.

The Triton kernel handles arbitrary batch size (1‑2), sequence length
(up to 1024) and hidden dimension (up to 768).  At least one operation
is performed in Triton as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,          # fp16*  (B, N, N, H)
    right_ptr,         # fp16*  (B, N, N, H)
    out_ptr,           # fp16*  (B, N, N, H)
    B, N, H,           # i64 scalar
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)   # block row (i)
    pid_n = tl.program_id(1)   # block col (j)
    pid_z = tl.program_id(2)   # batch * hidden index

    # decode batch and hidden index
    b = pid_z // H
    h = pid_z % H

    # range of rows / cols for this tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile A: (BLOCK_M, BLOCK_K)
        a_ptr = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + h * stride_left_h
        )
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile B (transposed) : (BLOCK_K, BLOCK_N)
        # We need right[b, j, k, h] -> map to (k, j)
        b_ptr = (
            right_ptr
            + b * stride_right_b
            + offs_j[None, :] * stride_right_j
            + offs_k[:, None] * stride_right_k
            + h * stride_right_h
        )
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # matrix multiply‑accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # write out result
    c_ptr = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h * stride_out_h
    )
    acc_fp16 = acc.to(tl.float16)
    tl.store(c_ptr, acc_fp16, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (bool/int/float) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to fp16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Apply mask & gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape (B, N, N) -> broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp16
            right = right * mask_fp16

    # ------------------------------------------------------------------
    # 7) Core reduction using Triton (batched GEMM per hidden slice)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # (B, N, N, H), fp16

    # Compute strides (contiguous layout)
    stride_left = left.stride()    # (stride_b, stride_i, stride_k, stride_h)
    stride_right = right.stride()
    stride_out = out.stride()

    # Tile sizes – these are good defaults for most GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),    # blocks over i dimension
        triton.cdiv(N, BLOCK_N),    # blocks over j dimension
        B * H,                      # batch * hidden
    )

    _tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
