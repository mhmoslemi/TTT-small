"""
custom_kernel: Triton‑accelerated outgoing TriMul forward (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (left/right) and gating (left/right/out) are performed
   with bias‑free linear layers (weights are cast to fp16).
4. If a mask is present and `nomask` is False, a tiny Triton kernel fuses the
   mask multiplication for the left and right tensors (post‑projection,
   post‑gate).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is realised as a batched matrix multiplication for each hidden slice
   (shape (B*H, N, N) @ (B*H, N, N)^T) using torch.bmm.
6. Apply a second LayerNorm over the hidden dimension, multiply by the output
   gate and project back to the original dimension.
7. Cast the result back to float32 (the original dtype of the input).

The Triton kernel is only used for step 4, satisfying the requirement of
including a custom kernel while keeping the implementation simple and fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _mask_apply_kernel(
    left_ptr,               # fp16 * (B, N, N, H)
    right_ptr,              # fp16 * (B, N, N, H)
    mask_ptr,               # fp16 * (B, N, N)
    left_out_ptr,           # fp16 * (B, N, N, H)
    right_out_ptr,          # fp16 * (B, N, N, H)
    total_elems,            # i64 scalar = B*N*N*H
    hidden_dim,             # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask multiplication for left/right tensors.

    left_out = left * mask[...,None]
    right_out = right * mask[...,None]
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # Index of corresponding mask element (mask has no hidden dimension)
    mask_idx = offs // hidden_dim

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # 1.0 = keep, 0.0 = drop

    tl.store(left_out_ptr + offs, left * m, mask=mask)
    tl.store(right_out_ptr + offs, right * m, mask=mask)


def _apply_mask(
    left: torch.Tensor,
    right: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Apply a binary mask (shape B,N,N) to the last dimension of left/right
    (shape B,N,N,H) using a Triton kernel.

    Returns masked left and right tensors (both fp16).
    """
    assert left.is_contiguous() and right.is_contiguous()
    assert mask.is_contiguous()
    total = left.numel()
    H = left.shape[-1]

    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = (triton.cdiv(total, block_size),)
    _mask_apply_kernel[grid](
        left,
        right,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None (0/1)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"     – optional bool, if True ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to fp16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))     # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse projection, gate and optional mask
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_mask(left.contiguous(),
                                 right.contiguous(),
                                 mask_fp16,
                                 block_size=1024)
    # else: keep left/right as they are (mask ignored)

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (B*H, N, N) @ (B*H, N, N)^T
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to float32 (original input dtype)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
