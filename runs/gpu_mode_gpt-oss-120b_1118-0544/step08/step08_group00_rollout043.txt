def batched_gemm_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a_ptr = A + batch_idx * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B tile (K x N), note B is stored as (batch, K, N) (i.e., we need B transposed)
        b_ptr = B + batch_idx * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write back
    c_ptr = C + batch_idx * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections (left/right) and gates (left/right/out) are performed
   with bias‑free Linear layers.  Mask (if present) and gates are fused
   element‑wise.
4. The core contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is implemented as a batched GEMM (A @ Bᵀ) using a custom Triton kernel.
   The tensor is reshaped to (B·H, N, N) so each (b,h) slice is an
   independent matrix multiplication.
5. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and project back to the original dimension.
6. Cast the result back to the input dtype.

The Triton kernel tiles the matrices (BLOCK_M×BLOCK_N) and iterates over
the K‑dimension in BLOCK_K steps, using FP16 arithmetic with FP32
accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _batched_gemm_kernel(
    A,                     # fp16* (batch, M, K)
    B,                     # fp16* (batch, K, N)
    C,                     # fp16* (batch, M, N)  – output
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B  (A: M×K, B: K×N) for each batch."""
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute the coordinates of the block this program works on
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: (M x K)
        a_ptr = A + batch * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile: (K x N)
        b_ptr = B + batch * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)                     # FP16 dot, accumulate in FP32

    # Write the result back to C
    c_ptr = C + batch * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass for the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor of shape [B, N, N] (optional, 0/1)
            - weights      : dict containing model parameters
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Prepare FP16 weights
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gates (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))     # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask, projection and gate
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Reshape to (B*H, N, N) for batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # 7) Launch Triton batched GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    M_dim = N
    N_dim = N
    K_dim = N

    stride_Ab = left_flat.stride(0)
    stride_Am = left_flat.stride(1)
    stride_Ak = left_flat.stride(2)

    stride_Bb = right_flat.stride(0)
    stride_Bk = right_flat.stride(1)
    stride_Bn = right_flat.stride(2)

    stride_Cb = out_flat.stride(0)
    stride_Cm = out_flat.stride(1)
    stride_Cn = out_flat.stride(2)

    grid = (
        B * H,
        (M_dim + BLOCK_M - 1) // BLOCK_M,
        (N_dim + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M_dim,
        N_dim,
        K_dim,
        stride_Ab,
        stride_Am,
        stride_Ak,
        stride_Bb,
        stride_Bk,
        stride_Bn,
        stride_Cb,
        stride_Cm,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)      # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
