@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, H,
                   stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                   stride_right_b, stride_right_i, stride_right_k, stride_right_h,
                   stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # Determine batch and hidden index
    pid = tl.program_id(0)  # linear index over B*H
    b = pid // H
    h = pid % H

    pid_i = tl.program_id(1)  # block idx i
    pid_j = tl.program_id(2)  # block idx j

    # compute start offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # ranges
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    # mask for within bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        # left_ptr + b*stride_left_b + offs_i[:,None]*stride_left_i + offs_k[None,:]*stride_left_k + h*stride_left_h
        left_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile transposed: (BLOCK_K, BLOCK_N)
        # right_ptr + b*stride_right_b + h*stride_right_h + offs_k[:, None]*stride_right_k + offs_j[None,:]*stride_right_i
        right_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # compute dot product
        acc += tl.dot(left, right)

    # Write back
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):

    # Set dtype and device
    dtype_in = input_tensor.dtype
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    norm_w = weights["norm.weight"].to(dtype_in)
    norm_b = weights["norm.bias"].to(dtype_in)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for compute
    compute_dtype = torch.float16
    x = x.to(compute_dtype)

    # Weights to fp16
    left_proj_w = weights["left_proj.weight"].to(compute_dtype)
    right_proj_w = weights["right_proj.weight"].to(compute_dtype)
    left_gate_w = weights["left_gate.weight"].to(compute_dtype)
    right_gate_w = weights["right_gate.weight"].to(compute_dtype)
    out_gate_w = weights["out_gate.weight"].to(compute_dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype)
    to_out_w = weights["to_out.weight"].to(compute_dtype)

    # Projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # shape (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Compute core reduction using Triton kernel
    B, N, _, H = left.shape

    # Allocate output tensor
    out = torch.empty_like(left, dtype=compute_dtype)

    stride_left = left.stride()
    # left: (B,N,N,H)
    s_bl, s_il, s_kl, s_hl = stride_left

    s_br, s_ir, s_kr, s_hr = stride_right
    stride_out = out.stride()
    s_bo, s_io, s_jo, s_ho = stride_out

"""
custom_kernel: Optimized outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for compute‑intensive parts.
3. Linear projections (`left_proj`, `right_proj`) and gating projections
   (`left_gate`, `right_gate`, `out_gate`) are performed in FP16.
4. If a mask is supplied, a tiny Triton kernel fuses projection,
   gating and masking for both left and right tensors.
5. Core reduction: for each hidden slice `h`,
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   Implemented as a batched matrix multiplication using
   `torch.matmul` on tensors of shape (B, H, N, N).
6. Output LayerNorm, multiplication by `out_gate`, and final linear
   projection back to `dim`.
7. Cast the result back to the original dtype.

Only the mask‑gate‑fusion step is written in Triton; the heavy GEMM
leverages cuBLAS via `torch.matmul`, which is the fastest available
routine on the target GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B,N,N,H)
    right_proj_ptr,  # fp16 * (B,N,N,H)
    left_gate_ptr,   # fp16 * (B,N,N,H)
    right_gate_ptr,  # fp16 * (B,N,N,H)
    mask_ptr,        # fp16 * (B,N,N)
    left_out_ptr,    # fp16 * (B,N,N,H)
    right_out_ptr,   # fp16 * (B,N,N,H)
    total_elems,     # i64 = B*N*N*H
    hidden_dim,      # i64 = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # each mask element repeats `hidden_dim` times
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    left_out = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gate and (optional) mask for both left and right tensors
    using a small Triton kernel.
    All tensors must be fp16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)
    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
        - mask         : torch.Tensor of shape [B, N, N] (0/1) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask" (optional) – if True, ignore mask

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(dtype_in)
    norm_b = weights["norm.bias"].to(dtype_in)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy parts
    # ------------------------------------------------------------------
    compute_dtype = torch.float16
    x = x.to(compute_dtype)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(compute_dtype)
    right_proj_w  = weights["right_proj.weight"].to(compute_dtype)
    left_gate_w   = weights["left_gate.weight"].to(compute_dtype)
    right_gate_w  = weights["right_gate.weight"].to(compute_dtype)
    out_gate_w    = weights["out_gate.weight"].to(compute_dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype)
    to_out_w      = weights["to_out.weight"].to(compute_dtype)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fusion via Triton if mask is used)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(compute_dtype).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched matmul over (B, H, N, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B, H, N, N) for batched matmul
    left_bhw = left.permute(0, 3, 1, 2)          # (B, H, N, N)
    right_bhw = right.permute(0, 3, 2, 1)        # (B, H, N, N)  (k ↔ j)

    # batched matrix multiplication (cuBLAS)
    out_bhw = torch.matmul(left_bhw, right_bhw)  # (B, H, N, N)

    # back to (B, N, N, H)
    out = out_bhw.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
