def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0)    # batch*hidden index
    pid_m = tl.program_id(1)     # block row i
    pid_n = tl.program_id(2)     # block col j

    # compute offsets
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    # mask for out-of-bounds
    mask_i = i_offset < N
    mask_j = j_offset < N

    # compute base pointers
    batch_offset = pid_bh * stride_l_batch
    left_ptrs = left_ptr + batch_offset + i_offset[:, None] * stride_l_i + k_offset[None, :] * stride_l_k
    right_ptrs = right_ptr + batch_offset + k_offset[:, None] * stride_r_k + j_offset[None, :] * stride_r_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        # load
        left = tl.load(left_ptrs, mask=mask_i[:, None] & (k_offset[None, :] < N), other=0.0)
        right = tl.load(right_ptrs, mask=(k_offset[:, None] < N) & mask_j[None, :], other=0.0)
        # dot
        acc += tl.dot(left, right)
        # increment pointers
        left_ptrs += BLOCK_K * stride_l_k
        right_ptrs += BLOCK_K * stride_r_k

    # write back
    out = acc.to(tl.float16)
    out_ptrs = out_ptr + batch_offset + i_offset[:, None] * stride_o_i + j_offset[None, :] * stride_o_j
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])

def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
    ):
   pid_bh = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # offsets
   i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

   # create mask flags
   mask_i = i < N
   mask_j = j < N

   # base pointer for this batch*hidden slice
   offset_batch = pid_bh * stride_l_batch

   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # loop over k
   for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
       k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
       mask_k = k < N

       # pointers
       left_ptrs = left_ptr + offset_batch + i[:, None] * stride_l_i + k[None, :] * stride_l_k

       # load with mask

       right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

       # accumulate dot product

def custom_kernel(data):
   input_tensor, mask, weights, config = data
   dim = config["dim"]
   hidden_dim = config["hidden_dim"]
   nomask = config.get("nomask", False)

   # layernorm
   x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

   # cast to fp16
   x = x.to(torch.float16)

   # weights convert to fp16
   left_proj_w = weights["left_proj.weight"].to(torch.float16)
   right_proj_w = weights["right_proj.weight"].to(torch.float16)
   left_gate_w = weights["left_gate.weight"].to(torch.float16)
   right_gate_w = weights["right_gate.weight"].to(torch.float16)
   out_gate_w = weights["out_gate.weight"].to(torch.float16)
   to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
   to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
   to_out_w = weights["to_out.weight"].to(torch.float16)

   # linear projections
   left_proj = torch.nn.functional.linear(x, left_proj_w)   # shape (B,N,N,hidden_dim)
   right_proj = torch.nn.functional.linear(x, right_proj_w)

   # gating
   left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
   right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
   out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

   # fuse projections with gating (and mask if present)
   if mask is not None and not nomask:
       # mask shape [B,N,N] -> unsqueeze
       mask_ = mask.to(torch.float16).unsqueeze(-1)
       left = left_proj * left_gate * mask_
       right = right_proj * right_gate * mask_
   else:
       left = left_proj * left_gate
       right = right_proj * right_gate

   # reshape for Triton kernel
   B, N, _, H = left.shape
   # permute to bring hidden dim before N,N
   left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
   right_perm = right.permute(0, 3, 1, 2).contiguous()
   BH = B * H
   left_flat = left_perm.view(BH, N, N)
   right_flat = right_perm.view(BH, N, N)
   out_flat = torch.empty_like(left_flat)

   # compute grid sizes
   BLOCK_M = 128
   BLOCK_N = 128
   BLOCK_K = 32
   n_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
   n_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

   # launch kernel
   tri_mul_kernel[(BH, n_blocks_m, n_blocks_n)](
       left_flat,
       right_flat,
       out_flat,
       N,
       left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
       right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
       out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
       BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
   )
   # reshape back
   out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

   # output layer norm
   out = torch.nn.functional.layer_norm(
       out,
       (hidden_dim,),
       weight=to_out_norm_w,
       bias=to_out_norm_b,
   )
   # apply out gate
   out = out * out_gate
   # final linear projection
   out = torch.nn.functional.linear(out, to_out_w)

   # cast back to input dtype (likely float32)
   out = out.to(input_tensor.dtype)
   return out
```

"""
TriMul outgoing kernel (AlphaFold3)
----------------------------------
Implements the forward pass of the Triangle Multiplicative Update (TriMul)
using a fused Triton kernel for the core reduction:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

Algorithm:
1. Layer‑norm the input over the channel dimension.
2. Cast to fp16 and apply linear projections + sigmoid gates.
3. Optionally apply the pairwise mask.
4. Fuse the N×N×H inner product across the hidden dimension with a
   custom Triton kernel (blocked GEMM).  This replaces the expensive
   `torch.einsum` / triple loop.
5. Layer‑norm the hidden result, multiply by the output gate and 
   project back to the original dimension.
6. Cast back to the original dtype.

The kernel processes all (batch * hidden) slices in parallel and
accumulates in fp32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    out_ptr,           # fp16 *
    N,                 # i32 scalar (sequence length)
    stride_l_batch,    # i64
    stride_l_i,        # i64
    stride_l_k,        # i64
    stride_r_batch,    # i64
    stride_r_j,        # i64
    stride_r_k,        # i64
    stride_o_batch,    # i64
    stride_o_i,        # i64
    stride_o_j,        # i64
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for TriMul.

    Computes for a given (batch*hidden) slice:
        out[i, j] = Σ_k left[i, k] * right[j, k]
    """
    pid_bh = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1)    # block row (i)
    pid_n = tl.program_id(2)    # block col (j)

    # Offsets for i and j within the sequence
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks for i and j
    mask_i = i < N
    mask_j = j < N

    # Base pointer for this (batch*hidden) slice
    batch_offset = pid_bh * stride_l_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k in tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Pointers to the current tiles
        left_tile_ptr = left_ptr + batch_offset + i[:, None] * stride_l_i + k[None, :] * stride_l_k
        right_tile_ptr = right_ptr + batch_offset + k[:, None] * stride_r_k + j[None, :] * stride_r_j

        # Load with masks (out‑of‑bounds elements are zero)
        left = tl.load(left_tile_ptr,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)
        right = tl.load(right_tile_ptr,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # Accumulate the dot product; tl.dot promotes to fp32
        acc += tl.dot(left, right)

    # Write the result back (cast to fp16)
    out_tile_ptr = out_ptr + batch_offset + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_tile_ptr,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32 or bool) or None
            - weights: dict with the model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast model weights to fp16 (no bias terms)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6) Fuse projection, gate and optional mask (pure PyTorch)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_ = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask_
        right = right_proj * right_gate * mask_
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core TriMul reduction via Triton (batched GEMM)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # N == seq_len, H == hidden_dim

    # Rearrange to (B*H, N, N) for a single batched GEMM launch
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)

    # Block sizes – tuned for Ampere GPUs (B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    n_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    n_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    # Launch the Triton kernel
    _tri_mul_kernel[(B * H, n_blocks_m, n_blocks_n)](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # left strides
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),# right strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),     # output strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9) Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
