"""
TriMul (outgoing) forward kernel – highly‑optimized implementation.

The algorithm follows the reference PyTorch TriMul but fuses several
operations to reduce kernel launches and memory traffic:

1. LayerNorm over the channel dimension (`dim`).
2. Cast to fp16 for the compute‑heavy part.
3. **Fused linear projection + gate**:
   For the left and right branches the projection matrix and the gate
   matrix are concatenated and a single matmul is performed:
        cat = [W_proj ; W_gate]   (2*hidden, dim)
        out = x @ cat.T
   The result is split into the projection and gate tensors,
   the gate side is passed through a sigmoid.
4. Optional mask handling – if a mask is provided and `nomask` is False,
   the mask is fused with projection/gate using a tiny Triton kernel.
5. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (A @ Bᵀ) with batch size B*hidden_dim.
   We reuse PyTorch’s `torch.bmm` which internally calls cuBLAS’s
   highly‑optimized batched GEMM.
6. Output LayerNorm, element‑wise multiplication with the output gate,
   and a final linear projection back to the original `dim`.
7. Cast back to the input dtype (usually float32).

Only the mask‑gate fusion is implemented in Triton; all other
heavy‑weight ops use cuBLAS via PyTorch, which together with the
fusion above yields a large speed‑up over the naïve implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _mask_gate_fusion_kernel(
    proj_ptr,           # fp16 * (total_elems)
    gate_ptr,           # fp16 * (total_elems)
    mask_ptr,           # fp16 * (mask_elems)  (broadcasted over hidden)
    out_ptr,            # fp16 * (total_elems)
    total_elems,        # i64 scalar = B*N*N*H
    hidden_dim,         # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse: out = proj * sigmoid(gate) * mask
    The mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    # index of the mask element (ignoring hidden dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    gated = proj * tl.sigmoid(gate) * m
    tl.store(out_ptr + offs, gated, mask=mask)


def _apply_mask_gate_fusion(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Apply the mask (if present) and sigmoid gate in a single Triton kernel.
    All tensors must be fp16, contiguous and have shape (B, N, N, H) for
    proj and gate, and (B, N, N) for mask.
    Returns:
        out: torch.Tensor of shape (B, N, N, H) = proj * sigmoid(gate) * mask
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _mask_gate_fusion_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (bool/int/float) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Prepare weights in fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Fused projection + gate for left and right branches
    #    (single matmul per branch)
    # ------------------------------------------------------------------
    # concatenate projection and gate weights
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # combined linear for left branch
    left_cat = F.linear(x, left_cat_w)            # (B, N, N, 2*H)
    left_proj, left_gate_pre = torch.split(left_cat, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate_pre)

    # combined linear for right branch
    right_cat = F.linear(x, right_cat_w)          # (B, N, N, 2*H)
    right_proj, right_gate_pre = torch.split(right_cat, hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate_pre)

    # output gate (separate linear)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fused via Triton when mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        # left = left_proj * left_gate * mask
        left = _apply_mask_gate_fusion(left_proj, left_gate, mask_fp16)
        # right = right_proj * right_gate * mask
        right = _apply_mask_gate_fusion(right_proj, right_gate, mask_fp16)
    else:
        # no mask or mask ignored – simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
            left = left * mask_fp16
            right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (A @ Bᵀ) with batch size B*H
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (BH, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (BH, N, N)

    # batched matrix multiplication: (BH, N, N) @ (BH, N, N)^T -> (BH, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore shape (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
