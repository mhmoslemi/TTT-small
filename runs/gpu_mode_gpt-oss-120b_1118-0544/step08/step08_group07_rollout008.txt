"""
TriMul (outgoing) forward pass – Triton‑accelerated mask & gate fusion.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for all compute‑intensive ops.
3. Linear projections → `left_proj`, `right_proj` (no bias).
4. Gating projections → `left_gate`, `right_gate`, `out_gate` (sigmoid).
5. Fuse projection, gate and optional mask with a tiny Triton kernel:
       left  = left_proj  * left_gate  * mask
       right = right_proj * right_gate * mask
   (mask is omitted when `nomask` is True or mask is None.)
6. Core reduction (TriMul):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication via `torch.bmm`
   on tensors of shape (B*H, N, N).
7. Layer‑norm over the hidden dimension, apply `out_gate`,
   and a final linear projection back to `dim`.
8. Cast the result back to the original dtype (usually float32).

Only the mask‑gate fusion step is written in Triton; all other
operations rely on highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B*N*N*H,)
    right_proj_ptr,  # fp16 * (B*N*N*H,)
    left_gate_ptr,   # fp16 * (B*N*N*H,)
    right_gate_ptr,  # fp16 * (B*N*N*H,)
    mask_ptr,        # fp16 * (B*N*N,)
    left_out_ptr,    # fp16 * (B*N*N*H,)
    right_out_ptr,   # fp16 * (B*N*N*H,)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse projection, gate and mask for both left and right tensors.
    All tensors are flattened to 1‑D pointers.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds
    mask = offs < total_elems

    # Each mask element repeats `hidden_dim` times in the flattened tensors
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left  = lp * lg * m
    right = rp * rg * m

    tl.store(left_out_ptr  + offs, left,  mask=mask)
    tl.store(right_out_ptr + offs, right, mask=mask)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"        – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype  # keep original dtype for final cast

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_w,
                     bias=norm_b)            # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections & gates (still FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)      # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Fuse projection, gate and optional mask (Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is contiguous FP16
        mask_fp16 = mask.to(torch.float16).contiguous()
        B, N, _, H = left_proj.shape
        total = left_proj.numel()               # B * N * N * H
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # Flatten tensors for the kernel
        left_proj_f   = left_proj.contiguous().view(-1)
        right_proj_f  = right_proj.contiguous().view(-1)
        left_gate_f   = left_gate.contiguous().view(-1)
        right_gate_f  = right_gate.contiguous().view(-1)
        mask_f        = mask_fp16.contiguous().view(-1)
        left_f   = left.view(-1)
        right_f  = right.view(-1)

        BLOCK_SIZE = 1024
        grid = (triton.cdiv(total, BLOCK_SIZE),)

        _proj_gate_mask_kernel[grid](
            left_proj_f,
            right_proj_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            left_f,
            right_f,
            total,
            H,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        # No mask → element‑wise fusion
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM: (B*H, N, N) @ (B*H, N, N)^T
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, N, N, H) -> (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (H,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)      # (B, N, N, H)
    out = out * out_gate
    out = F.linear(out, to_out_w)                # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
