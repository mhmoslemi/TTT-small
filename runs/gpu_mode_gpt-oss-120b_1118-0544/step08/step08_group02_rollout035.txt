def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ensure contiguous
    x = input_tensor

    # LayerNorm
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Convert to fp16 for performance
    x = x.to(torch.float16)

    # Cast weights
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left_proj = F.linear(x, w_left_proj)  # (B,N,N,hidden_dim)
    right_proj = F.linear(x, w_right_proj)

    left_gate = torch.sigmoid(F.linear(x, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x, w_out_gate))

    # Apply mask and gating
    if mask is not None and not nomask:
        # mask shape (B,N,N)
        mask = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but we ignore; still multiply? spec says ignore mask
            mask = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask
            right = right * mask

    # Core reduction using Triton
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_flat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)
    # transpose right to (B*H, N, N) => we need (batch, K, N) where K=N (since we want B^T)
    right_T = right_flat.transpose(1,2).contiguous()  # shape (B*H, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)  # (B*H, N, N)

    # strides
    stride_am = left_flat.stride(0)
    stride_ak = left_flat.stride(2)  # stride for K dimension (inner)
    stride_bk = right_T.stride(1)   # stride for K dimension in B (since shape (batch, K, N))
    stride_bn = right_T.stride(2)   # stride for N dimension in B
    stride_cm = out_flat.stride(0)
    stride_cn = out_flat.stride(2)

    # compute grid
    M = N
    N_dim = N
    K = N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N_dim + BLOCK_N - 1) // BLOCK_N
    total_batches = B * H
    grid = (total_batches * num_blocks_m * num_blocks_n,)

    matmul_transpose_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        M, N_dim, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()

    # Output layernorm
    out = F.layer_norm(out, (hidden_dim,), weight=w_to_out_norm, bias=b_to_out_norm)

    # Apply out gate and final linear
    out = out * out_gate
    out = F.linear(out, w_to_out)

    # Cast back
    out = out.to(input_tensor.dtype)

    return out

```python
"""
Efficient Triton implementation of the outgoing TriMul module.
...

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr
):
    pid = tl.program_id(0)
    # Compute block indices
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_blocks_m * num_blocks_n)
    pid_in_batch = pid % (num_blocks_m * num_blocks_n)
    block_m = pid_in_batch // num_blocks_n
    block_n = pid_in_batch % num_blocks_n

    # Offsets within the block
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for output write
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_out = mask_m[:, None] & mask_n[None, :]

    # Pointers to the beginning of the current batch
    a_batch_ptr = a_ptr + batch * stride_a_batch
    b_batch_ptr = b_ptr + batch * stride_b_batch
    c_batch_ptr = c_ptr + batch * stride_c_batch

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_offset = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K)

        # Compute pointers for this K‑block
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_a_row + (k_offset + k_range)[None, :] * stride_a_col
        b_ptrs = b_batch_ptr + (k_offset + k_range)[:, None] * stride_b_row + offs_n[None, :] * stride_b_col

        # Masks for loads (handle edge blocks)
        mask_a = (offs_m[:, None] < M) & (k_offset + k_range)[None, :] < K
        mask_b = (k_offset + k_range)[:, None] < K  & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

    # Write result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_out)

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul (AlphaFold3) using a Triton
    batched GEMM kernel for the O(N³) reduction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C]
        - mask        : torch.Tensor of shape [B, N, N] or None
        - weights     : dict of model parameters (float32)
        - config      : dict with keys:
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask" (optional) – if True, ignore the mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the heavy part of the computation
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast model weights to FP16 (bias‑free linear layers)
    # ------------------------------------------------------------------
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gates (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, w_left_proj)          # (B, N, N, H)
    right_proj = F.linear(x, w_right_proj)        # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, w_left_gate))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, w_right_gate)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, w_out_gate))     # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask (if present) and gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but instructed to ignore it → treat as ones
            mask = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask
            right = right * mask

    # ------------------------------------------------------------------
    # 6) Core O(N³) reduction via a batched GEMM (TriMul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (batch * hidden, N, N) where each slice is a matrix
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Transpose right so that we compute A @ B^T
    right_T = right_flat.transpose(1, 2).contiguous()

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Strides
    stride_a_batch = left_flat.stride(0)
    stride_a_row   = left_flat.stride(1)
    stride_a_col   = left_flat.stride(2)

    stride_b_batch = right_T.stride(0)
    stride_b_row   = right_T.stride(1)   # moves along K dimension
    stride_b_col   = right_T.stride(2)   # moves along N dimension

    stride_c_batch = out_flat.stride(0)
    stride_c_row   = out_flat.stride(1)
    stride_c_col   = out_flat.stride(2)

    M = N
    N_dim = N
    K = N

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N_dim + BLOCK_N - 1) // BLOCK_N
    total_batches = B * H
    grid = (total_batches * num_blocks_m * num_blocks_n,)

    matmul_transpose_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        M,
        N_dim,
        K,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Restore original shape (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gate, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )
    out = out * out_gate
    out = F.linear(out, w_to_out)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype of the input
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)

    return out

"""
Efficient Triton implementation of the outgoing TriMul (AlphaFold3) module.

The forward pass consists of:
1. LayerNorm over the channel dimension.
2. Linear projections (left/right) and gating (left/right/out) without bias.
3. Optional mask application (fused with projection/gating).
4. Core O(N³) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (A @ Bᵀ) using a custom Triton kernel.
5. Output LayerNorm, out‑gate multiplication and final linear projection back
   to the original channel dimension.
All heavy compute is performed in FP16; the final result is cast back to the
input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

def matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,

    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):

    Shapes:
        A : (batch, M, K)
        B : (batch, K, N)
        C : (batch, M, N)
    """

    # Number of blocks in each dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # Derive batch, block‑row and block‑col indices
    batch = pid // (num_blocks_m * num_blocks_n)
    pid_in_batch = pid % (num_blocks_m * num_blocks_n)
    block_m = pid_in_batch // num_blocks_n
    block_n = pid_in_batch % num_blocks_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_out = mask_m[:, None] & mask_n[None, :]

    # Base pointers for the current batch
    a_batch = a_ptr + batch * stride_a_batch
    b_batch = b_ptr + batch * stride_b_batch
    c_batch = c_ptr + batch * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)

        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # Mask present but we are instructed to ignore it → treat as ones
            mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp16
            right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6) Core O(N³) reduction via the Triton batched GEMM kernel
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # Reshape to (batch * hidden, N, N) where each slice is a matrix
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Transpose right so that we compute A @ Bᵀ
    right_T = right_flat.transpose(1, 2).contiguous()

    # Allocate output tensor
    out_flat = torch.empty_like(left_flat)

    # Strides (batch, row, col) for each tensor
    stride_a_batch = left_flat.stride(0)
    stride_a_row   = left_flat.stride(1)
    stride_a_col   = left_flat.stride(2)

    stride_b_batch = right_T.stride(0)
    stride_b_row   = right_T.stride(1)   # moves along K dimension
    stride_b_col   = right_T.stride(2)   # moves along N dimension

    stride_c_batch = out_flat.stride(0)
    stride_c_row   = out_flat.stride(1)
    stride_c_col   = out_flat.stride(2)

    M = N
    N_dim = N
    K = N

    # Block sizes – tuned for typical B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N_dim + BLOCK_N - 1) // BLOCK_N
    total_batches = B * H
    grid = (total_batches * num_blocks_m * num_blocks_n,)

    matmul_transpose_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        M,
        N_dim,
        K,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )
    out = out * out_gate
    out = F.linear(out, w_to_out)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype of the input
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)

    return out
```
