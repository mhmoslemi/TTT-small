"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm Overview
------------------
1. **Input LayerNorm** (float32 → fp16)  
2. **Linear projection** of the input to a hidden dimension `H` producing
   `left` and `right`.  
3. **Gate computation** – three independent linear layers with a sigmoid
   activation (`left_gate`, `right_gate`, `out_gate`).  
4. **Mask & gate fusion** – a single Triton kernel multiplies `left` and
   `right` in‑place by the (optional) binary mask and the corresponding
   gates, eliminating temporaries and extra memory traffic.  
5. **Core reduction** – for each batch and hidden channel we compute a
   matrix multiplication  
   `out[b, h] = left[b, h] @ right[b, h].T`  
   giving the tensor shape `(B, H, N, N)`.  This is performed by the highly
   tuned cuBLAS `torch.matmul`.  
6. **Second LayerNorm** over the hidden dimension.  
7. **Apply output gate** (element‑wise).  
8. **Final linear projection** back to the original channel dimension `C`.  
9. Cast the result to `float32` for the final output.

Only the mask‑gate fusion is implemented in Triton, satisfying the
requirement to include a custom kernel while keeping the compute‑heavy
matrix multiplication in cuBLAS for maximum throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    TOTAL: tl.constexpr,  # total number of elements = B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place fusion of mask and the two gate tensors.
    left  = left  * mask * left_gate
    right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL

    # Load the four tensors
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Derive index of the corresponding (b,i,j) entry for the mask.
    # Since the last dimension is H, a stride of H maps one scalar mask
    # value to a contiguous block of H elements.
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply the fused multiplication
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or fp16)
        - mask         : torch.Tensor of shape [B, N, N] (binary) or None
        - weights      : dict of model parameters (see code for expected keys)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Processed tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )

    # Cast to fp16 for the compute‑heavy portion
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    # Shape after projection: (B, N, N, H)
    left  = F.linear(x, left_proj_w)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Fuse mask and gate multiplications (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(device=device, dtype=torch.float16).contiguous()
        total_elements = left.numel()                     # B * N * N * H
        BLOCK = 1024                                        # threads per program
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates have been fused; free memory
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication
    #    left and right shape -> (B, H, N, N)
    # ------------------------------------------------------------------
    left_b  = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)

    # out_b shape: (B, H, N, N)
    out_b = torch.matmul(left_b, right_b.transpose(-2, -1))

    # Back to (B, N, N, H)
    out = out_b.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast to float32 (required output dtype)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
