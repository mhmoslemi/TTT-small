@triton.jit
def _batch_matmul_kernel(A_ptr, B_ptr, C_ptr,
                         stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
                         M, N, K,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(0) # row tile
    pid_n = tl.program_id(1) # col tile

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # load A block (M x K)
        a_ptrs = A_ptr + (offs_m[:, None] * stride_am + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak)
        b_ptrs = B_ptr + ((k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        a = tl.load(a_ptrs, mask=offs_m < M, other=0.0)
        b = tl.load(b_ptrs, mask=offs_n < N, other=0.0)
        acc += tl.dot(a, b)

    # write back
    c_ptrs = C_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

"""
TriMul (outgoing) forward implementation with Triton fusion.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (left/right) and gating (left/right/out) using
   PyTorch's highly‑optimized GEMM (`F.linear`).
4. Fuse projection, gate and optional mask for the left/right tensors
   with a small Triton kernel.
5. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (A @ Bᵀ) for each hidden slice.
   We implement it with a custom Triton batched‑matrix‑multiply kernel,
   which processes all `batch * hidden_dim` matrices in parallel.
6. Output LayerNorm, multiply by `out_gate`, and a final linear projection
   back to the original channel dimension.
7. Cast back to the input dtype (usually float32).

The implementation uses Triton for two parts:
* `proj_gate_mask_kernel` – fuses projection, gate and mask.
* `batched_matmul_kernel` – performs the N³ reduction as a batch of
  matrix multiplications.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# 1. Fuse projection, gate and mask (optional)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B*N*N*H)
    right_proj_ptr,  # fp16 * (B*N*N*H)
    left_gate_ptr,   # fp16 * (B*N*N*H)
    right_gate_ptr,  # fp16 * (B*N*N*H)
    mask_ptr,        # fp16 * (B*N*N)
    left_out_ptr,    # fp16 * (B*N*N*H)
    right_out_ptr,   # fp16 * (B*N*N*H)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # index of the element in the mask (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gate and mask for both left and right tensors using a Triton kernel.
    All tensors must be fp16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)
    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# 2. Batched matrix multiplication kernel (A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr,               # fp16 * (B*H, M, K)
    B_ptr,               # fp16 * (B*H, K, N)
    C_ptr,               # fp16 * (B*H, M, N)
    stride_a_batch, stride_ah, stride_ak,
    stride_b_batch, stride_bk, stride_bn,
    stride_c_batch, stride_ch, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)          # which (batch * hidden) matrix
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # apply boundary checks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch slice
    a_batch_ptr = A_ptr + batch * stride_a_batch
    b_batch_ptr = B_ptr + batch * stride_b_batch
    c_batch_ptr = C_ptr + batch * stride_c_batch

    # accumulator in fp32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)

        # load A tile (M x K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_ah
                                + k_range[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (k_range[None, :] < K),
                    other=0.0)

        # load B tile (K x N)
        b_ptrs = b_batch_ptr + (k_range[:, None] * stride_bk
                                + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=(k_range[:, None] < K) & mask_n[None, :],
                    other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # write back
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_ch
                            + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 3. Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] (bool/int/float) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to fp16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16) and gates (sigmoid)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (fused via Triton when mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            m = mask.to(torch.float16).unsqueeze(-1).contiguous()
            left = left * m
            right = right * m

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (A @ Bᵀ) using a Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) – contiguous layout required for the kernel
    left_flat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch_h = B * H
    left_view = left_flat.view(batch_h, N, N)
    right_view = right_flat.view(batch_h, N, N)

    out_view = torch.empty_like(left_view)  # (B*H, N, N), fp16

    # strides for the three‑dimensional tensors
    stride_a_batch, stride_ah, stride_ak = left_view.stride()
    stride_b_batch, stride_bk, stride_bn = right_view.stride()
    stride_c_batch, stride_ch, stride_cn = out_view.stride()

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        batch_h,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        left_view,
        right_view,
        out_view,
        stride_a_batch, stride_ah, stride_ak,
        stride_b_batch, stride_bk, stride_bn,
        stride_c_batch, stride_ch, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_view.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
