@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_ab, stride_am, stride_ak,
    stride_bb, stride_bk, stride_bn,
    stride_cb, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask outs-of-bounds
    m_mask = offs_m < M
    n_mask = offs_n < N

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offset
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        k_mask = offs_k < K

        # load a tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_batch * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=m_mask[:, None] & k_mask[None, :], other=0.0)

        # load b tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + pid_batch * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=k_mask[:, None] & n_mask[None, :], other=0.0)

        # accumulate

    # Cast output
    c = acc.to(tl.float16)

    # store
    c_ptrs = c_ptr + pid_batch * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=m_mask[:, None] & n_mask[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', False)

    # 1. Input layernorm
    x = torch.nn.functional.layer_norm(input_tensor,
               normalized_shape=(dim,),
               weight=weights['norm.weight'].to(device=device, dtype=torch.float32),
               bias=weights['norm.bias'].to(device=device, dtype=torch.float32))
    # cast to fp16
    x = x.to(torch.float16)
    # 2. Cast weights to fp16
    left_proj_w = weights['left_proj.weight'].to(device=device, dtype=torch.float16)
    right_proj_w = weights['right_proj.weight'].to(device=device, dtype=torch.float16)
    left_gate_w = weights['left_gate.weight'].to(device=device, dtype=torch.float16)
    right_gate_w = weights['right_gate.weight'].to(device=device, dtype=torch.float16)
    out_gate_w = weights['out_gate.weight'].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights['to_out_norm.weight'].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights['to_out_norm.bias'].to(device=device, dtype=torch.float16)
    to_out_w = weights['to_out.weight'].to(device=device, dtype=torch.float16)

    # 3. Linear projections
    left = torch.nn.functional.linear(x, left_proj_w)  # shape (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)

    # 4. Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # 5. Apply mask and gates
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()  # B*N*N*H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)
        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        del left_gate, right_gate
    else:
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # 6. Core reduction via Triton batched matmul
    # Prepare tensors
    B, N, _, _ = left.shape  # B, N, N, H
    # permute to (B, H, N, N)
    left_b = left.permute(0, 3, 1, 2).contiguous()
    right_b = right.permute(0, 3, 1, 2).contiguous()
    # transpose right for multiplication
    right_t = right_b.transpose(-2, -1).contiguous()  # (B, H, N, N)
    # collapse batch*H
    batch = B * hidden_dim
    left_flat = left_b.view(batch, N, N)
    right_flat = right_t.view(batch, N, N)

    # Prepare output placeholder
    out_flat = torch.empty_like(left_flat)  # fp16

    # stride extraction
    a_ptr = left_flat
    b_ptr = right_flat
    c_ptr = out_flat

    # compute strides
    stride_ab = a_ptr.stride(0)
    stride_am = a_ptr.stride(1)
    stride_ak = a_ptr.stride(2)
    stride_bb = b_ptr.stride(0)
    stride_bk = b_ptr.stride(1)  # row dimension (K)
    stride_bn = b_ptr.stride(2)
    stride_cb = c_ptr.stride(0)
    stride_cm = c_ptr.stride(1)
    stride_cn = c_ptr.stride(2)

    # block configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # compute grid
    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        a_ptr,
        b_ptr,
        c_ptr,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # reshape output back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # 7. Second LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 8. Apply out_gate
    out = out * out_gate

    # 9. Final linear projection
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # cast back to original dtype
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Input LayerNorm (float32) → cast to fp16.
2. Linear projections (left/right) from C → H (hidden_dim) using fp16 GEMM.
3. Compute three sigmoid gates (left, right, out) with Linear+sigmoid.
4. Fuse mask (if present) and gate multiplications via a small Triton kernel
   (in‑place, no extra temporaries).
5. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication (B·H independent GEMMs)
   using a custom Triton kernel that tiles the output and accumulates in fp32.
6. Layer‑normalize over hidden dimension H.
7. Apply the output gate (element‑wise).
8. Final linear projection H → C.
9. Cast back to the input dtype (float32).

The kernel works for batch sizes 1‑2, sequence lengths up to 1024 and
channel dimensions 128‑384.  It uses mixed‑precision (fp16 storage, fp32
accumulation) and fully exploits Tensor‑Core friendly tiling.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask and gates (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load values
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Derive mask index (one mask per (b,i,j) entry)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gates
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the core reduction
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_ab, stride_am, stride_ak,   # strides for A: (batch, M, K)
    stride_bb, stride_bk, stride_bn,   # strides for B: (batch, K, N)
    stride_cb, stride_cm, stride_cn,   # strides for C: (batch, M, N)
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program IDs
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / columns
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch_id * stride_ab \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + batch_id * stride_bb \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Cast back to fp16 for storage
    c = acc.to(tl.float16)

    # Store result
    c_ptrs = c_ptr + batch_id * stride_cb \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             c,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)  # fp16 for the rest of the pipeline

    # --------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # --------------------------------------------------------------
    # 4) Compute sigmoid gates
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 5) Fuse mask and gates (in‑place) via Triton
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and flattened
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()                     # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)
        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # --------------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, _ = left.shape

    # Rearrange to (B, H, N, N) for batched GEMM
    left_b  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Transpose right to match A @ B^T semantics
    right_t = right_b.transpose(-2, -1).contiguous()  # (B, H, N, N)

    # Collapse batch and hidden dimensions: (B*H, N, N)
    batch = B * hidden_dim
    left_flat  = left_b.view(batch, N, N)
    right_flat = right_t.view(batch, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides for the kernel
    stride_ab = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cb = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # Tile sizes (tuned for the B200 GPU)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_cm, stride_cn,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 8) Apply output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # --------------------------------------------------------------
    # 10) Cast back to original dtype if necessary
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
