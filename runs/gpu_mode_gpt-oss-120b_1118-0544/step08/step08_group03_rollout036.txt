def mask_gate_fusion_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    stride_b_l, stride_i_l, stride_j_l, stride_h_l,
    stride_b_r, stride_i_r, stride_j_r, stride_h_r,
    stride_b_m, stride_i_m, stride_j_m,
    stride_b_lg, stride_i_lg, stride_j_lg, stride_h_lg,
    stride_b_rg, stride_i_rg, stride_j_rg, stride_h_rg,
    H: tl.constexpr,
    N: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    total_ijb = tl.num_programs(0)  # we'll launch grid = B*N*N
    # compute indices
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offset base for this (b,i,j)
    left_base = left_ptr + b*stride_b_l + i*stride_i_l + j*stride_j_l
    right_base = right_ptr + b*stride_b_r + i*stride_i_r + j*stride_j_r
    left_gate_base = left_gate_ptr + b*stride_b_lg + i*stride_i_lg + j*stride_j_lg
    right_gate_base = right_gate_ptr + b*stride_b_rg + i*stride_i_rg + j*stride_j_rg
    mask_val = tl.load(mask_ptr + b*stride_b_m + i*stride_i_m + j*stride_j_m)

    # loop over hidden dimension in chunks
    for off in range(0, H, BLOCK_H):
        cur = tl.arange(0, BLOCK_H)
        mask = off + cur < H
        idx = off + cur
        # load vectors
        l = tl.load(left_base + idx * stride_h_l, mask=mask)
        r = tl.load(right_base + idx * stride_h_r, mask=mask)
        lg = tl.load(left_gate_base + idx * stride_h_lg, mask=mask)
        rg = tl.load(right_gate_base + idx * stride_h_rg, mask=mask)
        # apply

        r = r * mask_val * rg
        # store
        tl.store(left_base + idx * stride_h_l, l, mask=mask)
        tl.store(right_base + idx * stride_h_r, r, mask=mask)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input (FP32 for stability) → FP16.
2. Linear projections to hidden dimension (left, right) (FP16 GEMM).
3. Compute three sigmoid gates (left_gate, right_gate, out_gate) (FP16).
4. Fuse mask (if present) and gates into left/right in‑place with a Triton
   kernel that processes one (batch, i, j) element at a time and loops over
   the hidden dimension.
5. Core reduction: for each hidden head compute
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication
       out = (left_perm @ right_permᵀ)   with shape (B, H, N, N).
6. Layer‑norm over the hidden dimension.
7. Apply the output gate.
8. Final linear projection back to the original channel dimension.
9. Cast back to the original dtype.

The heavy O(N³·H) reduction is performed by cuBLAS (torch.matmul); only the
mask‑gate fusion is done in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    # strides for left
    stride_b_l, stride_i_l, stride_j_l, stride_h_l,
    # strides for right
    stride_b_r, stride_i_r, stride_j_r, stride_h_r,
    # strides for left gate
    stride_b_lg, stride_i_lg, stride_j_lg, stride_h_lg,
    # strides for right gate
    stride_b_rg, stride_i_rg, stride_j_rg, stride_h_rg,
    # strides for mask (B,N,N)
    stride_b_m, stride_i_m, stride_j_m,
    # compile‑time constants
    H: tl.constexpr, N: tl.constexpr, BLOCK_H: tl.constexpr
):
    """In‑place fusion of mask and left/right gates.

    For each (b, i, j) element:
        left[b,i,j,:]  = left[b,i,j,:]  * mask[b,i,j] * left_gate[b,i,j,:]
        right[b,i,j,:] = right[b,i,j,:] * mask[b,i,j] * right_gate[b,i,j,:]
    """
    pid = tl.program_id(0)

    # Decode (b,i,j) from pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Base pointers for this (b,i,j)
    left_base      = left_ptr      + b * stride_b_l  + i * stride_i_l  + j * stride_j_l
    right_base     = right_ptr     + b * stride_b_r  + i * stride_i_r  + j * stride_j_r
    left_gate_base = left_gate_ptr + b * stride_b_lg + i * stride_i_lg + j * stride_j_lg
    right_gate_base= right_gate_ptr+ b * stride_b_rg + i * stride_i_rg + j * stride_j_rg

    # Load scalar mask value (FP16)
    mask_val = tl.load(mask_ptr + b * stride_b_m + i * stride_i_m + j * stride_j_m)

    # Loop over hidden dimension in blocks
    for off in range(0, H, BLOCK_H):
        cur = tl.arange(0, BLOCK_H)
        mask = (off + cur) < H
        idx = off + cur

        # Load vectors
        l  = tl.load(left_base      + idx * stride_h_l,  mask=mask)
        r  = tl.load(right_base     + idx * stride_h_r,  mask=mask)
        lg = tl.load(left_gate_base + idx * stride_h_lg, mask=mask)
        rg = tl.load(right_gate_base+ idx * stride_h_rg, mask=mask)

        # Apply mask and gates
        l = l * mask_val * lg
        r = r * mask_val * rg

        # Store results back in‑place
        tl.store(left_base      + idx * stride_h_l,  l, mask=mask)
        tl.store(right_base     + idx * stride_h_r,  r, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see code for expected keys)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)  # work in fp16 from here on

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    # Shapes: (B, N, N, hidden_dim)
    left  = F.linear(x, left_proj_w)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Fuse mask and gates (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure tensors are contiguous for simple stride calculation
        left  = left.contiguous()
        right = right.contiguous()
        left_gate  = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp16 = mask.to(device=device, dtype=torch.float16).contiguous()

        # Strides (in elements)
        stride_b_l, stride_i_l, stride_j_l, stride_h_l = left.stride()
        stride_b_r, stride_i_r, stride_j_r, stride_h_r = right.stride()
        stride_b_lg, stride_i_lg, stride_j_lg, stride_h_lg = left_gate.stride()
        stride_b_rg, stride_i_rg, stride_j_rg, stride_h_rg = right_gate.stride()
        stride_b_m, stride_i_m, stride_j_m = mask_fp16.stride()

        B, N, _, _ = left.shape
        BLOCK_H = 64  # fits most hidden dimensions (<=768)

        grid = (B * N * N,)

        _mask_gate_fusion_kernel[grid](
            left, right, mask_fp16,
            left_gate, right_gate,
            stride_b_l, stride_i_l, stride_j_l, stride_h_l,
            stride_b_r, stride_i_r, stride_j_r, stride_h_r,
            stride_b_lg, stride_i_lg, stride_j_lg, stride_h_lg,
            stride_b_rg, stride_i_rg, stride_j_rg, stride_h_rg,
            stride_b_m, stride_i_m, stride_j_m,
            H=hidden_dim,
            N=N,
            BLOCK_H=BLOCK_H,
            num_warps=4,
        )
        # Gates are no longer needed
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for matmul
    left_b  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Matrix multiplication: (B, H, N, N) @ (B, H, N, N)^T -> (B, H, N, N)
    out_b = torch.matmul(left_b, right_b.transpose(-2, -1))

    # Restore original layout (B, N, N, H)
    out = out_b.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
