@triton.jit
def matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    batch_stride,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # i block
    pid_n = tl.program_id(1)  # j block
    pid_b = tl.program_id(2)  # batch

    # Offsets
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    # Compute base pointers
    a_base = a_ptr + pid_b * batch_stride
    b_base = b_ptr + pid_b * batch_stride
    c_base = c_ptr + pid_b * batch_stride

    # Mask for out-of-bound i/j
    i_valid = i_offset < N
    j_valid = j_offset < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_offset = k_start + tl.arange(0, BLOCK_K)
        # Mask for k dimension
        k_valid = k_offset < N

        # Load A block
        a_ptrs = a_base + i_offset[:, None] * N + k_offset[None, :]
        a = tl.load(a_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0)

        # Load B_T block (k,j)
        b_ptrs = b_base + k_offset[:, None] * N + j_offset[None, :]
        b = tl.load(b_ptrs, mask=k_valid[:, None] & j_valid[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)  # a: (M,K), b: (K,N)

    # Write back C block
    c_ptrs = c_base + i_offset[:, None] * N + j_offset[None, :]

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# Triton kernel for batched matrix multiplication with transposed second operand.
@triton.jit
def _matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,  # pointers to fp16 tensors
    N,                    # sequence length
    batch_stride,         # stride between batches = N*N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 3‑D grid: (i block, j block, batch)
    pid_m = tl.program_id(0)  # block row index (i)
    pid_n = tl.program_id(1)  # block column index (j)
    pid_b = tl.program_id(2)  # batch index (b * hidden_dim)

    # Compute start indices of the block
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    i_valid = i < N
    j_valid = j < N

    # Base pointer for this batch
    a_batch = a_ptr + pid_b * batch_stride
    b_batch = b_ptr + pid_b * batch_stride
    c_batch = c_ptr + pid_b * batch_stride

    # Accumulator in float32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_idx = k_start + tl.arange(0, BLOCK_K)
        k_valid = k_idx < N

        # Load a[i, k]   -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + i[:, None] * N + k_idx[None, :]
        a = tl.load(a_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0)

        # Load b_T[k, j] i.e. right[k, j] -> shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + k_idx[:, None] * N + j[None, :]
        b = tl.load(b_ptrs, mask=k_valid[:, None] & j_valid[None, :], other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)   # a: (M,K), b: (K,N) → (M,N)

    # Write back result
    c_ptrs = c_batch + i[:, None] * N + j[None, :]
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=i_valid[:, None] & j_valid[None, :])

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update
    (TriMul).  The implementation fuses the final O(N³) reduction using a custom
    Triton kernel while keeping the surrounding linear / layer‑norm operations
    in PyTorch (in fp16 for speed).  The output dtype matches the input dtype.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy work
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – also fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # ------------------------------------------------------------------
    # 5. Apply mask and gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) → (B, N, N, 1)
        mask_fp = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask_fp
        right = right_proj * right_gate * mask_fp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp
            right = right * mask_fp

    # ------------------------------------------------------------------
    # 6. Core O(N³) reduction using Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Rearrange to (B, H, N, N) then merge batch & head dimensions
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    # Launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    batch_stride = N * N  # because tensors are contiguous (row‑major)
    _matmul_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        batch_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output layer‑norm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 8. Cast back to the input dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward.

The implementation follows the PyTorch reference but replaces the
O(N³) reduction with a custom Triton kernel that computes,
for each batch‑head pair, C = A @ Bᵀ where A and B have shape (N, N).
All heavy arithmetic is performed in fp16 with a float32 accumulator.
LayerNorms, linear projections and gates are kept in PyTorch (fp16)
for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,       # fp16 pointers to (batch, N, N)
    N,                         # sequence length
    batch_stride,              # stride between successive (batch, *) slices = N*N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for each batch using tiled GEMM."""
    pid_m = tl.program_id(0)   # block row index (i)
    pid_n = tl.program_id(1)   # block column index (j)
    pid_b = tl.program_id(2)   # batch (combined batch * hidden_dim)

    # Block start coordinates
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    i_valid = i < N
    j_valid = j < N

    # Base pointers for this batch
    a_batch = a_ptr + pid_b * batch_stride
    b_batch = b_ptr + pid_b * batch_stride
    c_batch = c_ptr + pid_b * batch_stride

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_idx = k_start + tl.arange(0, BLOCK_K)
        k_valid = k_idx < N

        # Load A block (i, k) – shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + i[:, None] * N + k_idx[None, :]
        a = tl.load(a_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0)

        # Load Bᵀ block (k, j) = right[k, j] – shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + k_idx[:, None] * N + j[None, :]
        b = tl.load(b_ptrs, mask=k_valid[:, None] & j_valid[None, :], other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)      # (M,K) @ (K,N) → (M,N)

    # Write output block
    c_ptrs = c_batch + i[:, None] * N + j[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=i_valid[:, None] & j_valid[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor – torch.Tensor of shape [B, N, N, C]
            mask         – torch.Tensor of shape [B, N, N] or None
            weights      – dict of model parameters (float32)
            config       – dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (dtype matches input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the compute‑heavy parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # ------------------------------------------------------------------
    # 5) Apply mask and gating
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp
        right = right_proj * right_gate * mask_fp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp
            right = right * mask_fp

    # ------------------------------------------------------------------
    # 6) Core O(N³) reduction via Triton:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # Rearrange to (B, H, N, N) then merge batch & head dimensions
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (

        right_flat,
        out_flat,
        N,
        batch_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, output‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
