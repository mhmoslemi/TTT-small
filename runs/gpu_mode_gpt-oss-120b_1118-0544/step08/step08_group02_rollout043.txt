@triton.jit
def _bmm_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, M, N, K,
    stride_l_b, stride_l_m, stride_l_k,
    stride_r_b, stride_r_k, stride_r_n,
    stride_o_b, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    m_offset = pid_m * BLOCK_M
    n_offset = pid_n * BLOCK_N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute the size of each block (handling edge)
    # Actually we can compute mask for out-of-range rows/cols and apply at store.

    # Compute row indices for left A and out C
    row_idx = m_offset + tl.arange(0, BLOCK_M)
    col_idx = n_offset + tl.arange(0, BLOCK_N)

    # mask for valid rows/cols
    row_mask = row_idx < M
    col_mask = col_idx < N

    # For each tile in K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Load A block
        a_ptr = left_ptr + batch_id * stride_l_b + row_idx[:, None] * stride_l_m + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_l_k
        a = tl.load(a_ptr, mask=row_mask[:, None] & (k_offset + tl.arange(0, BLOCK_K) < K), other=0.0).to(tl.float32)

        # Load B_T block (right transposed)
        # We need to load B_T block of shape (BLOCK_K, BLOCK_N):
        # row indices = k_offset + tl.arange(0, BLOCK_K) (original column)
        # col indices = n_offset + tl.arange(0, BLOCK_N) (original row)
        b_ptr = right_ptr + batch_id * stride_r_b + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_r_n + (n_offset + tl.arange(0, BLOCK_N))[None, :] * stride_r_m
        # Note: stride_r_n is original column stride (1), stride_r_m is row stride (N)
        # But we must ensure correct order: row offset is original column index: index * stride_r_n, col offset is original row index: index * stride_r_m.
        b = tl.load(b_ptr, mask=(k_offset + tl.arange(0, BLOCK_K) < K)[:, None] & col_mask[None, :], other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)
    # Store result
    c_ptr = out_ptr + batch_id * stride_o_b + row_idx[:, None] * stride_o_m + col_idx[None, :] * stride_o_n
    # Apply mask for valid rows/cols
    mask = row_mask[:, None] & col_mask[None, :]
    tl.store(c_ptr, acc.to(tl.float16), mask=mask)

def custom_kernel(...):

    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Apply layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Cast to fp16
    x = x.to(torch.float16)

    # Convert weight tensors to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Projection
    left_proj = torch.nn.functional.linear(x, left_proj_w)  # (B,N,N,hidden)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Apply mask and gate
    if mask is not None and not nomask:
        # mask to fp16
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj, right_proj, left_gate, right_gate, mask_fp16, block_size=1024
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # Flatten batch and hidden
    B, N, _, H = left.shape
    # Ensure contiguity
    left = left.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Compute strides (in elements)
    stride_l_b = left_flat.stride(0)
    stride_l_m = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    stride_r_b = right_flat.stride(0)
    stride_r_k = right_flat.stride(1)
    stride_r_n = right_flat.stride(2)

    stride_o_b = out_flat.stride(0)
    stride_o_m = out_flat.stride(1)
    stride_o_n = out_flat.stride(2)

    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H, 
            triton.cdiv(N, BLOCK_M), 
            triton.cdiv(N, BLOCK_N))

    _bmm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H, N, N, N,
        stride_l_b, stride_l_m, stride_l_k,
        stride_r_b, stride_r_k, stride_r_n,
        stride_o_b, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out = out * out_gate

    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to fp16 for the heavy compute.
3. Linear projections (left/right) and gating (left/right/out) using fp16 weights.
4. Fuse mask, projection and gating for left/right branches with a small Triton kernel.
5. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (left @ rightᵀ) via a custom Triton kernel that
   processes (batch*hidden) matrices of size N×N in tiles.
6. Apply a second LayerNorm, multiply by the out‑gate and project back to the
   original channel dimension.
7. Cast the result back to the input dtype.

All heavy arithmetic is performed in fp16 (accumulated in fp32) and the kernel
covers both the mask‑gate fusion and the batched matrix multiplication.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, projection and gate for left/right tensors
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)
    right_gate_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    left_out_ptr,    # fp16 * (B, N, N, H)
    right_out_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # i64 = B * N * N * H
    hidden_dim,      # i64 = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each element belongs to a (b,i,j) entry; the mask index repeats each H entries
    mask_idx = offs // hidden_dim

    left_proj = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse mask, projection and gate for both left and right tensors using Triton.

    All tensors must be fp16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)

    Returns
    -------
    left, right : torch.Tensor
        Shape (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _bmm_kernel(
    left_ptr,          # fp16 * (BATCH, M, K)
    right_ptr,         # fp16 * (BATCH, K, N)   (original right, we will read its transpose)
    out_ptr,           # fp16 * (BATCH, M, N)
    BATCH: tl.constexpr,
    M: tl.constexpr,
    N: tl.constexpr,
    K: tl.constexpr,
    stride_l_b, stride_l_m, stride_l_k,
    stride_r_b, stride_r_m, stride_r_n,
    stride_o_b, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Starting offsets for this block
    m_offset = pid_m * BLOCK_M
    n_offset = pid_n * BLOCK_N

    # Row/column indices
    row_idx = m_offset + tl.arange(0, BLOCK_M)
    col_idx = n_offset + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    row_mask = row_idx < M
    col_mask = col_idx < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_offset = k * BLOCK_K
        k_idx = k_offset + tl.arange(0, BLOCK_K)

        k_mask = k_idx < K

        # --- Load A block (left) ---
        a_ptr = left_ptr + batch_id * stride_l_b \
                + row_idx[:, None] * stride_l_m \
                + k_idx[None, :] * stride_l_k
        a = tl.load(a_ptr, mask=row_mask[:, None] & k_mask[None, :], other=0.0)
        a = a.to(tl.float32)

        # --- Load Bᵀ block (right transposed) ---
        # Bᵀ[k, n] = right[n, k]
        b_ptr = right_ptr + batch_id * stride_r_b \
                + k_idx[:, None] * stride_r_n \
                + col_idx[None, :] * stride_r_m
        b = tl.load(b_ptr, mask=k_mask[:, None] & col_mask[None, :], other=0.0)
        b = b.to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Store the result
    c_ptr = out_ptr + batch_id * stride_o_b \
            + row_idx[:, None] * stride_o_m \
            + col_idx[None, :] * stride_o_n
    store_mask = row_mask[:, None] & col_mask[None, :]
    tl.store(c_ptr, acc.to(tl.float16), mask=store_mask)


# ----------------------------------------------------------------------
# Main entrypoint
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask" (optional) – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm over channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast weights to fp16 (all Linear layers are bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)   # (B, N, N, H)

    left_gate  = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid()  # (B, N, N, H)
    out_gate   = F.linear(x, out_gate_w).sigmoid()    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask + gating (fused via Triton when mask is used)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            block_size=1024,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (left @ rightᵀ) in Triton.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # left/right shape: (B, N, N, H)

    # reshape to (B*H, N, N) for batched GEMM
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left.view(B * H, N, N)   # (BATCH, M, K)
    right_flat = right.view(B * H, N, N) # (BATCH, K, N) – we will read its transpose

    out_flat = torch.empty_like(left_flat)

    # strides (in elements)
    stride_l_b = left_flat.stride(0)
    stride_l_m = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    stride_r_b = right_flat.stride(0)
    stride_r_m = right_flat.stride(1)  # row stride (original rows)
    stride_r_n = right_flat.stride(2)  # col stride (original cols)

    stride_o_b = out_flat.stride(0)
    stride_o_m = out_flat.stride(1)
    stride_o_n = out_flat.stride(2)

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                     # batch (BATCH)
        triton.cdiv(N, BLOCK_M),  # M dimension blocks
        triton.cdiv(N, BLOCK_N),  # N dimension blocks
    )

    _bmm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H, N, N, N,
        stride_l_b, stride_l_m, stride_l_k,
        stride_r_b, stride_r_m, stride_r_n,
        stride_o_b, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)

    return out
