    pid = tl.program_id(0)
    # number of blocks in m and n dimensions
    num_m_blocks = tl.cdiv(M, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    # total blocks per batch
    blocks_per_batch = num_m_blocks * num_n_blocks

    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    m_idx = block_idx // num_n_blocks
    n_idx = block_idx % num_n_blocks

    offs_m = m_idx * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = n_idx * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

"""
custom_kernel: Optimized outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension `dim`.
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (`left_proj`, `right_proj`) and gating projections
   (`left_gate`, `right_gate`, `out_gate`).
4. Fuse mask (if present), projection and gating for the left/right tensors
   with a tiny Triton kernel.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix‑multiply `left @ rightᵀ` for every head.
   Implemented with a custom Triton kernel that processes all
   (batch × hidden) matrices in one launch.
6. Apply output LayerNorm, multiply by `out_gate`, and a final linear
   projection back to the original `dim`.
7. Cast the result back to the original dtype.

All heavy arithmetic is performed in fp16 (accumulated in fp32 for
numerical safety).  At least one Triton kernel is used, satisfying the
requirement.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
#  Kernel 1: fuse projection, gating and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B,N,N,H)
    right_proj_ptr,  # fp16 * (B,N,N,H)
    left_gate_ptr,   # fp16 * (B,N,N,H)
    right_gate_ptr,  # fp16 * (B,N,N,H)
    mask_ptr,        # fp16 * (B,N,N)
    left_out_ptr,    # fp16 * (B,N,N,H)
    right_out_ptr,   # fp16 * (B,N,N,H)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # each mask element repeats `hidden_dim` times in the flat view
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    left_out  = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr  + offs, left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Apply (proj * gate * mask) for left and right tensors using Triton.
    All tensors must be fp16 and contiguous.
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out

@triton.jit
def _batched_matmul_transpose_kernel(
    a_ptr,                # fp16 * (B*H, M, K)
    b_ptr,                # fp16 * (B*H, N, K)   (original right)
    c_ptr,                # fp16 * (B*H, M, N)
    batch,                # i64 = B*H
    M, N, K,              # i64 dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of tiles per matrix
    num_m_blocks = tl.cdiv(M, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_m_blocks * num_n_blocks

    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    m_idx = block_idx // num_n_blocks
    n_idx = block_idx % num_n_blocks

    # tile offsets
    offs_m = m_idx * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = n_idx * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch_ptr = a_ptr + batch_idx * stride_a_batch
    b_batch_ptr = b_ptr + batch_idx * stride_b_batch
    c_batch_ptr = c_ptr + batch_idx * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m +

        acc += tl.dot(a, b)

    c = acc.to(tl.float16)

    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_m +
                            offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[:, None])


def _batched_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out = left @ rightᵀ for each head.
    left/right shape: (B, N, N, H)  (fp16)
    Returns out with same shape and dtype.
    """
    B, N, _, H = left.shape
    batch_heads = B * H

    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(batch_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(batch_heads, N, N)

    out_flat = torch.empty_like(left_flat)

    # strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_n, stride_b_k = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # block sizes – chosen to fit typical B200 SMEM limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_m_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_n_blocks = (N + BLOCK_N - 1) // BLOCK_N
    blocks_per_batch = num_m_blocks * num_n_blocks
    total_blocks = batch_heads * blocks_per_batch

    grid = (total_blocks,)

    _batched_matmul_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_heads,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C]
        - mask        : torch.Tensor of shape [B, N, N] or None
        - weights     : mapping name → torch.Tensor
        - config      : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(dtype_in)
    norm_b = weights["norm.bias"].to(dtype_in)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # ------------------------------------------------------------------
    compute_dtype = torch.float16
    x = x.to(compute_dtype)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(compute_dtype)
    right_proj_w  = weights["right_proj.weight"].to(compute_dtype)
    left_gate_w   = weights["left_gate.weight"].to(compute_dtype)
    right_gate_w  = weights["right_gate.weight"].to(compute_dtype)
    out_gate_w    = weights["out_gate.weight"].to(compute_dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype)
    to_out_w      = weights["to_out.weight"].to(compute_dtype)

    # ------------------------------------------------------------------
    # 4) Linear projections and gating (fp16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)            # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Fuse mask, projection and gate (via Triton when needed)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(compute_dtype).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: out = left @ rightᵀ  (batched matmul)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose(left, right)   # shape (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)                  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
