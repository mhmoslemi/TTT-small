"""
TriMul (outgoing) forward kernel – Triton accelerated.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast to FP16 for the heavy compute.
3. Linear projections (left / right) and gates (left / right / out) are
   bias‑free `nn.Linear`s – performed with `torch.nn.functional.linear`.
4. If a mask is present and `nomask` is False, a Triton kernel fuses
   projection, gate and mask for both left and right tensors.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden slice.
   Implemented as a tiled GEMM kernel in Triton (no cuBLAS).
6. Second LayerNorm over the hidden dimension, multiply by the output gate,
   and project back to the original dimension.
7. Cast the result back to the input dtype.

The kernel works for B ∈ {1,2}, N up to 1024 and hidden_dim = 128.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)
    right_gate_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    left_out_ptr,    # fp16 * (B, N, N, H)
    right_out_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # int64 scalar = B*N*N*H
    hidden_dim,      # int64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gate and mask for left/right tensors."""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard out‑of‑bounds
    mask = offs < total_elems

    # Each mask element corresponds to a (b,i,k) entry – it repeats H times.
    mask_idx = offs // hidden_dim

    # Load
    left_proj  = tl.load(left_proj_ptr + offs,  mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    left_gate  = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m          = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Fuse
    left_out  = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr  + offs, left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """Fuse projection, gate and mask with a Triton kernel."""
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem size
    stride_am, stride_ak, stride_an,   # a strides
    stride_bm, stride_bk, stride_bn,   # b strides
    stride_cm, stride_cm1, stride_cn,  # c strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   where A: (BH,M,K), B: (BH,K,N)."""
    pid = tl.program_id(0)   # batch*hidden index
    pid_m = tl.program_id(1) # block row
    pid_n = tl.program_id(2) # block col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers
    a_ptrs = a_ptr + pid * stride_am \
             + offs_m[:, None] * stride_ak \
             + offs_k[None, :] * stride_an
    b_ptrs = b_ptr + pid * stride_bm \
             + offs_k[:, None] * stride_bk \
             + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_an
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c_ptrs = c_ptr + pid * stride_cm \
             + offs_m[:, None] * stride_cm1 \
             + offs_n[None, :] * stride_cn
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_gemm(
    a: torch.Tensor,
    b: torch.Tensor,
    block_m: int = 64,
    block_n: int = 64,
    block_k: int = 64,
) -> torch.Tensor:
    """Batched matrix multiplication using the Triton GEMM kernel."""
    assert a.is_contiguous() and b.is_contiguous()
    BH, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty_like(a)

    grid = (
        BH,
        triton.cdiv(M, block_m),
        triton.cdiv(N, block_n),
    )

    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        a.stride(0), a.stride(1), a.stride(2),
        b.stride(0), b.stride(1), b.stride(2),
        c.stride(0), c.stride(1), c.stride(2),
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return c


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None (0/1)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gates (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)               # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)             # (B, N, N, H)

    left_gate = F.linear(x, left_gate_w).sigmoid()     # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid()   # (B, N, N, H)
    out_gate = F.linear(x, out_gate_w).sigmoid()       # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask, projection and gate (Triton kernel if needed)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
            block_size=1024,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but user asked to ignore it -> treat as all‑ones
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (left @ right^T) per hidden slice.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # transpose right so that its K dimension aligns with left's K
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    out_flat = _batched_gemm(left_flat, right_flat,
                             block_m=64, block_n=64, block_k=64)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
