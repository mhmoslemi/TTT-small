"""
TriMul (outgoing) forward pass implemented with Triton.
----------------------------------------------------
Algorithm
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast to FP16 for compute‑heavy kernels.
3. Linear projections (left / right) and gating (left / right / out) are
   performed with bias‑free Linear layers (weights are cast to FP16).
4. Optional mask is applied element‑wise after gating.
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (B*H, N, N) @ (B*H, N, N)^T
   realised by a custom Triton kernel that tiles the M, N and K
   dimensions (BLOCK_M/N/K = 64).  Accumulation is in FP32 and the
   result is stored in FP16.
6. Second LayerNorm over the hidden dimension, multiplication by the
   output gate and a final linear projection back to `dim`.
7. Cast the result back to the original input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Optional

@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix shapes
    stride_az, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bz, stride_bk, stride_bn,   # strides for B (batch, K, N)
    stride_cz, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B  where A,B,C are (batch, M, K), (batch, K, N), (batch, M, N)."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets of the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialise the accumulation buffer in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current blocks of A and B
        a_ptrs = (
            a_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # Load blocks with boundary guards
        a = tl.load(
            a_ptrs,
            mask=(mask_m[:, None] & mask_k[None, :]),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(mask_k[:, None] & mask_n[None, :]),
            other=0.0,
        )
        # Accumulate partial results
        acc += tl.dot(a, b)

    # Write the result back to C (cast to FP16)
    c = acc.to(tl.float16)
    c_ptrs = (
        c_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )

def _batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Perform batched matrix multiplication using the Triton kernel.
    a, b: tensors of shape (B, H, N, N) and dtype float16, contiguous.
    Returns c of shape (B, H, N, N), dtype float16.
    """
    B, H, N, _ = a.shape
    batch = B * H

    a_flat = a.view(batch, N, N)
    b_flat = b.view(batch, N, N)
    c_flat = torch.empty_like(a_flat)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_kernel[grid](
        a_flat,
        b_flat,
        c_flat,
        N,
        N,
        N,
        *a_flat.stride(),
        *b_flat.stride(),
        *c_flat.stride(),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c_flat.view(B, H, N, N)

def custom_kernel(
    data: Tuple[torch.Tensor, Optional[torch.Tensor], Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] or None (0/1)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gates (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_fp16, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x_fp16, right_proj_w)        # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))     # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask (if present) and combine projection with gate
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16)
        mask_fp16 = mask_fp16.unsqueeze(-1)  # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM using a Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # Rearrange to (B, H, N, N) for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Transpose the last two dimensions of right so that we compute
    # (i,k) @ (k,j) -> (i,j)
    right_T = right_perm.transpose(2, 3).contiguous()  # (B, H, N, N)

    # Batched matmul via Triton
    out_perm = _batched_matmul(left_perm, right_T)     # (B, H, N, N)

    # Rearrange back to (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
