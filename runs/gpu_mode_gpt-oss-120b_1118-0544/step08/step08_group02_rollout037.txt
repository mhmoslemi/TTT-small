"""
custom_kernel: Triton‑accelerated outgoing TriMul forward (AlphaFold3).

The algorithm follows the PyTorch reference:
1. Layer‑norm the input over the channel dimension.
2. Linear projections (left/right) and gating (left/right/out) in FP16.
3. Fuse mask, projection and gate for left/right tensors via a small Triton kernel.
4. Core O(N³) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom batched GEMM where each (b,h) pair is a separate
   matrix‑multiply (N×N) @ (N×N)ᵀ using a tiled Triton kernel.
5. Layer‑norm over the hidden dimension, multiply by the output‑gate and
   project back to the original channel dimension.
All heavy compute is performed in FP16 with FP32 accumulation for accuracy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Kernel 1: fuse mask, projection and gate (left/right)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B, N, N, H)
    right_proj_ptr,  # fp16 * (B, N, N, H)
    left_gate_ptr,   # fp16 * (B, N, N, H)
    right_gate_ptr,  # fp16 * (B, N, N, H)
    mask_ptr,        # fp16 * (B, N, N)
    left_out_ptr,    # fp16 * (B, N, N, H)
    right_out_ptr,   # fp16 * (B, N, N, H)
    total_elems,     # int64 = B * N * N * H
    hidden_dim,      # int64 = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # index of the mask element for each flattened element
    mask_idx = offs // hidden_dim

    left_proj = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    left_out = left_proj * left_gate * m
    right_out = right_proj * right_gate * m

    tl.store(left_out_ptr + offs, left_out, mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse mask, projection and gate for left/right tensors using Triton.
    All tensors must be contiguous FP16 with shape (B, N, N, H) and mask (B,N,N).
    Returns left, right tensors of the same shape.
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Kernel 2: core TriMul reduction (batched GEMM with transpose)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,   # fp16 * (B, N, N, H)
    right_ptr,  # fp16 * (B, N, N, H)
    out_ptr,    # fp16 * (B, N, N, H)
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # block row (i)
    pj = tl.program_id(1)           # block col (j)
    pb = tl.program_id(2)           # batch * hidden index

    # decode batch and hidden index
    b = pb // H
    h = pb % H

    i_start = pid * BLOCK_M
    j_start = pj * BLOCK_N

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K

        # ---------- load left block (M, K) ----------
        left_base = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + i_start * stride_left_i
            + k_start * stride_left_k
        )
        i_off = tl.arange(0, BLOCK_M)[:, None] * stride_left_i
        k_off = tl.arange(0, BLOCK_K)[None, :] * stride_left_k
        mask_i = (i_start + tl.arange(0, BLOCK_M)) < N
        mask_k = (k_start + tl.arange(0, BLOCK_K)) < N
        left = tl.load(
            left_base + i_off + k_off,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ---------- load right block (K, N) ----------
        # right is indexed as (b, j, k, h); we need (k, j) order for matmul
        right_base = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + k_start * stride_right_k
            + j_start * stride_right_j
        )
        k_off_r = tl.arange(0, BLOCK_K)[:, None] * stride_right_k
        j_off = tl.arange(0, BLOCK_N)[None, :] * stride_right_j
        mask_j = (j_start + tl.arange(0, BLOCK_N)) < N
        right = tl.load(
            right_base + k_off_r + j_off,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # ---------- multiply‑accumulate ----------
        acc += tl.dot(left, right)

    # ---------- store result ----------
    out_base = out_ptr + b * stride_out_b + h * stride_out_h
    i_idx = i_start + tl.arange(0, BLOCK_M)
    j_idx = j_start + tl.arange(0, BLOCK_N)
    mask_i = i_idx < N
    mask_j = j_idx < N
    store_mask = mask_i[:, None] & mask_j[None, :]

    ptrs = out_base + i_idx[:, None] * stride_out_i + j_idx[None, :] * stride_out_j
    tl.store(ptrs, acc.to(tl.float16), mask=store_mask)


def _tri_mul(
    left: torch.Tensor,
    right: torch.Tensor,
    block_m: int = 64,
    block_n: int = 64,
    block_k: int = 32,
) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using Triton.
    left/right: (B, N, N, H) FP16 tensors.
    Returns out of the same shape and dtype.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides (as int64)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    grid = (
        triton.cdiv(N, block_m),
        triton.cdiv(N, block_n),
        B * H,
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_rj,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return out


# ----------------------------------------------------------------------
# Main entrypoint
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask" (optional) – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast model weights to FP16 (bias‑free linear layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Gating (sigmoid) and optional mask fusion
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()   # (B, N, N, H)
    right_gate = F.linear(x, right_gate_w).sigmoid()  # (B, N, N, H)
    out_gate   = F.linear(x, out_gate_w).sigmoid()    # (B, N, N, H)

    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_fp16,
            block_size=1024,
        )
    else:
        # simple element‑wise fusion (mask ignored if absent or nomask)
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            mask_fp16 = mask.to(torch.float16).contiguous()
            left = left * mask_fp16[..., None]
            right = right * mask_fp16[..., None]

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction using a custom Triton kernel
    # ------------------------------------------------------------------
    out = _tri_mul(left, right, block_m=64, block_n=64, block_k=32)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
