```python
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak, stride_ah,
    batch_stride_b, stride_bk, stride_bn, stride_bh,
    batch_stride_c, stride_cm, stride_cn, stride_ch,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # batch * head id
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    batch_head = pid
    batch = batch_head // H
    head = batch_head % H

    # Compute offsets
    offs_m = tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers with batch and head offset
    a_base = a_ptr + batch * batch_stride_a + head * stride_ah
    b_base = b_ptr + batch * batch_stride_b + head * stride_bh
    c_base = c_ptr + batch * batch_stride_c + head * stride_ch

    # Compute block offsets
    block_i = pid_m * BLOCK_M
    block_j = pid_n * BLOCK_N

    # Load mask for bounds
    mask_m = block_i + offs_m < N
    mask_n = block_j + offs_n < N

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        # compute valid k mask
        mask_k = k_offset + offs_k < N

        # Load a block (M, K)
        a_ptrs = a_base + (block_i + offs_m)[:, None] * stride_am + (k_offset + offs_k)[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load b block (K, N)
        b_ptrs = b_base + (k_offset + offs_k)[:, None] * stride_bk + (block_j + offs_n)[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Convert to FP32
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # dot product
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_base + (block_i + offs_m)[:, None] * stride_cm + (block_j + offs_n)[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

```python
    norm_bias = weights["norm.bias"].to(input_tensor.dtype)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Cast to fp16
    compute_dtype = torch.float16
    x = x.to(compute_dtype)

    # Load weights to fp16
    left_proj_w = weights["left_proj.weight"].to(compute_dtype)
    right_proj_w = weights["right_proj.weight"].to(compute_dtype)
    left_gate_w = weights["left_gate.weight"].to(compute_dtype)
    right_gate_w = weights["right_gate.weight"].to(compute_dtype)
    out_gate_w = weights["out_gate.weight"].to(compute_dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype)
    to_out_w = weights["to_out.weight"].to(compute_dtype)

    # Projections
    left_proj = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply gating and mask
    if mask is not None and not nomask:
        mask_fp16 = mask.to(compute_dtype).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # core reduction using Triton
    # transpose right for matmul
    right_t = right.permute(0, 2, 1, 3).contiguous()  # (B,N,N,H) with swapped i,j

    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Compute strides
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_batch, stride_r_k, stride_r_j, stride_r_h = right_t.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H, tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N))

    batched_matmul_kernel[grid](
        left,
        right_t,
        out,
        stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
        stride_r_batch, stride_r_k, stride_r_j, stride_r_h,
        stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
        B, N, H,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Apply output layernorm, out_gate, final linear
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)
    out = out.to(input_tensor.dtype)

    return out
```

    pid_n = tl.program_id(2)

    batch_head = pid

    head = batch_head % H

    # Starting indices of the block
    block_i = pid_m * BLOCK_M
    block_j = pid_n * BLOCK_N

    # Offsets within a block

    offs_j = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    b_base = b_ptr + batch * stride_b_batch + head * stride_bh

"""
custom_kernel: Outgoing Triangle Multiplicative Update (TriMul) forward pass.
Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension `dim`.
2. Cast to fp16 for compute‑intensive parts.
3. Linear projections (`left_proj`, `right_proj`) and gating projections
   (`left_gate`, `right_gate`, `out_gate`) using the provided weights.
4. Fuse mask, projection and gate for `left` and `right` with a small Triton
   kernel (only when a mask is present).
5. Core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication using a custom Triton
   kernel that processes each (batch, head) pair as an independent matrix
   multiply.
6. Apply output LayerNorm, multiply by `out_gate`, and a final linear
   projection back to `dim`.
7. Cast the result back to the original dtype (usually float32).

The kernel in step 5 is mandatory to satisfy the “custom kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ------------------------------------------------------------
# Triton kernel: fuse projection, gating and mask (if mask present)
# ------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr,   # fp16 * (B,N,N,H)
    right_proj_ptr,  # fp16 * (B,N,N,H)
    left_gate_ptr,   # fp16 * (B,N,N,H)
    right_gate_ptr,  # fp16 * (B,N,N,H)
    mask_ptr,        # fp16 * (B,N,N)
    left_out_ptr,    # fp16 * (B,N,N,H)
    right_out_ptr,   # fp16 * (B,N,N,H)
    total_elems,     # i64 scalar = B*N*N*H
    hidden_dim,      # i64 scalar = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds
    valid = offs < total_elems

    # Each mask element is repeated `hidden_dim` times in the flattened tensor
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=valid, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    left_out  = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs, left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _apply_proj_gate_mask(
    left_proj: torch.Tensor,
    right_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse projection, gate and mask using Triton.
    All tensors must be fp16, contiguous and have shapes:
        left_proj, right_proj, left_gate, right_gate : (B, N, N, H)
        mask                                         : (B, N, N)
    Returns:
        left, right : (B, N, N, H) = proj * gate * mask
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_out,
        right_out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return left_out, right_out


# ------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul reduction
# ------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_am, stride_ak, stride_ah,
    stride_b_batch, stride_bk, stride_bj, stride_bh,
    stride_c_batch, stride_cm, stride_cn, stride_ch,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid = tl.program_id(0)     # combined batch * head
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block column

    # decode batch and head from pid
    batch_head = pid
    batch = batch_head // H
    head  = batch_head % H

    # block start indices
    block_i = pid_m * BLOCK_M
    block_j = pid_n * BLOCK_N

    # offsets inside a block
    offs_i = tl.arange(0, BLOCK_M)
    offs_j = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_i = block_i + offs_i < N
    mask_j = block_j + offs_j < N

    # base pointers for this batch/head
    a_base = a_ptr + batch * stride_a_batch + head * stride_ah
    b_base = b_ptr + batch * stride_b_batch + head * stride_bh
    c_base = c_ptr + batch * stride_c_batch + head * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K
        mask_k = cur_k + offs_k < N

        # load A block (M x K)
        a_ptrs = a_base + (block_i + offs_i)[:, None] * stride_am + (cur_k + offs_k)[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B block (K x N)
        b_ptrs = b_base + (cur_k + offs_k)[:, None] * stride_bk + (block_j + offs_j)[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # cast to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # dot product
        acc += tl.dot(a, b)

    # store result (cast back to fp16)
    c_ptrs = c_base + (block_i + offs_i)[:, None] * stride_cm + (block_j + offs_j)[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor of shape [B, N, N] (0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # --------------------------------------------------------
    # Unpack
    # --------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # --------------------------------------------------------
    norm_w = weights["norm.weight"].to(dtype_in)
    norm_b = weights["norm.bias"].to(dtype_in)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------
    # 2) Cast to fp16 for compute‑intensive parts
    # --------------------------------------------------------
    compute_dtype = torch.float16
    x = x.to(compute_dtype)

    # --------------------------------------------------------
    # 3) Cast all linear weights to fp16 (biases are None)
    # --------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(compute_dtype)
    right_proj_w  = weights["right_proj.weight"].to(compute_dtype)
    left_gate_w   = weights["left_gate.weight"].to(compute_dtype)
    right_gate_w  = weights["right_gate.weight"].to(compute_dtype)
    out_gate_w    = weights["out_gate.weight"].to(compute_dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype)
    to_out_w      = weights["to_out.weight"].to(compute_dtype)

    # --------------------------------------------------------
    # 4) Linear projections (FP16)
    # --------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------
    # 5) Apply mask & gating (fusion via Triton if mask is used)
    # --------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(compute_dtype).contiguous()
        left, right = _apply_proj_gate_mask(
            left_proj.contiguous(),
            right_proj.contiguous(),
            left_gate.contiguous(),
            right_gate.contiguous(),
            mask_fp16,
        )
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------
    # 6) Core reduction (TriMul) → batched matmul via Triton
    # --------------------------------------------------------
    # Transpose right so that we can perform a standard matrix multiplication
    # right_t[b, k, j, h] = right[b, j, k, h]
    right_t = right.permute(0, 2, 1, 3).contiguous()

    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides in element units
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_batch, stride_r_k, stride_r_j, stride_r_h = right_t.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Block sizes (tuned for B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H, tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N))

    _batched_matmul_kernel[grid](
        left,
        right_t,
        out,
        stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
        stride_r_batch, stride_r_k, stride_r_j, stride_r_h,
        stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
        B, N, H,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------
    # 8) Cast back to original dtype (usually float32)
    # --------------------------------------------------------
    out = out.to(dtype_in)

    return out
