"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Input LayerNorm (float32 for numerical stability) → fp16.
2. Linear projections `left` and `right` from C → H (hidden_dim) via F.linear.
3. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) with linear layers.
4. Fuse mask, left‑gate and right‑gate in‑place using a tiny Triton kernel
   (avoids extra temporaries and extra memory traffic).
5. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication (B×H, N, N) → (B×H, N, N)
   which is executed by cuBLAS/Tensor‑Cores.
6. Layer‑normalize over hidden dimension H.
7. Apply `out_gate` (element‑wise).
8. Final linear projection H → C.
9. Cast back to the original dtype.

Only the mask‑gate fusion is performed with Triton; the rest uses highly‑optimised
PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16 * left  (B,N,N,H) flattened
    right_ptr,         # fp16 * right (B,N,N,H) flattened
    mask_ptr,          # fp16 * mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16 * left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16 * right_gate (B,N,N,H) flattened
    H: tl.constexpr,   # hidden dimension
    total_elements: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and gates.
    left  = left  * mask * left_gate
    right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load values
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Broadcast mask (one value per (b,i,k) -> repeated H times)
    mask_idx = offs // H                     # integer division
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply fusion and store back
    tl.store(left_ptr + offs,  l * m * lg, mask=mask)
    tl.store(right_ptr + offs, r * m * rg, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict containing "dim" and "hidden_dim" (and optionally "nomask")

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    # Switch to half precision for the rest of the pipeline
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    # (B, N, N, H)
    left = F.linear(x, left_proj_w)          # bias is None (as in the reference)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Fuse mask and gates (in‑place) via Triton when a mask is present
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Cast mask to fp16 (0.0 / 1.0)
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()               # B * N * N * H
        BLOCK = 1024                                 # threads per program
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates are no longer needed after fusion
        del left_gate, right_gate
    else:
        # No mask – simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication over the third axis
    #    left[b,i,k,h] * right[b,j,k,h] -> out[b,i,j,h]
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) to use a single matmul per (B,H) pair.
    left_b  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Matrix multiplication: (B, H, N, N) @ (B, H, N, N)^T over the last two dims
    out_b = torch.matmul(left_b, right_b.transpose(-2, -1))

    # Back to original layout (B, N, N, H)
    out = out_b.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over the hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to the original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to the original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
