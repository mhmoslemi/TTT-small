def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
                     bias=weights["norm.bias"].to(device=device, dtype=torch.float32))
    x = x.to(torch.float16)

    # linear weights to fp16
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # projections
    left = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)

    # gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # apply mask and gates
    if mask is not None and not nomask:
        mask = mask.to(device=device, dtype=torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask * left_gate
        right = right * mask * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # Permute and reshape for batched matmul
    B, N, _, H = left.shape  # B, N, N, H (dim hidden_dim)
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)
    # reshape into (B*H, N, N)
    left = left.view(-1, N, N)
    right = right.view(-1, N, N)

    # Prepare output tensor
    out = torch.empty_like(left)  # (B*H, N, N), same dtype (fp16)

    # Compute strides for kernel
    a = left
    b = right
    c = out

    # shape: batch = a.shape[0]
    batch = a.shape[0]
    M = N
    N_dim = N
    K = N

    # Compute strides
    stride_am = a.stride(1)  # stride for M dimension
    stride_ak = a.stride(2)  # stride for K dimension
    batch_stride_a = a.stride(0)

    stride_bk = b.stride(1)  # after transposing we will treat b as (batch, K, N)
    stride_bn = b.stride(2)  # N dimension after transposition (i.e., stride of N)
    batch_stride_b = b.stride(0)

    stride_cm = c.stride(1)
    stride_cn = c.stride(2)
    batch_stride_c = c.stride(0)

    # Launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

        stride_cm=stride_cm, stride_cn=stride_cn,

@triton.jit
def batched_gemm(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # Compute block indices
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch = pid // (num_blocks_m * num_blocks_n)
    pid_in_batch = pid % (num_blocks_m * num_blocks_n)
    pid_m = pid_in_batch // num_blocks_n
    pid_n = pid_in_batch % num_blocks_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Compute k range
        k_range = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k_range < K

        # Load A
        a_ptrs = a_ptr + batch * batch_stride_a + offs_m[:, None] * stride_am + k_range[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B (transposed B: shape (batch, K, N))
        b_ptrs = b_ptr + batch * batch_stride_b + k_range[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul forward (outgoing) implementation.

Algorithm
---------
1. Layer‑normalize the input (float32) and cast to fp16.
2. Linear projections to hidden dimension `hidden_dim` (left/right).
3. Compute three sigmoid gates (left, right, out) via linear layers.
4. Apply mask (if present) and gates element‑wise.
5. Perform the core contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched matrix multiplication using a custom Triton kernel.
   The tensors are reshaped to (B*H, N, N) and the kernel computes
   C = A @ Bᵀ for each batch independently.
6. Layer‑normalize over the hidden dimension, apply the output gate
   and the final linear projection back to `dim`.
7. Cast the result to the original dtype.

The kernel fuses the O(N³) reduction and runs entirely in fp16
(accumulating in fp32 for precision).  Mask‑gate fusion is done with
standard PyTorch ops, satisfying the requirement of using Triton for
at least one core operation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dims
    stride_am, stride_ak,               # strides for A (batch, M, K)
    stride_bk, stride_bn,               # strides for B (batch, K, N) – B is stored as (batch, N, N)
    stride_cm, stride_cn,               # strides for C (batch, M, N)
    batch_stride_a, batch_stride_b, batch_stride_c,  # strides for batch dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # Number of blocks in each dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # Resolve batch and block indices
    batch_id = pid // (num_blocks_m * num_blocks_n)
    pid_in_batch = pid % (num_blocks_m * num_blocks_n)
    pid_m = pid_in_batch // num_blocks_n
    pid_n = pid_in_batch % num_blocks_n

    # Offsets of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k_range < K

        # A block: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + batch_id * batch_stride_a
            + offs_m[:, None] * stride_am
            + k_range[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B block (transposed): (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_ptr
            + batch_id * batch_stride_b
            + k_range[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate matrix product
        acc += tl.dot(a, b)

    # Write result back (cast to fp16)
    c_ptrs = (
        c_ptr
        + batch_id * batch_stride_c
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32/float16)
            - mask: torch.Tensor [B, N, N] (binary) or None
            - weights: dict of model parameters (see keys below)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32) and cast to fp16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Apply mask (if any) and gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape (B,N,N) -> broadcast over hidden dim
        mask_f16 = mask.to(device=device, dtype=torch.float16).unsqueeze(-1)
        left  = left * mask_f16 * left_gate
        right = right * mask_f16 * right_gate
    else:
        left  = left * left_gate
        right = right * right_gate
    # Gates no longer needed
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core contraction via Triton batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # Rearrange to (B, H, N, N) and flatten batch*head dimension
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left = left.view(-1, N, N)   # (B*H, N, N)
    right = right.view(-1, N, N) # (B*H, N, N)

    out = torch.empty_like(left)  # (B*H, N, N), fp16

    # Strides for the kernel (all tensors are contiguous)
    stride_am = left.stride(1)          # stride for M dimension (=N)
    stride_ak = left.stride(2)          # stride for K dimension (=1)
    batch_stride_a = left.stride(0)     # stride for batch dimension (=N*N)

    stride_bk = right.stride(1)         # stride for K dimension (=N)
    stride_bn = right.stride(2)         # stride for N dimension (=1)
    batch_stride_b = right.stride(0)    # =N*N

    stride_cm = out.stride(1)           # =N
    stride_cn = out.stride(2)           # =1
    batch_stride_c = out.stride(0)      # =N*N

    M = N
    K = N

    # Block sizes (tuned for B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: one program per (batch, block_m, block_n)
    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = left.shape[0] * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    _batched_gemm_kernel[grid](
        left,
        right,
        out,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, N, N, H)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
