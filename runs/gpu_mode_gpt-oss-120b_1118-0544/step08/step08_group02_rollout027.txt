@triton.jit
def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Identify program id over total number of blocks across all batches
    pid = tl.program_id(0)
    total_blocks_m = tl.cdiv(M, BLOCK_M)
    total_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = total_blocks_m * total_blocks_n
    batch = pid // blocks_per_batch
    block = pid % blocks_per_batch
    pid_m = block // total_blocks_n
    pid_n = block % total_blocks_n

    # Compute offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Prepare pointers for A and B blocks
    # A: (M,K)
    a_ptrs = a_ptr + batch * batch_stride_a + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    # B: (K,N)
    b_ptrs = b_ptr + batch * batch_stride_b + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (tl.arange(0, BLOCK_K)[None, :] + k*BLOCK_K < K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] + k*BLOCK_K < K) & (offs_n[None, :] < N), other=0.0)
        # Multiply-accumulate
        acc += tl.dot(a, b)
        # Increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c = acc.to(tl.float16)
    # Compute pointer for C
    c_ptrs = c_ptr + batch * batch_stride_c + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

def custom_kernel(data):

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Cast to fp16 for compute
    x_fp16 = x.to(torch.float16)

    # load weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left_proj = torch.nn.functional.linear(x_fp16, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x_fp16, right_proj_w)

    # Gates
    left_gate = torch.nn.functional.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()

    # Apply mask if present and not nomask
    if mask is not None and not nomask:
        # mask shape (B,N,N) -> need broadcast to (B,N,N,1)
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask_f

        right = right_proj * right_gate

    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # For right need to transpose last two dims (k,j)
    right_T = right_perm.permute(0, 1, 3, 2).contiguous()  # shape (B,H,N,N) with last two dims (k,j)
    # flatten batch*hidden_dim
    BH = B * H
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_T.view(BH, N, N)
    # allocate output
    out_flat = torch.empty_like(left_flat)  # fp16

    stride_am = left_flat.stride(1)  # stride between rows (M dimension)
    stride_ak = left_flat.stride(2)  # stride between columns (K dimension)

    # We can compute as left_flat.stride(0).
    batch_stride_a = left_flat.stride(0)
    # Similarly for right_flat: shape (BH, K, N)
    stride_bk = right_flat.stride(1)  # stride for K dimension
    stride_bn = right_flat.stride(2)  # stride for N dimension
    batch_stride_b = right_flat.stride(0)
    # Output:
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    batch_stride_c = out_flat.stride(0)

    # grid size: total blocks across all batches
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    total_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    total_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    blocks_per_batch = total_blocks_m * total_blocks_n
    total_blocks = BH * blocks_per_batch

    grid = (total_blocks,)

    _matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape out_flat back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # Cast back to original dtype

"""
custom_kernel: Triton‑accelerated outgoing TriMul forward (AlphaFold3).

The implementation follows the reference PyTorch module but fuses the
computationally heavy O(N³) reduction with a custom Triton kernel.
Steps
1. LayerNorm over the channel dimension.
2. Linear projections + gating (FP16).
3. Optional mask application (simple element‑wise multiplication).
4. Core reduction
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   performed as a batch of GEMMs (B·hidden_dim independent N×N
   matrix multiplications) using a tiled Triton kernel.
5. Output LayerNorm, output‑gate multiplication and final linear projection.
6. Cast back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _triton_batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all = seq_len)
    stride_am, stride_ak,               # strides for A (M, K)
    stride_bk, stride_bn,               # strides for B (K, N)
    stride_cm, stride_cn,               # strides for C (M, N)
    batch_stride_a, batch_stride_b, batch_stride_c,  # strides between batches
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B  where A has shape (BH, M, K) and B has shape (BH, K, N).
    The kernel processes all batches in a single launch; each program id
    corresponds to one output tile (blockM x blockN) of one batch.
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Compute batch index and block coordinates within that batch
    # ------------------------------------------------------------------
    total_blocks_m = tl.cdiv(M, BLOCK_M)
    total_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = total_blocks_m * total_blocks_n

    batch_id = pid // blocks_per_batch
    block_id = pid % blocks_per_batch
    pid_m = block_id // total_blocks_n
    pid_n = block_id % total_blocks_n

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ------------------------------------------------------------------
    # Pointers for the current batch
    # ------------------------------------------------------------------
    a_batch = a_ptr + batch_id * batch_stride_a
    b_batch = b_ptr + batch_id * batch_stride_b
    c_batch = c_ptr + batch_id * batch_stride_c

    # ------------------------------------------------------------------
    # Compute base pointers for the tiles (will be advanced in the K-loop)
    # ------------------------------------------------------------------
    a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # ------------------------------------------------------------------
    # Accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in blocks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # masks for out‑of‑bounds loads
        cur_k = k * BLOCK_K
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] + cur_k < K)
        mask_b = (offs_k[:, None] + cur_k < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

        # advance pointers to the next K‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write back the result
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, c, mask=mask_c)


# ----------------------------------------------------------------------
# Main entrypoint
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim"        – input channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask" (optional) – if True, ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = inp.device
    orig_dtype = inp.dtype

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast linear weights to FP16 (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gating (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)        # (B, N, N, H)

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5) Apply mask + gating (simple element‑wise; mask optional)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)          # (B, N, N, 1)
        left = left_proj * left_gate * mask_fp
        right = right_proj * right_gate * mask_fp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:  # mask present but ignored
            mask_fp = mask.to(torch.float16).unsqueeze(-1)
            left = left * mask_fp
            right = right * mask_fp

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction using a Triton batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H = hidden_dim
    # Permute to (B, H, N, N) so that each (b,d) pair is a separate matrix
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()        # (B, H, N, N)

    # Transpose the last two dims of `right` to get (k, j) ordering required by GEMM
    right_t = right_perm.permute(0, 1, 3, 2).contiguous()      # (B, H, N, N) with (k, j)

    # Collapse batch and hidden dimensions
    BH = B * H
    left_flat = left_perm.view(BH, N, N)        # (BH, M, K)
    right_flat = right_t.view(BH, N, N)         # (BH, K, N)
    out_flat = torch.empty_like(left_flat)      # (BH, M, N)

    # Strides for the kernel (in element counts)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    batch_stride_a = left_flat.stride(0)

    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)
    batch_stride_b = right_flat.stride(0)

    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    batch_stride_c = out_flat.stride(0)

    # Tiling parameters – chosen to fit into shared memory and provide good occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    total_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    blocks_per_batch = total_blocks_m * total_blocks_n
    grid = (BH * blocks_per_batch,)

    _triton_batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                     # M, N, K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
