def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # Compute block start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    bh = pid_bh
    b = bh // H
    h = bh % H

    # Compute pointers base
    left_base = left_ptr + b * stride_left_b + h * stride_left_d
    right_base = right_ptr + b * stride_right_b + h * stride_right_d
    out_base = out_ptr + b * stride_out_b + h * stride_out_d

    # Offsets for i and j
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # Masks for boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Prepare offsets for i and j
    # i_offset = offs_i[:, None] * stride_i
    # j_offset = offs_j[None, :] * stride_j
    i_offset = offs_i * stride_left_i
    j_offset = offs_j * stride_out_j  # out stride for j same as H
    # For left, we need i offset in left matrix
    # We'll compute per iteration over k

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, N, BLOCK_K):
        cur_k = tl.arange(0, BLOCK_K)
        cur_k = k + cur_k
        mask_k = cur_k < N
        # Offsets for k
        k_offset_left = cur_k * stride_left_k
        k_offset_right = cur_k * stride_right_k

        # Load left tile (i,k)
        # left_ptr + i_offset[:, None] + k_offset_left[None, :]
        left_ptrs = left_base + i_offset[:, None] + k_offset_left[None, :]
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile (k,j)
        # right_ptr + k_offset_right[:, None] + (j_start + tl.arange(0, BLOCK_N))*stride_right_i
        # Wait: right stride for j is stride_right_i because j is first N dim.
        # So j offset in right = (j_start + ...) * stride_right_i
        j_offset_right = offs_j * stride_right_i
        right_ptrs = right_base + k_offset_right[:, None] + j_offset_right[None, :]
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_base + i_offset[:, None] + j_offset[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

    device = input_tensor.device

    hidden_dim = config["hidden_dim"]

    x = x.to(torch.float16)
    
    # Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # Linear projections
    left = F.linear(x, left_proj_w)   # shape (B,N,N,hidden)
    right = F.linear(x, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (float32 → fp16).
2. Linear projections (`left`, `right`) from `dim` to `hidden_dim`.
3. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`).
4. Fuse mask (if present) and the left/right gates in‑place with a tiny Triton
   kernel, avoiding an extra temporary.
5. Core reduction:
   out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   implemented as a custom batched GEMM kernel that works directly on the
   (B, N, N, H) layout (no transpose/contiguous copies).
6. Layer‑norm over the hidden dimension, apply `out_gate`, and final linear
   projection back to `dim`.
All heavy arithmetic is performed in fp16 with fp32 accumulation for
numerical stability; the final result is cast back to the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    H: tl.constexpr,
    TOTAL_ELEMS: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place fusion of mask and left/right gates.
    left  = left  * mask * left_gate
    right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < TOTAL_ELEMS

    # load data
    l = tl.load(left_ptr + offs, mask=valid)
    r = tl.load(right_ptr + offs, mask=valid)
    lg = tl.load(left_gate_ptr + offs, mask=valid)
    rg = tl.load(right_gate_ptr + offs, mask=valid)

    # broadcast mask (one mask value per hidden element)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=valid)

    # write back
    tl.store(left_ptr + offs,  l * m * lg, mask=valid)
    tl.store(right_ptr + offs, r * m * rg, mask=valid)


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # left strides
    L_B: tl.constexpr, L_I: tl.constexpr, L_K: tl.constexpr, L_D: tl.constexpr,
    # right strides
    R_B: tl.constexpr, R_I: tl.constexpr, R_K: tl.constexpr, R_D: tl.constexpr,
    # out strides
    O_B: tl.constexpr, O_I: tl.constexpr, O_J: tl.constexpr, O_D: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_i = tl.program_id(0)   # i‑tile
    pid_j = tl.program_id(1)   # j‑tile
    pid_bh = tl.program_id(2)  # combined batch‑head index

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    bh = pid_bh
    b = bh // H
    h = bh % H

    # base pointers for the current (b, h)
    left_base  = left_ptr  + b * L_B + h * L_D
    right_base = right_ptr + b * R_B + h * R_D
    out_base   = out_ptr   + b * O_B + h * O_D

    # i / j indices within the tile
    i_idx = i0 + tl.arange(0, BLOCK_M)
    j_idx = j0 + tl.arange(0, BLOCK_N)

    mask_i = i_idx < N
    mask_j = j_idx < N

    # strides for the i and j dimensions
    i_offset = i_idx * L_I          # same for left and out (stride_i)
    j_offset = j_idx * O_J          # stride for j in the output
    j_offset_right = j_idx * R_I    # stride for j in the right operand

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension in blocks
    for k0 in range(0, N, BLOCK_K):
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # offsets for the k dimension
        k_offset_left  = k_idx * L_K
        k_offset_right = k_idx * R_K

        # load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i_offset[:, None] + k_offset_left[None, :]
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # load right tile: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + k_offset_right[:, None] + j_offset_right[None, :]
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # ACcumulate
        acc += tl.dot(left, right)

    # store the result
    out_ptrs = out_base + i_offset[:, None] + j_offset[None, :]
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see code for expected keys)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
    )
    x = x.to(torch.float16)   # fp16 for the rest of the pipeline

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)      # (B, N, N, H)
    right = F.linear(x, right_proj_w)     # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # Ensure contiguous layout for the kernels
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()

    # ------------------------------------------------------------------
    # 5) Fuse mask and gates (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)
        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates are no longer needed
        del left_gate, right_gate
    else:
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction (custom Triton GEMM)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # (B, N, N, H)

    # Strides (in elements) for left/right/out
    sL = left.stride()   # (B, N, N, H)
    sR = right.stride()
    sO = out.stride()

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        # left strides
        sL[0], sL[1], sL[2], sL[3],
        # right strides
        sR[0], sR[1], sR[2], sR[3],
        # out strides
        sO[0], sO[1], sO[2], sO[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast back to original dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
