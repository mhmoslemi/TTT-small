def _fuse_mask_gate(
    left_ptr, right_ptr, mask_ptr,
    left_gate_ptr, right_gate_ptr,
    H: tl.constexpr, total: tl.constexpr, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index
    idx = offs // H  # integer division
    m = tl.load(mask_ptr + idx, mask=mask, other=1.0) # mask default 1 if out-of-bounds? But mask size is B*N*N
    # Might need to cast mask to same dtype.
    l = l * lg * m
    r = r * rg * m

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input (FP32 for numerical stability) and cast to FP16.
2. Linear projections to the hidden dimension (`left`, `right`) using FP16 GEMM.
3. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`).
4. Fuse mask, left‑gate and right‑gate element‑wise multiplications in a
   single Triton kernel (in‑place, no temporaries).  If no mask is provided
   the fusion is done with a simple pointwise multiplication.
5. Core reduction: for each batch and hidden channel compute a matrix
   multiplication
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched `torch.bmm`.
6. Layer‑normalize over the hidden dimension.
7. Apply the output gate.
8. Final linear projection back to the original channel dimension.
9. Cast the result back to float32.

Only the mask‑gate fusion is implemented in Triton; the remaining
operations use highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_inplace(
    left_ptr,          # fp16* left  (B,N,N,H) flattened
    right_ptr,         # fp16* right (B,N,N,H) flattened
    mask_ptr,          # fp16* mask  (B,N,N)   flattened
    left_gate_ptr,     # fp16* left_gate  (B,N,N,H) flattened
    right_gate_ptr,    # fp16* right_gate (B,N,N,H) flattened
    H,                 # hidden dimension (runtime int)
    total_elements,    # total number of elements in left/right (runtime int)
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place fusion of mask and the two gate tensors:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load tensors
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # Broadcast mask across the hidden dimension
    mask_idx = offs // H                      # (b,i,j) index
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply fused multiplication
    tl.store(left_ptr + offs,  l * lg * m, mask=mask)
    tl.store(right_ptr + offs, r * rg * m, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (see keys below)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float32)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float32)

    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # Cast to fp16 for compute‑heavy stages
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension H
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Fuse mask and gate multiplications (in‑place) via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device=device, dtype=torch.float16)
        total_elements = left.numel()                     # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total_elements, BLOCK),)

        _mask_gate_fusion_inplace[grid](
            left,
            right,
            mask_fp16.view(-1),
            left_gate,
            right_gate,
            hidden_dim,
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # Gates no longer needed
        del left_gate, right_gate
    else:
        # No mask: simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6) Core reduction: batched matrix multiplication
    #    left and right are (B, N, N, H) → (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape for batched matmul
    left_b = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_b = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)
    out_b = torch.bmm(left_b, right_b)                     # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_b.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 10) Cast to float32 (required output dtype)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
