"""
TriMul outgoing (AlphaFold3) forward pass – Triton‑accelerated mask‑gate fusion.

Algorithm
---------
1. Layer‑norm over the input channel dimension.
2. Cast to FP16 for compute‑intensive ops.
3. Combine linear projections:
   - left_proj + left_gate   (2 × hidden_dim)
   - right_proj + right_gate (2 × hidden_dim)
   - out_gate                (hidden_dim)
   These are computed with a single torch.nn.functional.linear each.
4. Split the combined results, apply sigmoid to the gates,
   and fuse mask & gate multiplication with a tiny Triton kernel.
5. Core TriMul reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix‑multiply (torch.bmm) on
   tensors of shape (B × hidden_dim, N, N).
6. Output layer‑norm, multiply by out_gate, final linear projection back to dim.
7. Cast back to the original input dtype.

The Triton kernel only performs the element‑wise mask‑gate fusion,
which is the part that cannot be efficiently fused by torch.compile().
All heavy linear algebra (projections, reduction) uses cuBLAS via PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _mask_gate_fuse_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask, projection and gate for left/right tensors."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # each hidden entry repeats `hidden_dim` times in the flattened tensor
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    left_out = lp * lg * m
    right_out = rp * rg * m

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Combine linear projections to reduce the number of matmuls
    # ------------------------------------------------------------------
    # combine left_proj + left_gate
    W_left_comb = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)

    # combine right_proj + right_gate
    W_right_comb = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)

    # out_gate weight (H, dim)
    W_out_gate = weights["out_gate.weight"].to(torch.float16)

    # flatten input for linear layers
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)  # (B*N*N, dim)

    # left combined projection
    left_comb = F.linear(x_flat, W_left_comb)  # (B*N*N, 2*H)
    # right combined projection
    right_comb = F.linear(x_flat, W_right_comb)  # (B*N*N, 2*H)
    # out gate
    out_gate = F.linear(x_flat, W_out_gate)  # (B*N*N, H)

    # split combined results
    left_proj, left_gate = torch.split(left_comb, hidden_dim, dim=-1)
    right_proj, right_gate = torch.split(right_comb, hidden_dim, dim=-1)

    # reshape back to 4‑D tensors
    left_proj = left_proj.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_proj = right_proj.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 4) Fuse mask & gate multiplication with Triton (if mask is used)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is fp16, contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        total = left_proj.numel()
        BLOCK_SIZE = 16384  # tuneable
        grid = (triton.cdiv(total, BLOCK_SIZE),)

        _mask_gate_fuse_kernel[grid](
            left_proj, left_gate,
            right_proj, right_gate,
            mask_fp16,
            left, right,
            total,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        # No mask – simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched matrix multiplication (bmm).
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 7) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
