def tri_mul_kernel(left_ptr, right_ptr, out_ptr, batch, M, N, K, stride_lbatch, stride_lm, stride_lk, stride_rbatch, stride_rn, stride_rk, stride_obatch, stride_om, stride_on, BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

    batch_idx = tl.program_id(0)

    pid_m = tl.program_id(1)  # tile row index
    pid_n = tl.program_id(2)  # tile col index

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # k loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # load left tile
        l_ptrs = left_ptr + batch_idx*stride_lbatch + offs_m[:,None] * stride_lm + offs_k[None,:] * stride_lk

        # accumulate
        acc += tl.dot(left, right)  # shape (BLOCK_M, BLOCK_N)

"""
TriMul (outgoing) forward pass – Triton‑accelerated batched GEMM.

The heavy part of the TriMul operator is:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is a batched matrix multiplication of shape (B*H, N, N) @ (B*H, N, N)^T.
We implement this reduction with a custom Triton kernel that tiles the
matmul (BLOCK_M×BLOCK_N tiles, BLOCK_K reduction).  The rest of the
operations (LayerNorm, linear projections, gating, mask handling) are
performed in PyTorch (FP16) and the final output is cast back to the
input dtype (usually float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_bmm_kernel(
    left_ptr,          # fp16 * (batch, M, K)
    right_ptr,         # fp16 * (batch, N, K)
    out_ptr,           # fp16 * (batch, M, N)
    batch: tl.constexpr,
    M: tl.constexpr,
    N: tl.constexpr,
    K: tl.constexpr,
    stride_lbatch, stride_lrow, stride_lcol,
    stride_rbatch, stride_rrow, stride_rcol,
    stride_obatch, stride_orow, stride_ocol,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: out = left @ right.T   (left, right: (batch,M,K), (batch,N,K))."""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        l_ptrs = (
            left_ptr
            + pid_batch * stride_lbatch
            + offs_m[:, None] * stride_lrow
            + offs_k[None, :] * stride_lcol
        )
        l = tl.load(l_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile (no transposition needed because we multiply with right[j,k])
        r_ptrs = (
            right_ptr
            + pid_batch * stride_rbatch
            + offs_n[:, None] * stride_rrow
            + offs_k[None, :] * stride_rcol
        )
        r = tl.load(r_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Accumulate dot product
        acc += tl.dot(l, r)

    # Write back result in fp16
    out_ptrs = (
        out_ptr
        + pid_batch * stride_obatch
        + offs_m[:, None] * stride_orow
        + offs_n[None, :] * stride_ocol
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] (0/1) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Cast all linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections and gating (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Apply mask & gating (element‑wise fusion)
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6) Core reduction (TriMul) – batched GEMM in Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # left shape (B, N, N, H)

    # Rearrange to (batch, M, K) where batch = B*H, M = N, K = N
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Allocate output tensor
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements) for contiguous tensors
    stride_lbatch, stride_lrow, stride_lcol = left_flat.stride()
    stride_rbatch, stride_rrow, stride_rcol = right_flat.stride()
    stride_obatch, stride_orow, stride_ocol = out_flat.stride()

    # Tiling parameters – chosen to fit within shared memory and give good occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_tiles_m = triton.cdiv(N, BLOCK_M)
    num_tiles_n = triton.cdiv(N, BLOCK_N)
    grid = (B * H, num_tiles_m, num_tiles_n)

    _tri_mul_bmm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * H,
        N,
        N,
        N,
        stride_lbatch,
        stride_lrow,
        stride_lcol,
        stride_rbatch,
        stride_rrow,
        stride_rcol,
        stride_obatch,
        stride_orow,
        stride_ocol,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 8) Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
