def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Input may be torch.float32 or float16.

    # Extract config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Input layernorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to float16 for heavy ops
    x_fp16 = x.to(torch.float16)

    # Cast weights to float16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Projections and gates
    left_proj = F.linear(x_fp16, left_proj_w)  # [B,N,N,H]
    right_proj = F.linear(x_fp16, right_proj_w)
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))

    # Apply mask if present
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    B, N, _, H = left.shape

    # Core matmul
    # Permute to [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Flatten batch*H
    left_flat = left_perm.view(-1, N, N)  # (B*H, N, N)
    right_flat = right_perm.view(-1, N, N)

    # Compute batched matmul: (B*H, N, N) @ (B*H, N, N)^T
    # Use torch.bmm
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Hidden dim layernorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Output gate
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    # Multiply out * out_gate elementwise using Triton kernel
    out = _elemwise_mul(out, out_gate)

    # Final projection back to dim
    out = F.linear(out, to_out_w)

    # Cast back to original dtype of input_tensor (probably float32)
    out = out.to(input_tensor.dtype)

    return out

"""
custom_kernel: Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the reference PyTorch TriMul but replaces the
O(N³·H) reduction with a highly‑optimized batched GEMM (torch.bmm) and
uses a tiny Triton kernel for the element‑wise multiplication with the
output gate.  All heavy arithmetic is performed in float16 while the
input LayerNorm uses float32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton element‑wise multiplication (used for out * out_gate)
# --------------------------------------------------------------------------- #
@triton.jit
def _elemwise_mul_kernel(
    a_ptr, b_ptr, out_ptr, n_elements, BLOCK: tl.constexpr
):
    """Compute out = a * b for 1‑D contiguous tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton kernel on two tensors of identical shape."""
    assert a.shape == b.shape, "Shape mismatch in element‑wise mul."
    a = a.contiguous()
    b = b.contiguous()
    out = torch.empty_like(a)
    numel = a.numel()
    BLOCK = 1024
    grid = ((numel + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, numel, BLOCK=BLOCK)
    return out


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] or None (binary)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 2. Cast to half‑precision for the heavy path
    # ------------------------------------------------------------------- #
    x_fp16 = x.to(torch.float16)

    # Cast model weights to half precision
    device = x_fp16.device
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------- #
    # 3. Linear projections + gates
    # ------------------------------------------------------------------- #
    left_proj = F.linear(x_fp16, left_proj_w)               # [B, N, N, H]
    right_proj = F.linear(x_fp16, right_proj_w)             # [B, N, N, H]
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w)) # [B, N, N, H]

    # Apply mask before gating if required
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    left = left_proj * left_gate      # [B, N, N, H]
    right = right_proj * right_gate   # [B, N, N, H]

    B, N, _, H = left.shape
    assert H == hidden_dim, "Hidden dimension mismatch."

    # ------------------------------------------------------------------- #
    # 4. Core O(N³·H) reduction via batched GEMM (torch.bmm)
    #    out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]
    # ------------------------------------------------------------------- #
    # Rearrange to (B*H, N, N) matrices
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_perm.view(-1, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(-1, N, N) # (B*H, N, N)

    # Batched matrix multiplication with transposed right operand
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 5. Hidden‑dim LayerNorm (still in float16)
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------- #
    # 6. Output gate (element‑wise) – Triton kernel for multiplication
    # ------------------------------------------------------------------- #
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))  # [B, N, N, H]
    out = _elemwise_mul(out, out_gate)                     # still float16

    # ------------------------------------------------------------------- #
    # 7. Final projection back to original channel dimension
    # ------------------------------------------------------------------- #
    out = F.linear(out, to_out_w)  # [B, N, N, C]

    # Cast to the original dtype of the input tensor
    out = out.to(x.dtype)

    return out
