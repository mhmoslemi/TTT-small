"""
TriMul (outgoing) forward pass – Triton‑fused reduction.

Algorithm
---------
1. Layer‑norm on the input tensor.
2. FP16 linear projections for left/right and their gates.
3. Apply mask (if present) to the projections *before* gating.
4. Multiply each projection with its sigmoid‑gate.
5. Reshape the hidden dimension into the batch dimension and run a
   batched GEMM kernel (A @ Bᵀ) that implements the O(N³·H) reduction.
6. Reshape back to [B, N, N, H] and apply a second LayerNorm.
7. Multiply by the output gate and a final linear projection back to the
   original channel dimension.
8. Return the result in the original dtype (float32).

The heavy O(N³·H) step is performed by a custom Triton kernel that
processes each (batch, hidden) slice as an independent GEMM:
    C = A @ Bᵀ   with   A, B ∈ ℝ^{N×N}.
All other operations use PyTorch’s highly‑optimised kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def batched_gemm_forward(
    A_ptr, B_ptr, C_ptr,                # pointers
    batch, M, N, K,                     # dimensions
    stride_Ab, stride_Am, stride_Ak,    # strides for A (batch, M, K)
    stride_Bb, stride_Bn, stride_Bk,    # strides for B (batch, N, K)
    stride_Cb, stride_Cm, stride_Cn,    # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ  (A shape: batch×M×K, B shape: batch×N×K)."""
    batch_id = tl.program_id(2)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(0)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch
    A_batch = A_ptr + batch_id * stride_Ab
    B_batch = B_ptr + batch_id * stride_Bb
    C_batch = C_ptr + batch_id * stride_Cb

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M×K)
        a_ptrs = A_batch + (offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B tile transposed (K×N) – B is stored as (N,K), we read as (K,N)
        b_ptrs = B_batch + (offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Write result back in fp16
    c_ptrs = C_batch + (offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for the compute‑intensive path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask (if present) before gating, then gate
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    left = left_proj * left_gate   # shape [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·H) reduction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) then flatten batch and hidden dims
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left.shape
    batch_flat = B * H

    left_flat = left.view(batch_flat, N, N)   # [batch_flat, N, N]
    right_flat = right.view(batch_flat, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_N),   # program_id(0) – columns
        triton.cdiv(N, BLOCK_M),   # program_id(1) – rows
        batch_flat,                # program_id(2) – batch
    )

    batched_gemm_forward[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_flat, N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 7. Multiply by output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 8. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
