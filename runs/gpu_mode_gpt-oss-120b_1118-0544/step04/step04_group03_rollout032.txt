def elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = x.device
    orig_dtype = x.dtype

    # LayerNorm
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for compute
    x = x.to(torch.float16)

    # Linear projections
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply mask if needed
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Core reduction: batch gemm
    B, N, _, H = left.shape  # shape [B,N,N,H]
    # permute and reshape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_perm.view(B * H, N, N)  # (batch, M, K)
    right_flat = right_perm.view(B * H, N, N)  # (batch, K, N) -> need transpose
    # compute matmul left * right^T
    out_flat = torch.matmul(left_flat, right_flat.transpose(-2, -1))  # (batch, N, N)

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # to_out_norm layernorm
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    assert a.shape == b.shape, "Shape mismatch for elementwise multiplication."
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device
    orig_dtype = x.dtype

    # --------------------------------------------------------------------- #
    # 1. Input LayerNorm (float32 -> keep precision for norm)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------------- #
    # 2. Cast to fp16 for the heavy computation
    # --------------------------------------------------------------------- #
    x = x.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 3. Cast all linear weights to fp16 on the target device
    # --------------------------------------------------------------------- #
    def _to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    left_proj_w = _to_fp16(weights["left_proj.weight"])
    right_proj_w = _to_fp16(weights["right_proj.weight"])
    left_gate_w = _to_fp16(weights["left_gate.weight"])
    right_gate_w = _to_fp16(weights["right_gate.weight"])
    out_gate_w = _to_fp16(weights["out_gate.weight"])
    to_out_norm_w = _to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = _to_fp16(weights["to_out_norm.bias"])
    to_out_w = _to_fp16(weights["to_out.weight"])

    # --------------------------------------------------------------------- #
    # 4. Linear projections and gates
    # --------------------------------------------------------------------- #
    left_proj = F.linear(x, left_proj_w)            # [B, N, N, H]
    right_proj = F.linear(x, right_proj_w)          # [B, N, N, H]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # [B, N, N, H]

    # --------------------------------------------------------------------- #
    # 5. Optional mask handling (mask broadcast over H)
    # --------------------------------------------------------------------- #
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # --------------------------------------------------------------------- #
    # 6. Apply gates
    # --------------------------------------------------------------------- #
    left = left_proj * left_gate    # [B, N, N, H]
    right = right_proj * right_gate

    # --------------------------------------------------------------------- #
    # 7. Core O(N³·H) reduction: batched matrix multiplication
    #    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as (B*H) independent GEMMs: (N×N) @ (N×N)^T
    # --------------------------------------------------------------------- #
    B, N, _, H = left.shape

    # Move hidden dimension to batch axis for efficient GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_perm.view(B * H, N, N)            # (B*H, M, K)
    right_flat = right_perm.view(B * H, N, N)          # (B*H, K, N)

    # Batched GEMM: A @ Bᵀ
    out_flat = torch.matmul(left_flat, right_flat.transpose(-2, -1))  # (B*H, N, N)

    # Restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------------- #
    # 8. Hidden‑dim LayerNorm
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------------- #
    # 9. Multiply by output gate (Triton element‑wise kernel)
    # --------------------------------------------------------------------- #
    # cast gating tensor to same dtype (fp16)
    out_gate = out_gate.to(torch.float16)
    out = _elemwise_mul(out, out_gate)

    # --------------------------------------------------------------------- #
    # 10. Final linear projection back to original channel dim
    # --------------------------------------------------------------------- #
    out = F.linear(out, to_out_w)   # [B, N, N, dim]

    # Cast back to the dtype of the original input
    out = out.to(orig_dtype)

    return out

Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the reference PyTorch module but:
1. Performs the cheap element‑wise out‑gate multiplication with a custom
   Triton kernel.
2. Executes the O(N³·H) reduction as a batched half‑precision GEMM using
   PyTorch’s highly‑optimised cuBLAS (half‑precision tensor cores).
   The tensors are reshaped so that each (batch, hidden_dim) pair becomes an
   independent matrix multiplication:   out[b,i,j,d] = Σ_k left[b,i,k,d] *
   right[b,j,k,d]  →  left @ rightᵀ.
All heavy linear projections and layer‑norms are kept in fp16 for speed,
with the final result cast back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton element‑wise kernel (out‑gate multiplication)
# --------------------------------------------------------------------------- #
@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton kernel for element‑wise multiplication."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication."
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out

# --------------------------------------------------------------------------- #
