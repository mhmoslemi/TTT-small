@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab, # maybe offset for batch
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # coordinate offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # load a and b blocks iteratively over K
    a_ptrs = a_ptr + pid_batch * batch_stride_a + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_batch * batch_stride_b + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (tl.arange(0, BLOCK_K) < K - k*BLOCK_K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) < K - k*BLOCK_K)[:, None] & (offs_n < N)[None, :], other=0.0)
        # compute dot
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write out
    c_ptrs = c_ptr + pid_batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,  # maybe batch stride not needed? We'll just compute base pointers with batch offset.
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Base pointers for this batch
    a_batch_ptr = a_ptr + pid_batch * batch_stride_a
    b_batch_ptr = b_ptr + pid_batch * batch_stride_b
    c_batch_ptr = c_ptr + pid_batch * batch_stride_c

    # Compute offsets for rows and columns
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create block pointers for a and b
    a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_batch_ptr + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # mask for out of bounds
        k_off = k * BLOCK_K
        cur_k = tl.arange(0, BLOCK_K) + k_off
        # load a and b
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (cur_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(cur_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        # dot product
        acc += tl.dot(a, b)
        # move pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write to C
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,   # a row stride, a col stride
    stride_bk, stride_bn,   # b row stride (k?), b col stride (n)
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,   # batch strides: a_batch, b_batch, c_batch
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for rows and cols
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute base pointers for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_ab
    b_batch_ptr = b_ptr + pid_batch * stride_bb
    c_batch_ptr = c_ptr + pid_batch * stride_cb

    # Split rows into two halves for better occupancy (as in example)
    offs_m_1 = offs_m[:BLOCK_M // 2]
    offs_m_2 = offs_m[BLOCK_M // 2:]

    # Offsets for k
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers for A halves and B
    a_ptrs_1 = a_batch_ptr + (offs_m_1[:, None] * stride_am + offs_k[None, :] * stride_ak)
    a_ptrs_2 = a_batch_ptr + (offs_m_2[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # Accumulators
    acc_1 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)
    acc_2 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)

    # Loop over K

        a_1 = tl.load(a_ptrs_1)   # shape (BLOCK_M//2, BLOCK_K)
        a_2 = tl.load(a_ptrs_2)

        a_ptrs_2 += BLOCK_K * stride_ak

"""
TriMul (outgoing) forward pass – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to FP16 for the compute‑heavy path.
3. Linear projections for left/right (dim → hidden_dim) and their gates
   (dim → hidden_dim) plus the output gate (dim → hidden_dim).
4. Fuse mask, projection and gate with a tiny Triton kernel (if a mask is
   present).  The tensors stay in FP16.
5. Core O(N³) reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (left @ rightᵀ) with a custom Triton
   matmul kernel that uses Tensor‑cores (tl.dot) and processes each hidden
   head independently.
6. Output LayerNorm over the hidden dimension, multiply by out_gate and
   a final linear projection back to the original dimension.
7. Cast back to the input dtype.

Only the mask‑gate fusion and the batched GEMM are written in Triton; the
remaining operations use highly‑optimized PyTorch primitives.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel 1: fuse projection * gate * mask (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # total number of elements in proj/gate (=B*N*N*H)
    hidden_dim,        # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # mask index = element index // H  (broadcast over hidden dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Fuse projection, gate and mask using Triton.
    All tensors must be fp16, contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)   (binary 0/1)
    Returns proj * gate * mask  (broadcast over H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: batched matmul C = A @ B   (A: (batch, M, K), B: (batch, K, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,      # A: row stride, col stride
    stride_bk, stride_bn,      # B: row (K) stride, col (N) stride
    stride_cm, stride_cn,      # C: row stride, col stride
    stride_ab, stride_bb, stride_cb,   # batch strides for A, B, C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # batch * hidden_head index
    pid_m = tl.program_id(1)              # tile over M
    pid_n = tl.program_id(2)              # tile over N

    # Offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the start of the current batch
    a_batch = a_ptr + pid_batch * stride_ab
    b_batch = b_ptr + pid_batch * stride_bb
    c_batch = c_ptr + pid_batch * stride_cb

    # Split M dimension in two halves (as in Triton's tutorial)
    offs_m_1 = offs_m[:BLOCK_M // 2]
    offs_m_2 = offs_m[BLOCK_M // 2:]

    # K dimension offsets (used in every iteration)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to sub‑blocks of A and B
    a_ptrs_1 = a_batch + (offs_m_1[:, None] * stride_am + offs_k[None, :] * stride_ak)
    a_ptrs_2 = a_batch + (offs_m_2[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs   = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # Accumulators in FP32
    acc_1 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)
    acc_2 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)

    # Main loop over K
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a_1 = tl.load(a_ptrs_1)          # (BLOCK_M/2, BLOCK_K)
        a_2 = tl.load(a_ptrs_2)          # (BLOCK_M/2, BLOCK_K)
        b   = tl.load(b_ptrs)            # (BLOCK_K, BLOCK_N)

        acc_1 += tl.dot(a_1, b)
        acc_2 += tl.dot(a_2, b)

        # Advance pointers to the next K‑tile
        a_ptrs_1 += BLOCK_K * stride_ak
        a_ptrs_2 += BLOCK_K * stride_ak
        b_ptrs   += BLOCK_K * stride_bk

    # Write results back to C
    c_ptrs_1 = c_batch + (offs_m_1[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    c_ptrs_2 = c_batch + (offs_m_2[:, None] * stride_cm + offs_n[None, :] * stride_cn)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_1 = (mask_m[:BLOCK_M // 2][:, None] & mask_n[None, :])
    mask_2 = (mask_m[BLOCK_M // 2:][:, None] & mask_n[None, :])

    tl.store(c_ptrs_1, acc_1.to(tl.float16), mask=mask_1)
    tl.store(c_ptrs_2, acc_2.to(tl.float16), mask=mask_2)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul (outgoing) forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or fp16)
            - mask         : torch.Tensor of shape [B, N, N] (binary) or None
            - weights      : dict of model parameters (float32)
            - config       : dict containing:
                * "dim" (int)            – input channel dimension C
                * "hidden_dim" (int)     – hidden dimension H
                * "nomask" (bool, opt)   – if True, ignore mask
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 2. Cast to FP16 for the compute‑heavy part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Load weights (FP16) – all Linear layers are bias‑free
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask & gate (fused via Triton if mask is present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM: left @ rightᵀ
    # --------------------------------------------------------------
    B, N, _, H = left.shape

    # Rearrange tensors so that each head becomes an independent batch
    # left: (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # rightᵀ: (B, N, N, H) -> (B, H, N, N) with axes 1 and 2 swapped
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # Merge batch and head dimensions
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Allocate output tensor (FP16)
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements)
    stride_am = left_flat.stride(1)   # row stride (M)
    stride_ak = left_flat.stride(2)   # col stride (K)
    stride_bk = right_flat.stride(1)  # row stride (K)
    stride_bn = right_flat.stride(2)  # col stride (N)
    stride_cm = out_flat.stride(1)    # row stride (M)
    stride_cn = out_flat.stride(2)    # col stride (N)

    stride_ab = left_flat.stride(0)   # batch stride for A
    stride_bb = right_flat.stride(0)  # batch stride for B
    stride_cb = out_flat.stride(0)    # batch stride for C

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ab, stride_bb, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 9. Cast back to the original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
