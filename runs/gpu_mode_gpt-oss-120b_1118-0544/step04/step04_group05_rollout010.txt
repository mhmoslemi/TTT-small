"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C).
2. Project to hidden dimension H with two linear layers (left/right).
3. (Optional) multiply by binary mask broadcast over H.
4. Compute three sigmoid gates (left, right, out) from the normalized input.
5. Apply left/right gates to their projections.
6. Fold the hidden dimension into the batch dimension and compute the core
   reduction   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   This is a batched GEMM C = A @ Bᵀ with A and B of shape (B·H, N, N).
   For N ≤ 512 a hand‑written Triton kernel is used; for larger N we fall
   back to torch.bmm (cuBLAS).  The Triton kernel tiles the matrices and
   accumulates in fp32 for numerical stability.
7. Reshape back to (B,N,N,H), apply a second LayerNorm over H, multiply by
   the output gate and finally project back to the original channel dimension.
8. Cast the result back to the original dtype.

The kernel is written for Triton 3.3.1 and targets NVIDIA Hopper (B200)
devices.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (A,B shape: (batch, N, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix dimensions (all equal to seq_len)
    stride_ah, stride_am, stride_ak,    # A: batch, row, col strides
    stride_bh, stride_bk, stride_bn,    # B: batch, k, n strides (Bᵀ layout via swapped strides)
    stride_ch, stride_cm, stride_cn,    # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch (B·H)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Accumulator in fp32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main loop over K dimension (tiled)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        k_offset = k * BLOCK_K
        k_idx = k_offset + tl.arange(0, BLOCK_K)               # (BLOCK_K,)
        mask_k = k_idx < K

        # Load A tile
        a_ptr = A_ptr + pid_b * stride_ah \
                + (offs_m[:, None] * stride_am + k_idx[None, :] * stride_ak)
        a = tl.load(a_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (as transposed via swapped strides)
        b_ptr = B_ptr + pid_b * stride_bh \
                + (k_idx[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write out the result (convert to fp16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptr = C_ptr + pid_b * stride_ch \
            + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptr, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for tensors of shape (batch, N, N) using a Triton kernel.
    All tensors must be half‑precision and reside on CUDA.
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be on CUDA."
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "Tensors must be float16."
    assert A.shape == B.shape, "A and B must have the same shape."
    batch, N, K = A.shape
    assert N == K, "TriMul requires square sequence dimension."

    # Choose block sizes heuristically based on N
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 8
    else:  # N up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows
        (N + BLOCK_N - 1) // BLOCK_N,   # cols
        batch,                          # batch dimension (B·H)
    )

    C = torch.empty_like(A)

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),          # A batch, row, col
        B.stride(0), B.stride(2), B.stride(1),          # B as Bᵀ: batch, k, n (swap last two strides)
        C.stride(0), C.stride(1), C.stride(2),          # C batch, row, col
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )
    return C

# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask          : torch.Tensor of shape [B, N, N] or None (binary)
            - weights       : dict of model parameters (float32 tensors)
            - config        : dict with keys
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – optional bool, ignore mask when True

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ----------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------
    # 1) Input LayerNorm
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Work in FP16 for the heavy portion
    x = x.to(torch.float16)

    # ----------------------------------------------------------
    # 2) Cast linear weights to FP16 (LayerNorm weights already used)
    # ----------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ----------------------------------------------------------
    # 3) Linear projections (B,N,N,H)
    # ----------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B,N,N,H)
    right = F.linear(x, right_proj_w)   # (B,N,N,H)

    # ----------------------------------------------------------
    # 4) Optional mask (broadcast over hidden dim)
    # ----------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------
    # 5) Gates (sigmoid of linear projections)
    # ----------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    # ----------------------------------------------------------
    # 6) Apply gates to projections
    # ----------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 7) Fold hidden dimension into batch for GEMM
    # ----------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)
    left = left.view(B * H, N, N)                  # (B*H, N, N)
    right = right.view(B * H, N, N)                # (B*H, N, N)

    # ----------------------------------------------------------
    # 8) Core O(N³·H) reduction
    # ----------------------------------------------------------
    # Use Triton for up to moderate N, fallback to cuBLAS for very large seq lengths.
    if N <= 512:
        out_flat = _batched_matmul_triton(left, right)         # (B*H, N, N)
    else:
        out_flat = torch.bmm(left, right.transpose(-2, -1))   # (B*H, N, N)

    # ----------------------------------------------------------
    # 9) Reshape back to (B, N, N, H)
    # ----------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ----------------------------------------------------------
    # 10) Output LayerNorm over hidden dimension
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------------------------------------------
    # 11) Apply output gate
    # ----------------------------------------------------------
    out = out * out_gate

    # ----------------------------------------------------------
    # 12) Final linear projection back to original channel dimension
    # ----------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ----------------------------------------------------------
    # 13) Cast back to original dtype if needed
    # ----------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
