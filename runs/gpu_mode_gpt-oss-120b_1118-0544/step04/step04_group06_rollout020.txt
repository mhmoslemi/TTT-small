import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

def _apply_layer_norm(x, weight, bias):
    return torch.nn.functional.layer_norm(
        x,
        (weight.shape[0],),
        weight=weight,
        bias=bias,
    )

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)   # encodes batch and hidden dimension
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    i_off = i_base + tl.arange(0, BLOCK_I)
    j_off = j_base + tl.arange(0, BLOCK_J)

    i_mask = i_off < N
    j_mask = j_off < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, N, BLOCK_K):
        k_off = k + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # left tile: (i, k)
        left_ptrs = left_ptr + b * stride_lb + i_off[:, None] * stride_li + k_off[None, :] * stride_lk + h * stride_lh
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # right tile: (k, j)
        right_ptrs = right_ptr + b * stride_rb + k_off[:, None] * stride_rk + j_off[None, :] * stride_rj + h * stride_rh
        right = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0).to(tl.float32)

        # matrix multiplication over k dimension
        acc += tl.dot(left, right)

    # write output
    out_ptrs = out_ptr + b * stride_ob + i_off[:, None] * stride_oi + j_off[None, :] * stride_oj + h * stride_oh
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])

def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor,
                    block_i: int = 64, block_j: int = 64, block_k: int = 32) -> torch.Tensor:
    """Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h] using Triton."""
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # get strides (in elements)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    grid = (B * H, triton.cdiv(N, block_i), triton.cdiv(N, block_j))

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_I=block_i, BLOCK_J=block_j, BLOCK_K=block_k,
    )
    return out

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Implements the algorithm from AlphaFold3:
      1. Layer‑norm the input.
      2. Linear projections to a hidden dimension.
      3. Sigmoid gates (left, right, out) and optional mask fusion.
      4. Core reduction  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
         performed by a tiled Triton kernel.
      5. Output LayerNorm, output‑gate multiplication and final linear projection.
    Returns a tensor of shape [B, N, N, dim] with the same dtype as the input.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (fp16 compute)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float16)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    ).to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Gate projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5. Apply mask (if any) and gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape: (B,N,N) -> broadcast to (B,N,N,1)
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 6. Core TriMul reduction via Triton kernel
    # ------------------------------------------------------------------
    out = _tri_mul_triton(left, right,
                          block_i=64, block_j=64, block_k=32)  # tuned block sizes

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)

    # ------------------------------------------------------------------
    # 8. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out

TriMul (outgoing) forward pass – Triton accelerated core reduction.

Algorithm
---------

2. Cast to fp16 for the compute‑intensive part.
3. Linear projections to the hidden dimension and sigmoid gates.
4. Optionally fuse a mask with the gates (pure PyTorch – cheap relative to O(N³)).
5. Compute the main reduction

    i_mask = i_off < N
    j_mask = j_off < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # reduction over k
    for k in range(0, N, BLOCK_K):
        k_off = k + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # left[i, k, h]
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + i_off[:, None] * stride_li
            + k_off[None, :] * stride_lk
            + h * stride_lh
        )
        left = tl.load(
            left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0
        ).to(tl.float32)

        # right[k, j, h]  (note the swapped dimensions order for dot)
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + k_off[:, None] * stride_rk
            + j_off[None, :] * stride_rj
            + h * stride_rh
        )
        right = tl.load(
            right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0
        ).to(tl.float32)

        # accumulate dot product over the k‑tile
        acc += tl.dot(left, right)

    # store result
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + i_off[:, None] * stride_oi
        + j_off[None, :] * stride_oj
        + h * stride_oh
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])


def _tri_mul_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    block_i: int = 64,
    block_j: int = 64,
    block_k: int = 32,
) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel.
    left/right: tensors of shape [B, N, N, H] (fp16, contiguous)
    Returns a tensor of the same shape containing the reduction.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides expressed in element counts
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    grid = (B * H, triton.cdiv(N, block_i), triton.cdiv(N, block_j))

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_rj,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_I=block_i,
        BLOCK_J=block_j,
        BLOCK_K=block_k,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Triton‑accelerated outgoing TriMul forward pass.
    Input tuple: (input_tensor, mask, weights, config)
    Returns: tensor of shape [B, N, N, dim] with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (fp16 compute)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_b = weights["norm.bias"].to(device=device, dtype=torch.float16)
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_w, bias=norm_b
    ).to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Gate projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5. Apply mask (if present) and gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)          # (B,N,N,1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 6. Core TriMul reduction via Triton
    # ------------------------------------------------------------------
    out = _tri_mul_triton(left, right, block_i=64, block_j=64, block_k=32)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8. Cast back to the original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
