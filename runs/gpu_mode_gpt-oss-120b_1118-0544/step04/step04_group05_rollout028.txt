"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the 4‑D input tensor.
2. Cast to float16 for the heavy compute path.
3. Linear projections to the hidden dimension (left/right) and three
   gating projections (left_gate, right_gate, out_gate) using torch.nn.functional.linear.
4. If a mask is provided (and nomask is False) a small Triton kernel fuses
   mask multiplication, gate multiplication and writes the result back
   (left = left * left_gate * mask, same for right).
   When no mask is present the element‑wise multiplication is done in PyTorch.
5. Fold the hidden dimension into the batch dimension and perform the core
   contraction `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
   as a batched matrix multiplication `C = A @ Bᵀ` using torch.bmm
   (FP16 tensor cores give the best performance on Hopper/B200).
6. Reshape back to `[B, N, N, H]` and apply a second LayerNorm over the hidden
   dimension.
7. Multiply by the output gate and project back to the original channel dimension.
8. Cast the result back to the input dtype.

The only Triton component is the element‑wise mask‑gate fusion kernel,
which keeps the rest of the pipeline in highly‑optimized PyTorch/cuBLAS.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: fused mask * gate * tensor (in‑place)
# --------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    inp_ptr,          # input tensor (B,N,N,H) flattened
    gate_ptr,         # gate tensor same shape as input (flattened)
    mask_ptr,         # mask tensor (B,N,N) flattened
    out_ptr,          # output tensor (flattened, may be same as inp_ptr)
    total_elems,      # total number of elements in inp/gate (B*N*N*H)
    hidden_dim,       # H (used for mask broadcasting)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # load tensors
    a = tl.load(inp_ptr + offs, mask=mask, other=tl.float16(0.0))
    g = tl.load(gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # compute mask index: each block of size hidden_dim corresponds to one mask element
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))

    out = a * g * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_mask_and_gate(tensor, gate, mask, hidden_dim):
    """
    Fuse element‑wise multiplication with gate and mask using Triton.
    tensor, gate : [B, N, N, H] (float16, contiguous)
    mask          : [B, N, N]   (float16, contiguous)
    Returns tensor * gate * mask (broadcasted over H) as a new tensor.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    out = torch.empty_like(tensor)

    # flatten tensors for the kernel
    inp_ptr = tensor.contiguous().view(-1).data_ptr()
    gate_ptr = gate.contiguous().view(-1).data_ptr()
    mask_ptr = mask.contiguous().view(-1).data_ptr()
    out_ptr = out.data_ptr()

    BLOCK = 1024
    grid = ( (total + BLOCK - 1) // BLOCK, )
    mask_gate_fuse_kernel[grid](
        tl.pointer(inp_ptr),
        tl.pointer(gate_ptr),
        tl.pointer(mask_ptr),
        tl.pointer(out_ptr),
        total,
        hidden_dim,
        BLOCK=BLOCK,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask          : torch.Tensor of shape [B, N, N] or None (binary)
            - weights       : dict of model parameters (float32 tensors)
            - config        : dict with keys
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, ignore mask when True (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Load weights (cast to fp16)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Optional mask + gate fusion using Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is fp16 and contiguous
        mask_fp = mask.to(torch.float16, device=device).contiguous()
        # Fuse mask & gate into left/right
        left = _apply_mask_and_gate(left, left_gate, mask_fp, hidden_dim)
        right = _apply_mask_and_gate(right, right_gate, mask_fp, hidden_dim)
    else:
        # Simple element‑wise multiplication when no mask
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Core O(N³·H) reduction via batched GEMM (torch.bmm)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: C = A @ Bᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(-2, -1))

    # ------------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 12) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
