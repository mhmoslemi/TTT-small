"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (float32).
2. Cast to fp16 and apply the four linear layers:
   left_proj, right_proj, left_gate, right_gate, out_gate.
3. Fuse element‑wise mask (if present) and gate multiplications with a tiny
   Triton kernel (_gate_mask_kernel) to avoid extra Python loops.
4. Compute the core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   This is a batched matrix multiplication of shape (B*H, N, N) ×
   (B*H, N, N)ᵀ.  A custom Triton kernel (batched_matmul_kernel) performs
   the multiplication in fp16 with fp32 accumulation.
5. Apply the output layer‑norm, element‑wise out‑gate, and the final
   linear projection back to the original channel dimension.
6. Return the result in the original dtype (float32 by default).

Only the heavy element‑wise masking/gating and the O(N³) reduction are
implemented in Triton; all other operations use PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask and gate multiplications (optional)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape as above
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load operands
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share the same mask value
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------
# Kernel 2: batched matrix multiplication for the O(N³·H) reduction
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    batch, M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)      # traverses (batch * blocks_M)
    pid_n = tl.program_id(1)    # traverses blocks_N

    # ------------------------------------------------------------------
    # Compute batch index and block coordinates
    # ------------------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    batch_idx = pid // num_blocks_m
    block_m_idx = pid % num_blocks_m

    # Offsets for the current tile
    offs_m = block_m_idx * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the output tile
    c_ptrs = (c_ptr + batch_idx * stride_c_batch
              + offs_m[:, None] * stride_c_m
              + offs_n[None, :] * stride_c_n)

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in blocks
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptr + batch_idx * stride_a_batch
                    + offs_m[:, None] * stride_a_m
                    + offs_k[None, :] * stride_a_k,
                    mask=mask_a, other=0.0)

        b = tl.load(b_ptr + batch_idx * stride_b_batch
                    + offs_k[:, None] * stride_b_k
                    + offs_n[None, :] * stride_b_n,
                    mask=mask_b, other=0.0)

        # Perform the dot product (fp16 inputs, fp32 accumulation)
        acc += tl.dot(a, b)

    # Store the result (convert back to fp16)
    c = acc.to(tl.float16)
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, c, mask=mask_c)
