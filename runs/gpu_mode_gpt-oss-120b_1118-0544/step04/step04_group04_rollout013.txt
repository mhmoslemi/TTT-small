    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias = weights["norm.bias"].to(device).to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)  # shape [B,N,N,dim]

    # 2. Compute concatenated linear projections
    # Grab weight tensors (float32) to fp16
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    # Concatenate
    concat_weight = torch.cat([left_proj_weight,
                               right_proj_weight,
                               left_gate_weight,
                               right_gate_weight,
                               out_gate_weight], dim=0)  # shape [5*hidden_dim, dim]

    # Linear projection (no bias)
    proj = F.linear(x, concat_weight)  # shape [B,N,N,5*hidden_dim]

    # Split
    split_sizes = [hidden_dim, hidden_dim, hidden_dim, hidden_dim, hidden_dim]
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(proj, split_sizes, dim=-1)

    # Apply mask if needed
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # Apply gates to projections
    left = left_proj * left_gate
    right = right_proj * right_gate

    # 3. TriMul core using triton.ops.matmul
    B, N, _, _ = left.shape
    # reshape for batched matmul
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)  # (B*H,N,N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)  # (B*H,N,N)

    # Use triton.ops.matmul for FP16 matmul
    import triton.ops as ops
    out_mat = ops.matmul(left_mat, right_mat.transpose(-2, -1))  # (B*H,N,N)

    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # 4. to_out_norm (layernorm over hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # 5. Elementwise multiply out_gate (already fp16) and out
    # Could use torch multiplication directly; but we implement a tiny triton kernel for it for demonstration
    total_el = out.numel()
    BLOCK = 1024
    grid = ( (total_el + BLOCK - 1) // BLOCK, )
    # Launch triton kernel
    @triton.jit
    def elem_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
        pid = tl.program_id(0)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < size
        out_val = tl.load(out_ptr + offs, mask=mask)
        gate_val = tl.load(gate_ptr + offs, mask=mask)
        tl.store(out_ptr + offs, out_val * gate_val, mask=mask)

    elem_mul_kernel[grid](out, out_gate, total_el, BLOCK=BLOCK)

    # 6. Final linear projection to dim
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)  # shape [dim, hidden_dim]
    out = F.linear(out, to_out_weight)  # shape (B,N,N,dim)

    # Cast to float32 for output
    return out.to(torch.float32)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
import triton.ops as ops

@triton.jit
def elem_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out_val = tl.load(out_ptr + offs, mask=mask)
    gate_val = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out_val * gate_val, mask=mask)

def custom_kernel(data):
    """Outgoing TriMul forward pass using fused linear projections and Triton matmul.
    The heavy N³ computation is performed by Triton's optimized matmul kernel,
    while element‑wise gating is fused in a tiny Triton kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # Cast input to half precision early
    x = input_tensor.to(torch.float16)

    # 1. LayerNorm on input (C)
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias = weights["norm.bias"].to(device).to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # 2. Fuse all five linear projections into one matmul
    #   left_proj, right_proj, left_gate, right_gate, out_gate
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    # concatenate weights along output dimension
    concat_w = torch.cat([left_proj_w,
                          right_proj_w,
                          left_gate_w,
                          right_gate_w,
                          out_gate_w], dim=0)  # (5*H, dim)

    # Linear: (B,N,N,dim) @ transpose(concat_w) -> (B,N,N,5*H)
    proj = F.linear(x, concat_w)  # bias=False

    # Split into the five tensors
    split = torch.split(proj, hidden_dim, dim=-1)
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = split

    # Apply mask to projections if needed
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Gating
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # Combine projections with gates
    left = left_proj * left_gate   # (B,N,N,H)
    right = right_proj * right_gate

    # 3. TriMul core: batched GEMM for each hidden channel using Triton matmul
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Triton matmul (FP16) – computes left_mat @ right_matᵀ for each batch
    out_mat = ops.matmul(left_mat, right_mat.transpose(-2, -1))

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 4. LayerNorm on hidden dimension
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=1e-5)

    # 5. Fuse out_gate multiplication via a tiny Triton kernel
    total_el = out.numel()
    BLOCK = 1024
    grid = ( (total_el + BLOCK - 1) // BLOCK, )
    elem_mul_kernel[grid](out, out_gate, total_el, BLOCK=BLOCK)

    # 6. Final linear projection back to original dim
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # bias=False

    # Cast back to float32 as required by the interface
    return out.to(torch.float32)

"""
Outgoing TriMul (Triangle Multiplicative Update) forward pass.

The heavy O(N³) computation is performed by Triton's optimized matmul kernel
(`triton.ops.matmul`).  All linear projections are fused into a single matmul
to reduce memory traffic, and a tiny Triton kernel fuses the element‑wise
multiplication with the output gate.  The implementation works entirely in
FP16 (except the final cast to FP32) and supports optional masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
import triton.ops as ops


@triton.jit
def _elem_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    """out = out * gate  (flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out_val = tl.load(out_ptr + offs, mask=mask)
    gate_val = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out_val * gate_val, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple):
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float), may be ignored
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension H
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (over C)
    # ------------------------------------------------------------------
    x = input_tensor.to(torch.float16)
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias = weights["norm.bias"].to(device).to(torch.float16)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, C]  fp16

    # ------------------------------------------------------------------
    # 2. Fuse all five linear projections (left/right & gates) into one matmul
    # ------------------------------------------------------------------
    # Retrieve and cast projection weights (no bias in any of them)
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    # Concatenate along the output dimension: (5*H, C)
    concat_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0
    )

    # Linear projection: (B,N,N,C) @ (C,5*H).T -> (B,N,N,5*H)
    proj = F.linear(x, concat_w)  # bias=False

    # Split back into the five tensors
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(
        proj, hidden_dim, dim=-1
    )

    # ------------------------------------------------------------------
    # 3. Optional masking of the left/right projections
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4. Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # Combine projections with their gates
    left = left_proj * left_gate   # [B,N,N,H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul: batched GEMM over the sequence dimension.
    #    Implemented with Triton's optimized matmul kernel.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Reshape to (B*H, N, N) – each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Compute left_mat @ right_matᵀ for each batch using Triton
    out_mat = ops.matmul(left_mat, right_mat.transpose(-2, -1))  # (B*H, N, N)

    # Reshape back to (B,N,N,H)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6. LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # fp16

    # ------------------------------------------------------------------
    # 7. Fuse out_gate multiplication via a tiny Triton kernel
    # ------------------------------------------------------------------
    total_el = out.numel()
    BLOCK = 1024
    grid = ((total_el + BLOCK - 1) // BLOCK,)
    _elem_mul_kernel[grid](out, out_gate, total_el, BLOCK=BLOCK)

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)  # (C, H)
    out = F.linear(out, to_out_w)  # (B,N,N,C)  fp16

    # Return in float32 as required by the interface
    return out.to(torch.float32)
