class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the reference PyTorch version but fuses the heavy
O(N³·H) reduction with a custom Triton batched‑GEMM kernel
(left @ rightᵀ).  Linear projections for the left/right pathways are combined
with their corresponding gates to halve the number of input‑tensor reads.
All costly tensor math is performed in float16 with float32 accumulation,
while the final result is cast back to the input dtype.

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernels

@triton.jit
def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dims (M×K)·(K×N)→(M×N)
    stride_am, stride_ak,              # strides for A (batch‑local)
    stride_bk, stride_bn,              # strides for B (interpreted as K×N)
    stride_cm, stride_cn,              # strides for C
    stride_ab, stride_bb, stride_cb,   # strides for the batch dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM: C = A @ Bᵀ  (A,B shape: batch×M×K)"""

    pid = tl.program_id(0)          # output tile id (i‑j)
    batch = tl.program_id(1)        # batch index (B·H)

    # output tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # linear offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the K‑dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = k_offset + offs_k
        mask_k = cur_k < K

        # A is (M, K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B is stored as (K, N) – we load it as Bᵀ
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)          # fp16·fp16 → fp32 accumulation

    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )

def _elemwise_mul_kernel(
    a_ptr, b_ptr, out_ptr, size,
    BLOCK: tl.constexpr,
):
    """Element‑wise multiplication of two 1‑D contiguous tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

### Approach and Rationale
The core operation of TriMul involves computing an element-wise product and summing over a sequence dimension, which originally had a high complexity (O(N^3)) and low arithmetic intensity. To optimize this, the new approach leverages the Gemm-like structure of the operation by re-indexing the tensors to perform a standard matrix multiplication. Specifically:
- The left and right tensors are flattened along their sequence (N) and hidden (H) dimensions.
- The hidden dimension is replicated to align with the sequence dimension, allowing the operation to be viewed as a batched matrix multiplication.
This re-indexing technique increases arithmetic intensity by enabling the use of fast CPU-like matrix operations, thereby reducing the role of memory accesses. The approach is different from the previous one by structuring the computation around Gemm primitives from the outset.

### Mathematical Specification
1. **Input**:
   - `x`: Tensor of shape `[B, N, N, C]`
2. **LayerNorm**:
   - Applied to `x` along channel dimension `C`, parameterized by scalar `γ` and `β`.
3. **Projection and Gate**:
   - Compute two projections (to hidden_dim `H`) and their corresponding gates.
4. **Mask Integration**:
   - If a mask `m` is provided, it is broadcast over `H` and used to zero out elements.
5. **Core Operation**:
   - For each batch and channel, compute an intermediate tensor `y` of shape `[B, N, N, H]` via:
     - Flattening `x` to form two batches with dimensions `[B_H, H]` and `[B_N, N]`.
     - Performing matrix multiplication `y = B @ C`, where `B` and `C` are appropriately permuted versions of the projections and gates.
   - The resulting `y` is then reshaped back to `[B, N, N, H]`.
6. **Final Projection**:
   - Interpolate back to the original channel dimension `C` using a linear projection.
