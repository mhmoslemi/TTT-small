def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', False)

    # 1. LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])
    # Convert to half for speed
    x_h = x.to(torch.float16)

    # 2. Compute left, right projections via linear
    # left_proj = x_h @ left_proj_weight.T + bias? Actually bias not included. Use F.linear
    left = torch.nn.functional.linear(x_h, weights['left_proj.weight'].to(dtype=torch.float16))
    right = torch.nn.functional.linear(x_h, weights['right_proj.weight'].to(dtype=torch.float16))

    # 3. Compute gates
    left_gate = torch.nn.functional.linear(x_h, weights['left_gate.weight'].to(dtype=torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, weights['right_gate.weight'].to(dtype=torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, weights['out_gate.weight'].to(dtype=torch.float16)).sigmoid()

    # 4. Apply mask if any
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # 5. Apply gates
    left = left * left_gate
    right = right * right_gate

    # 6. Core multiplication via Triton
    B, N, _, H = left.shape  # B,N,N,H
    # Rearrange to (B*H, N, N)
    left_flat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_flat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # compute strides
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)  # K dimension (fast) treat as 1
    stride_b_n = right_flat.stride(1)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Choose block sizes:
    BLOCK_M = 64  # can be 128 for larger N but 64 ok
    BLOCK_N = 64
    BLOCK_K = 64

    # number of blocks
    grid = ( ( (N + BLOCK_M - 1)//BLOCK_M ) * ( (N + BLOCK_N - 1)//BLOCK_N ), B*H )

    _matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape output back: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()

    # 7. to_out_norm (layernorm over hidden dim)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'])

    # 8. out_gate multiplication
    out = out * out_gate

    # 9. Final projection
    out = torch.nn.functional.linear(out, weights['to_out.weight'].to(torch.float16))

    # Cast back to original dtype (float32)
    out = out.to(input_tensor.dtype)

    return out

"""
TriMul outgoing implementation with Triton.

The heavy O(N³·H) reduction (for each hidden dimension d we compute
left[..., d] @ right[..., d].T) is performed by a custom Triton batched
GEMM kernel. All other operations (LayerNorms, linear projections, gating,
masking) are carried out with regular PyTorch calls. The kernel works in
float32 (no loss of precision) and supports both masked and un‑masked
inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton batched GEMM kernel: C = A @ Bᵀ where A,B have shape (batch, M, K)
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix dimensions (M×K)·(K×N)→(M×N)
    stride_am, stride_ak,               # strides for A (M, K)
    stride_bk, stride_bn,               # strides for B (K, N) – note the transposed load
    stride_cm, stride_cn,               # strides for C (M, N)
    stride_ab, stride_bb, stride_cb,    # strides for the batch dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ Bᵀ.
    A : (batch, M, K)
    B : (batch, K, N)   (interpreted as the transpose of the original tensor)
    C : (batch, M, N)
    """
    pid = tl.program_id(0)      # tile index over (M, N)
    batch = tl.program_id(1)    # batch index (covers B * H)

    # -------------------------------------------------
    # Tile coordinates
    # -------------------------------------------------
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------------------------------------------------
    # Base pointers for this batch
    # -------------------------------------------------
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Main K-loop
    # -------------------------------------------------
    num_k_iter = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_iter):
        k_offset = k * BLOCK_K
        cur_k = k_offset + offs_k
        mask_k = cur_k < K

        # Load A block (M, K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B block (K, N) – treating the stored tensor as transposed
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp32 accumulation

    # -------------------------------------------------
    # Write back
    # -------------------------------------------------
    c = acc.to(tl.float16) if tl.dtype(a) == tl.float16 else acc.to(tl.float32)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    device = x.device
    dtype = x.dtype

    # -------------------------------------------------
    # 1. Input LayerNorm
    # -------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -------------------------------------------------
    # 2. Linear projections & gates
    # -------------------------------------------------
    # Projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # -------------------------------------------------
    # 3. Optional mask
    # -------------------------------------------------
    if mask is not None and not nomask:
        # mask shape [B, N, N] -> broadcast over hidden dim
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------
    # 4. Apply gates
    # -------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5. Core O(N³·H) reduction via Triton
    # -------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # Rearrange to (batch = B*H, M = N, K = N)
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_t.view(B * H, N, N)   # (batch, M, K)
    right_flat = right_t.view(B * H, N, N) # (batch, K, N) – will be read as transposed

    out_flat = torch.empty_like(left_flat)  # (batch, M, N)

    # Strides (in elements, not bytes)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)   # M stride
    stride_a_k = left_flat.stride(2)   # K stride

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # K stride (fast) – we treat K as innermost
    stride_b_n = right_flat.stride(1)   # N stride (outer)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Tiling parameters – chosen to fit typical N ≤ 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                     # M, N, K (all equal to seq_len)
        stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 6. Hidden‑dim LayerNorm
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -------------------------------------------------
    # 7. Output gate (element‑wise)
    # -------------------------------------------------
    out = out * out_gate

    # -------------------------------------------------
    # 8. Final projection back to original channel dimension
    # -------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    # Cast back to the input dtype (should already be float32)
    out = out.to(dtype)

    return out
