"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑intensive part.
3. Linear projections:
      left  = x @ W_left   (B,N,N,H)
      right = x @ W_right
   Gate projections (sigmoid):
      left_gate  = σ(x @ W_left_gate)
      right_gate = σ(x @ W_right_gate)
      out_gate   = σ(x @ W_out_gate)        # (B,N,N,H)
4. Fuse mask, projection and gate in a single Triton kernel:
      left  = left_proj  * left_gate  * mask
      right = right_proj * right_gate * mask
   (mask is broadcast over the hidden dimension H).
   If no mask is supplied the multiplication is performed with a simple
   elementwise product.
5. Core reduction (batched matrix multiplication):
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented with `torch.bmm` on the reshaped tensors.
6. Output LayerNorm over hidden dimension H.
7. Multiply by `out_gate` and apply final linear projection back to `dim`.
8. Cast back to the original dtype (usually float32).

The implementation mixes Triton (step 4) and highly‑optimized PyTorch
primitives for the remaining operations.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# --------------------------------------------------------------
# Triton kernel: (proj * gate * mask) – mask broadcast over H
# --------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,   # fp16* (total,)
    gate_ptr,   # fp16* (total,)
    mask_ptr,   # fp16* (total_mask = B*N*N,)
    out_ptr,    # fp16* (total,)
    total_elems,  # int
    hidden_dim,   # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)  # [0, BLOCK_SIZE)

    # guard for the last (partial) block
    mask = offsets < total_elems

    # compute mask index (each H entries share the same mask element)
    mask_idx = offsets // hidden_dim

    # load values
    proj = tl.load(proj_ptr + offsets, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offsets, mask=mask, other=0.0)
    msk  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * msk
    tl.store(out_ptr + offsets, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Fuse projection, gate and optional mask using Triton.
    All tensors must be fp16 and contiguous.
    proj, gate : (B,N,N,H)
    mask       : (B,N,N) (0/1 values, fp16)
    Returns proj*gate*mask (broadcast mask over H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple containing:
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask (optional) : torch.Tensor of shape [B, N, N] (bool or 0/1)
            - weights : dict of model parameters (float32)
            - config : dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack arguments – support both 3‑tuple (no mask) and 4‑tuple.
    # ------------------------------------------------------------------
    if len(data) == 4:
        input_tensor, mask, weights, config = data
    elif len(data) == 3:
        input_tensor, weights, config = data
        mask = None
    else:
        raise ValueError("data must be a 3‑ or 4‑tuple.")

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast weights to fp16 (no bias for linear layers)
    # ------------------------------------------------------------------
    w_left   = weights["left_proj.weight"].to(torch.float16)
    w_right  = weights["right_proj.weight"].to(torch.float16)
    w_left_g = weights["left_gate.weight"].to(torch.float16)
    w_right_g= weights["right_gate.weight"].to(torch.float16)
    w_out_g  = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, w_left)   # (B,N,N,H)
    right_proj = F.linear(x, w_right)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, w_left_g).sigmoid()
    right_gate = F.linear(x, w_right_g).sigmoid()
    out_gate   = F.linear(x, w_out_g).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Apply mask + gate (fusion via Triton if needed)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous, and broadcastable.
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        # No mask – simple elementwise multiplication.
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched matrix multiplication over the sequence dim.
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm → out‑gate → final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )
    out = out * out_gate
    out = F.linear(out, w_to_out)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype.
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
