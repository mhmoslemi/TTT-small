@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
   batch = tl.program_id(1)
   pid = tl.program_id(0)
   num_pid_n = tl.cdiv(N, BLOCK_N)
   pid_m = pid // num_pid_n
   pid_n = pid % num_pid_n
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   mask_m = offs_m < M
   mask_n = offs_n < N

   a_batch = a_ptr + batch * stride_ab
   b_batch = b_ptr + batch * stride_bb
   c_batch = c_ptr + batch * stride_cb

   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K
        a = tl.load(a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
                     mask=mask_m[:, None] & mask_k[None, :],
                     other=0.0)
        b = tl.load(b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
                     mask=mask_k[:, None] & mask_n[None, :],
                     other=0.0)
        acc += tl.dot(a, b)
   c = acc.to(tl.float16)
   tl.store(c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
             c, mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul outgoing update (AlphaFold3) – fast Triton implementation.

The forward pass consists of:
1. LayerNorm on the input tensor.
2. Two fused linear+gate projections (left/right) with optional mask.
3. Core O(N³·H) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication left @ rightᵀ for each hidden
   channel h. Implemented with a highly‑tuned Triton GEMM kernel.
4. Hidden‑dim LayerNorm, output gate and final linear projection back to
   the original channel dimension.
All heavy arithmetic is performed in fp16 with fp32 accumulation; the
result is cast back to fp32 before returning.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton batched GEMM kernel (C = A @ Bᵀ)
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dims (M×K)·(K×N)→(M×N)
    stride_am, stride_ak,              # strides for A (M,K)
    stride_bk, stride_bn,              # strides for B (K,N) – we read Bᵀ
    stride_cm, stride_cn,              # strides for C (M,N)
    stride_ab, stride_bb, stride_cb,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM: C = A @ Bᵀ."""
    pid = tl.program_id(0)          # output tile id (i‑j)
    batch = tl.program_id(1)        # batch index (B·H)

    # Compute tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # Linear offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # Load A tile (M × K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load Bᵀ tile (K × N) – B is stored as (K,N) but we access it as transposed
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)  # fp16·fp16 → fp32 accumulation

    # Write back result in fp16
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[:, None],
    )


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for the heavy path
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 2. Cast model weights to fp16
    # ------------------------------------------------------------------- #
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------- #
    # 3. Linear projections + gates (combined)
    # ------------------------------------------------------------------- #
    # Shape after linear: [B, N, N, 2*H]
    left_comb = F.linear(x_fp16, left_comb_w)   # bias omitted (bias=False)
    right_comb = F.linear(x_fp16, right_comb_w)

    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    if mask is not None and not nomask:
        # Ensure mask is fp16 and broadcastable
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Apply gates
    left = left_proj * left_gate      # [B, N, N, H]
    right = right_proj * right_gate   # [B, N, N, H]

    # ------------------------------------------------------------------- #
    # 4. Core O(N³·H) reduction via batched GEMM (Triton)
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape  # (B, N, N, H)

    # Rearrange to (batch = B*H, M = N, K = N) for A and (batch, K = N, N) for Bᵀ
    left_t = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_flat = left_t.view(B * H, N, N)    # (batch, M, K)
    right_flat = right_t.view(B * H, N, N)  # (batch, K, N) – will be read as transposed

    out_flat = torch.empty_like(left_flat)  # (batch, M, N) fp16

    # Strides (in elements)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(1)
    stride_b_n = right_flat.stride(2)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Tiling parameters – tuned for typical sequence lengths ≤1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                 # M, N, K
        stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 5. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------- #
    # 6. Output gate (element‑wise) and final projection
    # ------------------------------------------------------------------- #
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()  # [B, N, N, H]
    out = out * out_gate                               # fused element‑wise multiplication

    out = F.linear(out, to_out_w)                      # [B, N, N, C]

    # Cast result back to float32 (original input dtype)
    out = out.to(torch.float32)

    return out
