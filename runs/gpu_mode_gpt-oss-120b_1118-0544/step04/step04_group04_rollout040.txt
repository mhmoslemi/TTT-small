    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # Load weights
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias = weights["norm.bias"].to(device).to(torch.float16)
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # 1. LayerNorm (float32). Use torch.nn.functional.layer_norm.
    # To avoid storing weight and bias in FP16 for layer_norm (which might require float32),
    # we can compute in float32 then cast.
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight.to(torch.float32),
        bias=norm_bias.to(torch.float32),
        eps=1e-5,
    )
    # Cast to fp16
    x_fp16 = x.to(torch.float16)

    # 2. Linear projections
    left_proj = torch.nn.functional.linear(x_fp16, left_proj_weight)  # shape [B,N,N,hidden]

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm over the input channel dimension (float32 → fp16).
2. Four linear projections (left/right and their gates) in fp16.
3. Optional mask is applied to the left/right projections.
4. Gating via sigmoid and element‑wise multiplication.
5. Batched GEMM: for each hidden channel d we compute
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a matrix multiplication left @ rightᵀ.  The GEMM is
   performed by a custom Triton kernel that processes all
   (batch * hidden) matrices in a single launch.
6. Layer‑norm over the hidden dimension.
7. out_gate = sigmoid(linear(x)), fused element‑wise multiplication
   with the Triton `elementwise_mul_kernel`.
8. Final linear projection back to the original channel dimension.
9. Cast the result to float32 (required by the interface).

Only the core matrix multiplication and the final element‑wise
multiplication are executed in Triton; all other operations use
efficient PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    """out = out * gate  (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out * gate, mask=mask)


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matmul C = A @ Bᵀ  where A,B are (batch, M, K) and (batch, K, N)."""
    batch_id = tl.program_id(0)          # which (batch * hidden) matrix
    pid = tl.program_id(1)               # combined (block_row, block_col) id
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * stride_batch_a
    b_base = b_ptr + batch_id * stride_batch_b

    a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = kk * BLOCK_K
        k_valid = (cur_k + offs_k) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write‑back
    c = acc.to(tl.float16)
    c_base = c_ptr + batch_id * stride_batch_c
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             c,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float)
            - weights: dict of model weight tensors (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ---------- 1. Load & cast weights ----------
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias   = weights["norm.bias"].to(device).to(torch.float16)

    left_proj_weight  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)

    left_gate_weight  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight   = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # ---------- 2. Input LayerNorm (float32 → fp16) ----------
    # LayerNorm expects weight/bias of same dtype as input; we use float32 for stability.
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight.to(torch.float32),
        bias=norm_bias.to(torch.float32),
        eps=1e-5,
    )
    x_fp16 = x.to(torch.float16)          # [B, N, N, dim]

    # ---------- 3. Linear projections ----------
    left_proj  = F.linear(x_fp16, left_proj_weight)   # [B, N, N, hidden]
    right_proj = F.linear(x_fp16, right_proj_weight)

    # ---------- 4. Optional mask ----------
    if not nomask:
        mask_fp16 = mask.to(device).to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj  = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ---------- 5. Gates ----------
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_weight))

    left  = left_proj * left_gate   # [B, N, N, hidden]
    right = right_proj * right_gate

    # ---------- 6. Batched GEMM via Triton ----------
    B, N, _, _ = left.shape
    # Bring hidden dimension to the second axis
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, hidden, N, N]

    left_mat  = left.view(B * hidden_dim, N, N)   # (batch*hidden, M=N, K=N)
    right_mat = right.view(B * hidden_dim, N, N)  # (batch*hidden, K=N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat)  # fp16

    # Strides for the kernel
    stride_batch_a = left_mat.stride(0)
    stride_am      = left_mat.stride(1)   # row stride (M)
    stride_ak      = left_mat.stride(2)   # col stride (K)

    stride_batch_b = right_mat.stride(0)
    stride_bk      = right_mat.stride(2)  # row stride for K
    stride_bn      = right_mat.stride(1)  # col stride for N (j)

    stride_batch_c = out_mat.stride(0)
    stride_cm      = out_mat.stride(1)    # row stride (M)
    stride_cn      = out_mat.stride(2)    # col stride (N)

    M = N
    N_val = N
    K = N
    num_batches = B * hidden_dim

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        num_batches,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N_val + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M,
        N_val,
        K,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_batch_a,
        stride_batch_b,
        stride_batch_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- 7. Output layer‑norm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # still fp16

    # ---------- 8. Apply out_gate (elementwise Triton kernel) ----------
    total_el = out.numel()
    BLOCK = 1024
    grid_gate = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid_gate](
        out,
        out_gate,
        total_el,
        BLOCK=BLOCK,
    )

    # ---------- 9. Final linear projection ----------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim] (fp16)

    # Cast to float32 as required by the API
    return out.to(torch.float32)
```
