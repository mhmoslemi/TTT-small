---------
1. Layer‑norm the input tensor (B,N,N,C) (fp16).
2. Linear projections → left/right (B,N,N,H) and corresponding gates
   (B,N,N,H) using fp16 weights.
3. Fuse projection * gate * mask (if a mask is present) in a single
   Triton kernel, broadcasting the mask over the hidden dimension.
4. Reshape the tensors to (B·H, N, N) and perform a batched GEMM:
     out[b·h, i, j] = Σ_k left[b·h, i, k] * right[b·h, k, j]
   which implements the einsum `'b i k d, b j k d -> b i j d'`.
5. Reshape back to (B,N,N,H), apply a second LayerNorm, multiply by the
   output‑gate, and linearly project back to the original channel
   dimension C.
6. Cast the result back to the input dtype (usually float32).

Only the mask‑gate fusion is implemented in Triton; the heavy matrix
multiplication leverages PyTorch’s highly‑optimized batched GEMM (cuBLAS /
Tensor‑cores) on fp16 data.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total_proj,) where total_proj = B*N*N*H
    gate_ptr,          # fp16 * (total_gate,) same size as proj
    mask_ptr,          # fp16 * (total_mask,) where total_mask = B*N*N
    out_ptr,           # fp16 * (total_proj,)
    total_elems,       # i32   – total number of elements in proj/gate
    hidden_dim,        # i32   – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # Compute mask index (broadcast over hidden dimension)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 8192,
) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using the Triton kernel above.
    All tensors must be on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input tensor [B, N, N, C]
        torch.Tensor,          # mask tensor [B, N, N] (or None)
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # --------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------- layer norm (input) -----------------------
    # Cast input to fp16 for all subsequent ops
    x_fp16 = input_tensor.to(torch.float16)

    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = torch.nn.functional.layer_norm(
        x_fp16,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # (B,N,N,dim) fp16

    # ---------------------------- cast weights ---------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)       # (C, H)

    # ------------------------ linear projections -------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------- gates -----------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()   # (B,N,N,H)
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()    # (B,N,N,H)

    # -------------------------- mask & gate fusion ----------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------- core O(N³·H) reduction via batched GEMM -------
    # Reshape to (B*H, N, N) and perform left @ right^T.
    # left: (B,N,N,H) -> (B,H,N,N) where dims are (i,k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right: (B,N,N,H) -> (B,H,N,N) where dims are (k,j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B = left_perm.shape[0]
    H = left_perm.shape[1]
    N = left_perm.shape[2]

    left_mat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication (fp16) – uses cuBLAS/Tensor Cores
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # (B, N, N, H)

    # ------------------- second LayerNorm & output gate ---------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection ----------------------
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # -------------------------- cast back --------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
