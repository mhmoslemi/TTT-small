"""
TriMul (outgoing) forward pass – Triton‑accelerated mask & gate fusion.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the provided LN parameters.
2. Cast the normalized tensor to fp16 for compute‑intensive ops.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are applied with fp16 weights.
4. If a mask is present (and not disabled) a Triton kernel fuses the
   elementwise multiplication of projection, gate and mask, avoiding
   extra memory passes.
5. Core reduction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix‑multiply of shape (B*H, N, N) × (B*H, N, N)ᵀ.
   The reduction is performed with torch.bmm (cuBLAS) – already optimal.
6. Apply a second LayerNorm (over the hidden dimension), multiply by the
   output gate and project back to the original channel dimension.
7. Cast the result back to the original dtype (usually float32).

Only the mask‑gate fusion is done in Triton; all other heavy work uses
high‑performance torch kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_kernel(
    proj_ptr,      # fp16* (B*N*N*H,)
    gate_ptr,      # fp16* (B*N*N*H,)
    mask_ptr,      # fp16* (B*N*N,)
    out_ptr,       # fp16* (B*N*N*H,)
    total_elems,   # i64  – total number of elements in proj/gate
    hidden_dim,    # i64  – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse elementwise multiplication: out = proj * gate * mask
    where mask is broadcasted over the hidden dimension.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # Bounds check for the flattened tensor of size total_elems
    mask = offs < total_elems

    # Compute index into the mask (each mask element repeats hidden_dim times)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * mval
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse elementwise multiplication proj * gate * mask (broadcast over hidden_dim)
    using a Triton kernel. All tensors must be fp16 and contiguous.

    Args:
        proj: (B,N,N,H) tensor.
        gate: (B,N,N,H) tensor.
        mask: (B,N,N) tensor (fp16, 0/1).
        block_size: kernel block size.

    Returns:
        Tensor of shape (B,N,N,H) = proj*gate*mask
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],   # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16 on the same device
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,C) -> (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (same shape)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask + gates (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM (out = left @ rightᵀ)
    #    Shapes: left, right : (B,N,N,H) → (B*H, N, N) × (B*H, N, N)ᵀ
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N) contiguous view for both tensors
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication over the reduction axis k
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype (e.g., float32)
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
