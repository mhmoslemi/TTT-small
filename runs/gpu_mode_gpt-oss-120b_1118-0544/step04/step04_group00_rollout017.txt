@triton.jit
def _apply_gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, total_elems, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # Compute mask index = offs // hidden_dim
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # default mask 1 (no mask) - but we only call kernel when mask present.
    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask)

"""
TriMul (outgoing) forward pass – highly fused implementation.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑heavy part.
3. Fuse the five linear projections (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single matrix‑multiply
   by concatenating their weight matrices.  Split the result and
   apply sigmoid to the three gate tensors.
4. Fuse mask, gating and projection for the left/right tensors
   with a tiny Triton kernel (elementwise prod with broadcast mask).
5. Core reduction (tri‑mul) : for each hidden channel `h`,
   out[...,h] = left[...,h] @ right[...,h].T.
   Implemented as a batched GEMM using `torch.bmm` on tensors
   reshaped to (B*H, N, N).
6. Output LayerNorm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
7. Cast back to the original dtype (usually float32).

The kernel in step 4 is required to satisfy the “use Triton” rule.
All heavy linear work is kept in FP16 to exploit Tensor‑cores.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: elementwise prod of proj * gate * mask (mask broadcasted)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,    # fp16 * (B*N*N*H,)
    gate_ptr,    # fp16 * (B*N*N*H,)
    mask_ptr,    # fp16 * (B*N*N,)
    out_ptr,     # fp16 * (B*N*N*H,)
    total_elems, # int = B*N*N*H
    hidden_dim,  # int = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask_valid = offs < total_elems

    # mask index (broadcast over hidden dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask_valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask_valid, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask_valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse projection, gate and (optional) mask using Triton.
    All tensors must be contiguous fp16.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)   (broadcast over H)
    Returns prod = proj * gate * mask
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (binary) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys
                * "dim" (int)          – input channel dimension C
                * "hidden_dim" (int)   – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over channel dim (C)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to FP16 for the heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Fuse five linear projections into one matmul
    # ------------------------------------------------------------------
    # Weight shapes:
    #   left_proj.weight   : (hidden_dim, dim)
    #   right_proj.weight  : (hidden_dim, dim)
    #   left_gate.weight   : (hidden_dim, dim)
    #   right_gate.weight  : (hidden_dim, dim)
    #   out_gate.weight    : (hidden_dim, dim)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)
    proj_concat = torch.nn.functional.linear(x_flat, w_concat)  # (B*N*N, 5*hidden_dim)
    proj_concat = proj_concat.view(B, N, N, -1)

    h = hidden_dim
    left_proj  = proj_concat[..., 0:h]
    right_proj = proj_concat[..., h:2 * h]
    left_gate  = proj_concat[..., 2 * h:3 * h].sigmoid()
    right_gate = proj_concat[..., 3 * h:4 * h].sigmoid()
    out_gate   = proj_concat[..., 4 * h:5 * h].sigmoid()

    # ------------------------------------------------------------------
    # 3. Apply mask + gate (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and not nomask:
            # mask present but nomask=False handled above; otherwise ignore
            left = left * mask.unsqueeze(-1).to(torch.float16)
            right = right * mask.unsqueeze(-1).to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched GEMM over hidden dim.
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * h, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * h, N, N)

    # batched matrix multiply: (B*H, N, N) @ (B*H, N, N)^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, h, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (h,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # final linear projection back to `dim`
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out.view(-1, h), to_out_w)  # (B*N*N, dim)
    out = out.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 6. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
