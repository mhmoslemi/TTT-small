   fused matmul per side (proj + gate weights concatenated).
3. Optional masking is applied before gating.
4. The core reduction  out[b,i,j,d] = Σₖ left[b,i,k,d]·right[b,j,k,d]
   is performed as a batched GEMM (A @ Bᵀ) where each hidden dimension is
   treated as an independent batch; the kernel uses float16 inputs with
   float32 accumulation.
5. Hidden‑dim LayerNorm, output gate and final linear projection back to the
   original channel dimension.
6. The result is cast back to the original input dtype (float32 in the tests).

Only the O(N³·H) contraction is executed in Triton; all other ops use PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton batched GEMM kernel (A @ Bᵀ) for the TriMul contraction
# --------------------------------------------------------------------------- #
@triton.jit
def _triton_gemm_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                        # matrix dimensions (A: M×K, Bᵀ: K×N → C: M×N)
    stride_am, stride_ak,           # strides for A (row, col)
    stride_bk, stride_bn,           # strides for B (k‑dim, n‑dim) – B is stored as (K,N)
    stride_cm, stride_cn,           # strides for C (row, col)
    stride_ab, stride_bb, stride_cb,  # stride for the batch dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # tile index within a batch
    batch = tl.program_id(1)        # batch (B*H)

    # tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # A is (M, K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B is stored as (K, N); we load it transposed to perform A @ Bᵀ
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp16·fp16 → fp32

    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------------------- #
# Entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------- #
    # Unpack
    # ------------------------------------------------------------------- #
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    orig_dtype = x.dtype
    device = x.device

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for the heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 2. Prepare fused projection+gate weights (fp16)
    # ------------------------------------------------------------------- #
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------- #
    # 3. Linear proj + gates (fused)
    # ------------------------------------------------------------------- #
    # [B, N, N, 2*H]
    left_comb = F.linear(x, left_comb_w)
    right_comb = F.linear(x, right_comb_w)

    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    left = left_proj * left_gate   # [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------- #
    # 4. Core contraction via Triton batched GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape to (batch = B*H, M = N, K = N) for A and B
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_t.view(B * H, N, N)   # (batch, M, K)
    right_flat = right_t.view(B * H, N, N) # (batch, K, N) – stored as (j,k)

    out_flat = torch.empty_like(left_flat)  # (batch, M, N), fp16

    # strides for kernel
    stride_a_m = left_flat.stride(1)   # row stride (M)
    stride_a_k = left_flat.stride(2)   # col stride (K)
    stride_b_k = right_flat.stride(2)  # K stride (inner)
    stride_b_n = right_flat.stride(1)  # N stride (outer)
    stride_c_m = out_flat.stride(1)    # row stride for C
    stride_c_n = out_flat.stride(2)    # col stride for C

    stride_a_batch = left_flat.stride(0)
    stride_b_batch = right_flat.stride(0)
    stride_c_batch = out_flat.stride(0)

    # Tiling parameters (tuned for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _triton_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_a_m,
        stride_a_k,
        stride_b_k,
        stride_b_n,
        stride_c_m,
        stride_c_n,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 5. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------- #
    # 6. Output gate (element‑wise) and final projection
    # ------------------------------------------------------------------- #
    out_gate = F.linear(x, out_gate_w).sigmoid()   # [B, N, N, H]
    out = out * out_gate

    out = F.linear(out, to_out_w)   # [B, N, N, C]

    # Cast back to original dtype (float32 in the test suite)
    out = out.to(orig_dtype)
    return out
```
