@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_WARPS: tl.constexpr, NUM_STAGES: tl.constexpr
):
    pid = tl.program_id(0)
    batch = tl.program_id(1)

    # Compute block index
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers
    a = a_ptr + batch * stride_ab
    b = b_ptr + batch * stride_bb
    c = c_ptr + batch * stride_cb

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        a_ptrs = a + (offs_m[:, None] * stride_am + cur_k[None,:] * stride_ak)
        b_ptrs = b + (cur_k[:, None] * stride_bk + offs_n[None,:] * stride_bn)

        a_vals = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None,:], other=0.0)
        b_vals = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None,:], other=0.0)

        acc += tl.dot(a_vals, b_vals)

    # Write back
    c_ptrs = c + (offs_m[:, None] * stride_cm + offs_n[None,:] * stride_cn)
    c_vals = acc.to(tl.float16)
    tl.store(c_ptrs, c_vals, mask=mask_m[:, None] & mask_n[None,:])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy O(N³·H) contraction is implemented as a batched GEMM:
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
which is mathematically a batch of matrix multiplications
    left[b,h] @ right[b,h].T  (N×N @ N×Nᵀ → N×N).

All linear projections, gating, and LayerNorms are performed in
float16 on the host (torch) while the reduction uses a custom Triton
kernel with float32 accumulation.  The final result is cast back to the
original input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM: C = A @ Bᵀ  (A,B: batch×M×K, batch×K×N)."""
    pid = tl.program_id(0)          # tile id within a batch
    batch = tl.program_id(1)        # batch index (B * hidden_dim)

    # -------------------------------------------------------------
    # Tile coordinates
    # -------------------------------------------------------------
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # -------------------------------------------------------------
    # Offsets within the tile
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------------------------------------------------------------
    # Base pointers for the current batch
    # -------------------------------------------------------------
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    # -------------------------------------------------------------
    # Accumulator (float32 for numerical stability)
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Main K‑loop
    # -------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)        # fp16·fp16 → fp32

    # -------------------------------------------------------------
    # Write back result (cast to fp16)
    # -------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] matching the dtype of input_tensor.
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Prepare weights in fp16 (device‑side)
    # -----------------------------------------------------------------
    device = x.device
    dtype = torch.float16

    # combine projection + gate for left/right to halve reads
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype)          # (2*H, C)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype)          # (2*H, C)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype)   # (H, C)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype)       # (C, H)

    # -----------------------------------------------------------------
    # 3. Linear projections + gates (combined)
    # -----------------------------------------------------------------
    # x : [B, N, N, C]
    left_comb = F.linear(x, left_comb_w)    # [B, N, N, 2*H]
    right_comb = F.linear(x, right_comb_w)  # [B, N, N, 2*H]

    left_proj = left_comb[..., :hidden_dim]                     # [B,N,N,H]
    left_gate = left_comb[..., hidden_dim:].sigmoid()           # [B,N,N,H]
    right_proj = right_comb[..., :hidden_dim]                   # [B,N,N,H]
    right_gate = right_comb[..., hidden_dim:].sigmoid()         # [B,N,N,H]

    # optional masking (broadcast over hidden dim)
    if mask is not None and not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # apply gates
    left = left_proj * left_gate    # [B,N,N,H]
    right = right_proj * right_gate # [B,N,N,H]

    # -----------------------------------------------------------------
    # 4. Core batched GEMM (left @ rightᵀ) via Triton
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # reshape to (batch = B*H, M = N, K = N)
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_t.view(B * H, N, N)   # (batch, M, K)
    right_flat = right_t.view(B * H, N, N) # (batch, K, N) – we will treat as transposed

    out_flat = torch.empty_like(left_flat)  # (batch, M, N), fp16

    # strides
    stride_a_m = left_flat.stride(1)   # stride for M dimension
    stride_a_k = left_flat.stride(2)   # stride for K dimension
    stride_b_k = right_flat.stride(2)  # K dimension (1) – transposed view
    stride_b_n = right_flat.stride(1)  # N dimension (N)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    stride_a_batch = left_flat.stride(0)
    stride_b_batch = right_flat.stride(0)
    stride_c_batch = out_flat.stride(0)

    # -------------------- Tiling parameters -------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_tiles_m * num_tiles_n, B * H)

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                # M, N, K (all = seq_len)
        stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 6. Output gate (element‑wise)
    # -----------------------------------------------------------------
    out_gate = F.linear(x, out_gate_w).sigmoid()   # [B, N, N, H]
    out = out * out_gate

    # -----------------------------------------------------------------
    # 7. Final projection to original dimension
    # -----------------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B, N, N, C]

    # Cast back to the dtype of the original input
    out = out.to(x.dtype)

    return out
