@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offset < size
    a = tl.load(a_ptr + offset, mask=mask)
    b = tl.load(b_ptr + offset, mask=mask)
    tl.store(out_ptr + offset, a * b, mask=mask)

def custom_kernel(data):
    x, mask, weights, config = data
    # Cast x to float32? It may be forward with float32.
    orig_dtype = x.dtype
    device = x.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # Cast weights to half
    dtype = torch.float16
    # Combined projection weight
    left_comb_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(device=device, dtype=dtype)
    right_comb_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(device=device, dtype=dtype)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype)

    # Projections and gates combined
    # left_comb: B,N,N,2*H
    left_comb = F.linear(x, left_comb_w)  # shape [B,N,N,2*H]
    right_comb = F.linear(x, right_comb_w)  # shape [B,N,N,2*H]

    # split
    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    if mask is not None and not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Reshape for batched matmul
    B, N, _, H = left.shape
    # permute to B,H,N,N
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Compute out via batched matmul: left @ right^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # shape [B*H,N,N]

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Hidden dim layernorm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Output gate
    out_gate = F.linear(x, out_gate_w).sigmoid()
    # Elementwise multiplication via Triton
    out = elemwise_mul(out, out_gate)

    # Final linear projection
    out = F.linear(out, to_out_w)

    # Cast back to original dtype
    out = out.to(orig_dtype)
    return out

"""Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The kernel fuses the cheap element‑wise part (output‑gate scaling) with a
custom Triton kernel while delegating the dominant O(N³·H) reduction to a
batched GEMM (torch.bmm) that runs on tensor‑cores.  All heavy arithmetic is
performed in fp16 and the result is cast back to the original dtype.

The entry point is `custom_kernel(data)`, where `data` is a tuple
(input_tensor, mask, weights, config) matching the AlphaFold3 TriMul module.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _elemwise_mul_kernel(
    a_ptr,
    b_ptr,
    out_ptr,
    size,
    BLOCK: tl.constexpr,
):
    """Element‑wise multiplication kernel (a * b)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton element‑wise multiplication kernel."""
    assert a.shape == b.shape, "Shapes must match."
    size = a.numel()
    out = torch.empty_like(a)
    # Choose a block size that is a power‑of‑two for good occupancy.
    BLOCK = 65536
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out

    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask : torch.Tensor of shape [B, N, N] or None
        - weights : dict of model parameters (float32)
        - config : dict containing at least "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # ------------------------------------------------------------------ #
    # Unpack inputs
    # ------------------------------------------------------------------ #
    x, mask, weights, config = data
    orig_dtype = x.dtype
    device = x.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (dim)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Work in fp16 for the compute‑intensive parts
    x = x.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 2. Cast module weights to fp16
    # ------------------------------------------------------------------ #
    dtype = torch.float16
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype)          # (2*H, C)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype)          # (2*H, C)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype)

    # ------------------------------------------------------------------ #
    # 3. Linear projections + gates (combined)
    # ------------------------------------------------------------------ #
    # Combined linear gives [B, N, N, 2*H]
    left_comb = F.linear(x, left_comb_w)
    right_comb = F.linear(x, right_comb_w)

    # Split projection and gate
    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # Optional masking (broadcast over hidden dimension)
    if mask is not None and not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Apply gates
    left = left_proj * left_gate      # [B, N, N, H]
    right = right_proj * right_gate   # [B, N, N, H]

    # ------------------------------------------------------------------ #
    # 4. Core O(N³·H) reduction using batched GEMM
    # ------------------------------------------------------------------ #
    B, N, _, H = left.shape

    # Move hidden dimension to the batch axis: [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch*hidden into a single GEMM batch dimension
    left_flat = left_perm.view(B * H, N, N)          # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)        # (B*H, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # Restore the original layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 5. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------ #
    # 6. Output gate (element‑wise) – Triton kernel
    # ------------------------------------------------------------------ #
    out_gate = F.linear(x, out_gate_w).sigmoid()   # [B, N, N, H]
    out = _elemwise_mul(out, out_gate)

    # ------------------------------------------------------------------ #
    # 7. Final projection back to input dimension
    # ------------------------------------------------------------------ #
    out = F.linear(out, to_out_w)   # [B, N, N, C]

    # Cast back to the original dtype of the input
    out = out.to(orig_dtype)
    return out
