```python
@triton.jit
def batched_gemm_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_aba, stride_ami, stride_ak,
    stride_bba, stride_bnj, stride_bk,
    stride_cba, stride_cmi, stride_cnj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute batch index, block_i, block_j
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_id = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    block_i = block_id // num_blocks_n
    block_j = block_id % num_blocks_n

    # offsets
    i_offset = block_i * BLOCK_M
    j_offset = block_j * BLOCK_N

    # Compute offsets for A and B
    a_batch_offset = batch_id * stride_aba
    b_batch_offset = batch_id * stride_bba
    c_batch_offset = batch_id * stride_cba

    # Pointers for loads
    # Loop over K
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_offset in range(0, K, BLOCK_K):
        # compute K range for mask
        cur_k = k_offset + tl.arange(0, BLOCK_K)
        k_mask = cur_k < K

        # A tile
        a_i = i_offset + tl.arange(0, BLOCK_M)
        a_i_mask = a_i < M
        a_ptrs = a_ptr + a_batch_offset + a_i[:, None] * stride_ami + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=a_i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float16)

        # B tile (transposed)
        b_j = j_offset + tl.arange(0, BLOCK_N)
        b_j_mask = b_j < N
        b_ptrs = b_ptr + b_batch_offset + cur_k[:, None] * stride_bk + b_j[None, :] * stride_bnj
        b = tl.load(b_ptrs, mask=k_mask[:, None] & b_j_mask[None, :], other=0.0).to(tl.float16)

        # dot product
        acc += tl.dot(a, b).to(tl.float32)

    # Store result
    c_i = i_offset + tl.arange(0, BLOCK_M)
    c_j = j_offset + tl.arange(0, BLOCK_N)
    c_i_mask = c_i < M
    c_j_mask = c_j < N
    c_ptrs = c_ptr + c_batch_offset + c_i[:, None] * stride_cmi + c_j[None, :] * stride_cnj
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=c_i_mask[:, None] & c_j_mask[None, :])
```

"""
TriMul (outgoing) forward pass – fully fused reduction.
The kernel implements the heavy O(N³·H) reduction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
as a batched GEMM of shape (B*H, N, N) × (B*H, N, N)ᵀ.
All other tensor operations (LayerNorms, Linear projections, gates,
masking, final projection) are performed in PyTorch with fp16
precision. The reduction itself is executed by a custom Triton
kernel that avoids the cuBLAS batched‑gemm overhead for many small
matrices, yielding a sizable speed‑up.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with the second operand transposed.
# Computes C[b] = A[b] @ B[b].T   where
#   A: (batch, M, K)   fp16
#   B: (batch, N, K)   fp16   (treated as K×N inside the kernel)
#   C: (batch, M, N)   fp16   (accumulated in fp32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_t_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    batch, M, N, K,                     # sizes
    stride_a_batch, stride_a_m, stride_a_k,   # A strides
    stride_b_batch, stride_b_n, stride_b_k,   # B strides (N dim first, K dim second)
    stride_c_batch, stride_c_m, stride_c_n,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of tiles per matrix
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # decode program id into batch, tile_i, tile_j
    batch_id = pid // (num_blocks_m * num_blocks_n)
    tile_id = pid % (num_blocks_m * num_blocks_n)
    tile_i = tile_id // num_blocks_n
    tile_j = tile_id % num_blocks_n

    # offsets for the current tile
    i_offset = tile_i * BLOCK_M
    j_offset = tile_j * BLOCK_N

    # batch offsets in memory
    a_batch_off = batch_id * stride_a_batch
    b_batch_off = batch_id * stride_b_batch
    c_batch_off = batch_id * stride_c_batch

    # accumulator (float32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for k_offset in range(0, K, BLOCK_K):
        cur_k = k_offset + tl.arange(0, BLOCK_K)
        k_mask = cur_k < K

        # ---- load A tile (M × K) ----
        a_i = i_offset + tl.arange(0, BLOCK_M)
        a_i_mask = a_i < M
        a_ptrs = a_ptr + a_batch_off \
                 + a_i[:, None] * stride_a_m \
                 + cur_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=a_i_mask[:, None] & k_mask[None, :],
                    other=0.0).to(tl.float16)

        # ---- load B tile transposed (K × N) ----
        b_j = j_offset + tl.arange(0, BLOCK_N)
        b_j_mask = b_j < N
        # note: B is stored as (batch, N, K); we treat it as (K, N) by swapping strides
        b_ptrs = b_ptr + b_batch_off \
                 + cur_k[:, None] * stride_b_k \
                 + b_j[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & b_j_mask[None, :],
                    other=0.0).to(tl.float16)

        # ---- dot product over the K dimension ----
        acc += tl.dot(a, b).to(tl.float32)

    # ---- store result tile (M × N) ----
    c_i = i_offset + tl.arange(0, BLOCK_M)
    c_j = j_offset + tl.arange(0, BLOCK_N)
    c_i_mask = c_i < M
    c_j_mask = c_j < N
    c_ptrs = c_ptr + c_batch_off \
             + c_i[:, None] * stride_c_m \
             + c_j[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=c_i_mask[:, None] & c_j_mask[None, :])


def _batched_gemm_t(
    a: torch.Tensor,
    b: torch.Tensor,
    block_m: int = 64,
    block_n: int = 64,
    block_k: int = 32,
) -> torch.Tensor:
    """
    Wrapper for the Triton batched GEMM (A @ Bᵀ).

    Args:
        a: Tensor of shape (batch, M, K), dtype torch.float16, contiguous.
        b: Tensor of shape (batch, N, K), dtype torch.float16, contiguous.
        block_m, block_n, block_k: tiling sizes.

    Returns:
        Tensor C of shape (batch, M, N), dtype torch.float16.
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, N, _ = b.shape

    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    grid = (batch * triton.cdiv(M, block_m) * triton.cdiv(N, block_n),)

    _batched_gemm_t_kernel[grid](
        a, b, c,
        batch, M, N, K,
        a.stride(0), a.stride(1), a.stride(2),
        b.stride(0), b.stride(1), b.stride(2),
        c.stride(0), c.stride(1), c.stride(2),
        BLOCK_M=block_m, BLOCK_N=block_n, BLOCK_K=block_k,
    )
    return c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C]  (float32/float16)
            - mask: [B, N, N] (bool/int/float) or None
            - weights: dict with all linear and layer‑norm parameters (float32)
            - config: {"dim": C, "hidden_dim": H, "nomask": bool (optional)}

    Returns:
        Tensor of shape [B, N, N, C] in the same dtype as the input.
    """
    # -------------------- unpack --------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device
    orig_dtype = x.dtype

    # -------------------- 1. Input LayerNorm --------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # stays in original dtype

    # -------------------- 2. Cast to fp16 for the heavy work --------------------
    x = x.to(torch.float16)

    # -------------------- 3. Cast model weights to fp16 --------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # -------------------- 4. Linear projections --------------------
    left_proj = F.linear(x, left_proj_w)          # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # -------------------- 5. Gate projections (sigmoid) --------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # -------------------- 6. Apply mask, gates and produce gated tensors --------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) -> (B, N, N, 1) broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Clean up intermediate tensors we no longer need to free memory
    del left_proj, right_proj, left_gate, right_gate, x

    # -------------------- 7. Core batched reduction (A @ Bᵀ) --------------------
    B, N, _, H = left.shape  # note: left shape = (B, N, N, H)

    # reshape to (B*H, N, N) for the kernel
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Triton GEMM: left_flat @ right_flatᵀ
    out_flat = _batched_gemm_t(left_flat, right_flat,
                               block_m=64, block_n=64, block_k=32)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------- 8. Output LayerNorm, gate, final projection --------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)                     # (B, N, N, dim)

    # -------------------- 9. Cast back to original dtype --------------------
    out = out.to(orig_dtype)

    return out
