"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate with
batched FP16 GEMM for the O(N³·H) reduction.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the provided
   `norm.{weight,bias}` (cast to FP16 for the compute‑heavy path).
2. Cast the normalized tensor to FP16.
3. Four linear projections (left/right and their gates) are performed
   with FP16 weights.  The gate tensors are passed through a sigmoid.
4. If a mask is supplied (and `nomask` is False) we fuse
   `proj * gate * mask` in a single Triton kernel (`_gate_mask_kernel`);
   otherwise we simply multiply the projection and gate tensors.
5. Core reduction:
   For each hidden channel `h` we compute a matrix product
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`,
   i.e. a batched GEMM of shape (B·H, N, N) @ (B·H, N, N)ᵀ.
   This is performed with `torch.matmul` on FP16 tensors, which
   maps to cuBLAS/Tensor‑core kernels.
6. Layer‑norm over the hidden dimension, multiplication by the output gate,
   and a final linear projection back to the original channel dimension.
7. Cast the result back to the input dtype (usually float32) and return.

Only the mask‑gate fusion is implemented in Triton; the large matrix
multiplication leverages the highly‑optimized cuBLAS path.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32   – total number of elements in proj/gate
    hidden_dim,        # i32   – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard
    mask = offs < total_elems

    # Index into the mask (broadcast over the hidden dimension)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    msk  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * msk
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 8192) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using Triton.
    All tensors must be on the same CUDA device, dtype torch.float16,
    and contiguous.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],               # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    # Cast weights to fp16 for the compute‑heavy part
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias   = weights["norm.bias"].to(torch.float16)

    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)   # fp16 compute

    # ----------------------- cast linear weights -------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------- linear projections ------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------- gates ----------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------- mask & gate fusion -----------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------- core O(N³·H) reduction ------------------------
    # Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # This is a batched GEMM: (B,H,N,N) @ (B,H,N,N)^T
    # Rearrange to (B*H, N, N)
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Merge B and H into a single batch dimension for torch.matmul
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # Batched matrix multiplication; note the transpose on the second operand
    out_mat = torch.matmul(left_mat, right_mat.transpose(-2, -1))  # (B*H, N, N)

    # Restore original dimensions: (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------ cast back --------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
