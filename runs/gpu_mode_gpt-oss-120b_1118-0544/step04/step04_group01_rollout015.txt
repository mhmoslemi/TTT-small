"""
TriMul (outgoing) forward pass – Triton‑fused mask‑gate.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) with learned weight/bias.
2. Cast to fp16 for all compute‑heavy ops.
3. Four linear projections (left/right + their gates) are performed with
   fp16 weights (torch.nn.functional.linear).  Gates are passed through a
   sigmoid.
4. If a mask is present (and not disabled by config["nomask"]), a small
   Triton kernel fuses `proj * gate * mask` for each of the two streams.
5. Core O(N³·H) reduction is expressed as a batched matrix multiplication:
   for each hidden dimension h we compute   out_h = L_h @ R_hᵀ,
   where L_h and R_h are the left/right tensors of shape (N,N).
   This is implemented with torch.bmm, which maps to cuBLAS/Tensor‑core.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output‑gate, and project back to the original channel dimension.
7. Cast the result back to the input dtype.

Only the mask‑gate fusion is performed in Triton; all matrix‑multiply heavy
operations use the highly‑optimized cuBLAS paths.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32   – total number of elements in proj/gate
    hidden_dim,        # i32   – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask index: each hidden dim shares the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using a Triton kernel.
    All tensors must be contiguous, on the same CUDA device, and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],        # hidden_dim (H)
        BLOCK_SIZE=block_size,
    )
    return out

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # Apply mask & gate (fused via Triton when possible)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B,N,N) -> fp16, contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·H) reduction using batched GEMM: out = left @ rightᵀ
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_ = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_ = right.permute(0, 3, 1, 2).reshape(B * H, N, N)
    # batched matrix multiply: (B*H, N, N) @ (B*H, N, N)ᵀ
    out = torch.bmm(left_, right_.transpose(1, 2))
    # reshape back to (B, N, N, H)
    out = out.reshape(B, H, N, N).permute(0, 2, 3, 1)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # Second LayerNorm over hidden dimension, then apply output gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
