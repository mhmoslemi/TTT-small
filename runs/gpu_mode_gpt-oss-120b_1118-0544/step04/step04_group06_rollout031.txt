@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    total_batches = B * H
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_batch = pid // (num_pid_m * num_pid_n)
    pid_rem = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator with zeros in fp32 (reduce precision errors)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # load A [batch, m, k]
        a_ptrs = left_ptr + pid_batch * stride_l_batch + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < N) & (offs_k[None, :] < N), other=0.0).to(tl.float32)

        # load B [batch, k, n]; note right is stored as (batch, k, n)
        b_ptrs = right_ptr + pid_batch * stride_r_batch + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_n
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < N) & (offs_n[None, :] < N), other=0.0).to(tl.float32)

        # block dot
        acc += tl.dot(a, b)

    # write back result
    out_ptrs = out_ptr + pid_batch * stride_o_batch + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    tl.store(out_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < N) & (offs_n[None, :] < N))

        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp)
    else:

        right = right_proj * right_gate

    B, N, _, H = left.shape
    left_perm = left.permute(0,3,1,2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0,3,2,1).contiguous() # (B, H, N, N) with k axis before j

    left_flat = left_perm.view(B*H, N, N)
    right_flat = right_perm.view(B*H, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Launch Triton kernel
    total_batches = B * H
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (total_batches * num_pid_m * num_pid_n,)

        out_flat,
        B, H, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape out back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()

    # Output layer norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # Apply out_gate
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, to_out_weight)

    # Cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out

def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,

    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    pid_batch = pid // (num_pid_m * num_pid_n)
    pid_rem = pid % (num_pid_m * num_pid_n)

"""
TriMul (outgoing) – Triton‑accelerated implementation.
The kernel fuses the heavy O(N³·H) reduction (left @ rightᵀ) across the
hidden dimension H by treating (B·H) as the batch axis and performing a
batched GEMM with tiled blocks.  A separate Triton kernel fuses the
element‑wise mask, projection and gate multiplications.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Kernel: fuse projection * gate * mask  (shape: B,N,N,H)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total_elems,)
    gate_ptr,          # fp16 * (total_elems,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (total_elems,)
    total_elems,       # i64  – total number of elements in proj/gate
    hidden_dim,        # i64  – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # each mask element repeats `hidden_dim` times in the flattened view
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mval = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mval
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse ``proj * gate * mask`` using a Triton kernel.
    All inputs must be fp16, contiguous, and broadcastable as described.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],               # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel: batched matrix multiplication for the TriMul reduction
#   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
#   treated as (B*H, N, N) @ (B*H, N, N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 * (B*H, N, N)
    right_ptr,         # fp16 * (B*H, N, N)   – already transposed (k, j)
    out_ptr,           # fp16 * (B*H, N, N)
    B, H, N,           # i64 scalars
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    total_batches = B * H
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    pid_batch = pid // (num_pid_m * num_pid_n)
    pid_rem   = pid % (num_pid_m * num_pid_n)
    pid_m     = pid_rem // num_pid_n
    pid_n     = pid_rem % num_pid_n

    # offsets for the output block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: (batch, m, k)
        a_ptrs = (left_ptr
                  + pid_batch * stride_l_batch
                  + offs_m[:, None] * stride_l_m
                  + offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # right (already transposed): (batch, k, n)
        b_ptrs = (right_ptr
                  + pid_batch * stride_r_batch
                  + offs_k[:, None] * stride_r_k
                  + offs_n[None, :] * stride_r_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)          # a,b fp16 → acc fp32

    # write back
    out_ptrs = (out_ptr
                + pid_batch * stride_o_batch
                + offs_m[:, None] * stride_o_m
                + offs_n[None, :] * stride_o_n)
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradients).
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Convert model weights to fp16 (same device)
    # ------------------------------------------------------------------
    def _fp16(w):
        return w.to(device=device, dtype=torch.float16)

    left_proj_w   = _fp16(weights["left_proj.weight"])
    right_proj_w  = _fp16(weights["right_proj.weight"])
    left_gate_w   = _fp16(weights["left_gate.weight"])
    right_gate_w  = _fp16(weights["right_gate.weight"])
    out_gate_w    = _fp16(weights["out_gate.weight"])
    to_out_norm_w = _fp16(weights["to_out_norm.weight"])
    to_out_norm_b = _fp16(weights["to_out_norm.bias"])
    to_out_w      = _fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 4) Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)    # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gate projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6) Fuse mask + gate + projection (optional)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core reduction: batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N) -> (batch, k, j)

    left_flat = left_perm.view(B * H, N, N)                    # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)                  # (B*H, N, N) (k, j)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # kernel launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    total_batches = B * H
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    total_programs = total_batches * grid_m * grid_n

    _tri_mul_kernel[(total_programs,)](
        left_flat,
        right_flat,
        out_flat,
        B, H, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)          # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)
    return out
