"""
custom_kernel: Triton‑accelerated outgoing TriMul.

Algorithm
---------
1. Layer‑norm the input tensor (float32) using the provided weights.
2. Cast to float16 for the heavy compute path.
3. Linear projections to hidden dimension (left, right) and linear gates
   (left_gate, right_gate, out_gate). Apply sigmoid to gate outputs.
4. Fuse element‑wise mask (if present) and gate multiplication with a tiny
   Triton kernel (in‑place on left/right).
5. Compute the contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   by reshaping left/right to (B*H, N, N) and using a batched GEMM
   (torch.bmm) which runs on cuBLAS/Tensor‑cores.
6. Apply a second LayerNorm on the hidden dimension, multiply by out_gate,
   and finally project back to the original channel dimension.
7. Return the result in the original dtype (float32).

The kernel only implements the mask‑and‑gate fusion; the O(N³·H) reduction
leverages highly‑optimized cuBLAS batched matmul.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------
# Triton kernel that multiplies left/right by their gates and
# optionally by a broadcast mask (mask shape [B, N, N]).
# ---------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr,               # fp16 * (B*N*N*H)
    right_ptr,              # fp16 * (B*N*N*H)
    left_gate_ptr,          # fp16 * (B*N*N*H)
    right_gate_ptr,         # fp16 * (B*N*N*H)
    mask_ptr,               # fp16 * (B*N*N)   (optional)
    total_elems,            # i32 scalar : B*N*N*H
    hidden_dim,             # i32 scalar : H
    HAS_MASK: tl.constexpr, # compile‑time bool
    BLOCK_SIZE: tl.constexpr # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask entry corresponds to a block of size hidden_dim in the flattened tensors
        mask_idx = offs // hidden_dim
        mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * lgate * mval
        right = right * rgate * mval
    else:
        left = left * lgate
        right = right * rgate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor of shape (B, N, N, C)
            mask         : torch.Tensor of shape (B, N, N) or None
            weights      : dict of model parameters (float32)
            config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on the input (float32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # Cast to half‑precision for compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Load linear weights (bias‑free) and cast to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B, N, N, H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # Fuse mask (if any) and gate multiplication via Triton kernel
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask broadcast to hidden dimension
        mask_fp = mask.to(device).to(torch.float16)
        # ensure contiguity for all tensors
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp = mask_fp.contiguous()

        # flatten
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_fp.view(-1)  # shape (B*N*N,)

        total = left_flat.numel()
        BLOCK = 32768  # reasonable block size for fp16

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – simple element‑wise multiplication with gates (still in‑place)
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·H) contraction via batched GEMM (cuBLAS / Tensor cores)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matmul: left @ rightᵀ  (equivalent to the Einstein sum)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output LayerNorm, gating, and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # Cast back to the original dtype (float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
