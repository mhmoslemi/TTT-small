@triton.jit
def batched_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Determine program id for output block (M,N)
    pid = tl.program_id(0)  # block index in M/N grid
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n
    # Also need batch id for B*H? We'll use program_id(1) for batch dimension:
    batch_pid = tl.program_id(1)  # each batch (combined B*H)
    # compute indices for batch:
    b = batch_pid // H
    h = batch_pid % H

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute pointers for left (A) and right (B^T)
    # left[b, h, i, k]
    a_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + offs_m[:, None] * stride_left_i + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    # right[b, h, j, k] -> we need B^T so treat as right[b, h, k, j]
    # we can load as right with k index varying in inner dimension, j outer.
    b_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + tl.arange(0, BLOCK_K)[:, None] * stride_right_k + offs_n[None, :] * stride_right_j

    # accumulator

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load A block
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < N), other=0.0)

        # advance pointers
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections (left/right) and gate projections (left/right/out) using
   torch.nn.functional.linear (bias‑free) – weights are cast to FP16.
4. Fuse mask, projection and gate in a small Triton kernel
   (proj * gate * mask).  When no mask is supplied a simple elementwise
   multiplication is used.
5. Core triangular update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication with a transpose on the right
   operand.  Implemented as a custom Triton kernel that processes
   (B·hidden_dim) independent GEMMs of size N×N.
6. Apply output LayerNorm on the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
7. Cast the result back to the original dtype (usually float32).
"""

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------
# Kernel 1: fuse (proj * gate * mask)  – mask broadcast over H
# --------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # int32  (B*N*N*H)
    hidden_dim,        # int32  (H)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)          # [0, BLOCK_SIZE)

    valid = offs < total_elems

    # each mask element is repeated `hidden_dim` times in the flattened proj/gate
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    proj, gate : (B, N, N, H)  (fp16)
    mask       : (B, N, N)     (fp16, 0/1)
    Returns proj * gate * mask  (broadcast over H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# --------------------------------------------------------------
# Kernel 2: batched triangular multiplication (A @ B^T)
# --------------------------------------------------------------
@triton.jit
def _batched_tri_mul_kernel(
    left_ptr,          # fp16 * (B*H*N*N,)
    right_ptr,         # fp16 * (B*H*N*N,)
    out_ptr,           # fp16 * (B*H*N*N,)
    B,                 # int32
    H,                 # int32
    N,                 # int32
    stride_left_b, stride_left_h, stride_left_i, stride_left_j,
    stride_right_b, stride_right_h, stride_right_i, stride_right_j,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # program IDs
    # ------------------------------------------------------------------
    batch_id = tl.program_id(1)               # 0 <= batch_id < B*H
    pid = tl.program_id(0)                    # blocks over M and N

    # decode batch dimensions
    b = batch_id // H
    h = batch_id % H

    # compute M/N block coordinates
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # main K-loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # pointers for A (left) – shape (B,H,N,N)  where dims are (b,h,i,k)
        a_ptrs = left_ptr + b * stride_left_b + h * stride_left_h \
                 + offs_m[:, None] * stride_left_i \
                 + k_range[None, :] * stride_left_j

        # pointers for B (right) – we need B_T : (b,h,k,j)
        b_ptrs = right_ptr + b * stride_right_b + h * stride_right_h \
                 + k_range[:, None] * stride_right_i \
                 + offs_n[None, :] * stride_right_j

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # write back result (FP16)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h \
               + offs_m[:, None] * stride_out_i \
               + offs_n[None, :] * stride_out_j

    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask         : torch.Tensor of shape [B, N, N] or None (0/1)
            - weights      : dict of model parameters (float32)
            - config       : dict containing:
                * "dim" (int)        – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim), dtype = input dtype

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy part
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear weights to FP16 (biases are None)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections (+ sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask and gating (fused via Triton if mask is present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core TriMul reduction: batched GEMM (A @ B^T)
    #    Shapes after permute: (B, H, N, N)
    # --------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left.shape
    out = torch.empty_like(left)

    # strides (in elements)
    stride_left_b, stride_left_h, stride_left_i, stride_left_j = left.stride()
    stride_right_b, stride_right_h, stride_right_i, stride_right_j = right.stride()
    stride_out_b, stride_out_h, stride_out_i, stride_out_j = out.stride()

    # block sizes – can be tuned; chosen to fit typical SM shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (num_pid_m * num_pid_n, B * H)

    _batched_tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        H,
        N,
        stride_left_b,
        stride_left_h,
        stride_left_i,
        stride_left_j,
        stride_right_b,
        stride_right_h,
        stride_right_i,
        stride_right_j,
        stride_out_b,
        stride_out_h,
        stride_out_i,
        stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8. Restore original layout (B, N, N, H)
    # --------------------------------------------------------------
    out = out.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # --------------------------------------------------------------
    # 9. Output LayerNorm, gating, final projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 10. Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(dtype_in)

    return out
```
