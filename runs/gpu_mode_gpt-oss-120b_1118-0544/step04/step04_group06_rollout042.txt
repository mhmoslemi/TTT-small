@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    # get program ids
    pid0 = tl.program_id(0)  # batch * Hblocks
    pid1 = tl.program_id(1)  # i block
    pid2 = tl.program_id(2)  # j block

    # compute batch and hidden block indices
    # total hidden blocks
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    b = pid0 // num_h_blocks
    h_block = pid0 % num_h_blocks

    i0 = pid1 * BLOCK_I
    j0 = pid2 * BLOCK_J
    h0 = h_block * BLOCK_H

    # compute offsets for each thread in tile
    # i and j offsets for each element in tile
    i = i0 + tl.arange(0, BLOCK_I)
    j = j0 + tl.arange(0, BLOCK_J)
    k = tl.arange(0, BLOCK_K)
    h = h0 + tl.arange(0, BLOCK_H)

    # bounds
    i_valid = i < N
    j_valid = j < N
    h_valid = h < H

    # create a mask for invalid i/j/h for final store
    # allocate accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over k dimension in tiles
    for k_start in range(0, N, BLOCK_K):
        cur_K = tl.minimum(BLOCK_K, N - k_start)
        # Compute offsets for left: left[b, i, k, h] (i varies, k range, h vary)
        # need to broadcast i dimension (size BLOCK_I) and k dim cur_K and h dim BLOCK_H
        # we load left as shape (BLOCK_I, cur_K, BLOCK_H)
        # compute pointer offsets
        # offset for left: b*stride_l_b + i*stride_l_i + (k_start + k)*stride_l_k + h*stride_l_h
        # Use broadcasting with tl.arange for i, k, h.
        # Use tl.load with shape (BLOCK_I, cur_K, BLOCK_H)

        # compute offsets:
        k_offsets = k_start + tl.arange(0, cur_K)

        # left pointer base for current batch
        left_base = left_ptr + b * stride_l_b

        # compute offsets using tl.broadcasted shape
        left_ptrs = left_base + \
            i[:, None, None] * stride_l_i + \
            k_offsets[None, :, None] * stride_l_k + \
            h[None, None, :] * stride_l_h

        left_tile = tl.load(left_ptrs, mask=(i_valid[:,None,None] & (k_offsets[None,:,None] < N) & h_valid[None,None,:]), other=0.0)

        # right tile: right[b, j, k, h]
        right_base = right_ptr + b * stride_r_b
        right_ptrs = right_base + \
            j[:, None, None] * stride_r_i + \
            k_offsets[None, :, None] * stride_r_k + \
            h[None, None, :] * stride_r_h

        right_tile = tl.load(right_ptrs, mask=(j_valid[:,None,None] & (k_offsets[None,:,None] < N) & h_valid[None,None,:]), other=0.0)

        # Now compute outer product across i and j per h: accumulate
        # left_tile shape (BLOCK_I, cur_K, BLOCK_H)
        # right_tile shape (BLOCK_J, cur_K, BLOCK_H)
        # We need for each h: sum_{k} left_tile[i,k,h] * right_tile[j,k,h]

        # Use loop over k dimension for each tile chunk
        for kk in range(cur_K):
            # left_vec shape (BLOCK_I, BLOCK_H)
            left_vec = left_tile[:, kk, :]  # (BLOCK_I, BLOCK_H)
            # right_vec shape (BLOCK_J, BLOCK_H)
            right_vec = right_tile[:, kk, :]  # (BLOCK_J, BLOCK_H)

            # Expand dims for broadcasting
            left_exp = left_vec[:, None, :]   # (BLOCK_I,1,BLOCK_H)
            right_exp = right_vec[None, :, :] # (1,BLOCK_J,BLOCK_H)

            acc += left_exp * right_exp

    # After loop, store accumulator to out
    # Cast to fp16
    out_ptrs = out_ptr + b * stride_o_b + i[:, None, None] * stride_o_i + j[None, :, None] * stride_o_j + h[None, None, :] * stride_o_h

    # store only valid entries
    mask_store = i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_store)

    left_ptr, right_ptr, out_ptr,
    B, N, H,

    stride_r_b, stride_r_i, stride_r_k, stride_r_h,

    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr

    pid1 = tl.program_id(1)  # i block
    pid2 = tl.program_id(2)  # j block

    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Apply mask if needed
    if mask is not None and not nomask:
        # ensure mask shape broadcast
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

    # Core reduction via Triton
    out_hidden = _tri_mul_reduce(left, right)   # shape (B,N,N,H) fp16

    # Convert to float32 for LayerNorm
    out_hidden_fp32 = out_hidden.to(torch.float32)

        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # Apply out_gate (cast to float32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection to dim
    out = torch.nn.functional.linear(out_norm, to_out_w)

    # Cast back to original dtype
    out = out.to(orig_dtype)

    return out

"""
TriMul (outgoing) forward pass – Triton fused reduction.

The kernel computes, for each batch b, hidden channel h and pair (i, j):
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

All LayerNorms, linear projections and gates are performed with PyTorch.
The O(N³·H) contraction is executed by a custom Triton kernel that
tiles the output (i, j) and the hidden dimension h, loads tiles of the
left and right tensors and performs a batched matrix multiplication via
broadcasted multiplication and reduction over the K dimension.
Masking and gating are fused before the kernel.
The final output is LayerNorm‑ed, gated, linearly projected back to the
original dimension and returned in the input dtype.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    # ------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------
    pid0 = tl.program_id(0)          # combined batch * hidden‑block index
    pid1 = tl.program_id(1)          # i‑tile
    pid2 = tl.program_id(2)          # j‑tile

    num_h_blocks = tl.cdiv(H, BLOCK_H)
    b = pid0 // num_h_blocks
    h_block = pid0 % num_h_blocks

    i0 = pid1 * BLOCK_I
    j0 = pid2 * BLOCK_J
    h0 = h_block * BLOCK_H

    # ------------------------------------------------------------
    # Index ranges
    # ------------------------------------------------------------
    i = i0 + tl.arange(0, BLOCK_I)
    j = j0 + tl.arange(0, BLOCK_J)
    k = tl.arange(0, BLOCK_K)
    h_range = h0 + tl.arange(0, BLOCK_H)

    i_mask = i < N
    j_mask = j < N
    h_mask = h_range < H

    # ------------------------------------------------------------
    # Accumulator (float32 for precision)
    # ------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------
    for k_offset in range(0, N, BLOCK_K):
        k_idx = k_offset + k
        k_valid = k_idx < N

        # ---- left tile (i, k) -------------------------------------------------
        left_ptrs = left_ptr + \
            b * stride_l_b + \
            i[:, None] * stride_l_i + \
            k_idx[None, :] * stride_l_k
        left_tile = tl.load(
            left_ptrs,
            mask=(i_mask[:, None] & k_valid[None, :]),
            other=0.0,
        )  # shape (BLOCK_I, BLOCK_K)

        # ---- right tile (k, j, h) -------------------------------------------
        right_ptrs = right_ptr + \
            b * stride_r_b + \
            k_idx[:, None, None] * stride_r_k + \
            j[None, :, None] * stride_r_i + \
            h_range[None, None, :] * stride_r_h
        right_mask = (
            k_valid[:, None, None] &          # K dimension
            j_mask[None, :, None] &           # J dimension
            h_mask[None, None, :]             # H dimension
        )
        right_tile = tl.load(
            right_ptrs,
            mask=right_mask,
            other=0.0,
        )  # shape (BLOCK_K, BLOCK_J, BLOCK_H)

        # ---- broadcast & multiply, reduce over K -------------------------------
        # left_tile: (BLOCK_I, BLOCK_K) -> (BLOCK_I, BLOCK_K, 1, 1)
        left_exp = left_tile[:, :, None, None]
        # right_tile: (BLOCK_K, BLOCK_J, BLOCK_H) -> (BLOCK_K, 1, BLOCK_J, BLOCK_H)
        right_exp = right_tile[:, None, :, :]

        prod = left_exp * right_exp                     # (BLOCK_I, BLOCK_K, BLOCK_J, BLOCK_H)
        acc += tl.sum(prod, axis=1)                     # reduce K -> (BLOCK_I, BLOCK_J, BLOCK_H)

    # ------------------------------------------------------------
    # Store result (fp16)
    # ------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_o_b + \
        i[:, None, None] * stride_o_i + \
        j[None, :, None] * stride_o_j + \
        h_range[None, None, :] * stride_o_h

    store_mask = (
        i_mask[:, None, None] &
        j_mask[None, :, None] &
        h_mask[None, None, :]
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=store_mask)


def _tri_mul_reduce(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Batched O(N³·H) reduction using the Triton kernel above."""
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # fp16, same device

    # strides (int64)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – chosen to fit in registers and shared memory
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_K = 32
    BLOCK_H = 8

    grid = (
        B * ((H + BLOCK_H - 1) // BLOCK_H),          # pid0
        (N + BLOCK_I - 1) // BLOCK_I,                # pid1
        (N + BLOCK_J - 1) // BLOCK_J,                # pid2
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    return out


def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C] (float32/float16)
            - mask: [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)  # cheap compute dtype

    # ------------------------------------------------------------------
    # 2. Cast model weights to fp16 where appropriate
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # LayerNorm after reduction stays in float32 for stability
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 3. Linear projections & gates (all fp16)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply mask and gates (mask is broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * left_gate * mask_fp
        right = right_proj * right_gate * mask_fp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 5. Core O(N³·H) reduction via Triton
    # ------------------------------------------------------------------
    out_hidden = _tri_mul_reduce(left, right)        # (B,N,N,hidden_dim), fp16

    # ------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out_hidden_fp32 = out_hidden.to(torch.float32)

    out_norm = torch.nn.functional.layer_norm(
        out_hidden_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out_norm = out_norm * out_gate.to(torch.float32)

    out = torch.nn.functional.linear(out_norm, to_out_w)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype and return
    # ------------------------------------------------------------------
    return out.to(orig_dtype)
