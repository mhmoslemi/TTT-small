    """
    Triton‑accelerated outgoing TriMul.
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Save original dtype
    orig_dtype = x.dtype

    device = x.device
    dtype = torch.float16

    # 2. Cast weights to FP16
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=dtype)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype)

    # 3. Projections & gates
    left = F.linear(x, left_proj_w)          # [B,N,N,H]
    right = F.linear(x, right_proj_w)        # [B,N,N,H]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    if mask is not None and not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    left = left * left_gate
    right = right * right_gate

    # 4. Batched GEMM via Triton (left @ right^T)
    B, N, _, H = left.shape

    # reshape to (B*H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_t.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)   # fp16

    # stride values
    stride_am = left_flat.stride(1)   # stride for M dimension (row)
    stride_ak = left_flat.stride(2)   # stride for K dimension (col)
    batch_stride_a = left_flat.stride(0)

    stride_bk = right_flat.stride(2)   # stride for K dimension after transposition (original last dim)
    stride_bn = right_flat.stride(1)   # stride for N dimension (original middle dim)
    batch_stride_b = right_flat.stride(0)

    stride_cm = out_flat.stride(1)   # M stride
    stride_cn = out_flat.stride(2)   # N stride
    batch_stride_c = out_flat.stride(0)

    M = N = K = N

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        M, N, K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )

    # reshape back: (B, H, N, N) -> (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 5. Hidden dim LayerNorm
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # 6. Multiply by out gate
    out = out * out_gate

    # 7. Final projection back to original dim
    out = F.linear(out, to_out_w)

    # Cast back to original dtype
    out = out.to(orig_dtype)

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) – Triton implementation.

The module computes
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where `left = (x @ W_left) * σ(x @ W_left_gate)` and
`right = (x @ W_right) * σ(x @ W_right_gate)`.
The heavy O(N³·H) reduction is performed as a batched GEMM
(left @ rightᵀ) using a custom Triton kernel.  All heavy arithmetic
uses FP16 with FP32 accumulation; the final result is cast back to the
original input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    batch_stride_a, batch_stride_b, batch_stride_c,   # batch strides
    stride_am, stride_ak,                     # A strides (M, K)
    stride_bk, stride_bn,                     # B strides (K, N) – B will be read transposed
    stride_cm, stride_cn,                     # C strides (M, N)
    M, N, K,                                  # matrix dimensions (all equal to seq_len)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  (A,B shape: batch×M×K, batch×K×N)."""
    pid = tl.program_id(0)          # output tile ID (i‑j)
    batch = tl.program_id(1)        # batch index = B * hidden_dim

    # tile grid
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch * batch_stride_a
    b_batch = b_ptr + batch * batch_stride_b
    c_batch = c_ptr + batch * batch_stride_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile (transposed view): (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)          # fp16·fp16 → fp32 accumulation

    # write back
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # --------------------------------------------------------------------- #
    # unpack inputs
    # --------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    orig_dtype = x.dtype
    device = x.device

    # --------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Work in FP16 for the heavy path
    x = x.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 2. Cast model weights to FP16
    # --------------------------------------------------------------------- #
    to_fp16 = lambda t: t.to(device=device, dtype=torch.float16)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w      = to_fp16(weights["to_out.weight"])

    # --------------------------------------------------------------------- #
    # 3. Linear projections + gates
    # --------------------------------------------------------------------- #
    left = F.linear(x, left_proj_w)     # [B, N, N, H]
    right = F.linear(x, right_proj_w)   # [B, N, N, H]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 4. Core O(N³·H) reduction using Triton (batched GEMM)
    # --------------------------------------------------------------------- #
    B, N, _, H = left.shape

    # reshape to (batch = B*H, M = N, K = N)
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_t.view(B * H, N, N)   # (batch, M, K)
    right_flat = right_t.view(B * H, N, N) # (batch, K, N) – we will treat it transposed

    out_flat = torch.empty_like(left_flat)  # (batch, M, N) FP16

    # strides for kernel
    stride_am = left_flat.stride(1)                # M‑stride
    stride_ak = left_flat.stride(2)                # K‑stride
    batch_stride_a = left_flat.stride(0)

    stride_bk = right_flat.stride(2)               # K‑stride after transposition
    stride_bn = right_flat.stride(1)               # N‑stride after transposition
    batch_stride_b = right_flat.stride(0)

    stride_cm = out_flat.stride(1)                 # M‑stride
    stride_cn = out_flat.stride(2)                 # N‑stride
    batch_stride_c = out_flat.stride(0)

    M = N = K = N

    # tiling parameters – tuned for N up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        M,
        N,
        K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )

    # restore layout (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 5. Hidden‑dim LayerNorm
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------------- #
    # 6. Apply output gate (element‑wise)
    # --------------------------------------------------------------------- #
    out = out * out_gate

    # --------------------------------------------------------------------- #
    # 7. Final projection back to original channel dimension
    # --------------------------------------------------------------------- #
    out = F.linear(out, to_out_w)   # [B, N, N, C]

    # Cast back to the original dtype of the input
    out = out.to(orig_dtype)

    return out
