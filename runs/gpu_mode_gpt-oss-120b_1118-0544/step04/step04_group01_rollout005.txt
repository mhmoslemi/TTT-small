"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate + FP16 batched GEMM.

Algorithm
---------
1. Layer‑normalize the 4‑D input (B,N,N,C) using the provided LayerNorm
   parameters.
2. Cast the normalized tensor to FP16 (half‑precision) for all heavy work.
3. Compute four linear projections (left/right and their gates) with FP16
   weights.  Gate tensors are passed through a sigmoid.
4. If a mask is provided and ``config["nomask"]`` is False, fuse
   ``proj * gate * mask`` with a small Triton kernel that multiplies the
   projection, gate and broadcasted mask in a single pass.
5. Core update: for each hidden channel d,
   ``out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]``.
   This is exactly a batched matrix multiplication:
   ``out_{b,d} = left_{b,d} @ right_{b,d}.T`` where each matrix is (N,N).
   We perform this with a single FP16 ``torch.bmm`` over a reshaped batch
   dimension (B * hidden_dim).  This leverages cuBLAS/Tensor‑cores and is
   far faster than ``torch.einsum``.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output‑gate, and project back to the original channel dimension with a
   final linear layer.
7. Cast the result back to the original input dtype (usually FP32).

Only the mask‑gate fusion is implemented in Triton; the rest uses
high‑performance PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32   – total number of elements in proj/gate
    hidden_dim,        # i32   – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # guard against out‑of‑bounds
    valid = offs < total_elems

    # each consecutive block of `hidden_dim` elements shares the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse ``proj * gate * mask`` using the Triton kernel above.
    All tensors must be on the same CUDA device, contiguous, and FP16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],               # hidden_dim (broadcast factor)
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict of model parameters (float32)
            - config: dict with at least:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------- LayerNorm (input) ------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for compute‑heavy path
    x = x.to(torch.float16)

    # ----------------------- FP16 weight conversion ----------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections ---------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------- gates ------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ----------------------- mask & gate fusion -------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------- core O(N³·H) reduction via batched GEMM -------
    # Reshape to (B*H, N, N) for batched matrix multiplication
    B, N, _, _ = left.shape
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # out_mat[bh] = left_mat[bh] @ right_mat[bh].T   -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------ cast back ---------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
