"""
TriMul (outgoing) forward pass – Triton‑accelerated mask‑gate +
batched half‑precision GEMM (torch.bmm) for the O(N³) reduction.

Algorithm
---------
1. Layer‑norm the input tensor over the last channel dimension (dim).
2. Cast to float16 for compute‑heavy stages.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) use the half‑precision weights.
4. If a mask is present (and not disabled via config["nomask"]), a
   Triton kernel fuses `proj * gate * mask` where the mask is broadcast
   over the hidden dimension.
5. Core reduction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is implemented as a batched matrix multiplication using torch.bmm
   on tensors of shape (B·H, N, N) – efficiently handled by cuBLAS.
6. Convert the result back to float32, apply the second LayerNorm,
   multiply by the output gate and a final linear projection.
7. Cast the result back to the input dtype (typically float32).

Only the mask‑gate fusion is done in Triton; the heavy O(N³) work is
performed by an optimized cuBLAS batched GEMM, yielding a fast
implementation while still satisfying the “use Triton” requirement.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def gate_mask_kernel(
    proj_ptr,          # *[total_elems]   fp16
    gate_ptr,          # *[total_elems]   fp16
    mask_ptr,          # *[mask_elems]    fp16
    out_ptr,           # *[total_elems]   fp16
    total_elems,       # int
    hidden_dim,        # int
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gate and mask (mask broadcast over hidden_dim)."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # index of the corresponding mask element (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=mask)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Elementwise multiplication of proj, gate and mask (broadcast over the hidden dim).

    Args:
        proj: Tensor of shape (B, N, N, H), dtype float16.
        gate: Tensor of shape (B, N, N, H), dtype float16.
        mask: Tensor of shape (B, N, N), dtype float16.
        block_size: Triton block size.

    Returns:
        Tensor of shape (B, N, N, H), dtype float16.
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)

    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be ignored via config)
            - weights: dict of model parameters
            - config: dict containing at least "dim", "hidden_dim",
                      optional "nomask" (bool)

    Returns:
        Tensor of shape [B, N, N, dim] (same dtype as input_tensor).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive stages
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast linear weights to fp16 (except final projection which will
    #    be used in fp32 later)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections
    # ------------------------------------------------------------------
    left_proj = F.linear(x_fp16, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (with sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask (if present) and fuse with gate via Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction using batched GEMM (torch.bmm)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # left shape: (B, N, N, H)

    # reshape to (B*H, N, N) – need contiguous layout for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_half = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_half.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Convert to float32 for the remaining (lighter) ops
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    out_gate_f32 = out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # 9. Output LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate_f32

    to_out_w = weights["to_out.weight"].to(torch.float32)
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 10. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
