import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0) # tile id
    batch = tl.program_id(1) # batch index

    # compute tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch base pointers
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B^(T) i.e., B as (K,N)
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # write back
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )
```

def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul) forward pass.

    The heavy O(N³·H) reduction is expressed as a batched matrix multiply
    `out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]`,
    which is implemented by a fused Triton kernel operating on half‑precision
    tensors and accumulating in fp32.
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------
    # 1. Input layer norm
    # -------------------------------------------------
    # x is [B,N,N,dim], torch.float32 or float16
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(x.dtype),
        bias=weights["norm.bias"].to(x.dtype),
    )
    # Cast to fp16 for the heavy part
    x = x.to(torch.float16)

    device = x.device
    dtype = torch.float16

    # -------------------------------------------------
    # 2. Linear projections + gates (combined)
    # -------------------------------------------------
    # Combine projection and gate weights to reduce reads
    left_comb_w = torch.cat(

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy O(N³·H) reduction is expressed as a batched matrix multiply

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to `out = left @ rightᵀ` for each hidden dimension `d`.
A fused Triton kernel performs the batched GEMM in fp16 with fp32 accumulation,
while all remaining linear/gating/normalisation steps are carried out in PyTorch.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton batched GEMM kernel (computes C = A @ B where B is already transposed)
# --------------------------------------------------------------------------- #
@triton.jit
def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (A: M×K, B: K×N -> C: M×N)
    stride_am, stride_ak,              # strides for A (row, col)
    stride_bk, stride_bn,              # strides for B (row=K, col=N)
    stride_cm, stride_cn,              # strides for C (row, col)
    stride_ab, stride_bb, stride_cb,   # strides for the batch dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)      # tile id within a batch
    batch = tl.program_id(1)    # batch index (B * hidden_dim)

    # --------------------------------------------------------------------- #
    # Compute tile coordinates
    # --------------------------------------------------------------------- #
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------------- #
    # Base pointers for this batch
    # --------------------------------------------------------------------- #
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------------- #
    # Loop over K dimension in BLOCK_K chunks
    # --------------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B is stored as (K, N) – we load it directly (no explicit transpose)
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)          # fp16·fp16 → fp32 accumulation

    # --------------------------------------------------------------------- #
    # Write result
    # --------------------------------------------------------------------- #
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[:, None],
    )

# Entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int)          – input channel dimension C
                * "hidden_dim" (int)   – hidden dimension H
                * "nomask" (bool)      – if True, ignore mask (optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input).
    """
    # ------------------------------------------------------------------- #
    # Unpack
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    orig_dtype = x.dtype
    device = x.device

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(x.dtype),
        bias=weights["norm.bias"].to(x.dtype),
    )
    # Cast to half precision for the compute‑heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 2. Linear projections + gates (combined to reduce reads)
    # ------------------------------------------------------------------- #
    # concatenate proj and gate weights: [2*H, C]
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)

    # combined linear
    left_comb = F.linear(x, left_comb_w)    # [B, N, N, 2*H]
    right_comb = F.linear(x, right_comb_w)  # [B, N, N, 2*H]

    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # Apply mask (if any) *before* gating, exactly as the reference implementation
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Gated projections
    left = left_proj * left_gate   # [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------- #
    # 3. Core reduction: batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape  # (B, N, N, H)

    # Bring hidden dimension to the batch axis
    left_t = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Transpose right for the “rightᵀ” in the einsum
    right_t_T = right_t.permute(0, 1, 3, 2).contiguous()  # [B, H, N, N] (K × N)

    # Flatten batch & hidden dim -> (B*H, N, N)
    left_flat = left_t.view(B * H, N, N)
    right_flat = right_t_T.view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)   # fp16 output buffer

    # Strides (in elements) for the kernel
    stride_a_batch = left_flat.stride(0)     # batch stride
    stride_a_m = left_flat.stride(1)         # row stride (M)
    stride_a_k = left_flat.stride(2)         # col stride (K)

    stride_b_batch = right_flat.stride(0)    # batch stride
    stride_b_k = right_flat.stride(1)        # K stride
    stride_b_n = right_flat.stride(2)        # N stride

    stride_c_batch = out_flat.stride(0)      # batch stride
    stride_c_m = out_flat.stride(1)          # M stride
    stride_c_n = out_flat.stride(2)          # N stride

    # Block sizes – tuned for seq‑len up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (tiles over M×N, batch dimension)
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                     # M, N, K (all equal to seq_len)
        stride_a_m,
        stride_a_k,
        stride_b_k,
        stride_b_n,
        stride_c_m,
        stride_c_n,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 4. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ------------------------------------------------------------------- #
    # 5. Output gate and final projection
    # ------------------------------------------------------------------- #
    out_gate = F.linear(
        x, weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    ).sigmoid()                     # [B, N, N, H]

    out = out * out_gate

    out = F.linear(
        out, weights["to_out.weight"].to(device=device, dtype=torch.float16)
    )                               # [B, N, N, C]

    # ------------------------------------------------------------------- #
    # 6. Cast back to the original dtype
    # ------------------------------------------------------------------- #
    out = out.to(orig_dtype)
    return out
```
