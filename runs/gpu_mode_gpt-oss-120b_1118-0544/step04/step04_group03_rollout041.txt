@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(1)
    pid = tl.program_id(0)

    # compute tile coordinates
    #tile row
    pid_m = pid // tl.cdiv(N, BLOCK_N)
    pid_n = pid % tl.cdiv(N, BLOCK_N)
    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # pointers offset for this batch
    a_batch_ptr = a_ptr + batch_id * stride_batch_a
    b_batch_ptr = b_ptr + batch_id * stride_batch_b
    c_batch_ptr = c_ptr + batch_id * stride_batch_c

    # read and compute
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        a = tl.load(a_batch_ptr + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak), mask=(offs_m[:, None] < M) & mask_k[None, :], other=0.0)
        b = tl.load(b_batch_ptr + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn), mask=mask_k[:, None] & (offs_n[None, :] < N), other=0.0)

        acc += tl.dot(a, b)   # a,b are half
    c = acc.to(tl.float16)
    tl.store(c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn), c, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

def custom_kernel(data):
    x, mask, weights, config = data
    bs, N, _, C = x.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Cast to half
    x_h = x_norm.to(torch.float16)

    # Load weights in half
    w_left_comb = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(device=x.device, dtype=torch.float16)
    w_right_comb = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(device=x.device, dtype=torch.float16)
    w_out_gate = weights["out_gate.weight"].to(device=x.device, dtype=torch.float16)
    w_to_out_norm_weight = weights["to_out_norm.weight"].to(device=x.device, dtype=torch.float16)
    w_to_out_norm_bias = weights["to_out_norm.bias"].to(device=x.device, dtype=torch.float16)
    w_to_out = weights["to_out.weight"].to(device=x.device, dtype=torch.float16)

    # Linear combined
    left_comb = F.linear(x_h, w_left_comb)  # shape [B,N,N,2*H]
    right_comb = F.linear(x_h, w_right_comb)

    # Split
    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    if mask is not None and not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left = left_proj * left_gate
    right = right_proj * right_gate

    # reshape for batched matmul
    # shape B,N,N,H -> B,H,N,N -> flatten batch H
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_t.shape

    left_flat = left_t.view(B*H, N, N)
    right_flat = right_t.view(B*H, N, N)

    # compute batched matmul
    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Hidden layernorm
    out = F.layer_norm(out, (hidden_dim,), weight=w_to_out_norm_weight, bias=w_to_out_norm_bias)

    # Output gate
    out_gate = F.linear(x_h, w_out_gate).sigmoid()
    # elementwise multiply . use Triton kernel
    out = _elemwise_mul(out, out_gate)

    # Final projection
    out = F.linear(out, w_to_out)

    # Cast back to original dtype (original input dtype)
    out = out.to(x.dtype)

    return out

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑normalize the input tensor.
2. Fuse left/right linear projections and their gates into a single matmul per
   pathway (weight shape (2·H, C)).  Split the result into projection and gate,
   apply the optional mask and the sigmoid gate.
3. Reshape the gated tensors to (B·H, N, N) and perform a batched matrix
   multiplication   out = left @ rightᵀ   using torch.bmm (Tensor‑Core fast).
4. Layer‑normalize the hidden dimension, multiply by the output gate
   (element‑wise Triton kernel), and project back to the original channel
   dimension.
5. Cast the result back to the input dtype.

At least one operation (the element‑wise multiplication) is executed in a
custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------- #
# Triton kernel for element‑wise multiplication (used for out‑gate)
# --------------------------------------------------------------- #
@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton kernel for element‑wise multiplication."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication."
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out


# --------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask        : torch.Tensor of shape [B, N, N] (or None)
            - weights     : dict of model parameters (float32)
            - config      : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as the input tensor).
    """
    # ----------------------------------------------------------- #
    # 1. unpack inputs
    # ----------------------------------------------------------- #
    x, mask, weights, config = data
    orig_dtype = x.dtype                     # keep original dtype for the output
    device = x.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------- #
    # 2. Input LayerNorm (float32, then cast to fp16 for the heavy path)
    # ----------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.to(torch.float16)                # fp16 for the rest of the computation

    # ----------------------------------------------------------- #
    # 3. Load model weights in fp16
    # ----------------------------------------------------------- #
    w_left_comb = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)           # (2·H, C)

    w_right_comb = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)           # (2·H, C)

    w_out_gate = weights["out_gate.weight"].to(device=device, dtype=torch.float16)  # (H, C)
    w_to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    w_to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    w_to_out = weights["to_out.weight"].to(device=device, dtype=torch.float16)      # (C, H)

    # ----------------------------------------------------------- #
    # 4. Linear projections + gates (fused proj+gate)
    # ----------------------------------------------------------- #
    # shape after linear: [B, N, N, 2·H]
    left_comb = F.linear(x_h, w_left_comb)
    right_comb = F.linear(x_h, w_right_comb)

    # split projection / gate
    left_proj = left_comb[..., :hidden_dim]                # [B, N, N, H]
    left_gate = left_comb[..., hidden_dim:].sigmoid()      # [B, N, N, H]

    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # optional mask (broadcast over the hidden dimension)
    if mask is not None and not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # apply gates
    left = left_proj * left_gate                         # [B, N, N, H]
    right = right_proj * right_gate

    # ----------------------------------------------------------- #
    # 5. Core O(N³·H) reduction – batched matrix multiplication
    #    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # ----------------------------------------------------------- #
    # reshape to (B·H, N, N) for torch.bmm
    left_t = left.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]

    B, H, N, _ = left_t.shape

    left_flat = left_t.view(B * H, N, N)                # (batch, M, K)
    right_flat = right_t.view(B * H, N, N)              # (batch, K, N)

    # batched matmul: (B·H, N, N) @ (B·H, N, N)ᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------- #
    # 6. Hidden‑dim LayerNorm
    # ----------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm_w,
        bias=w_to_out_norm_b,
    )

    # ----------------------------------------------------------- #
    # 7. Output gate (element‑wise) – Triton kernel
    # ----------------------------------------------------------- #
    out_gate = F.linear(x_h, w_out_gate).sigmoid()      # [B, N, N, H]
    out = _elemwise_mul(out, out_gate)

    # ----------------------------------------------------------- #
    # 8. Final projection back to the original channel dimension
    # ----------------------------------------------------------- #
    out = F.linear(out, w_to_out)                       # [B, N, N, C]

    # ----------------------------------------------------------- #
    # 9. Cast back to the original dtype of the input tensor
    # ----------------------------------------------------------- #
    out = out.to(orig_dtype)

    return out
