@triton.jit
def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    mask_ptr,               # optional; can be null pointer (use 0 offset)
    B, M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    stride_mask_batch, stride_mask_m, stride_mask_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    batch_id = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # compute offsets
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute masks for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for out (initialised with zeros)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointer calculations
        left_ptrs = left_ptr + batch_id * stride_left_batch + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        right_ptrs = right_ptr + batch_id * stride_right_batch + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n  # note transpose

        # load
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # apply mask if present
        if HAS_MASK:
            # mask for left: mask_ptr[batch_id, i, k]
            mask_left_ptrs = mask_ptr + batch_id * stride_mask_batch + offs_m[:, None] * stride_mask_m + offs_k[None, :] * stride_mask_n
            mask_left = tl.load(mask_left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=1.0)
            left = left * mask_left

            # mask for right: mask_ptr[batch_id, j, k] but note we loaded right using transposed indices: for each (k, j) we need mask at (j, k)
            mask_right_ptrs = mask_ptr + batch_id * stride_mask_batch + offs_n[:, None] * stride_mask_m + offs_k[None, :] * stride_mask_n
            mask_right = tl.load(mask_right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=1.0)
            right = right * mask_right

        acc += tl.dot(left, right)  # left: (BM, BK) right: (BK, BN) -> (BM, BN)

    # store
    out_ptrs = out_ptr + batch_id * stride_out_batch + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # 1. LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16
    x = x.to(torch.float16)

    # Load weights and cast
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # 2. Linear projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # 3. Gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # 4. Apply gates (elementwise)
    left = left_proj * left_gate   # (B,N,N,H)
    right = right_proj * right_gate

    B, N, _, _ = left.shape

    # 5. Flatten hidden dim into batch dimension for matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous()
    # reshape to (B*H, N, N)
    left_flat = left_flat.view(B * hidden_dim, N, N)
    right_flat = right_flat.view(B * hidden_dim, N, N)

    # 6. Prepare mask if needed
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()  # (B,N,N)
        has_mask = True
    else:
        # dummy mask
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask = False

    # 7. Allocate output tensor
    out_flat = torch.empty_like(left_flat)   # (B*H,N,N) fp16

    # Compute strides
    left_stride_b, left_stride_m, left_stride_k = left_flat.stride()
    right_stride_b, right_stride_n, right_stride_k = right_flat.stride()
    out_stride_b, out_stride_m, out_stride_n = out_flat.stride()
    if has_mask:
        mask_stride_b, mask_stride_m, mask_stride_n = mask_fp16.stride()
    else:
        mask_stride_b = mask_stride_m = mask_stride_n = 0

    # Kernel launch
    import triton
    import triton.language as tl

    # Define kernel (we will define below, inside the file)
    # Blum.

    # Launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute grid size
    grid = (B * hidden_dim,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    # call kernel with HAS_MASK = 1 or 0
    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        mask_fp16,
        B, N, N, N,
        left_stride_b, left_stride_m, left_stride_k,
        right_stride_b, right_stride_n, right_stride_k,
        out_stride_b, out_stride_m, out_stride_n,
        mask_stride_b, mask_stride_m, mask_stride_n,
        hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        HAS_MASK=1 if has_mask else 0,
    )
    # 8. Reshape back to (B,N,N,H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 9. Output layernorm and projection
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    out = out.to(orig_dtype)

    return out
```

```python
@triton.jit
def _batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr, mask_ptr,
    B, M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    stride_mask_batch, stride_mask_m, stride_mask_n,
    hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute batch and hidden offset for mask
    if HAS_MASK:
        batch_id = pid // hidden_dim
        # hidden_id = pid % hidden_dim  # we don't need hidden_id for mask
    else:
        batch_id = pid

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load left tile (M x K)
        left_ptrs = left_ptr + pid * stride_left_batch + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load right tile (K x N) - note transposed
        right_ptrs = right_ptr + pid * stride_right_batch + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        if HAS_MASK:
            # left mask (M x K)
            mask_left_ptrs = mask_ptr + batch_id * stride_mask_batch + offs_m[:, None] * stride_mask_m + offs_k[None, :] * stride_mask_n
            mask_left = tl.load(mask_left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=1.0)
            left = left * mask_left

            # right mask (N x K) but we need mask at [batch_id, j, k]
            mask_right_ptrs = mask_ptr + batch_id * stride_mask_batch + offs_n[:, None] * stride_mask_m + offs_k[None, :] * stride_mask_n
            mask_right = tl.load(mask_right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=1.0)
            right = right * mask_right

        # dot product
        acc += tl.dot(left, right)  # (BLOCK_M, BLOCK_N)

    # store results
    out_ptrs = out_ptr + pid * stride_out_batch + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated fusion.

Algorithm
---------
1. Layer‐norm the input tensor (B,N,N,C) with provided LN parameters.
2. Cast to float16 for heavy compute.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) using FP16 weights; apply sigmoid on gates.
4. Fuse projection with gate (elementwise product). The mask (if present)
   is *not* materialised; it is applied on‑the‑fly inside the batched
   matrix‑multiply kernel.
5. Core O(N³·H) reduction:
   For each hidden slice d, compute  out[b,i,j,d] = Σₖ left[b,i,k,d] *
   right[b,j,k,d]  by a custom Triton kernel that performs a batched
   GEMM (left @ rightᵀ) with optional mask multiplication.
6. Reshape the result back to (B,N,N,H), apply a second LayerNorm over H,
   multiply by out_gate and linearly project back to the original channel
   dimension C.
7. Restore the original dtype (usually float32).

Only the reduction (step 5) is implemented in Triton; all other ops use
highly‑optimised PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _batched_matmul_kernel(
    left_ptr,               # fp16* (B*H, N, N)
    right_ptr,              # fp16* (B*H, N, N)
    out_ptr,                # fp16* (B*H, N, N)
    mask_ptr,               # fp16* (B, N, N)  (broadcasted over H)
    B, M, N, K,             # int64 scalars (B = batch, M=N=seq_len, K=N)
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    stride_mask_batch, stride_mask_m, stride_mask_n,
    hidden_dim,             # int64 – size of hidden dimension H
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    # program ids
    batch_id = tl.program_id(0)      # 0 <= batch_id < B*H
    pid_m = tl.program_id(1)         # block row
    pid_n = tl.program_id(2)         # block col

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # when mask is used we need the original batch index (without H)
    if HAS_MASK:
        batch_no_h = batch_id // hidden_dim
    else:
        batch_no_h = batch_id

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ----- load left tile (M x K) -----
        left_ptrs = (
            left_ptr
            + batch_id * stride_left_batch
            + offs_m[:, None] * stride_left_m
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # ----- load right tile (K x N) – note the transpose -----
        right_ptrs = (
            right_ptr
            + batch_id * stride_right_batch
            + offs_k[:, None] * stride_right_k
            + offs_n[None, :] * stride_right_n
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # ----- optional mask multiplication -----
        if HAS_MASK:
            # mask for left[i,k]
            mask_left_ptrs = (
                mask_ptr
                + batch_no_h * stride_mask_batch
                + offs_m[:, None] * stride_mask_m
                + offs_k[None, :] * stride_mask_n
            )
            mask_left = tl.load(
                mask_left_ptrs,
                mask=mask_m[:, None] & mask_k[None, :],
                other=1.0,
            )
            left = left * mask_left

            # mask for right[j,k]  (note j <-> n)
            mask_right_ptrs = (
                mask_ptr
                + batch_no_h * stride_mask_batch
                + offs_n[:, None] * stride_mask_m
                + offs_k[None, :] * stride_mask_n
            )
            mask_right = tl.load(
                mask_right_ptrs,
                mask=mask_n[:, None] & mask_k[None, :],
                other=1.0,
            )
            right = right * mask_right

        # ----- matrix multiply and accumulate -----
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N) in FP32

    # store result
    out_ptrs = (
        out_ptr
        + batch_id * stride_out_batch
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict containing all module parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to FP16 (ensure same device)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections and gates (all in FP16)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)      # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse projection with gate (mask will be applied inside GEMM)
    # ------------------------------------------------------------------
    left = left_proj * left_gate       # (B,N,N,H)
    right = right_proj * right_gate    # (B,N,N,H)

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 6. Fold hidden dimension into the batch dimension for GEMM
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (B*H,N,N)
    right = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # 7. Prepare mask (optional)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()   # (B,N,N)
        has_mask = True
    else:
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)
        has_mask = False

    # ------------------------------------------------------------------
    # 8. Allocate output buffer
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # (B*H,N,N), fp16

    # ------------------------------------------------------------------
    # 9. Strides (in elements, not bytes)
    # ------------------------------------------------------------------
    stride_left_b, stride_left_m, stride_left_k = left.stride()
    stride_right_b, stride_right_n, stride_right_k = right.stride()
    stride_out_b, stride_out_m, stride_out_n = out.stride()
    if has_mask:
        stride_mask_b, stride_mask_m, stride_mask_n = mask_fp16.stride()
    else:
        stride_mask_b = stride_mask_m = stride_mask_n = 0

    # ------------------------------------------------------------------
    # 10. Launch Triton batched GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        left,
        right,
        out,
        mask_fp16,
        B,
        N,
        N,
        N,
        stride_left_b,
        stride_left_m,
        stride_left_k,
        stride_right_b,
        stride_right_n,
        stride_right_k,
        stride_out_b,
        stride_out_m,
        stride_out_n,
        stride_mask_b,
        stride_mask_m,
        stride_mask_n,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HAS_MASK=1 if has_mask else 0,
    )

    # ------------------------------------------------------------------
    # 11. Reshape back to (B,N,N,H)
    # ------------------------------------------------------------------
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 12. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 13. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
