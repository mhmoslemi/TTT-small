"""
TriMul (outgoing) forward pass – Triton‑accelerated mask‑gate fusion.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for compute‑intensive operations.
3. Linear projections:
      left  = x @ W_left          (B,N,N,H)
      right = x @ W_right
4. Gate projections (sigmoid):
      left_gate  = σ(x @ W_left_gate)   (B,N,N,H)
      right_gate = σ(x @ W_right_gate)
      out_gate   = σ(x @ W_out_gate)    (B,N,N,H)
5. Fuse mask, gating and projection in a single Triton kernel
   (applied only when a mask is provided).  The kernel computes
        out = proj * gate * mask   (mask broadcast over H).
   If no mask is present we fall back to a pure PyTorch elementwise
   multiplication.
6. Core reduction (Einstein sum) using a batched matrix multiplication:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented with `torch.einsum` (which calls cuBLAS).
7. Output LayerNorm over the hidden dimension H.
8. Multiply by `out_gate` and apply the final linear projection back to `dim`.
9. Cast the result back to the input dtype (usually float32).

The kernel in step 5 is required to satisfy the “must use Triton” rule.
All heavy linear algebra is performed in fp16 for speed, while the
final output is returned in the original dtype.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: elementwise proj * gate * mask   (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # int: total number of elements in proj/gate (B*N*N*H)
    hidden_dim,        # int: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)          # [0, BLOCK_SIZE)

    # guard for the last (partial) block
    mask = offs < total_elems

    # each hidden entry shares the same mask element
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse projection, sigmoid‑gate and optional mask using Triton.
    All tensors must be fp16 and contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)  (binary 0/1)
    Returns     proj * gate * mask   (broadcast over H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/0‑1) or None
            - weights: dict of parameters (float32)
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * optional "nomask" (bool) – if True, ignore mask even if provided
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim` (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim) – same dtype as input_tensor

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to fp16 (biases are None except to_out_norm)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, H)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid (still fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Fuse mask, projection and gate (Triton kernel) if mask is present
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        # Apply fusion for left and right separately
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        # No mask – simple elementwise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction:   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with torch.einsum (efficient batched GEMM)
    # ------------------------------------------------------------------
    # left, right shape: (B, N, N, H)
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm over hidden dimension H (fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # Multiply by out_gate
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 10. Cast back to the original dtype (typically float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
