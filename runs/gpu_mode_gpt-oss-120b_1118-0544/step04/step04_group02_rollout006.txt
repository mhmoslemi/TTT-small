"""
Outgoing TriMul forward pass (AlphaFold3 style) with Triton acceleration.

Algorithm
---------
1. Layer‑norm the 4‑D input (B, N, N, C) using the provided LN weights.
2. Cast to FP16 for the compute‑heavy stages.
3. Linear projections:
   left_proj  = x @ left_proj_weight   (B,N,N,H)
   right_proj = x @ right_proj_weight  (B,N,N,H)
4. Gate projections + sigmoid:
   left_gate  = sigmoid(x @ left_gate_weight)   (B,N,N,H)
   right_gate = sigmoid(x @ right_gate_weight)  (B,N,N,H)
   out_gate   = sigmoid(x @ out_gate_weight)    (B,N,N,H)
5. Fuse mask, projection and gate with a small Triton kernel
   (if a mask is present).  The mask is broadcast over the hidden
   dimension H.
6. Core O(N³·H) reduction:
   For each batch and hidden channel we compute a GEMM:
       out[b, h, i, j] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   This is a batched matrix multiplication of shape (B*H, N, N) against
   the transpose of the same shape, implemented by `tri_mul_matmul_kernel`.
   FP16 arithmetic with FP32 accumulation is used.
7. Reshape back to (B, N, N, H), apply a second LayerNorm over H,
   multiply by `out_gate` and project back to the original channel
   dimension with a final linear layer.
8. Cast the result back to the input dtype (typically float32).

The implementation mixes high‑level PyTorch (layer‑norm, linear, sigmoid)
with two Triton kernels:
* `gate_mask_kernel`   – elementwise `proj * gate * mask`.
* `tri_mul_matmul_kernel` – batched GEMM for the O(N³) reduction.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask  (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    proj_ptr,          # *[total_elems]   fp16
    gate_ptr,          # *[total_elems]   fp16
    mask_ptr,          # *[mask_elems]    fp16
    out_ptr,           # *[total_elems]   fp16
    total_elems,       # int
    hidden_dim,        # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # map flat index to corresponding mask element (broadcast over hidden dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(out_ptr + offs, proj * gate * mask, mask=valid)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor,
                        block_size: int = 1024) -> torch.Tensor:
    """Fuse projection, gate and mask (mask broadcast over hidden dim)."""
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) core of TriMul
# Computes: out = left @ rightᵀ   with shapes (B*H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to flattened (B*H,N,N) tensors
    batch, M, N, K,                     # integer sizes (batch = B*H, M=N, N=N, K=N)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch dimension
    pid_m = tl.program_id(0)   # tile over M
    pid_n = tl.program_id(1)   # tile over N

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_base = a_ptr + pid_b * stride_a_batch
    b_base = b_ptr + pid_b * stride_b_batch
    c_base = c_ptr + pid_b * stride_c_batch

    # boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block (M x K)
        a_offset = offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(
            a_base + a_offset,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B block transposed (K x N)
        b_offset = offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(
            b_base + b_offset,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)   # FP32 accumulation

    # Write back
    c_offset = offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_base + c_offset, acc.to(tl.float16), mask=mask)

            left = left_proj * left_gate
            right = right_proj * right_gate

        # --------------------------------------------------------------
        # 7. Core O(N³·H) reduction – batched GEMM on flattened (B*H,N,N)
        # --------------------------------------------------------------
        B, N, _, H = left.shape  # (B, N, N, H)

        # reshape to (B*H, N, N) – need contiguous layout for the kernel
        left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
        right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

        out_flat = torch.empty_like(left_flat)   # fp16

        # strides in elements
        stride_a_batch = left_flat.stride(0)
        stride_a_m     = left_flat.stride(1)
        stride_a_k     = left_flat.stride(2)

        stride_b_batch = right_flat.stride(0)
        stride_b_n     = right_flat.stride(1)   # row index (j)
        stride_b_k     = right_flat.stride(2)   # column index (k)

        stride_c_batch = out_flat.stride(0)
        stride_c_m     = out_flat.stride(1)
        stride_c_n     = out_flat.stride(2)

        # Choose block sizes (tuned for typical N values)
        if N <= 256:
            BLOCK_M = 64
            BLOCK_N = 64
        else:
            BLOCK_M = 128
            BLOCK_N = 128
        BLOCK_K = 32

        grid = (
            math.ceil(N / BLOCK_M),
            math.ceil(N / BLOCK_N),
            B * H,
        )

        tri_mul_matmul_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            B * H,
            N,
            N,
            N,
            stride_a_batch,
            stride_a_m,
            stride_a_k,
            stride_b_batch,
            stride_b_n,
            stride_b_k,
            stride_c_batch,
            stride_c_m,
            stride_c_n,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

        # reshape back to (B, N, N, H)
        out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

        # --------------------------------------------------------------
        # 8. Output LayerNorm, output‑gate and final projection
        # --------------------------------------------------------------
        out = F.layer_norm(
            out,
            (H,),
            weight=to_out_norm_w,
            bias=to_out_norm_b,
        )
        out = out * out_gate
        out = F.linear(out, to_out_w)   # (B, N, N, dim)

        # --------------------------------------------------------------
        # 9. Cast back to original dtype
        # --------------------------------------------------------------
        out = out.to(input_tensor.dtype)
        return out
