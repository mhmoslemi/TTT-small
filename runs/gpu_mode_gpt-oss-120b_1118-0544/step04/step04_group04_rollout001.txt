"""
Outgoing TriMul forward pass (AlphaFold3).
...
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    """In-place element‑wise multiplication out *= gate for 1‑D flattened tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out * gate, mask=mask)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor with shape [B, N, N, dim] (float32)
            - mask: torch.Tensor with shape [B, N, N] (bool or float), may be ignored
            - weights: dict of model weights (float32 tensors) matching the
                       module definition.
            - config: dict with keys
                * "dim" (int) – input channel size C
                * "hidden_dim" (int) – hidden projection size
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B, N, N, dim], float32

    # -----------------------------------------------------------------
    # 2. Cast to FP16 for the heavy arithmetic
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x_fp16, right_proj_weight)

    # -----------------------------------------------------------------
    # 4. Optional masking (applied before the gates)
    # -----------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; broadcast to hidden_dim
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp
        right = right * mask_fp

    # -----------------------------------------------------------------
    # 5. Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))

    left = left * left_gate      # [B, N, N, hidden_dim]
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Batched GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    We treat (B * hidden_dim) as the batch dimension for torch.bmm.
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # Rearrange to (B, hidden_dim, N, N) and then collapse batch*hidden
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left.view(B * hidden_dim, N, N)      # (B*hidden_dim, N, N)
    right_mat = right.view(B * hidden_dim, N, N)

    # Matrix multiply left @ rightᵀ for each (B*hidden_dim) slice
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden_dim, N, N)

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7. Output LayerNorm over hidden dimension
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # still fp16

    # -----------------------------------------------------------------
    # 8. Out‑gate (linear + sigmoid) and fused element‑wise multiplication
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # [B, N, N, hidden_dim]

    # Fuse using a tiny Triton kernel
    total_elements = out.numel()
    BLOCK = 1024
    grid = ((total_elements + BLOCK - 1) // BLOCK,)
    # Ensure tensors are contiguous for the kernel
    out_contig = out.contiguous()
    gate_contig = out_gate.contiguous()
    elementwise_mul_kernel[grid](
        out_contig,
        gate_contig,
        total_elements,
        BLOCK=BLOCK,
    )
    out = out_contig

    # -----------------------------------------------------------------
    # 9. Final linear projection back to original dimension
    # -----------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_weight)   # [B, N, N, dim], fp16

    # Cast to float32 as required by the interface
    return out.to(torch.float32)

"""
Outgoing TriMul forward pass (AlphaFold3).

The implementation follows the reference PyTorch module:
1. Layer‑norm over the input channels.
2. Four half‑precision linear projections (left/right and their gates).
3. Optional masking of the left/right projections.
4. Sigmoid gating and element‑wise multiplication.
5. Batched GEMM:   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   (implemented with torch.bmm on a (B·hidden) batch).
6. Layer‑norm over the hidden dimension.
7. Out‑gate (linear + sigmoid) fused with an element‑wise multiplication
   performed by a tiny Triton kernel.
8. Final linear projection back to the original dimension.
All heavy arithmetic is performed in FP16; the final result is cast to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    """In‑place out *= gate for flattened 1‑D tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out * gate, mask=mask)

    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float). May be ignored.
            - weights: dict of model weights (float32 tensors) matching the
              module definition.
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, the mask is not used.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------

    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim], float32

    # -----------------------------------------------------------------
    # 2. Cast to FP16 for the heavy arithmetic
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x_fp16, right_proj_weight)

    # -----------------------------------------------------------------
    if not nomask:

        left = left * mask_fp
        right = right * mask_fp

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))

    left = left * left_gate          # [B, N, N, hidden_dim]
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Batched GEMM:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as (B·hidden) independent matmuls.
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape

    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)

    right_mat = right.view(B * hidden_dim, N, N)

    # left @ rightᵀ for each slice
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)
