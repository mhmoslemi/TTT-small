"""
TriMul (outgoing) forward pass – Triton‑fused gating + cuBLAS batched GEMM.

Algorithm
---------
1. Layer‑norm the input tensor (B, N, N, C).
2. Cast to fp16 for compute‑heavy stages.
3. Linear projections:
       left_proj  = x·W_left   (B,N,N,H)
       right_proj = x·W_right  (B,N,N,H)
   Gate logits:
       left_gate  = sigmoid(x·W_lgate)
       right_gate = sigmoid(x·W_rgate)
       out_gate   = sigmoid(x·W_outgate)
4. Fuse projection·gate·mask (if a mask is present) with a tiny Triton kernel
   that broadcasts the 3‑D mask over the hidden dimension H.
5. Core O(N³·H) reduction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (B·H independent N×N matmuls) using
   torch.bmm, which leverages highly tuned cuBLAS kernels.
6. Apply a second LayerNorm over the hidden dimension, multiply by the output
   gate and project back to the original channel dimension.
7. Cast the result back to the input dtype (usually float32).

Only the gate‑mask multiplication is performed with Triton; the heavy matrix
multiplication leverages cuBLAS for maximal throughput.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: elementwise proj * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    proj_ptr,          # *[total_elems]   fp16
    gate_ptr,          # *[total_elems]   fp16
    mask_ptr,          # *[mask_elems]    fp16
    out_ptr,           # *[total_elems]   fp16
    total_elems,       # int
    hidden_dim,        # int, size of H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # map flat index to mask element (broadcast over hidden dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=mask)


def apply_gate_and_mask(proj: torch.Tensor,
                        gate: torch.Tensor,
                        mask: torch.Tensor,
                        block_size: int = 65536) -> torch.Tensor:
    """
    proj, gate : (B, N, N, H)  fp16
    mask       : (B, N, N)     fp16
    Returns    : proj * gate * mask  (fp16)
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)

    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,dim) -> (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate logits + sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Multiply projection, gate and (optional) mask
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction – batched GEMM using cuBLAS (torch.bmm)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape to (B*H, N, N) where the second operand is transposed on the K dim
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # batched matrix multiplication
    out_flat = torch.bmm(left_flat, right_flat)          # (B*H, N, N), fp16

    # restore to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)                       # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
```
