Define kernel function:

@triton.jit
def bmm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0) # tile index
    batch = tl.program_id(1) # batch index (B*H)

    # tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute base pointers for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # load a
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load b (K,N)
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # store output
    c = acc.to(tl.float16)

    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )

"""
TriMul outgoing kernel accelerated with Triton.
Inputs:
    - input_tensor: [B, N, N, C] (float32 or float16)
    - mask: [B, N, N] or None (binary mask)
    - weights: dict of parameters
    - config: dict containing "dim" and "hidden_dim"
The heavy O(N³·H) reduction is performed as a batched GEMM
(L = left @ rightᵀ) using a custom Triton kernel.
All compute‑intensive work is done in float16 with float32
accumulation; the final result is cast back to the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _bmm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix dimensions
    stride_am, stride_ak,               # strides for A (M,K)
    stride_bk, stride_bn,               # strides for B (K,N)
    stride_cm, stride_cn,               # strides for C (M,N)
    stride_ab, stride_bb, stride_cb,    # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A: M×K, B: K×N)."""
    pid = tl.program_id(0)      # tile index (i‑j)
    batch = tl.program_id(1)    # batch index (B·H)

    # Compute tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Main K‑loop
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # Load A block (M × K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B block (K × N)
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp16·fp16 → fp32

    # Store result
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # --------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_fp16 = x.to(dtype_fp16)

    # --------------------------------------------------------------------- #
    # 2. Cast model weights to half precision (on the same device)
    # --------------------------------------------------------------------- #
    # combine left projection & gate weights to halve linear passes
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype_fp16)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype_fp16)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # --------------------------------------------------------------------- #
    # 3. Linear projections + gates (combined)
    # --------------------------------------------------------------------- #
    # left_comb: [B, N, N, 2*H]
    left_comb = F.linear(x_fp16, left_comb_w)          # half precision
    right_comb = F.linear(x_fp16, right_comb_w)        # half precision

    left_proj = left_comb[..., :hidden_dim]            # [B, N, N, H]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # optional mask (broadcast over hidden dimension)
    if mask is not None and not nomask:
        mask_f = mask.to(dtype_fp16).unsqueeze(-1)    # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # apply gates
    left = left_proj * left_gate        # [B, N, N, H]
    right = right_proj * right_gate     # [B, N, N, H]

    # --------------------------------------------------------------------- #
    # 4. Core O(N³·H) reduction via batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------------- #
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape to (batch = B*H, M = N, K = N) for A and (batch, K = N, N) for Bᵀ
    # left_perm: (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right needs to be transposed on the two N axes: (B, H, N, N) where axes (k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    left_flat = left_perm.view(B * H, N, N)   # (batch, M, K)
    right_flat = right_perm.view(B * H, N, N) # (batch, K, N)

    out_flat = torch.empty_like(left_flat)    # (batch, M, N), fp16

    # strides for the kernel
    stride_a_batch = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_c_batch = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # block sizes – tuned for typical seq‑len ≤ 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _bmm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,  # M
        N,  # N
        N,  # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 5. Hidden‑dim LayerNorm
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------------- #
    # 6. Output gate (element‑wise sigmoid) and multiplication
    # --------------------------------------------------------------------- #
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # [B, N, N, H]
    out = out * out_gate

    # --------------------------------------------------------------------- #
    # 7. Final projection back to original channel dimension
    # --------------------------------------------------------------------- #
    out = F.linear(out, to_out_w)   # [B, N, N, C]

    # Cast back to the original dtype of the input tensor
    out = out.to(input_tensor.dtype)

    return out

"""
TriMul outgoing kernel accelerated with Triton.

The expensive O(N³·H) reduction is expressed as a batched matrix‑multiply:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
which is equivalent to a batched GEMM `left @ rightᵀ`.  The kernel
operates in fp16 with fp32 accumulation and the surrounding logic
(layer‑norms, gates, linear projections) is performed with PyTorch.
The final result is returned in the original dtype of the input tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _bmm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix dimensions
    stride_am, stride_ak,               # strides for A (M, K)
    stride_bk, stride_bn,               # strides for B (K, N)
    stride_cm, stride_cn,               # strides for C (M, N)
    stride_ab, stride_bb, stride_cb,    # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B  where A is (M×K) and B is (K×N)."""
    pid = tl.program_id(0)      # tile index (i‑j)
    batch = tl.program_id(1)    # batch index (B·H)

    # tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # out‑of‑bounds masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the current batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp16·fp16 → fp32 accumulation

    # write result
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with the model parameters (float32)
            - config: dict containing at least "dim" and "hidden_dim";
                      optional key "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # --------------------------------------------------------------------- #
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_fp16 = x.to(dtype_fp16)

    # --------------------------------------------------------------------- #
    # 2. Cast model weights to fp16 (all heavy ops in half precision)
    # --------------------------------------------------------------------- #
    # combine projection & gate weights to halve the number of linear calls
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype_fp16)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype_fp16)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # --------------------------------------------------------------------- #
    # 3. Linear projections + gates (combined)
    # --------------------------------------------------------------------- #
    left_comb = F.linear(x_fp16, left_comb_w)          # [B, N, N, 2*H]
    right_comb = F.linear(x_fp16, right_comb_w)        # [B, N, N, 2*H]

    left_proj = left_comb[..., :hidden_dim]            # [B, N, N, H]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # optional mask (broadcast over hidden dimension)
    if mask is not None and not nomask:
        mask_f = mask.to(dtype_fp16).unsqueeze(-1)    # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # apply gates
    left = left_proj * left_gate      # [B, N, N, H]
    right = right_proj * right_gate   # [B, N, N, H]

    # --------------------------------------------------------------------- #
    # 4. Core O(N³·H) reduction via batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------------- #
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape to (batch = B*H, M = N, K = N) for A and (batch, K = N, N) for Bᵀ
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # [B, H, N, N] (i↔k)

    left_flat = left_perm.view(B * H, N, N)    # (batch, M, K)
    right_flat = right_perm.view(B * H, N, N)  # (batch, K, N)

    out_flat = torch.empty_like(left_flat)     # (batch, M, N)

    # strides for Triton kernel
    stride_a_batch = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_c_batch = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # tile sizes (tuned for N ≤ 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _bmm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 5. Hidden‑dim LayerNorm
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------------- #
    # 6. Output gate (element‑wise) and multiplication
    # --------------------------------------------------------------------- #
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # [B, N, N, H]
    out = out * out_gate

    # --------------------------------------------------------------------- #
    # 7. Final projection back to original channel dimension
    # --------------------------------------------------------------------- #
    out = F.linear(out, to_out_w)   # [B, N, N, C]

    # Cast back to the original dtype of the input tensor
    out = out.to(input_tensor.dtype)

    return out
