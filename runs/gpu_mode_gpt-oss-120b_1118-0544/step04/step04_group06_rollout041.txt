def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # cast to fp16
    x_fp16 = x.to(torch.float16)

    # Weights to fp16
    left_proj_w = weights["left_proj.weight"].to(x.device, torch.float16)
    right_proj_w = weights["right_proj.weight"].to(x.device, torch.float16)
    left_gate_w = weights["left_gate.weight"].to(x.device, torch.float16)
    right_gate_w = weights["right_gate.weight"].to(x.device, torch.float16)
    out_gate_w = weights["out_gate.weight"].to(x.device, torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(x.device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(x.device, torch.float16)
    to_out_w = weights["to_out.weight"].to(x.device, torch.float16)

    # projections
    left_proj = torch.nn.functional.linear(x_fp16, left_proj_w)  # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x_fp16, right_proj_w)

    # gates
    left_gate = torch.nn.functional.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()

    # mask apply
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # core TriMul reduction using Triton
    B, N, _, H = left.shape

    # allocate output tensor (fp16)
    out_tensor = torch.empty_like(left, dtype=torch.float16)

    # compute strides
    stride_b = left.stride(0)
    stride_i = left.stride(1)  # i dimension stride (N*H)
    stride_k = left.stride(2)  # k dimension stride (H)
    stride_h = left.stride(3)  # H dimension stride (1)

    # right strides are same
    stride_b_r = right.stride(0)
    stride_i_r = right.stride(1)  # j index
    stride_k_r = right.stride(2)
    stride_h_r = right.stride(3)

    # output strides
    stride_b_o = out_tensor.stride(0)
    stride_i_o = out_tensor.stride(1)
    stride_j_o = out_tensor.stride(2)
    stride_h_o = out_tensor.stride(3)

    # choose block sizes
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 32  # hidden tile size

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M,
          (N + BLOCK_N - 1) // BLOCK_N,
          B * ((H + BLOCK_D - 1) // BLOCK_D) )
    )

    trimul_kernel[grid](
        left,
        right,
        out_tensor,
        B, N, H,
        stride_b, stride_i, stride_k, stride_h,
        stride_b_r, stride_i_r, stride_k_r, stride_h_r,
        stride_b_o, stride_i_o, stride_j_o, stride_h_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # now out_tensor shape (B,N,N,H)
    # Apply to_out_norm layernorm, out_gate, and final linear
    out = torch.nn.functional.layer_norm(out_tensor, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)

    # cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out

def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):

    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)  # batch * hidden tile

    # compute i and j base
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    j = j0 + tl.arange(0, BLOCK_N)

    mask_j = j < N

TriMul (outgoing) – Triton‑accelerated implementation
-----------------------------------------------------

Algorithm
~~~~~~~~~
1. Layer‑norm the input tensor (B,N,N,C).
2. Cast to fp16 for the compute‑intensive part.
3. Linear projections (left/right) and gate projections (left_gate,

4. Apply mask (if present) and element‑wise gates.
5. Core TriMul reduction:

The kernel tiles:

    pid_i = tl.program_id(0)          # tile over i
    pid_j = tl.program_id(1)          # tile over j
    pid_bd = tl.program_id(2)         # combined batch & hidden‑tile dimension

    # ----- tile origin -----
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # batch index and hidden‑tile start
    b = pid_bd // num_hidden_tiles
    d0 = (pid_bd % num_hidden_tiles) * BLOCK_D

    # ----- intra‑tile indices -----
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    d = d0 + tl.arange(0, BLOCK_D)

    mask_i = i < N
    mask_j = j < N
    mask_d = d < H

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ----- reduction over k -----
    for k in range(0, N, BLOCK_K):
        k_range = k + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # left tile  (M, K, D)
        left_off = (
            b * stride_bl
            + i[:, None, None] * stride_il
            + k_range[None, :, None] * stride_kl
            + d[None, None, :] * stride_hl
        )
        left = tl.load(
            left_ptr + left_off,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # right tile (N, K, D)
        right_off = (
            b * stride_br
            + j[:, None, None] * stride_ir
            + k_range[None, :, None] * stride_kr
            + d[None, None, :] * stride_hr
        )
        right = tl.load(
            right_ptr + right_off,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        )

        # matmul for each hidden slice
        for d_idx in tl.static_range(0, BLOCK_D):
            # (M, K) @ (K, N)ᵀ  → (M, N)
            acc[:, :, d_idx] += tl.dot(
                left[:, :, d_idx], right[:, :, d_idx], trans_b=True
            )

    # ----- write output -----
    out_off = (
        b * stride_bo
        + i[:, None, None] * stride_io
        + j[None, :, None] * stride_jo
        + d[None, None, :] * stride_ho
    )
    tl.store(
        out_ptr + out_off,
        acc.to(tl.float16),
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C] (float32/float16) on CUDA
            - mask: [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # -------------------------------------------------------
    # 1. Input LayerNorm
    # -------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -------------------------------------------------------
    # 2. Cast to fp16 for heavy compute
    # -------------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------------
    # 3. Prepare fp16 weights
    # -------------------------------------------------------
    def to_fp16(t): return t.to(device=device, dtype=torch.float16)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w      = to_fp16(weights["to_out.weight"])

    # -------------------------------------------------------
    # 4. Linear projections
    # -------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # -------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # -------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # -------------------------------------------------------
    # 6. Apply mask + gates
    # -------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------------------------------------
    # 7. Core TriMul reduction (Triton kernel)
    # -------------------------------------------------------
    B, N_seq, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float16)

    # Strides (as used by the kernel)
    stride_bl, stride_il, stride_kl, stride_hl = left.stride()
    stride_br, stride_ir, stride_kr, stride_hr = right.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out.stride()

    # Tiling parameters (tuned for typical hardware)
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 32
    BLOCK_D = 8   # hidden‑dim tile; 8 * 16 * 16 = 2048 floats ≈ 8 KB per block

    num_hidden_tiles = (hidden_dim + BLOCK_D - 1) // BLOCK_D

    grid = (
        (N_seq + BLOCK_M - 1) // BLOCK_M,
        (N_seq + BLOCK_N - 1) // BLOCK_N,
        B * num_hidden_tiles,
    )

    _trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N_seq,
        hidden_dim,
        stride_bl,
        stride_il,
        stride_kl,
        stride_hl,
        stride_br,
        stride_ir,
        stride_kr,
        stride_hr,
        stride_bo,
        stride_io,
        stride_jo,
        stride_ho,
        num_hidden_tiles,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # -------------------------------------------------------
    # 8. Output LayerNorm, gate & final projection
    # -------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # -------------------------------------------------------
    # 9. Cast back to original dtype
    # -------------------------------------------------------
    out = out.to(orig_dtype)
    return out
