Define:

@triton.jit

def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_head, M, N, K,
    stride_ah, stride_ak, stride_a_batch,
    stride_bh, stride_bk, stride_b_batch,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(2)   # batch_head index
    pid_i = tl.program_id(0)    # tile row (i)
    pid_j = tl.program_id(1)    # tile col (j)

    # compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute pointers for A and B for current batch_head
    a_ptrs = a_ptr + pid_bh * stride_a_batch + offs_i[:, None] * stride_ah + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_bh * stride_b_batch + offs_j[:, None] * stride_bh + offs_k[None, :] * stride_bk
    # Note that we load B in shape (BLOCK_N, BLOCK_K) needing B[j,k]

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load fragments
        a = tl.load(a_ptrs, mask=(offs_i[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_j[:, None] < N) & (offs_k[None, :] < K), other=0.0)
        # Compute dot product across K: (BLOCK_M, BLOCK_K) @ (BLOCK_N, BLOCK_K).T
        acc += tl.dot(a, b, trans_b=True)
        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Cast accumulator to output type
    c = acc.to(tl.float16)

    # Store results
    c_ptrs = c_ptr + pid_bh * stride_c_batch + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    mask_c = (offs_i[:, None] < M) & (offs_j[None, :] < N)
    tl.store(c_ptrs, c, mask=mask_c)
```

```
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_ah, stride_ak,
    stride_b_batch, stride_bh, stride_bk,
    stride_c_batch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(2)   # batch*head dimension
    pid_i = tl.program_id(0)    # output row block
    pid_j = tl.program_id(1)    # output col block

    # Compute offset indices
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to blocks of A and B (B is accessed as [j, k])
    a_ptrs = a_ptr + pid_bh * stride_a_batch + offs_i[:, None] * stride_ah + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_bh * stride_b_batch + offs_j[:, None] * stride_bh + offs_k[None, :] * stride_bk

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A and B blocks
        a = tl.load(a_ptrs, mask=(offs_i[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_j[:, None] < N) & (offs_k[None, :] < K), other=0.0)
        # Compute dot product across K dimension (A @ B^T)
        acc += tl.dot(a, b, trans_b=True)
        # Advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Cast accumulator to output type (fp16)
    c = acc.to(tl.float16)

    # Store result
    c_ptrs = c_ptr + pid_bh * stride_c_batch + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    mask_c = (offs_i[:, None] < M) & (offs_j[None, :] < N)
    tl.store(c_ptrs, c, mask=mask_c)

def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Implements the AlphaFold3 TriMul operation using:
    - LayerNorm on input.
    - Linear projections + gating (FP16).
    - Optional mask.
    - Batched matrix multiplication L @ R^T computed with a custom Triton kernel.
    - Second LayerNorm, out_gate gating (Triton elementwise mul), final projection.
    Returns tensor of shape [B, N, N, C] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1) Input LayerNorm (float32)
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, N, N, dim], float32

    # Cast to fp16 for rest of the computation
    x_fp16 = x.to(torch.float16)

    # 2) Linear projections (FP16)
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    left_proj = F.linear(x_fp16, left_proj_weight)   # (B, N, N, hidden)
    right_proj = F.linear(x_fp16, right_proj_weight)

    # 3) Optional mask
    if not nomask:
        mask_f = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # 4) Gating
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))

    left = left_proj * left_gate   # (B, N, N, hidden)
    right = right_proj * right_gate

    # 5) Core TriMul multiplication using Triton
    # reshape to (B*hidden, N, N)
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    batch_head = B * hidden_dim
    left_mat = left.view(batch_head, N, N)
    right_mat = right.view(batch_head, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat)  # fp16

    # strides (in elements)
    stride_a_batch = left_mat.stride(0)
    stride_ah = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_bh = right_mat.stride(1)
    stride_bk = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch_head,
    )

    tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_a_batch, stride_ah, stride_ak,
        stride_b_batch, stride_bh, stride_bk,
        stride_c_batch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 6) to_out_norm (LayerNorm over hidden dim)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # (B, N, N, hidden), fp16

    # 7) out_gate gating (sigmoid) and elementwise multiplication using Triton
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # (B,N,N,hidden)

    # Elementwise multiplication kernel
    total = out.numel()
    BLOCK = 1024
    grid_elem = ( (total + BLOCK - 1) // BLOCK, )
    elementwise_mul_kernel[grid_elem](
        out,
        out_gate,
        total,
        BLOCK=BLOCK,
    )

    # 8) Final linear projection back to dim
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_weight)   # (B, N, N, dim), fp16

    # Return float32
    return out.to(torch.float32)

"""
Outgoing TriMul (Triangle Multiplicative Update) forward pass.

This implementation follows the AlphaFold3 TriMul module but replaces the
expensive batched matrix multiplication with a custom Triton kernel.
The steps are:
1. Layer‑norm on the input (float32).
2. FP16 linear projections for left/right tensors and their gates.
3. Optional masking of the left/right projections.
4. Gating via sigmoid and element‑wise multiplication.
5. Batched GEMM L @ Rᵀ computed with a Triton kernel (FP16, FP32 accumulation).
6. Layer‑norm on the hidden dimension.
7. Out‑gate (sigmoid) applied with a tiny Triton element‑wise kernel.
8. Final linear projection back to the original channel dimension.
The final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    """1‑D element‑wise multiplication: out = out * gate."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out * gate, mask=mask)


@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_ah, stride_ak,
    stride_b_batch, stride_bh, stride_bk,
    stride_c_batch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ Bᵀ for tensors with shape
        A: (batch_head, M, K)
        B: (batch_head, N, K)   (note: we use B’s rows as the second index)
    The kernel processes a tile of size BLOCK_M×BLOCK_N per program instance.
    """
    pid_bh = tl.program_id(2)          # batch_head index (B * hidden)
    pid_i = tl.program_id(0)           # output‑row tile
    pid_j = tl.program_id(1)           # output‑col tile

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)

    for k in range(num_k_blocks):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        mask_i = offs_i < M
        mask_j = offs_j < N
        mask_k = k_offset < K

        # A block: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_bh * stride_a_batch
            + offs_i[:, None] * stride_ah
            + k_offset[None, :] * stride_ak,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B block (treated as rows j, columns k): (BLOCK_N, BLOCK_K)
        b = tl.load(
            b_ptr
            + pid_bh * stride_b_batch
            + offs_j[:, None] * stride_bh
            + k_offset[None, :] * stride_bk,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Accumulate A @ Bᵀ
        acc += tl.dot(a, b, trans_b=True)

    # Write back (cast to fp16)
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_bh * stride_c_batch
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj,
        c,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask        : torch.Tensor [B, N, N] (bool/float) – may be ignored.
            - weights     : dict of model weights (float32 tensors)
            - config      : dict with keys
                * "dim"        – input channel dimension C
                * "hidden_dim"– hidden projection dimension H
                * "nomask"    – optional bool, skip masking if True

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim], float32

    # -----------------------------------------------------------------
    # 2) Linear projections & gates (FP16)
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_proj = F.linear(x_fp16, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x_fp16, right_proj_w)

    # -----------------------------------------------------------------
    # 3) Optional mask
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))

    left = left_proj * left_gate    # (B, N, N, H)
    right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5) Batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with a custom Triton kernel.
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape

    # Move hidden dim to the second axis and make tensors contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    batch_head = B * hidden_dim
    left_mat = left.view(batch_head, N, N)
    right_mat = right.view(batch_head, N, N)

    out_mat = torch.empty_like(left_mat)  # FP16 output

    # Strides in number of elements (not bytes)
    stride_a_batch = left_mat.stride(0)
    stride_ah = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_bh = right_mat.stride(1)
    stride_bk = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch_head,
    )

    tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_a_batch, stride_ah, stride_ak,
        stride_b_batch, stride_bh, stride_bk,
        stride_c_batch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # (B, N, N, H), fp16

    # -----------------------------------------------------------------
    # 7) Out‑gate (sigmoid) and element‑wise multiplication (Triton)
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))  # (B, N, N, H)

    total = out.numel()
    BLOCK = 1024
    grid_elem = ((total + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid_elem](
        out,
        out_gate,
        total,
        BLOCK=BLOCK,
    )

    # -----------------------------------------------------------------
    # 8) Final linear projection back to input dimension
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, C), fp16

    # Return as float32
    return out.to(torch.float32)
