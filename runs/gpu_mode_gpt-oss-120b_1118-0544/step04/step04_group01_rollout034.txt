"""
TriMul (outgoing) forward pass – Triton‑fused core reduction.
Algorithm
---------
1. Layer‑norm the input (float32) → fp16.
2. Four linear projections (left/right) and three gates (left, right, out) in fp16.
3. Fuse `proj * gate * mask` (when a mask is present) with a small Triton kernel.
4. Compute the O(N³·H) reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   via a custom batched GEMM Triton kernel that treats (batch, hidden) as a single
   program dimension and tiles the (i, j, k) axes.
5. Apply a second LayerNorm on the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original channel dimension.
6. Cast the result back to the input dtype (float32).

The kernel uses fp16 arithmetic with fp32 accumulation for numerical stability.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask (optional)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE
    offs = offset + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # mask index is the element index stripped of the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0).to(tl.float16)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0).to(tl.float16)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0).to(tl.float16)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 4096) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using Triton.
    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],               # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for right (note: we read the transpose, so k‑row, j‑col)
    stride_r_b, stride_r_k, stride_r_j, stride_r_h,
    # strides for out
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)       # combined batch * hidden
    pid_m = tl.program_id(1)     # block row (i)
    pid_n = tl.program_id(2)     # block col (j)

    # decode batch and hidden index
    b = pid // H
    h = pid % H

    # block offsets in i and j dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for left: (i, k)
        left_ptrs = left_ptr + \
            b * stride_l_b + \
            h * stride_l_h + \
            offs_i[:, None] * stride_l_i + \
            offs_k[None, :] * stride_l_k

        # pointers for right (transposed): (k, j)
        right_ptrs = right_ptr + \
            b * stride_r_b + \
            h * stride_r_h + \
            offs_k[:, None] * stride_r_k + \
            offs_j[None, :] * stride_r_j

        # masks
        mask_left = mask_i[:, None] & mask_k[None, :]
        mask_right = mask_k[:, None] & mask_j[None, :]

        a = tl.load(left_ptrs, mask=mask_left, other=0.0).to(tl.float32)
        b_mat = tl.load(right_ptrs, mask=mask_right, other=0.0).to(tl.float32)

        acc += tl.dot(a, b_mat)

    # store result (cast back to fp16)
    out_ptrs = out_ptr + \
        b * stride_o_b + \
        h * stride_o_h + \
        offs_i[:, None] * stride_o_i + \
        offs_j[None, :] * stride_o_j

    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype  # usually torch.float32

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)   # compute heavy parts in fp16

    # ------------------------------------------------------------------
    # cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # linear projections
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)    # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # optional mask & gate fusion (project * gate * mask)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # core O(N³·H) reduction via Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out_hidden = torch.empty_like(left)  # (B,N,N,H)

    # strides (as Python ints)
    stride_l = left.stride()   # (b,i,k,h)
    stride_r = right.stride()
    stride_o = out_hidden.stride()

    # block sizes – tuned for A100/B200 style GPUs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                                 # batch * hidden
        triton.cdiv(N, BLOCK_M),               # i‑blocks
        triton.cdiv(N, BLOCK_N),               # j‑blocks
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out_hidden,
        B, N, H,
        # left strides
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        # right strides (note the transposition: k‑row, j‑col)
        stride_r[0], stride_r[2], stride_r[1], stride_r[3],
        # output strides
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # second LayerNorm on hidden dimension, output gate, final linear
    # ------------------------------------------------------------------
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out_hidden = out_hidden * out_gate

    # final projection back to original dim
    out = torch.nn.functional.linear(out_hidden, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
