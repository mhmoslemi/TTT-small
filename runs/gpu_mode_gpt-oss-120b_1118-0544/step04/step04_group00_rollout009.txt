"""
TriMul (outgoing) forward pass – Triton‑accelerated mask & gate fusion.

Algorithm
---------
1. Layer‑norm the input tensor over its channel dimension `dim`.
2. Cast to fp16 (the compute‑heavy part runs in half‑precision).
3. Linear projections:
      left_proj  = x @ W_left   (B,N,N,H)
      right_proj = x @ W_right
   Gate projections (sigmoid):
      left_gate  = σ(x @ W_left_gate)
      right_gate = σ(x @ W_right_gate)
      out_gate   = σ(x @ W_out_gate)   # (B,N,N,H)
4. Fuse mask, gate and projection in a single Triton kernel
   (applied only when a mask is present).  The kernel computes
        proj * gate * mask   (mask broadcast over the hidden dim H).
5. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication performed with
   ``torch.bmm`` after reshaping to (B*H, N, N).
6. Output LayerNorm over the hidden dimension H.
7. Multiply by `out_gate`, apply the final linear projection back to `dim`.
8. Cast the result back to the original dtype (usually float32).

The implementation mixes Triton (step 4) and high‑performance PyTorch
primitives (steps 1‑3, 5‑8) to keep the code simple while still achieving
significant speed‑ups on NVIDIA GPUs.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: (proj * gate * mask) – mask broadcast over H
# --------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # total number of elements in proj/gate (=B*N*N*H)
    hidden_dim,        # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)          # [0, BLOCK_SIZE)

    # mask for valid elements (last block may be partial)
    valid = offs < total_elems

    # index of the corresponding mask element (broadcast over H)
    mask_idx = offs // hidden_dim

    # load tensors
    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_proj_gate_mask(proj: torch.Tensor,
                          gate: torch.Tensor,
                          mask: torch.Tensor,
                          block_size: int = 1024) -> torch.Tensor:
    """
    Fuse projection, gate and mask using Triton.
    All tensors must be contiguous fp16.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)  (binary 0/1)
    Returns proj * gate * mask broadcast over H.
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask         : torch.Tensor of shape [B, N, N] or None
            - weights      : dict of model parameters (float32)
                Keys required:
                    "norm.weight", "norm.bias",
                    "left_proj.weight", "right_proj.weight",
                    "left_gate.weight", "right_gate.weight", "out_gate.weight",
                    "to_out_norm.weight", "to_out_norm.bias",
                    "to_out.weight"
            - config       : dict with entries
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to fp16 (no bias in these layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (same shape) + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask and gating (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous, and has shape (B, N, N)
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_proj_gate_mask(left_proj, left_gate, mask_fp16)
        right = _apply_proj_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        # No mask – simple elementwise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    left, right : (B, N, N, H)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T  ->  (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
