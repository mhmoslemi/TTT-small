Pseudo:

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    # strides
    stride_b_lh, stride_i_lh, stride_k_lh, stride_h_lh,
    stride_b_rh, stride_j_rh, stride_k_rh, stride_h_rh,
    stride_b_oh, stride_i_oh, stride_j_oh, stride_h_oh,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    # program ids
    pid_bh = tl.program_id(0)  # combined batch and hidden group
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # determine batch id and hidden slice
    num_h_groups = tl.cdiv(H, BLOCK_H)
    batch_id = pid_bh // num_h_groups
    h_group = pid_bh % num_h_groups
    h_start = h_group * BLOCK_H
    h_end = tl.minimum(h_start + BLOCK_H, H)
    cur_h = h_end - h_start

    # compute offsets for i, j within N
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # pointers for left (b,i,k,h) and right (b,j,k,h)
    # We'll iterate over k in blocks.
    # Create accumulator for output tile (BLOCK_M, BLOCK_N, cur_h)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32) # but shape must be compile time; allocate max BLOCK_H and later mask.

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        # compute k offsets and mask sizes
        k_valid = tl.minimum(N - k0, BLOCK_K)
        # load left tile:
        # left shape (B,N,N,H) -> need left[b, i, k0:k0+BLOCK_K, h]
        # offset: base + batch_id*stride_b_lh + i*stride_i_lh + (k0+offset_k)*stride_k_lh + h*stride_h_lh
        # We need to broadcast over h dimension.

        # Create offsets for i and k
        offs_i = i0 + tl.arange(0, BLOCK_M)
        offs_j = j0 + tl.arange(0, BLOCK_N)
        offs_k = k0 + tl.arange(0, BLOCK_K)
        offs_h = h_start + tl.arange(0, BLOCK_H)

        # mask for out-of-bounds
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, row_stride_a, col_stride_a,
    batch_stride_b, row_stride_b, col_stride_b,
    batch_stride_c, row_stride_c, col_stride_c,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute k loop offsets
    k = 0

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Bound checking:
        a_mask = (offs_m < M) & (offs_k < K)
        b_mask = (offs_k < K) & (offs_n < N)

        # Compute pointers
        a_ptrs = a_ptr + batch_id * batch_stride_a + offs_m[:, None] * row_stride_a + offs_k[None, :] * col_stride_a
        b_ptrs = b_ptr + batch_id * batch_stride_b + offs_k[:, None] * row_stride_b + offs_n[None, :] * col_stride_b

        # Load
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Compute dot product accumulate
        acc += tl.dot(a, b)

    # Store result
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = c_ptr + batch_id * batch_stride_c + offs_m[:, None] * row_stride_c + offs_n[None, :] * col_stride_c
    tl.store(c_ptrs, acc, mask=c_mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # cast to fp16
    x = x.to(torch.float16)

    # linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # apply mask and gating
    if mask is not None and not nomask:
        mask = mask.to(torch.float16).unsqueeze(-1)  # broadcast across hidden
        left = left * left_gate * mask
        right = right * right_gate * mask
    else:
        left = left * left_gate
        right = right * right_gate

    # Move hidden dimension to batch for batched gemm
    # left: (B,N,N,H) -> (B,H,N,N) -> contiguous -> view
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape

    # Flatten batch dimension: (B*H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # compute out using Triton kernel
    out_mat = triton_matmul(left_mat, right_mat, N, N, N)

    # reshape
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # shape (B,N,N,H)

    # output norm and gate
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate

    # final linear
    out = F.linear(out, weights["to_out.weight"])

    # cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

The implementation follows the reference PyTorch module but moves the
O(N³·H) contraction (the einsum) to a custom Triton batched GEMM kernel.
All linear layers, LayerNorms and activations stay in half‑precision to
leverage Tensor‑Cores, while the final output is cast back to the input
dtype (usually float32).  The kernel operates on a flattened batch
dimension (batch × hidden), so each hidden channel is processed as an
independent matrix multiplication.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to A, B, C
    batch, M, N, K,                     # int scalars
    stride_az, stride_am, stride_ak,    # strides for A
    stride_bz, stride_bk, stride_bn,    # strides for B (B is accessed transposed)
    stride_cz, stride_cm, stride_cn,    # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes C = A @ B^T for each batch element.
    A: (batch, M, K)  – row‑major
    B: (batch, N, K)  – will be read as (batch, K, N) via swapped strides
    C: (batch, M, N)
    All tensors are in fp16; accumulation is in fp32.
    """
    pid_z = tl.program_id(0)   # batch index
    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile col

    # Compute offsets for the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for A (batch, M, K)
        a_ptrs = (
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        # Pointers for B (batch, K, N) – note swapped strides for transposition
        b_ptrs = (
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk   # k as row
            + offs_n[None, :] * stride_bn   # n as column
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write back result in fp16
    c = acc.to(tl.float16)
    c_ptrs = (
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ using the Triton kernel above.
    A, B: tensors of shape (batch, N, N), dtype fp16, contiguous.
    Returns C of shape (batch, N, N), dtype fp16.
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2, "Inner dimensions must match"

    # Output tensor
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Strides (in elements)
    stride_az, stride_am, stride_ak = A.stride()
    stride_bz, stride_bn, stride_bk = B.stride()
    # For B we need transposed access: row stride = stride_bk, col stride = stride_bn
    stride_bz = stride_bz
    stride_bk = stride_bk   # K‑dimension stride (as row after transposition)
    stride_bn = stride_bn   # N‑dimension stride (as column after transposition)

    stride_cz, stride_cm, stride_cn = C.stride()

    # Block sizes – chosen to fit typical Tensor‑Core kernels
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        math.ceil(M / BLOCK_M),
        math.ceil(N / BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        batch,
        M,
        N,
        K,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask_tensor: torch.Tensor of shape (B, N, N) or None
            - weights: dict with keys matching the module parameters
            - config: dict with at least "dim" and "hidden_dim";
                      optional "nomask" (bool, default False)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as the input.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- LayerNorm on the input -----
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Cast to fp16 for the compute‑heavy part
    x = x.to(torch.float16)

    # ----- Load linear weights in fp16 -----
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, torch.float16)

    # ----- Linear projections -----
    left = F.linear(x, left_proj_w)      # (B, N, N, H)
    right = F.linear(x, right_proj_w)    # (B, N, N, H)

    # ----- Gates (sigmoid) -----
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ----- Apply mask (if any) and gates -----
    if mask is not None and not nomask:
        mask_fp = mask.to(device, torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left = left * left_gate
        right = right * right_gate

    # ----- Prepare tensors for batched GEMM -----
    # Move hidden dimension to the batch axis and make them contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # Flatten batch × hidden -> (B*H, N, N)
    A = left_perm.view(B * H, N, N)
    B_mat = right_perm.view(B * H, N, N)

    # ----- Core O(N³·H) contraction using Triton -----
    out_mat = _batched_gemm(A, B_mat)   # (B*H, N, N) in fp16

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----- Output LayerNorm -----
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ----- Apply output gate -----
    out = out * out_gate

    # ----- Final linear projection back to dim -----
    out = F.linear(out, to_out_w)   # (B, N, N, C) in fp16

    # ----- Cast back to original dtype -----
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
