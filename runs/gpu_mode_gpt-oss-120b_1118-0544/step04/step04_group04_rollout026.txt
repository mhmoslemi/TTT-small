Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The kernel performs the heavy O(N³·C_hidden) contraction:
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
where left/right are the projected and gated pairwise representations.
The computation is expressed as a batched GEMM (A @ Bᵀ) over the
sequence dimension using a fused Triton kernel to avoid costly reshapes
and to exploit Tensor‑cores. All other operations (LayerNorms, linear
projections, gating, final linear) are done with PyTorch torch.nn.functional
for simplicity. The result is returned in float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul contraction.
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    # pointers
    A_ptr, B_ptr, C_ptr,
    # dimensions
    N, H,                     # sequence length, hidden_dim
    # strides (in elements)
    stride_A_batch, stride_A_i, stride_A_k, stride_A_h,
    stride_B_batch, stride_B_j, stride_B_k, stride_B_h,
    stride_C_batch, stride_C_i, stride_C_j, stride_C_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for tensors of shape
        A : (B, N, N, H)   (as A[b, i, k, h])
        B : (B, N, N, H)   (as B[b, j, k, h])
    The output C has the same shape as A/B.
    The kernel is launched with a 3‑D grid:
        pid0 = batch * hidden index,
        pid1 = tile over i (rows),
        pid2 = tile over j (cols).
    """
    # ------------------------------------------------------------------ #
    # 0) program ids
    # ------------------------------------------------------------------ #
    pid_bh = tl.program_id(0)     # combined batch*hidden
    pid_i  = tl.program_id(1)     # tile over i (rows)
    pid_j  = tl.program_id(2)     # tile over j (cols)

    # ------------------------------------------------------------------ #
    # 1) decode batch and hidden index
    # ------------------------------------------------------------------ #
    hidden = H
    b = pid_bh // hidden          # batch index
    h = pid_bh - b * hidden        # hidden index (mod)

    # ------------------------------------------------------------------ #
    # 2) compute tile start offsets
    # ------------------------------------------------------------------ #
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # ranges for the tile (may exceed N, will be masked)
    i_range = i_base + tl.arange(0, BLOCK_M)
    j_range = j_base + tl.arange(0, BLOCK_N)

    # ------------------------------------------------------------------ #
    # 3) compute base pointers for this (b,h) slice
    # ------------------------------------------------------------------ #
    a_base = A_ptr + b * stride_A_batch + h * stride_A_h
    b_base = B_ptr + b * stride_B_batch + h * stride_B_h
    c_base = C_ptr + b * stride_C_batch + h * stride_C_h

    # ------------------------------------------------------------------ #
    # 4) accumulation
    # ------------------------------------------------------------------ #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k_offset in range(0, N, BLOCK_K):
        k_range = k_offset + tl.arange(0, BLOCK_K)

        # masks for out‑of‑bounds accesses
        mask_i = i_range < N
        mask_j = j_range < N
        mask_k = k_range < N

        # ------------------------------------------------------------------
        # load A block: shape (BLOCK_M, BLOCK_K)
        # A[b, i, k, h]  -> strides (i, k)
        # ------------------------------------------------------------------
        a_ptrs = a_base + i_range[:, None] * stride_A_i + k_range[None, :] * stride_A_k
        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & mask_k[None, :]),
                    other=0.0)

        # ------------------------------------------------------------------
        # load B block transposed: shape (BLOCK_K, BLOCK_N)
        # We need Bᵀ, i.e. B[b, j, k, h] accessed as (k, j)
        # ------------------------------------------------------------------
        b_ptrs = b_base + k_range[:, None] * stride_B_k + j_range[None, :] * stride_B_j
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_j[None, :]),
                    other=0.0)

        # ------------------------------------------------------------------
        # inner product (TensorCore: FP16 inputs, FP32 accumulation)
        # ------------------------------------------------------------------
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # -------------------------------------------------------------------- #
    # 5) write result (cast back to FP16)
    # -------------------------------------------------------------------- #
    c = acc.to(tl.float16)

    # store only valid elements
    mask_store = (i_range[:, None] < N) & (j_range[None, :] < N)
    c_ptrs = c_base + i_range[:, None] * stride_C_i + j_range[None, :] * stride_C_j
    tl.store(c_ptrs, c, mask=mask_store)


# --------------------------------------------------------------------------- #
# Entry point expected by the evaluator
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys "dim" and "hidden_dim" (ints) and optional
                      "nomask" (bool)
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # -------------------------------------------------------------------- #
    # 1) Input LayerNorm (float32)
    # -------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"].to(device)
    norm_bias   = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # -------------------------------------------------------------------- #
    # 2) Cast to FP16 for the heavy compute
    # -------------------------------------------------------------------- #
    x_fp16 = x.to(torch.float16)

    # -------------------------------------------------------------------- #
    # 3) Linear projections (no bias)
    # -------------------------------------------------------------------- #
    left_proj_weight  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight   = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_weight)   # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_weight)

    # -------------------------------------------------------------------- #
    # 4) Optional mask (broadcast on hidden dim)
    # -------------------------------------------------------------------- #
    if not nomask:
        mask_fp16 = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------------------------- #
    # 5) Gating (sigmoid)
    # -------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))

    left = left * left_gate    # (B,N,N,hidden)
    right = right * right_gate

    # -------------------------------------------------------------------- #
    # 6) Core TriMul contraction via Triton
    # -------------------------------------------------------------------- #
    B, N, _, _ = left.shape
    out = torch.empty_like(left)   # FP16, same layout (B,N,N,hidden)

    # strides (in elements) for the original layout (B,N,N,hidden)
    sA_b, sA_i, sA_k, sA_h = left.stride()
    sB_b, sB_j, sB_k, sB_h = right.stride()
    sC_b, sC_i, sC_j, sC_h = out.stride()

    # block sizes – chosen to fit Tensor‑Core tiles
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # grid: (batch*hidden, tiles_i, tiles_j)
    grid = (
        B * hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        N, hidden,
        sA_b, sA_i, sA_k, sA_h,
        sB_b, sB_j, sB_k, sB_h,
        sC_b, sC_i, sC_j, sC_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------------------------- #
    # 7) Post‑processing: LayerNorm over hidden, out‑gate, final linear
    # -------------------------------------------------------------------- #
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B,N,N,hidden) FP16

    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))   # (B,N,N,hidden) FP16
    out = out * out_gate    # element‑wise (still FP16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_weight)   # (B,N,N,dim) FP16

    # -------------------------------------------------------------------- #
    # 8) Cast back to float32 as required by the interface
    # -------------------------------------------------------------------- #
    return out.to(torch.float32)
