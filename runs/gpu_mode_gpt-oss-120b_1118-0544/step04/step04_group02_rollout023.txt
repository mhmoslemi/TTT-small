"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (B, N, N, C) using PyTorch.
2. Cast to FP16 for compute‑intensive stages.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are performed with FP16 weights via torch.nn.functional.linear.
4. When a mask is present, a small Triton kernel fuses
   projection × gate × mask (mask is broadcast over the hidden dimension H).
5. Core O(N³·H) reduction:
       out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   Implemented as a batched GEMM using torch.bmm on the flattened
   (B·H, N, N) tensors (cuBLAS/Tensor‑core fast path).
6. Apply the second LayerNorm, multiply by the output gate and project
   back to the original channel dimension.
7. Cast the result back to the input dtype (usually float32).

Only the mask‑gate fusion is performed in Triton, satisfying the
requirement to have at least one custom kernel.
"""

def gate_mask_kernel(

    gate_ptr,          # *[total_elements]  fp16
    mask_ptr,          # *[mask_elements]  fp16
    out_ptr,           # *[total_elements]  fp16
    total,             # int
    hidden_dim,        # int
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection * gate * mask (mask broadcast over hidden dim)."""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # map flat index to mask element (mask is size B*N*N)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=mask)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Elementwise multiplication proj * gate * mask, where mask is broadcast
    over the hidden dimension. All tensors must be contiguous and on CUDA.
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (math.ceil(total / block_size),)
    mask_flat = mask.view(-1)  # (B*N*N,)

    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],         # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask:        torch.Tensor of shape [B, N, N] (or None)
            - weights:     dict of torch.Tensors containing model parameters
            - config:      dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive stages
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast model weights to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B,N,N,dim) -> (B,N,N,hidden_dim)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6. Fuse mask and gate (if applicable)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core O(N³·H) reduction – batched GEMM via torch.bmm
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape to (B*H, N, N) for batched matrix multiplication
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM: (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)      # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
