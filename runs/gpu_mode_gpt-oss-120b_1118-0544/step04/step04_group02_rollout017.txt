def gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                    total_elements, hidden_dim,
                    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # compute mask idx: each hidden dimension block size = hidden_dim
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the input tensor (B, N, N, C).
2. Cast to FP16 for the compute‑intensive path.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) using the provided FP16 weights.
4. Fuse mask, projection, and gate using a Triton kernel
   (elementwise `proj * gate * mask`).  When no mask is required, a
   simple elementwise multiplication is used.
5. Core O(N³·H) reduction: for each batch and hidden channel compute
   a batched matrix multiplication
       out[b, h] = left[b, h] @ right[b, h].T
   This is performed with torch.bmm which uses cuBLAS tensor‑core
   kernels (FP16×FP16→FP16 with FP32 accumulation).
6. Apply the second LayerNorm, multiply by the output gate and project
   back to the original channel dimension.
7. Cast the result back to the input dtype.

The implementation mixes high‑level PyTorch (layer_norm, linear,
sigmoid) with a custom Triton kernel for the elementwise
proj × gate × mask fusion.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: elementwise proj * gate * mask   (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    proj_ptr,          # *[total_elems]  fp16
    gate_ptr,          # *[total_elems]  fp16
    mask_ptr,          # *[mask_elems]  fp16
    out_ptr,           # *[total_elems]  fp16
    total_elems,       # int, total number of elements in proj/gate
    hidden_dim,        # int, size of hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # map flat index to mask element (broadcast over hidden dimension)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Fused elementwise multiplication of a projection, its gate and a mask.
    Args:
        proj: (B, N, N, H) half
        gate: (B, N, N, H) half
        mask: (B, N, N) half (will be broadcast over H)
    Returns:
        Tensor of same shape as proj (half) = proj * gate * mask
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

# Entry point -----------------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    data = (input_tensor, mask, weights, config)
    Returns: Tensor [bs, seq_len, seq_len, dim] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive stages
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to FP16 (no bias in linear layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,dim) -> (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Fuse mask, projection and gate (Triton) or simple multiplication
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction – batched GEMM (B*H, N, N) using torch.bmm
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape to (B*H, N, N) for batched matrix multiplication
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # compute left @ rightᵀ for each (batch, hidden) pair
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype (typically float32)
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
