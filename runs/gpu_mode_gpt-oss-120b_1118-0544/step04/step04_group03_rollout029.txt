        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape result back to [B,N,N,H]

    out = F.layer_norm(

        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],

    out = F.linear(out, weights["to_out.weight"].to(device=device, dtype=torch.float16))

    # Cast back to original dtype
    out = out.to(orig_dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The kernel computes
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
for every batch b and hidden dimension d.  This is a batched
matrix‑multiplication (left @ rightᵀ) with shape (B*H, N, N) where
H = hidden_dim.  The heavy O(N³·H) reduction is performed by a
tiled half‑precision GEMM kernel written in Triton; all other
operations (layer‑norms, linear projections, gating, masking) are
handled in PyTorch.  The implementation follows the reference
PyTorch TriMul forward pass but fuses linear+gate weights and
executes the cubic reduction fully on the GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (M×K)·(K×N) → (M×N)
    stride_am, stride_ak,              # strides for A (row, col)
    stride_bk, stride_bn,              # strides for B (k‑dim, n‑dim) – B is read transposed
    stride_cm, stride_cn,              # strides for C (row, col)
    stride_ab, stride_bb, stride_cb,   # strides for the batch dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ Bᵀ
    A shape: (batch, M, K)
    B shape: (batch, K, N)  (stored as (batch, N, K) and accessed transposed)
    C shape: (batch, M, N)
    """
    pid = tl.program_id(0)          # tile index over (M, N) plane
    batch = tl.program_id(1)        # batch index (B * hidden_dim)

    # tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # A[i, k]
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Bᵀ[k, j]  (original B is stored as (N, K))
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp16·fp16 → fp32

    # write result
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[:, None],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with model parameters (float32)
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    torch.cuda.empty_cache()  # ensure a clean launch environment
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    orig_dtype = x.dtype
    device = x.device

    # --------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to half precision for the heavy path
    x = x.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 2. Combined projection + gate linear layers (reduce memory traffic)
    # --------------------------------------------------------------------- #
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)           # (2*H, C)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)           # (2*H, C)

    # Linear transforms
    left_comb = F.linear(x, left_comb_w)   # [B, N, N, 2*H]
    right_comb = F.linear(x, right_comb_w) # [B, N, N, 2*H]

    # Split into projection and gate parts
    left_proj = left_comb[..., :hidden_dim]                         # [B,N,N,H]
    left_gate = left_comb[..., hidden_dim:].sigmoid()               # [B,N,N,H]
    right_proj = right_comb[..., :hidden_dim]                       # [B,N,N,H]
    right_gate = right_comb[..., hidden_dim:].sigmoid()             # [B,N,N,H]

    # --------------------------------------------------------------------- #
    # 3. Optional masking (applied before gating, as in the reference)
    # --------------------------------------------------------------------- #
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------------------------- #
    # 4. Apply gates
    # --------------------------------------------------------------------- #
    left = left_proj * left_gate   # [B,N,N,H]
    right = right_proj * right_gate

    # --------------------------------------------------------------------- #
    # 5. Core O(N³·H) reduction via tiled batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------------- #
    B, N, _, H = left.shape  # H == hidden_dim

    # Re‑order to (batch, hidden, N, N) then flatten batch*hidden
    left_t = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_flat = left_t.view(B * H, N, N)   # (batch, M, K)
    right_flat = right_t.view(B * H, N, N) # (batch, K, N) – will be read transposed

    out_flat = torch.empty_like(left_flat)  # (batch, M, N)

    # Strides for kernel
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)   # row stride (M)
    stride_a_k = left_flat.stride(2)   # col stride (K)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # k stride (original last dim = 1)
    stride_b_n = right_flat.stride(1)   # n/j stride (original first dim = N)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Tile sizes – tuned for typical sequence lengths ≤ 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                     # M, N, K (all equal to seq_len)
        stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 6. Hidden‑dim LayerNorm
    # --------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------------- #
    # 7. Output gate (element‑wise) and final projection
    # --------------------------------------------------------------------- #
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate = F.linear(x, out_gate_w).sigmoid()   # [B, N, N, H]

    out = out.to(orig_dtype)

    return out
