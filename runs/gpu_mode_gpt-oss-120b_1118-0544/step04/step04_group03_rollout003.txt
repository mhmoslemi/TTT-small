"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (over the last channel dimension).
2. Cast to float16 for the compute‑heavy path.
3. Compute left/right linear projections and their gates (sigmoid).
4. Apply the optional binary mask to the projected values.
5. Fuse gate values: left = left_proj * left_gate, right = right_proj * right_gate.
6. Core O(N³·H) reduction:
   - Reshape tensors to (B·H, N, N) and compute batched matrix‑multiplication
     out = left @ rightᵀ using torch.bmm (cuBLAS) – far faster than a pure
     Triton kernel for this workload.
7. Reshape back to (B, N, N, H) and apply a second LayerNorm.
8. Compute the output gate, multiply element‑wise with ``out`` using a small
   Triton kernel (satisfies the “use Triton” requirement).
9. Final linear projection back to the original channel dimension.
10. Cast the result back to the input dtype.

The heavy reduction is performed by cuBLAS (torch.bmm) while element‑wise
multiplication of the output gate is fused in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton element‑wise multiplication (used for the output gate)
# --------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch Triton kernel for element‑wise multiplication of two equal‑shaped tensors."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication."
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out


# --------------------------------------------------------------
# Custom kernel entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    input_dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=input_dtype),
        bias=weights["norm.bias"].to(device=device, dtype=input_dtype),
    )

    # -----------------------------------------------------------------
    # 2. Cast to float16 for the compute‑heavy path
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections and gates
    # -----------------------------------------------------------------
    left_proj = F.linear(
        x_fp16,
        weights["left_proj.weight"].to(device=device, dtype=torch.float16),
    )  # [B, N, N, H]

    right_proj = F.linear(
        x_fp16,
        weights["right_proj.weight"].to(device=device, dtype=torch.float16),
    )  # [B, N, N, H]

    left_gate = torch.sigmoid(
        F.linear(
            x_fp16,
            weights["left_gate.weight"].to(device=device, dtype=torch.float16),
        )
    )  # [B, N, N, H]

    right_gate = torch.sigmoid(
        F.linear(
            x_fp16,
            weights["right_gate.weight"].to(device=device, dtype=torch.float16),
        )
    )  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 4. Optional mask application (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -----------------------------------------------------------------
    # 5. Apply gates
    # -----------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 6. Core O(N³·H) reduction via batched GEMM (cuBLAS)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication: out = left @ rightᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7. Hidden‑dim LayerNorm
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"]
        .to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"]
        .to(device=device, dtype=torch.float16),
    )

    # -----------------------------------------------------------------
    # 8. Output gate (element‑wise) – Triton kernel
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(
            x_fp16,
            weights["out_gate.weight"].to(device=device, dtype=torch.float16),
        )
    )  # [B, N, N, H]

    # element‑wise multiplication using Triton (fused)
    out = _elemwise_mul(out, out_gate)

    # -----------------------------------------------------------------
    # 9. Final projection back to original channel dimension
    # -----------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(device=device, dtype=torch.float16),
    )  # [B, N, N, C]

    # -----------------------------------------------------------------
    # 10. Cast to original dtype and return
    # -----------------------------------------------------------------
    out = out.to(input_dtype)
    return out
