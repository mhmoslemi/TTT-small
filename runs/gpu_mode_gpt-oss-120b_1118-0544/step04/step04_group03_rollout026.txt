def custom_kernel(data):
    x, mask, weights, config = data
    bs, N, _, dim = x.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # Step 1 layernorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # cast to float16
    x_norm_fp16 = x_norm.to(torch.float16)
    # Step 2 linear concat for left
    left_comb_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(x.device, dtype=torch.float16)
    right_comb_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(x.device, dtype=torch.float16)
    left_comb = torch.nn.functional.linear(x_norm_fp16, left_comb_w)  # shape (B,N,N,2*H)
    right_comb = torch.nn.functional.linear(x_norm_fp16, right_comb_w)
    # split
    left_proj = left_comb[..., :hidden]
    left_gate = left_comb[..., hidden:].sigmoid()
    right_proj = right_comb[..., :hidden]
    right_gate = right_comb[..., hidden:].sigmoid()
    # apply mask if needed
    if mask is not None and not nomask:
        mask_f = mask.to(x_norm_fp16.dtype).unsqueeze(-1)  # shape (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f
    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Step 4: core reduction using Triton kernel
    B, N, _, H = left.shape
    # permute dims for matmul
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B,H,N,N)
    # flatten batch
    batch = B * H
    left_flat = left_t.view(batch, N, N)
    right_flat = right_t.view(batch, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)  # shape (batch, N, N), fp16

    # compute strides
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)
    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)  # original k dimension (last)
    stride_b_n = right_flat.stride(1)
    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # define block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # or 64 - choose 64? Let's try 64 but ensure smem fits. 128x128x64 may exceed.
    # compute grid
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, batch)
    # launch kernel
    _triton_gemm_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8  # may be tuned automatically
    )
    # reshape
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # hidden layernorm
    out = torch.nn.functional.layer_norm(out, (hidden,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    # out_gate
    out_gate = torch.nn.functional.linear(x_norm_fp16, weights["out_gate.weight"].to(x.device, dtype=torch.float16)).sigmoid()
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(x.device, dtype=torch.float16))
    # cast back to original dtype (float32)
    out = out.to(x.dtype)
    return out

    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,

    batch = tl.program_id(1)

    # compute block indices
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_n = offs_n < N

"""
TriMul (outgoing) forward pass with Triton‑accelerated O(N³·H) reduction.

The heavy contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched GEMM (left @ rightᵀ) where each hidden
dimension is treated as an independent batch element.  A custom Triton
kernel performs the batched matmul in FP16 with FP32 accumulation.  All
other operations (LayerNorms, linear projections, sigmoids, gating and
final projection) are carried out with PyTorch but fused where possible.
The kernel is tiled with BLOCK_M/N/K = 128/128/32 (auto‑tuned) and
uses two program dimensions: (tile_id, batch_id).  The final output is
returned in the same dtype as the input (float32 by default).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton GEMM kernel: computes C = A @ Bᵀ for many batches (A: M×K, B: N×K)
# --------------------------------------------------------------------------- #
@triton.autotune(
    configs=[
        triton.Config(
            {"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 64, "NUM_WARPS": 4},
            num_stages=2,
        ),
        triton.Config(
            {"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 32, "NUM_WARPS": 8},
            num_stages=2,
        ),
        triton.Config(
            {"BLOCK_M": 128, "BLOCK_N": 64, "BLOCK_K": 64, "NUM_WARPS": 8},
            num_stages=2,
        ),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # tensor pointers
    M, N, K,                           # matrix dimensions (all int)
    stride_am, stride_ak,              # A strides (M‑dim, K‑dim)
    stride_bk, stride_bn,              # B strides (K‑dim, N‑dim) – B is read as Bᵀ
    stride_cm, stride_cn,              # C strides (M‑dim, N‑dim)
    stride_ab, stride_bb, stride_cb,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # tile id over (M, N)
    batch = tl.program_id(1)        # batch id (B * hidden)

    # ------------------------------------------------------------------- #
    # tile coordinates
    # ------------------------------------------------------------------- #
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # base pointers for this batch
    # ------------------------------------------------------------------- #
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # loop over K dimension
    # ------------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # A[m, k]
        a = tl.load(
            a_batch
            + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Bᵀ[k, n] = B[n, k]
        b = tl.load(
            b_batch
            + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp16·fp16 → fp32 accumulation

    # ------------------------------------------------------------------- #
    # write result
    # ------------------------------------------------------------------- #
    c = acc.to(tl.float16)
    tl.store(
        c_batch
        + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------------------- #
# Entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------- #
    # unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    device = x.device
    dtype_input = x.dtype
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------- #
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float16)               # work in FP16 for the heavy path

    # ------------------------------------------------------------------- #
    # 2. Linear projections + gates (combined for better memory traffic)
    # ------------------------------------------------------------------- #
    # concatenate proj and gate weights (out_features stacked)
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)

    left_comb = F.linear(x_norm, left_comb_w)    # [B, N, N, 2*H]
    right_comb = F.linear(x_norm, right_comb_w)  # [B, N, N, 2*H]

    # split into projection and gate parts
    left_proj = left_comb[..., :hidden]
    left_gate = left_comb[..., hidden:].sigmoid()
    right_proj = right_comb[..., :hidden]
    right_gate = right_comb[..., hidden:].sigmoid()

    # optional mask (broadcast over hidden dim)
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # apply gates
    left = left_proj * left_gate      # [B, N, N, H]
    right = right_proj * right_gate   # [B, N, N, H]

    # ------------------------------------------------------------------- #
    # 3. Core O(N³·H) reduction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape   # (B, N, N, H)

    # reshape to (batch = B*H, M = N, K = N) for A and (batch, K = N, N) for Bᵀ
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_t.view(B * H, N, N)    # (batch, M, K)
    right_flat = right_t.view(B * H, N, N)  # (batch, K, N) – will be read as Bᵀ
    out_flat = torch.empty_like(left_flat)  # (batch, M, N), fp16

    # strides (in elements)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # last dim is K when read as Bᵀ
    stride_b_n = right_flat.stride(1)   # column dim after transpose

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # tile sizes (tuned for N up to 1024)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        stride_a_m,
        stride_a_k,
        stride_b_k,
        stride_b_n,
        stride_c_m,
        stride_c_n,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 4. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
    )

    # ------------------------------------------------------------------- #
    # 5. Output gate (element‑wise)
    # ------------------------------------------------------------------- #
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate = F.linear(x_norm, out_gate_w).sigmoid()      # [B, N, N, H]
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 6. Final projection back to original channel dimension
    # ------------------------------------------------------------------- #
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)   # [B, N, N, C]

    # ------------------------------------------------------------------- #
    # 7. Cast to original dtype and return
    # ------------------------------------------------------------------- #
    out = out.to(dtype_input)
    return out
