import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, total_elems, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out
```

    x = x.to(torch.float16)

        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
    
    # Core reduction via batched GEMM (left @ right^T) per hidden dim
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_reshaped = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    out = out_reshaped.reshape(B, H, N, N).permute(0, 2, 3, 1)   # (B,N,N,H)

        (hidden_dim,),

        bias=to_out_norm_b,
    )

    out = out * out_gate

    out = torch.nn.functional.linear(out, to_out_w)

    return out

"""
TriMul (outgoing) forward‑pass – Triton‑fused gate‑mask multiplication.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) in FP32 and cast to FP16.
2. Four linear projections (left/right + their gates) are performed in FP16.
3. If a mask is present, a custom Triton kernel fuses
   `proj * gate * mask` for each element, broadcasting the mask over the
   hidden dimension.
4. Core O(N³·H) reduction:
   for each hidden channel d we need  out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].
   This is realised as a batched GEMM:
   (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N) which is highly optimized
   in cuBLAS/Tensor‑cores.
5. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate and a final linear projection back to the original channel dimension.
6. Cast the result back to the input dtype.

Only the element‑wise gate‑mask multiplication is implemented in Triton;
all heavy matrix‑multiplications use PyTorch’s optimized kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate * mask) on a flattened view.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32   – total number of elements in proj/gate
    hidden_dim,        # i32   – H (used to broadcast mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds
    mask = offs < total_elems

    # mask is broadcast over the hidden dimension:
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)   # fp16
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)   # fp16
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # fp16

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using the Triton kernel.
    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dictionary with model parameters (float32)
            - config: dictionary with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # Input LayerNorm (FP32) → cast to FP16 for compute‑heavy ops
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Cast model weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # Fuse mask, projection and gate
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·H) reduction via batched GEMM:
    #   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_reshaped  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).reshape(B * H, N, N)
    # batched matmul with transpose on the second operand
    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))
    # reshape back to (B, N, N, H)
    out = out_reshaped.reshape(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # Second LayerNorm on hidden dimension and output gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # Cast back to input dtype if necessary
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
