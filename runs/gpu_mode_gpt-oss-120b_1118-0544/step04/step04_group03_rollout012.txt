def triton_gemm_outgoing(A_ptr, B_ptr, C_ptr,
                         M, N, K,
                         stride_am, stride_ak,
                         stride_bk, stride_bn,
                         stride_cm, stride_cn,
                         stride_ab, stride_bb, stride_cb,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    batch = tl.program_id(1)

    # compute tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_ptr = A_ptr + batch * stride_ab
    b_ptr = B_ptr + batch * stride_bb
    c_ptr = C_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M x K)
        a = tl.load(a_ptr + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B stored as (N, K) (since we want B^T). In our input we have B shape (N, K), we need B^T(k,n) = B[n,k].

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the PyTorch reference but fuses the O(N³·H) reduction
with a custom Triton batched GEMM kernel (C = A @ Bᵀ).  Linear projections,
gating, and LayerNorms are performed with torch.nn.functional for simplicity.
All heavy arithmetic is carried out in float16 with float32 accumulation,
and the final result is cast back to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton batched GEMM kernel (C = A @ Bᵀ)
# A : (batch, M, K)   – left tensor
# B : (batch, N, K)   – right tensor (treated as K×N in the kernel)
# C : (batch, M, N)
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to A, B, C
    M, N, K,                           # matrix dims
    stride_am, stride_ak,              # strides for A (M‑dim, K‑dim)
    stride_bk, stride_bn,              # strides for B (K‑dim, N‑dim) – B is stored as (N,K)
    stride_cm, stride_cn,              # strides for C (M‑dim, N‑dim)
    stride_ab, stride_bb, stride_cb,   # strides for batch dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # tile index over (M,N)
    batch = tl.program_id(1)        # batch index (B * hidden_dim)

    # tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile (M×K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile interpreted as (K×N) -> we load Bᵀ
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) · (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # store result
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------------------- #
# Entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    device = x.device
    orig_dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # Cast model weights to half precision (same device as x)
    # ------------------------------------------------------------------- #
    dtype = torch.float16
    norm_w = weights["norm.weight"].to(device=device, dtype=dtype)
    norm_b = weights["norm.bias"].to(device=device, dtype=dtype)

    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=dtype)

    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype)

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # Work in float16 for the heavy path
    x = x.to(dtype)

    # ------------------------------------------------------------------- #
    # 2. Linear projections + gates
    # ------------------------------------------------------------------- #
    left_proj = F.linear(x, left_proj_w)            # [B, N, N, H]
    right_proj = F.linear(x, right_proj_w)          # [B, N, N, H]

    # Apply mask before gating if required
    if mask is not None and not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)      # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))

    left = left_proj * left_gate    # [B, N, N, H]
    right = right_proj * right_gate # [B, N, N, H]

    # ------------------------------------------------------------------- #
    # 3. Core O(N³·H) reduction via batched GEMM (C = left @ rightᵀ)
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # reshape to (batch = B*H, M = N, K = N) for both operands
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_t.view(B * H, N, N)    # (batch, M, K)
    right_flat = right_t.view(B * H, N, N)  # (batch, N, K) – we'll treat as K×N in kernel

    out_flat = torch.empty_like(left_flat)  # (batch, M, N)

    # strides for the kernel
    stride_am = left_flat.stride(1)   # M‑dim step (N)
    stride_ak = left_flat.stride(2)   # K‑dim step (1)

    stride_bk = right_flat.stride(2)  # K‑dim step (1) – for transposed view
    stride_bn = right_flat.stride(1)  # N‑dim step (N)

    stride_cm = out_flat.stride(1)    # M‑dim step (N)
    stride_cn = out_flat.stride(2)    # N‑dim step (1)

    stride_ab = left_flat.stride(0)   # batch stride for A
    stride_bb = right_flat.stride(0)  # batch stride for B
    stride_cb = out_flat.stride(0)    # batch stride for C

    # Blocking parameters (tuned for typical seq_len ≤ 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                     # M, N, K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ab,
        stride_bb,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 4. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ------------------------------------------------------------------- #
    # 5. Output gate (element‑wise) and final projection
    # ------------------------------------------------------------------- #
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B, N, N, H]
    out = out * out_gate

    out = F.linear(out, to_out_w)   # back to [B, N, N, C]

    # Cast back to original dtype
    out = out.to(orig_dtype)

    return out
