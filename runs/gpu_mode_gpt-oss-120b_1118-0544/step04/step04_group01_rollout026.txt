import triton
import triton.language as tl

@triton.jit
def gate_mask_fusion_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                            total_elems, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # compute mask index: offset // hidden_dim
    mask_idx = offs // hidden_dim

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                      total_elems, hidden_dim,
                      BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # broadcast mask over hidden_dim
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse projection * gate * mask using a Triton kernel.
    All tensors must be contiguous fp16 on the same CUDA device.
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Arguments:
        data: (input, mask, weights, config)
            - input: torch.Tensor, shape [B, N, N, C]
            - mask: torch.Tensor, shape [B, N, N] (may be None)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> fp16)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast weights to fp16 for compute‑heavy ops
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5. Optional mask + gate fusion (Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core O(N³·H) reduction via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    which is a matmul of (N×N) × (N×N) for each hidden dim.
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) for batched matmul
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # batched matrix multiplication: (B*H, N, N) x (B*H, N, N)^T
    # Use torch.matmul which dispatches to cuBLAS/TensorCore for fp16.
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))
    # back to (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7. Second LayerNorm + output gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```

"""
Outgoing Triangle Multiplicative Update (TriMul) – forward pass.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) using the provided LayerNorm
   parameters (float32).  Cast the result to fp16 for all compute‑heavy ops.
2. Cast all model weights that participate in the heavy computation to fp16.
3. Compute four linear projections of the normalized tensor:
      * left_proj, right_proj  – hidden dimension H
      * left_gate, right_gate, out_gate – hidden dimension H, followed by sigmoid.
4. If a mask is supplied (and not ignored), fuse `proj * gate * mask` with a
   Triton kernel that broadcasts the mask over the hidden dimension.
   Otherwise perform a simple element‑wise multiplication.
5. Core O(N³·H) reduction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication of shape (B·H, N, N) × (B·H, N, N)ᵀ
   performed by `torch.matmul`, which utilizes cuBLAS/Tensor‑cores for fp16.
6. Apply a second LayerNorm on the hidden dimension, multiply by `out_gate`,
   and project back to the original channel dimension with a final linear.
7. Cast the result back to the input dtype (typically float32).

A Triton kernel is used for the mask‑gate‑projection fusion; all other
operations rely on PyTorch’s highly‑optimized kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_kernel(
    proj_ptr,      # fp16 * (total_elems,)
    gate_ptr,      # fp16 * (total_elems,)
    mask_ptr,      # fp16 * (mask_elems,)
    out_ptr,       # fp16 * (total_elems,)
    total_elems,   # i32 – total number of elements in proj/gate (B*N*N*H)
    hidden_dim,    # i32 – H, used for broadcasting mask
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask index: each consecutive hidden_dim elements share the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using the Triton kernel above.
    All tensors must be contiguous fp16 on the same CUDA device.
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],      # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    dtype_in = input_tensor.dtype
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → fp16
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(dtype_in)
    norm_b = weights["norm.bias"].to(dtype_in)
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_w, bias=norm_b
    )
    x = x.to(torch.float16)  # heavy work in fp16

    # ------------------------------------------------------------------
    # 2. Cast model weights to fp16 for compute‑intensive ops
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)    # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5. Mask & gate fusion (Triton) or simple element‑wise multiplication
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core O(N³·H) reduction via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) for batched matmul
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T  → (B*H, N, N)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))

    # back to (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7. Second LayerNorm on hidden dimension and output gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
