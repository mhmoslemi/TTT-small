"""
TriMul (outgoing) forward pass – Triton‑fused mask‑gate and
batched half‑precision GEMM reduction.

Algorithm
---------
1. Layer‑normalize the input tensor `x` (shape B,N,N,C) using the supplied
   weight and bias (cast to fp16).  Cast the result to fp16 for the
   compute‑heavy part.
2. Linear projections:
      left  = x @ left_proj.weight.T        (B,N,N,H)
      right = x @ right_proj.weight.T       (B,N,N,H)
   Compute gate values with separate linear layers followed by sigmoid.
3. Fuse element‑wise `proj * gate * mask` (if a mask is present) in a
   Triton kernel to avoid an extra memory pass.
4. Core reduction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication for every hidden channel `d`.
   We reshape to (B*H,N,N) and use `torch.bmm` which runs on cuBLAS/Tensor
   Cores in fp16.
5. Apply a second LayerNorm on the hidden dimension, multiply by the
   output gate, and a final linear projection back to the original channel
   dimension.
6. Cast the result back to the input dtype (float32).

All heavy arithmetic is performed in FP16; only the final result is
converted back to the original dtype.  The mask‑gate fusion kernel is the
required Triton component.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # *fp16  (B*N*N*H,)
    gate_ptr,          # *fp16  (B*N*N*H,)
    mask_ptr,          # *fp16  (B*N*N,)
    out_ptr,           # *fp16  (B*N*N*H,)
    total_elems,       # i32    total number of elements in proj/gate
    hidden_dim,        # i32    H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask_valid = offs < total_elems

    # Compute mask index (each mask element repeats `hidden_dim` times)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask_valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask_valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=mask_valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 4096) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using a Triton kernel.
    All tensors must be fp16, contiguous and on the same CUDA device.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],               # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    # Cast LayerNorm parameters to fp16
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )
    x = x.to(torch.float16)  # fp16 for the rest

    # ----------------------- cast linear weights -------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections -------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------- gates -----------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ----------------------- mask & gate fusion -------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------- core O(N³·H) reduction via batched GEMM -------
    # Reshape to (B*H, N, N) for batched matrix multiply
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_mat = left_perm.view(B * hidden_dim, N, N)            # (B*H, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N).transpose(1, 2)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------ cast back --------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
