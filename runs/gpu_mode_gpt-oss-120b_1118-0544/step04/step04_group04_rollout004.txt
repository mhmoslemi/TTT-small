"""
Optimized outgoing TriMul forward pass.

Algorithm:
1. LayerNorm over the input channels (float32).
2. Cast to fp16 and perform a *single* batched linear projection that
   concatenates all five required projections (left, right, left_gate,
   right_gate, out_gate).  This reduces the number of GEMM calls from 5 to 1.
3. Apply optional mask to left/right projections.
4. Apply sigmoid gates to left_gate, right_gate and out_gate.
5. Fuse left/right projections with their gates.
6. Compute the core O(N³) contraction using a single torch.einsum:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   This avoids costly permutes and copies.
7. LayerNorm over the hidden dimension.
8. Multiply with out_gate using a tiny Triton element‑wise kernel.
9. Final linear projection back to the original dimensionality and cast to float32.

Only the final element‑wise multiplication is implemented in Triton,
meeting the requirement of having at least one custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    """Element‑wise out = out * gate for flattened tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out * gate, mask=mask)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float). Ignored if config["nomask"] is True.
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    eps = 1e-5

    # -------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim] float32

    # -------------------------------------------------
    # 2. Cast to fp16 and combine all linear projections
    #    into one GEMM.
    # -------------------------------------------------
    x_fp16 = x.to(torch.float16)  # [B, N, N, dim]

    # gather weights for the five linear layers (no bias)
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    # concatenate along the output‑feature dimension
    combined_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    # single linear projection
    proj_all = F.linear(x_fp16, combined_w)  # [B, N, N, 5*hidden_dim]

    # split the concatenated result
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=-1
    )  # each: [B, N, N, hidden_dim]

    # -------------------------------------------------
    # 3. Optional mask (applied before gating)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(device).to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 4. Gating (sigmoid)
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate_sig = torch.sigmoid(out_gate)

    # -------------------------------------------------
    # 5. Fuse projections with gates
    # -------------------------------------------------
    left = left_proj * left_gate      # [B, N, N, hidden]
    right = right_proj * right_gate   # [B, N, N, hidden]

    # -------------------------------------------------
    # 6. Core O(N³) contraction via einsum
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -------------------------------------------------
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)  # [B, N, N, hidden]

    # -------------------------------------------------
    # 7. Output LayerNorm over hidden dimension
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # [B, N, N, hidden] fp16

    # -------------------------------------------------
    # 8. Multiply with out_gate (element‑wise) via Triton
    # -------------------------------------------------
    total_el = out.numel()
    BLOCK = 1024
    grid = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        out,
        out_gate_sig,
        total_el,
        BLOCK=BLOCK,
    )

    # -------------------------------------------------
    # 9. Final linear projection back to original dim
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)  # (dim, hidden)
    out = F.linear(out, to_out_w)  # [B, N, N, dim] fp16

    # Cast to float32 as required by the interface
    return out.to(torch.float32)
