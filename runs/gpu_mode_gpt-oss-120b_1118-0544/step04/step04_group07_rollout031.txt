### Triton LayerNorm Kernel Implementation

Goal: For each row (size C), compute normalized output = (x - mean) / sqrt(var + eps) * weight + bias.

We will implement using two passes: first pass compute sum and sumsq. second pass compute normalized values.

Observe: In Triton, we cannot allocate dynamic arrays; we use fixed BLOCK per thread.

We'll use `num_iters = tl.cdiv(C, BLOCK)` for loop.

Implementation details:

```
@triton.jit
def layernorm_kernel(
    X_ptr, Y_ptr,          # pointers to input and output, shape (rows, C)
    weight_ptr, bias_ptr,  # shape (C,)
    eps,                   # scalar
    C: tl.constexpr,
    BLOCK: tl.constexpr,
):
    row = tl.program_id(0)
    # compute base offset:
    base = row * C
    # compute sum and sumsq
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)
    num_iters = tl.cdiv(C, BLOCK)
    for i in range(num_iters):
        col = i * BLOCK + tl.arange(0, BLOCK)
        mask = col < C
        x = tl.load(X_ptr + base + col, mask=mask, other=0.0)
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32)
        sum_sq += tl.sum(x_f32 * x_f32)
    # compute mean and inv_std
    mean = sum_val / C
    var = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # second pass
    for i in range(num_iters):
        col = i * BLOCK + tl.arange(0, BLOCK)
        mask = col < C
        x = tl.load(X_ptr + base + col, mask=mask, other=0.0)
        x_f32 = x.to(tl.float32)
        w = tl.load(weight_ptr + col, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + col, mask=mask, other=0.0).to(tl.float32)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operation.
    The heavy reductions are performed by cuBLAS batched GEMM,
    while both input and output LayerNorms are implemented in pure Triton.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- Input LayerNorm (Triton) ----------
    x_flat = input_tensor.contiguous().view(-1, dim)  # shape (rows, dim)
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias = weights["norm.bias"].contiguous()
    x_norm_flat = layernorm_triton(x_flat, norm_weight, norm_bias, eps=eps, block=128)
    x_norm = x_norm_flat.view_as(input_tensor)  # (B,N,N,dim)

    # ---------- Cast to fp16 for compute ----------
    x_fp16 = x_norm.to(torch.float16)

    # ---------- Linear projections ----------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16).contiguous()
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16).contiguous()
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16).contiguous()
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16).contiguous()
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16).contiguous()
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16).contiguous()

    # projections
    left = F.linear(x_fp16, left_proj_w)   # (B,N,N,H)
    right = F.linear(x_fp16, right_proj_w)  # (B,N,N,H)

    # gates
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    # mask handling
    if mask is not None and not config.get("nomask", False):
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left = left * left_gate
        right = right * right_gate

    # ---------- Core O(N³·H) via batched GEMM ----------
    B, N, _, H = left.shape
    # reshape for batched matmul
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H,N,N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H,N,N)

    # batch matmul: left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H,N,N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- Output LayerNorm (Triton) ----------
    out_flat = out.view(-1, H)
    out_norm_flat = layernorm_triton(out_flat, to_out_norm_w, to_out_norm_b, eps=eps, block=128)
    out_norm = out_norm_flat.view_as(out)

    # ---------- Apply output gate ----------
    out_norm = out_norm.to(torch.float16) * out_gate

    # ---------- Final linear projection ----------
    out_final = F.linear(out_norm, to_out_w)  # (B,N,N,dim)

    # Cast back to original dtype
    if out_final.dtype != dtype_in:
        out_final = out_final.to(dtype_in)

    return out_final

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

The implementation follows the reference PyTorch module but replaces the
LayerNorms with a pure Triton kernel, keeping the heavy O(N³·H) reduction
as a batched GEMM (torch.bmm). All linear layers and the einsum are
executed in fp16 to exploit tensor‑cores, while the final output is
returned in the input dtype (float32 by default). The kernel fuses the
mean/variance computation and the affine transformation of LayerNorm.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _layernorm_kernel(
    X_ptr, Y_ptr,            # [rows, C] input & output (flattened)
    weight_ptr, bias_ptr,    # [C] affine parameters
    eps,                     # scalar epsilon
    C: tl.constexpr,         # channel dimension (compile‑time)
    BLOCK: tl.constexpr,     # block size for the inner loop (compile‑time)
):
    """
    Row‑wise LayerNorm:
        y = (x - mean) / sqrt(var + eps) * weight + bias

    Each program instance processes one row (i.e. a vector of length C).
    The reduction over C is performed with `tl.sum`, which aggregates the
    values across the BLOCK threads of the program.
    """
    row = tl.program_id(0)                     # which row we are normalising
    base = row * C                              # offset of the row start

    # ---- first pass: compute mean and variance ----
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)
    num_iters = tl.cdiv(C, BLOCK)               # number of blocks needed

    for i in range(num_iters):
        col = i * BLOCK + tl.arange(0, BLOCK)   # column indices inside the row
        mask = col < C
        x = tl.load(X_ptr + base + col, mask=mask, other=0.0)
        xf = x.to(tl.float32)
        sum_val += tl.sum(xf)                   # sum over the BLOCK threads
        sum_sq  += tl.sum(xf * xf)              # sum of squares

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ---- second pass: write normalised output with affine transformation ----
    for i in range(num_iters):
        col = i * BLOCK + tl.arange(0, BLOCK)
        mask = col < C
        x = tl.load(X_ptr + base + col, mask=mask, other=0.0)
        xf = x.to(tl.float32)

        w = tl.load(weight_ptr + col, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + col,   mask=mask, other=0.0).to(tl.float32)

        y = ((xf - mean) * inv_std) * w + b
        tl.store(Y_ptr + base + col, y.to(x.dtype), mask=mask)


def _layernorm_triton(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor,
                      eps: float = 1e-5, block: int = 128) -> torch.Tensor:
    """
    Helper that launches the Triton LayerNorm kernel.
    `x` must be 2‑D (rows, C) and contiguous.
    """
    rows, C = x.shape
    y = torch.empty_like(x)
    grid = (rows,)
    _layernorm_kernel[grid](
        x, y,
        weight, bias,
        eps,
        C,
        BLOCK=block,
    )
    return y


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using Triton for LayerNorms.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] with the same dtype as input.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # -------------------------------------------------
    # 1. Input LayerNorm (Triton)
    # -------------------------------------------------
    x_flat = input_tensor.contiguous().view(-1, dim)               # (B*N*N, dim)
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias   = weights["norm.bias"].contiguous()
    x_norm_flat = _layernorm_triton(x_flat, norm_weight, norm_bias,
                                   eps=eps, block=128)
    x_norm = x_norm_flat.view_as(input_tensor)                     # (B,N,N,dim)

    # -------------------------------------------------
    # 2. Cast to fp16 for the compute‑heavy part
    # -------------------------------------------------
    x_fp16 = x_norm.to(torch.float16)

    # -------------------------------------------------
    # 3. Linear projections and gates (fp16)
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16).contiguous()
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16).contiguous()
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16).contiguous()
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16).contiguous()
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16).contiguous()
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16).contiguous()

    # projections
    left  = F.linear(x_fp16, left_proj_w)    # (B, N, N, hidden_dim)
    right = F.linear(x_fp16, right_proj_w)   # (B, N, N, hidden_dim)

    # gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # (B, N, N, hidden_dim)

    # -------------------------------------------------
    # 4. Apply mask (if present) and fuse with gates
    # -------------------------------------------------
    if mask is not None and not nomask:
        # mask shape (B, N, N) -> broadcast over hidden_dim
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left  = left * left_gate
        right = right * right_gate

    # -------------------------------------------------
    # 5. Core O(N³·H) reduction via batched GEMM
    # -------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # left @ rightᵀ  (equivalent to the Einstein sum)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))       # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 6. Output LayerNorm (Triton)
    # -------------------------------------------------
    out_flat = out.view(-1, H)
    out_norm_flat = _layernorm_triton(out_flat, to_out_norm_w, to_out_norm_b,
                                     eps=eps, block=128)
    out_norm = out_norm_flat.view_as(out)          # (B,N,N,H)

    # -------------------------------------------------
    # 7. Apply output gate and final projection
    # -------------------------------------------------
    out_norm = out_norm.to(torch.float16) * out_gate
    out_final = F.linear(out_norm, to_out_w)       # (B,N,N,dim)

    # -------------------------------------------------
    # 8. Cast back to original dtype
    # -------------------------------------------------
    if out_final.dtype != dtype_in:
        out_final = out_final.to(dtype_in)

    return out_final
