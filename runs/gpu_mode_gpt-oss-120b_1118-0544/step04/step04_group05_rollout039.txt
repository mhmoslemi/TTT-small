def _tri_mul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    assert A.shape == B.shape
    batch, N, K = A.shape
    assert N == K
    device = A.device
    # Block sizes heuristics
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
        num_warps = 2
    elif N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 8
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
              (N + BLOCK_N - 1)//BLOCK_N,
              batch)
    C = torch.empty_like(A)
    # compute strides
    stride_abatch = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)
    stride_bbatch = B.stride(0)
    stride_bi = B.stride(1)
    stride_bk = B.stride(2)
    stride_cbatch = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    _tri_mul_kernel[grid](
        A, B, C,
        N, N, N,
        stride_am, stride_ak,
        stride_bi, stride_bk,
        stride_cm, stride_cn,
        stride_abatch, stride_bbatch, stride_cbatch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )
    return C

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The function implements the forward pass of the TriMul module used in AlphaFold3.
It performs:
  1. LayerNorm on the input tensor.
  2. Linear projections to a hidden dimension (left/right).
  3. Optional binary mask application.
  4. Sigmoid gates (left, right, out) derived from the normalized input.
  5. Gated left/right tensors.
  6. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
     implemented as a batched GEMM (A @ Bᵀ) with a custom Triton kernel
     where the hidden dimension is folded into the batch dimension.
  7. LayerNorm on the hidden dimension, multiplication by out‑gate.
  8. Final linear projection back to the original channel dimension.
All heavy arithmetic is done in FP16 for speed; the final result is cast back
to the original dtype.  At least the O(N³·H) reduction is performed in the
Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ where A and B have shape
# (batch, N, N) and the reduction is over the last dimension (k).
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,                     # matrix dimensions (all equal to seq_len)
    stride_am, stride_ak,        # A: row stride, col stride
    stride_bi, stride_bk,        # B: row (j) stride, col (k) stride
    stride_cm, stride_cn,        # C: row stride, col stride
    stride_ab, stride_bb, stride_cb,   # batch strides for A, B, C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch index (includes hidden dimension)
    pid_m = tl.program_id(0)   # block row index
    pid_n = tl.program_id(1)   # block column index

    # --------------------------------------------------------------
    # Offsets within the block
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Base pointers for this batch
    # --------------------------------------------------------------
    a_batch = pid_b * stride_ab
    b_batch = pid_b * stride_bb
    c_batch = pid_b * stride_cb

    # Pointers to the current tile of A and B
    a_ptrs = A_ptr + a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + b_batch + offs_n[:, None] * stride_bi + offs_k[None, :] * stride_bk

    # Accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):

    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2) Switch to FP16 for the heavy path
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast model weights to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections to hidden dimension
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_w)       # (B, N, N, H)
    right = F.linear(x, right_proj_w)      # (B, N, N, H)

    # --------------------------------------------------------------
    # 5) Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(dtype=x.dtype).unsqueeze(-1)   # (B, N, N, 1)
        left  = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 6) Gates (sigmoid of linear transforms)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # --------------------------------------------------------------
    # 7) Apply gates to projections
    # --------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 8) Fold hidden dimension into batch and run batched GEMM
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N) -> (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Core reduction: out_flat[b,i,j] = Σ_k left_flat[b,i,k] * right_flat[b,j,k]
    out_flat = _tri_mul_triton(left_flat, right_flat)   # (B*H, N, N)

    # --------------------------------------------------------------
    # 9) Reshape back to (B, N, N, H)
    # --------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 10) Output LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 11) Apply output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 12) Final linear projection back to original channel dimension
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # --------------------------------------------------------------
    # 13) Cast back to original dtype if needed
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
