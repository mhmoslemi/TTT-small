"""
TriMul (outgoing) forward pass – Triton‑accelerated reduction.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the provided LN parameters.
2. Cast the normalized tensor to fp16 for the compute‑intensive part.
3. Apply three linear projections (left, right) and three gating projections.
4. Fuse elementwise multiplication of projection, gate and optional mask
   (mask is broadcast over the hidden dimension).
5. Core reduction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply of shape (B*H,N,N) × (B*H,N,N)ᵀ.
   Implemented with a custom Triton kernel that processes all batches in a
   single launch, hiding the overhead of cuBLAS for many batch entries.
6. Reshape to (B,N,N,H), apply a second LayerNorm over the hidden dimension,
   multiply by the output gate and project back to the original channel dimension.
7. Cast the result back to the input dtype (usually float32).

The Triton kernel performs a tiled GEMM with fp16 inputs and fp32 accumulation,
supporting arbitrary sequence lengths (up to the tested 1024) and hidden
dimensions.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) where A,B have shape (batch, N, K)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all int64)
    stride_am, stride_ak, stride_ab,   # A strides: (row, col, batch)
    stride_bk, stride_bn, stride_bb,   # B strides (after B is transposed): (row, col, batch)
    stride_cm, stride_cn, stride_cb,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b,m,n] = Σ_k A[b,m,k] * B[b,n,k]    (B is already transposed)."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ----- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile for current batch
    a_base = a_ptr + batch_id * stride_ab
    b_base = b_ptr + batch_id * stride_bb
    c_base = c_ptr + batch_id * stride_cb

    # Row/col offsets inside the tile (no k offset yet)
    a_tile_ptr = a_base + offs_m[:, None] * stride_am
    b_tile_ptr = b_base + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over K dimension ---------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Global k indices for this tile
        k_offset = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K)
        k_valid = (k_offset + k_range) < K

        # Load A tile (M x K)
        a = tl.load(
            a_tile_ptr + (k_offset + k_range)[None, :] * stride_ak,
            mask=mask_m[:, None] & k_valid[None, :],
            other=0.0,
        )
        # Load B tile (N x K) – note B is stored as (K, N) after transpose
        b = tl.load(
            b_tile_ptr + (k_offset + k_range)[:, None] * stride_bk,
            mask=k_valid[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate
        acc += tl.dot(a, b)

    # Store result, cast back to fp16
    c = acc.to(tl.float16)
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: optional fused mask‑gate multiplication (kept for completeness)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * mval, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """Fuse proj * gate * mask (mask broadcast over hidden_dim) via Triton."""
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C] (float32/float16)
            - mask: [B, N, N] (bool/int/float) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16 (ensure same device)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections and gates (all fp16)
    # ------------------------------------------------------------------
    left_proj   = torch.nn.functional.linear(x, left_proj_w)      # (B,N,N,H)
    right_proj  = torch.nn.functional.linear(x, right_proj_w)

    left_gate_pre  = torch.nn.functional.linear(x, left_gate_w)
    right_gate_pre = torch.nn.functional.linear(x, right_gate_w)
    out_gate_pre   = torch.nn.functional.linear(x, out_gate_w)

    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate = out_gate_pre.sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse projection, gate and optional mask
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        # Broadcast mask over hidden_dim and fuse via Triton for better locality
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core reduction: batched GEMM (left @ rightᵀ) using Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Transpose right so that we compute A @ Bᵀ as A @ (rightᵀ)
    right_T = right_flat.transpose(1, 2).contiguous()

    out_flat = torch.empty_like(left_flat)

    # ----- kernel launch configuration ---------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    batch = B * H
    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Strides (int64) for the three tensors
    stride_am, stride_ak, stride_ab = left_flat.stride(1), left_flat.stride(2), left_flat.stride(0)
    stride_bk, stride_bn, stride_bb = right_T.stride(1), right_T.stride(2), right_T.stride(0)
    stride_cm, stride_cn, stride_cb = out_flat.stride(1), out_flat.stride(2), out_flat.stride(0)

    _batched_gemm_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        N,               # M
        N,               # N
        N,               # K
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to (B,N,N,H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)
    return out
