"""
TriMul (outgoing) forward pass – Triton‑fused gate‑mask and
batched half‑precision GEMM.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) with learnable weight/bias.
2. Cast to fp16 for the heavy compute path.
3. Linear projections (left/right) and their gates are computed with
   fp16 weights; gates are passed through a sigmoid.
4. If a mask is provided (and nomask=False) a small Triton kernel fuses
   `proj * gate * mask` in a single pass; otherwise we use plain
   element‑wise multiplication.
5. The core O(N³·H) reduction is performed as a batched matrix multiply:
   for each hidden dimension h we compute   L_h @ R_hᵀ,
   where L_h,R_h ∈ ℝ^{N×N}.  This is expressed with a single torch.bmm
   over a batch of size B·H, which maps to cuBLAS/Tensor‑core GEMM.
6. A second LayerNorm (over the hidden dimension) and multiplication by
   an output gate are applied.
7. Final linear projection back to the original channel dimension.
8. Cast the result back to the input dtype (normally float32).

Only the gate‑mask fusion is written in Triton; the batched GEMM uses
highly‑optimized cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused projection * gate * mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total_elems,)
    gate_ptr,          # fp16 * (total_elems,)
    mask_ptr,          # fp16 * (mask_elems,)  where mask_elems = total_elems // hidden_dim
    out_ptr,           # fp16 * (total_elems,)
    total_elems,       # i32
    hidden_dim,        # i32
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # index of the mask element (broadcast over hidden dim)
    mask_index = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_index, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse element‑wise multiplication of projection, gate and mask.
    All tensors must be contiguous fp16 on the same CUDA device.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out

# ----------------------------------------------------------------------

                              Dict[str, Any]]) -> torch.Tensor:

    Args:
        data: (input_tensor, mask, weights, config)

            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    dtype_in = input_tensor.dtype

        input_tensor,

    )

    # ----------------------- cast weights --------------------------------
    w_left_proj  = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate  = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate   = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    w_to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out     = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections ---------------------------
    left_proj  = torch.nn.functional.linear(x, w_left_proj)   # [B,N,N,H]
    right_proj = torch.nn.functional.linear(x, w_right_proj)

    # --------------------------- gates -----------------------------------
    left_gate  = torch.nn.functional.linear(x, w_left_gate).sigmoid()
    right_gate = torch.nn.functional.linear(x, w_right_gate).sigmoid()
    out_gate   = torch.nn.functional.linear(x, w_out_gate).sigmoid()   # [B,N,N,H]

    # ----------------------- mask & gate fusion -------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16, block_size=1024)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16, block_size=1024)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------- core O(N³·H) reduction via batched GEMM -------
    # Reshape to (B, H, N, N) and then to (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left.shape

    left_mat = left.view(B * H, N, N)                     # [B*H, N, N]
    right_t = right.transpose(2, 3).contiguous()         # transpose last two dims
    right_mat = right_t.view(B * H, N, N)                 # [B*H, N, N]

    out_mat = torch.bmm(left_mat, right_mat)              # [B*H, N, N]

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm_w,
        bias=w_to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, w_to_out)   # [B, N, N, C]

    # ------------------------ cast back --------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
```
