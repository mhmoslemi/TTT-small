@triton.jit
def matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_am, stride_ak,
    stride_bk, stride_bn,  # B shape (batch, K, N) i.e., stride batch, stride K, stride N
    stride_cm, stride_cn,
    batch, M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute pointers
    a_ptrs = A_ptr + pid_batch * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_batch * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # load accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask = (offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask = (offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        acc += tl.dot(a, b, out_type=tl.float32) # or just using tl.dot
        # shift pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
    # cast down to fp16 if needed?
    c = acc.to(tl.float16)
    # store
    c_ptrs = C_ptr + pid_batch * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask = (offs_m < M)[:, None] & (offs_n < N)[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # LayerNorm
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = torch.nn.functional.layer_norm(input_tensor,
            (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Cast to fp16
    x_fp16 = x.half()

    # Linear weights to fp16
    left_proj_weight = weights["left_proj.weight"].to(device).half()
    right_proj_weight = weights["right_proj.weight"].to(device).half()
    left_gate_weight = weights["left_gate.weight"].to(device).half()
    right_gate_weight = weights["right_gate.weight"].to(device).half()
    out_gate_weight = weights["out_gate.weight"].to(device).half()
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).half()
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).half()
    to_out_weight = weights["to_out.weight"].to(device).half()

    # Projections
    left_proj = torch.nn.functional.linear(x_fp16, left_proj_weight)   # shape (B,N,N,hidden)
    right_proj = torch.nn.functional.linear(x_fp16, right_proj_weight)

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_weight))

    left = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask:
        mask_f = mask.to(device).half().unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # Transform to (batch, hidden, N, N) then flatten batch*hidden
    B, N, _, _ = left.shape
    left_reshaped = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch*hidden dimension
    batch = B * hidden_dim
    left_mat = left_reshaped.view(batch, N, N)
    right_mat = right_reshaped.view(batch, N, N)

    # Allocate out mat
    out_mat = torch.empty_like(left_mat)   # fp16

    # Compute strides
    stride_left_batch, stride_left_m, stride_left_k = left_mat.stride()
    stride_right_batch, stride_right_n, stride_right_k = right_mat.stride()
    stride_out_batch, stride_out_m, stride_out_n = out_mat.stride()

    # Launch Triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch, N, N, N,
        stride_left_batch, stride_left_m, stride_left_k,
        stride_right_batch, stride_right_n, stride_right_k,
        stride_out_batch, stride_out_m, stride_out_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape out back to (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    out = torch.nn.functional.layer_norm(out,
            (hidden_dim,),
            weight=to_out_norm_weight,
            bias=to_out_norm_bias,
            eps=1e-5)

    # out_gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_weight))

    # Apply gate (elementwise)
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, to_out_weight)

    # Cast back to float32
    return out.float()

def tri_mul_kernel(

    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Bounds
    M_end = M
    N_end = N
    K_end = K

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M_end
    mask_n = offs_n < N_end
    mask_k = offs_k < K_end

                     bias=norm_bias,
                     eps=1e-5)
    # Cast to half
    x_h = x.half()

    # Linear weights (FP16)
    left_proj_w = weights["left_proj.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()
    left_gate_w = weights["left_gate.weight"].to(device).half()
    right_gate_w = weights["right_gate.weight"].to(device).half()
    out_gate_w = weights["out_gate.weight"].to(device).half()
    to_out_norm_w = weights["to_out_norm.weight"].to(device).half()
    to_out_norm_b = weights["to_out_norm.bias"].to(device).half()
    to_out_w = weights["to_out.weight"].to(device).half()

    right_proj = F.linear(x_h, right_proj_w)

    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))

    right = right_proj * right_gate

        left = left * mask_f
        right = right * mask_f

    right_t = right.permute(0, 3, 1, 2).contiguous()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

        (N + BLOCK_N - 1) // BLOCK_N,

        out_mat,

        stride_right_batch, stride_right_n, stride_right_k,
        stride_out_batch, stride_out_m, stride_out_n,

"""
Outgoing Triangle Multiplicative Update (TriMul) – fast Triton implementation.

The kernel computes, for every batch `b` and hidden channel `d`:
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are the projected and gated pair representations.
The computation is a batched GEMM (left @ rightᵀ) that is performed entirely
in FP16 with Tensor‑Core acceleration.  A small amount of surrounding work
(layer‑norms, gating, final linear projection) is kept in PyTorch because it
is cheap compared to the O(N³) core.  The implementation avoids the costly
permutes of the reference code by fusing the batch × hidden dimension into
the first program‑id dimension of the Triton kernel.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # FP16, shape (B, M, K)
    right_ptr,              # FP16, shape (B, N, K) – we load it as (K, N)
    out_ptr,                # FP16, shape (B, M, N)
    batch, M, N, K,         # integer sizes
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: out = left @ rightᵀ  (all tensors in FP16)."""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Starting indices of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds handling
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers to the current tile of left (M×K) and right (K×N)
    a_ptrs = (left_ptr
              + pid_batch * stride_left_batch
              + offs_m[:, None] * stride_left_m
              + offs_k[None, :] * stride_left_k)
    b_ptrs = (right_ptr
              + pid_batch * stride_right_batch
              + offs_k[:, None] * stride_right_k   # K dimension
              + offs_n[None, :] * stride_right_n)  # N dimension

    # Accumulator in FP32 to preserve precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Tensor‑core dot (FP16 inputs, FP32 accumulator)
        acc += tl.dot(a, b, out_type=tl.float32)

        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # Write the result back in FP16
    c = acc.to(tl.float16)
    c_ptrs = (out_ptr
              + pid_batch * stride_out_batch
              + offs_m[:, None] * stride_out_m
              + offs_n[None, :] * stride_out_n)

    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N]; may be ignored if config["nomask"]
            - weights: dictionary of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim], float32

    # Cast to FP16 for the heavy arithmetic
    x_h = x.half()

    # ------------------------------------------------------------------
    # 2. Linear projections and gates (FP16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()
    left_gate_w = weights["left_gate.weight"].to(device).half()
    right_gate_w = weights["right_gate.weight"].to(device).half()
    out_gate_w = weights["out_gate.weight"].to(device).half()

    left_proj = F.linear(x_h, left_proj_w)    # (B, N, N, hidden)
    right_proj = F.linear(x_h, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(device).half().unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 4. Batched GEMM via Triton  (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Bring hidden dimension to the batch axis
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    batch = B * hidden_dim
    left_mat = left_t.view(batch, N, N)    # (batch, M=N, K=N)
    right_mat = right_t.view(batch, N, N)  # (batch, N, K=N)

    out_mat = torch.empty_like(left_mat)   # (batch, N, N), fp16

    # Strides (in elements, not bytes)
    stride_la, stride_lm, stride_lk = left_mat.stride()
    stride_rb, stride_rn, stride_rk = right_mat.stride()
    stride_co, stride_cm, stride_cn = out_mat.stride()

    # Tile sizes – chosen to hit Tensor Cores well
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch, N, N, N,
        stride_la, stride_lm, stride_lk,
        stride_rb, stride_rn, stride_rk,
        stride_co, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to the original 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Post‑processing (LayerNorm, gating, final projection)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).half()
    to_out_norm_b = weights["to_out_norm.bias"].to(device).half()
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # (B, N, N, hidden), fp16

    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))  # (B, N, N, hidden), fp16
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).half()
    out = F.linear(out, to_out_w)  # (B, N, N, dim), fp16

    # Return float32 as required by the interface
    return out.float()
