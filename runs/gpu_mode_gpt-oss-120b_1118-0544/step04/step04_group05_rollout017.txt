"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1️⃣ Layer‑normalize the input tensor [B, N, N, C].
2️⃣ Project to hidden dimension H with two linear layers (left/right).
3️⃣ Apply optional mask (broadcast over H).
4️⃣ Compute three sigmoid gates (left, right, out) from the normalized input.
5️⃣ Multiply left/right by their gates.
6️⃣ Fold H into the batch dimension and compute the core reduction
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is a batched GEMM C = A @ Bᵀ with A,B shaped (B·H, N, N).
   The heavy O(N³·H) work is performed by a custom Triton kernel.
7️⃣ Reshape back to (B, N, N, H), layer‑norm over H, multiply by the output gate,
   and linearly project back to the original channel dimension C.
8️⃣ Return the result in the original dtype (float32 or float16).

The kernel uses fp16 arithmetic with fp32 accumulation for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# A, B : (batch, M, K) and (batch, N, K) respectively.
# C    : (batch, M, N)
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix dimensions
    stride_ah, stride_am, stride_ak,    # A strides (batch, row, col)
    stride_bh, stride_bk, stride_bn,    # B strides (batch, k, n)  <-- B is treated as transposed
    stride_ch, stride_cm, stride_cn,    # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch index (B*H)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column

    # ---------------------------------------------------------
    # Offsets for this program
    # ---------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---------------------------------------------------------
    # Pointers to the start of the tile
    # ---------------------------------------------------------
    a_ptrs = A_ptr + pid_b * stride_ah + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + pid_b * stride_bh + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # ---------------------------------------------------------
    # Accumulator
    # ---------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------
    # Main loop over K dimension
    # ---------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        k_off = k * BLOCK_K
        cur_k = tl.arange(0, BLOCK_K) + k_off
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ---------------------------------------------------------
    # Write back
    # ---------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + pid_b * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for tensors of shape (batch, N, N) using Triton.
    A, B must be half‑precision and contiguous on CUDA.
    """
    assert A.shape == B.shape, "A and B must have the same shape"
    batch, N, K = A.shape
    assert N == K, "TriMul expects square sequence dimension"

    # Heuristic block sizes tuned for typical N values
    if N <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
        num_warps = 2
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks in M dimension
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks in N dimension
        batch,                          # batch (B * H)
    )

    C = torch.empty_like(A)

    _batched_matmul_kernel[grid](
        A, B, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),          # A: batch, row, col
        B.stride(0), B.stride(2), B.stride(1),          # B as Bᵀ: batch, k, n
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )
    return C

# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor [B, N, N] or None (binary)
            - weights      : dict of model parameters (float32 tensors)
            - config       : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, if True ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Work in half precision for the expensive part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Cast all linear weights to half precision
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections to hidden_dim
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5️⃣ Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Apply gates to projections
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Fold hidden dimension into batch for GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    left  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8️⃣ Core O(N³·H) reduction via Triton batched GEMM
    # ------------------------------------------------------------------
    # Compute C = left @ rightᵀ   (right is treated as transposed inside the kernel)
    out_flat = _batched_matmul_triton(left, right)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 9️⃣ Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 🔟 Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 1️⃣2️⃣ Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 1️⃣3️⃣ Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
