@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    size,   # N
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to start of batch
    a_batch_ptr = a_ptr + batch_id * batch_stride_a
    b_batch_ptr = b_ptr + batch_id * batch_stride_b
    c_batch_ptr = c_ptr + batch_id * batch_stride_c

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(size, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # mask for k dimension
        mask_k = cur_k < size

        # Load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * size) + cur_k[None, :]
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < size) & mask_k[None, :], other=0.0)

        # Load B_T: shape (BLOCK_K, BLOCK_N)
        # We want B_T[k, n] = B[n, k]
        b_ptrs = b_batch_ptr + (offs_n[None, :] * size) + cur_k[:, None]
        b = tl.load(b_ptrs, mask=mask_k[:, None] & (offs_n[None, :] < size), other=0.0)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * size) + offs_n[None, :]
    mask = (offs_m[:, None] < size) & (offs_n[None, :] < size)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the AlphaFold3 TriMul module but optimizes the
core O(N³) operation with a custom Triton kernel that computes a batched
matrix multiplication:
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
where each hidden channel `d` is treated as an independent batch.  The
remaining linear projections, gating, and LayerNorms are performed with
PyTorch (FP16) and a small Triton kernel is used for the final element‑wise
gate multiplication.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    """In‑place out = out * gate for flattened tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    tl.store(out_ptr + offs, out * gate, mask=mask)


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    size,                     # N (seq length)
    BLOCK_M: tl.constexpr,    # tile size in M dimension (i)
    BLOCK_N: tl.constexpr,    # tile size in N dimension (j)
    BLOCK_K: tl.constexpr,    # tile size in reduction (k)
):
    """Compute C = A @ Bᵀ for a batch of square matrices.

    A, B, C shape per batch: (size, size).  The batch dimension is
    encoded in the program_id(0).  All tensors are stored row‑major.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers to the start of the current batch
    a_batch = a_ptr + batch_id * batch_stride_a
    b_batch = b_ptr + batch_id * batch_stride_b
    c_batch = c_ptr + batch_id * batch_stride_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(size, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < size

        # ----- Load A (M x K) -----
        a_ptrs = a_batch + (offs_m[:, None] * size) + cur_k[None, :]
        mask_a = (offs_m[:, None] < size) & mask_k[None, :]
        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16)

        # ----- Load Bᵀ (K x N) -----
        # Bᵀ[k, n] = B[n, k]  (original B is row‑major)
        b_ptrs = b_batch + (offs_n[None, :] * size) + cur_k[:, None]
        mask_b = mask_k[:, None] & (offs_n[None, :] < size)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16)

        # Accumulate
        acc += tl.dot(a, b)   # result shape (BLOCK_M, BLOCK_N) -> promoted to float32 via acc

    # ----- Store result -----
    c_ptrs = c_batch + (offs_m[:, None] * size) + offs_n[None, :]
    mask_c = (offs_m[:, None] < size) & (offs_n[None, :] < size)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor shape [B, N, N] (bool/float) – may be ignored if config["nomask"] is True
            - weights: dict of model weight tensors (float32)
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  fp32

    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    left_proj = F.linear(x_fp16, left_proj_weight)    # (B, N, N, hidden)
    right_proj = F.linear(x_fp16, right_proj_weight) # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply to projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))   # (B, N, N, hidden)
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))

    left = left_proj * left_gate   # (B, N, N, hidden)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Batched GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a Triton batched GEMM (A @ Bᵀ) where each hidden
    #    channel is treated as an independent batch.
    # ------------------------------------------------------------------
    # Rearrange to (batch, N, N) where batch = B * hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(-1, N, N)   # (B*hidden, N, N)
    right_mat = right_perm.view(-1, N, N)

    # Allocate output tensor
    out_mat = torch.empty_like(left_mat)   # (B*hidden, N, N), fp16

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    batch = left_mat.shape[0]  # B * hidden_dim
    grid = (batch, tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N))

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        left_mat.stride(0),
        right_mat.stride(0),
        out_mat.stride(0),
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6. to_out_norm (LayerNorm over hidden)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # (B, N, N, hidden)  fp16

    # ------------------------------------------------------------------
    # 7. out_gate (linear + sigmoid) and element‑wise multiply (Triton)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))   # (B, N, N, hidden) fp16

    total_elements = out.numel()
    BLOCK_EL = 1024
    grid_el = ( (total_elements + BLOCK_EL - 1) // BLOCK_EL, )
    elementwise_mul_kernel[grid_el](
        out,
        out_gate,
        total_elements,
        BLOCK=BLOCK_EL,
    )

    # ------------------------------------------------------------------
    # 8. Final projection back to original dimensionality
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_weight)   # (B, N, N, dim)  fp16

    # Return FP32 as required by the interface
    return out.to(torch.float32)
