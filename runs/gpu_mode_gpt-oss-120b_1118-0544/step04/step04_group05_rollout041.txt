def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

        weight=weights["norm.weight"],
        bias=weights["norm.bias"],

    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the 4‑D input (B,N,N,C) using half‑precision
   parameters.
2. Cast to float16 for the heavy path.
3. Linear projections to the hidden dimension H (`left_proj` and `right_proj`).
4. Compute three sigmoid gates (left, right, out) from the normalized input.
5. Fuse the element‑wise multiplication of the projections with their
   respective gates via a tiny Triton kernel (in‑place, no extra memory).
6. Apply an optional binary mask (broadcasted over H) with a regular
   PyTorch multiplication.
7. Fold the hidden dimension into the batch dimension and compute the core
   reduction:   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   This is a batched matrix multiplication C = A @ Bᵀ performed with
   torch.bmm (cuBLAS), which is the fastest path on B200.
8. Reshape back to (B,N,N,H), apply a second LayerNorm over H,
   multiply by the output gate, and project back to the original channel
   dimension C.
9. Cast the result to the original dtype.

Only the element‑wise gate fusion uses a Triton kernel, satisfying the
requirement to employ Triton while keeping the overall runtime low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: in‑place element‑wise multiplication (a *= b)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,               # pointer to tensor A (output, also input)
    b_ptr,               # pointer to tensor B
    N,                   # total number of elements (runtime int)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE, dtype=tl.int64)
    mask = offs < N

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)

    tl.store(a_ptr + offs, a * b, mask=mask)   # in‑place write back to A


def _apply_elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """In‑place a = a * b using Triton."""
    assert a.numel() == b.numel(), "Tensor sizes must match for elementwise mul."
    total = a.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        a,
        b,
        total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
        num_stages=2,
    )
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C]
            - mask          : torch.Tensor of shape [B, N, N] or None (binary)
            - weights       : dict of model parameters (float32 tensors)
            - config        : dict with keys
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – optional bool, ignore mask when True

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (FP16)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_b = weights["norm.bias"].to(device=device, dtype=torch.float16)

    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )
    x = x.to(torch.float16)   # work in half precision from here

    # ------------------------------------------------------------------
    # 2) Cast linear weights to half
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Fuse projection * gate using Triton (in‑place)
    # ------------------------------------------------------------------
    _apply_elementwise_mul(left, left_gate)
    _apply_elementwise_mul(right, right_gate)

    # ------------------------------------------------------------------
    # 6) Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1) -> broadcast
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 7) Fold hidden dimension into batch and perform batched GEMM with cuBLAS
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # B, seq_len, seq_len, hidden_dim
    # (B, N, N, H) -> (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Core reduction: out_flat = left @ rightᵀ
    out_flat = torch.bmm(left, right.transpose(1, 2))   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim) in half precision

    # ------------------------------------------------------------------
    # 12) Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
