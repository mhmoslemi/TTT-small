"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the input 4‑D tensor (B, N, N, C).
2. Cast to fp16 for the compute‑intensive part.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) with fp16 weights.
4. If a mask is present, a small Triton kernel fuses projection * gate *
   mask (broadcast over the hidden dimension H).
5. Core reduction:
       out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   This is a batch of H·B matrix multiplications of size N×N.
   We reshape to (B·H, N, N) and use torch.bmm (cuBLAS) which runs on
   tensor‑cores and is highly optimized.
6. Apply the second LayerNorm, multiply by the output gate and project
   back to the original channel dimension.
7. Cast the result back to the input dtype.

At least one operation (gate × mask) is performed in Triton.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: elementwise proj * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    proj_ptr,          # *[total_elems]   fp16
    gate_ptr,          # *[total_elems]   fp16
    mask_ptr,          # *[mask_elems]    fp16
    out_ptr,           # *[total_elems]   fp16
    total_elems,       # int
    hidden_dim,        # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask index: one mask element per hidden_dim consecutive proj elements
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def apply_gate_and_mask(proj: torch.Tensor,
                        gate: torch.Tensor,
                        mask: torch.Tensor,
                        block_size: int = 4096) -> torch.Tensor:
    """
    Fuse projection, gate and optional mask.
    All tensors are fp16 and have shapes:
        proj : (B, N, N, H)
        gate : (B, N, N, H)
        mask : (B, N, N)   (broadcast over H)
    Returns:
        Tensor of shape (B, N, N, H) = proj * gate * mask
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (math.ceil(total / block_size),)

    # flatten mask for fast indexed load
    mask_flat = mask.contiguous().view(-1)

    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Entry point -----------------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, C] (float32)
            mask: torch.Tensor [B, N, N] (bool/float) or None
            weights: dict of model weights (torch.Tensor)
            config: dict with keys "dim" and "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, C] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy stages
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast weights to fp16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections to hidden dimension
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)   # (B, N, N, H)

    # --------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 6. Fuse mask and gate (if mask is present and not disabled)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core O(N³·H) reduction – batch of B*H matrix multiplications
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batch matrix multiplication (right is transposed on the reduction axis)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # --------------------------------------------------------------
    # 9. Cast back to the original dtype (usually float32)
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
