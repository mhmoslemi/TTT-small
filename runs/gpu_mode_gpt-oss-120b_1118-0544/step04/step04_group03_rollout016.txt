```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batch_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)   # tile id over output M,N
    batch = tl.program_id(1) # batch index

    # coordinates for tile
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets within tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointer for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # Load A block
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B block (as KxN)
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # write back
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )

```python
def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul operation.
    Implements the forward pass described in the AlphaFold3 paper.
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on input
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # cast to half for compute
    x_fp16 = x.to(torch.float16)

    device = x_fp16.device
    dtype = torch.float16

    # Prepare combined linear weight for left and its gate
    left_comb_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(device=device, dtype=dtype)
    right_comb_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(device=device, dtype=dtype)

    # linear + gate preactivation
    left_comb = F.linear(x_fp16, left_comb_weight)   # shape [B,N,N,2*H]
    right_comb = F.linear(x_fp16, right_comb_weight) # shape [B,N,N,2*H]

    # split
    left_proj = left_comb[..., :hidden_dim]  # [B,N,N,H]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # mask (if any)
    if not nomask and mask is not None:
        mask_fp = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Core batched GEMM via Triton: out = left @ right^T per hidden dim
    B, N, _, H = left.shape

    # reshape for batched gemm
    # left: [B,N,N,H] -> [B,H,N,N] -> view (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # striding
    stride_a_m = left_flat.stride(1)  # N
    stride_a_k = left_flat.stride(2)  # 1
    stride_b_k = right_flat.stride(1)  # N (since we treat as K,N)
    stride_b_n = right_flat.stride(2)  # 1
    stride_c_m = out_flat.stride(1)    # N
    stride_c_n = out_flat.stride(2)    # 1

    stride_a_batch = left_flat.stride(0)  # N*N
    stride_b_batch = right_flat.stride(0)
    stride_c_batch = out_flat.stride(0)

    # block sizes, tune for typical N <= 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # compute grid
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _batch_gemm_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B,N,N,H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # hidden dimension LayerNorm
    out = F.layer_norm(out, (H,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # output gate
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(device=device, dtype=dtype)).sigmoid()
    out = out * out_gate

    # final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"].to(device=device, dtype=dtype))

    # cast to original dtype (float32)
    out = out.to(x.dtype)

    return out
```

"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the PyTorch reference:
1. LayerNorm on the input.
2. Linear projections + gates (left/right) are fused into two linear calls.
3. Optional masking applied to the projected values.
4. Gated projections are multiplied using a batched GEMM kernel
   (out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]),
   i.e. a set of (B·hidden_dim) matrix multiplications of shape N×N.
5. Hidden‑dim LayerNorm, output gate, and final linear projection back
   to the original channel dimension.

All heavy computations (projections, gates, GEMM) are performed in float16
with float32 accumulation for numerical stability. The final result is cast
to the original input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batch_gemm_kernel(
    a_ptr, b_ptr, c_ptr,           # pointers
    M, N, K,                       # matrix sizes
    stride_am, stride_ak,          # A strides (M dim, K dim)
    stride_bk, stride_bn,          # B strides (K dim, N dim) – B is stored as K×N (i.e. transposed)
    stride_cm, stride_cn,          # C strides (M dim, N dim)
    stride_ab, stride_bb, stride_cb,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B^T  for each batch (A: M×K, B: N×K)."""
    pid = tl.program_id(0)          # tile index over output matrix
    batch = tl.program_id(1)        # batch index (B * hidden_dim)

    # tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # A block (M×K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B block (K×N) – we treat stored B as K×N (i.e. transposed)
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp16·fp16 → fp32

    # write back
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[:, None],
    )


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # --------------------------------------------------------------- #
    # Unpack
    # --------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device
    fp16 = torch.float16

    # --------------------------------------------------------------- #
    # 1. Input LayerNorm
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_fp16 = x.to(fp16)

    # --------------------------------------------------------------- #
    # 2. Linear projections + gates (fused)
    # --------------------------------------------------------------- #
    # combine projection and gate weights to halve linear calls
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=fp16)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=fp16)

    # shape: [B, N, N, 2*H]
    left_comb = F.linear(x_fp16, left_comb_w)
    right_comb = F.linear(x_fp16, right_comb_w)

    # split projection and gate pre‑activations
    left_proj = left_comb[..., :hidden_dim]               # [B,N,N,H]
    left_gate = left_comb[..., hidden_dim:].sigmoid()    # [B,N,N,H]
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # --------------------------------------------------------------- #
    # 3. Mask (if any) – applied only to the projected tensors
    # --------------------------------------------------------------- #
    if (mask is not None) and (not nomask):
        mask_fp = mask.to(fp16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # --------------------------------------------------------------- #
    # 4. Apply gates
    # --------------------------------------------------------------- #
    left = left_proj * left_gate    # [B,N,N,H]
    right = right_proj * right_gate

    # --------------------------------------------------------------- #
    # 5. Core O(N³·H) reduction via batched GEMM (TriMul)
    # --------------------------------------------------------------- #
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) for batched GEMM
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B,H,N,N]
    left_flat = left_t.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)

    out_flat = torch.empty_like(left_flat, dtype=fp16)

    # strides (row‑major contiguous tensors)
    stride_a_m = left_flat.stride(1)   # N
    stride_a_k = left_flat.stride(2)   # 1
    stride_b_k = right_flat.stride(1)  # N
    stride_b_n = right_flat.stride(2)  # 1
    stride_c_m = out_flat.stride(1)    # N
    stride_c_n = out_flat.stride(2)    # 1

    stride_a_batch = left_flat.stride(0)   # N*N
    stride_b_batch = right_flat.stride(0)
    stride_c_batch = out_flat.stride(0)

    # block sizes – tuned for typical N ≤ 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _batch_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                               # M, N, K
        stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------- #
    # 6. Hidden‑dim LayerNorm
    # --------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=fp16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=fp16),
    )

    # --------------------------------------------------------------- #
    # 7. Output gate (element‑wise)
    # --------------------------------------------------------------- #
    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(device=device, dtype=fp16),
    ).sigmoid()                     # [B,N,N,H]
    out = out * out_gate

    # --------------------------------------------------------------- #
    # 8. Final linear projection back to original channel dimension
    # --------------------------------------------------------------- #
    out = F.linear(
        out,
        weights["to_out.weight"].to(device=device, dtype=fp16),
    )                                 # [B,N,N,dim]

    # Cast back to the original dtype (typically float32)
    out = out.to(x.dtype)
    return out
