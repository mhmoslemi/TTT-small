@triton.jit
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_am, stride_ak,
                        stride_bk, stride_bn,
                        stride_cm, stride_cn,
                        batch_stride_a, batch_stride_b, batch_stride_c,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    batch = tl.program_id(2)  # batch dimension

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create mask for valid rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointer calculations inclusive batch offset
        a_ptrs = A_ptr + batch * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = B_ptr + batch * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

    # Write back
    c_ptrs = C_ptr + batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

@triton.jit
def gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate and a
batched GEMM kernel for the O(N³·H) reduction.

Algorithm
---------
1. Layer‑norm the input (B,N,N,C).
2. Cast to fp16 and apply three linear projections:
   * left_proj, right_proj   → (B,N,N,H)
   * left_gate, right_gate, out_gate → (B,N,N,H) (sigmoid)
3. Fuse element‑wise multiplication with the optional mask using a
   Triton kernel (proj * gate * mask) to avoid extra memory passes.
4. Compute the core reduction
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM (A @ Bᵀ) where the batch dimension is B·H.
   The GEMM is implemented as a Triton kernel with tiling over
   M, N and K.
5. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate, and project back to the original channel dimension.
6. Cast the result back to the original dtype (usually float32).

Only the mask‑gate fusion and the batched GEMM are performed in Triton;
all other ops use PyTorch’s highly‑optimized kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def _gate_mask_kernel(
    proj_ptr,      # fp16 *
    gate_ptr,      # fp16 *
    mask_ptr,      # fp16 *
    out_ptr,       # fp16 *
    total_elems,   # i64
    hidden_dim,    # i64
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse proj * gate * mask (mask broadcast over hidden_dim)."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    in_bounds = offs < total_elems

    # mask index repeats each value hidden_dim times
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=in_bounds, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=in_bounds, other=0.0)
    mval = tl.load(mask_ptr + mask_idx, mask=in_bounds, other=1.0)

    out = proj * gate * mval
    tl.store(out_ptr + offs, out, mask=in_bounds)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """Fuse elementwise mul of proj, gate and mask using Triton."""
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)

    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,                              # dimensions (int64)
    stride_am, stride_ak,                 # A strides
    stride_bk, stride_bn,                 # Bᵀ strides (k, n)
    stride_cm, stride_cn,                 # C strides
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM C = A @ Bᵀ
    A: (batch, M, K)
    B: (batch, N, K)   -> accessed as (batch, K, N) via stride_bk/stride_bn
    C: (batch, M, N)
    """
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    batch = tl.program_id(2)          # batch index

    # Offsets for the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers with batch offset
        a_ptrs = (A_ptr + batch * batch_stride_a
                  + offs_m[:, None] * stride_am
                  + offs_k[None, :] * stride_ak)
        b_ptrs = (B_ptr + batch * batch_stride_b
                  + offs_k[:, None] * stride_bk
                  + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) in FP32

    # Write back
    c_ptrs = (C_ptr + batch * batch_stride_c
              + offs_m[:, None] * stride_cm
              + offs_n[None, :] * stride_cn)

    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main custom kernel
# ----------------------------------------------------------------------


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16
    # ------------------------------------------------------------------
    def _to_fp16(t): return t.to(device=device, dtype=torch.float16)

    left_proj_w   = _to_fp16(weights["left_proj.weight"])
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])
    to_out_norm_w = _to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = _to_fp16(weights["to_out_norm.bias"])
    to_out_w      = _to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,C) -> (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (same shape) + sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask + gates (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM C = A @ Bᵀ
    #    A and B have shape (B, N, N, H) -> treat batch = B*H
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Rearrange to (B, H, N, N) and flatten batch dimension
    left_view = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left_view.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_view.view(B * H, N, N) # (B*H, N, N)

    out_flat = torch.empty_like(left_flat)    # (B*H, N, N)

    # Strides for the flattened tensors (batch, M, K) and (batch, K, N)
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_m, stride_b_k = right_flat.stride()
    # For Bᵀ we need stride over K (original column) and N (original row)
    stride_bk = stride_b_k               # stride for K dimension
    stride_bn = stride_b_m               # stride for N dimension

    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # Kernel launch configuration
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks over M dimension
        triton.cdiv(N, BLOCK_N),   # blocks over N dimension
        B * H                      # batch dimension
    )

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_a_m, stride_a_k,
        stride_bk, stride_bn,
        stride_c_m, stride_c_n,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
