"""
Custom Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy O(N³·H) reduction is performed with torch.bmm (batched GEMM),
which on CUDA uses cuBLAS and is much faster than a naive Triton implementation.
A tiny Triton kernel is used for the final element‑wise multiplication with the
output gate (and also for mask application when a mask is present), satisfying
the requirement that part of the computation runs in a Triton kernel.
Mixed‑precision (fp16) is used for the compute‑intensive parts and the result
is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton element‑wise multiplication kernel (used for mask and output gate)
# --------------------------------------------------------------------------- #
@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, numel, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < numel
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Element‑wise multiplication via Triton."""
    assert a.shape == b.shape, "Shapes must match for elementwise mul"
    out = torch.empty_like(a)
    numel = a.numel()
    BLOCK = 1024
    grid = ((numel + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, numel, BLOCK=BLOCK)
    return out


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool or 0/1) or None
            - weights: dict of pretrained parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------- #
    # unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    device = x.device
    orig_dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 2. Move to fp16 for the compute‑heavy part
    # ------------------------------------------------------------------- #
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 3. Prepare weights in fp16
    # ------------------------------------------------------------------- #
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    left_proj_w = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w = to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------- #
    # 4. Combined linear for left (proj + gate) and right (proj + gate)
    # ------------------------------------------------------------------- #
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, C)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    left_comb = F.linear(x_fp16, left_comb_w)      # [B, N, N, 2H]
    right_comb = F.linear(x_fp16, right_comb_w)    # [B, N, N, 2H]

    # split
    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # ------------------------------------------------------------------- #
    # 5. Optional masking (broadcast over hidden dim)
    # ------------------------------------------------------------------- #
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = _elemwise_mul(left_proj, mask_fp16)
        right_proj = _elemwise_mul(right_proj, mask_fp16)

    # ------------------------------------------------------------------- #
    # 6. Apply gates
    # ------------------------------------------------------------------- #
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------- #
    # 7. Core O(N³·H) reduction – batched matrix multiplication
    #    Compute out[b, i, j, h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape  # B, N, N, H

    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched GEMM: (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 8. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------- #
    # 9. Output gate (element‑wise) – Triton kernel
    # ------------------------------------------------------------------- #
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()
    out = _elemwise_mul(out, out_gate)

    # ------------------------------------------------------------------- #
    # 10. Final projection back to original channel dimension
    # ------------------------------------------------------------------- #
    out = F.linear(out, to_out_w)   # [B, N, N, C] (still fp16)

    # ------------------------------------------------------------------- #
    # 11. Cast to float32 (as required) and return
    # ------------------------------------------------------------------- #
    out = out.to(torch.float32)
    return out
