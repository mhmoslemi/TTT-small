@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out * gate, mask=mask)

Alternatively we can provide a slightly larger block size (e.g., 8192). But we need to stay within shared memory etc.

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # Get weight tensors
    # All weights are stored maybe as float32; we convert to appropriate dtype.

    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    # 1. LayerNorm (FP32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Cast to fp16
    x_fp16 = x.to(torch.float16)

    # 2. Linear projections
    left_proj = F.linear(x_fp16, left_proj_weight.to(torch.float16))
    right_proj = F.linear(x_fp16, right_proj_weight.to(torch.float16))

    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # 3. Gates
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight.to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight.to(torch.float16)))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # 4. Main contraction (einsum)
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # 5. to_out_norm (LayerNorm over hidden)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight.to(torch.float16), bias=to_out_norm_bias.to(torch.float16), eps=1e-5)

    # 6. out_gate
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight.to(torch.float16)))
    # Ensure shape matches out: [B, N, N, hidden_dim]

    # 7. elementwise multiplication with Triton
    total_el = out.numel()
    BLOCK = 65536  # might be large; but limit within 2**31? Must be multiple of 2? Use block size 32768 maybe.

    # Determine grid
    grid = ( (total_el + BLOCK - 1) // BLOCK, )
    elementwise_mul_kernel[grid](out, out_gate, total_el, BLOCK=BLOCK)

    # 8. final linear projection to dim
    out = F.linear(out, to_out_weight.to(torch.float16))

    # Cast to float32
    return out.to(torch.float32)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask, dtype=tl.float16)
    gate = tl.load(gate_ptr + offs, mask=mask, dtype=tl.float16)
    tl.store(out_ptr + offs, out * gate, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Performs LayerNorm, linear projections, gating, and the core
    reduction out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] using a
    fused einsum, followed by a second LayerNorm, out_gate multiplication
    (via a tiny Triton kernel), and the final linear projection.
    Input and weights are assumed to be on the target CUDA device.
    Output is in float32.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # Load weights and cast to appropriate dtype (fp16)
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # 1. Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] (float32)

    # Cast to fp16 for the heavy compute
    x_fp16 = x.to(torch.float16)

    # 2. Linear projections (no bias)
    left_proj = F.linear(x_fp16, left_proj_weight)   # (B, N, N, hidden)
    right_proj = F.linear(x_fp16, right_proj_weight)

    # 3. Optional mask (broadcast over hidden dim)
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # 4. Gates
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))

    # 5. Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # 6. Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)

    # 7. LayerNorm over hidden dimension (fp16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden) fp16

    # 8. out_gate (linear + sigmoid)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # (B, N, N, hidden) fp16

    # 9. Element‑wise multiplication via Triton
    total_el = out.numel()
    BLOCK = 1024
    grid = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        out,
        out_gate,
        total_el,
        BLOCK=BLOCK,
    )

    # 10. Final linear projection back to original dim
    out = F.linear(out, to_out_weight)  # (B, N, N, dim) fp16

    # 11. Cast back to float32 as required by the interface
    return out.to(torch.float32)

"""
Outgoing TriMul forward pass (AlphaFold3).

The implementation follows the reference PyTorch module but
* keeps the heavy arithmetic in fp16 to exploit Tensor Cores,
* replaces the explicit batched GEMM with a single `torch.einsum`,
  eliminating costly permutes and copies,
* fuses the final element‑wise multiplication with a tiny Triton kernel,
  satisfying the requirement to use Triton and reducing memory traffic.

Input:
    x:  [B, N, N, C]               (float32)
    mask: [B, N, N]                (bool/float, optional)
    weights: dict with all linear / layer‑norm parameters
    config: {"dim": C, "hidden_dim": H, "nomask": bool}

Output:
    Tensor of shape [B, N, N, C] (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    """out = out * gate   (both fp16, flattened 1‑D view)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask, dtype=tl.float16)
    gate = tl.load(gate_ptr + offs, mask=mask, dtype=tl.float16)
    tl.store(out_ptr + offs, out * gate, mask=mask)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) – may be ignored
        - weights      : dict of model weight tensors (float32)
        - config       : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns
    -------
    torch.Tensor
        Processed tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # Load weights (convert to half for the compute‑intensive path)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias   = weights["norm.bias"].to(device)

    left_proj_weight  = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)

    left_gate_weight  = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight   = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, C]  (float32)

    # ------------------------------------------------------------------
    # Cast to fp16 for the remaining heavy work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)                     # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → shape [B, N, N, hidden]
    # ------------------------------------------------------------------
    left_proj  = F.linear(x_fp16, left_proj_weight)   # (B, N, N, H)
    right_proj = F.linear(x_fp16, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional masking (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left_proj  = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate      # (B, N, N, H)
    right = right_proj * right_gate    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Core contraction:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented via a single einsum (avoids explicit permutes).
    # ------------------------------------------------------------------
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7. LayerNorm over the hidden dimension (still fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),

    )  # (B, N, N, H) fp16

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)               # (B, N, N, C) fp16

    # 11. Cast to float32 as required by the interface
    # ------------------------------------------------------------------
    return out.to(torch.float32)
