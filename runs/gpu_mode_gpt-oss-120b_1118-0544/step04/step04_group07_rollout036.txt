    left_ptr, right_ptr, out_ptr,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_m, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,

):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # loop over K

        # load left tile (M x K)

        # load right tile (N x K) but transposed
        # We need right_T tile: shape (BLOCK_K, BLOCK_N) where row = k, col = n

        # So for B_T we need pointer: base + pid_batch*stride_r_batch + (k_offset[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_m)

@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_m, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BATCH: tl.constexpr,
    N: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Mask for boundaries
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load left tile
        # left[b,i,k]
        left_ptrs = left_ptr + pid_batch * stride_l_batch + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=(offs_m[:, None] < N) & (offs_k[None, :] < N), other=0.0).to(tl.float32)

        # Load right tile transposed: need right[b, j, k] -> we need right_T[k, j]
        # original right layout: stride_r_m = N (row stride), stride_r_k = 1 (col stride)
        # For B_T we need to index row = k, col = j
        right_ptrs = right_ptr + pid_batch * stride_r_batch + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_m
        right = tl.load(right_ptrs, mask=(offs_k[:, None] < N) & (offs_n[None, :] < N), other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(left, right)

    # Write result

@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_m, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    N: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program IDs
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load left tile
        left_ptrs = left_ptr + pid_batch * stride_l_batch + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=(offs_m[:, None] < N) & (offs_k[None, :] < N),
                       other=0.0).to(tl.float32)

        # Load right tile transposed (right_T)
        right_ptrs = right_ptr + pid_batch * stride_r_batch + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_m
        right = tl.load(right_ptrs,
                        mask=(offs_k[:, None] < N) & (offs_n[None, :] < N),
                        other=0.0).to(tl.float32)

"""
custom_kernel: Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update).

The implementation follows the reference PyTorch module but moves the two
compute‑heavy parts to Triton:
  1) Element‑wise fusion of mask, left‑gate and right‑gate (optional).
  2) A batched GEMM kernel that computes, for each hidden dimension,
     out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]  (i.e. left @ rightᵀ).

All linear layers, LayerNorms and activations run in FP16 for tensor‑core
speed; the final result is cast back to the input dtype (float32 by default).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------------- #
# 1) Optional mask + gate fusion
# ----------------------------------------------------------------------------- #
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each mask element corresponds to `hidden_dim` elements in the flat view
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store in‑place
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------------- #
# 2) Batched GEMM kernel: out = left @ rightᵀ   (batch = B * hidden_dim)
# ----------------------------------------------------------------------------- #
@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_m, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    N: tl.constexpr,                     # sequence length (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # which (B * hidden) matrix
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile column

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # ----- left tile (M x K) -----
        left_ptrs = (
            left_ptr
            + pid_batch * stride_l_batch
            + offs_m[:, None] * stride_l_m
            + offs_k[None, :] * stride_l_k
        )
        left = tl.load(
            left_ptrs,
            mask=(offs_m[:, None] < N) & (offs_k[None, :] < N),
            other=0.0,
        )  # fp16

        # ----- right tile transposed (K x N) -----
        #   right is stored row‑major (j,k). We need rightᵀ so we
        #   index (k,j) → row = k, col = j.
        right_ptrs = (
            right_ptr
            + pid_batch * stride_r_batch
            + offs_k[:, None] * stride_r_k   # k → row
            + offs_n[None, :] * stride_r_m   # j → column
        )
        right = tl.load(
            right_ptrs,
            mask=(offs_k[:, None] < N) & (offs_n[None, :] < N),
            other=0.0,
        )  # fp16

        # Accumulate
        acc += tl.dot(left, right)  # fp16 inputs, fp32 accumulation

    # ----- Store result (convert back to fp16) -----
    out_ptrs = (
        out_ptr
        + pid_batch * stride_o_batch
        + offs_m[:, None] * stride_o_m
        + offs_n[None, :] * stride_o_n
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=(offs_m[:, None] < N) & (offs_n[None, :] < N),
    )


# ----------------------------------------------------------------------------- #
# Entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------- #
    # 0) Load weights (cast to appropriate dtype)
    # --------------------------------------------------------------------- #
    # LayerNorm on the input
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)

    # Linear weights (FP16 for compute)
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # --------------------------------------------------------------------- #
    # 1) Input LayerNorm
    # --------------------------------------------------------------------- #
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)                     # fp16 for the rest

    # --------------------------------------------------------------------- #
    # 2) Projections
    # --------------------------------------------------------------------- #
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # --------------------------------------------------------------------- #
    # 3) Gates
    # --------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # --------------------------------------------------------------------- #
    # 4) Optional mask & element‑wise fusion (Triton)
    # --------------------------------------------------------------------- #
    if mask is not None and not nomask:
        # mask: (B, N, N) → (B, N, N, 1) for broadcast
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)

        # Flatten everything for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.squeeze(-1).contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024  # chosen to give enough work per program

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask → simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5) Batched GEMM: out = left @ rightᵀ   (B * H independent N×N matmuls)
    # --------------------------------------------------------------------- #
    B, N_seq, _, _ = left.shape  # N_seq == seq_len
    # move hidden dim into the batch dimension
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch_h, N, _ = left.shape  # batch_h = B * H
    left = left.view(batch_h, N, N)
    right = right.view(batch_h, N, N)

    # Prepare output tensor
    out = torch.empty_like(left)  # (batch_h, N, N), dtype=fp16

    # Strides (all tensors are contiguous after the view)
    stride_l_batch = left.stride(0)
    stride_l_m     = left.stride(1)
    stride_l_k     = left.stride(2)

    stride_r_batch = right.stride(0)
    stride_r_m     = right.stride(1)
    stride_r_k     = right.stride(2)

    stride_o_batch = out.stride(0)
    stride_o_m     = out.stride(1)
    stride_o_n     = out.stride(2)

    # Block sizes – tweakable if needed
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_h,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_gemm_kernel[grid](
        left,
        right,
        out,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_m,
        stride_r_k,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        N=N,                     # compile‑time constant for the kernel
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 6) Output LayerNorm, gating and final projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)    # (B, N, N, dim)

    # Cast back to the original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
