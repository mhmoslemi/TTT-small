"""
custom_kernel: Outgoing Triangle Multiplicative Update (TriMul) – Triton‑accelerated.

The forward pass follows the reference PyTorch implementation:

1. Layer‑norm over the input channel dimension.
2. Linear projections (left/right) and gated projections (left/right) computed in
   float16.  An optional mask (if provided) is applied before the gates.
3. Core O(N³·H) reduction: for each batch and hidden channel d,
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   This is a batched GEMM C = A @ Bᵀ with A, B ∈ ℝ^{N×N} and is computed by a
   custom Triton kernel that uses FP16 inputs with FP32 accumulation.
4. Layer‑norm over the hidden dimension, element‑wise output‑gate, and final
   linear projection back to the original channel dimension.
All heavy arithmetic is performed in float16; the final result is cast back
to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM C = A @ Bᵀ
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all equal to seq_len)
    stride_am, stride_ak,              # strides for A (M, K)
    stride_bk, stride_bn,              # strides for B (K, N) – B is read transposed
    stride_cm, stride_cn,              # strides for C (M, N)
    stride_ab, stride_bb, stride_cb,   # strides for the batch dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM: C = A @ Bᵀ
    A, B, C have shape (batch, M, K), (batch, N, K), (batch, M, N) respectively.
    """
    pid = tl.program_id(0)          # tile index over (M,N)
    batch = tl.program_id(1)        # batch index (B * hidden_dim)

    # Compute tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = k_offset + offs_k
        mask_k = cur_k < K

        # Load a tile of A (M x K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a tile of Bᵀ (K x N) – note the transposed access pattern
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16·FP16 → FP32 accumulation
        acc += tl.dot(a, b)

    # Write result back as float16
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass for the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict containing
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device
    orig_dtype = x.dtype

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (over channel dim C)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Work in float16 for the heavy compute
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 2. Linear projections + gates (float16)
    # ------------------------------------------------------------------- #
    # Cast weights to half precision once
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    left_proj_w = to_half(weights["left_proj.weight"])   # (H, C)
    right_proj_w = to_half(weights["right_proj.weight"])

    left_gate_w = to_half(weights["left_gate.weight"])
    right_gate_w = to_half(weights["right_gate.weight"])
    out_gate_w = to_half(weights["out_gate.weight"])

    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w = to_half(weights["to_out.weight"])

    # Projections
    left_proj = F.linear(x_h, left_proj_w)   # [B,N,N,H]
    right_proj = F.linear(x_h, right_proj_w)

    # Optional masking before gating
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))

    # Apply gates
    left = left_proj * left_gate    # [B,N,N,H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------- #
    # 3. Core O(N³·H) reduction via batched GEMM (Triton)
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape  # (B, N, N, H)

    # Re‑arrange to (batch, M, K) where batch = B*H, M = K = N
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_t.view(B * H, N, N)   # (batch, M=N, K=N)
    right_flat = right_t.view(B * H, N, N) # (batch, N, K=N) – will be used transposed

    # Output buffer
    out_flat = torch.empty_like(left_flat)  # (batch, M, N), float16

    # Strides (in elements, not bytes)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)   # M stride
    stride_a_k = left_flat.stride(2)   # K stride

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)  # column stride (K dimension after transposition)
    stride_b_n = right_flat.stride(1)  # row stride (N dimension after transposition)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)    # M stride
    stride_c_n = out_flat.stride(2)    # N stride

    # Blocking parameters – tuned for N up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 4. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------- #
    # 5. Output gate (element‑wise) and final projection
    # ------------------------------------------------------------------- #
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B,N,N,H]
    out = out * out_gate

    out = F.linear(out, to_out_w)   # [B,N,N,C]

    # Cast back to original dtype
    out = out.to(orig_dtype)
    return out
