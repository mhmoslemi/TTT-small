@triton.jit
def trmul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # Compute block indices
    grid_i = tl.cdiv(N, BLOCK_M)
    grid_j = tl.cdiv(N, BLOCK_N)
    batch_id = pid // (grid_i * grid_j)
    rem = pid % (grid_i * grid_j)
    pid_i = rem // grid_j
    pid_j = rem % grid_j

    # Offsets for this block at given batch
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Batch offset
    batch_offset = batch_id * stride_left_batch

    # Pointers for A and B tiles
    a_ptrs = left_ptr + batch_offset + offs_i[:, None] * stride_left_i + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    b_ptrs = right_ptr + batch_offset + tl.arange(0, BLOCK_K)[:, None] * stride_right_k + offs_j[None, :] * stride_right_j

    # Init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Load A and B
        a = tl.load(a_ptrs, mask=offs_i < N and tl.arange(0, BLOCK_K) + k*BLOCK_K < N)
        b = tl.load(b_ptrs, mask=tl.arange(0, BLOCK_K) + k*BLOCK_K < N and offs_j < N)

        # dot
        acc += tl.dot(a, b)

        # Increment pointers
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # Store result
    c_ptrs = out_ptr + batch_offset + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask= (offs_i < N)[:, None] & (offs_j < N)[None, :])
```

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_aba, stride_abi, stride_abk,
    stride_bba, stride_bbk, stride_bbj,
    stride_cca, stride_cci, stride_ccj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # Compute block indices.
    # grid dimensions
    grid_m = tl.cdiv(M, BLOCK_M)
    grid_n = tl.cdiv(N, BLOCK_N)
    batch_size = batch

    # Compute coordinates.
    total_blocks_per_batch = grid_m * grid_n
    batch_idx = pid // total_blocks_per_batch
    pid_in_batch = pid % total_blocks_per_batch
    block_i = pid_in_batch // grid_n
    block_j = pid_in_batch % grid_n

    # Offsets for i and j within each block
    i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # offsets for k
    k = tl.arange(0, BLOCK_K)

    # Compute offsets for pointers
    # pointer to start of current batch matrix
    a_batch_offset = a_ptr + batch_idx * stride_aba

```python
import torch
import triton
import triton.language as tl

# Triton kernel for applying projection, gate and optional mask
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Determine which mask element corresponds (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * mask_val, mask=mask)

def _apply_gate_and_mask(proj, gate, mask, block_size=1024):
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _apply_gate_mask_kernel[grid](
        proj, gate, mask, out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out

# Triton kernel for batched GEMM (C = A @ B^T) where each batch corresponds to one hidden channel
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr
):
    pid = tl.program_id(0)

    # Tile indices
    num_i = tl.cdiv(M, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_i * num_j

    batch_idx = pid // blocks_per_batch
    pid_in_batch = pid % blocks_per_batch
    tile_i = pid_in_batch // num_j
    tile_j = pid_in_batch % num_j

    # Coordinates within the matrix
    i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for valid rows / cols
    mask_i = i < M
    mask_j = j < N

    # Pointers to the start of the current batch matrix
    a_batch_ptr = a_ptr + batch_idx * stride_a_batch
    b_batch_ptr = b_ptr + batch_idx * stride_b_batch
    c_batch_ptr = c_ptr + batch_idx * stride_c_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k = tl.cdiv(K, BLOCK_K)

    for kk in range(0, num_k):
        k_base = kk * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch_ptr + i[:, None] * stride_a_i + k[None, :] * stride_a_k,
            mask=mask_i[:, None] & (k < K)[None, :],
            other=0.0
        )
        # Load B tile (as B^T): shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_batch_ptr + k[:, None] * stride_b_k + j[None, :] * stride_b_j,
            mask=(k < K)[:, None] & mask_j[None, :],
            other=0.0
        )
        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c = c_batch_ptr + i[:, None] * stride_c_i + j[None, :] * stride_c_j
    mask = mask_i[:, None] & mask_j[None, :]
    tl.store(c, acc.to(tl.float16), mask=mask)

def _batched_gemm_fp16(a, b, block_m=64, block_n=64, block_k=32):
    """
    a, b: tensors of shape (B*H, N, N), dtype fp16, contiguous.
    Returns c of same shape.
    """
    assert a.shape == b.shape
    batch, N, _ = a.shape
    c = torch.empty_like(a)

    grid = (batch * triton.cdiv(N, block_m) * triton.cdiv(N, block_n),)

    _batched_gemm_kernel[grid](
        a_ptr = a,
        b_ptr = b,
        c_ptr = c,
        batch = batch,
        stride_a_batch = a.stride(0),
        stride_a_i = a.stride(1),
        stride_a_k = a.stride(2),
        stride_b_batch = b.stride(0),
        stride_b_k = b.stride(1),
        stride_b_j = b.stride(2),
        stride_c_batch = c.stride(0),
        stride_c_i = c.stride(1),
        stride_c_j = c.stride(2),
        BLOCK_M = block_m,
        BLOCK_N = block_n,
        BLOCK_K = block_k,
        M = N,
        N = N,
        K = N,
    )
    return c

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C]
            - mask         : torch.Tensor of shape [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict containing:
                * "dim" (int)        – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – whether to ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ---------- LayerNorm ----------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ---------- Cast to fp16 for heavy compute ----------
    x = x.to(torch.float16)

    # ---------- Load weights in fp16 ----------
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    # ---------- Linear projections ----------
    left_proj = torch.nn.functional.linear(x, w_left)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, w_right)

    # ---------- Gates ----------
    left_gate = torch.nn.functional.linear(x, w_left_gate).sigmoid()
    right_gate = torch.nn.functional.linear(x, w_right_gate).sigmoid()
    out_gate = torch.nn.functional.linear(x, w_out_gate).sigmoid()

    # ---------- Apply mask + gates ----------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ---------- Core O(N^3) reduction ----------
    B, N, _, H = left.shape
    # Move hidden dim to a batch dimension for efficient GEMM
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch and hidden into one dimension
    left_flat = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)

    # Compute batched GEMM: out = left @ right^T
    out_flat = _batched_gemm_fp16(left_flat, right_flat, block_m=64, block_n=64, block_k=32)

    # Restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ---------- Output processing ----------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, w_to_out)  # (B,N,N,dim)

    # ---------- Cast back ----------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated mask & gate fusion,
and a custom batched GEMM kernel for the O(N³) contraction.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# 1. Fuse projection, gate and optional mask (fp16 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # int = B*N*N*H
    hidden_dim,        # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # mask is broadcast over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(out_ptr + offs, proj * gate * mask, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """Fuse proj * gate * mask using Triton. All tensors must be fp16 and contiguous."""
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# 2. Batched GEMM kernel: C = A @ Bᵀ   (each batch = one hidden channel)
#    A, B, C are (batch, N, N) fp16 tensors, accumulator in fp32.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
):
    pid = tl.program_id(0)

    # tiles per matrix
    num_i = tl.cdiv(M, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_i * num_j

    batch_idx = pid // blocks_per_batch
    pid_in_batch = pid % blocks_per_batch
    tile_i = pid_in_batch // num_j
    tile_j = pid_in_batch % num_j

    # coordinates of the tile
    i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < M
    mask_j = j < N

    # batch base pointers
    a_batch = a_ptr + batch_idx * stride_a_batch
    b_batch = b_ptr + batch_idx * stride_b_batch
    c_batch = c_ptr + batch_idx * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for kk in range(0, num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)

        # A tile (M×K)
        a = tl.load(
            a_batch + i[:, None] * stride_a_i + k[None, :] * stride_a_k,
            mask=mask_i[:, None] & (k < K)[None, :],
            other=0.0,
        )
        # B tile interpreted as Bᵀ (K×N)
        b = tl.load(
            b_batch + k[:, None] * stride_b_k + j[None, :] * stride_b_j,
            mask=(k < K)[:, None] & mask_j[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)          # fp16 → fp32 accumulation

    # write back
    c = c_batch + i[:, None] * stride_c_i + j[None, :] * stride_c_j
    tl.store(c, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def _batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor,
                       block_m: int = 64, block_n: int = 64, block_k: int = 32) -> torch.Tensor:
    """
    Batched GEMM for tensors of shape (B*H, N, N), fp16.
    Returns a tensor of the same shape.
    """
    assert a.shape == b.shape
    batch, N, _ = a.shape
    c = torch.empty_like(a)

    grid = (batch * triton.cdiv(N, block_m) * triton.cdiv(N, block_n),)

    _batched_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        batch=batch,
        stride_a_batch=a.stride(0),
        stride_a_i=a.stride(1),
        stride_a_k=a.stride(2),
        stride_b_batch=b.stride(0),
        stride_b_k=b.stride(1),
        stride_b_j=b.stride(2),
        stride_c_batch=c.stride(0),
        stride_c_i=c.stride(1),
        stride_c_j=c.stride(2),
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        M=N,
        N=N,
        K=N,
    )
    return c


# ----------------------------------------------------------------------
# 3. Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C]
            - mask         : torch.Tensor of shape [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    orig_dtype = input_tensor.dtype

    # 1️⃣ LayerNorm over the channel dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # 2️⃣ Cast to fp16 for the compute‑heavy part
    x = x.to(torch.float16)

    # 3️⃣ Load all linear weights in fp16
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    # 4️⃣ Linear projections
    left_proj = F.linear(x, w_left)    # (B, N, N, H)
    right_proj = F.linear(x, w_right)

    # 5️⃣ Gating (sigmoid)
    left_gate = F.linear(x, w_left_gate).sigmoid()
    right_gate = F.linear(x, w_right_gate).sigmoid()
    out_gate = F.linear(x, w_out_gate).sigmoid()

    # 6️⃣ Fuse mask & gates (or simple elementwise if no mask)
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 7️⃣ Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    B, N, _, H = left.shape

    # bring hidden dim to the batch axis for efficient GEMM
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)

    out_flat = _batched_gemm_fp16(left_flat, right_flat,
                                  block_m=64, block_n=64, block_k=32)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 8️⃣ Output LayerNorm, out‑gate, final linear projection
    out = F.layer_norm(
        out,
        (H,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )
    out = out * out_gate
    out = F.linear(out, w_to_out)   # (B, N, N, dim)

    # 9️⃣ Cast back to original dtype
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
