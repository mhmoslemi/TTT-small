@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B_H, N,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)  # 0.. B_H-1
    pid_m = tl.program_id(1)  # tile row
    pid_n = tl.program_id(2)  # tile col

    # Offsets for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Compute linear idx for batch* hidden gemm
    batch_idx = pid_batch

    # Pointers for A and B
    a_ptrs = left_ptr + batch_idx * stride_l_batch + offs_m[:, None] * stride_l_m + tl.arange(0, BLOCK_K)[None, :] * stride_l_k
    b_ptrs = right_ptr + batch_idx * stride_r_batch + tl.arange(0, BLOCK_K)[:, None] * stride_r_k + offs_n[None, :] * stride_r_j

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=offs_m < N and tl.arange(0,BLOCK_K)<N, other=0.0)
        b = tl.load(b_ptrs, mask=tl.arange(0,BLOCK_K) < N and offs_n < N, other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_l_k
        b_ptrs += BLOCK_K * stride_r_k
    # write
    out_ptrs = out_ptr + batch_idx * stride_o_batch + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    tl.store(out_ptrs, acc.to(tl.float16), mask=(offs_m < N)[:, None] & (offs_n < N)[None, :])
```

    left_ptr, right_ptr, out_ptr,
    B_H, N,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute masks
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers

            a_ptr + (k_offset + cur_k)[None, :] * stride_l_k,

    left_ptr, right_ptr, out_ptr,
    B_H, N,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to start of tiles for a and b
    a_ptrs = left_ptr + pid * stride_l_batch + offs_m[:, None] * stride_l_m + tl.arange(0, BLOCK_K)[None, :] * stride_l_k
    b_ptrs = right_ptr + pid * stride_r_batch + tl.arange(0, BLOCK_K)[:, None] * stride_r_k + offs_n[None, :] * stride_r_n

        a = tl.load(a_ptrs, mask=mask_m[:, None] & k_valid[None, :], other=0.0)

"""
Outgoing Triangle Multiplicative Update (TriMul) – forward pass.

The kernel performs the core O(N³) operation:
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication left @ rightᵀ for each hidden
channel.  A custom Triton GEMM kernel processes all (batch * hidden)
matrices in parallel, keeping everything in FP16 to exploit Tensor
Cores.  The surrounding LayerNorms, linear projections, gating and the
final output projection are carried out with PyTorch (FP16 where
possible).  An additional tiny Triton kernel fuses the element‑wise
multiplication by the output gate.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
import math


@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_hidden, N,
    # strides for left
    stride_l_batch, stride_l_m, stride_l_k,
    # strides for right
    stride_r_batch, stride_r_j, stride_r_k,
    # strides for output
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul operation.
    Each program instance computes a tile (BLOCK_M x BLOCK_N) of one
    (batch * hidden) matrix: out = left @ rightᵀ.
    """
    pid = tl.program_id(0)          # index over batch * hidden
    pid_m = tl.program_id(1)        # tile row
    pid_n = tl.program_id(2)        # tile column

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the start of the current tile (k = 0)
    a_ptrs = (left_ptr + pid * stride_l_batch
              + offs_m[:, None] * stride_l_m
              + tl.arange(0, BLOCK_K)[None, :] * stride_l_k)

    b_ptrs = (right_ptr + pid * stride_r_batch
              + tl.arange(0, BLOCK_K)[:, None] * stride_r_k
              + offs_n[None, :] * stride_r_j)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        valid_k = (tl.arange(0, BLOCK_K) + k_offset) < N

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & valid_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=valid_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

        # advance k‑dimension pointers
        a_ptrs += BLOCK_K * stride_l_k
        b_ptrs += BLOCK_K * stride_r_k

    # Write the result tile
    c_ptrs = (out_ptr + pid * stride_o_batch
              + offs_m[:, None] * stride_o_m
              + offs_n[None, :] * stride_o_n)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


@triton.jit
def _elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    """out = out * gate (1‑D fused kernel)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out_val = tl.load(out_ptr + offs, mask=mask)
    gate_val = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out_val * gate_val, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)                     # [B, N, N, dim], float32

    B, N, _, _ = x.shape

    # -----------------------------------------------------------------
    # 2. Cast to FP16 for the heavy arithmetic
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (no bias) – FP16
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)      # [B, N, N, hidden]
    right = F.linear(x_fp16, right_proj_w)

    # -----------------------------------------------------------------
    # 4. Optional masking
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(device).to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 5. Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left = left * left_gate      # [B, N, N, hidden]
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Reshape for batched GEMM: (B*hidden, N, N)
    # -----------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left = left.view(B * hidden_dim, N, N)
    right = right.view(B * hidden_dim, N, N)

    # -----------------------------------------------------------------
    # 7. Batched GEMM using Triton (out = left @ rightᵀ)
    # -----------------------------------------------------------------
    out = torch.empty_like(left)   # (B*hidden, N, N), FP16

    # strides of the contiguous tensors
    stride_l_batch = left.stride(0)
    stride_l_m = left.stride(1)    # row dimension (i)
    stride_l_k = left.stride(2)    # column dimension (k)

    stride_r_batch = right.stride(0)
    stride_r_j = right.stride(1)   # row dimension (j)
    stride_r_k = right.stride(2)   # column dimension (k)

    stride_o_batch = out.stride(0)
    stride_o_m = out.stride(1)
    stride_o_n = out.stride(2)

    # Tile sizes – chosen to fit in shared memory and use Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        B * hidden_dim,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_j,
        stride_r_k,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8. Output LayerNorm (over hidden dimension)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)                     # FP16

    # -----------------------------------------------------------------
    # 9. Fuse out_gate multiplication with a tiny Triton kernel
    # -----------------------------------------------------------------
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)
    total_el = out_flat.numel()
    BLOCK = 1024
    grid_mul = ((total_el + BLOCK - 1) // BLOCK,)
    _elementwise_mul_kernel[grid_mul](
        out_flat,
        gate_flat,
        total_el,
        BLOCK=BLOCK,
    )
    out = out_flat.view_as(out)   # shape (B, N, N, hidden)

    # -----------------------------------------------------------------
    # 10. Final linear projection back to original dim
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, dim), FP16

    # Return in float32 as required by the interface
    return out.to(torch.float32)
