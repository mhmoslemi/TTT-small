@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
   pid = tl.program_id(0)  # tile id (i,j)
   batch = tl.program_id(1)  # batch index

   # Determine block indices
   num_pid_n = tl.cdiv(N, BLOCK_N)
   pid_m = pid // num_pid_n
   pid_n = pid % num_pid_n

   # off offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   offs_k = tl.arange(0, BLOCK_K)

   mask_m = offs_m < M
   mask_n = offs_n < N

   a_batch = a_ptr + batch * stride_ab
   b_batch = b_ptr + batch * stride_bb
   c_batch = c_ptr + batch * stride_cb

   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for k in range(0, tl.cdiv(K, BLOCK_K)):
       cur_k = k * BLOCK_K + offs_k
       mask_k = cur_k < K

       a = tl.load(
           a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
           mask=mask_m[:, None] & mask_k[None, :],
           other=0.0
       )
       # b is (N,K) but we need B^T, so we load (cur_k, offs_n)
       b = tl.load(
           b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
           mask=mask_k[:, None] & mask_n[None, :],
           other=0.0
       )
       acc += tl.dot(a, b) # shape (BLOCK_M,BLOCK_N)
   c = acc.to(tl.float16)
   tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :]
   )

def custom_kernel(data):
    x, mask, weights, config = data
    bs, seq_len, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = x.device
    dtype = x.dtype

    # LayerNorm
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Cast to fp16 for speed
    x_h = x_norm.to(torch.float16)

    # Linear projections
    left_proj = F.linear(x_h, weights["left_proj.weight"].to(device, torch.float16))
    right_proj = F.linear(x_h, weights["right_proj.weight"].to(device, torch.float16))
    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].to(device, torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].to(device, torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(device, torch.float16)))

    # Apply mask to projections before gating if mask is provided
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # reshape for batched matmul:
    # left: [B,N,N,hidden_dim] -> permute to [B, hidden_dim, N, N]
    left_t = left.permute(0, 3, 1, 2).contiguous()

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy O(N³·H) contraction is implemented as a batched GEMM:
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
which is equivalent to a batched matrix multiplication
    left[b,:, :,h] @ right[b,:, :,h].T .
All linear projections, gating, masking and final projections are performed
with PyTorch (mixed‑precision). The GEMM kernel operates on fp16 data with
fp32 accumulation and is tiled with BLOCK_M/N/K = 64 for good occupancy on
A100 / B200 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (M×K)·(K×N) → (M×N)
    stride_am, stride_ak,              # strides for A (M, K)
    stride_bi, stride_bj,              # strides for B (row i, col j) – we load Bᵀ
    stride_cm, stride_cn,              # strides for C (M, N)
    stride_ab, stride_bb, stride_cb,   # batch strides for A, B, C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM: C = A @ Bᵀ
    A : (batch, M, K)   – fp16
    B : (batch, N, K)   – fp16 (loaded as transposed)
    C : (batch, M, N)   – fp16 output, fp32 accumulation
    """
    pid = tl.program_id(0)          # tile index (i‑j)
    batch = tl.program_id(1)        # batch × hidden index

    # tile grid
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # A[i,k]
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Bᵀ[k,j] == B[j,k]; we load B with swapped indices
        b = tl.load(
            b_batch + (offs_n[:, None] * stride_bi + cur_k[None, :] * stride_bj),
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp16·fp16 → fp32

    # write result
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[:, None],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int)          – input channel dimension C
                * "hidden_dim" (int)   – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with dtype matching the input tensor.
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    x, mask, weights, cfg = data
    device = x.device
    orig_dtype = x.dtype
    bs, seq_len, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    # --------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to half‑precision for the heavy path
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 2. Linear projections & gates (fused two‑linear calls per side)
    # --------------------------------------------------------------------- #
    # fuse proj + gate weights to halve the number of linear passes
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)

    left_comb = F.linear(x_h, left_comb_w)   # [B, N, N, 2*H]
    right_comb = F.linear(x_h, right_comb_w)  # [B, N, N, 2*H]

    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # --------------------------------------------------------------------- #
    # 3. Optional mask (applied before gating, as in reference code)
    # --------------------------------------------------------------------- #
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------------------------- #
    # 4. Core O(N³·H) contraction via batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------------- #
    # Rearrange to (batch, hidden, N, N) and flatten batch×hidden
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_t.shape
    batch_hidden = B * H

    left_flat = left_t.view(batch_hidden, N, N)   # (batch, M, K)
    right_flat = right_t.view(batch_hidden, N, N) # (batch, N, K) – will be read transposed

    out_flat = torch.empty_like(left_flat)  # (batch, M, N), fp16

    # Strides (in elements, not bytes)
    stride_a_batch = left_flat.stride(0)
    stride_am = left_flat.stride(1)   # M stride
    stride_ak = left_flat.stride(2)   # K stride

    stride_b_batch = right_flat.stride(0)
    stride_bi = right_flat.stride(1)  # row (i) stride – will be used as “row” after transpose
    stride_bj = right_flat.stride(2)  # col (j) stride

    stride_c_batch = out_flat.stride(0)
    stride_cm = out_flat.stride(1)    # M stride
    stride_cn = out_flat.stride(2)    # N stride

    # Tiling parameters – tuned for N up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, batch_hidden)

    _batched_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,                     # M, N, K
        stride_am, stride_ak,
        stride_bi, stride_bj,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 5. Hidden‑dim LayerNorm
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------------- #
    # 6. Output gate (element‑wise)
    # --------------------------------------------------------------------- #
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))  # [B, N, N, H]
    out = out * out_gate

    # --------------------------------------------------------------------- #
    # 7. Final projection back to original channel dimension
    # --------------------------------------------------------------------- #
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)  # [B, N, N, C]

    # Cast back to the original dtype (usually float32)
    out = out.to(orig_dtype)
    return out
