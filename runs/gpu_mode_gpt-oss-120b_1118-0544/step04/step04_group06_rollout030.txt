def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16
    x = x.to(torch.float16)

    # Cast weights to fp16
    left_proj_w = weights["left_proj.weight"].to(device=x.device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=x.device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=x.device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=x.device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=x.device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=x.device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=x.device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=x.device, dtype=torch.float16)

    # Linear projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # Gate projections + sigmoid
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Apply mask if present
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16)
        mask_fp16 = mask_fp16.unsqueeze(-1)  # shape (B,N,N,1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Prepare for batched GEMM: (B,H,N,N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    # Allocate output tensor
    out = torch.empty_like(left)

    # Compute strides
    l_stride_b, l_stride_h, l_stride_i, l_stride_k = left.stride()
    r_stride_b, r_stride_h, r_stride_j, r_stride_k = right.stride()
    o_stride_b, o_stride_h, o_stride_i, o_stride_j = out.stride()

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_batches = B * H
    grid = (total_batches * grid_m * grid_n,)

    batched_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        H,
        N,
        N,
        N,
        l_stride_b,
        l_stride_h,
        l_stride_i,
        l_stride_k,
        r_stride_b,
        r_stride_h,
        r_stride_j,
        r_stride_k,
        o_stride_b,
        o_stride_h,
        o_stride_i,
        o_stride_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B,N,N,H)
    out = out.permute(0, 2, 3, 1).contiguous()

    # Output layer norm over hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # Gate
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype
    out = out.to(input_tensor.dtype)

    return out

```python
@triton.jit
def batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, M, N, K,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_j, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # total number of output blocks per batch
    grid_m = tl.cdiv(M, BLOCK_M)
    grid_n = tl.cdiv(N, BLOCK_N)
    total_batches = B * H

    # determine which batch and which tile we are processing
    batch_idx = pid // (grid_m * grid_n)
    tile_idx = pid % (grid_m * grid_n)
    m_tile = tile_idx // grid_n
    n_tile = tile_idx % grid_n

    # compute offsets within M and N dimensions
    m_offset = m_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    n_offset = n_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out-of-bounds
    mask_m = m_offset < M
    mask_n = n_offset < N

    # compute batch and hidden indices
    b_idx = batch_idx // H
    h_idx = batch_idx - b_idx * H

    # base pointers for this (b,h) pair
    left_base = left_ptr + b_idx * stride_l_b + h_idx * stride_l_h
    right_base = right_ptr + b_idx * stride_r_b + h_idx * stride_r_h
    out_base = out_ptr + b_idx * stride_o_b + h_idx * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # load left tile [M, K]
        left_ptrs = left_base + (m_offset[:, None] * stride_l_i) + (k_offset[None, :] * stride_l_k)
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load right tile [K, N] (note that right is indexed as (k, n))
        right_ptrs = right_base + (k_offset[:, None] * stride_r_k) + (n_offset[None, :] * stride_r_j)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # accumulate
        acc += tl.dot(left, right)

    # write out result
    out = acc.to(tl.float16)
    out_ptrs = out_base + (m_offset[:, None] * stride_o_i) + (n_offset[None, :] * stride_o_j)
    tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass – Triton‑accelerated batched GEMM.

Algorithm
---------
1️⃣ Layer‑norm the input (B,N,N,C) with the provided LN parameters.
2️⃣ Compute the four linear projections (left/right & three gates) in FP32.
3️⃣ Fuse mask (if present) and gates with elementwise multiplication.
4️⃣ Cast the gated tensors to FP16 and reshape to (B, H, N, N) where H=hidden_dim.
5️⃣ Run a custom Triton kernel that performs a batched GEMM:
      out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
   The kernel tiles the M/N dimensions (i/j) and iterates over K (k) using
   `tl.dot`, thus exploiting Tensor‑Core FP16 matrix multiplication.
6️⃣ Reshape the result back to (B,N,N,H) and apply a second LayerNorm,
   the output gate and the final linear projection to the original channel
   dimension C.
7️⃣ Cast the result back to the input dtype (usually FP32) and return.

The heavy O(N³·H) reduction is performed entirely on the GPU with a
single Triton kernel, yielding a large speed‑up over the PyTorch+cuBLAS
baseline.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, H, M, N, K,                         # sizes
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_j, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: for each (b,h) compute out = left @ rightᵀ."""
    pid = tl.program_id(0)

    # number of tiles per output matrix
    grid_m = tl.cdiv(M, BLOCK_M)
    grid_n = tl.cdiv(N, BLOCK_N)
    total_batches = B * H

    # decode program id → batch and tile coordinates
    batch_idx = pid // (grid_m * grid_n)
    tile_idx = pid % (grid_m * grid_n)
    m_tile = tile_idx // grid_n
    n_tile = tile_idx % grid_n

    # offsets within the M and N dimensions
    m_offset = m_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    n_offset = n_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # out‑of‑bounds masks for the output tile
    mask_m = m_offset < M
    mask_n = n_offset < N

    # decode batch (b) and hidden (h) indices
    b_idx = batch_idx // H
    h_idx = batch_idx - b_idx * H

    # base pointers for this (b,h) slice
    left_base  = left_ptr  + b_idx * stride_l_b + h_idx * stride_l_h
    right_base = right_ptr + b_idx * stride_r_b + h_idx * stride_r_h
    out_base   = out_ptr   + b_idx * stride_o_b + h_idx * stride_o_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # left tile:   [M, K]
        left_ptrs = left_base + (m_offset[:, None] * stride_l_i) \
                                 + (k_offset[None, :] * stride_l_k)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile:  [K, N]  (note the order: k then n)
        right_ptrs = right_base + (k_offset[:, None] * stride_r_k) \
                                   + (n_offset[None, :] * stride_r_j)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        acc += tl.dot(left, right)   # FP16 dot, accumulate in FP32

    # write the result back (FP16)
    out = acc.to(tl.float16)
    out_ptrs = out_base + (m_offset[:, None] * stride_o_i) \
                         + (n_offset[None, :] * stride_o_j)
    tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]
                 ) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C] (float32/float16)
            - mask:        [B, N, N]          or None
            - weights: dict of model parameters (float32)
            - config:  dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask"      – optional bool, ignore mask if True

    Returns:
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # float32

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections & gates (keep FP32 for accuracy)
    # -----------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # -----------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and elementwise gating
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # mask → [B,N,N,1] for broadcasting over hidden dim
        mask_fp = mask.to(x.dtype).unsqueeze(-1)
        left  = left_proj * left_gate * mask_fp
        right = right_proj * right_gate * mask_fp
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ Cast to FP16 and reshape for batched GEMM
    # -----------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)
    # shape → (B, H, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape

    # output buffer (FP16)
    out = torch.empty_like(left)

    # stride vectors (in elements)
    lsb, lsh, lsi, lsk = left.stride()
    rsb, rsh, rsj, rsk = right.stride()
    osb, osh, osi, osj = out.stride()

    # -----------------------------------------------------------------
    # 5️⃣ Batched GEMM via Triton
    # -----------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_batches = B * H
    grid = (total_batches * grid_m * grid_n,)

    _batched_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        H,
        N,          # M
        N,          # N
        N,          # K
        lsb,
        lsh,
        lsi,
        lsk,
        rsb,
        rsh,
        rsj,
        rsk,
        osb,
        osh,
        osi,
        osj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Reshape back, LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    out = out.permute(0, 2, 3, 1).contiguous()            # (B,N,N,H)
    out = out.to(torch.float32)                           # for stable LN

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out = out * out_gate                                   # out_gate is FP32
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to original dtype and return
    # -----------------------------------------------------------------
    out = out.to(orig_dtype)
    return out
