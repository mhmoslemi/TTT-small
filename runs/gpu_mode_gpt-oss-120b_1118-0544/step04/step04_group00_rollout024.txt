@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak, stride_bn, stride_bk,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # batch id
    batch_idx = tl.program_id(0)
    # block id for output matrix
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    
    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute actual pointer for a and b blocks
    a_ptrs = a_ptr + batch_idx * batch_stride_a + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch_idx * batch_stride_b + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn
    
    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load a and b block, with mask for OOB
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (tl.arange(0,BLOCK_K)[None,:] + k*BLOCK_K < K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0,BLOCK_K)[:, None] + k*BLOCK_K < K) & (offs_n[None,:] < N), other=0.0)
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
    # store
    c_ptrs = c_ptr + batch_idx * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (`dim`).
2. Cast to FP16 for compute‑heavy parts.
3. Linear projections:
       left  = x @ W_left   (B×N×N×H)
       right = x @ W_right
   Gate projections (sigmoid):
       left_gate  = σ(x @ W_left_gate)
       right_gate = σ(x @ W_right_gate)
       out_gate   = σ(x @ W_out_gate)   (B×N×N×H)
4. Fuse mask, gate and projection with a small Triton kernel
   (applied only when a mask is present).
5. Core reduction – batched matrix multiplication
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton GEMM that processes all
   (batch * H) matrices in parallel.
6. Output LayerNorm over the hidden dimension `H`.
7. Multiply by `out_gate`, apply the final linear projection back to `dim`.
8. Cast the result back to the original dtype (usually float32).

Only the mask‑gate fusion and the N³ reduction are done in Triton;
all other ops use highly‑optimised PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask   (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # total number of elements in proj/gate (=B*N*N*H)
    hidden_dim,        # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)          # [0, BLOCK_SIZE)

    valid = offs < total_elems

    # map element index to corresponding mask element (broadcast over H)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(out_ptr + offs, proj * gate * mask, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Fuse projection, gate and mask using Triton.
    All tensors must be fp16 and contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)   (binary 0/1, fp16)
    Returns proj * gate * mask  (broadcast over H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ B
#   A : (batch, M, K)    (row‑major)
#   B : (batch, K, N)    (row‑major)
#   C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,          # A: (M, K)
    stride_bk, stride_bn,          # B: (K, N)
    stride_cm, stride_cn,          # C: (M, N)
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets of the current output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile (k = 0)
    a_tile_ptr = a_ptr + batch_idx * batch_stride_a + \
                 offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_tile_ptr = b_ptr + batch_idx * batch_stride_b + \
                 tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K axis
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        k_offs = tl.arange(0, BLOCK_K)
        mask_k = k_offs < K

        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

        # Advance to next K‑tile
        a_tile_ptr += BLOCK_K * stride_ak
        b_tile_ptr += BLOCK_K * stride_bk

    # Write back the result
    c_tile_ptr = c_ptr + batch_idx * batch_stride_c + \
                 offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm(a: torch.Tensor, b: torch.Tensor,
                  block_m: int = 64, block_n: int = 64, block_k: int = 64) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    a : (B, M, K)   fp16, contiguous
    b : (B, K, N)   fp16, contiguous
    Returns c : (B, M, N)  fp16, contiguous
    """
    B, M, K = a.shape
    assert b.shape[0] == B and b.shape[1] == K
    N = b.shape[2]

    c = torch.empty((B, M, N), dtype=a.dtype, device=a.device)

    # Strides in element units
    batch_stride_a = a.stride(0)
    batch_stride_b = b.stride(0)
    batch_stride_c = c.stride(0)

    stride_am = a.stride(1)
    stride_ak = a.stride(2)

    stride_bk = b.stride(1)
    stride_bn = b.stride(2)

    stride_cm = c.stride(1)
    stride_cn = c.stride(2)

    grid = (B,
            triton.cdiv(M, block_m),
            triton.cdiv(N, block_n))

    _batched_gemm_kernel[grid](
        a,
        b,
        c,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        M,
        N,
        K,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return c

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C]  (float32 or float16)
            - mask         : torch.Tensor of shape [B, N, N] or None (binary 0/1)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys
                * "dim" (int)        – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Load all linear weights in FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, H)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Fuse mask, projection and gate (Triton) if mask is present
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM (left @ right^T) in Triton.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # Transpose right on the (i,k) axes to obtain the needed K‑N layout
    right_t = right.transpose(1, 2).contiguous()   # (B, N, N, H) where axes are (k, j)

    # Collapse batch and hidden dimensions for the GEMM
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, M=N, K=N)
    right_flat = right_t.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H, K=N, N)

    # Batch GEMM (A @ B)
    out_flat = _batched_gemm(left_flat, right_flat,
                             block_m=64, block_n=64, block_k=64)   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
