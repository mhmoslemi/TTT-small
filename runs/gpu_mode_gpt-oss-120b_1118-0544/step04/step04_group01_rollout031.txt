@triton.jit
def _batched_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    total_heads, N,
    stride_l_head, stride_l_i, stride_l_k,
    stride_r_head, stride_r_k, stride_r_j,
    stride_o_head, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # compute number of blocks
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    # compute head id, block row, block col
    head = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_id // num_blocks_n
    pid_n = block_id % num_blocks_n

    # compute offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for i, j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left block [i, k] for this head
        # left offset: head*stride_l_head + offs_i[:,None]*stride_l_i + offs_k[None,:]*stride_l_k
        left_ptrs = left_ptr + head * stride_l_head + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # load right transposed block [k, j] for this head
        # right offset: head*stride_r_head + offs_k[:, None]*stride_r_k + offs_j[None, :] * stride_r_j
        right_ptrs = right_ptr + head * stride_r_head + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float16)

        # accumulate dot product
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + head * stride_o_head + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass – Triton fused implementation.

Algorithm
---------
1. Layer‑norm the input (B,N,N,C) using the provided LN weights.
2. Cast to fp16 for the heavy compute path.
3. Linear projections for the left/right streams and their gates.
4. Fuse projection * gate * mask (if a mask is supplied) with a tiny
   Triton kernel.
5. Compute the O(N³·H) reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply of shape (B·H, N, N) × (B·H, N, N)ᵀ
   and is carried out by a custom Triton GEMM kernel.
6. Apply a second LayerNorm on the hidden dimension, multiply by the
   output gate, and perform the final linear projection back to C.
7. Cast the result back to the original dtype.

Only the mask‑gate fusion and the heavy GEMM are executed in Triton,
while the remaining ops use PyTorch’s highly‑optimized kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Kernel 1: proj * gate * mask  (broadcast mask over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,   # fp16 * (B*N*N*H,)
    gate_ptr,   # fp16 * (B*N*N*H,)
    mask_ptr,   # fp16 * (B*N*N,)
    out_ptr,    # fp16 * (B*N*N*H,)
    total_elems,   # i32   total elements in proj/gate
    hidden_dim,    # i32   H (used for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # each hidden‑dim block shares the same mask entry
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0).to(tl.float16)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0).to(tl.float16)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0).to(tl.float16)

    tl.store(out_ptr + offs, proj * gate * mask, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 4096) -> torch.Tensor:
    """Fuse `proj * gate * mask` via Triton."""
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the O(N³·H) reduction
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    left_ptr,   # fp16 * (heads, N, N)
    right_ptr,  # fp16 * (heads, N, N)  (original layout, accessed as [k, j])
    out_ptr,    # fp16 * (heads, N, N)
    total_heads, N,
    stride_l_head, stride_l_i, stride_l_k,
    stride_r_head, stride_r_k, stride_r_j,
    stride_o_head, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of blocks in each dimension
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # decode program id into head, block-row, block-col
    head = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_id // num_blocks_n
    pid_n = block_id % num_blocks_n

    # block offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[head, i, k]
        left_ptrs = left_ptr + head * stride_l_head \
                    + offs_i[:, None] * stride_l_i \
                    + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float16)

        # right[head, k, j]   (we read transposed)
        right_ptrs = right_ptr + head * stride_r_head \
                     + offs_k[:, None] * stride_r_k \
                     + offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float16)

        acc += tl.dot(left, right)   # fp16 dot → fp32 accumulation

    # write result
    out_ptrs = out_ptr + head * stride_o_head \
               + offs_i[:, None] * stride_o_i \
               + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _batched_gemm(left: torch.Tensor,
                  right: torch.Tensor,
                  block_m: int = 64,
                  block_n: int = 64,
                  block_k: int = 64) -> torch.Tensor:
    """
    Batched GEMM: out = left @ right⁻ᵀ  where left/right shape = (B, H, N, N)
    Returns tensor of shape (B, N, N, H) (fp16).
    """
    B, H, N, _ = left.shape
    heads = B * H

    # flatten heads dimension
    left_flat = left.view(heads, N, N)
    right_flat = right.view(heads, N, N)
    out_flat = torch.empty_like(left_flat)

    # strides (in elements)
    stride_l_head = left_flat.stride(0)
    stride_l_i = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    stride_r_head = right_flat.stride(0)
    stride_r_j = right_flat.stride(1)   # original j dimension
    stride_r_k = right_flat.stride(2)   # original k dimension

    stride_o_head = out_flat.stride(0)
    stride_o_i = out_flat.stride(1)
    stride_o_j = out_flat.stride(2)

    # launch grid
    num_blocks_m = (N + block_m - 1) // block_m
    num_blocks_n = (N + block_n - 1) // block_n
    total_programs = heads * num_blocks_m * num_blocks_n
    grid = (total_programs,)

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        heads, N,
        stride_l_head, stride_l_i, stride_l_k,
        stride_r_head, stride_r_k, stride_r_j,
        stride_o_head, stride_o_i, stride_o_j,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be ignored)
            - weights: dict with model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # work in fp16
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5. Fuse proj * gate * mask (if needed)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core O(N³·H) reduction via a custom batched GEMM
    #    left/ right shape: (B, N, N, H) -> (B, H, N, N)
    # ------------------------------------------------------------------
    # bring hidden dim to second axis for convenient flattening
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    out = _batched_gemm(left, right)               # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7. Second LayerNorm + output gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
