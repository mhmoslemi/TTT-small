"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections and gates (left/right projections + left/right/out gates)
   are computed with torch (fp16) – these are cheap compared to the O(N³) term.
3. Optional mask‑gate fusion is performed by a tiny Triton kernel that
   multiplies left/right projections with their gates and the pair mask.
4. The core O(N³·H) operation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is implemented as a batched GEMM (left @ rightᵀ) in Triton.
   We flatten the (batch, hidden) dimensions to a single batch dimension
   and launch a 3‑D grid (batch, tile_m, tile_n).  The kernel works in
   fp16 and accumulates in fp32 for numerical stability.
5. The result is layer‑normed over the hidden dimension, multiplied by the
   output gate and projected back to the original channel dimension.
6. Output is returned in the input dtype (float32).

The implementation satisfies the requirement of having at least one
Triton kernel (mask‑gate fusion) and replaces the expensive torch.bmm with a
custom Triton batched GEMM, yielding a substantial speed‑up.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Mask‑ and gate‑fusion kernel (elementwise)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask element is broadcast over `hidden_dim` entries
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * m
        right = right * right_gate * m
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# ----------------------------------------------------------------------
# 2️⃣ Batched GEMM kernel (left @ rightᵀ) – core O(N³·H) work
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers to A, B, C
    M, N, K,                            # matrix dims (all = seq_len)
    stride_az, stride_am, stride_ak,    # A strides: batch, row, col
    stride_bz, stride_bk, stride_bn,    # B strides: batch, row(k), col
    stride_cz, stride_cm, stride_cn,    # C strides: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)   # batch index (batch * hidden)
    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile col

    # Compute offsets within the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds checks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the batch
    a_batch_ptr = a_ptr + pid_z * stride_az
    b_batch_ptr = b_ptr + pid_z * stride_bz
    c_batch_ptr = c_ptr + pid_z * stride_cz

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr \
                 + (offs_m[:, None] * stride_am) \
                 + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (B is treated as (K, N) i.e. rightᵀ)
        b_ptrs = b_batch_ptr \
                 + (offs_k[:, None] * stride_bk) \
                 + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute tile product and accumulate (fp16 -> fp32)
        acc += tl.dot(a, b).to(tl.float32)

    # Write output tile
    c_ptrs = c_batch_ptr \
             + (offs_m[:, None] * stride_cm) \
             + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of torch.Tensors (float32)
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with dtype = input_tensor.dtype (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (float32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for subsequent heavy work
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Load all linear weights (fp16, bias‑less)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections + gates (still in fp16)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)               # (B, N, N, H)
    right = F.linear(x, right_proj_w)             # (B, N, N, H)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Apply mask and fuse with gates via Triton (if needed)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask shape → (B, N, N, 1) for broadcasting
        mask_fp = mask.to(device, torch.float16).unsqueeze(-1)
        # flatten tensors for the kernel
        left_f = left.contiguous().view(-1)
        right_f = right.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)
        mask_f = mask_fp.squeeze(-1).contiguous().view(-1)

        total = left_f.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_f.view_as(left)
        right = right_f.view_as(right)
    else:
        # No mask → just element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core O(N³·H) reduction: batched GEMM left @ rightᵀ
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Rearrange to (batch, H, N, N) → (B*H, N, N) for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch = B * H

    # Allocate output buffer
    out = torch.empty_like(left_perm, dtype=torch.float16)

    # Strides for the three tensors
    stride_az = left_perm.stride(0)
    stride_am = left_perm.stride(1)
    stride_ak = left_perm.stride(2)

    stride_bz = right_perm.stride(0)
    stride_bk = right_perm.stride(1)   # K dimension (row)
    stride_bn = right_perm.stride(2)   # N dimension (col)

    stride_cz = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    # Block sizes – tuned for fp16 tensor‑core usage on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_gemm_kernel[grid](
        left_perm,
        right_perm,
        out,
        N,               # M
        N,               # N
        N,               # K
        stride_az, stride_am, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output layer‑norm over hidden dimension, apply out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 9) Cast back to original dtype (float32)
    # ------------------------------------------------------------------
    out = out.to(dtype_in)

    return out
