@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_az, stride_am, stride_ak,  # stride for batch, M, K (a)
    stride_bz, stride_bk, stride_bn,  # stride for batch, K, N (b)
    stride_cz, stride_cm, stride_cn,  # stride for batch, M, N (c)
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(0) # batch id
    pid_m = tl.program_id(1) # block row
    pid_n = tl.program_id(2) # block col

    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # check boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) # maybe accumulation in fp32 for better accuracy

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load a and b
        a = tl.load(a_ptr + pid_z*stride_az + offs_m[:, None] * stride_am + offs_k[None,:] * stride_ak, mask=mask_m[:,None] & mask_k[None,:], other=0.0)
        b = tl.load(b_ptr + pid_z*stride_bz + offs_k[:, None] * stride_bk + offs_n[None,:] * stride_bn, mask=mask_k[:,None] & mask_n[None,:], other=0.0)

        # dot
        acc += tl.dot(a, b)  # a shape (BLOCK_M, BLOCK_K) , b shape (BLOCK_K, BLOCK_N)
    # write back
    c = acc.to(tl.float16)
    tl.store(c_ptr + pid_z*stride_cz + offs_m[:, None] * stride_cm + offs_n[None,:] * stride_cn, c, mask=mask_m[:,None] & mask_n[None,:])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input tensor (float32 for numerical stability).
2. Cast to fp16 and apply six linear projections:
   left_proj, right_proj, left_gate, right_gate, out_gate, to_out.
3. Fuse gating (sigmoid) and optional mask using a small Triton kernel.
4. Compute the core contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batch of H independent matrix multiplications (size N×N) with a
   custom Triton GEMM kernel (B×H batched matmul of left @ rightᵀ).
5. Apply output layer‑norm, multiply by out_gate and a final linear projection.
6. Return the result in the original input dtype (float32).

Both the elementwise gate/mask fusion and the batched matmul are
implemented in Triton, satisfying the requirement to use a kernel
while keeping the overall runtime well below the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse left/right projection with their gates and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,            # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,  # same shape
    mask_ptr,                       # [B,N,N] flattened (broadcasted over H)
    total_elems, hidden_dim,        # runtime scalars
    HAS_MASK: tl.constexpr,         # compile‑time bool
    BLOCK_SIZE: tl.constexpr,       # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # load tensors (fp16)
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden_dim elements share one mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (left @ rightᵀ) for all hidden channels
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_az, stride_am, stride_ak,   # a: (batch, M, K)
    stride_bz, stride_bk, stride_bn,   # b: (batch, K, N)
    stride_cz, stride_cm, stride_cn,   # c: (batch, M, N)
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)  # batch id
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    # starting offsets of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A and B tiles
        a = tl.load(
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot: (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # Write back in fp16
    c = acc.to(tl.float16)
    tl.store(
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C) on CUDA
            - mask: torch.Tensor of shape (B, N, N) or None (also on CUDA)
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, mask is ignored

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on the input (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # Cast to float16 for compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Load linear weights (fp16, bias=False in the reference)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections (fp16)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))  # (B, N, N, H)

    # ------------------------------------------------------------------
    # Fuse gating (and optional mask) with a Triton kernel
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Convert mask to fp16 and flatten
        mask_fp = mask.to(device).to(torch.float16)
        mask_flat = mask_fp.contiguous().view(-1)           # (B*N*N,)
    else:
        # Dummy mask (will never be read)
        mask_flat = torch.empty(0, device=device, dtype=torch.float16)

    # Ensure contiguous tensors for the kernel
    left       = left.contiguous()
    right      = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # Flatten all tensors to 1‑D
    total_elems = left.numel()          # B * N * N * H
    left_flat  = left.view(-1)
    right_flat = right.view(-1)
    lgate_flat = left_gate.view(-1)
    rgate_flat = right_gate.view(-1)

    BLOCK = 1024
    grid_gate = lambda meta: (triton.cdiv(total_elems, meta["BLOCK_SIZE"]),)

    _gate_mask_kernel[grid_gate](
        left_flat,
        right_flat,
        lgate_flat,
        rgate_flat,
        mask_flat,
        total_elems,
        hidden_dim,
        HAS_MASK= (mask is not None and not nomask),
        BLOCK_SIZE=BLOCK,
    )

    # Reshape back to original layout
    left = left_flat.view_as(left)
    right = right_flat.view_as(right)

    # ------------------------------------------------------------------
    # Core contraction: batched matmul left @ rightᵀ for every hidden channel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape          # B, N, N, H
    # (B, H, N, N) layout for easy batching
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)       # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)      # (B*H, N, N) = rightᵀ

    out_mat = torch.empty_like(left_mat)  # fp16 output

    # Determine block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_matmul = (
        (B * H, ),                               # batch dimension
        (triton.cdiv(N, BLOCK_M),),              # row blocks
        (triton.cdiv(N, BLOCK_N),),              # column blocks
    )

    _batched_matmul_kernel[grid_matmul](
        left_mat,
        right_mat,
        out_mat,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate                      # element‑wise gate
    out = F.linear(out, to_out_w)             # (B, N, N, C)

    # Cast back to original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
