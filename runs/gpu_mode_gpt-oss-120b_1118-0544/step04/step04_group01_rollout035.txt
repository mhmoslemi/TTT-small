    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M, BLOCK_N, BLOCK_K: tl.constexpr
):

    pid1 = tl.program_id(1)  # j block
    pid2 = tl.program_id(2)  # batch_hidden index

    bh = pid2
    b = bh // H
    d = bh % H

    # offsets
    i_start = pid0 * BLOCK_M
    j_start = pid1 * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_k = tl.arange(0, BLOCK_K)

    mask_j = (j_start + tl.arange(0, BLOCK_N)) < N
    mask_k = offs_k < N

    # same for right: we want right block (k,j) shape (BLOCK_K, BLOCK_N) using row stride = stride_right_k (k) and column stride = stride_right_j (j)

"""
TriMul (outgoing) forward pass – Triton fused O(N³·H) reduction.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) with the provided LN weights.
2. Cast everything to fp16 for compute‑heavy paths.
3. Linear projections for “left”, “right” and their gates.
4. Optional binary mask is broadcast over the hidden dimension and applied
   together with the gates (pure torch elementwise – fast enough).
5. Core reduction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication A·Bᵀ for each hidden slice.
   A custom Triton kernel processes a (BLOCK_M×BLOCK_N) tile of the
   output while looping over the k‑dimension in bricks of size BLOCK_K.
   The kernel receives the full stride information of the three tensors
   and runs one program per (batch, hidden) slice.
6. Layer‑norm on the hidden dimension, multiplication by the output gate,
   and a final linear projection back to the original channel dimension.
7. Cast the result back to the input dtype.

The heavy O(N³·H) work is performed entirely in Triton, satisfying the
requirement that “part of the operations” be implemented in a kernel.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing triangle multiplicative update
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,                     # pointers
    B, N, H,                                          # shape params
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_i = tl.program_id(0)   # i‑block
    pid_j = tl.program_id(1)   # j‑block
    pid_h = tl.program_id(2)   # batch*hidden program

    # map pid_h -> batch and hidden index
    b = pid_h // H
    d = pid_h % H

    # start offsets for this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # ranges within the tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # boundary masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the k‑dimension
    for k in range(0, N, BLOCK_K):
        cur_k = k + offs_k
        mask_k = cur_k < N

        # ------------------------------------------------------------------
        # Load left tile   shape (BLOCK_M, BLOCK_K)
        # offset = b*stride_left_b + d*stride_left_d
        #        + i*stride_left_i + k*stride_left_k
        # ------------------------------------------------------------------
        left_ptrs = left_ptr + b*stride_left_b + d*stride_left_d \
                    + offs_i[:, None] * stride_left_i \
                    + cur_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # ------------------------------------------------------------------
        # Load right tile  shape (BLOCK_K, BLOCK_N) (k as rows, j as cols)
        # offset = b*stride_right_b + d*stride_right_d
        #        + k*stride_right_k + j*stride_right_j
        # ------------------------------------------------------------------
        right_ptrs = right_ptr + b*stride_right_b + d*stride_right_d \
                     + cur_k[:, None] * stride_right_k \
                     + offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # ------------------------------------------------------------------
        # Accumulate dot product: (M,K) * (K,N) -> (M,N)
        # tl.dot works with FP16 inputs and FP32 accumulator.
        # ------------------------------------------------------------------
        acc += tl.dot(left, right)

    # ----------------------------------------------------------------------
    # Store the result tile (cast back to FP16)
    # ----------------------------------------------------------------------
    out_ptrs = out_ptr + b*stride_out_b + d*stride_out_d \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict with model parameters (float32)
            - config: dictionary with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # Input layer‑norm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Cast to fp16 for all heavy computation
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # Gates (with sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # Apply mask (if any) and combine with gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B,N,N) -> broadcast over hidden dim
        mask = mask.to(torch.float16).unsqueeze(-1)       # (B,N,N,1)
        left  = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguity for Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # Core O(N³·H) reduction using Triton
    # ------------------------------------------------------------------
    B_, N_, _, H_ = left.shape
    assert (B_, N_, H_) == (B, N, hidden_dim), "Shape mismatch."

    # Allocate output tensor (fp16)
    out_hidden = torch.empty_like(left)  # (B,N,N,H)

    # Tile sizes – tuned for typical B200 GPUs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Strides (in elements) for each tensor
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out_hidden.stride()

    # Grid: (i‑tiles, j‑tiles, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out_hidden,
        B,
        N,
        hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_j, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Second LayerNorm over hidden dimension, gate, and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate                               # (B,N,N,H)

    # Final linear projection back to original channel dimension
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # Cast back to the original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
