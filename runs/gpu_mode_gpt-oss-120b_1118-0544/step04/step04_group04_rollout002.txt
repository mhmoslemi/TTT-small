"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The implementation follows the AlphaFold3 TriMul module:
1. Layer‑norm over the input channel dimension (float16 compute).
2. Linear projections for left/right vectors and their gates.
3. Optional masking of the left/right projections.
4. Gating via sigmoid.
5. Batched matrix multiplication (per hidden channel) implemented with
   torch.matmul on tensors of shape (B, H, N, N). This uses cuBLAS/Tensor
   Cores and is far faster than a naïve Python loop.
6. Layer‑norm over the hidden dimension.
7. Element‑wise multiplication with the output‑gate using a tiny Triton
   kernel (required by the task).
8. Final linear projection back to the original dimension.
All heavy arithmetic is performed in float16; the result is cast back
to float32 before returning.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    """out = out * gate   (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out * gate, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float). May be ignored if config["nomask"] is True.
            - weights: dict of model weight tensors (float32)
            - config: dict containing:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension H
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # -------------------------------------------------
    # 1. Input LayerNorm (float32 → cast to fp16 later)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim], float32

    # -------------------------------------------------
    # 2. Cast to fp16 and compute linear projections
    # -------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Linear weights (fp16)
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # (B, N, N, H)
    right = F.linear(x_fp16, right_proj_w)  # (B, N, N, H)

    # -------------------------------------------------
    # 3. Optional mask
    # -------------------------------------------------
    if not nomask:
        # mask: (B, N, N) → (B, N, N, 1) → fp16
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------
    # 4. Gating (sigmoid)
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))  # (B, N, N, H)

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5. Batched GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as (B, H, N, N) @ (B, H, N, N)^T
    # -------------------------------------------------
    # Move hidden dimension to batch axis for efficient matmul
    left = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2) # (B, H, N, N)

    # Matrix multiplication per (B, H) batch
    out = torch.matmul(left, right.transpose(-2, -1))  # (B, H, N, N)

    # Back to original layout
    out = out.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # -------------------------------------------------
    # 6. to_out_norm (LayerNorm over hidden dim)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # (B, N, N, H), fp16

    # -------------------------------------------------
    # 7. out_gate (linear + sigmoid) and element‑wise multiply via Triton
    # -------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))  # (B, N, N, H), fp16

    total_elements = out.numel()
    BLOCK = 1024
    grid = ( (total_elements + BLOCK - 1) // BLOCK, )
    _elementwise_mul_kernel[grid](
        out.data_ptr(),
        out_gate.data_ptr(),
        total_elements,
        BLOCK=BLOCK,
    )

    # -------------------------------------------------
    # 8. Final projection back to original dim
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, C), fp16

    # Cast to float32 as required by the external interface
    return out.to(torch.float32)
```
