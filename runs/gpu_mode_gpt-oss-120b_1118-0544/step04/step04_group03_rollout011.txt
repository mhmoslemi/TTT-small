4. Apply the optional binary mask to the projections.
5. Apply sigmoid to the gate logits and multiply (proj * gate) to obtain
   ``left`` and ``right`` tensors of shape [B, N, N, H].
6. Reshape to a batched matrix‑multiplication problem:
   for each batch b and channel h we need
       out[b, :, :, h] = left[b, :, :, h] @ right[b, :, :, h].T
   i.e. a GEMM of size N×N × N×N for B*H independent batches.
   A custom Triton kernel performs this batched GEMM efficiently.
7. Reshape the result back to [B, N, N, H] and apply a second LayerNorm.
8. Multiply by the output gate (sigmoid of a linear projection of the
   original normalized input).
9. Final linear projection back to the original channel dimension and cast
   to the input dtype.

Only the O(N³·H) reduction (step 6) is implemented in Triton; all other
operations use regular PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton batched GEMM kernel: C = A @ Bᵀ
# A : (batch, M, K)   – row‑major (stride_am = K, stride_ak = 1)
# B : (batch, K, N)   – stored as (K, N) but we load it transposed:
#                      row stride = stride_bk (=1), col stride = stride_bn (=K)
# C : (batch, M, N)
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # base pointers
    M, N, K,                           # matrix dimensions (all = seq_len)
    stride_am, stride_ak,              # A strides (row, col)
    stride_bk, stride_bn,              # B strides (row = K‑dim, col = N‑dim)
    stride_cm, stride_cn,              # C strides (row, col)
    stride_ab, stride_bb, stride_cb,   # batch strides for A, B, C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)           # tile index (i‑j)
    batch = tl.program_id(1)         # batch index (B*H)

    # tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch‑specific base pointers
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Bᵀ tile: shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp16·fp16 → fp32

    # write result
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------------------- #
# Entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32 or float16)
            - mask        : torch.Tensor [B, N, N] (binary) or None
            - weights     : dict of model parameters (float32)
            - config      : dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask
    Returns:
        torch.Tensor [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    orig_dtype = x.dtype
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # operate in float16 for the heavy part
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 2. Cast and fuse linear weights (proj + gate)
    # ------------------------------------------------------------------- #
    # left path: (proj, gate) → [2*H, C]
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------- #
    # 3. Linear proj + gate (fused)
    # ------------------------------------------------------------------- #
    left_comb = F.linear(x_fp16, left_comb_w)   # [B, N, N, 2*H]
    right_comb = F.linear(x_fp16, right_comb_w) # [B, N, N, 2*H]

    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    left = left_proj * left_gate   # [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------- #
    # 4. Batched GEMM via Triton: out = left @ rightᵀ  (per channel)
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape  # (B, N, N, H)

    # bring channel dim to batch dimension: (B*H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_t.view(B * H, N, N)   # (batch, M=N, K=N)
    right_flat = right_t.view(B * H, N, N) # (batch, K=N, N)

    out_flat = torch.empty_like(left_flat)  # (batch, M, N) fp16

    # strides
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)   # row stride for K (original column)
    stride_b_n = right_flat.stride(1)   # col stride for N (original row)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # block sizes – tuned for typical seq_len ≤ 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _batched_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,                             # M, N, K
        stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 5. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------- #
    # 6. Output gate
    # ------------------------------------------------------------------- #
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()   # [B, N, N, H]
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------- #
    out = F.linear(out, to_out_w)   # [B, N, N, C]

    # cast to original dtype
    out = out.to(orig_dtype)
    return out
```
