import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _gate_mask_fusion_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # each hidden element repeats mask per hidden_dim
    mask_idx = offs // hidden_dim

    p = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = p * g * m
    tl.store(out_ptr + offs, out, mask=mask)

def _apply_gate_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_fusion_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    This implementation:
    1. Layer‑norms the input.
    2. Computes left/right linear projections and their sigmoid gates.
    3. Fuses (proj * gate * mask) with a small Triton kernel when a mask is present.
    4. Performs the O(N³·H) contraction using a batched fp16 GEMM (torch.bmm) which
       maps to cuBLAS/Tensor‑core kernels.
    5. Applies a second LayerNorm, the output gate, and the final linear projection.
    6. Returns the result in the original dtype.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
        - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
        - weights: dict of model parameters (float32)
        - config: dict containing at least ``dim`` and ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (same dtype as the input)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device
    orig_dtype = x.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for computation heavy paths
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16 (single conversion)
    # ------------------------------------------------------------------
    def _to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    left_proj_w = _to_fp16(weights["left_proj.weight"])
    right_proj_w = _to_fp16(weights["right_proj.weight"])
    left_gate_w = _to_fp16(weights["left_gate.weight"])
    right_gate_w = _to_fp16(weights["right_gate.weight"])
    out_gate_w = _to_fp16(weights["out_gate.weight"])
    to_out_norm_w = _to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = _to_fp16(weights["to_out_norm.bias"])
    to_out_w = _to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 4. Linear projections
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate values
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask + gate (fused via Triton when mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask broadcast to hidden_dim
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction: batched GEMM over the reduction dimension k
    #    left, right : (B,N,N,H)
    #    -> (B*H, N, N) for torch.bmm
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T  => (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_fusion_kernel(
    proj_ptr,                 # fp16* (B*N*N*H,)
    gate_ptr,                 # fp16* (B*N*N*H,)
    mask_ptr,                 # fp16* (B*N*N,)
    out_ptr,                  # fp16* (B*N*N*H,)
    total_elems,              # i64 total elements in proj/gate
    hidden_dim,               # i64 H  (size of the last dim)
    BLOCK_SIZE: tl.constexpr, # compile‑time block size
):
    """Fuse ``proj * gate * mask`` where ``mask`` is broadcast over the hidden
    dimension.  Each thread works on one element of the flattened ``proj``/``gate``
    tensors."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # every hidden element repeats the same mask entry
    mask_idx = offs // hidden_dim

    p = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = p * g * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse element‑wise ``proj * gate * mask`` using a Triton kernel.
    All tensors must be fp16, contiguous and on the same device.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _gate_mask_fusion_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],        # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    The algorithm follows the reference PyTorch implementation but
    fuses the mask‑gate multiplication in a Triton kernel and
    uses fp16 Tensor‑core GEMMs for the heavy ``O(N³·H)`` contraction.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        * input_tensor – shape [B, N, N, C] (float32/float16)
        * mask – shape [B, N, N] (bool/int/float) or None
        * weights – dict of parameters (float32)
        * config – must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor – shape [B, N, N, C] (same dtype as input)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device
    orig_dtype = x.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Convert all weights to fp16 once
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w      = to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,C) -> (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Fuse mask, projection and gate (if mask present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj,  left_gate,  mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM (Tensor‑core friendly)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # (B, H, N, N) views → (B*H, N, N) matrices
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication with transpose on the second operand
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm → output gate → final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype and return
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)
    return out
