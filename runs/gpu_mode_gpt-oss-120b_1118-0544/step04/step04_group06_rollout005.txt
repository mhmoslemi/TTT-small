import torch
import triton
import triton.language as tl
from torch.nn import functional as F
from typing import Tuple, Dict, Any

@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,  # *fp16
    gate_ptr,  # *fp16
    mask_ptr,  # *fp16
    out_ptr,   # *fp16
    total_elems,  # i64
    hidden_dim: tl.constexpr,  # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # compute index into mask: each mask element repeats hidden_dim times
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * mval
    tl.store(out_ptr + offs, out, mask=mask)

def _apply_gate_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    proj, gate: (B,N,N,H) fp16 contiguous
    mask: (B,N,N) fp16 contiguous
    returns proj*gate*mask broadcasted over H.
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_fuse_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Arguments
    ---------
    data: tuple containing
        - input_tensor: (B, N, N, C)   float32/float16
        - mask: (B, N, N)  (bool/int/float) or None
        - weights: dict of model parameters (float32)
        - config: dict with keys
            * "dim" (int) – input channel dimension C
            * "hidden_dim" (int) – hidden dimension H
            * "nomask" (bool, optional)

    Returns
    -------
    Tensor of shape (B, N, N, C) with same dtype as input.
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive part
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Cast weights to FP16 on the same device
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # -----------------------------------------------------------------
    # 4. Linear projections : (B,N,N,C) -> (B,N,N,H)
    # -----------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # shape (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 5. Gates (sigmoid)
    # -----------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # -----------------------------------------------------------------
    # 6. Fuse mask, projection and gate (Triton) when needed
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 7. Core O(N³·H) reduction: batched GEMM across hidden dimension
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication (left * rightᵀ)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8. Output LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 9. Cast back to original datatype
    # -----------------------------------------------------------------
    out = out.to(orig_dtype)

    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

The kernel fuses three element‑wise operations:
    proj * gate * mask
where `proj` and `gate` have shape (B, N, N, H) and `mask` (B, N, N) is
broadcasted over the hidden dimension H.  The heavy O(N³·H) reduction is
performed with a batched GEMM using torch.bmm (cuBLAS), which is the
fastest approach on modern GPUs.  All compute‑intensive parts run in
float16; the final result is cast back to the input dtype.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_fuse_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i64  – total number of elements in proj/gate
    hidden_dim,        # i64  – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse proj * gate * mask (mask broadcasted over hidden_dim)."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # bounds check
    mask = offs < total_elems

    # each mask element repeats hidden_dim times in the flattened tensor
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * mval
    tl.store(out_ptr + offs, out, mask=mask)

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # -----------------------------------------------------------------
    # 6. Fuse mask, projection and gate (Triton) when a mask is present
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 7. Core reduction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    This is a batched GEMM over (B*H, N, N) matrices.
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # (B, H, N, N) layout for cuBLAS
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8. Output LayerNorm, gating, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    # -----------------------------------------------------------------
    # 9. Cast back to original dtype
    # -----------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
```
