"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate.

Algorithm
---------
1. Layer‑normalize the input tensor (B, N, N, C).
2. Cast to fp16 for all compute‑intensive ops.
3. Four linear projections (left/right + their gates) are performed with
   fp16 weights. Sigmoid is applied to the gate outputs.
4. If a mask is supplied, a tiny Triton kernel fuses `proj * gate * mask`
   in a single pass (otherwise we just multiply element‑wise).
5. Core O(N³·H) reduction is realized as a batched matrix multiplication:
   for each head `h` we compute `out_h = left_h @ right_h.T`.
6. A second LayerNorm is applied, the result is multiplied by the output
   gate, and a final linear projection brings the tensor back to the
   original channel dimension.
7. Cast the result back to the input dtype.

Only the mask‑gate fusion is done in Triton; the heavy matrix
multiplications use PyTorch’s highly‑optimized cuBLAS/Tensor‑core paths.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # index into the mask (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0).to(tl.float16)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0).to(tl.float16)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0).to(tl.float16)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse ``proj * gate * mask`` using the Triton kernel above.
    All tensors must be on the same CUDA device, contiguous and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],               # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for heavy compute
    x = x.to(torch.float16)

    # ----------------------- cast weights --------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections ---------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------- gates -----------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ----------------------- mask & gate fusion -------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ----------------- core O(N³·H) reduction via batched GEMM ----------
    # Reshape to (B*H, N, N) for matmul
    B, N, _, _ = left.shape
    left_mat  = left.reshape(B * hidden_dim, N, N)
    right_mat = right.reshape(B * hidden_dim, N, N)

    # Compute left @ right^T for each head
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.reshape(B, N, N, hidden_dim)

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------ cast back --------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
