"""
Outgoing TriMul (Triangle Multiplicative Update) forward pass.

Algorithm
---------
1. Layer‑norm over the input channel dimension (FP32).
2. Linear projections (left, right) and their gates (left_gate, right_gate,
   out_gate) using FP32 matrix multiplication.
3. Optional masking of left/right projections.
4. Apply sigmoid gates and element‑wise multiply projections with gates.
5. Core O(N³) contraction:
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d],
   implemented as a batch of independent GEMMs via torch.bmm.
6. Layer‑norm over the hidden dimension.
7. Fuse the element‑wise multiplication with the out_gate using a small
   Triton kernel.
8. Final linear projection back to the original dimension.
9. Return FP32 tensor of shape [B, N, N, dim].

The heavy element‑wise scaling with out_gate is executed in Triton to satisfy
the requirement of having at least one custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for element‑wise multiplication (out * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    """out[0:size] *= gate[0:size]  (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out * gate, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float), may be
                    ignored if config["nomask"] is True
            - weights: dict mapping weight names to torch.Tensor (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] float32

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Linear projections (FP32)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device)    # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(device)  # (hidden, dim)

    left = F.linear(x, left_proj_w)   # (B, N, N, hidden)
    right = F.linear(x, right_proj_w) # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_f = mask.to(device).to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device)    # (hidden, dim)
    right_gate_w = weights["right_gate.weight"].to(device)  # (hidden, dim)
    out_gate_w = weights["out_gate.weight"].to(device)      # (hidden, dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B,N,N,hidden)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batch GEMM over (B*hidden) matrices.
    # ------------------------------------------------------------------
    # Re‑order to (B, hidden, N, N) and flatten batch & hidden.
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    left_mat = left.view(B * hidden_dim, N, N)   # (B*hidden, N, N)
    right_mat = right.view(B * hidden_dim, N, N) # (B*hidden, N, N)

    # Matrix multiply left @ rightᵀ for each (B*hidden) slice
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)

    # Reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. to_out_norm (LayerNorm over hidden dimension)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 7. out_gate (linear + sigmoid) and element‑wise multiply via Triton
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # (B, N, N, hidden)

    # Flatten tensors for the Triton kernel
    out_flat = out.view(-1).contiguous()
    out_gate_flat = out_gate.view(-1).contiguous()

    total_elements = out_flat.numel()
    BLOCK = 16384  # reasonable block size for 32‑bit values
    grid = ( (total_elements + BLOCK - 1) // BLOCK, )

    elementwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        total_elements,
        BLOCK=BLOCK,
    )

    # Restore original shape after kernel execution
    out = out_flat.view_as(out)

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device)   # (dim, hidden)
    out = F.linear(out, to_out_w)                    # (B,N,N,dim)

    # Ensure output dtype is float32 as required by the interface
    return out.to(torch.float32)
