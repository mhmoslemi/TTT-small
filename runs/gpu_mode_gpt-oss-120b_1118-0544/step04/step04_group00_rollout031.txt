    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

"""
TriMul (outgoing) forward pass – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension `dim`.
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (left/right) and gate projections (left/right/out).
4. Fuse mask, projection and gate (if a mask is present) with a small Triton
   elementwise kernel.
5. Core O(N³) reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix multiplication using a custom Triton GEMM
   kernel (no Python loops, accumulator in fp32, final store in fp16).
6. Output LayerNorm over the hidden dimension, multiply by `out_gate`,
   final linear projection back to the original `dim`.
7. Cast back to the input dtype (usually float32) and return.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1. Elementwise kernel: proj * gate * mask (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,    # fp16 *
    gate_ptr,    # fp16 *
    mask_ptr,    # fp16 *
    out_ptr,     # fp16 *
    total_elems,  # int
    hidden_dim,   # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each mask element corresponds to hidden_dim projected elements
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(out_ptr + offs, proj * gate * mask, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Fuse projection, gate and (optional) mask using Triton.
    All tensors must be fp16 and contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)    (binary 0/1)
    Returns proj * gate * mask  (broadcast over H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# 2. Batched matrix multiplication kernel (left @ right^T)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,              # pointers
    batch, M, N, K,                    # dimensions
    stride_ah, stride_am, stride_ak,   # strides for A (BH, M, K)
    stride_bh, stride_bk, stride_bn,   # strides for B (BH, K, N) – note transposed view
    stride_ch, stride_cm, stride_cn,   # strides for C (BH, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * stride_ah
    b_base = b_ptr + batch_id * stride_bh

    # accumulator in fp32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (transposed view): shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Store result in fp16
    c_ptrs = c_ptr + batch_id * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 3. Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C]
            - mask         : torch.Tensor of shape [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict containing:
                * "dim" (int)         – input channel dimension C
                * "hidden_dim" (int)  – hidden dimension H
                * "nomask" (bool)     – if True, ignore mask
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive parts
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask and gating (fused via Triton if mask present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask broadcasted over hidden dimension
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³) reduction using batched GEMM Triton kernel
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    BH = B * H  # combined batch + head dimension

    # Rearrange to (BH, N, N) where the two leading axes are batch*head
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(BH, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(BH, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)  # fp16

    # Strides for A (left)
    stride_ah = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    # Strides for B (right) – transposed view via swapped K/N strides
    stride_bh = right_flat.stride(0)
    # right_flat is (BH, N, N) with strides (N*N, N, 1)
    stride_bk = right_flat.stride(2)   # = 1  (stride of K after transposition)
    stride_bn = right_flat.stride(1)   # = N  (stride of N after transposition)

    # Strides for C (output)
    stride_ch = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # Block sizes (tuned for H200 / Tensor cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N, N, N,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
