"""
custom_kernel
==============

Implements the outgoing Triangle Multiplicative Update (TriMul) forward pass
used in AlphaFold3‑style models.

Algorithm
---------
1. Layer‑normalize the input tensor (B, N, N, C) with learned weight/bias.
2. Cast to FP16 for the compute‑heavy part.
3. Compute four linear projections (left/right and their gates) with FP16
   weights; apply sigmoid to the gate outputs.
4. Fuse the element‑wise operations `proj * gate * mask` (if a mask is
   provided) in a single Triton kernel to avoid extra memory traffic.
5. Core O(N³·H) reduction:
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication:
   out = (left.permute(0,3,1,2) @ right.permute(0,3,2,1))
   yielding shape (B, N, N, H).
6. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate, and project back to the original channel dimension with a linear.
7. Cast the result back to the input dtype (typically FP32).

The mask‑gate fusion kernel is the only Triton component; the heavy O(N³)
reduction leverages cuBLAS batched GEMM via `torch.matmul`, which is far
faster than the naïve `torch.einsum` used in the reference implementation.
"""

    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # Index in the mask (broadcast over the hidden dimension)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using the Triton kernel above.
    All tensors must be contiguous, on the same CUDA device and of dtype torch.float16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for the compute‑heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # Gates (with sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # Fuse mask, projection and gate (if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·H) reduction via batched GEMM (cuBLAS)
    #   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #   = left_perm @ right_permᵀ where permutes bring H into batch
    # ------------------------------------------------------------------
    # left: (B, N, N, H) -> (B, H, N, N)
    # right: (B, N, N, H) -> (B, H, N, N) then transpose inner dims (K,J)
    out = torch.matmul(
        left.permute(0, 3, 1, 2),    # (B, H, N, N)
        right.permute(0, 3, 2, 1)    # (B, H, N, N) where inner dim is K
    )                               # (B, H, N, N)
    out = out.permute(0, 2, 3, 1)    # (B, N, N, H)

    # ------------------------------------------------------------------
    # Second LayerNorm and output gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
