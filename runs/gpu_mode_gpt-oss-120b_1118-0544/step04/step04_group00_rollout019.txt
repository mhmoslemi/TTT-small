    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)

"""
TriMul (outgoing) forward pass – Triton‑accelerated mask‑gate fusion.

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension `dim`.
2. Cast the normalized tensor to FP16 for the compute‑heavy part.
3. Linear projections (left/right) and gate projections (left/right/out) are
   performed with FP16 weights.
4. If a mask is present (and `nomask` is False) a Triton kernel fuses
   projection, gate and mask:  out = proj * gate * mask  (mask broadcast over
   the hidden dimension).
   Otherwise a simple elementwise multiplication is used.
5. Core update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication and is carried out with
   `torch.einsum` (equivalent to a batched GEMM over the hidden dimension).
6. Output LayerNorm over the hidden dimension, followed by elementwise
   multiplication with `out_gate` (performed by a small Triton kernel).
7. Final linear projection back to `dim`.
8. Cast the result back to the original input dtype.

The implementation uses Triton for the mask‑gate fusion (step 4) and for the
final gate multiplication (step 6); all other heavy work relies on highly
optimized PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# --------------------------------------------------------------
# Triton kernels
# --------------------------------------------------------------

@triton.jit
def _mask_gate_kernel(
    proj_ptr,       # fp16* (B*N*N*H,)
    gate_ptr,       # fp16* (B*N*N*H,)
    mask_ptr,       # fp16* (B*N*N,)
    out_ptr,        # fp16* (B*N*N*H,)
    total_elems,    # int: total number of elements in proj/gate
    hidden_dim,     # int: H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    mask_idx = offs // hidden_dim          # broadcast mask index

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    tl.store(out_ptr + offs, proj * gate * mask, mask=valid)


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor,
                     mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """Fuse projection, gate and mask (mask broadcast over hidden_dim)."""
    assert proj.shape == gate.shape
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)

    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


@triton.jit
def _mul_gate_kernel(
    tensor_ptr,   # fp16* (numel,)
    gate_ptr,     # fp16* (numel,)
    out_ptr,      # fp16* (numel,)
    total_elems,  # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    a = tl.load(tensor_ptr + offs, mask=valid, other=0.0)
    b = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=valid)


def _apply_out_gate(tensor: torch.Tensor, gate: torch.Tensor,
                    block_size: int = 1024) -> torch.Tensor:
    """Elementwise multiplication tensor *= gate (in‑place)."""
    assert tensor.shape == gate.shape
    total = tensor.numel()
    grid = (triton.cdiv(total, block_size),)

    _mul_gate_kernel[grid](
        tensor,
        gate,
        tensor,          # write back in‑place
        total,
        BLOCK_SIZE=block_size,
    )
    return tensor


# --------------------------------------------------------------
# Triton‑accelerated TriMul (outgoing) forward pass
# --------------------------------------------------------------

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim" (int)        – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    orig_dtype = input_tensor.dtype
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask and gating (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is half‑precision and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction (batched outer product over the sequence dimension)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with torch.einsum which internally uses highly‑optimized
    #    batched GEMM kernels.
    # ------------------------------------------------------------------
    out = torch.einsum("bikd,bjkd->bijd", left, right)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # Multiply by out_gate – fused via a tiny Triton kernel
    out = _apply_out_gate(out, out_gate)

    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
