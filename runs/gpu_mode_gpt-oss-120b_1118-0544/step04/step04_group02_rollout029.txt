import torch
import torch.nn.functional as F
import triton
import triton.language as tl
import math

@triton.jit
def gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    hidden_dim, total_elems,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # compute mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)

def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Fuse projection, gate sigmoid, and mask.
    proj, gate: (B, N, N, H) fp16
    mask: (B, N, N) fp16
    Returns tensor same shape as proj.
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)
    hidden_dim = proj.shape[-1]

    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        hidden_dim,
        total,
        BLOCK_SIZE=block_size,
    )
    return out

def custom_kernel(data):
    """
    TriMul (outgoing) forward pass – Triton‑accelerated core.

    Steps:
    1. LayerNorm input.
    2. FP16 cast for compute.
    3. Linear projections (left/right) and gate projections.
    4. Fuse mask and gate with a tiny Triton kernel (if mask present).
    5. Batched GEMM (left @ rightᵀ) using torch.bmm on the flattened (B*H, N, N) tensors.
    6. Output LayerNorm, output‑gate, final linear projection.
    7. Cast back to original dtype.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. Input LayerNorm
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # 2. Cast to FP16 for heavy compute
    x = x.to(torch.float16)

    # 3. Cast weights to FP16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # 4. Linear projections to hidden_dim
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # 5. Gate projections + sigmoid
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # 6. Fuse mask and gate (via Triton kernel) or elementwise
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 7. Core reduction: batched GEMM left @ rightᵀ per hidden channel
    B, N, _, H = left.shape   # (B, N, N, H)

    # reshape to (B, H, N, N) then flatten batch*H dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # batched matrix multiplication; right is transposed on the K axis
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 8. Output LayerNorm, output gate, final projection
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # 9. Cast back to original dtype (usually float32)
    out = out.to(input_tensor.dtype)
    return out
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the 4‑D input.
2. Cast to FP16 for compute‑heavy stages.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) with FP16 weights.
4. If a mask is present, fuse `proj * gate * mask` with a tiny Triton
   kernel (broadcasting the mask over the hidden dimension).
5. Core O(N³·H) reduction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply: for each (b,h) compute
   left[b,h] @ right[b,h]ᵀ.  We use torch.bmm (cuBLAS) on the
   flattened (B·H, N, N) tensors – this gives tensor‑core speed.
6. Output LayerNorm, multiply by the output gate and a final linear
   projection back to the original channel dimension.
7. Cast the result back to the input dtype (usually float32).

Only the mask‑gate fusion is performed in Triton; the heavy GEMM uses
highly‑optimized cuBLAS via torch.bmm.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    proj_ptr,       # *[total_elems]   fp16
    gate_ptr,       # *[total_elems]   fp16
    mask_ptr,       # *[mask_elems]    fp16 (contiguous)
    out_ptr,        # *[total_elems]   fp16
    hidden_dim,     # int, size of hidden dimension H
    total_elems,    # int, total number of elements in proj/gate
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # mask is broadcast over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Fuse projection, gate sigmoid and mask.
    proj, gate : (B, N, N, H)  fp16
    mask       : (B, N, N)     fp16
    Returns tensor of shape (B, N, N, H) fp16.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    # flatten mask for 1‑D indexing inside the kernel
    mask_flat = mask.contiguous().view(-1)

    hidden_dim = proj.shape[-1]

    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        hidden_dim,
        total,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    data = (input_tensor, mask, weights, config)
    Returns: Tensor [bs, seq_len, seq_len, dim] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive stages
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections to hidden_dim
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Fuse mask and gates (Triton) or simple elementwise multiplication
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction – batched GEMM via torch.bmm
    #    Compute out = left @ rightᵀ for each (batch, hidden) pair.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # (B, N, N, H)

    # Move hidden dimension to the batch axis and make tensors contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Flatten batch*hidden into a single batch dimension
    left_flat = left_perm.view(B * H, N, N)    # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
