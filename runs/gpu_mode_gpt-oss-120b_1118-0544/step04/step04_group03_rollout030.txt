@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid = tl.program_id(0)
    batch = tl.program_id(1)

    # tile indices
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers
    a_batch = a_ptr + batch * stride_batch_a
    b_batch = b_ptr + batch * stride_batch_b
    c_batch = c_ptr + batch * stride_batch_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)

    # store
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :]
    )

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor.
2. Fuse the two linear projections (proj & gate) for the left and right
   pathways by concatenating the weights and using a single F.linear call.
3. Apply the mask (if present) and the sigmoid gates.
4. Compute the core O(N³·H) reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication C = A @ Bᵀ where each batch
   corresponds to a (batch, hidden_dim) pair.  The multiplication is
   performed with a custom Triton kernel that works in FP16 with FP32
   accumulation.
5. Apply a hidden‑dim LayerNorm, an output gate, and a final linear
   projection back to the original channel dimension.
All heavy arithmetic is done in float16; the final result is cast back
to the dtype of the original input (typically float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM (A @ Bᵀ) for the TriMul core reduction
# --------------------------------------------------------------------------- #
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions (all = seq_len)
    stride_am, stride_ak,         # strides for A: (M, K)
    stride_bk, stride_bn,         # strides for B: (K, N) – stored as K×N
    stride_cm, stride_cn,         # strides for C: (M, N)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ  for a batch of matrices.
    A: (batch, M, K)      – fp16
    B: (batch, K, N)      – fp16 (interpreted as transposed)
    C: (batch, M, N)      – fp16 output (accumulated in fp32)
    """
    pid = tl.program_id(0)          # tile id over (M, N)
    batch = tl.program_id(1)        # batch index (B * hidden_dim)

    # --------------------------------------------------------------- #
    # Tile coordinates
    # --------------------------------------------------------------- #
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------- #
    # Base pointers for this batch
    # --------------------------------------------------------------- #
    a_batch = a_ptr + batch * stride_batch_a
    b_batch = b_ptr + batch * stride_batch_b
    c_batch = c_ptr + batch * stride_batch_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------- #
    # Loop over the K dimension
    # --------------------------------------------------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # A block: (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B block (as transposed): (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp16·fp16 → fp32 accumulation

    # --------------------------------------------------------------- #
    # Write out the result
    # --------------------------------------------------------------- #
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[:, None],
    )


# --------------------------------------------------------------------------- #
# Optional element‑wise multiplication kernel (used for output gating)
# --------------------------------------------------------------------------- #
@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch Triton element‑wise multiplication on two tensors of equal shape."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication."
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # --------------------------------------------------------------- #
    # 1) Unpack inputs
    # --------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device
    # --------------------------------------------------------------- #
    # 2) Input LayerNorm (float32 -> fp16 afterwards)
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------- #
    # 3) Fuse projection + gate linear layers
    # --------------------------------------------------------------- #
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)          # (2*H, C)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)          # (2*H, C)

    # Combined linear projections (produces proj and gate logits together)
    left_comb = F.linear(x, left_comb_w)    # [B, N, N, 2*H]
    right_comb = F.linear(x, right_comb_w)  # [B, N, N, 2*H]

    # Split into projection & gate components
    left_proj = left_comb[..., :hidden_dim]                # [B,N,N,H]
    left_gate = left_comb[..., hidden_dim:].sigmoid()      # [B,N,N,H]
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # --------------------------------------------------------------- #
    # 4) Apply optional mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------- #
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)    # [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------------------- #
    # 5) Apply gates
    # --------------------------------------------------------------- #
    left = left_proj * left_gate      # [B,N,N,H]
    right = right_proj * right_gate   # [B,N,N,H]

    # --------------------------------------------------------------- #
    # 6) Core O(N³·H) reduction via batched GEMM (Triton)
    # --------------------------------------------------------------- #
    B, N, _, H = left.shape

    # Rearrange to (B*H, N, N) layout for the kernel
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output placeholder
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Strides for the kernel (all tensors are contiguous)
    stride_a_batch = left_flat.stride(0)
    stride_am = left_flat.stride(1)      # M‑stride
    stride_ak = left_flat.stride(2)      # K‑stride

    stride_b_batch = right_flat.stride(0)
    stride_bk = right_flat.stride(1)     # K‑stride (as rows)
    stride_bn = right_flat.stride(2)     # N‑stride (as columns)

    stride_c_batch = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # Tile sizes – chosen to map well to tensor cores on the B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # 32 is a multiple of the WMMA K dimension

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # more warps → better occupancy on large tiles
        num_stages=3,         # pipeline depth
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------- #
    # 7) Hidden‑dim LayerNorm
    # --------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------- #
    # 8) Output gate (element‑wise multiplication, Triton kernel)
    # --------------------------------------------------------------- #
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()   # [B,N,N,H]
    out = _elemwise_mul(out, out_gate)

    # --------------------------------------------------------------- #
    # 9) Final linear projection back to original channel dimension
    # --------------------------------------------------------------- #
    out = F.linear(out, weights["to_out.weight"])

    # --------------------------------------------------------------- #
    # 10) Cast to the original input dtype and return
    # --------------------------------------------------------------- #
    out = out.to(x.dtype)      # x is still float16; original input dtype is preserved in `data[0]`
    out = out.to(data[0].dtype)  # ensure we match the dtype of the original tensor

    return out
