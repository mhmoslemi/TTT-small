"""
TriMul (outgoing) forward pass – Triton‑accelerated mask & gate fusion.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension `dim`.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections:
      left  = x @ W_left   (shape B×N×N×H)
      right = x @ W_right
   Gate projections (sigmoid):
      left_gate  = σ(x @ W_left_gate)
      right_gate = σ(x @ W_right_gate)
      out_gate   = σ(x @ W_out_gate)          # shape B×N×N×H
4. Fuse mask, gate and projection in a single Triton kernel
   (applied only when a mask is present).  The kernel computes
        proj * gate * mask   (mask broadcast over the hidden dimension H).
   When no mask is provided we fall back to a simple element‑wise multiply.
5. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented with a batched GEMM (torch.bmm) which invokes cuBLAS/Tensor‑cores.
6. Output LayerNorm over the hidden dimension H.
7. Multiply by `out_gate`, apply the final linear projection back to `dim`.
8. Cast the result back to the original dtype (usually FP32).

Only the mask‑gate fusion is performed in Triton; the rest relies on
high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16 * (total_elems = B*N*N*H)
    gate_ptr,          # fp16 * (total_elems)
    mask_ptr,          # fp16 * (B*N*N)
    out_ptr,           # fp16 * (total_elems)
    total_elems,       # int64 scalar
    hidden_dim,        # int64 scalar (H)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard for the last (partial) block
    mask_valid = offs < total_elems

    # The index within the mask (broadcast over H)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask_valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask_valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=mask_valid)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor, block_size: int = 8192) -> torch.Tensor:
    """
    Fuse projection, gate and optional binary mask using Triton.

    Args:
        proj: (B, N, N, H) fp16 tensor.
        gate: (B, N, N, H) fp16 tensor.
        mask: (B, N, N)   fp16 tensor (0/1).
        block_size: number of elements processed per program.

    Returns:
        Tensor of shape (B, N, N, H) fp16 = proj * gate * mask.
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights      : dict of model parameters (float32 tensors)
                              Expected keys:
                              ["norm.weight", "norm.bias",
                               "left_proj.weight", "right_proj.weight",
                               "left_gate.weight", "right_gate.weight", "out_gate.weight",
                               "to_out_norm.weight", "to_out_norm.bias",
                               "to_out.weight"]
            - config       : dict with entries:
                              "dim"         (int)    – input channel dimension C
                              "hidden_dim" (int)    – hidden dimension H
                              "nomask"      (bool)   – if True, ignore mask (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ---------------------------------------------------------
    # Unpack arguments
    # ---------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ---------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # ---------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim), dtype = input dtype

    # ---------------------------------------------------------
    # 2. Cast to FP16 for heavy compute
    # ---------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ---------------------------------------------------------
    # 3. Cast model weights to FP16 (no bias in Linear layers)
    # ---------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ---------------------------------------------------------
    # 4. Linear projections (B, N, N, hidden_dim)
    # ---------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x_fp16, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x_fp16, right_proj_w)

    # ---------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ---------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()

    # ---------------------------------------------------------
    # 6. Apply mask and gating
    # ---------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous and has the right shape
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        # No mask – plain elementwise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ---------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ---------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Matrix multiplication (B*H, N, N) @ (B*H, N, N)^T → (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ---------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ---------------------------------------------------------
    # 9. Cast back to the original dtype (usually float32)
    # ---------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
