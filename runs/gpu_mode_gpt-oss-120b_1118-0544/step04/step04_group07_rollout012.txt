    pid_n = tl.program_id(2)

    # compute block start indices

    mask_n = offs_n < N

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    ...

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

    # Cast to fp16
    x = x.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"].to(device).to(torch.float16))
    right = torch.nn.functional.linear(x, weights["right_proj.weight"].to(device).to(torch.float16))

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"].to(device).to(torch.float16)))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"].to(device).to(torch.float16)))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"].to(device).to(torch.float16)))

    # Apply gates and mask
    left = left * left_gate
    right = right * right_gate
    if mask is not None and not config.get("nomask", False):
        mask = mask.to(device).to(torch.float16).unsqueeze(-1) # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # Core multiplication via Triton
    # reshape to (batch_h, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, N2 = left_perm.shape
    assert N == N2

    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # compute strides
    stride_a_batch, stride_a_m, stride_a_n = left_flat.stride()
    stride_b_batch, stride_b_m, stride_b_n = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # In our kernel we treat A as (batch, M, K) = (batch, N, N)
    # B as (batch, K, N) = transposed; So stride_bk = stride_b_n; stride_bj = stride_b_m
    stride_ah = stride_a_m
    stride_ak = stride_a_n
    stride_abatch = stride_a_batch

    stride_bj = stride_b_m  # original row stride
    stride_bbatch = stride_b_batch

    stride_ch = stride_c_m
    stride_cj = stride_c_n
    stride_cbatch = stride_c_batch

    BLOCK_K = 64

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The forward pass follows the reference PyTorch implementation:
1. Layer‑norm on the input.
2. Linear projections (left/right) and three gates (left, right, out) in fp16.
3. Fuse mask (if present) and gate multiplications.
4. Core O(N³·H) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   performed by a custom batched GEMM Triton kernel that processes the
   hidden dimension as an extra batch axis (B·H independent matmuls).
5. Output LayerNorm, out‑gate multiplication and final linear projection.
All heavy arithmetic is done in float16 with float32 accumulators; the
result is cast back to the input dtype (float32 by default).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched matrix multiplication A @ Bᵀ
# A : (batch, M, K)   (fp16)
# B : (batch, N, K)   (fp16)  – we treat B as transposed inside the kernel
# C : (batch, M, N)   (fp16)  output
# --------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A, B, C,
    stride_ah, stride_ak, stride_abatch,   # A strides: row, col, batch
    stride_bk, stride_bj, stride_bbatch,   # B strides (transposed view)
    stride_ch, stride_cj, stride_cbatch,   # C strides
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = A + batch_id * stride_abatch \
                 + offs_m[:, None] * stride_ah \
                 + offs_k[None, :] * stride_ak
        mask_a = mask_m[:, None] & (offs_k[None, :] < K)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # Load B tile (K x N) – note the transposed view
        b_ptrs = B + batch_id * stride_bbatch \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bj
        mask_b = (offs_k[:, None] < K) & mask_n[None, :]
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b).to(tl.float32)

    # Write back FP16 result
    c = acc.to(tl.float16)
    c_ptrs = C + batch_id * stride_cbatch \
             + offs_m[:, None] * stride_ch \
             + offs_n[None, :] * stride_cj
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, c, mask=mask_c)


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, dim)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape (B, N, N, dim) (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"].to(device),
        bias=weights["norm.bias"].to(device),
    )
    # Cast to fp16 for the compute‑heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections and gates (all fp16)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3. Apply gates and optional mask (fused element‑wise)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    if mask is not None and not nomask:
        # mask shape -> (B, N, N, 1) for broadcasting over hidden dim
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4. Core O(N³·H) contraction via Triton batched GEMM
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and flatten the first two dims.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    left_flat = left_perm.view(batch_total, N, N)
    right_flat = right_perm.view(batch_total, N, N)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Strides for the flattened tensors
    stride_a_batch, stride_a_m, stride_a_n = left_flat.stride()
    stride_b_batch, stride_b_m, stride_b_n = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # For Bᵀ we swap the row/col strides (k is original col, j is original row)
    stride_ah = stride_a_m
    stride_ak = stride_a_n
    stride_abatch = stride_a_batch

    stride_bk = stride_b_n          # k‑dimension stride (original column)
    stride_bj = stride_b_m          # j‑dimension stride (original row)
    stride_bbatch = stride_b_batch

    stride_ch = stride_c_m
    stride_cj = stride_c_n
    stride_cbatch = stride_c_batch

    # Tiling parameters (tuned for B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_total,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        stride_ah, stride_ak, stride_abatch,
        stride_bk, stride_bj, stride_bbatch,
        stride_ch, stride_cj, stride_cbatch,
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
