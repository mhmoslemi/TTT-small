@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr,
                          M, N, K,
                          stride_am, stride_ak, stride_ab,
                          stride_bk, stride_bn, stride_bb,
                          stride_cm, stride_cn, stride_cb,
                          BLOCK_M: tl.constexpr,
                          BLOCK_N: tl.constexpr,
                          BLOCK_K: tl.constexpr):

   pid = tl.program_id(0) # tile id
   batch = tl.program_id(1) # batch id

   # compute tile coordinates
   num_pid_n = tl.cdiv(N, BLOCK_N)
   pid_m = pid // num_pid_n
   pid_n = pid % num_pid_n

   # offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   offs_k = tl.arange(0, BLOCK_K)

   # boundary masks
   mask_m = offs_m < M
   mask_n = offs_n < N

   # base pointers
   a_batch_ptr = a_ptr + batch * stride_ab
   b_batch_ptr = b_ptr + batch * stride_bb
   c_batch_ptr = c_ptr + batch * stride_cb

   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for k in range(0, tl.cdiv(K, BLOCK_K)):
       cur_k = k * BLOCK_K + offs_k
       mask_k = cur_k < K

       a = tl.load(a_batch_ptr + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
                   mask=mask_m[:, None] & mask_k[None, :], other=0.0)  # shape (BLOCK_M, BLOCK_K)
       b = tl.load(b_batch_ptr + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
                   mask=mask_k[:, None] & mask_n[None, :], other=0.0)  # shape (BLOCK_K, BLOCK_N)

       acc += tl.dot(a, b)  # fp16 * fp16 => fp32

   c = acc.to(tl.float16)
   tl.store(c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
            c,
            mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul outgoing update (AlphaFold3) – Triton accelerated core reduction.

Algorithm:
1. Layer‑norm the input over the channel dimension.
2. Fuse linear projection + gating for the left and right pathways:
   * Concatenate the projection and gate weight matrices (no bias) and apply a
     single F.linear.  Split the result, apply sigmoid to the gate part and
     mask the projection (if a mask is provided).
   * Multiply projection × gate → left / right tensors of shape [B,N,N,H].
3. Compute the batched matrix multiplication:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is equivalent to a batched GEMM left @ rightᵀ for each hidden slice.
   We reshape (B,N,N,H) → (B·H, N, N) and launch a Triton kernel that handles
   the batch dimension via `program_id(1)`.  All arithmetic is fp16 with
   fp32 accumulation.
4. Layer‑norm over the hidden dimension, multiply by the output gate
   (computed from the normalized input), and project back to the original
   channel dimension.
All heavy work (step 3) runs on the GPU via Triton; the remaining
operations use torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to A, B, C
    M, N, K,                           # matrix dims (M×K)·(K×N) → (M×N)
    stride_am, stride_ak, stride_ab,   # strides for A (M, K, batch)
    stride_bk, stride_bn, stride_bb,   # strides for B (K, N, batch)
    stride_cm, stride_cn, stride_cb,   # strides for C (M, N, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A: (M,K), B: (K,N))."""
    pid = tl.program_id(0)      # tile index within a batch
    batch = tl.program_id(1)    # batch index (B·H)

    # tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # (BLOCK_M, BLOCK_K)
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)      # fp16·fp16 → fp32

    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    B, N, _, C = x.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    orig_dtype = x.dtype
    device = x.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm over channel dimension
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(dtype_fp16)  # work in half precision from now on

    # ------------------------------------------------------------------- #
    # 2. Fuse projection + gate for left and right pathways
    # ------------------------------------------------------------------- #
    # combine weights: (proj, gate) → shape (2*H, C)
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype_fp16)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype_fp16)

    # linear that outputs [B, N, N, 2*H]
    left_comb = F.linear(x, left_comb_w)    # (proj || gate)
    right_comb = F.linear(x, right_comb_w)

    # split
    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # ------------------------------------------------------------------- #
    # 3. Optional mask (applied only to the raw projections)
    # ------------------------------------------------------------------- #
    if mask is not None and not nomask:
        mask_f = mask.to(dtype_fp16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # apply gates
    left = left_proj * left_gate   # [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------- #
    # 4. Output gate (computed from the normalized input)
    # ------------------------------------------------------------------- #
    out_gate = F.linear(x, weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)).sigmoid()  # [B,N,N,H]

    # ------------------------------------------------------------------- #
    # 5. Core O(N^3·H) reduction via batched GEMM (Triton)
    # ------------------------------------------------------------------- #
    # reshape for batched GEMM: treat hidden dim as additional batch dimension
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (B*H, M=N, K=N)
    right_t = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)   # (B*H, K=N, N)

    out_flat = torch.empty_like(left_flat)  # will hold (B*H, N, N) in fp16

    # strides for kernel launch
    stride_a_batch = left_flat.stride(0)     # N*N
    stride_am = left_flat.stride(1)          # N
    stride_ak = left_flat.stride(2)          # 1

    stride_b_batch = right_t.stride(0)       # N*N
    stride_bk = right_t.stride(1)            # N
    stride_bn = right_t.stride(2)            # 1

    stride_c_batch = out_flat.stride(0)      # N*N
    stride_cm = out_flat.stride(1)           # N
    stride_cn = out_flat.stride(2)           # 1

    M = N
    K = N
    # tile sizes – tuned for typical N ≤ 1024 on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * hidden_dim)

    _batched_gemm_kernel[grid](
        left_flat,
        right_t,
        out_flat,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_a_batch,
        stride_bk,
        stride_bn,
        stride_b_batch,
        stride_cm,
        stride_cn,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 6. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 7. Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8. Final projection back to original channel dimension
    # ------------------------------------------------------------------- #
    out = F.linear(
        out,
        weights["to_out.weight"].to(device=device, dtype=dtype_fp16),
    )

    # Cast to the original dtype of the input tensor
    out = out.to(orig_dtype)
    return out
