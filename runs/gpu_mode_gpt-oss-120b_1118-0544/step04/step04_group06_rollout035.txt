@triton.jit
def triu_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # tile indices
    i_tile = tl.program_id(0)
    j_tile = tl.program_id(1)
    bh = tl.program_id(2)

    # compute batch and hidden index
    b = bh // H
    d = bh % H

    # base offsets
    i_offset = i_tile * BLOCK_M
    j_offset = j_tile * BLOCK_N

    # create offsets
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask bounds for i, j, k
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # pointers
    left_ptrs = left_ptr + b * stride_left_batch + d * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
    right_ptrs = right_ptr + b * stride_right_batch + d * stride_right_h + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_idx in range(num_k_tiles):
        # load left block
        cur_k = k_idx * BLOCK_K
        # compute actual k offsets for this tile
        cur_offs_k = cur_k + offs_k
        # recompute masks for this tile (cur_k + offs_k < N)
        mask_k_tile = cur_offs_k < N
        # offsets pointer for left
        left_tile_ptrs = left_ptr + b * stride_left_batch + d * stride_left_h + offs_i[:, None] * stride_left_i + cur_offs_k[None, :] * stride_left_k
        right_tile_ptrs = right_ptr + b * stride_right_batch + d * stride_right_h + cur_offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j

        left_tile = tl.load(left_tile_ptrs, mask=mask_i[:, None] & mask_k_tile[None, :], other=0.0).to(tl.float32)
        right_tile = tl.load(right_tile_ptrs, mask=mask_k_tile[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(left_tile, right_tile)

    # store result
    out_ptrs = out_ptr + b * stride_out_batch + d * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    # Convert accumulate to fp16

def tri_mul_forward_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    i_tile = tl.program_id(0)
    j_tile = tl.program_id(1)
    bh = tl.program_id(2)

    b = bh // H
    d = bh % H

    # base offsets
    i_base = i_tile * BLOCK_M
    j_base = j_tile * BLOCK_N

    # offsets
    i_offs = i_base + tl.arange(0, BLOCK_M)
    j_offs = j_base + tl.arange(0, BLOCK_N)
    mask_i = i_offs < N
    mask_j = j_offs < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of tiles in K
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_base = k_iter * BLOCK_K
        k_offs = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_offs < N

        # pointers for left and right blocks
        left_ptrs = left_ptr + b * stride_left_batch + d * stride_left_h + i_offs[:, None] * stride_left_i + k_offs[None, :] * stride_left_k
        right_ptrs = right_ptr + b * stride_right_batch + d * stride_right_h + k_offs[:, None] * stride_right_k + j_offs[None, :] * stride_right_j

        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # cast to fp32 for accumulation
        left_f32 = left.to(tl.float32)
        right_f32 = right.to(tl.float32)
        acc += tl.dot(left_f32, right_f32)

    # write back
    out_ptrs = out_ptr + b * stride_out_batch + d * stride_out_h + i_offs[:, None] * stride_out_i + j_offs[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel for the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Apply LayerNorm on the input tensor (shape [B,N,N,C]).
2. Cast to FP16 and perform three linear projections:
   - left_proj, right_proj      : (B,N,N,H)
   - left_gate, right_gate, out_gate (sigmoid) : (B,N,N,H)
3. Fuse element‑wise multiplication of projection, gate and optional mask
   with a small Triton kernel.
4. Core O(N³·H) reduction:
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   implemented as a tiled batched GEMM (left @ rightᵀ) in Triton.
5. Apply second LayerNorm over the hidden dimension, multiply by out_gate
   and project back to the original channel dimension.
6. Cast the result back to the original input dtype.

Only the mask‑gate fusion and the large reduction are performed in Triton;
all other ops use PyTorch (highly optimized cuBLAS for the linear layers).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask (mask broadcast over hidden)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i64 – total number of elements in proj/gate
    hidden_dim,        # i64 – H (mask repeats this many times)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # bounds
    mask = offs < total_elems

    # mask index (each mask element repeats hidden_dim times)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * mval, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse proj * gate * mask (mask broadcast over hidden_dim) using Triton.
    All tensors must be FP16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³·H) reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_forward_kernel(
    left_ptr,       # fp16 * (B*N*N*H,)
    right_ptr,      # fp16 * (B*N*N*H,)
    out_ptr,        # fp16 * (B*N*N*H,)
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids: (i_tile, j_tile, bh) where bh = b*H + d
    i_tile = tl.program_id(0)
    j_tile = tl.program_id(1)
    bh = tl.program_id(2)

    b = bh // H            # batch index
    d = bh % H             # hidden index

    # base offsets for this tile
    i_base = i_tile * BLOCK_M
    j_base = j_tile * BLOCK_N

    # offsets within the tile
    i_offs = i_base + tl.arange(0, BLOCK_M)
    j_offs = j_base + tl.arange(0, BLOCK_N)

    mask_i = i_offs < N
    mask_j = j_offs < N

    # accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k_tiles):
        k_base = k_tile * BLOCK_K
        k_offs = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_offs < N

        # left[ b, i, k, d ]
        left_ptrs = left_ptr + b * stride_left_batch \
                    + d * stride_left_h \
                    + i_offs[:, None] * stride_left_i \
                    + k_offs[None, :] * stride_left_k
        # right[ b, j, k, d ] -> we need the transpose (k, j) for dot
        right_ptrs = right_ptr + b * stride_right_batch \
                     + d * stride_right_h \
                     + k_offs[:, None] * stride_right_k \
                     + j_offs[None, :] * stride_right_j

        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # write back result (cast back to fp16)
    out_ptrs = out_ptr + b * stride_out_batch \
               + d * stride_out_h \
               + i_offs[:, None] * stride_out_i \
               + j_offs[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_reduce(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    Both inputs are FP16 tensors of shape [B, N, N, H].
    Returns a FP16 tensor of the same shape.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # FP16, same layout

    # strides
    lsb, lsi, lsk, lsh = left.stride()
    rsb, rsj, rsk, rsh = right.stride()
    osb, osi, osj, osh = out.stride()

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j tiles
        B * H,                          # batch*hidden
    )

    _tri_mul_forward_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        lsb,
        lsi,
        lsk,
        lsh,
        rsb,
        rsj,
        rsk,
        rsh,
        osb,
        osi,
        osj,
        osh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor: [B, N, N, C] (float32 or float16)
            mask: [B, N, N] (bool/float) or None
            weights: dict of model parameters (float32)
            config: {"dim": C, "hidden_dim": H, "nomask": bool (optional)}
    Returns:
        Tensor of shape [B, N, N, C] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float16)
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # --------------------------------------------------------------
    # 2. Cast to FP16 for all heavy compute
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Cast all linear weights to FP16
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w  = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (B,N,N,C) -> (B,N,N,H)
    # --------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 6. Apply mask + gates (fused via Triton when mask is present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7. Core reduction: out = left @ rightᵀ  (batched GEMM over hidden dim)
    # --------------------------------------------------------------
    out = _tri_mul_reduce(left, right)   # shape (B,N,N,H), FP16

    # --------------------------------------------------------------
    # 8. Output LayerNorm, output gate, final projection back to C
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    out = out * out_gate
    out = F.linear(out, to_out_w)    # (B,N,N,dim)

    # --------------------------------------------------------------
    # 9. Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(orig_dtype)
    return out
