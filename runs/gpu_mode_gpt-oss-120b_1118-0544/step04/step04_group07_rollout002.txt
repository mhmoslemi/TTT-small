"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input (dim‑dimensional) tensor.
2. Linear projections → left/right (hidden_dim) and three gating projections.
3. Fuse mask (if present) and gate multiplications for left/right with a tiny
   Triton kernel (element‑wise, no synchronization).
4. Compute the O(N³·H) reduction as a batched GEMM:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a batched matrix multiplication
        left_mat @ right_matᵀ
   with shape (B·H, N, N).
5. Layer‑norm the hidden result, apply the output gate, and finally a linear
   projection back to the original channel dimension.
6. Cast the result back to the original dtype.

Only the mask‑and‑gate fusion is performed in Triton; the heavy GEMM uses
torch.bmm (cuBLAS) which already exploits Tensor Cores on the target B200.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr,            # [B,N,N,H] flattened pointer (fp16)
    right_ptr,           # same shape
    left_gate_ptr,       # same shape
    right_gate_ptr,      # same shape
    mask_ptr,            # [B,N,N] flattened pointer (fp16)
    total_elems,         # total number of elements in left/right (int)
    hidden_dim,          # H (int)
    HAS_MASK: tl.constexpr,   # compile‑time bool
    BLOCK_SIZE: tl.constexpr, # compile‑time int
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    If HAS_MASK=False, the mask load/ multiplication is omitted.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # Each hidden_dim entries share the same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)   (float32/float16)
            - mask: torch.Tensor of shape (B, N, N) or None.  If present and
                    config["nomask"] is False, it will be applied.
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int)          – input channel dimension C
                * "hidden_dim" (int)   – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) in the same dtype as input_tensor.
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    ##################################################################
    # 1. Input layer‑norm
    ##################################################################
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Work in float16 for the heavy part
    x = x.to(torch.float16)

    ##################################################################
    # 2. Linear projections and gating
    ##################################################################
    # Linear weights (float16)
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # Projections
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # Gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    ##################################################################
    # 3. Fuse mask and gate multiplication (Triton)
    ##################################################################
    use_mask = (mask is not None) and (not nomask)
    if use_mask:
        # Convert mask to half and broadcast over hidden_dim
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)

        # Flatten tensors for the kernel
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.squeeze(-1).contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024  # Tunable; must be multiple of 32 for best occupancy
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask → simple element‑wise multiplication with gates
        left = left * left_gate
        right = right * right_gate

    ##################################################################
    # 4. Core O(N³·H) reduction via batched GEMM (cuBLAS)
    ##################################################################
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matmul: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    ##################################################################
    # 5. Output layer‑norm, gate, and final projection
    ##################################################################
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the input dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
