@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offset < size
    a = tl.load(a_ptr + offset, mask=mask)
    b = tl.load(b_ptr + offset, mask=mask)
    tl.store(out_ptr + offset, a * b, mask=mask)

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to half for heavy ops
    dtype = torch.float16
    x_half = x.to(dtype)

    device = x_half.device

    # cast weights to half
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=dtype)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype)

    # linear projections
    # x_half shape [B,N,N,dim] -> flatten last dims? F.linear expects (batch..., dim_in). It works on any shape: treat last dimension as in_features.
    left_proj = torch.nn.functional.linear(x_half, left_proj_w)  # shape [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x_half, right_proj_w)

    left_gate = torch.nn.functional.linear(x_half, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_half, right_gate_w).sigmoid()

    # mask
    if mask is not None and not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    left = left_proj * left_gate
    right = right_proj * right_gate

    # reshape for batched gemm
    B, N, _, H = left.shape
    # permute to B,H,N,N
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # flatten batch
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)
    # compute batched GEMM
    # out_flat shape (B*H, N, N)
    # Use torch.bmm
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # hidden dim layer norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # out gate
    out_gate = torch.nn.functional.linear(x_half, out_gate_w).sigmoid()
    # elementwise multiply using triton
    out = _elemwise_mul(out, out_gate)

    # final projection
    out = torch.nn.functional.linear(out, to_out_w)

    # cast to original dtype (maybe float32)
    out = out.to(x.dtype)

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The kernel follows the reference PyTorch TriMul implementation but
optimizes the O(N³·H) reduction by reshaping the tensors so that the
core operation becomes a batched matrix multiplication:
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to (left ⟂ rightᵀ) for each hidden channel d.
We use torch.bmm (cuBLAS) for this massive GEMM and a tiny Triton
kernel for the final element‑wise multiplication (output‑gate).  All
heavy arithmetic is performed in float16, with layer‑norms and final
projection cast back to the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Tiny Triton kernel for element‑wise multiplication (out * out_gate)
# --------------------------------------------------------------------------- #
@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton element‑wise multiplication kernel."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication."
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Work in float16 for the heavy path
    dtype = torch.float16
    x_h = x.to(dtype)

    device = x_h.device

    # ------------------------------------------------------------------- #
    # 2. Cast model weights to half precision
    # ------------------------------------------------------------------- #
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=dtype)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype)

    # ------------------------------------------------------------------- #
    # 3. Linear projections + gates
    # ------------------------------------------------------------------- #
    left_proj = F.linear(x_h, left_proj_w)          # [B, N, N, H]
    right_proj = F.linear(x_h, right_proj_w)        # [B, N, N, H]

    left_gate = F.linear(x_h, left_gate_w).sigmoid()
    right_gate = F.linear(x_h, right_gate_w).sigmoid()

    # Optional mask (broadcast over the hidden dimension)
    if mask is not None and not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)      # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Apply gates
    left = left_proj * left_gate                    # [B, N, N, H]
    right = right_proj * right_gate                 # [B, N, N, H]

    # ------------------------------------------------------------------- #
    # 4. Core O(N³·H) reduction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape
    # (B, H, N, N) layout for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Collapse batch & hidden dimension
    left_flat = left_perm.view(B * H, N, N)              # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)            # (B*H, N, N)

    # Batched matrix multiplication: (left) @ (right)ᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # ------------------------------------------------------------------- #
    # 5. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------- #
    # 6. Output gate (element‑wise) – Triton kernel
    # ------------------------------------------------------------------- #
    out_gate = F.linear(x_h, out_gate_w).sigmoid()   # [B, N, N, H]
    out = _elemwise_mul(out, out_gate)               # fused multiplication

    # ------------------------------------------------------------------- #
    # 7. Final projection back to original channel dimension
    # ------------------------------------------------------------------- #
    out = F.linear(out, to_out_w)                     # [B, N, N, C]

    # Cast back to the original dtype of the input tensor
    out = out.to(x.dtype)

    return out
