def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor,
                                      (dim,),
                                      weight=weights["norm.weight"],
                                      bias=weights["norm.bias"])
    # cast to fp16
    x = x.to(torch.float16)

    # Prepare weight tensors in fp16 and transposed for projection and gate
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16).t().contiguous()
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16).t().contiguous()
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16).t().contiguous()
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16).t().contiguous()
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16).t().contiguous()
    # bias not needed as linear has no bias.

    B, N, _, C = x.shape

    # allocate outputs
    left = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
    right = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # mask handling
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
    else:
        mask_fp16 = None

    # compute left and right using kernel
    _proj_gate_mask(x, left_proj_w, left_gate_w, mask_fp16, left)
    _proj_gate_mask(x, right_proj_w, right_gate_w, mask_fp16, right)

    # compute out_gate using torch linear + sigmoid
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # core reduction using batched GEMM
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # apply out_gate
    out = out * out_gate

    # output layernorm (hidden_dim)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(out,
                                         (hidden_dim,),
                                         weight=to_out_norm_w,
                                         bias=to_out_norm_b)

    # final linear to dim
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)

    # cast back
    out = out.to(dtype)

    return out
```

    x_ptr, proj_w_ptr, gate_w_ptr, mask_ptr, out_ptr,
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_pw0, stride_pw1,
    stride_gw0, stride_gw1,
    stride_maskb, stride_maski, stride_maskj,

def _proj_gate_mask_kernel(
    x_ptr, proj_w_ptr, gate_w_ptr, mask_ptr, out_ptr,
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_pw0, stride_pw1,
    stride_gw0, stride_gw1,
    stride_maskb, stride_maski, stride_maskj,
    stride_outb, stride_outi, stride_outj, stride_outh,
    B, N, C, H,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    APPLY_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    rows = row_start + tl.arange(0, BLOCK_M)
    mask = rows < (B * N * N)

    # indices
    b = rows // (N * N)
    rem = rows % (N * N)
    i = rem // N
    j = rem % N

    # offsets base for x and out
    row_offset = b * stride_xb + i * stride_xi + j * stride_xj
    out_row_offset = b * stride_outb + i * stride_outi + j * stride_outj

    # accumulator per output tile (BLOCK_M, H)
    acc_proj = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, H), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(C, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_K = tl.minimum(BLOCK_K, C - k_offset)

        # offsets for x tile
        k_range = tl.arange(0, BLOCK_K)
        # mask for valid k load
        k_valid = (k_offset + k_range) < C

        x_ptrs = x_ptr + row_offset[:, None] + (k_offset + k_range)[None, :] * stride_xc
        x = tl.load(x_ptrs, mask=mask[:, None] & k_valid[None, :], other=0.0)  # shape (M, BLOCK_K)

        w_proj_ptrs = proj_w_ptr + (k_offset + k_range)[:, None] * stride_pw0 + tl.arange(0, H)[None, :] * stride_pw1

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Cast to fp16 for compute‑intensive ops.
3. Fuse linear projection, gating (sigmoid) and optional mask into a single
   Triton kernel, producing the left and right tensors of shape
   (B, N, N, H) where H = hidden_dim.
4. Compute the out‑gate tensor with a regular linear + sigmoid (small
   relative cost).
5. Perform the core O(N³·H) reduction as a batched GEMM:
      out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   using torch.bmm on the (B*H, N, N) view.
6. Apply the out‑gate, a second LayerNorm, and the final linear projection.
7. Cast back to the original dtype.

Only the projection‑gate‑mask fusion is accelerated with Triton; the
reduction uses cuBLAS which is already optimal on the target GPU.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused projection + gate + optional mask
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    x_ptr, proj_w_ptr, gate_w_ptr, mask_ptr, out_ptr,          # pointers
    stride_xb, stride_xi, stride_xj, stride_xc,                # x strides
    stride_pw0, stride_pw1,                                    # proj weight strides (C, H)
    stride_gw0, stride_gw1,                                    # gate weight strides (C, H)
    stride_maskb, stride_maski, stride_maskj,                  # mask strides (B, N, N)
    stride_outb, stride_outi, stride_outj, stride_outh,        # out strides (B, N, N, H)
    B, N, C, H,                                                # dimensions
    BLOCK_M: tl.constexpr,                                     # rows processed per program
    BLOCK_K: tl.constexpr,                                     # inner‑K tile size
    APPLY_MASK: tl.constexpr,                                  # compile‑time flag
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    rows = row_start + tl.arange(0, BLOCK_M)                # (M,)
    row_mask = rows < (B * N * N)

    # decode flat row index -> (b, i, j)
    b = rows // (N * N)
    rem = rows % (N * N)
    i = rem // N
    j = rem % N

    # base offsets for input & output
    x_base = b * stride_xb + i * stride_xi + j * stride_xj
    out_base = b * stride_outb + i * stride_outi + j * stride_outj

    # accumulators (float32 for precision)
    acc_proj = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, H), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(C, BLOCK_K)):
        k_off = k * BLOCK_K
        cur_k = tl.minimum(BLOCK_K, C - k_off)                 # actual K for last tile
        k_idx = tl.arange(0, BLOCK_K)

        # mask for valid K elements
        k_valid = (k_off + k_idx) < C

        # ----- load X tile -------------------------------------------------
        x_ptrs = x_ptr + x_base[:, None] + (k_off + k_idx)[None, :] * stride_xc
        x_tile = tl.load(x_ptrs,
                         mask=row_mask[:, None] & k_valid[None, :],
                         other=0.0)                               # (M, K)

        # ----- load projection weight tile ----------------------------------
        proj_w_ptrs = proj_w_ptr + (k_off + k_idx)[:, None] * stride_pw0 + tl.arange(0, H)[None, :] * stride_pw1
        proj_w_tile = tl.load(proj_w_ptrs,
                              mask=k_valid[:, None],
                              other=0.0)                         # (K, H)

        # ----- load gate weight tile ----------------------------------------
        gate_w_ptrs = gate_w_ptr + (k_off + k_idx)[:, None] * stride_gw0 + tl.arange(0, H)[None, :] * stride_gw1
        gate_w_tile = tl.load(gate_w_ptrs,
                              mask=k_valid[:, None],
                              other=0.0)                         # (K, H)

        # ----- accumulate dot products ---------------------------------------
        acc_proj += tl.dot(x_tile, proj_w_tile)                 # (M, H)
        acc_gate += tl.dot(x_tile, gate_w_tile)                 # (M, H)

    # ----- sigmoid on gate accumulator -------------------------------------
    gate_sig = 1.0 / (1.0 + tl.exp(-acc_gate))

    # ----- optional mask ----------------------------------------------------
    if APPLY_MASK:
        mask_ptrs = mask_ptr + b * stride_maskb + i * stride_maski + j * stride_maskj
        mask_val = tl.load(mask_ptrs, mask=row_mask, other=1.0)   # (M,)
        mask_val = mask_val[:, None]                             # broadcast over H
        acc_proj = acc_proj * mask_val

    # ----- final output ------------------------------------------------------
    out_val = acc_proj * gate_sig                                 # (M, H)

    # store
    h_idx = tl.arange(0, H)
    out_ptrs = out_ptr + out_base[:, None] + h_idx[None, :] * stride_outh
    tl.store(out_ptrs, out_val, mask=row_mask[:, None])


def _proj_gate_mask(x: torch.Tensor,
                    proj_w: torch.Tensor,
                    gate_w: torch.Tensor,
                    mask: torch.Tensor | None,
                    out: torch.Tensor,
                    block_m: int = 64,
                    block_k: int = 32) -> None:
    """
    Fuse linear projection, gate (sigmoid) and optional mask.

    Arguments
    ---------
    x      : (B, N, N, C)   fp16
    proj_w : (C, H)         fp16, transposed weight (original weight.t())
    gate_w : (C, H)         fp16, transposed weight
    mask   : (B, N, N)      fp16 or None
    out    : (B, N, N, H)   fp16   (pre‑allocated)
    """
    B, N, _, C = x.shape
    H = proj_w.shape[1]

    # strides
    stride_xb, stride_xi, stride_xj, stride_xc = x.stride()
    stride_pw0, stride_pw1 = proj_w.stride()
    stride_gw0, stride_gw1 = gate_w.stride()
    if mask is not None:
        stride_maskb, stride_maski, stride_maskj = mask.stride()
    else:
        stride_maskb = stride_maski = stride_maskj = 0
    stride_outb, stride_outi, stride_outj, stride_outh = out.stride()

    total_rows = B * N * N
    grid = (triton.cdiv(total_rows, block_m),)

    _proj_gate_mask_kernel[grid](
        x_ptr=x,
        proj_w_ptr=proj_w,
        gate_w_ptr=gate_w,
        mask_ptr=mask if mask is not None else x,  # dummy pointer when not used
        out_ptr=out,
        stride_xb=stride_xb,
        stride_xi=stride_xi,
        stride_xj=stride_xj,
        stride_xc=stride_xc,
        stride_pw0=stride_pw0,
        stride_pw1=stride_pw1,
        stride_gw0=stride_gw0,
        stride_gw1=stride_gw1,
        stride_maskb=stride_maskb,
        stride_maski=stride_maski,
        stride_maskj=stride_maskj,
        stride_outb=stride_outb,
        stride_outi=stride_outi,
        stride_outj=stride_outj,
        stride_outh=stride_outh,
        B=B,
        N=N,
        C=C,
        H=H,
        BLOCK_M=block_m,
        BLOCK_K=block_k,
        APPLY_MASK=1 if mask is not None else 0,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor of shape [B, N, N, C] (float32/float16)
        mask         : torch.Tensor of shape [B, N, N] (bool/int/float) or None
        weights      : dict of model parameters (float32)
        config       : dict with keys "dim" (int) and "hidden_dim" (int)

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (same dtype as input)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16).contiguous()

    B, N, _, C = x.shape

    # ------------------------------------------------------------------
    # 3. Prepare weights (fp16, transposed for GEMM)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16).t().contiguous()
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16).t().contiguous()
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16).t().contiguous()
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16).t().contiguous()
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16).t().contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 4. Optional mask (fp16) – broadcasted later inside kernel
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
    else:
        mask_fp16 = None

    # ------------------------------------------------------------------
    # 5. Fused projection + gate (+ mask) for left/right
    # ------------------------------------------------------------------
    left = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
    right = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    _proj_gate_mask(x, left_proj_w, left_gate_w, mask_fp16, left)
    _proj_gate_mask(x, right_proj_w, right_gate_w, mask_fp16, right)

    # ------------------------------------------------------------------
    # 6. Compute out_gate (linear + sigmoid) – separate small GEMM
    # ------------------------------------------------------------------
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the summed dimension k
    #    Shapes: left/right : (B, N, N, H) → (B*H, N, N)
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) then view as (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Apply out_gate, output LayerNorm, final projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = torch.nn.functional.linear(out, to_out_w)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
