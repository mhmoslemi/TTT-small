"""Implementation of the outgoing TriMul operator (AlphaFold3) using Triton.

The kernel fuses:
  * elementwise multiplication of projected values, gates and optional mask,
  * batched matrix multiply `out = left @ rightᵀ` for each hidden channel.
The rest of the operations (LayerNorm, sigmoid, final projection) are done
with PyTorch for simplicity. The computation is performed in float32 to keep
numerical accuracy while still benefiting from Triton's fast GEMM kernel.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel: proj * gate * mask (broadcast mask over hidden dimension)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    proj_ptr,   # *[total_elems]  (float32 or float16)
    gate_ptr,   # *[total_elems]
    mask_ptr,   # *[mask_elems]   (broadcast over hidden dim)
    out_ptr,    # *[total_elems]
    total_elems,  # int
    hidden_dim,   # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # map flat offset to mask element (mask is broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Fuse (proj * gate * mask) using a Triton kernel.
    proj, gate : (B, N, N, H)   float32
    mask       : (B, N, N)      float32  (will be broadcast over H)
    Returns a tensor of same shape as proj.
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)

    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel: batched GEMM for the TriMul core reduction.
# Computes out = left @ rightᵀ for each hidden channel.
# Shapes:
#   left  : (B*H, M, K)
#   right : (B*H, N, K)
#   out   : (B*H, M, N)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to flattened tensors
    batch, M, N, K,                     # integer sizes (batch = B*H)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)  # batch dimension
    pid_m = tl.program_id(0)  # output M tile
    pid_n = tl.program_id(1)  # output N tile

    # offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_base = a_ptr + pid_b * stride_a_batch
    b_base = b_ptr + pid_b * stride_b_batch
    c_base = c_ptr + pid_b * stride_c_batch

    # masks for out-of-bounds handling
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block (M x K)
        a_offset = offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(
            a_base + a_offset,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B block transposed (K x N) from original (N x K)
        b_offset = offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(
            b_base + b_offset,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c_offset = offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_base + c_offset, acc.to(tl.float32), mask=mask)  # store as FP32


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using Triton.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (or None)
        - weights      : dict of model parameters
        - config       : dict containing 'dim', 'hidden_dim' and optional 'nomask'

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------
    # 1. Input LayerNorm
    # -----------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----------------------------------------------------------------
    # 2. Linear projections (dim -> hidden_dim) and gate projections
    # ----------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ----------------------------------------------------------------
    # 3. Fuse mask, projection and gate (Triton) if a mask is present
    # ----------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is float32 and contiguous
        mask_fp = mask.to(input_tensor.dtype).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 4. Core O(N³·hidden_dim) reduction: batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # bring hidden dimension into the batch axis
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    batch = B * H
    M = N
    N_ = N
    K = N

    # strides (in elements)
    stride_l_batch = left_flat.stride(0)
    stride_l_m = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_n = right_flat.stride(1)   # rows (j)
    stride_r_k = right_flat.stride(2)   # cols (k)

    stride_o_batch = out_flat.stride(0)
    stride_o_m = out_flat.stride(1)
    stride_o_n = out_flat.stride(2)

    # Tuned block sizes (work well for N up to 1024)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        math.ceil(M / BLOCK_M),
        math.ceil(N_ / BLOCK_N),
        batch,
    )

    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch,
        M,
        N_,
        K,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_n,
        stride_r_k,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5. Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    # --------------------------------------------------------------
    # 6. Cast back to original dtype (usually float32)
    # --------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out

    # 7. Cast back to the original dtype (usually float32)
    # -------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
