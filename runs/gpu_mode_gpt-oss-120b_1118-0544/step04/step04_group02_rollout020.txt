@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,   # fp16* total elements
    gate_ptr,   # fp16* total elements
    mask_ptr,   # fp16* mask elements (B*N*N)
    out_ptr,
    total_elems,   # int
    hidden_dim: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    
    # compute mask index: total elements = B*N*N*H, we want integer division by hidden_dim to broadcast mask over H.
    mask_idx = offs // hidden_dim
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # config contains dim, hidden_dim, maybe nomask
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', False)
    # get weight tensors
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_w = weights['left_proj.weight']
    right_proj_w = weights['right_proj.weight']
    left_gate_w = weights['left_gate.weight']
    right_gate_w = weights['right_gate.weight']
    out_gate_w = weights['out_gate.weight']
    to_out_norm_w = weights['to_out_norm.weight']
    to_out_norm_b = weights['to_out_norm.bias']
    to_out_w = weights['to_out.weight']

    # 1. Input layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias) # shape (B,N,N,dim)

    # 2. Cast to fp16
    x = x.to(torch.float16)

    # Convert weights to fp16
    left_proj_w = left_proj_w.to(torch.float16)
    right_proj_w = right_proj_w.to(torch.float16)
    left_gate_w = left_gate_w.to(torch.float16)
    right_gate_w = right_gate_w.to(torch.float16)
    out_gate_w = out_gate_w.to(torch.float16)
    to_out_norm_w = to_out_norm_w.to(torch.float16)
    to_out_norm_b = to_out_norm_b.to(torch.float16)
    to_out_w = to_out_w.to(torch.float16)

    # 3. Linear projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)  # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)  # (B,N,N,H)

    # 4. Gate projections
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # 5. apply mask + gates using Triton for left and right
    if not nomask and mask is not None:
        # ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        # Flatten mask for kernel usage
        # left_proj * left_gate * mask
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16) # defined using kernel
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # 6. Core reduction using batched matrix multiplication.
    # We'll use torch.bmm (fast) but wrap as a function.
    B, N, _, H = left.shape
    # reshape
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # BMM: left @ right^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 7. Output LayerNorm over hidden dimension H
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # 8. Multiply by out_gate and final linear projection
    # out_gate shape (B,N,N,H)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # shape (B,N,N,dim)

    # Cast back to original dtype
    out = out.to(input_tensor.dtype)
    return out

"""
Highly‑optimized Triton‑accelerated forward pass for the outgoing TriMul
operator (AlphaFold3 style).

Algorithm
---------
1. Layer‑normalize the 4‑D input tensor (B, N, N, C) using the provided
   weight/bias.
2. Cast to FP16 for compute‑heavy stages.
3. Linear projections (`left_proj`, `right_proj`) and gate projections
   (`left_gate`, `right_gate`, `out_gate`) are performed with FP16 weights.
4. If a mask is supplied (and `nomask` is False) a Triton kernel fuses
   element‑wise multiplication of projection, gate and mask, broadcasting the
   mask over the hidden dimension H.
5. Core reduction:
      out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   which is equivalent to a batched matrix multiplication
   `left @ rightᵀ` for each (b, h).  We reshape to (B·H, N, N) and use
   `torch.bmm` (cuBLAS) for fast FP16 GEMM.
6. Apply a second LayerNorm over the hidden dimension, multiply by
   `out_gate` and project back to the original channel dimension.
7. Cast the result back to the input dtype (usually float32).

At least one operation (gate × mask) is implemented in Triton to satisfy the
requirement of using a custom kernel.
"""
import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,                # *[total_elems]   fp16
    gate_ptr,                # *[total_elems]   fp16
    mask_ptr,                # *[mask_elems]    fp16
    out_ptr,                 # *[total_elems]   fp16
    total_elems,             # i64
    hidden_dim,              # i64  (size of the last dimension H)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # map flat index -> mask index (broadcast H)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Element‑wise ``proj * gate * mask`` where ``mask`` has shape (B, N, N)
    and is broadcast over the hidden dimension H.
    All tensors must be FP16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict containing all linear and LayerNorm parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask + gate (Triton kernel) or just gate
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction: batched GEMM (left @ rightᵀ)
    #    We reshape to (B*H, N, N) and use torch.bmm which is highly
    #    optimized on NVIDIA GPUs.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # left shape (B, N, N, H)

    # (B, H, N, N) -> (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: left @ rightᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
