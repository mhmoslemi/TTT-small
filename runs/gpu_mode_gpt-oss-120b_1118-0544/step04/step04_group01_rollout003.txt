"""
TriMul (outgoing) forward pass – Triton‑fused mask‑gate and
batched GEMM for the O(N³·H) reduction.

Algorithm
---------
1. Layer‑norm the [B,N,N,C] input.
2. Cast to fp16 for the compute‑heavy path.
3. Four linear projections (left/right + their gates) are performed
   with fp16 weights.  A Triton kernel fuses `proj * gate * mask` so
   that the mask and sigmoid gate are applied in a single pass.
   If `nomask` is True the mask is omitted and a simple element‑wise
   multiply is used.
4. The core update
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix multiplication:
   for each hidden dimension d we compute  out_d = left_d @ right_dᵀ.
   This is implemented with torch.bmm after permuting the tensors to
   shape (B·H, N, N), which lets cuBLAS/Tensor‑Core handle the O(N³·H)
   work efficiently.
5. Apply a second LayerNorm, multiply by the output gate and a final
   linear projection back to the original channel dimension.
6. Cast the result back to the input dtype.

Only the mask‑gate fusion is performed in Triton; all large GEMM
operations use the highly‑optimized cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total_elems,)
    gate_ptr,          # fp16 * (total_elems,)
    mask_ptr,          # fp16 * (mask_elems,) where mask_elems = B*N*N
    out_ptr,           # fp16 * (total_elems,)
    total_elems,       # i32   – total number of elements in proj/gate
    hidden_dim,        # i32   – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # Compute mask index (broadcast mask over hidden dimension)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using Triton.
    All tensors must be on the same CUDA device, dtype torch.float16,
    and contiguous.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],   # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype
    bs, seq_len, _, _ = input_tensor.shape

    # ------------------- LayerNorm (input) -------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for the compute‑heavy path
    x = x.to(torch.float16)

    # ----------------------- cast weights --------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections ---------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------- gates ------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ----------------------- mask & gate fusion -------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ----------------- core O(N³·H) reduction via batched GEMM ----------
    # Reshape to (B, H, N, N) then fuse batch and hidden dim.
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_reshape = left.view(bs * hidden_dim, seq_len, seq_len)   # (B*H, N, N)
    right_reshape = right.view(bs * hidden_dim, seq_len, seq_len) # (B*H, N, N)

    # batched matrix multiplication: left @ rightᵀ
    out = torch.bmm(left_reshape, right_reshape.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # ------------------- second LayerNorm & output gate ------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------ cast back ----------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
