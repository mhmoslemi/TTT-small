"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Apply LayerNorm to the input tensor (B, N, N, C).
2. Cast to fp16 for the compute‑intensive part.
3. Compute linear projections and gating projections (all fp16).
4. If a mask is present, fuse `proj * gate * mask` with a small Triton
   element‑wise kernel; otherwise just multiply by the gate.
5. Core O(N³·H) reduction:
      out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   This is a batch of independent GEMMs (B·H of size N×N).  A custom
   Triton kernel performs the batched GEMM, using fp16 arithmetic with
   fp32 accumulation.
6. Apply a second LayerNorm, multiply by the output gate and project back
   to the original channel dimension.
7. Cast the result back to the original dtype (usually float32).

Only the elementwise mask‑gate fusion and the batched GEMM are implemented
in Triton; the remaining operations use PyTorch for simplicity.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse projection, gate and optional mask
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    proj_ptr,          # *[total_elems]  fp16
    gate_ptr,          # *[total_elems]  fp16
    mask_ptr,          # *[mask_elems]  fp16
    out_ptr,           # *[total_elems]  fp16
    total_elems,       # int
    hidden_dim,        # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # each mask element is broadcast over `hidden_dim` consecutive entries
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=valid)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` where `mask` has shape (B, N, N) and is
    broadcast over the hidden dimension.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)
    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)

    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the core O(N³·H) reduction
# Computes out = left @ rightᵀ for each (batch, hidden) pair.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to flattened tensors
    batch, M, N, K,                     # integers (batch = B*H)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)  # batch dimension
    pid_m = tl.program_id(0)  # tile along M
    pid_n = tl.program_id(1)  # tile along N

    # Offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for the current batch
    a_base = a_ptr + pid_b * stride_a_batch
    b_base = b_ptr + pid_b * stride_b_batch
    c_base = c_ptr + pid_b * stride_c_batch

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator (FP32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block (M x K)
        a_offset = offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(
            a_base + a_offset,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B block (K x N) – note the transposition
        b_offset = offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(
            b_base + b_offset,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate (dot returns FP32)
        acc += tl.dot(a, b)

    # Write back result (cast to fp16)
    c_offset = offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_base + c_offset, acc.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] (may be None)
        - weights      : dict of torch.Tensors containing all model weights
        - config       : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,C) -> (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Fuse mask and gate (if present) or simple multiplication
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the reduction axis K
    #    Shapes:
    #        left, right : (B, N, N, H)
    #    We view them as (B*H, N, N) and compute
    #        out_flat = left_flat @ right_flatᵀ
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)

    batch = B * H
    M = N
    K = N
    # Strides in element units
    stride_l_batch = left_flat.stride(0)   # B*H stride
    stride_l_m     = left_flat.stride(1)   # row stride (N)
    stride_l_k     = left_flat.stride(2)   # col stride (1)

    stride_r_batch = right_flat.stride(0)
    stride_r_k     = right_flat.stride(1)   # row stride (N) – this is K dimension
    stride_r_n     = right_flat.stride(2)   # col stride (1) – this is N dimension

    stride_o_batch = out_flat.stride(0)
    stride_o_m     = out_flat.stride(1)
    stride_o_n     = out_flat.stride(2)

    # Tuned block sizes (fp16 tensor‑core friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        math.ceil(M / BLOCK_M),
        math.ceil(N / BLOCK_N),
        batch,
    )

    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch,
        M,
        N,
        K,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_n,
        stride_r_k,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before proceeding
    torch.cuda.synchronize()

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype (typically float32)
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
