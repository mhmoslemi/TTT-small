    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # get weight tensors
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    # LayerNorm
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # Linear projections (float32)
    left_proj = F.linear(x, left_proj_weight)  # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_weight)

    if not nomask:
        mask_f = mask.to(device).to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Gates (float32)
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Prepare for batched matmul: Permute to (B, hidden_dim, N, N)
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch and hidden dim
    batch_hidden = B * hidden_dim
    left_flat = left.view(batch_hidden, N, N)
    right_flat = right.view(batch_hidden, N, N)

    # Cast to fp16 for kernel
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Allocate output
    out_fp16 = torch.empty_like(left_fp16)

    # Compute strides
    stride_a_batch = left_fp16.stride(0)
    stride_a_m = left_fp16.stride(1)
    stride_a_k = left_fp16.stride(2)

    stride_b_batch = right_fp16.stride(0)
    stride_b_n = right_fp16.stride(1)   # N dimension
    stride_b_k = right_fp16.stride(2)   # K dimension

    stride_c_batch = out_fp16.stride(0)
    stride_c_m = out_fp16.stride(1)
    stride_c_n = out_fp16.stride(2)

        c_ptr=out_fp16.data_ptr(),
        M=M,
        N=N,
        K=K,
        stride_am=stride_a_m,
        stride_ak=stride_a_k,
        stride_bk=stride_b_k,
        stride_bn=stride_b_n,
        stride_cm=stride_c_m,
        stride_cn=stride_c_n,
        stride_batch_a=stride_a_batch,
        stride_batch_b=stride_b_batch,
        stride_batch_c=stride_c_batch,

        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,

                       weight=to_out_norm_weight,

    out = out * out_gate

    # Final linear projection to dim
    out = F.linear(out, to_out_weight)

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers with batch offset
    a_batch = a_ptr + batch_id * stride_batch_a
    b_batch = b_ptr + batch_id * stride_batch_b
    c_batch = c_ptr + batch_id * stride_batch_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_valid = k_offset < K

        # Load A block (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=k_valid[None, :], other=0.0)

        # Load B block (K x N) - transposed
        b_ptrs = b_batch + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=k_valid[:, None], other=0.0)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16))

"""
Outgoing TriMul (Triangle Multiplicative Update) – forward pass.

Algorithm
---------
1. Layer‑norm over the input channel dimension (float32).
2. Four linear projections (left/right and three gates) in float32.
3. Optional mask multiplication.
4. Apply sigmoid gates to the projections.
5. Fuse the O(N³) contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   by:
   a) reshaping to (B·D, N, N) where D = hidden_dim,
   b) performing a batched GEMM (left @ rightᵀ) with a custom Triton
      kernel that uses fp16 compute and fp32 accumulation.
6. Layer‑norm over the hidden dimension.
7. Multiply by the output gate (float32).
8. Final linear projection back to the original dimension.
9. Return a float32 tensor of shape [B, N, N, dim].

Only the heavy matrix multiplication (step 5) is implemented in Triton;
all other ops use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all int64)
    stride_am, stride_ak,              # A: (M‑stride, K‑stride)
    stride_bk, stride_bn,              # B transposed: (K‑stride, N‑stride)
    stride_cm, stride_cn,              # C: (M‑stride, N‑stride)
    stride_batch_a, stride_batch_b, stride_batch_c,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  where A,B shape = (batch, N, N)."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # batch offsets
    a_batch = a_ptr + batch_id * stride_batch_a
    b_batch = b_ptr + batch_id * stride_batch_b
    c_batch = c_ptr + batch_id * stride_batch_c

    # masks for out‑of‑bounds (handles any size, although test sizes are multiples of 32)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # ----- load A (M × K) -----
        a_ptrs = a_batch + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak
        mask_a = mask_m[:, None] & mask_k[None, :]
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # ----- load Bᵀ (K × N) -----
        b_ptrs = b_batch + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn
        mask_b = mask_k[:, None] & mask_n[None, :]
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # ----- dot product (fp16×fp16 → fp32) -----
        acc += tl.dot(a, b)

    # write result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config["nomask"] is True
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2. Linear projections (float32)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)

    left_proj = F.linear(x, left_proj_weight)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (float32)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(device).to(x.dtype).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4. Gates (float32) and apply
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left_proj * left_gate       # (B, N, N, hidden_dim)
    right = right_proj * right_gate    # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5. Batched GEMM: out = left @ rightᵀ  (per hidden channel)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # reshape to (B, hidden_dim, N, N) then collapse batch & hidden
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, hidden_dim, N, N)

    batch_hidden = B * hidden_dim
    left_flat = left.view(batch_hidden, N, N).contiguous()
    right_flat = right.view(batch_hidden, N, N).contiguous()

    # cast to fp16 for the kernel
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)

    # strides (in elements)
    stride_a_batch = left_fp16.stride(0)
    stride_a_m = left_fp16.stride(1)
    stride_a_k = left_fp16.stride(2)

    stride_b_batch = right_fp16.stride(0)
    stride_b_n = right_fp16.stride(1)   # N dimension
    stride_b_k = right_fp16.stride(2)   # K dimension

    stride_c_batch = out_fp16.stride(0)
    stride_c_m = out_fp16.stride(1)
    stride_c_n = out_fp16.stride(2)

    M = N
    K = N

    # block sizes (power‑of‑two that divide all test seq lengths)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        batch_hidden,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        a_ptr=left_fp16.data_ptr(),
        b_ptr=right_fp16.data_ptr(),
        c_ptr=out_fp16.data_ptr(),
        M=M,
        N=N,
        K=K,
        stride_am=stride_a_m,
        stride_ak=stride_a_k,
        stride_bk=stride_b_k,
        stride_bn=stride_b_n,
        stride_cm=stride_c_m,
        stride_cn=stride_c_n,
        stride_batch_a=stride_a_batch,
        stride_batch_b=stride_b_batch,
        stride_batch_c=stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output layer‑norm (float32)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden_dim) float32

    # ------------------------------------------------------------------
    # 7. Output gate multiplication (float32)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(device)
    out = F.linear(out, to_out_weight)   # (B, N, N, dim) float32

    return out
