"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The kernel fuses mask+gate element‑wise multiplication and computes the
core O(N³·H) contraction `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
using a tiled batched matrix‑multiply (`left @ rightᵀ`) written in
pure Triton.  All heavy arithmetic is performed in fp16 with
float32 accumulation; the rest of the pipeline (LayerNorms, linear
projections, sigmoids) is kept in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel that fuses element‑wise mask and gate multiplication.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [total_elements] fp16
    left_gate_ptr, right_gate_ptr,     # same
    mask_ptr,                          # [B*N*N] fp16
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share one mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)

# ----------------------------------------------------------------------
# Batched matrix‑multiply kernel that computes
#   out = left @ rightᵀ    (left/right: (M, N, N) where M = B*H)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    left_ptr,          # fp16 pointer to (M, N, N)
    right_ptr,         # fp16 pointer to (M, N, N)
    out_ptr,           # fp16 pointer to (M, N, N)
    M, N,              # runtime scalars (M = B*H)
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_k,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # 0 … M‑1
    pid_m = tl.program_id(1)       # block row index
    pid_n = tl.program_id(2)       # block col index

    # Offsets for the output block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator (float32 for stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        a_ptr = left_ptr + pid_batch * stride_l_batch \
                + offs_m[:, None] * stride_l_i \
                + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # rightᵀ block: shape (BLOCK_K, BLOCK_N)
        # we need element right[k, j] → after transposition becomes (k, j)
        b_ptr = right_ptr + pid_batch * stride_r_batch \
                + offs_k[:, None] * stride_r_i \
                + offs_n[None, :] * stride_r_k
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write the result back (fp16)
    c_ptr = out_ptr + pid_batch * stride_o_batch \
            + offs_m[:, None] * stride_o_i \
            + offs_n[None, :] * stride_o_k
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operation.

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as the input.
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # Config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for the compute‑heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, torch.float16)

    # Project to hidden dimension
    left = F.linear(x, left_proj_w)      # (B, N, N, H)
    right = F.linear(x, right_proj_w)    # (B, N, N, H)

    # Gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3) Fuse mask and gates (Triton)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask is broadcast over hidden dim
        mask_fp = mask.to(device, torch.float16).unsqueeze(-1)     # (B, N, N, 1)
        # Flatten tensors for the kernel
        B, N, _, H = left.shape
        total = B * N * N * H
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.squeeze(-1).contiguous().view(-1)   # (B*N*N,)

        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)
        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_flat.view(B, N, N, H)
        right = right_flat.view(B, N, N, H)
    else:
        # No mask – just element‑wise gate multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Core O(N³·H) contraction via Triton batched matmul
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # reuse after possible broadcast
    M = B * H  # flatten batch × hidden dim

    # Rearrange to (M, N, N) where each slice corresponds to a hidden channel
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(M, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(M, N, N)

    out_mat = torch.empty_like(left_mat)  # fp16

    # Strides (in elements)
    stride_l_batch = left_mat.stride(0)
    stride_l_i     = left_mat.stride(1)
    stride_l_k     = left_mat.stride(2)

    stride_r_batch = right_mat.stride(0)
    stride_r_i     = right_mat.stride(1)
    stride_r_k     = right_mat.stride(2)

    stride_o_batch = out_mat.stride(0)
    stride_o_i     = out_mat.stride(1)
    stride_o_k     = out_mat.stride(2)

    # Block sizes – tuned for typical N (powers of two up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (M,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M, N,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_i, stride_r_k,
        stride_o_batch, stride_o_i, stride_o_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to input dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

The kernel works for batch sizes 1‑2, sequence lengths up to 1024 and
hidden dimension 128 (as used in AlphaFold‑3).  The final output is
returned in the original dtype of the input tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Fuse mask and gate multiplication (optional mask)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [total_elements] fp16
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B*N*N] fp16
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries share the same mask entry
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)

# ----------------------------------------------------------------------
# Batched matmul kernel computing out = left @ rightᵀ
# left, right, out: (M, N, N) where M = B * hidden_dim
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    left_ptr,          # fp16 pointer (M, N, N)
    right_ptr,         # fp16 pointer (M, N, N)
    out_ptr,           # fp16 pointer (M, N, N)
    M, N,              # runtime scalars
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_k,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # 0 … M‑1
    pid_m = tl.program_id(1)       # output‑row block index
    pid_n = tl.program_id(2)       # output‑col block index

    # Offsets for the current output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: (BLOCK_M, BLOCK_K)  → rows i, cols k
        a_ptr = left_ptr + pid_batch * stride_l_batch \
                + offs_m[:, None] * stride_l_i \
                + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # rightᵀ block: (BLOCK_K, BLOCK_N) → rows k, cols j
        b_ptr = right_ptr + pid_batch * stride_r_batch \
                + offs_k[:, None] * stride_r_i \
                + offs_n[None, :] * stride_r_k
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store result (fp16)
    c_ptr = out_ptr + pid_batch * stride_o_batch \
            + offs_m[:, None] * stride_o_i \
            + offs_n[None, :] * stride_o_k
    tl.store(c_ptr, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32)
            - config: dict containing:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as the input.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, torch.float32)
    norm_b = weights["norm.bias"].to(device, torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for the compute‑heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device, torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device, torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device, torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device, torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device, torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device, torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, torch.float16)
    to_out_w      = weights["to_out.weight"].to(device, torch.float16)

    # Project to hidden dimension
    left = F.linear(x, left_proj_w)      # (B, N, N, H)
    right = F.linear(x, right_proj_w)    # (B, N, N, H)

    # Gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3) Fuse mask and gates (optional)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask broadcast over the hidden dimension
        mask_fp = mask.to(device, torch.float16).unsqueeze(-1)   # (B, N, N, 1)

        B, N, _, H = left.shape
        total = B * N * N * H

        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat= right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.squeeze(-1).contiguous().view(-1)   # (B*N*N,)

        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        left = left_flat.view(B, N, N, H)
        right = right_flat.view(B, N, N, H)
    else:
        # No mask – just element‑wise gate multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Core O(N³·H) contraction via batched matmul (Triton)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    M = B * H                                 # flattened batch × hidden

    # Rearrange to (M, N, N) where each slice corresponds to one hidden channel
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(M, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(M, N, N)

    out_mat = torch.empty_like(left_mat)      # fp16 output

    # Strides (in elements)
    stride_l_batch = left_mat.stride(0)
    stride_l_i     = left_mat.stride(1)
    stride_l_k     = left_mat.stride(2)

    stride_r_batch = right_mat.stride(0)
    stride_r_i     = right_mat.stride(1)
    stride_r_k     = right_mat.stride(2)

    stride_o_batch = out_mat.stride(0)
    stride_o_i     = out_mat.stride(1)
    stride_o_k     = out_mat.stride(2)

    # Block sizes – chosen to give good occupancy across the supported N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        M,                                 # batch × hidden
        triton.cdiv(N, BLOCK_M),           # rows
        triton.cdiv(N, BLOCK_N),           # cols
    )

    _tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M, N,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_i, stride_r_k,
        stride_o_batch, stride_o_i, stride_o_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to original dtype if required
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
