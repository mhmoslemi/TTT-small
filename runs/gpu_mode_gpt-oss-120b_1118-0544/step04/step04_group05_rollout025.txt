```
def batched_matmul_triton(A, B):
    assert A.shape == B.shape
    batch, N, _ = A.shape
    M=N; K=N
    # choose tile sizes
    if N <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
        num_warps = 2
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    else: # >512
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = ((N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            batch)

    C = torch.empty_like(A)
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(2), B.stride(1),  # B transposed layout
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=3,
    )
    return C
```

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (B, N, N, C).
2. Cast to fp16 and compute linear projections (left/right) and three sigmoid
   gates (left, right, out) using the provided weights.
3. Apply optional pairwise mask to the projections.
4. Fuse projections with their respective gates.
5. Fold the hidden dimension H into the batch dimension and compute the core
   reduction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched matrix multiplication C = A @ Bᵀ with a custom Triton kernel.
6. Reshape back to (B, N, N, H), apply a second LayerNorm over H, multiply by
   the output gate and project back to the original channel dimension.
7. Cast the result back to the input dtype.

The heavy O(N³·H) work is performed entirely by the Triton kernel, satisfying
the requirement to use Triton for part of the computation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ where A,B have shape (B*H, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes (M=N=seq_len, K=seq_len)
    stride_ah, stride_am, stride_ak,    # A strides: batch, row, col
    stride_bh, stride_bk, stride_bn,    # B transposed strides: batch, K, N
    stride_ch, stride_cm, stride_cn,    # C strides: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch index (B*H)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column

    # ------------------------------------------------------------------
    # Tile offset computation
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the current batch
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # ------------------------------------------------------------------
    # Pointers for the first K‑tile
    # ------------------------------------------------------------------
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # fp16 dot product accumulated in fp32
        acc += tl.dot(a, b, out_dtype=tl.float32)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for tensors of shape (batch, N, N) using Triton.
    All tensors must be fp16 and on CUDA.
    """
    assert A.shape == B.shape, "A and B must have identical shapes"
    batch, N, K = A.shape
    M = N
    assert N == K, "TriMul requires square sequence dimension"

    # ------------------------------------------------------------------
    # Heuristic block sizes and warps
    # ------------------------------------------------------------------
    if N <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
        num_warps = 2
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    else:  # N > 512
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    C = torch.empty_like(A)

    _batched_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),   # A: batch, row, col
        B.stride(0), B.stride(2), B.stride(1),   # B transposed: batch, K, N
        C.stride(0), C.stride(1), C.stride(2),   # C: batch, row, col
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=3,
    )
    return C

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask          : torch.Tensor of shape [B, N, N] (binary) or None
            - weights       : dict of model parameters (float32 tensors)
            - config        : dict with keys
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – optional bool, ignore mask when True

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ----------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------
    # 1. Input LayerNorm
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----------------------------------------------------------
    # 2. Cast to fp16 for heavy compute
    # ----------------------------------------------------------
    x = x.to(torch.float16)

    # ----------------------------------------------------------
    # 3. Cast weights to fp16 once
    # ----------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ----------------------------------------------------------
    # 4. Linear projections (B,N,N,H)
    # ----------------------------------------------------------
    left = F.linear(x, left_proj_w)    # (B,N,N,H)
    right = F.linear(x, right_proj_w)  # (B,N,N,H)

    # ----------------------------------------------------------
    # 5. Optional mask (binary, broadcast over hidden dim)
    # ----------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------
    # 6. Gates (sigmoid of linear projections)
    # ----------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    # ----------------------------------------------------------
    # 7. Apply gates to projections
    # ----------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 8. Fold hidden dimension into batch for GEMM: (B*H, N, N)
    # ----------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)
    left = left.view(B * H, N, N)   # (B*H, N, N)
    right = right.view(B * H, N, N) # (B*H, N, N)

    # ----------------------------------------------------------
    # 9. Core O(N³·H) reduction via Triton batched GEMM
    # ----------------------------------------------------------
    out_flat = _batched_gemm_triton(left, right)   # (B*H, N, N)

    # ----------------------------------------------------------
    # 10. Reshape back to (B, N, N, H)
    # ----------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ----------------------------------------------------------
    # 11. Output LayerNorm over hidden dimension
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------------------------------------------
    # 12. Apply output gate
    # ----------------------------------------------------------
    out = out * out_gate

    # ----------------------------------------------------------
    # 13. Final linear projection back to original channel dimension
    # ----------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ----------------------------------------------------------
    # 14. Cast back to original dtype if needed
    # ----------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
