def triton_batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    total_tiles = num_tiles_m * num_tiles_n

    batch_idx = pid // total_tiles
    tile_idx = pid % total_tiles
    pid_m = tile_idx // num_tiles_n
    pid_n = tile_idx % num_tiles_n

    # compute offsets
    a_batch_offset = batch_idx * stride_a_batch
    b_batch_offset = batch_idx * stride_b_batch
    c_batch_offset = batch_idx * stride_c_batch

    # row and col indexes
    row_offset = pid_m * BLOCK_M
    col_offset = pid_n * BLOCK_N

    # create thread offsets
    offs_m = row_offset + tl.arange(0, BLOCK_M)
    offs_n = col_offset + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out of bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        cur_k = k + offs_k
        mask_k = cur_k < K

        # Compute load pointers with broadcasting
        a_ptrs = a_ptr + a_batch_offset + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        b_ptrs = b_ptr + b_batch_offset + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row

        # Load with masks

        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Cast to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = c_ptr + c_batch_offset + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    c = acc.to(tl.float16)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. LayerNorm the input tensor (B, N, N, C) with provided LN parameters.
2. Cast to fp16 for the heavy compute path.
3. Linear projections (left/right) and three gate projections (left, right, out)
   are performed with fp16 weights; gates are passed through sigmoid.
4. Optional mask (broadcast over the hidden dimension) is fused with the
   projection * gate product using ordinary PyTorch (the cost is negligible).
5. Core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication over the hidden dimension.
   We implement it with a custom Triton kernel that treats the hidden dimension
   as an additional batch axis, i.e. a GEMM of shape (B*H, N, N) × (B*H, N, N)ᵀ.
   The kernel tiles the matrices (64×64) and accumulates in fp32.
6. Apply a second LayerNorm over the hidden dimension, multiply by the output
   gate and linearly project back to the original channel dimension.
7. Cast the result back to the input dtype (usually float32) and return.

The implementation mixes PyTorch for the inexpensive elementwise parts and a
high‑performance Triton kernel for the N³ reduction.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel for batched matrix multiplication:
#   C[b] = A[b] @ B[b]^T   with A,B shape (batch, M, K) and (batch, N, K)
#   Here M = N = K (square), and we treat the original right matrix as
#   transposed by swapping its stride arguments.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    batch, M, N, K,                    # dimensions
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of tiles per output matrix
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_tiles_m * num_tiles_n

    # decode program id into batch, tile row, tile col
    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    pid_m = tile_idx // num_tiles_n
    pid_n = tile_idx % num_tiles_n

    # offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this batch
    a_batch_off = batch_idx * stride_a_batch
    b_batch_off = batch_idx * stride_b_batch
    c_batch_off = batch_idx * stride_c_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + offs_k                      # (BLOCK_K,)

        # masks for out‑of‑bounds accesses
        mask_a = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        mask_b = (cur_k[:, None] < K) & (offs_n[None, :] < N)

        # load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + a_batch_off + offs_m[:, None] * stride_a_row + cur_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # load B tile as transposed (BLOCK_K, BLOCK_N)
        # we treat the original right matrix as Bᵀ by swapping its stride arguments
        b_ptrs = b_ptr + b_batch_off + cur_k[:, None] * stride_b_col + offs_n[None, :] * stride_b_row
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # promote to fp32 for better accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_N)

    # write back result in fp16
    c_ptrs = c_ptr + c_batch_off + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

# ----------------------------------------------------------------------
# Helper to launch the batched GEMM kernel
# ----------------------------------------------------------------------
def _batched_matmul(a: torch.Tensor, b: torch.Tensor,
                    BLOCK_M=64, BLOCK_N=64, BLOCK_K=64) -> torch.Tensor:
    """
    a, b: tensors of shape (BATCH, N, N) and dtype fp16, contiguous.
    Returns c of the same shape and dtype.
    """
    assert a.is_contiguous() and b.is_contiguous()
    BATCH, N, _ = a.shape
    c = torch.empty_like(a)

    # strides for contiguous (batch, row, col) layout
    stride_a_batch, stride_a_row, stride_a_col = a.stride()
    stride_b_batch, stride_b_row, stride_b_col = b.stride()
    stride_c_batch, stride_c_row, stride_c_col = c.stride()

    # total number of program instances
    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = BATCH * num_tiles_m * num_tiles_n

    grid = (total_programs,)

    _batched_matmul_kernel[grid](
        a, b, c,
        BATCH, N, N, N,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row, stride_b_col,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return c

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                              Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor, shape [B, N, N] (bool/int/float) or None
            - weights: dict of model parameters (float32)
            - config: dict containing
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16 (no bias in linear layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections and gates (fp16)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)          # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5. Apply mask (if any) and combine with gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        mask_exp = mask_fp16.unsqueeze(-1)            # (B,N,N,1)
        left = left_proj * left_gate * mask_exp
        right = right_proj * right_gate * mask_exp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core N³ reduction via Triton batched GEMM
    #    Reshape to (B*H, N, N) where H = hidden_dim
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N) contiguous layout
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Triton GEMM: left_flat @ right_flatᵀ
    out_flat = _batched_matmul(left_flat, right_flat)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)    # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8. Cast back to original dtype and return
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)
    return out
