@triton.jit
def mask_gate_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, total_elements, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < total_elements

    # Compute mask index: each group of hidden_dim corresponds to one mask element.
    mask_idx = offsets // hidden_dim

    proj = tl.load(proj_ptr + offsets, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offsets, mask=mask, other=0.0)
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * mval
    tl.store(out_ptr + offsets, out, mask=mask)

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    bh = pid  # combined batch and head index
    b = bh // H
    h = bh % H

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    i_offset = i_start + tl.arange(0, BLOCK_I)
    j_offset = j_start + tl.arange(0, BLOCK_J)

    # mask for valid i,j
    mask_i = i_offset < N
    mask_j = j_offset < N

    # base offset for batch and head
    offset_base = b * stride_lb + h * stride_lh  # same for left/right/out

    # initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_offset = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left tile (i,k)
        left_ptrs = left_ptr + offset_base + i_offset[:, None] * stride_li + k_offset[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile (j,k) transposed: shape (BLOCK_K, BLOCK_J)
        right_ptrs = right_ptr + offset_base + k_offset[:, None] * stride_rk + j_offset[None, :] * stride_ri
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Compute out tile: left (BLOCK_I, BLOCK_K) @ right (BLOCK_K, BLOCK_J) -> (BLOCK_I, BLOCK_J)
        acc += tl.dot(left, right)  # tl.dot supports matrix multiplication

    # write output
    out_ptrs = out_ptr + offset_base + i_offset[:, None] * stride_oi + j_offset[None, :] * stride_oj
    # Cast accumulator to fp16
    out_val = acc.to(tl.float16)
    tl.store(out_ptrs, out_val, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass – Triton‑accelerated core reduction.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to fp16 for the compute‑heavy part.
3. Fuse the four linear projections (left/right and their gates) into a single
   matmul:   X @ W_all  where   W_all = [W_left; W_left_gate;
                                        W_right; W_right_gate].
   The result is split into left/right projections and gates, and the gates
   are passed through a sigmoid.
4. Compute `out_gate = σ(X @ W_out_gate)`.
5. (Optional) Fuse mask, gate and projection with a tiny Triton kernel:
        out = proj * gate * mask   (mask broadcast over the hidden dim).
6. Core triangular multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that performs a batched GEMM
   over the `k` dimension using fp16 inputs and fp32 accumulation.
7. Apply LayerNorm over the hidden dimension, multiply by `out_gate`,
   and apply the final linear projection back to `dim`.
8. Cast the result back to the original dtype (usually float32).

The implementation uses two Triton kernels (mask‑gate fusion & the O(N³)
reduction) while all other operations are performed with high‑performance
PyTorch primitives.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse projection * gate * mask (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,            # fp16 * (total_elems,)
    gate_ptr,            # fp16 * (total_elems,)
    mask_ptr,            # fp16 * (mask_elems,)
    out_ptr,             # fp16 * (total_elems,)
    total_elems,         # int   = B * N * N * H
    hidden_dim,          # int   = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems
    # each hidden_dim elements share the same mask element
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor,
                     mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """Fuse `proj * gate * mask` with Triton.
    All tensors must be fp16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: core triangular multiplication (outgoing version)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,            # fp16 * (B, N, N, H)
    right_ptr,           # fp16 * (B, N, N, H)
    out_ptr,             # fp16 * (B, N, N, H)
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    bh_id = tl.program_id(0)          # combined batch & head index
    i_tile = tl.program_id(1)         # i‑tile
    j_tile = tl.program_id(2)         # j‑tile

    # decode batch and head
    b = bh_id // H
    h = bh_id % H

    # start offsets for this tile
    i_start = i_tile * BLOCK_I
    j_start = j_tile * BLOCK_J

    # ranges for i, j
    i_range = i_start + tl.arange(0, BLOCK_I)
    j_range = j_start + tl.arange(0, BLOCK_J)

    # masks for out‑of‑bounds i/j
    mask_i = i_range < N
    mask_j = j_range < N

    # base offsets for the current (b, h) slice
    base_offset = b * stride_lb + h * stride_lh   # same for left/right/out

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k‑dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # ----- load left tile (i, k) -----
        left_ptrs = left_ptr + base_offset \
                    + i_range[:, None] * stride_li \
                    + k_range[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ----- load right tile transposed (k, j) -----
        # we load as (k, j) so that tl.dot(left, right) works directly
        right_ptrs = right_ptr + base_offset \
                     + k_range[:, None] * stride_rk \
                     + j_range[None, :] * stride_ri
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # matrix multiplication for this k‑block
        acc += tl.dot(left, right)      # (BLOCK_I, BLOCK_J)

    # write the result
    out_ptrs = out_ptr + base_offset \
               + i_range[:, None] * stride_oi \
               + j_range[None, :] * stride_oj
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul(left: torch.Tensor, right: torch.Tensor,
             block_i: int = 32, block_j: int = 32, block_k: int = 32) -> torch.Tensor:
    """Batched triangular multiplication using the Triton kernel above."""
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides (must be ints)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    grid = (B * H,               # combined batch & head
            triton.cdiv(N, block_i),
            triton.cdiv(N, block_j))

    _tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_ri, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_I=block_i, BLOCK_J=block_j, BLOCK_K=block_k,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config) or
              (input_tensor, weights, config) if no mask.
            - input_tensor : torch.Tensor [B, N, N, C]
            - mask         : torch.Tensor [B, N, N] (optional)
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack arguments (support both 3‑ and 4‑tuple inputs)
    # ------------------------------------------------------------------
    if len(data) == 4:
        input_tensor, mask, weights, config = data
    elif len(data) == 3:
        input_tensor, weights, config = data
        mask = None
    else:
        raise ValueError("data must be a tuple of 3 or 4 elements.")

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm over channel dimension `dim`
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim), dtype = input dtype

    # --------------------------------------------------------------
    # 2. Cast to fp16 for heavy compute
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Fuse left/right projections and their gates
    # --------------------------------------------------------------
    # concatenate weights: [W_left; W_left_gate; W_right; W_right_gate]
    w_all = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"]
    ], dim=0).to(torch.float16)               # (4*H, dim)

    # single linear projection
    all_proj = torch.nn.functional.linear(x, w_all)   # (B, N, N, 4*H)

    # split
    left_proj = all_proj[..., :hidden_dim]
    left_gate = all_proj[..., hidden_dim:2 * hidden_dim].sigmoid()
    right_proj = all_proj[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate = all_proj[..., 3 * hidden_dim:].sigmoid()

    # --------------------------------------------------------------
    # 4. Out gate (separate linear)
    # --------------------------------------------------------------
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()               # (B, N, N, H)

    # --------------------------------------------------------------
    # 5. Apply mask & gating (fused via Triton when a mask is present)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6. Core triangular multiplication (outgoing)
    # --------------------------------------------------------------
    out = _tri_mul(left, right,
                   block_i=32, block_j=32, block_k=32)   # (B, N, N, H)

    # --------------------------------------------------------------
    # 7. Output LayerNorm, gating, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate

    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )   # (B, N, N, dim)

    # --------------------------------------------------------------
    # 8. Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
