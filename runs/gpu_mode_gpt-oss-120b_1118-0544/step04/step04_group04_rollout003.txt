"""
Outgoing TriMul (Triangle Multiplicative Update) forward pass.

Algorithm:
1. Layer‑norm the input pair representation (float32).
2. Cast to fp16 and compute four linear projections:
   - left_proj, right_proj (hidden dimension)
   - left_gate, right_gate, out_gate (sigmoids)
3. Optional masking of left/right projections.
4. Apply gates: left = left_proj * left_gate, right = right_proj * right_gate.
5. Core O(N³·H) multiplication:
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   Implemented with a single torch.einsum, which leverages an
   optimized GPU kernel without extra memory copies.
6. Layer‑norm over the hidden dimension.
7. Fuse element‑wise multiplication with out_gate using a tiny Triton kernel.
8. Final linear projection back to the original channel dimension.
9. Return result in float32.

All heavy arithmetic is performed in fp16 to utilize Tensor Cores.
At least one operation (the gating multiplication) is executed in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    """out = out * gate  (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out * gate, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config["nomask"] is True
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension H
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)  # float32
    norm_bias = weights["norm.bias"].to(device)     # float32
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    left_proj = F.linear(x_fp16, left_proj_weight)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x_fp16, right_proj_weight)

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5. Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))

    left = left_proj * left_gate      # (B, N, N, hidden_dim)
    right = right_proj * right_gate   # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6. Core multiplication: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented via einsum to avoid extra memory shuffles.
    # ------------------------------------------------------------------
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)  # fp16

    # ------------------------------------------------------------------
    # 7. Output layer‑norm over hidden dimension (fp16)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden_dim), fp16

    # ------------------------------------------------------------------
    # 8. Fuse element‑wise multiplication with out_gate (Triton)
    # ------------------------------------------------------------------
    total_elements = out.numel()
    BLOCK = 1024
    grid = ((total_elements + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        out,
        out_gate,
        total_elements,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 9. Final linear projection back to original dim (fp16)
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)  # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)  # (B, N, N, dim), fp16

    # ------------------------------------------------------------------
    # 10. Cast to float32 as required by the interface
    # ------------------------------------------------------------------
    return out.to(torch.float32)
