    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # Layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # load weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)  # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # Gates (pre-sigmoid)
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B,N,N,H)

    if mask is not None and not nomask:
        # mask shape (B,N,N)
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core reduction via batched GEMM
    # reshape to (B*H, N, N)
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2)  # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1)  # (B, H, N, N) where k is rows
    left_mat = left_perm.reshape(B * hidden_dim, N, N)
    right_mat = right_perm.reshape(B * hidden_dim, N, N)

    # matmul: (B*H, N, N) @ (B*H, N, N) -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)
    out = out_mat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

"""
Highly‑optimized implementation of the outgoing TriMul (Triangle Multiplicative Update)
operator from AlphaFold3.

Algorithm
---------
1️⃣ Layer‑normalize the input tensor (B,N,N,C).
2️⃣ Cast to fp16 for the compute‑intensive part.
3️⃣ Compute four linear projections (left/right) and three linear gates
   (left_gate, right_gate, out_gate) using cuBLAS (torch.nn.functional.linear).
4️⃣ If a mask is present, fuse `proj * gate * mask` via a tiny Triton kernel
   (reducing memory traffic and kernel launches).  Otherwise use a simple
   element‑wise multiplication.
5️⃣ Core O(N³·H) reduction:
   For each hidden channel `h` we perform a batched matrix multiplication
   `out_{b,h} = left_{b,h} @ right_{b,h}.T`, where
   `left_{b,h}` and `right_{b,h}` are (N×N) matrices indexed as
   (i,k) and (j,k) respectively.  This is realised with `torch.bmm`.
6️⃣ Layer‑norm on the hidden dimension, multiply by the output gate,
   and apply the final linear projection back to the original channel size.
7️⃣ Cast the result back to the input dtype.

The only Triton component is the mask‑gate fusion kernel; the rest
leverages highly‑optimised cuBLAS/Tensor‑core paths.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total_proj_elements,)
    gate_ptr,          # fp16 * (total_proj_elements,)
    mask_ptr,          # fp16 * (mask_elements = B*N*N,)
    out_ptr,           # fp16 * (total_proj_elements,)
    total_elems,       # i32   – total number of elements in proj/gate (B*N*N*H)
    hidden_dim,        # i32   – H (size of hidden dim)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # each hidden dimension repeats mask, so compute mask index
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0).to(tl.float16)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0).to(tl.float16)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0).to(tl.float16)

    tl.store(out_ptr + offs, proj * gate * m, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using Triton.
    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out

    # Cast model weights to fp16 (biases are present only for LayerNorms)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections (no bias) – shape (B, N, N, H)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # Linear gates followed by sigmoid – shape (B,N,N,H)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # Fuse projection * gate * mask (if mask present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B,N,N) → fp16, contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·H) reduction via batched GEMM
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left.shape = (B, N, N, H)

    # Rearrange to (B, H, N, N) where the reduction dimension is the innermost
    left_perm  = left.permute(0, 3, 1, 2)          # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1)         # (B, H, k, j)

    # Collapse batch and hidden dimensions for a single torch.bmm call
    left_mat  = left_perm.reshape(B * hidden_dim, N, N)   # (B*H, N, N)
    right_mat = right_perm.reshape(B * hidden_dim, N, N)  # (B*H, N, N)

    # Matrix multiplication: (i,k) · (k,j) -> (i,j)
    out_mat = torch.bmm(left_mat, right_mat)               # (B*H, N, N)

    # Restore original layout (B, N, N, H)
    out = out_mat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # Second LayerNorm on the hidden dimension H
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # Apply output gate (broadcasted over the hidden dimension)
    out = out * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to original channel dimension C
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)
    return out
```
