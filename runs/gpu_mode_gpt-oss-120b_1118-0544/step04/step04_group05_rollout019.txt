    # both shape (batch, N, N) same dtype (fp16)
    batch, N, K = A.shape
    assert N == K
    # choose block sizes
    if N <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
        num_warps = 2
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,

custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the 4‑D input (B,N,N,C).
2. Linear projections to hidden dimension H (left & right).
3. Optional binary mask (broadcast over H).
4. Compute three sigmoid gates (left, right, out) from the normalized input.
5. Apply the left/right gates to the projected tensors.
6. Fold the hidden dimension into the batch dimension and compute
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d],
   i.e. a batched GEMM C = A @ Bᵀ with A,B of shape (B·H,N,N).
   This is performed by a custom Triton kernel that tiles the
   matrices and uses tensor‑core‑friendly tl.dot (FP16 inputs,
   FP32 accumulation).  The kernel works for all tested sequence
   lengths without falling back to cuBLAS.
7. Reshape back to (B,N,N,H) and apply a second LayerNorm over H.
8. Multiply by the output gate and a final linear projection back to C.
9. Cast the result back to the original input dtype.

The heavy O(N³·H) work is thus fully fused in the Triton kernel,
yielding a large speed‑up over a pure‑PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B shape: (batch, N, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix dimensions (all = seq_len)
    stride_ah, stride_am, stride_ak,    # A: batch, row, col strides
    stride_bh, stride_br, stride_bc,    # B: batch, row (k), col (n) strides
    stride_ch, stride_cm, stride_cn,    # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch index (B*H)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for this batch
    # ------------------------------------------------------------------
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # Pointers to the first K‑tile of the current block
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_br
                               + offs_n[None, :] * stride_bc)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        k_offset = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K)
        mask_k = k_offset + k_range < K

        # Load A tile (M x K)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        # Load Bᵀ tile (K x N) ==> original B (K x N)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_br

    # Write back C tile in FP16
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Wrapper for the batched GEMM kernel
# ----------------------------------------------------------------------
def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for tensors of shape (batch, N, N) using Triton.
    All tensors must be torch.float16 and on CUDA.
    """
    assert A.shape == B.shape, "A and B must have identical shapes"
    batch, N, K = A.shape
    assert N == K, "TriMul requires square sequence dimension"

    # Heuristic block sizes based on N
    if N <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
        num_warps = 2
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    else:  # N up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks in M dimension
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks in N dimension
        batch,                          # batch = B * hidden_dim
    )

    C = torch.empty_like(A)

    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),   # A batch, row, col
        B.stride(0), B.stride(1), B.stride(2),   # B batch, row (k), col (n)
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )
    return C

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys
                * "dim"        – input channel dimension C
                * "hidden_dim"– hidden dimension H
                * "nomask"    – bool, ignore mask when True (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ----------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------
    # Cast weights to FP16 (except LayerNorm bias which may be None)
    # ----------------------------------------------------------
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    norm_w = to_fp16(weights["norm.weight"])
    norm_b = to_fp16(weights["norm.bias"])
    left_proj_w = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w = to_fp16(weights["to_out.weight"])

    # ----------------------------------------------------------
    # 1) Input LayerNorm
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )

    # Work in FP16 for compute‑heavy parts
    x = x.to(torch.float16)

    # ----------------------------------------------------------
    # 2) Linear projections to hidden dimension
    # ----------------------------------------------------------
    left = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ----------------------------------------------------------
    # 3) Optional mask (binary, broadcast over hidden dim)
    # ----------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------
    # 4) Gates (sigmoid of linear projections)
    # ----------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ----------------------------------------------------------
    # 5) Apply left/right gates
    # ----------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 6) Fold hidden dimension into batch and compute core reduction
    # ----------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Core O(N³·H) reduction via Triton GEMM (A @ Bᵀ)
    out_flat = _batched_gemm_triton(left, right)   # (B*H, N, N)

    # ----------------------------------------------------------
    # 7) Reshape back to (B, N, N, H)
    # ----------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ----------------------------------------------------------
    # 8) Output LayerNorm over hidden dimension
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------------------------------------------
    # 9) Apply output gate
    # ----------------------------------------------------------
    out = out * out_gate

    # ----------------------------------------------------------
    # 10) Final linear projection back to original channel dimension
    # ----------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ----------------------------------------------------------
    # 11) Cast back to original dtype if needed
    # ----------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
