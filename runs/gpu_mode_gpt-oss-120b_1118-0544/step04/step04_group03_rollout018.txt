Let's design a simpler _batched_matmul_kernel similar to known Triton tutorial style:

```
@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    batch = tl.program_id(1)

    # tile indices
    grid_m = tl.cdiv(M, BLOCK_M)
    grid_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // grid_n
    pid_n = pid % grid_n

    # compute offsets
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # (K loop)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # load A
        a = tl.load(a_ptr + batch * batch_stride_a
            + off_m[:, None] * stride_am
            + cur_k[None, :] * stride_ak,
            mask=(off_m[:, None] < M) & (cur_k[None, :] < K), other=0.0)
        # load B (K,N)
        b = tl.load(b_ptr + batch * batch_stride_b
            + cur_k[:, None] * stride_bk
            + off_n[None, :] * stride_bn,
            mask=(cur_k[:, None] < K) & (off_n[None, :] < N), other=0.0)
        acc += tl.dot(a, b)
    # store
    c = acc.to(tl.float16)
    tl.store(c_ptr + batch * batch_stride_c
        + off_m[:, None] * stride_cm
        + off_n[None, :] * stride_cn,
        c,
        mask=(off_m[:, None] < M) & (off_n[None, :] < N))
```

"""
TriMul (outgoing) optimized with Triton.

The core operation computes, for each batch b and hidden dimension h,
    out[b, i, j, h] = Σ_k  left[b, i, k, h] * right[b, j, k, h]
which is a batched matrix multiplication:
    out_h = left_h @ right_hᵀ  with left_h, right_h ∈ ℝ^{N×N}.
We fuse the linear projections + gates (using concatenated weights) and
execute the O(N³·H) reduction with a custom Triton GEMM kernel.
All heavy arithmetic is performed in float16 with float32 accumulation.
The final result is returned in the original dtype of the input tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton batched GEMM kernel (C = A @ Bᵀ)
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dims (M×K)·(K×N)→(M×N)
    stride_am, stride_ak,         # strides for A (M, K)
    stride_bk, stride_bn,         # strides for B (K, N)   (B is already transposed)
    stride_cm, stride_cn,         # strides for C (M, N)
    stride_ab, stride_bb, stride_cb,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM: C = A @ Bᵀ.
    A: (batch, M, K), B: (batch, K, N) (already transposed w.r.t. original right),
    C: (batch, M, N).
    """
    pid = tl.program_id(0)        # tile index for (M, N)
    batch = tl.program_id(1)      # batch index (B * hidden_dim)

    # Compute tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # Load A (M × K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B (K × N) – B is already the transposed view of the original right
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)      # fp16·fp16 → fp32

    # Write output
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    orig_dtype = x.dtype
    device = x.device

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 2. Cast to half-precision for compute‑heavy path
    # ------------------------------------------------------------------- #
    x = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 3. Prepare weights (half‑precision)
    # ------------------------------------------------------------------- #
    dtype = torch.float16
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype)            # (2*H, C)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype)            # (2*H, C)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype)          # (H, C)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype)    # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype)      # (H,)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype)              # (C, H)

    # ------------------------------------------------------------------- #
    # 4. Linear projections + gates (combined)
    # ------------------------------------------------------------------- #
    # Combined linear: [proj | gate] → (2*H)
    left_comb = F.linear(x, left_comb_w)    # [B, N, N, 2*H]
    right_comb = F.linear(x, right_comb_w)  # [B, N, N, 2*H]

    # Split
    left_proj = left_comb[..., :hidden_dim]                     # [B, N, N, H]
    left_gate = left_comb[..., hidden_dim:].sigmoid()           # [B, N, N, H]
    right_proj = right_comb[..., :hidden_dim]                   # [B, N, N, H]
    right_gate = right_comb[..., hidden_dim:].sigmoid()         # [B, N, N, H]

    # Apply mask if present and not ignored
    if mask is not None and not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Apply gates
    left = left_proj * left_gate    # [B, N, N, H]
    right = right_proj * right_gate # [B, N, N, H]

    # ------------------------------------------------------------------- #
    # 5. Core reduction: batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # left: (i, k)   – keep order (i,k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    # right needs to be transposed to (k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k, j)

    # Collapse batch & hidden dimensions for the kernel
    left_flat = left_perm.view(B * H, N, N)   # (batch, M=N, K=N)
    right_flat = right_perm.view(B * H, N, N) # (batch, K=N, N)

    out_flat = torch.empty_like(left_flat)    # (batch, M, N), fp16

    # Strides
    stride_a_batch = left_flat.stride(0)
    stride_am = left_flat.stride(1)      # M stride
    stride_ak = left_flat.stride(2)      # K stride

    stride_b_batch = right_flat.stride(0)
    stride_bk = right_flat.stride(1)     # K stride
    stride_bn = right_flat.stride(2)     # N stride

    stride_c_batch = out_flat.stride(0)
    stride_cm = out_flat.stride(1)       # M stride
    stride_cn = out_flat.stride(2)       # N stride

    # Tile sizes – chosen to fit typical N ≤ 1024 and H ≤ 384
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_tiles_m * num_tiles_n, B * H)

    _batched_gemm_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,                     # M, N, K (all N)
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 6. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------- #
    # 7. Output gate (element‑wise) – PyTorch (fast enough)
    # ------------------------------------------------------------------- #
    out_gate = F.linear(x, out_gate_w).sigmoid()   # [B, N, N, H]
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8. Final projection back to original channel dimension
    # ------------------------------------------------------------------- #
    out = F.linear(out, to_out_w)   # [B, N, N, C]

    # Cast back to the original dtype of the input tensor
    out = out.to(orig_dtype)

    return out
