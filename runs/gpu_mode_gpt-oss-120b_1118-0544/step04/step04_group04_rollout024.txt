@triton.jit
def batch_matmul_kernel(A_ptr, B_ptr, C_ptr, M, N, K,
                        stride_am, stride_ak,
                        stride_bk, stride_bn,
                        stride_cm, stride_cn,
                        batch_stride_a, batch_stride_b, batch_stride_c,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
                        BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)  # batch index
    block_m = tl.program_id(1)
    block_n = tl.program_id(2)

    # compute offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute k loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        a_ptrs = A_ptr + pid * batch_stride_a + (offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak)
        b_ptrs = B_ptr + pid * batch_stride_b + (k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (k_offset[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k_offset[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        acc += tl.dot(a, b)
    # store
    c_ptrs = C_ptr + pid * batch_stride_c + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batch_matmul_fp16_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers to the start of the batch
    a_batch_ptr = A_ptr + batch_id * stride_batch_a
    b_batch_ptr = B_ptr + batch_id * stride_batch_b
    c_batch_ptr = C_ptr + batch_id * stride_batch_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # a[b, m, k]
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak)
        # b[b, k, n]
        b_ptrs = b_batch_ptr + (k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (k_offset[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k_offset[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    """out = out * gate (flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out * gate, mask=mask)


def custom_kernel(data):
    """
    Outgoing TriMul forward pass optimized with Triton.
    Performs LayerNorm, linear projections, gating, a batched
    matrix multiplication (left @ right^T) via a custom Triton kernel,
    a second Triton kernel for element‑wise gating, and final projection.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_fp16 = torch.float16
    dtype_fp32 = torch.float32

    # ---------- 1. Input LayerNorm (float32 for stability) ----------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    # LayerNorm expects normalized_shape = (dim,)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim], float32

    # Cast to FP16 for the heavy compute
    x_fp16 = x.to(dtype_fp16)

    # ---------- 2. Linear projections ----------
    left_proj_weight = weights["left_proj.weight"].to(device).to(dtype_fp16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(dtype_fp16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(dtype_fp16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(dtype_fp16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(dtype_fp16)

    left_proj = F.linear(x_fp16, left_proj_weight)   # (B, N, N, hidden)
    right_proj = F.linear(x_fp16, right_proj_weight)

    # ---------- 3. Optional masking ----------
    if not nomask:
        mask_f = mask.to(device).to(dtype_fp16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ---------- 4. Gating ----------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))

    left = left_proj * left_gate   # (B, N, N, hidden)
    right = right_proj * right_gate

    # ---------- 5. Batched GEMM via Triton ----------
    B, N, _, _ = left.shape  # batch, seq_len, seq_len, hidden

    # Permute to (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right needs to be transposed on the last two dims before permuting
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, hidden, N, N) where dim0=N_k, dim1=N_j

    batch = B * hidden_dim

    # Collapse the first two dimensions -> (batch, N, N)
    left_mat = left_perm.view(batch, N, N)
    right_mat = right_perm.view(batch, N, N)
    out_mat = torch.empty_like(left_mat)  # allocate output (batch, N, N)

    # Strides (in elements)
    stride_batch_a = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_batch_b = right_mat.stride(0)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)

    stride_batch_c = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batch_matmul_fp16_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_batch_a=stride_batch_a,
        stride_batch_b=stride_batch_b,
        stride_batch_c=stride_batch_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- 6. to_out_norm ----------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(dtype_fp16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(dtype_fp16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden), fp16

    # ---------- 7. out_gate multiplication ----------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # (B, N, N, hidden)
    total_el = out.numel()
    BLOCK = 1024
    grid_gate = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid_gate](
        out,
        out_gate,
        total_el,
        BLOCK=BLOCK,
    )

    # ---------- 8. Final projection ----------
    to_out_weight = weights["to_out.weight"].to(device).to(dtype_fp16)
    out = F.linear(out, to_out_weight)  # (B, N, N, dim), fp16

    # Return float32 as required
    return out.to(dtype_fp32)

"""
Outgoing TriMul (Triangle Multiplicative Update) forward pass.

The implementation follows the AlphaFold3 TriMul module:
1. LayerNorm over the input channel dimension (float32 for numerical stability).
2. Four linear projections (left/right and their gates) in FP16.
3. Optional masking of the left/right projections.
4. Gating via sigmoid and element‑wise multiplication.
5. Batched matrix multiplication   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   performed by a custom Triton kernel that treats each (batch * hidden) pair as an
   independent GEMM.
6. LayerNorm over the hidden dimension.
7. Out‑gate (linear + sigmoid) fused with element‑wise multiplication via a second
   Triton kernel.
8. Final linear projection back to the original dimension and cast to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batch_matmul_fp16_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication in FP16 (A @ B) -> C.

    A : (batch, M, K)   fp16
    B : (batch, K, N)   fp16
    C : (batch, M, N)   fp16   (accumulated in fp32)
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = A_ptr + batch_id * stride_batch_a
    b_batch = B_ptr + batch_id * stride_batch_b
    c_batch = C_ptr + batch_id * stride_batch_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)

        a_ptrs = a_batch + (offs_m[:, None] * stride_am + k_off[None, :] * stride_ak)
        b_ptrs = b_batch + (k_off[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (k_off[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(k_off[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        acc += tl.dot(a, b)          # fp16 * fp16 -> fp32 accumulation

    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    """Flattened element‑wise multiplication: out = out * gate."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out * gate, mask=mask)


def custom_kernel(data):
    """
    Outgoing TriMul forward pass optimized with Triton.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, C]  (float32)
            mask         : torch.Tensor [B, N, N]    (bool or float)
            weights      : dict of model parameters (float32)
            config       : dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    fp16 = torch.float16
    fp32 = torch.float32

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, dim], fp32

    x_fp16 = x.to(fp16)

    # ------------------------------------------------------------------
    # 2. Linear projections (FP16, bias=False)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(fp16)
    right_proj_w = weights["right_proj.weight"].to(device).to(fp16)
    left_gate_w = weights["left_gate.weight"].to(device).to(fp16)
    right_gate_w = weights["right_gate.weight"].to(device).to(fp16)
    out_gate_w = weights["out_gate.weight"].to(device).to(fp16)

    left_proj = F.linear(x_fp16, left_proj_w)      # (B, N, N, hidden)
    right_proj = F.linear(x_fp16, right_proj_w)   # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(device).to(fp16).unsqueeze(-1)   # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))

    left = left_proj * left_gate     # (B, N, N, hidden)
    right = right_proj * right_gate  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 5. Batched GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as left @ rightᵀ using a Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # (B, hidden, N, N) where the last two axes are (i,k) for left and (k,j) for right
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # transpose the last two dims

    batch = B * hidden_dim

    # Collapse the (B, hidden) dimensions → (batch, N, N)
    left_mat = left_perm.view(batch, N, N)
    right_mat = right_perm.view(batch, N, N)
    out_mat = torch.empty_like(left_mat)  # (batch, N, N), fp16

    # strides in elements
    stride_batch_a = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_batch_b = right_mat.stride(0)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)

    stride_batch_c = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batch_matmul_fp16_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_batch_a=stride_batch_a,
        stride_batch_b=stride_batch_b,
        stride_batch_c=stride_batch_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore the original 4‑D layout: (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. to_out_norm (LayerNorm over hidden dimension)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(fp16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    ).contiguous()   # (B, N, N, hidden), fp16

    # ------------------------------------------------------------------
    # 7. out_gate multiplication (fused via Triton)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # (B, N, N, hidden)

    total_el = out.numel()
    BLOCK = 1024
    grid_gate = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid_gate](
        out,
        out_gate,
        total_el,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 8. Final projection back to original dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(fp16)
    out = F.linear(out, to_out_w)   # (B, N, N, dim), fp16

    # Return as float32 as required by the interface
    return out.to(fp32)
