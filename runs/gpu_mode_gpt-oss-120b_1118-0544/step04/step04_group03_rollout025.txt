@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    batch = tl.program_id(1)

    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch_ptr = a_ptr + batch * stride_ab
    b_batch_ptr = b_ptr + batch * stride_bb
    c_batch_ptr = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        a = tl.load(
            a_batch_ptr + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_batch_ptr + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    c = acc.to(tl.float16)
    tl.store(
        c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )
```

def custom_kernel(data):
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1 LayerNorm on dim
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # 2 Linear projections + gates
    # Apply in original dtype (float32) but convert later to fp16 for heavy compute
    left_proj = F.linear(x, weights["left_proj.weight"])
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_proj = F.linear(x, weights["right_proj.weight"])
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))

    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Convert to half for GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape  # B, N, N, H

    # permute to [B, H, N, N] and ensure contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch*H dimension
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)  # shape [B*H, N, N], dtype fp16

    # strides
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    # for transposed load: swap k and n strides
    stride_b_k = right_flat.stride(2)   # original col stride = 1
    stride_b_n = right_flat.stride(1)   # original row stride = N

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # tile size
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _batched_gemm_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B,N,N,H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 3 Hidden dim LayerNorm
    out = F.layer_norm(out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # 4 Output gating
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))
    out = out * out_gate

    # 5 Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])
    return out
```

def custom_kernel(data):
    """...
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # keep original dtype
    orig_dtype = x.dtype

    # 1. Input LayerNorm over dim
    x = F.layer_norm(x, (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # 2. Projections & gates (keep in original dtype)
    # left projection and gate
    left_proj = F.linear(x, weights["left_proj.weight"])
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    # right projection and gate
    right_proj = F.linear(x, weights["right_proj.weight"])
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))

    # 3. Optional mask (broadcast over hidden dim)
    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # 4. Apply gates
    left = left_proj * left_gate      # [B,N,N,H]
    right = right_proj * right_gate

    # 5. Cast to half for heavy GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape  # H == hidden_dim

    # Permute to (B, H, N, N) and make contiguous for easier stride handling
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch*H dimension => (B*H, N, N)
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements)
    stride_a_batch = left_flat.stride(0)   # batch stride
    stride_a_m = left_flat.stride(1)       # row stride (M dimension)
    stride_a_k = left_flat.stride(2)       # column stride (K dimension)

    stride_b_batch = right_flat.stride(0)
    # For B^T we swap the k and n strides
    stride_b_k = right_flat.stride(2)      # original col stride = 1
    stride_b_n = right_flat.stride(1)      # original row stride = N

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Tile sizes – chosen to give good occupancy up to N=1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    @triton.jit
    def _batched_gemm_kernel(
        a_ptr, b_ptr, c_ptr,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ab, stride_bb, stride_cb,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        pid = tl.program_id(0)
        batch = tl.program_id(1)

        num_pid_n = tl.cdiv(N, BLOCK_N)
        pid_m = pid // num_pid_n
        pid_n = pid % num_pid_n

        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        offs_k = tl.arange(0, BLOCK_K)

        mask_m = offs_m < M
        mask_n = offs_n < N

        a_batch_ptr = a_ptr + batch * stride_ab
        b_batch_ptr = b_ptr + batch * stride_bb
        c_batch_ptr = c_ptr + batch * stride_cb

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k in range(0, tl.cdiv(K, BLOCK_K)):
            cur_k = k * BLOCK_K + offs_k
            mask_k = cur_k < K

            a = tl.load(
                a_batch_ptr + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
                mask=mask_m[:, None] & mask_k[None, :],
                other=0.0,
            )
            b = tl.load(
                b_batch_ptr + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
                mask=mask_k[:, None] & mask_n[None, :],
                other=0.0,
            )
            acc += tl.dot(a, b)

        c = acc.to(tl.float16)
        tl.store(
            c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
            c,
            mask=mask_m[:, None] & mask_n[:, None],
        )

    _batched_gemm_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Hidden‑dim LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(out.dtype),
        bias=weights["to_out_norm.bias"].to(out.dtype),
    )

    # 7. Output gate
    # compute gate in half precision
    x_fp16 = x.to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)))
    out = out * out_gate

    # 8. Final linear back to original dim
    out = F.linear(out, weights["to_out.weight"].to(out.dtype))

    # Cast back to original dtype of the input
    out = out.to(orig_dtype)
    return out

def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

    The implementation follows the reference PyTorch version but fuses the
    O(N³·H) reduction with a custom batched GEMM kernel (left @ rightᵀ) written
    in Triton.  Linear projections and gates are performed in PyTorch, the
    heavy matrix multiplication is done in half‑precision with float32
    accumulation, and the remaining LayerNorm / output‑gate steps are
    executed with PyTorch.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    # ------------------------------------------------------------------ #
    # unpack inputs
    # ------------------------------------------------------------------ #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # remember original dtype for final cast
    orig_dtype = x.dtype

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm over channel dimension `dim`
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------ #
    # 2. Linear projections + gates (still in input dtype)
    # ------------------------------------------------------------------ #
    left_proj = F.linear(x, weights["left_proj.weight"])
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))

    right_proj = F.linear(x, weights["right_proj.weight"])
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))

    # ------------------------------------------------------------------ #
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------ #
    # 4. Apply gates
    # ------------------------------------------------------------------ #
    left = left_proj * left_gate                         # [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------ #
    # 5. Heavy O(N³·H) reduction: batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------ #
    # Cast to fp16 for the GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape  # H == hidden_dim

    # reshape to (B, H, N, N) and flatten batch*H dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_perm.view(B * H, N, N)            # (batch, M, K)
    right_flat = right_perm.view(B * H, N, N)          # (batch, K, N) – we will use it transposed

    out_flat = torch.empty_like(left_flat)            # (batch, M, N)

    # strides (in elements)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    # we need Bᵀ → swap k and n strides
    stride_b_k = right_flat.stride(2)   # original column stride  (1)
    stride_b_n = right_flat.stride(1)   # original row stride     (N)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # tile sizes – good trade‑off for N up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    @triton.jit
    def _batched_gemm_kernel(
        a_ptr, b_ptr, c_ptr,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ab, stride_bb, stride_cb,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        """C = A @ Bᵀ  (batched, fp16 inputs, fp32 accumulation)."""
        pid = tl.program_id(0)          # tile index within a batch
        batch = tl.program_id(1)        # flattened batch*hidden index

        num_pid_n = tl.cdiv(N, BLOCK_N)
        pid_m = pid // num_pid_n
        pid_n = pid % num_pid_n

        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        offs_k = tl.arange(0, BLOCK_K)

        mask_m = offs_m < M
        mask_n = offs_n < N

        # base pointers for this batch
        a_batch = a_ptr + batch * stride_ab
        b_batch = b_ptr + batch * stride_bb
        c_batch = c_ptr + batch * stride_cb

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k in range(0, tl.cdiv(K, BLOCK_K)):
            cur_k = k * BLOCK_K + offs_k
            mask_k = cur_k < K

            a = tl.load(
                a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
                mask=mask_m[:, None] & mask_k[None, :],
                other=0.0,
            )
            b = tl.load(
                b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
                mask=mask_k[:, None] & mask_n[None, :],
                other=0.0,
            )
            acc += tl.dot(a, b)   # fp16·fp16 → fp32

        c = acc.to(tl.float16)
        tl.store(
            c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
            c,
            mask=mask_m[:, None] & mask_n[None, :],
        )

    # launch the kernel
    _batched_gemm_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_a_m, stride_a_k,
        stride_b_k, stride_b_n,
        stride_c_m, stride_c_n,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 6. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(out.dtype),
        bias=weights["to_out_norm.bias"].to(out.dtype),
    )

    # ------------------------------------------------------------------ #
    # 7. Output gate (element‑wise)
    # ------------------------------------------------------------------ #
    x_fp16 = x.to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)))
    out = out * out_gate

    # ------------------------------------------------------------------ #
    # 8. Final linear projection back to original channel dimension
    # ------------------------------------------------------------------ #
    out = F.linear(out, weights["to_out.weight"].to(out.dtype))

    # cast to the dtype of the original input
    out = out.to(orig_dtype)
    return out
