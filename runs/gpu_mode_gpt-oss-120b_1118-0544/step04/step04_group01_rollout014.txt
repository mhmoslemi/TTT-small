"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate, batched GEMM for O(N³) reduction.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C) using provided weights.
2. Cast to fp16 for all heavy computation.
3. Four linear projections (left/right & their gates) are performed in fp16.
4. If a mask is supplied, a Triton kernel fuses ``proj * gate * mask``.
   Otherwise a simple point‑wise multiplication is used.
5. Core TriMul reduction:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is equivalent to a batch of hidden‑dim independent matmuls:
   left_d @ right_dᵀ.
   We reshape (B,N,N,H) → (B*H,N,N) and use a single
   torch.bmm (highly‑optimized cuBLAS) to compute all H matmuls
   simultaneously.
6. Apply a second LayerNorm, multiply by the output gate and a final
   linear projection back to the original channel dimension.
7. Cast the result back to the original dtype (float32).

Only the mask‑gate fusion is written in Triton; all matrix‑heavy work
leverages cuBLAS via torch.bmm, yielding a large speed‑up over the
reference torch.einsum implementation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total elements in proj/gate (B*N*N*H)
    hidden_dim,        # i32 – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # each hidden‑dim element shares the same mask, so compute the mask index
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0).to(tl.float16)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0).to(tl.float16)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0).to(tl.float16)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 2048) -> torch.Tensor:
    """
    Fuse ``proj * gate * mask`` using a Triton kernel.
    All tensors must be on the same CUDA device, contiguous and fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],               # hidden_dim (H)
        BLOCK_SIZE=block_size,
    )
    return out

                              torch.Tensor,

        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ----------------------------------------------------------------------
    # 7. Core TriMul reduction using batched GEMM (fast cuBLAS path)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    ↔ for each d: out_d = left_d @ right_dᵀ
    # ----------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat  = left.reshape(B * H, N, N)
    right_flat = right.reshape(B * H, N, N)

    # Batch matrix multiplication: (B*H, N, N) @ (B*H, N, N)ᵀ -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------------
    # 8. Second LayerNorm over hidden dimension H
    # ----------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------------------------------------------------------
    # 9. Apply output gate and final linear projection back to C
    # ----------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ----------------------------------------------------------------------
    # 10. Cast back to original dtype (float32) and return
    # ----------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)
    return out
