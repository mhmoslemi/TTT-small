"""
TriMul (outgoing) forward pass – Triton‑accelerated projection+gate+mask fusion.

Algorithm
---------
1. Layer‑norm the input over the channel dimension `dim`.
2. Cast to FP16 for the heavy compute.
3. Fuse each projection (left/right) with its gate:
   - Compute a single linear layer of size 2·hidden_dim, producing
     `[proj , gate_raw]` concatenated on the hidden axis.
   - A Triton kernel reads the concatenated tensor, applies the sigmoid
     to the gate part and multiplies by the optional mask, writing the
     final tensor of shape (B, N, N, hidden_dim).
4. Compute the output‑gate in the usual way (linear + sigmoid).
5. Core reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   performed with torch.einsum (batched matrix multiplication over the
   sequence dimension).
6. Apply the output LayerNorm, multiply by the output‑gate, and project
   back to the original channel dimension.
7. Cast back to the input dtype.

The Triton kernel is the only non‑PyTorch component and fuses three
operations (projection, gating, masking) into a single pass, reducing
memory traffic and improving overall runtime.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _proj_gate_mask_kernel(
    comb_ptr,          # fp16 * (B*N*N*2*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_out,         # scalar, total number of output elements = B*N*N*H
    hidden_dim,        # H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)                # [0, BLOCK_SIZE)

    valid = offs < total_out

    # linear index → (base, h) where base = b*i*j index, h = hidden offset
    base = offs // hidden_dim
    h = offs % hidden_dim

    # offsets inside the concatenated tensor (proj | gate_raw)
    proj_off = base * (2 * hidden_dim) + h
    gate_off = proj_off + hidden_dim

    proj = tl.load(comb_ptr + proj_off,
                   mask=valid, other=tl.float16(0.0))
    gate_raw = tl.load(comb_ptr + gate_off,
                       mask=valid, other=tl.float16(0.0))
    mask_val = tl.load(mask_ptr + base,
                       mask=valid, other=tl.float16(1.0))

    # sigmoid
    gate = 1.0 / (1.0 + tl.exp(-gate_raw))

    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C]
            - mask         : torch.Tensor of shape [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict containing:
                * "dim" (int)        – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim), dtype = dtype_in

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 4. Prepare mask (fp16). If no mask or nomask, use an all‑ones mask.
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
    else:
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 5. Left projection + gate (fused)
    # ------------------------------------------------------------------
    left_comb_weight = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*H, dim)
    left_comb = torch.nn.functional.linear(x, left_comb_weight)     # (B, N, N, 2*H)

    left = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
    total_out = left.numel()
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total_out, BLOCK_SIZE),)

    _proj_gate_mask_kernel[grid](
        left_comb,
        mask_fp16,
        left,
        total_out,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    del left_comb  # free memory

    # ------------------------------------------------------------------
    # 6. Right projection + gate (fused)
    # ------------------------------------------------------------------
    right_comb_weight = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)
    right_comb = torch.nn.functional.linear(x, right_comb_weight)      # (B, N, N, 2*H)

    right = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
    total_out = right.numel()
    grid = (triton.cdiv(total_out, BLOCK_SIZE),)

    _proj_gate_mask_kernel[grid](
        right_comb,
        mask_fp16,
        right,
        total_out,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    del right_comb  # free memory

    # ------------------------------------------------------------------
    # 7. Output gate (separate linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8. Core reduction (batched matrix multiplication over the sequence dim)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 9. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 10. Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
