):
    pid = tl.program_id(0)          # tile id over (M,N)
    batch = tl.program_id(1)        # batch dimension (B*H)

    # compute tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out-of-bound
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # load B transposed: original B is (batch, N, K)
        # we need B_T of shape (K, N)
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        acc += tl.dot(a, b)   # FP16*FP16 -> FP32 accumulation

    # write back result as FP16
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[:, None],
    )

def _batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication where each batch computes:
        out[b] = left[b] @ right[b].T
    left, right: shape (B*H, N, N) float16
    Returns: out of same shape.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B_H, N, N2 = left.shape
    assert N == N2
    out = torch.empty_like(left)

    # strides
    stride_a_batch = left.stride(0)
    stride_am = left.stride(1)
    stride_ak = left.stride(2)

    stride_b_batch = right.stride(0)
    # for B_T: k dimension is dim 2, n dimension is dim 1
    stride_bk = right.stride(2)
    stride_bn = right.stride(1)

    stride_c_batch = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B_H)

    _tri_mul_gemm[grid](
        left,
        right,
        out,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------ #

    )
    right_gate = torch.sigmoid(

    if mask is not None and not nomask:

        right_proj = right_proj * mask_fp16

    # 4. Apply gates
    # ------------------------------------------------------------------ #
    left = left_proj * left_gate          # [B, N, N, H]
    right = right_proj * right_gate

    # Free intermediate tensors that are no longer needed
    del left_proj, right_proj, left_gate, right_gate

    # 5. Core O(N³·H) reduction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------ #
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for the GEMM

    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    out = F.layer_norm(

        weight=weights["to_out_norm.weight"],
