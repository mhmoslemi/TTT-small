@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems,
    hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * m
        right = right * right_gate * m
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)
```

    3. Fuse gate multiplication and optional mask with a Triton kernel.

    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)

"""
Triton‑accelerated outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the input pair representation.
2. Project the normalized tensor to the hidden dimension H with three
   linear layers (left_proj, right_proj, gate_proj) in FP16.
3. Compute element‑wise sigmoid gates (left_gate, right_gate, out_gate).
4. Fuse gate multiplication and optional mask with a tiny Triton kernel:
   left  = left  * left_gate  * mask
   right = right * right_gate * mask
5. Perform the triangular multiplicative contraction
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a single `torch.einsum` (effectively a batched GEMM).
6. Apply a second LayerNorm on the hidden dimension, multiply by `out_gate`,
   and project back to the original channel dimension C.
7. Cast the result back to the input dtype.

The kernel only handles the element‑wise masking/gating; the heavy
O(N³·H) reduction is performed by cuBLAS via `torch.einsum`, which is
highly optimized on B200 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr,               # fp16 pointer to left  [B,N,N,H] flattened
    right_ptr,              # fp16 pointer to right [B,N,N,H] flattened
    left_gate_ptr,          # fp16 pointer to left_gate  same shape
    right_gate_ptr,         # fp16 pointer to right_gate same shape
    mask_ptr,               # fp16 pointer to mask [B,N,N] flattened
    total_elems,            # total number of elements in left/right
    hidden_dim,             # H
    HAS_MASK: tl.constexpr, # compile‑time bool
    BLOCK_SIZE: tl.constexpr # compile‑time int
):
    """Fuse mask, left‑gate and right‑gate into left/right in‑place."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden_dim entries correspond to the same mask entry
        mask_idx = offs // hidden_dim          # integer division
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * m
        right = right * right_gate * m
    else:
        left = left * left_gate
        right = right * right_gate

    # write back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) in the same dtype as `input_tensor`.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------ Input LayerNorm ------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # cast to fp16 for the compute‑heavy part
    x = x.to(torch.float16)

    # ------------------ Load linear weights (fp16) ------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------ Projections ------------------
    left = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------ Gates (sigmoid) ------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ------------------ Fuse mask & gates (Triton) ------------------
    if mask is not None and not nomask:
        # mask shape (B, N, N) -> fp16 broadcast over hidden dim
        mask_fp = mask.to(device).to(torch.float16)

        # flatten everything for the kernel
        total = left.numel()                     # B * N * N * H
        left_flat      = left.contiguous().view(-1)
        right_flat     = right.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat      = mask_fp.contiguous().view(-1)   # (B*N*N,)

        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back to original layout
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # no mask → simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------ Triangular multiplicative contraction ------------------
    # out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)   # (B, N, N, H)

    # ------------------ Output LayerNorm ------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ------------------ Output gate ------------------
    out = out * out_gate

    # ------------------ Final projection back to C ------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
