def _gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, total_elems, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # compute mask index for broadcasting
    mask_idx = offs // hidden_dim
    # load
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask)

"""
TriMul (outgoing) forward pass – Triton‑fused mask‑gate multiplication.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C).
2. Cast to fp16 for the compute‑heavy part.
3. Linear projections (left/right) and their gates are computed with fp16
   weights.  The mask (if present) and gates are fused in a single Triton
   kernel:   out = proj * gate * mask   (broadcast mask over hidden dim).
4. Core O(N³·H) reduction:
   – reshape left to (B*H, N, N)   (i,k)
   – reshape right to (B*H, N, N) with the (j,k) axes swapped → (k,j)
   – batched matrix multiplication (torch.bmm) performs the sum over k
     for every hidden channel independently.
5. Second LayerNorm on the hidden dimension, multiplication by the output
   gate, and a final linear projection back to the original channel size.
6. Cast back to the input dtype (usually float32) and return.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask (broadcast mask over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16*  (B*N*N*H,)
    gate_ptr,          # fp16*  (B*N*N*H,)
    mask_ptr,          # fp16*  (B*N*N,)
    out_ptr,           # fp16*  (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – size of hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    off = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = off < total_elems

    # index into mask (broadcast over hidden dim)
    mask_idx = off // hidden_dim

    proj = tl.load(proj_ptr + off, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + off, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * mask_val
    tl.store(out_ptr + off, out, mask=mask)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` with a Triton kernel.
    All tensors must be fp16, contiguous and on the same CUDA device.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],               # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input tensor [B, N, N, C]
        torch.Tensor,                     # mask tensor [B, N, N] (or None)
        Dict[str, torch.Tensor],          # weights dict
        Dict[str, Any],                   # config dict
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Cast to fp16 for the compute‑heavy path
    x = x.to(torch.float16)

    # ----------------------- cast weights --------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections ---------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------- gates -----------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ----------------------- mask & gate fusion -------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ----------------- core O(N³·H) reduction (batched GEMM) ------------
    B, N, _, _ = left.shape

    # reshape to (B*H, N, N) for batched matrix multiplication
    left_bmm  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B*H, I, K)
    right_bmm = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)  # (B*H, K, J)

    out_bmm = torch.bmm(left_bmm, right_bmm)                              # (B*H, I, J)

    out = out_bmm.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)           # (B, I, J, H)

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------ cast back --------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
