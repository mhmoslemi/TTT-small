"""
TriMul (outgoing) forward pass – Triton‑fused mask+gate and batched GEMM.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C).
2. Cast to fp16 for the heavy part.
3. Linear projections (left/right) and their gates are computed with
   torch.nn.functional.linear.  If a mask is present, a small Triton kernel
   fuses `proj * gate * mask`.
4. The core O(N³·H) reduction
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is implemented as a batched matrix‑multiplication kernel
   (batch = B·H) that computes many independent GEMMs in a single
   Triton launch.
5. Second layer‑norm on the hidden dimension, multiplication by the
   output gate, and a final linear projection back to the original
   channel dimension.
6. Cast the result back to the input dtype.

The implementation keeps the heavy compute on the GPU with fp16/Tensor‑Core
operations while preserving full numerical correctness.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse proj * gate * mask (mask is broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total_elems,)
    gate_ptr,          # fp16 * (total_elems,)
    mask_ptr,          # fp16 * (mask_elems,)   (broadcast over hidden dim)
    out_ptr,           # fp16 * (total_elems,)
    total_elems,       # i32   – total number of elements in proj/gate
    hidden_dim,        # i32   – H (used for mask broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # the mask is indexed per (b,i,k) element, i.e. one entry per hidden‑dim block
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """Fuse `proj * gate * mask` using a Triton kernel.
    All tensors must be fp16, contiguous and on the same CUDA device."""
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for tensors of shape (batch, M, K) @ (batch, K, N)
# ----------------------------------------------------------------------
@triton.jit
def _batch_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)        # which batch (i.e. B*H)
    pid_m = tl.program_id(1)           # row tile
    pid_n = tl.program_id(2)           # column tile

    # ==== tile offsets ====
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds rows / cols
    m_mask = offs_m < M
    n_mask = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ==== loop over K dimension ====
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # mask for k
        k_mask = offs_k < K

        # pointers
        a_offset = batch_id * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_offset = batch_id * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        # load tiles
        a = tl.load(a_ptr + a_offset,
                    mask=(m_mask[:, None] & k_mask[None, :]), other=0.0).to(tl.float16)
        b = tl.load(b_ptr + b_offset,
                    mask=(k_mask[:, None] & n_mask[None, :]), other=0.0).to(tl.float16)

        # fused dot‑product (fp16 -> fp32 accumulation)
        acc += tl.dot(a, b)

    # ==== write back ====
    c_offset = batch_id * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptr + c_offset,
             acc.to(tl.float16),
             mask=(m_mask[:, None] & n_mask[None, :]))

# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict containing at least "dim" and "hidden_dim";
                      optional "nomask" flag.

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input layer‑norm
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # cast to fp16 for the compute‑heavy part
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Cast weights to fp16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections
    # --------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 4. Gates
    # --------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 5. Fuse mask + gate (if needed)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6. Core O(N³·H) reduction via batched GEMM (Triton)
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # reshape to (batch, M, K) and (batch, K, N) where batch = B*H
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)  i,k
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)  k,j

    batch = B * H
    left_mat  = left_perm.view(batch, N, N)   # (batch, M, K)
    right_mat = right_perm.view(batch, N, N)  # (batch, K, N)
    out_mat   = torch.empty((batch, N, N), dtype=torch.float16, device=device)

    # strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    # block sizes – chosen to fit typical Tensor‑Core tiles
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _batch_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch,
        N,   # M
        N,   # N
        N,   # K
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Second LayerNorm, output gate and final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 8. Cast back to original dtype
    # --------------------------------------------------------------
    out = out.to(dtype_in)

    return out
