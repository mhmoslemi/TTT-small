def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Cast to half
    x = x.to(torch.float16)

    # linear transformations
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"].to(x.dtype).to(x.device))
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"].to(x.dtype).to(x.device))

    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"].to(x.dtype).to(x.device)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(x.dtype).to(x.device)).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"].to(x.dtype).to(x.device)).sigmoid()

    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape B,N,N,1
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    left = left_proj * left_gate
    right = right_proj * right_gate

    B, N, _, H = left.shape

    # reshape for batched matmul
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H,N,N)
    # compute out
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    out = _elemwise_mul(out, out_gate)

    # final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(out.dtype).to(out.device))

    out = out.to(x.dtype) # but x is half, we need original dtype= (maybe float32). Use dtype_orig = data[0].dtype.

def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)
```

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm:
1. Layer‑norm the input tensor.
2. Cast to float16 for the compute‑heavy part.
3. Linear projections for left/right pathways and their gates (sigmoid).
4. Optional mask application.
5. Fuse gate into projection: left = left_proj * left_gate, right = right_proj * right_gate.
6. Reshape to (B*H, N, N) and perform a batched matrix multiplication
   `out_flat = left_flat @ right_flat^T` using torch.bmm (cuBLAS).
7. Reshape back to (B, N, N, H) and apply a second LayerNorm.
8. Multiply with an output gate (element‑wise) using a tiny Triton kernel.
9. Final linear projection back to the original channel dimension.
10. Cast result to the original input dtype.

At least one operation (element‑wise multiplication with the output gate) is
implemented in Triton to satisfy the requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton element‑wise multiplication kernel (used for the output gate)
# --------------------------------------------------------------------------- #
@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Element‑wise multiplication using Triton."""
    assert a.shape == b.shape, "Shapes must match for element‑wise mul."
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 65536  # 64K elements per program; adjust as needed.
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out

# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # --------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    orig_dtype = x.dtype
    device = x.device

    # --------------------------------------------------------------- #
    # 1. Input LayerNorm (float32) then cast to float16
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------- #
    # 2. Cast all linear weights to float16 on the correct device
    # --------------------------------------------------------------- #
    w_left_proj = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    w_right_proj = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    w_left_gate = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    w_right_gate = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    w_out_gate = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    w_to_out = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------- #
    # 3. Linear projections + gates
    # --------------------------------------------------------------- #
    left_proj = F.linear(x, w_left_proj)                # [B, N, N, H]
    right_proj = F.linear(x, w_right_proj)              # [B, N, N, H]

    left_gate = F.linear(x, w_left_gate).sigmoid()      # [B, N, N, H]
    right_gate = F.linear(x, w_right_gate).sigmoid()    # [B, N, N, H]
    out_gate = F.linear(x, w_out_gate).sigmoid()        # [B, N, N, H]

    # --------------------------------------------------------------- #
    # 4. Optional masking (broadcast over hidden dim)
    # --------------------------------------------------------------- #
    if mask is not None and not nomask:
        mask_f = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------------------- #
    # 5. Apply gates
    # --------------------------------------------------------------- #
    left = left_proj * left_gate        # [B, N, N, H]
    right = right_proj * right_gate     # [B, N, N, H]

    # Free intermediate tensors early to reduce peak memory
    del left_proj, right_proj, left_gate, right_gate
    torch.cuda.empty_cache()

    B, N, _, H = left.shape

    # --------------------------------------------------------------- #
    # 6. Core O(N³·H) reduction via batched GEMM (cuBLAS)
    #    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------- #
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T → (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Clean up large temporaries
    del left_flat, right_flat, out_flat, left, right
    torch.cuda.empty_cache()

    # --------------------------------------------------------------- #
    # 7. Hidden‑dim LayerNorm
    # --------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )

    # --------------------------------------------------------------- #
    # 8. Apply output gate (element‑wise) using Triton
    # --------------------------------------------------------------- #
    out = _elemwise_mul(out, out_gate)

    # --------------------------------------------------------------- #
    # 9. Final linear projection back to original channel dimension
    # --------------------------------------------------------------- #
    out = F.linear(out, w_to_out)  # bias=False by design

    # --------------------------------------------------------------- #
    # 10. Cast back to original input dtype (typically float32)
    # --------------------------------------------------------------- #
    out = out.to(orig_dtype)

    return out
