@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_i, stride_out_j,
    B, N, H,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr):
    pid_g = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute batch and hidden indices
    batch_idx = pid_g // H
    hidden_idx = pid_g % H

    # Base pointers for this (batch, hidden) slice
    left_base = left_ptr + batch_idx * stride_left_batch + hidden_idx * tl.constexpr(1)  # hidden stride=1
    right_base = right_ptr + batch_idx * stride_right_batch + hidden_idx * tl.constexpr(1)
    out_base = out_ptr + batch_idx * stride_out_batch + hidden_idx * tl.constexpr(1)

    # Compute start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets within block
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # Masks for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M x K)
        left = tl.load(
            left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load right tile transposed (K x N)
        right = tl.load(
            right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_n,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # Convert to fp32 for accumulation
        left = left.to(tl.float32)
        right = right.to(tl.float32)

        acc += tl.dot(left, right)

    # Cast accumulator to fp16 for store
    out = acc.to(tl.float16)

    # Store result
    tl.store(
        out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j,
        out,
        mask=mask_i[:, None] & mask_j[None, :],
    )

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

The implementation follows the reference PyTorch module but moves the
O(N³·H) reduction (the einsum) into a fused Triton kernel.  All
linear projections, layer‑norms and gates are kept in fp16 to exploit
tensor cores, while the final output is cast back to the input dtype.
The kernel computes, for every batch b and hidden channel h:

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

without any intermediate materialisation of transposed tensors.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel for the core einsum
# ----------------------------------------------------------------------------- #
@triton.jit
def _trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    # Strides for the left tensor (shape [B,N,N,H])
    stride_left_batch, stride_left_i, stride_left_k,
    # Strides for the right tensor (shape [B,N,N,H]) – note the transposed view
    stride_right_batch, stride_right_k, stride_right_n,
    # Strides for the output tensor (shape [B,N,N,H])
    stride_out_batch, stride_out_i, stride_out_j,
    # Compile‑time constants
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out = Σ_k left[..., i, k, h] * right[..., j, k, h]

    Grid dimensions:
        axis 0 : (batch * hidden)   → pid_g
        axis 1 : block row (i)      → pid_i
        axis 2 : block col (j)      → pid_j
    """
    pid_g = tl.program_id(0)   # combined batch & hidden index
    pid_i = tl.program_id(1)   # block row index
    pid_j = tl.program_id(2)   # block col index

    # -------------------------------------------------
    # Decode batch and hidden indices from pid_g
    # -------------------------------------------------
    batch_idx = pid_g // H
    hidden_idx = pid_g % H

    # Base pointers for this (batch, hidden) slice
    left_base = left_ptr + batch_idx * stride_left_batch + hidden_idx
    right_base = right_ptr + batch_idx * stride_right_batch + hidden_idx
    out_base = out_ptr + batch_idx * stride_out_batch + hidden_idx

    # -------------------------------------------------
    # Offsets for the current tile
    # -------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Main K‑loop (k dimension is summed over)
    # -------------------------------------------------
    for k in range(0, N, BLOCK_K):
        k_start = k
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k]  → shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_base
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right[j, k] transposed to (k, j) → shape (BLOCK_K, BLOCK_N)
        right = tl.load(
            right_base
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_n,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # Convert to fp32 and accumulate
        acc += tl.dot(left.to(tl.float32), right.to(tl.float32))

    # -------------------------------------------------
    # Store the result (cast back to fp16)
    # -------------------------------------------------
    out = acc.to(tl.float16)
    tl.store(
        out_base
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j,
        out,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------------- #
# Python entry point
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C) (float32 or float16)
            - mask        : torch.Tensor of shape (B, N, N) or None
            - weights     : dict of model parameters (float32)
            - config      : dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------- LayerNorm on the input ------------------- #
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for the heavy part
    x = x.to(torch.float16)

    # ------------------- Load linear weights (fp16) ---------------- #
    def to_fp16(t): return t.to(device).to(torch.float16)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w      = to_fp16(weights["to_out.weight"])

    # ------------------- Linear projections ------------------------ #
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # ------------------- Gates (sigmoid) -------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ------------------- Apply mask & gates ----------------------- #
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------- Core O(N³·H) reduction (Triton) -------- #
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # (B,N,N,H), fp16

    # Strides of the tensors (in elements)
    # left / right / out have layout (B, N, N, H) → (batch, i, k, h)
    stride_left = left.stride()   # (B*N*N*H, N*H, H, 1)
    stride_right = right.stride()
    stride_out = out.stride()

    # Strides for the transposed view of right (k as row, j as column)
    # row‑stride = original k‑stride, column‑stride = original i‑stride
    stride_right_batch = stride_right[0]
    stride_right_k = stride_right[2]          # original k‑stride (H)
    stride_right_n = stride_right[1]          # original i‑stride (N*H)

    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _trmul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        stride_left_batch=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_right_batch=stride_right_batch,
        stride_right_k=stride_right_k,
        stride_right_n=stride_right_n,
        stride_out_batch=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        B=B,
        N=N,
        H=H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------- Output LayerNorm ------------------------- #
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ------------------- Apply output gate ------------------------ #
    out = out * out_gate

    # ------------------- Final linear projection ----------------- #
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to the original dtype if needed
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
