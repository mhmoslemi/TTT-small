"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑normalize the 4‑D input (B,N,N,C) using the provided weight/bias.
2. Cast to fp16 and compute four linear projections:
   * left_proj, right_proj  – hidden dimension H
   * left_gate, right_gate, out_gate – same shape, then sigmoid.
3. Fuse (proj * gate * mask) with a tiny Triton kernel when a mask is present.
   The kernel broadcasts the mask over the hidden dimension.
4. Core O(N³·H) reduction:
   * Reshape left/right to (B·H, N, N) and perform a batched GEMM
     `out = left @ rightᵀ` via `torch.bmm`.  This maps to cuBLAS/Tensor‑core
     code and is far faster than `torch.einsum`.
5. Layer‑norm over the hidden dimension, multiply by the output gate,
   and apply the final linear projection back to the original channel size.
6. Cast the result back to the input dtype (typically float32).

Only the mask‑gate fusion is implemented in Triton; the heavy reduction uses
the highly‑optimized PyTorch batched matrix‑multiply kernel.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and optional mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # Compute the corresponding mask index (broadcast over H)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using the Triton kernel above.
    All tensors must be contiguous and on the same CUDA device, dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,               # input tensor [B, N, N, C]
        torch.Tensor,               # mask tensor  [B, N, N] (or None)
        Dict[str, torch.Tensor],    # model weights (float32)
        Dict[str, Any],             # config dict
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict containing model parameters (float32)
            - config: dict with keys:
                  * "dim" (int) – input channel dimension C
                  * "hidden_dim" (int) – hidden dimension H
                  * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------- LayerNorm (input) -------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for compute‑heavy path
    x = x.to(torch.float16)

    # ----------------------- cast weights --------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections ---------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------- gates -----------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ----------------------- mask & gate fusion -------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------- core O(N³·H) reduction -----------------------
    # Reshape to (B*H, N, N) and perform batched matmul left @ rightᵀ
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_3d  = left_perm.view(B * H, N, N)               # (B*H, N, N)
    right_3d = right_perm.view(B * H, N, N)              # (B*H, N, N)

    out_3d = torch.bmm(left_3d, right_3d.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_3d.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- hidden‑dim LayerNorm & output gate -------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------ cast back --------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
