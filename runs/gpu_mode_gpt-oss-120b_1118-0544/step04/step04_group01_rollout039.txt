@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_abh, stride_ak, stride_ab,
    stride_bbh, stride_bk, stride_bb,
    stride_cbh, stride_ch, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # 1D grid for all blocks
    # Compute batch index, block row, block col
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_id = pid // (num_blocks_m * num_blocks_n)
    pid = pid % (num_blocks_m * num_blocks_n)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    # Offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Compute mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A
        a_ptrs = a_ptr + batch_id * stride_abh + (offs_m[:, None] * stride_ak) + (offs_k[None, :] * stride_ab)
        b_ptrs = b_ptr + batch_id * stride_bbh + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bb)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Convert to fp16 then compute
        acc += tl.dot(a, b)  # dot expects (BLOCK_M, BLOCK_K) and (BLOCK_K, BLOCK_N) both fp16

    # Write back
    c_ptrs = c_ptr + batch_id * stride_cbh + (offs_m[:, None] * stride_ch) + (offs_n[None, :] * stride_cb)
    # convert acc to fp16 maybe, but keep fp32? We'll store as fp16; conversion done in store.
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

    # a: (B*H, N, N) fp16 contiguous?
    # b: (B*H, N, N) fp16 contiguous? maybe not contiguous but view. Let's ensure they are contiguous.

        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate and a custom
batched GEMM kernel for the O(N³·H) contraction.

Algorithm
---------
1. Layer‑normalize the input (float32) and cast to fp16.
2. Four linear projections (left/right and their gates) are performed in fp16.
3. If a mask is present and not disabled, a Triton kernel fuses
   `proj * gate * mask` element‑wise.
4. The core contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is equivalent to a batched matrix multiplication
        left[b,d] @ right[b,d]ᵀ
   for each hidden dimension `d`.  A custom Triton kernel performs this
   batched GEMM efficiently (FP16 compute, FP32 accumulation).
5. The hidden result is Layer‑normed, multiplied by the output gate,
   and projected back to the original channel dimension.
6. The final tensor is cast back to the input dtype.

Only the mask‑gate fusion and the batched GEMM are implemented in Triton;
all other ops use PyTorch (which already leverages cuBLAS/Tensor‑cores).
"""

import math
from typing import Tuple, Dict, Any

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32   – total number of elements in proj/gate
    hidden_dim,        # i32   – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask_valid = offs < total_elems

    # broadcast mask over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask_valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask_valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=mask_valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using a Triton kernel.
    All tensors must be on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B) where A and B are
# (batch, M, K) and (batch, K, N) respectively.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_abh, stride_ak, stride_ab,   # A strides: (batch, row, col)
    stride_bbh, stride_bk, stride_bb,   # B strides: (batch, row(K), col(N))
    stride_cbh, stride_ch, stride_cb,   # C strides: (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # Compute indices of the block
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch_id = pid // blocks_per_batch
    pid_in_batch = pid % blocks_per_batch
    block_m = pid_in_batch // num_blocks_n
    block_n = pid_in_batch % num_blocks_n

    # Offsets for rows and cols within the block
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for A and B
        a_ptrs = a_ptr + batch_id * stride_abh \
                 + (offs_m[:, None] * stride_ak) \
                 + (offs_k[None, :] * stride_ab)

        b_ptrs = b_ptr + batch_id * stride_bbh \
                 + (offs_k[:, None] * stride_bk) \
                 + (offs_n[None, :] * stride_bb)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Both a and b are fp16; dot does FP16 mul -> FP32 accumulation
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + batch_id * stride_cbh \
             + (offs_m[:, None] * stride_ch) \
             + (offs_n[None, :] * stride_cb)

    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_fp16(
    a: torch.Tensor,
    b: torch.Tensor,
    block_m: int = 64,
    block_n: int = 64,
    block_k: int = 64,
) -> torch.Tensor:
    """
    Batched matrix multiplication (fp16 inputs, fp32 accumulation) using Triton.
    a: (batch, M, K)   fp16
    b: (batch, K, N)   fp16
    Returns c: (batch, M, N) fp16
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    # Strides in elements
    stride_abh, stride_ak, stride_ab = a.stride()
    stride_bbh, stride_bk, stride_bb = b.stride()
    stride_cbh, stride_ch, stride_cb = c.stride()

    num_blocks_m = (M + block_m - 1) // block_m
    num_blocks_n = (N + block_n - 1) // block_n
    total_blocks = batch * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        batch,
        M,
        N,
        K,
        stride_abh,
        stride_ak,
        stride_ab,
        stride_bbh,
        stride_bk,
        stride_bb,
        stride_cbh,
        stride_ch,
        stride_cb,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return c

    x = x.to(torch.float16)

    # ----------------------- cast weights --------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections ---------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------- gates -----------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ----------------------- mask & gate fusion -------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------- core O(N³·H) contraction ----------------------
    # Reshape to (batch*hidden, N, N) and compute batched GEMM with transpose
    B = left.shape[0]

    # Bring hidden dimension to the batch axis
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left.view(B * hidden_dim, left.shape[2], left.shape[3])   # (B*H, N, N)
    # Transpose the second operand on the last two dimensions
    right_T = right.permute(0, 1, 3, 2).contiguous().view(B * hidden_dim,
                                                         right.shape[2],
                                                         right.shape[3])  # (B*H, N, N)

    # Batched GEMM (fp16 compute, fp32 accumulate)
    out_flat = _batched_matmul_fp16(left_flat, right_T,
                                    block_m=64, block_n=64, block_k=64)  # (B*H, N, N)

    # Restore original layout (B, N, N, H)
    out = out_flat.view(B, hidden_dim, input_tensor.shape[1], input_tensor.shape[2]) \
                     .permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------- second LayerNorm & output gate -----------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------- final linear projection -----------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------ cast back --------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
