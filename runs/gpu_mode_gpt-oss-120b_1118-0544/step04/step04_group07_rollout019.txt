def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on input
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to half for compute
    x = x.to(torch.float16)

    # Load weights to half
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)

    # Linear projections
    left = F.linear(x, left_proj_w)  # [B,N,N,H]
    right = F.linear(x, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply mask and gates
    if mask is not None and not nomask:

        # Make it broadcastable: unsqueeze(-1) -> (B,N,N,1)

        # Flatten for kernel

        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_fp = mask_fp.contiguous()

        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat = mask_fp.view(-1)

            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,

            BLOCK_SIZE=BLOCK,

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input tensor.
2. Project the normalized tensor to a hidden dimension (left/right) and
   compute three gate tensors (left, right, out) via linear + sigmoid.
3. Fuse mask (if present) and the left/right gates in a small Triton kernel.
4. Perform the core O(N³·H) reduction:
   For each (batch, hidden) pair compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication
       left_mat @ right_matᵀ   with shape (B·H, N, N).
   This multiplication is implemented in pure Triton with fp16
   arithmetic and fp32 accumulation.
5. Apply a second layer‑norm, the output gate and a final linear
   projection back to the original dimension.
6. Cast the result back to the input dtype (float32).

Only the mask‑gate fusion and the batched matrix multiplication are
implemented in Triton; the remaining lightweight operations use
torch.nn.functional for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Mask + left/right gate fusion kernel (optional)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] flattened
    left_gate_ptr, right_gate_ptr,     # same shape
    mask_ptr,                          # [B,N,N] flattened
    total_elems, hidden_dim,           # runtime scalars
    HAS_MASK: tl.constexpr,            # compile‑time bool
    BLOCK_SIZE: tl.constexpr,          # compile‑time int
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each mask element is repeated `hidden_dim` times in the flattened view
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------
# 2) Batched matrix multiplication kernel (left @ right.T)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    stride_am, stride_ak,                    # A strides (row, col)
    stride_bk, stride_bn,                    # Bᵀ strides (k, n) – note swapped
    stride_cm, stride_cn,                    # C strides (row, col)
    M, N, K,                                 # matrix dimensions (rows, cols, inner)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # batch index (0 .. B*H-1)

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + pid_b * stride_batch_a
    b_batch = b_ptr + pid_b * stride_batch_b
    c_batch = c_ptr + pid_b * stride_batch_c

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load Bᵀ tile (K x N)  → original B is (N x K)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back C
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------

    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32)
            - config: dict containing:
                * "dim" (int) – input channels C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on input
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to half for the heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Load linear weights (half precision)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B, N, N, H)

    # ------------------------------------------------------------------
    # Apply mask and left/right gates (fusion kernel when mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure contiguous tensors
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask = mask.to(device).to(torch.float16).contiguous()

        # flatten
        left_flat      = left.view(-1)
        right_flat     = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_flat      = mask.view(-1)                # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³·H) reduction via Triton batched matmul
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)   (row‑major)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # allocate output tensor
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # strides for A (left)
    stride_am = left_mat.stride(1)          # row stride = N
    stride_ak = left_mat.stride(2)          # col stride = 1
    stride_batch_a = left_mat.stride(0)     # batch stride = N*N

    # strides for Bᵀ (right transposed view)
    # original right stride: (batch, row, col) = (N*N, N, 1)
    # after transposition we need (k, n) where k corresponds to original col
    stride_bk = right_mat.stride(2)         # original col stride = 1  -> stride for k
    stride_bn = right_mat.stride(1)         # original row stride = N  -> stride for n
    stride_batch_b = right_mat.stride(0)    # N*N

    # strides for C (output)
    stride_cm = out_mat.stride(1)           # row stride = N
    stride_cn = out_mat.stride(2)           # col stride = 1
    stride_batch_c = out_mat.stride(0)      # N*N

    ROWS = N
    COLS = N
    K = N
    BATCH = B * H

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(ROWS, BLOCK_M),
        triton.cdiv(COLS, BLOCK_N),
        BATCH,
    )

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        ROWS,
        COLS,
        K,
        stride_batch_a,
        stride_batch_b,
        stride_batch_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)          # (B, N, N, dim)

    # cast back to the original input dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
