```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    assert a.shape == b.shape, "Shapes mismatch for multiplication"
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out
```

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # cast to fp16 for heavy compute
    x_fp16 = x.to(torch.float16)

    # cast model weights to fp16
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    # linear projections
    left_proj = F.linear(x_fp16, w_left_proj)  # [B,N,N,hidden_dim]
    right_proj = F.linear(x_fp16, w_right_proj)

    # gating
    left_gate = torch.sigmoid(F.linear(x_fp16, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_fp16, w_right_gate))

    # mask
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = elemwise_mul(left_proj, mask_fp16)
        right_proj = elemwise_mul(right_proj, mask_fp16)

    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Reshape for batched GEMM
    B, N, _, H = left.shape  # H==hidden_dim
    # shape to (B*H,N,N) for left and right
    left_batched = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_batched = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # compute batched matmul: left @ right^T
    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))

    # reshape back
    out = out_batched.view(B, H, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]

    # hidden dim layernorm
    out = F.layer_norm(out, (hidden_dim,), weight=w_to_out_norm, bias=b_to_out_norm)

    # out gate
    out_gate = torch.sigmoid(F.linear(x_fp16, w_out_gate))  # [B,N,N,hidden_dim]
    out = elemwise_mul(out, out_gate)

    # final linear projection
    out = F.linear(out, w_to_out)  # [B,N,N,dim]

    # cast back to original dtype (float32)
    out = out.to(x.dtype)

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for element-wise multiplication of two tensors (flattened)
@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Element‑wise multiplication using a Triton kernel."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication."
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out

    The heavy O(N³·H) reduction is performed by a batched GEMM via torch.bmm,
    while a small Triton kernel fuses the final element‑wise gating.
    """

    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    """Multiply two 1‑D tensors element‑wise."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    tl.store(out_ptr + offs, a * b, mask=mask)
