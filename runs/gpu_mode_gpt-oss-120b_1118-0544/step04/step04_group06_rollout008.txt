@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, # seq_len
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_i, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,h,i,k]
        left_ptrs = left_ptr + \
            pid_bh * stride_left_b + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k

        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right[b,h,j,k] need transposed to (k,j)
        right_ptrs = right_ptr + \
            pid_bh * stride_right_b + \
            offs_k[:, None] * stride_right_k + \
            offs_j[None, :] * stride_right_i  # note swapped positions

        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)  # left (BLOCK_M,BLOCK_K) * right (BLOCK_K,BLOCK_N) -> (BLOCK_M,BLOCK_N)

    # Store result
    out_ptrs = out_ptr + \
        pid_bh * stride_out_b + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j

    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) using the provided LN parameters.
2. Cast to fp16 for compute‑intensive parts.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are applied with fp16 weights.
4. If a mask is present, a small Triton kernel fuses the elementwise
   multiplication of projection, gate and mask, avoiding extra passes.
5. Core reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each (b,h).  The reduction
   is performed by a custom Triton kernel that tiles the i/j dimensions
   (BLOCK_M × BLOCK_N) and iterates over k in blocks (BLOCK_K), using
   fp32 accumulation for numerical stability.
6. Output LayerNorm over the hidden dimension, elementwise multiplication
   by the output gate, and a final linear projection back to the original
   channel dimension.
7. Cast back to the original input dtype.

The implementation uses Triton for (a) mask‑gate fusion and (b) the O(N³·H)
reduction, while the remaining linear layers are executed with PyTorch.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, gate and projection (elementwise)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,      # fp16 * (total elements)
    gate_ptr,      # fp16 * (total elements)
    mask_ptr,      # fp16 * (total mask elements)
    out_ptr,       # fp16 * (total elements)
    total_elems,   # i64 – total number of elements in proj/gate
    hidden_dim,    # i64 – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # bounds
    active = offs < total_elems

    # each mask element repeats hidden_dim times
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=active, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=active, other=0.0)
    mval = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)

    out = proj * gate * mval
    tl.store(out_ptr + offs, out, mask=active)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """Fuse proj * gate * mask using Triton."""
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _trimul_reduce_kernel(
    left_ptr,   # fp16 * (BH, N, N)
    right_ptr,  # fp16 * (BH, N, N)
    out_ptr,    # fp16 * (BH, N, N)
    N: tl.constexpr,                # sequence length
    stride_l_bh, stride_l_i, stride_l_k,  # left strides
    stride_r_bh, stride_r_i, stride_r_k,  # right strides
    stride_o_bh, stride_o_i, stride_o_j,  # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # i‑tile
    pid_j = tl.program_id(1)   # j‑tile
    pid_bh = tl.program_id(2)  # combined batch & hidden

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in blocks
    for k_block in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_block * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[bh, i, k]  -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            pid_bh * stride_l_bh + \
            offs_i[:, None] * stride_l_i + \
            offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right[bh, j, k] need to be transposed to (k, j) for dot
        right_ptrs = right_ptr + \
            pid_bh * stride_r_bh + \
            offs_k[:, None] * stride_r_k + \
            offs_j[None, :] * stride_r_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate product over k
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N)

    # Store the result
    out_ptrs = out_ptr + \
        pid_bh * stride_o_bh + \
        offs_i[:, None] * stride_o_i + \
        offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C] torch.Tensor (float32/float16)
            - mask: [B, N, N] torch.Tensor or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,C) -> (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (same shape) and sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask + gates (fused via Triton when mask present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a Triton tiled batched GEMM.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Rearrange to (B*H, N, N) for a simple 3‑D view.
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)

    # Strides for the flattened tensors (int64)
    stride_l_bh, stride_l_i, stride_l_k = left_flat.stride()
    stride_r_bh, stride_r_i, stride_r_k = right_flat.stride()
    stride_o_bh, stride_o_i, stride_o_j = out_flat.stride()

    # Tile sizes – chosen to fit typical shared‑memory limits.
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),    # i tiles
        triton.cdiv(N, BLOCK_N),    # j tiles
        B * H                       # combined batch‑hidden dimension
    )

    _trimul_reduce_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_l_bh, stride_l_i, stride_l_k,
        stride_r_bh, stride_r_i, stride_r_k,
        stride_o_bh, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
