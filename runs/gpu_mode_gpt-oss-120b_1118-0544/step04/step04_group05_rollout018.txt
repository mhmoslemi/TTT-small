"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C).
2. Linear projections → tensors L,R of shape (B,N,N,H).
3. Optional binary mask (broadcast over H) applied to L,R.
4. Compute three sigmoid gates (left, right, out) from the normalized input.
5. Multiply L,R by their respective gates.
6. Core O(N³·H) reduction:
   out[b,i,j,d] = Σₖ L[b,i,k,d] * R[b,j,k,d]
   This is a batched GEMM C = A @ Bᵀ where A and B have shape (B·H,N,N).
   Implemented with a custom Triton kernel that streams over the K‑dimension
   using tf32/FP16 tensor‑core dot products (accumulated in FP32).
7. Reshape back to (B,N,N,H), apply a second LayerNorm, multiply by the output
   gate and project back to the original channel dimension.
8. Return a tensor of shape (B,N,N,C) in the original dtype.

Only the heavy batched GEMM is written in Triton; all other operations use
PyTorch (which is cheap compared to the O(N³·H) work).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B shape: (batch, N, N))
# ------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix dims (M=N=seq_len, K=N)
    stride_ah, stride_am, stride_ak,    # A: batch, row, col strides
    stride_bh, stride_bk, stride_bn,    # Bᵀ: batch, k, n strides
    stride_ch, stride_cm, stride_cn,    # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch index (B*H)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col

    # --------------------------------------------------------------
    # Compute block offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Compute base pointers for this batch
    # --------------------------------------------------------------
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # Pointers to the beginning of the current tile in A and Bᵀ
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    # --------------------------------------------------------------
    # Accumulator (FP32 for numeric stability)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot: (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

        # Advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Write result (store in FP16, keep FP32 accumulation inside)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ where A,B have shape (batch, N, N).
    All tensors must be on CUDA and in float16.
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    assert A.shape == B.shape
    batch, N, K = A.shape
    assert N == K, "TriMul expects square matrices."

    # --------------------------------------------------------------
    # Choose block sizes based on N (heuristic)
    # --------------------------------------------------------------
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:  # N up to 1024 in tests
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # M‑blocks
        (N + BLOCK_N - 1) // BLOCK_N,   # N‑blocks
        batch,                          # batch dimension (B*H)
    )

    C = torch.empty_like(A)

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),   # A: batch, row, col
        B.stride(0), B.stride(2), B.stride(1),   # B as Bᵀ: batch, k, n
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )
    return C


# ------------------------------------------------------------------
# Entry point
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask          : torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights       : dict of model parameters (float32 tensors)
            - config        : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, ignore mask when True (default False)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ----------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------
    # LayerNorm (input)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Work in FP16 for the heavy path
    x = x.to(torch.float16)

    # ----------------------------------------------------------
    # Cast weights to FP16 (no bias in linear layers)
    # ----------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ----------------------------------------------------------
    # Linear projections (B,N,N,H)
    # ----------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w)  # (B,N,N,H)

    # ----------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # ----------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------
    # Gates (sigmoid of linear projections)
    # ----------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    # ----------------------------------------------------------
    # Apply gates to projections
    # ----------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # Fold hidden dimension into the batch dimension
    # ----------------------------------------------------------
    B, N, _, H = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)
    left = left.view(B * H, N, N)                  # (B*H, N, N)
    right = right.view(B * H, N, N)                # (B*H, N, N)

    # ----------------------------------------------------------
    # Core O(N³·H) reduction via Triton GEMM (C = A @ Bᵀ)
    # ----------------------------------------------------------
    out_flat = _batched_matmul_triton(left, right)   # (B*H, N, N)

    # ----------------------------------------------------------
    # Reshape back to (B, N, N, H)
    # ----------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ----------------------------------------------------------
    # Output LayerNorm over hidden dimension
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------------------------------------------
    # Apply output gate
    # ----------------------------------------------------------
    out = out * out_gate

    # ----------------------------------------------------------
    # Final linear projection back to original channel dimension
    # ----------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ----------------------------------------------------------
    # Cast back to original dtype if needed
    # ----------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
