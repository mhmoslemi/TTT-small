"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the PyTorch reference:
1. Layer‑normalize the input over the channel dimension.
2. Apply left/right linear projections and optional binary mask.
3. Compute three sigmoid gates (left, right, out) from the normalized input.
4. Multiply left/right by their gates.
5. Fold the hidden dimension into the batch dimension and compute
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]          (batched GEMM).
   This is performed by a custom Triton kernel that computes
   C = A @ Bᵀ for A,B of shape (batch, N, N) in FP16 with FP32 accumulation.
6. Reshape back, apply a second LayerNorm, multiply by out_gate,
   and project back to the original channel dimension.
All heavy computation (the O(N³·H) contraction) is done in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all equal to seq_len)
    stride_ah, stride_am, stride_ak,   # A: batch, row, col strides
    stride_bh, stride_bk, stride_bn,   # B: batch, row(K), col(N) strides  (B is already transposed)
    stride_ch, stride_cm, stride_cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ where A,B have shape (batch, N, N)."""
    pid_b = tl.program_id(2)   # batch index (folded B*H)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column

    # -------- tile offsets ----------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------- pointers to tile ----------
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        k_offset = k * BLOCK_K
        k_valid = (k_offset + tl.arange(0, BLOCK_K)) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # -------- store result ----------
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for tensors of shape (batch, N, N) using Triton.
    A, B must be half‑precision and CUDA tensors.
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    assert A.shape == B.shape
    batch, N, K = A.shape
    assert N == K, "TriMul requires square matrices"

    # Heuristic block sizes – good for N up to 1024 and hidden_dim ≤ 128
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 8
    else:  # up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows
        (N + BLOCK_N - 1) // BLOCK_N,   # cols
        batch,                          # batch dimension (folded B*H)
    )

    C = torch.empty_like(A)

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        N,
        N,
        N,
        A.stride(0), A.stride(1), A.stride(2),   # A strides: batch, row, col
        B.stride(0), B.stride(1), B.stride(2),   # B strides: batch, row(K), col(N)
        C.stride(0), C.stride(1), C.stride(2),   # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32 or float16)
            - mask          : torch.Tensor [B, N, N] (binary) or None
            - weights       : dict of model parameters (float32 tensors)
            - config        : dict with keys
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, ignore mask when True (optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] with the same dtype as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Work in FP16 for the heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Cast linear weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # Optional mask
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------------------------------------------------------
    # Apply gates to projections
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Fold hidden dimension into batch dimension for GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    left  = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    left  = left.view(B * H, N, N)                   # (B*H, N, N)
    right = right.view(B * H, N, N)                  # (B*H, N, N)

    # ------------------------------------------------------------------
    # Core O(N³·H) reduction via Triton (or fallback to cuBLAS for large N)
    # ------------------------------------------------------------------
    if N >= 512:
        # cuBLAS (torch.bmm) is faster for very large N on the target hardware
        out_flat = torch.bmm(left, right.transpose(-2, -1))
    else:
        # Use Triton kernel: compute left @ rightᵀ
        right_t = right.transpose(-2, -1).contiguous()
        out_flat = _batched_matmul_triton(left, right_t)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # Apply output gate and final projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
