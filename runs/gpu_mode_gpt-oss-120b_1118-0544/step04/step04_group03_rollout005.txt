"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor.
2. Fuse linear projections and gates:
   * left = (input @ left_proj) ⊙ sigmoid(input @ left_gate)
   * right = (input @ right_proj) ⊙ sigmoid(input @ right_gate)
   Optionally apply the pairwise mask to left/right before gating.
3. Core O(N³·H) reduction:
   * For each hidden dimension h, compute left_h @ right_hᵀ
   * Implemented as a batched GEMM kernel in Triton (half‑precision
     with float32 accumulation).
4. Layer‑normalize the hidden result, apply the output gate, and project
   back to the original channel dimension.
All heavy arithmetic runs in fp16; the final output is cast back to the
input dtype (typically fp32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A : (batch, M, K)  – fp16
#   B : (batch, K, N)  – fp16   (interpreted as Bᵀ in the kernel)
#   C : (batch, M, N)  – fp16 (accumulated in fp32)
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A strides: (M, K)
    stride_bk, stride_bn,          # B strides: (K, N)
    stride_cm, stride_cn,          # C strides: (M, N)
    stride_ab, stride_bb, stride_cb,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)      # output tile index (i‑j)
    batch = tl.program_id(1)    # batch index (B·H)

    # ---- tile coordinates -------------------------------------------------
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # ---- offsets within the tile ------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ---- masks for out‑of‑bounds -----------------------------------------
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- base pointers for this batch ------------------------------------
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    # ---- accumulator -------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K dimension --------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # Load A tile (M × K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (K × N) – we read it as Bᵀ so that the inner product is a·b
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate fp16·fp16 -> fp32
        acc += tl.dot(a, b)

    # ---- write result -------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, C]
        - mask        : torch.Tensor of shape [B, N, N] or None
        - weights     : dict of model parameters (float32 tensors)
        - config      : dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device
    orig_dtype = x.dtype

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (fp16 for speed)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 2. Cast model weights to fp16 (they are stored as fp32)
    # ------------------------------------------------------------------- #
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------- #
    # 3. Linear projections + gates (fused)
    # ------------------------------------------------------------------- #
    # Combined projection+gate: shape [B, N, N, 2*H]
    left_comb = F.linear(x, left_comb_w)   # [B, N, N, 2*H]
    right_comb = F.linear(x, right_comb_w) # [B, N, N, 2*H]

    # Split projections and gates
    left_proj = left_comb[..., :hidden_dim]                # [B, N, N, H]
    left_gate = left_comb[..., hidden_dim:].sigmoid()      # [B, N, N, H]
    right_proj = right_comb[..., :hidden_dim]              # [B, N, N, H]
    right_gate = right_comb[..., hidden_dim:].sigmoid()    # [B, N, N, H]

    # Optional mask (broadcast over hidden dim)
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)      # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Apply gates
    left = left_proj * left_gate   # [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------- #
    # 4. Core O(N³·H) reduction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape   # B, N, N, H

    # Reshape to (batch = B*H, M = N, K = N) for both operands
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_t.view(B * H, N, N)   # (batch, M, K)
    right_flat = right_t.view(B * H, N, N) # (batch, K, N) – we will treat it as transposed in the kernel

    out_flat = torch.empty_like(left_flat)  # (batch, M, N), fp16

    # Strides for the kernel
    stride_a_m = left_flat.stride(1)   # K dimension stride for A (row stride)
    stride_a_k = left_flat.stride(2)   # column stride for A
    stride_b_k = right_flat.stride(1)  # K stride for B
    stride_b_n = right_flat.stride(2)  # N stride for B
    stride_c_m = out_flat.stride(1)    # M stride for C
    stride_c_n = out_flat.stride(2)    # N stride for C

    stride_ab = left_flat.stride(0)    # batch stride for A
    stride_bb = right_flat.stride(0)   # batch stride for B
    stride_cb = out_flat.stride(0)     # batch stride for C

    # Tile sizes – tuned for typical seq_len ≤ 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)  # (output tiles, batch)

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_a_m,
        stride_a_k,
        stride_b_k,
        stride_b_n,
        stride_c_m,
        stride_c_n,
        stride_ab,
        stride_bb,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 5. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------- #
    # 6. Output gate (element‑wise)
    # ------------------------------------------------------------------- #
    out_gate = F.linear(x, out_gate_w).sigmoid()   # [B, N, N, H]
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 7. Final projection back to original channel dimension
    # ------------------------------------------------------------------- #
    out = F.linear(out, to_out_w)   # [B, N, N, dim]

    # Cast back to the original dtype of the input
    out = out.to(orig_dtype)

    return out
```
