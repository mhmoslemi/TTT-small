@triton.jit
def triton_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    
    pid = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    
    # compute block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    
    # compute pointer offsets for batch
    batch_idx = pid
    a_batch_ptr = a_ptr + batch_idx * batch_stride
    b_batch_ptr = b_ptr + batch_idx * batch_stride
    c_batch_ptr = c_ptr + batch_idx * batch_stride
    
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
        # Load with mask
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b, out_dtype=tl.float32)
    
    # write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) forward pass implemented with a fused Triton GEMM.

Algorithm
---------
1. Layer‑norm over the input channel dimension (float32 → FP16).
2. Four linear projections (left, right, left_gate, right_gate, out_gate) in
   FP16, followed by sigmoid for the gates.
3. Optional masking of the left/right projections.
4. Element‑wise gating: left = left_proj * left_gate,
   right = right_proj * right_gate.
5. Batched GEMM:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   i.e. for each hidden channel d we compute a matrix multiplication
   left[d] @ right[d].T.  This is performed by a custom Triton kernel that
   processes the (batch × hidden) dimension as the first grid axis and tiles
   the N×N output with BLOCK_M×BLOCK_N blocks, accumulating in FP32.
6. Layer‑norm over the hidden dimension.
7. Multiply by out_gate (sigmoid‑gated linear projection) – elementwise,
   performed with a tiny Triton kernel.
8. Final linear projection back to the original dimension.
9. Cast result to float32.

The heavy O(N³·hidden) contraction is fully executed on the GPU using
Tensor‑cores (FP16 × FP16 → FP32 accumulation), which yields the required
sub‑millisecond runtimes on the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ   (A,B shape: [batch, M, K]; C shape: [batch, M, N])"""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch
    a_batch = a_ptr + pid_batch * batch_stride
    b_batch = b_ptr + pid_batch * batch_stride
    c_batch = c_ptr + pid_batch * batch_stride

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A: (M, K)   → rows = offs_m, cols = offs_k
        a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        # Bᵀ: (K, N) → rows = offs_k, cols = offs_n  (note the swapped strides)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Write back C
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


@triton.jit
def _elemwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    """out = out * gate   (1‑D flattened tensors)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out * gate, mask=mask)


def custom_kernel(data):
    """TriMul (outgoing) forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored if config["nomask"] is True
            - weights: dict of model tensors (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- 1. Input LayerNorm ----------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # Cast to FP16 for the heavy part
    x_fp16 = x.to(torch.float16)

    # ---------- 2. Linear projections ----------
    # weight shapes: (out_features, in_features)
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_proj = F.linear(x_fp16, left_proj_w)          # [B, N, N, H]
    right_proj = F.linear(x_fp16, right_proj_w)

    # ---------- 3. Optional mask ----------
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ---------- 4. Gating ----------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    left = left_proj * left_gate          # [B, N, N, H]
    right = right_proj * right_gate

    # ---------- 5. Batched GEMM (Triangular multiplication) ----------
    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden
    # Rearrange to (B, H, N, N) and flatten batch*hidden
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    batch_hidden = B * hidden_dim
    left_mat = left.view(batch_hidden, N, N)   # (B*H, N, N)
    right_mat = right.view(batch_hidden, N, N)

    # Output tensor
    out_mat = torch.empty_like(left_mat)

    # Strides (int64)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    stride_bk = right_mat.stride(2)   # col stride of original right (used as row stride after transpose)
    stride_bn = right_mat.stride(1)   # row stride of original right (used as column stride after transpose)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    batch_stride = left_mat.stride(0)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        batch_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ---------- 6. Output LayerNorm ----------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # [B, N, N, H]  (FP16)

    # ---------- 7. out_gate element‑wise multiplication (Triton) ----------
    total_elements = out.numel()
    BLOCK = 1024
    grid_gate = ( (total_elements + BLOCK - 1) // BLOCK, )
    _elemwise_mul_kernel[grid_gate](
        out,
        out_gate,
        total_elements,
        BLOCK=BLOCK,
    )

    # ---------- 8. Final linear projection ----------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # [B, N, N, C]  (FP16)

    # Cast back to float32 as required by the API
    return out.to(torch.float32)
