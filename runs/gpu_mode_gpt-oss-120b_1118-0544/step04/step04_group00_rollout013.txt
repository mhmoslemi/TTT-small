@triton.jit
def _gate_mask_fusion_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, total_elements, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements
    # compute mask index
    mask_idx = offs // hidden_dim
    # load
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

    # Layernorm

    # cast to fp16

    # Convert weights to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Apply mask and gates
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)

        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core reduction via torch.einsum
    # result shape (B,N,N,hidden_dim)
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to original dtype

    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated mask‑gate fusion.

Algorithm
---------
1. Input LayerNorm over the channel dimension `dim`.
2. Cast to fp16 for the compute‑heavy path.
3. Linear projections & gate projections (all in fp16):
       left_proj  = x @ W_left
       right_proj = x @ W_right
       left_gate  = σ(x @ W_left_gate)
       right_gate = σ(x @ W_right_gate)
       out_gate   = σ(x @ W_out_gate)
4. Fuse mask, projection and gate in a single Triton kernel
   (applied only when a mask is provided).  The kernel computes
       proj * gate * mask   (mask broadcast over the hidden dimension).
   When no mask is present a simple elementwise multiplication is used.
5. Core reduction (the “triangle” update):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden channel.
   Implemented with a single `torch.einsum` to avoid costly
   permute/reshape copies.
6. Output LayerNorm over the hidden dimension `hidden_dim`.
7. Multiply by `out_gate` and apply the final linear projection back
   to the original dimension `dim`.
8. Cast the result back to the input dtype (usually float32).

All heavy elementwise work (mask‑gate fusion) is performed by a
custom Triton kernel; the expensive N³ reduction uses a highly‑optimized
einsum implementation.
"""

def _gate_mask_fusion_kernel(
    proj_ptr,          # fp16*  (B*N*N*H,)
    gate_ptr,          # fp16*  (B*N*N*H,)
    mask_ptr,          # fp16*  (B*N*N,)
    out_ptr,           # fp16*  (B*N*N*H,)
    total_elems,       # int    total number of elements in proj/gate (=B*N*N*H)
    hidden_dim,        # int    H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse projection, gate and optional mask."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # each mask element corresponds to `hidden_dim` consecutive projection elems
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 8192) -> torch.Tensor:
    """
    Triton‑fused elementwise multiplication:
        result = proj * gate * mask   (mask broadcast over hidden_dim)

    All tensors must be fp16 and contiguous.
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_fusion_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C]   (float32)
            - mask         : torch.Tensor of shape [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                  * "dim"          – input channel dimension C
                  * "hidden_dim"   – hidden dimension H
                  * "nomask" (optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim), dtype = input dtype

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to fp16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (same shape) + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Fuse mask, projection and gate (Triton) if mask is present
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask is broadcast over the hidden dimension
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with a single einsum to avoid extra copies.
    # ------------------------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
