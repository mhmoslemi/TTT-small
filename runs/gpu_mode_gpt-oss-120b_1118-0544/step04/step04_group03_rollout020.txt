def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_a_batch, stride_b_batch, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, GROUP_M: tl.constexpr
):
    pid = tl.program_id(0)
    batch = tl.program_id(1)

    # Compute block indices
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    # map to groups
    pid_m = (pid // num_pid_n)
    pid_n = pid % num_pid_n

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_ptrs = a_ptr + batch * stride_a_batch + (offs_m[:, None] * stride_am + 0 * stride_ak)
    b_ptrs = b_ptr + batch * stride_b_batch + (0 * stride_bk + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs + cur_k[None, :] * stride_ak,
                    mask=(mask_m[:, None] & mask_k[None, :]), other=0.0)
        b = tl.load(b_ptrs + cur_k[:, None] * stride_bk,
                    mask=(mask_k[:, None] & mask_n[None, :]), other=0.0)
        acc += tl.dot(a, b)

    # Store
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + batch * stride_c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1️⃣ Layer‑norm the input tensor.
2️⃣ Fuse left/right linear projections with their gates:
   * Concatenate proj‑weight and gate‑weight → single F.linear call.
   * Split the result, apply sigmoid to the gate part.
3️⃣ Apply optional mask (broadcast over the hidden dimension) to the
   projected tensors and then multiply by the gates.
4️⃣ Core O(N³·H) reduction:
   * For each hidden channel d compute   out[...,d] = left[...,d] @ right[...,d]ᵀ .
   * This is a batched GEMM with batch = B·H.
   * Implemented with a tiled Triton kernel (float16 inputs, float32 acc).
5️⃣ Hidden‑dim LayerNorm + output gate (sigmoid).
6️⃣ Final linear projection back to the original channel dimension.
7️⃣ Cast the result to the original dtype (float32).

The heavy matrix multiplication is fully performed in Triton, while the
remaining tensor operations use PyTorch (FP16 for speed, FP32 for final
output).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B shape: batch × M × K)
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (M, K)
    stride_bk, stride_bn,          # B: (K, N)   (B is already transposed)
    stride_cm, stride_cn,          # C: (M, N)
    stride_a_batch, stride_b_batch, stride_c_batch,  # batch stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM (A @ Bᵀ) with fp32 accumulation."""
    batch_id = tl.program_id(1)          # batch = B * H
    pid = tl.program_id(0)               # tile id for (i, j)

    # Compute block coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile: (M, K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (already transposed): (K, N)
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp16·fp16 → fp32

    # Write back
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------- #
    # unpack inputs
    # --------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device
    dtype_fp16 = torch.float16

    # --------------------------------------------------------------- #
    # 1. LayerNorm on the input
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(dtype_fp16)   # fp16 for the heavy path

    # --------------------------------------------------------------- #
    # 2. Prepare fused projection+gate weights (fp16)
    # --------------------------------------------------------------- #
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype_fp16)       # (2*H, C)

    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype_fp16)       # (2*H, C)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # --------------------------------------------------------------- #
    # 3. Linear projections + gates (fused)
    # --------------------------------------------------------------- #
    # Shape after linear: [B, N, N, 2*H]
    left_comb = F.linear(x, left_comb_w)   # proj | gate
    right_comb = F.linear(x, right_comb_w)

    # Split
    left_proj = left_comb[..., :hidden_dim]                # [B, N, N, H]
    left_gate = left_comb[..., hidden_dim:].sigmoid()      # [B, N, N, H]

    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # Optional mask (broadcast over hidden dim)
    if mask is not None and not nomask:
        mask_f = mask.to(dtype_fp16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Apply gates
    left = left_proj * left_gate      # [B, N, N, H]
    right = right_proj * right_gate   # [B, N, N, H]

    # --------------------------------------------------------------- #
    # 4. Core O(N³·H) reduction via Triton GEMM (batched)
    # --------------------------------------------------------------- #
    B, N, _, H = left.shape   # B, seq_len, _, hidden_dim

    # Rearrange to (batch, H, N, N) → (B*H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()

    # Transpose right for the gemm (so we compute A @ Bᵀ)
    right_t_T = right_t.transpose(-2, -1).contiguous()   # [B, H, N, N]

    # Merge batch and hidden dimensions
    batch_dim = B * H
    A = left_t.view(batch_dim, N, N)          # (batch, M=N, K=N)
    B_T = right_t_T.view(batch_dim, N, N)     # (batch, K=N, N)

    # Output tensor
    C = torch.empty_like(A, dtype=dtype_fp16)   # (batch, M, N)

    # Strides (int64)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)
    stride_a_batch = A.stride(0)

    stride_bk = B_T.stride(1)
    stride_bn = B_T.stride(2)
    stride_b_batch = B_T.stride(0)

    stride_cm = C.stride(1)
    stride_cn = C.stride(2)
    stride_c_batch = C.stride(0)

    # Tile sizes (tuned for typical N ≤ 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_tiles_m * num_tiles_n, batch_dim)

    _batched_gemm_kernel[grid](
        A, B_T, C,
        N, N, N,                      # M, N, K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_a_batch, stride_b_batch, stride_c_batch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out_hidden = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------- #
    # 5. Hidden‑dim LayerNorm
    # --------------------------------------------------------------- #
    out_hidden = F.layer_norm(
        out_hidden,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------- #
    # 6. Output gate (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------- #
    out_gate = F.linear(x, out_gate_w).sigmoid()   # [B, N, N, H]
    out_hidden = out_hidden * out_gate

    # --------------------------------------------------------------- #
    # 7. Final linear projection back to original dimension
    # --------------------------------------------------------------- #
    out = F.linear(out_hidden, to_out_w)   # [B, N, N, C]

    # Cast to the original dtype (float32)
    out = out.to(x.dtype)

    return out
