@triton.jit
def batched_gemm_kernel(a_ptr, b_ptr, c_ptr,
                        batch_stride_a, batch_stride_b, batch_stride_c,
                        stride_am, stride_ak,
                        stride_bk, stride_bn,
                        stride_cm, stride_cn,
                        M, N, K,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    # compute total number of blocks per batch
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_id = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    block_m = block_id // num_blocks_n
    block_n = block_id % num_blocks_n

    # compute offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute valid mask for M,N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to output
    c_ptrs = c_ptr + batch_id * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # use float32 accumulation

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offset
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile
        a_ptrs = a_ptr + batch_id * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B tile (B is transposed; we assume stride_bk and stride_bn passed accordingly)
        b_ptrs = b_ptr + batch_id * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # compute dot product using tl.dot
        acc += tl.dot(a, b)
    # finalize and store
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul)

The kernel implements the heavy O(N³·H) contraction of the TriMul module as a
single batched GEMM in Triton, avoiding the per‑batch cuBLAS launch overhead.
All linear projections, gating, and layer‑norms are performed with PyTorch
(FP16 compute, FP32 accumulation for stability).  The final output is returned
in the original dtype of the input tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused batched matrix multiplication
# Computes   C[b] = A[b] @ B[b].T   for each batch b.
# A, B have shape (B, M, K) and (B, K, N) respectively (row‑major).
# The kernel receives B, M, N, K as runtime arguments and uses
# BLOCK_M/N/K tiling.  Accumulation is performed in float32.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # number of tiles per matrix
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # derive batch index and tile indices
    batch_id = pid // (num_blocks_m * num_blocks_n)
    tile_id = pid % (num_blocks_m * num_blocks_n)
    tile_m = tile_id // num_blocks_n
    tile_n = tile_id % num_blocks_n

    # block offsets
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the output tile
    c = c_ptr + batch_id * batch_stride_c \
        + offs_m[:, None] * stride_cm \
        + offs_n[None, :] * stride_cn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M x K)
        a = tl.load(
            a_ptr + batch_id * batch_stride_a
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # B tile (K x N) – we treat B as transposed, so we swap strides
        b = tl.load(
            b_ptr + batch_id * batch_stride_b
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # dot product (FP16 * FP16 -> FP32)
        acc += tl.dot(a, b)

    # write result back in FP16
    c = acc.to(tl.float16)
    tl.store(c, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32)
            - config: dict containing "dim" (C) and "hidden_dim" (H)
    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # Cast to fp16 for the compute‑heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)

    left = F.linear(x, left_proj_w)          # (B, N, N, H)
    right = F.linear(x, right_proj_w)        # (B, N, N, H)

    # ------------------------------------------------------------------
    # Gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # Optional mask application (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) -> (B, N, N, 1)
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core contraction: batched GEMM left @ rightᵀ
    # ------------------------------------------------------------------
    B, N0, _, H = left.shape
    assert N0 == N0, "Non‑square pairwise dimension not supported"
    N = N0

    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    batch = left_mat.shape[0]   # = B * H

    # output tensor
    out_mat = torch.empty_like(left_mat)

    # strides (row‑major)
    stride_am = left_mat.stride(1)   # = K (N)
    stride_ak = left_mat.stride(2)   # = 1
    batch_stride_a = left_mat.stride(0)

    # right is transposed inside the kernel, so swap its inner strides
    stride_bk = right_mat.stride(2)   # original col stride = 1
    stride_bn = right_mat.stride(1)   # original row stride = N
    batch_stride_b = right_mat.stride(0)

    stride_cm = out_mat.stride(1)   # = N
    stride_cn = out_mat.stride(2)   # = 1
    batch_stride_c = out_mat.stride(0)

    # Tiling parameters (tuned for B200, fp16)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch * num_blocks_m * num_blocks_n,)

    _batched_gemm_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        batch_stride_a=batch_stride_a,
        batch_stride_b=batch_stride_b,
        batch_stride_c=batch_stride_c,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        M=N,
        N=N,
        K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, C), fp16

    # Cast back to the original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)
    return out
