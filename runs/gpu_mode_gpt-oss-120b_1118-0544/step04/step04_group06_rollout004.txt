    left_proj = torch.nn.functional.linear(x, left_proj_w)  # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # Gate projections
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()

"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate.
---------------------------------------------------------

* LayerNorm on the input (float32) → fp16.
* Linear projections (left/right) and gate projections (left_gate,
  right_gate, out_gate) are computed in fp16.
* If a mask is present (and not disabled) we fuse the elementwise
  multiplication `proj * gate * mask` with a tiny Triton kernel,
  avoiding an extra memory pass.
* Core reduction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
  which is a batched matrix‑multiply of shape (B·H, N, N) × (B·H, N, N)ᵀ.
  We reuse highly‑optimized cuBLAS via `torch.bmm`.
* Output LayerNorm, output‑gate multiplication and final linear projection
  return a tensor of shape [B, N, N, dim] (original dtype).

Only the mask‑gate fusion is implemented in Triton; all other heavy
operations use native PyTorch for maximal performance on the target GPU.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i64  – total number of elements in proj/gate
    hidden_dim,        # i64  – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse elementwise multiplication: out = proj * gate * mask,
    where mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # Bounds
    mask_valid = offs < total_elems

    # Compute mask index (each mask element repeats hidden_dim times)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask_valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask_valid, other=0.0)
    mval = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    out = proj * gate * mval
    tl.store(out_ptr + offs, out, mask=mask_valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Apply `proj * gate * mask` using a Triton kernel.
    All tensors must be fp16, contiguous and have matching device.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],               # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out

    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication using cuBLAS (torch.bmm)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8. Cast back to original dtype (e.g., float32)
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
