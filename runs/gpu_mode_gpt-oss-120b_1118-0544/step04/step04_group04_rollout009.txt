@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
    # Acquire program ids:
    pid_m = tl.program_id(0)  # tile_i
    pid_n = tl.program_id(1)  # tile_j
    pid_z = tl.program_id(2)  # batch * hidden

    # Determine batch and hidden index:
    b = pid_z // H
    h = pid_z % H

    # Compute offsets for i and j:
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Guard for out of bounds:
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator:
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension:
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Compute pointers
        # left_ptr: [b, i, k, h]
        a_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + offs_i[:, None] * stride_left_i + k_offset[None, :] * stride_left_k
        # right_ptr: [b, j, k, h] but we need shape (K,N) when loaded:
        b_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + k_offset[:, None] * stride_right_k + offs_j[None, :] * stride_right_i

        # Load with masks for boundaries
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Multiply-accumulate
        acc += tl.dot(a, b)  # computes sum over K dimension

    # Write back
    # out_ptr: [b, i, j, h] (B,N,N,H)
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1. LayerNorm: using torch.nn.functional.layer_norm
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)
    # Cast to fp16
    x = x.to(torch.float16)

    # 2. Projections and gates
    # linear can be done via F.linear
    left_proj = F.linear(x, weights["left_proj.weight"].to(device).to(torch.float16))
    right_proj = F.linear(x, weights["right_proj.weight"].to(device).to(torch.float16))
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(device).to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(device).to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(device).to(torch.float16)))

    # Optional mask:
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Apply gates:
    left = left_proj * left_gate  # (B,N,N,hidden_dim)
    right = right_proj * right_gate

    # 3. Core batched GEMM via Triton
    # Prepare output tensor out with same dtype (fp16)
    out = torch.empty_like(left)   # shape (B,N,N,hidden_dim)

    # Get strides:
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()
    B, N, _, H = left.shape

    # Decide block sizes based on N size: typical 64x64 tile, K 32
    BLOCK_M = 64
    BLOCK_N = 64

"""
Outgoing TriMul (Triangle Multiplicative Update) forward pass.

Algorithm:
1. Layer‑norm over the input channel dimension (FP16).
2. Four linear projections (left/right and their gates) – FP16, bias‑free.
3. Optional masking of the left/right projections.
4. Apply sigmoid gates.
5. Batched GEMM: for each batch b and hidden channel h compute
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]  (i.e. A @ Bᵀ)
   Implemented with a fused Triton kernel that tiles the M×N output and
   iterates over K, accumulating in FP32 for accuracy.
6. Layer‑norm over the hidden dimension.
7. Out‑gate (linear + sigmoid) and element‑wise multiplication (Triton).
8. Final linear projection back to the original dimension and cast to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = left @ rightᵀ for each (b, h) pair."""
    pid_m = tl.program_id(0)  # tile over i dimension
    pid_n = tl.program_id(1)  # tile over j dimension
    pid_z = tl.program_id(2)  # batch * hidden

    # decode batch and hidden indices
    b = pid_z // H
    h = pid_z % H

    # i / j offsets for this tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (M, K)
        a_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile (treated as Bᵀ): shape (K, N)
        b_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # multiply‑accumulate
        acc += tl.dot(a, b)  # result is FP32

    # write output (cast back to FP16)
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


@triton.jit
def _elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    """out = out * gate  (1‑D flattened tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out * gate, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float), may be ignored if config["nomask"] is True
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------------------
    # 1. Input LayerNorm (FP16)
    # ----------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float16)
    norm_b = weights["norm.bias"].to(device).to(torch.float16)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )
    x = x.to(torch.float16)  # switch to fp16 for the heavy work

    # ----------------------------------------------------------------------
    # 2. Linear projections & gates (FP16, bias‑free)
    # ----------------------------------------------------------------------
    left_proj = F.linear(
        x, weights["left_proj.weight"].to(device).to(torch.float16)
    )
    right_proj = F.linear(
        x, weights["right_proj.weight"].to(device).to(torch.float16)
    )
    left_gate = torch.sigmoid(
        F.linear(
            x, weights["left_gate.weight"].to(device).to(torch.float16)
        )
    )
    right_gate = torch.sigmoid(
        F.linear(
            x, weights["right_gate.weight"].to(device).to(torch.float16)
        )
    )
    out_gate = torch.sigmoid(
        F.linear(
            x, weights["out_gate.weight"].to(device).to(torch.float16)
        )
    )

    # ----------------------------------------------------------------------
    # 3. Optional mask (applied after projection, before gating)
    # ----------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # ----------------------------------------------------------------------
    # 4. Apply gates
    # ----------------------------------------------------------------------
    left = left_proj * left_gate   # (B, N, N, hidden_dim)
    right = right_proj * right_gate

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ----------------------------------------------------------------------
    # 5. Batched GEMM via Triton: out = left @ rightᵀ across K
    # ----------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # (B, N, N, hidden_dim), fp16

    # Strides (int64)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tile sizes – tuned for typical B200 Tensor‑Core performance
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    _triton_trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ----------------------------------------------------------------------
    # 6. Output LayerNorm over hidden dimension (FP16)
    # ----------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )

    # ----------------------------------------------------------------------
    # 7. Multiply by out_gate (element‑wise) via Triton
    # ----------------------------------------------------------------------
    total_el = out.numel()
    BLOCK = 1024
    grid_mul = ((total_el + BLOCK - 1) // BLOCK,)
    _elementwise_mul_kernel[grid_mul](
        out,
        out_gate,
        total_el,
        BLOCK=BLOCK,
    )

    # ----------------------------------------------------------------------
    # 8. Final linear projection back to original dim (FP16)
    # ----------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, dim)

    # Return in float32 as required by the interface
    return out.to(torch.float32)
