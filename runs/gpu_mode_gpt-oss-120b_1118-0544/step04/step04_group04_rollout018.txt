    stride_cm, stride_cn,

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    input_tensor, mask, weights, config = data
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device

    # get weight tensors
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    # 1. LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # cast to fp16
    x_fp16 = x.to(torch.float16)

    # 2. Linear projections
    # Note: F.linear expects weight shape (out_features, in_features)
    left_proj = torch.nn.functional.linear(x_fp16, left_proj_weight)   # shape (B,N,N,hidden)
    right_proj = torch.nn.functional.linear(x_fp16, right_proj_weight)

    # 3. Apply mask if needed
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # 4. Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_weight))
    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # 5. Batched GEMM using Triton kernel
    # reshape for batch
    B_h = B * hidden_dim
    left_ = left.permute(0,3,1,2).contiguous()

"""
Outgoing TriMul (Triangle Multiplicative Update) forward pass.

The implementation follows the AlphaFold3 TriMul module:
1. Layer‑norm over the input channel dimension (FP32).
2. Four linear projections (left/right) and three gating projections (FP16).
3. Optional masking of the left/right projections.
4. Gating via sigmoid and element‑wise multiplication.
5. Batched GEMM: for each hidden channel we compute
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is a matrix multiplication left @ rightᵀ performed with a custom
   Triton kernel (batch‑wise FP16 GEMM, accumulation in FP32).
6. Layer‑norm over the hidden dimension.
7. out_gate (linear + sigmoid) and element‑wise multiplication via Triton.
8. Final linear projection back to the original dimension (FP16) and cast to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,  # A: (M, K)
    stride_bk, stride_bn,  # Bᵀ: (K, N)   (B is stored as (N,N) and accessed as transposed)
    stride_cm, stride_cn,  # C: (M, N)
    batch_stride,          # M*N for one batch matrix
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  (A,B shape = [batch, M, K])"""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this batch
    a_batch = a_ptr + pid_batch * batch_stride
    b_batch = b_ptr + pid_batch * batch_stride
    c_batch = c_ptr + pid_batch * batch_stride

    # masks for out‑of‑bounds handling
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A tile  (M x K)
        a = tl.load(
            a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=(mask_m[:, None] & mask_k[None, :]),
            other=0.0,
        )
        # Load Bᵀ tile (K x N) -> original B accessed as B[n, k]
        b = tl.load(
            b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=(mask_k[:, None] & mask_n[None, :]),
            other=0.0,
        )
        acc += tl.dot(a, b)

        a_batch += BLOCK_K * stride_ak
        b_batch += BLOCK_K * stride_bk

    # Store result
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=(mask_m[:, None] & mask_n[None, :]),
    )


@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    """out = out * gate  (1‑D flattened tensors)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out * gate, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float), may be ignored if config["nomask"] is True
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------
    # Load all weights (FP32 -> appropriate dtype)
    # ------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)

    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # Cast to FP16 for the heavy compute
    x_fp16 = x.to(torch.float16)

    # ------------------------------
    # 2. Linear projections (FP16)
    # ------------------------------
    left_proj = F.linear(x_fp16, left_proj_weight)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x_fp16, right_proj_weight)

    # ------------------------------
    # 3. Optional mask
    # ------------------------------
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))

    left = left_proj * left_gate      # (B, N, N, hidden_dim)
    right = right_proj * right_gate   # (B, N, N, hidden_dim)

    # ------------------------------
    # 5. Batched GEMM: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with a custom Triton kernel.
    # ------------------------------
    # Rearrange to (B, hidden, N, N) -> (B*hidden, N, N)
    left_ = left.permute(0, 3, 1, 2).contiguous()
    right_ = right.permute(0, 3, 1, 2).contiguous()
    B_h = B * hidden_dim
    left_mat = left_.view(B_h, N, N)
    right_mat = right_.view(B_h, N, N)

    out_mat = torch.empty_like(left_mat)  # FP16 output

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Strides (in elements)
    stride_am = left_mat.stride(1)      # N
    stride_ak = left_mat.stride(2)      # 1
    # For Bᵀ we need stride over K (rows of transposed) = original column stride = 1
    # and stride over N (columns of transposed) = original row stride = N
    stride_bk = right_mat.stride(2)     # 1
    stride_bn = right_mat.stride(1)     # N
    stride_cm = out_mat.stride(1)       # N
    stride_cn = out_mat.stride(2)       # 1
    batch_stride = left_mat.stride(0)   # N * N

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------
    # 6. to_out_norm (LayerNorm over hidden dimension)
    # ------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden_dim), fp16

    # ------------------------------
    # 7. out_gate (linear + sigmoid) and element‑wise multiply via Triton
    # ------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # (B, N, N, hidden_dim)

    total_el = out.numel()
    BLOCK = 1024
    grid_gate = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid_gate](
        out,
        out_gate,
        total_el,
        BLOCK=BLOCK,
    )

    # ------------------------------
    # 8. Final projection back to original dim
    # ------------------------------
    out = F.linear(out, to_out_weight)  # (B, N, N, dim), fp16

    # Cast to float32 as required by the interface
    return out.to(torch.float32)
