    BLOCK_N = 64

        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,

    out = F.linear(out, to_out_w)

    # Cast back to original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    B, H,
    SEQ_LEN: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # decode batch and hidden indices
    b = pid_bh // H
    d = pid_bh % H

    # Range indices within block
    range_i = tl.arange(0, BLOCK_M)
    range_j = tl.arange(0, BLOCK_N)
    # compute actual indices, will be used for masks
    i = i0 + range_i
    j = j0 + range_j

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension tiles
    for k_iter in range(0, tl.cdiv(SEQ_LEN, BLOCK_K)):
        k0 = k_iter * BLOCK_K
        range_k = tl.arange(0, BLOCK_K)
        k = k0 + range_k

        # mask for k out of bounds
        mask_k = k < SEQ_LEN

        # pointers
        left_ptrs = left_ptr + b * stride_l_b + i[:, None] * stride_l_i + k[None, :] * stride_l_k + d * stride_l_h
        right_ptrs = right_ptr + b * stride_r_b + j[:, None] * stride_r_i + k[None, :] * stride_r_k + d * stride_r_h

        left_tile = tl.load(left_ptrs, mask=mask_k[None, :], other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_k[None, :], other=0.0)

        # accumulate dot product (left_tile shape MxK, right_tile shape NxK)
        acc += tl.dot(left_tile, right_tile, trans_b=True)

    # Apply mask for output indices beyond N
    mask_i = i < SEQ_LEN
    mask_j = j < SEQ_LEN
    mask = mask_i[:, None] & mask_j[None, :]

    # Store result
    out_ptrs = out_ptr + b * stride_o_b + i[:, None] * stride_o_i + j[None, :] * stride_o_j + d * stride_o_h
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask)

Triton implementation of the outgoing Triangle Multiplicative Update (TriMul).

The kernel computes, for each batch element `b` and hidden channel `d`,
    out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]

):
    pid_i = tl.program_id(0)      # tile row index
    pid_j = tl.program_id(1)      # tile col index
    pid_bh = tl.program_id(2)     # combined batch * hidden index

    # start indices of the tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # decode batch and hidden channel from the flattened index
    b = pid_bh // H
    d = pid_bh % H

    # range vectors
    range_i = tl.arange(0, BLOCK_M)
    range_j = tl.arange(0, BLOCK_N)
    range_k = tl.arange(0, BLOCK_K)

    i = i0 + range_i                      # absolute i indices
    j = j0 + range_j                      # absolute j indices

    # masks for out‑of‑bounds i/j (used for loading and storing)
    mask_i = i < SEQ_LEN
    mask_j = j < SEQ_LEN

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in tiles
    for kk in range(0, tl.cdiv(SEQ_LEN, BLOCK_K)):
        k0 = kk * BLOCK_K
        k = k0 + range_k                  # absolute k indices
        mask_k = k < SEQ_LEN

        # ---- left tile: shape (BLOCK_M, BLOCK_K) ----
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + i[:, None] * stride_l_i
            + k[None, :] * stride_l_k
            + d * stride_l_h
        )
        left_mask = mask_i[:, None] & mask_k[None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0)

        # ---- right tile (transposed for GEMM): shape (BLOCK_K, BLOCK_N) ----
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + k[:, None] * stride_r_k
            + j[None, :] * stride_r_i
            + d * stride_r_h
        )
        right_mask = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0)

        # outer‑product accumulation (matrix multiplication)
        acc += tl.dot(left_tile, right_tile)   # (M×K)·(K×N) → (M×N)

    # ---- store the result -------------------------------------------------
    out_mask = mask_i[:, None] & mask_j[None, :]
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + i[:, None] * stride_o_i
        + j[None, :] * stride_o_j
        + d * stride_o_h
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=out_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm on the input ----------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # cast to fp16 for the compute‑heavy path
    x = x.to(torch.float16)

    # ---------- Load linear weights ----------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # ---------- Linear projections ----------
    left = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w) # (B, N, N, H)

    # ---------- Gates (sigmoid) ----------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ---------- Apply mask (if any) and fuse with gates ----------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask_fp
        right = right * right_gate * mask_fp
    else:
        left = left * left_gate
        right = right * right_gate

    # ---------- O(N³·H) reduction via Triton ----------
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # allocate fp16 output tensor

    # strides (must be Python ints)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tile sizes – tuned for the B200
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _tri_mul_kernel[grid](
        left, right, out,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        B, H,
        SEQ_LEN=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection back to dim ----------
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # cast back to the original dtype
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
