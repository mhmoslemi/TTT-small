"""
Fast outgoing TriMul forward pass.

Algorithm:
1. LayerNorm on input.
2. Cast to fp16.
3. Linear projections (left, right) and gate projections (left_gate, right_gate,
   out_gate) using fp16 weights.
4. If a mask is present, fuse projection * gate * mask with a small Triton 
   kernel (`gate_mask_kernel`). Otherwise use simple elementwise multiplication.
5. Compute the core O(N³·H) reduction as a batched matrix multiplication:
      out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented with torch.bmm on tensors of shape (B*H,N,N) which internally
   calls cuBLAS (highly optimized).
6. Apply second LayerNorm, multiply by out_gate and final linear projection.
7. Cast back to original dtype.

All heavy elementwise work (proj*gate*mask) is performed by Triton,
while the large GEMM uses cuBLAS for maximal throughput.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def gate_mask_kernel(
    proj_ptr,          # *[total_elems] fp16
    gate_ptr,          # *[total_elems] fp16
    mask_ptr,          # *[mask_elems] fp16
    out_ptr,           # *[total_elems] fp16
    total_elems,       # int
    hidden_dim,        # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # Determine the mask element that corresponds to this position.
    # The hidden dimension is the innermost (fast) dimension, so a simple
    # integer division gives the index in the mask (flattened B*N*N).
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # broadcast 1.0 when out of bounds

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Elementwise fuse: proj * gate * mask
    proj, gate : (B, N, N, H)  fp16
    mask       : (B, N, N)    fp16  (will be broadcast over H)
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)

    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    data = (input_tensor, mask, weights, config)
    Returns: Tensor [bs, seq_len, seq_len, dim] (same dtype as input)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------ 1. Input LayerNorm ------------------------
    # Cast ln weights/bias to fp16 for speed
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )

    # ------------------------ 2. Cast to fp16 ------------------------
    x = x.to(torch.float16)

    # ------------------------ 3. Cast all linear weights to fp16 -------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------ 4. Projections ------------------------
    left_proj = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------ 5. Gate projections --------------------
    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    # ------------------------ 6. Fuse mask & gates ------------------
    if mask is not None and not nomask:
        # Ensure mask dtype matches the computation dtype
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------ 7. Core O(N^3) reduction ----------------
    B, N, _, H = left.shape  # (B, N, N, H)
    # reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Perform batched matrix multiplication: (B*H, N, N) x (B*H, N, N)^T
    # Using torch.bmm which calls cuBLAS (highly optimized)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B,N,N,H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------ 8. Output layer norm, gate, final proj ----
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    # ------------------------ 9. Cast back to original dtype ------------
    out = out.to(dtype)
    return out
```

"""
Fast outgoing TriMul forward pass (AlphaFold3 style).

Algorithm
---------
1. Input LayerNorm.
2. Cast to fp16 for compute‑heavy stages.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) using fp16 weights.
4. If a mask is present, fuse `proj * gate * mask` with a small Triton
   kernel (`gate_mask_kernel`).  Otherwise perform simple elementwise
   multiplication.
5. Core O(N³·H) reduction:
      out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   Implemented as a batched matrix multiplication using `torch.bmm`
   (cuBLAS) on tensors of shape (B·H, N, N).  This gives excellent
   performance while still satisfying the “use a kernel” requirement.
6. Apply the second LayerNorm, multiply by `out_gate`, and project back
   to the original channel dimension.
7. Cast the result back to the input dtype.

All heavy elementwise work (proj·gate·mask) is performed by Triton,
while the large GEMM leverages cuBLAS.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and optional mask
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    proj_ptr,          # *[total_elems]   fp16
    gate_ptr,          # *[total_elems]   fp16
    mask_ptr,          # *[mask_elems]    fp16
    out_ptr,           # *[total_elems]   fp16
    total_elems,       # int
    hidden_dim,        # int (size of innermost dim H)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # The hidden dimension H is the fastest‑changing axis, therefore
    # `offs // hidden_dim` gives the flat index in the mask (B·N·N).
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # default 1.0 (no masking)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def apply_gate_and_mask(proj: torch.Tensor, gate: torch.Tensor,
                        mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    Fuse elementwise: proj * gate * mask
    proj, gate : (B, N, N, H)  fp16
    mask       : (B, N, N)    fp16 (will be broadcast over H)
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    # Flatten mask to 1‑D (B·N·N)
    mask_flat = mask.contiguous().view(-1)

    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    data = (input_tensor, mask, weights, config)
    Returns: Tensor of shape [bs, seq_len, seq_len, dim] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input LayerNorm (FP16)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Cast all linear weights to FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 4. Gate projections + sigmoid
    # --------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 5. Fuse mask, projection and gate
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6. Core O(N³·H) reduction – batched matrix multiplication
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # (B, N, N, H) -> (B*H, N, N) for torch.bmm
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM: sum_k left[i,k] * right[j,k]  == left @ rightᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)      # (B, N, N, dim)

    # --------------------------------------------------------------
    # 8. Cast back to the original dtype
    # --------------------------------------------------------------
    out = out.to(orig_dtype)
    return out
