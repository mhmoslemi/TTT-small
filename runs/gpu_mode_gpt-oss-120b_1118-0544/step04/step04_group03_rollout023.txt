@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    batch = tl.program_id(1)
    # compute tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    # mask
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch_ptr = a_ptr + batch * stride_ab
    b_batch_ptr = b_ptr + batch * stride_bb
    c_batch_ptr = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K
        # Load A: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch_ptr + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B^T: we have B stored as (N, K), we need B^T shape (K,N)
        # If B is stored as (K,N) we load directly, else we load transposed.
        # For our use, B is (K,N) where K=N.
        b = tl.load(
            b_batch_ptr + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)
    # Store result
    c = acc.to(tl.float16)
    tl.store(
        c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )

"""
custom_kernel: Triton‑accelerated outgoing TriMul (Triangle Multiplicative Update).

The implementation follows the PyTorch reference:
1. LayerNorm on the input.
2. Two fused linear layers (projection + gate) for the left and right paths.
3. Optional masking of the projected tensors.
4. Core O(N³·H) reduction implemented as a batched GEMM:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   (i.e. a batched matrix multiplication left @ rightᵀ for each hidden channel
   using a custom Triton kernel with float16 arithmetic and float32 accumulation).
5. Hidden‑dim LayerNorm, output gate and final linear projection back to the
   original channel dimension.
All heavy arithmetic (the N³ reduction) is performed in Triton, while the
remaining lightweight operations use PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM C = A @ Bᵀ
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (M×K)·(K×N) → (M×N)
    stride_am, stride_ak,              # strides for A (rows, cols)
    stride_bk, stride_bn,              # strides for B (interpreted as K×N)
    stride_cm, stride_cn,              # strides for C (rows, cols)
    stride_ab, stride_bb, stride_cb,   # strides for the batch dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM: C = A @ Bᵀ.
    A: (batch, M, K), B: (batch, K, N) (stored as original (batch, N, K) and
    accessed transposed). Accumulates in float32 and writes float16."""
    pid = tl.program_id(0)          # tile index over output (M×N)
    batch = tl.program_id(1)        # batch index (B * hidden_dim)

    # Compute tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # Load A tile (M × K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load Bᵀ tile (K × N) – B is stored as (N, K), we access transposed view
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp16·fp16 → fp32

    # Write result
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[:, None],
    )


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    orig_dtype = x.dtype
    device = x.device

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (float32 for stability, then cast to half)
    # ------------------------------------------------------------------- #
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 2. Prepare half‑precision weights
    # ------------------------------------------------------------------- #
    dtype_h = torch.float16
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype_h)          # (2*H, C)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype_h)          # (2*H, C)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype_h)   # (H, C)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_h)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_h)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_h)       # (C, H)

    # ------------------------------------------------------------------- #
    # 3. Linear projections + gates (combined)
    # ------------------------------------------------------------------- #
    # Combined linear gives [B, N, N, 2*H]
    left_comb = F.linear(x_h, left_comb_w)      # (B, N, N, 2*H)
    right_comb = F.linear(x_h, right_comb_w)    # (B, N, N, 2*H)

    # Split projection & gate
    left_proj = left_comb[..., :hidden_dim]                     # (B, N, N, H)
    left_gate = left_comb[..., hidden_dim:].sigmoid()           # (B, N, N, H)
    right_proj = right_comb[..., :hidden_dim]                  # (B, N, N, H)
    right_gate = right_comb[..., hidden_dim:].sigmoid()        # (B, N, N, H)

    # Optional masking (mask broadcast over hidden dimension)
    if mask is not None and not nomask:
        mask_f = mask.to(dtype_h).unsqueeze(-1)   # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------- #
    # 4. Core O(N³·H) reduction: batched GEMM left @ rightᵀ
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape   # (B, N, N, H)

    # Rearrange to (batch = B*H, M = N, K = N)
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left_t.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)   # (B*H, N, N), half

    # Strides (in elements)
    stride_a_m = left_flat.stride(1)   # row stride of A
    stride_a_k = left_flat.stride(2)   # col stride of A
    stride_b_k = right_flat.stride(2)  # row stride of Bᵀ (original col stride)
    stride_b_n = right_flat.stride(1)  # col stride of Bᵀ (original row stride)
    stride_c_m = out_flat.stride(1)    # row stride of C
    stride_c_n = out_flat.stride(2)    # col stride of C

    stride_ab = left_flat.stride(0)    # batch stride for A
    stride_bb = right_flat.stride(0)   # batch stride for B
    stride_cb = out_flat.stride(0)     # batch stride for C

    # Choose block sizes (tuned for N up to 1024)
    BLOCK_M = 64 if N <= 256 else 128
    BLOCK_N = BLOCK_M
    BLOCK_K = 64

    # Grid: (tiles over M×N, batch)
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,               # M
        N,               # N (output cols)
        N,               # K
        stride_a_m,
        stride_a_k,
        stride_b_k,
        stride_b_n,
        stride_c_m,
        stride_c_n,
        stride_ab,
        stride_bb,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 5. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------- #
    # 6. Output gate (element‑wise) and final projection
    # ------------------------------------------------------------------- #
    out_gate = F.linear(x_h, out_gate_w).sigmoid()   # (B, N, N, H)
    out = out * out_gate                              # element‑wise, half precision

    out = F.linear(out, to_out_w)                     # (B, N, N, C), half

    # ------------------------------------------------------------------- #
    # 7. Cast back to original dtype and return
    # ------------------------------------------------------------------- #
    out = out.to(orig_dtype)
    return out
