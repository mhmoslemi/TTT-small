    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,

    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

    batch = tl.program_id(1)

    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_n = offs_n < N

    a_batch = a_ptr + batch * stride_batch_a
    b_batch = b_ptr + batch * stride_batch_b

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the reference PyTorch module but fuses the costly
O(N³·H) reduction into a single batched half‑precision GEMM kernel.
The algorithm:
1) Layer‑norm the input.
2) Jointly compute left/right projections and their gates (two linear layers
   concatenated) in float16.
3) Apply optional mask and element‑wise gate multiplication.
4) Reshape to (B·H, N, N) and run a custom Triton kernel that computes
   out_h = left_h @ right_hᵀ for every hidden channel h (batched GEMM).
5) Layer‑norm on the hidden dimension, apply an output gate and a final
   projection back to the original channel dimension.
All heavy arithmetic is performed in float16 with float32 accumulation; the
final result is cast back to the original input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
# A, B: (batch, M, K)  (here M = K = N)
# C: (batch, M, N)
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions
    stride_am, stride_ak,         # A strides (row, col)
    stride_bk, stride_bn,         # B strides (row, col) – B is interpreted as (K, N)
    stride_cm, stride_cn,         # C strides (row, col)
    stride_batch_a, stride_batch_b, stride_batch_c,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)            # tile index (i‑j)
    batch = tl.program_id(1)          # batch index (B·H)

    # tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch * stride_batch_a
    b_batch = b_ptr + batch * stride_batch_b
    c_batch = c_ptr + batch * stride_batch_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in BLOCK_K steps
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B block loaded as transposed: (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)          # FP16·FP16 → FP32 accumulation

    # write result
    c = acc.to(tl.float16)

        c,
        mask=mask_m[:, None] & mask_n[None, :],

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------- #
    # unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    orig_dtype = input_tensor.dtype
    device = input_tensor.device

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 2. Cast to float16 for the heavy path
    # ------------------------------------------------------------------- #
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 3. Linear projections + gates (combined)
    # ------------------------------------------------------------------- #
    # concatenate proj and gate weights for left and right paths
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)               # (2*H, C)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)               # (2*H, C)

    # joint linear passes
    left_comb = F.linear(x_fp16, left_comb_w)   # [B, N, N, 2*H]
    right_comb = F.linear(x_fp16, right_comb_w) # [B, N, N, 2*H]

    # split into proj / gate
    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # optional masking (broadcast over hidden dim)
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------- #
    # 4. Core O(N³·H) reduction via Triton batched GEMM
    # ------------------------------------------------------------------- #
    # reshape to (B*H, N, N) to treat each hidden channel as an independent batch
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.empty_like(left_flat)  # (B*H, N, N), float16

    # strides needed by the kernel
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    stride_batch_a = left_flat.stride(0)
    stride_batch_b = right_flat.stride(0)
    stride_batch_c = out_flat.stride(0)

    # tile sizes – tuned for typical sequence lengths up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * hidden_dim)

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,   # M
        N,   # N
        N,   # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_batch_a,
        stride_batch_b,
        stride_batch_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 5. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(

        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
