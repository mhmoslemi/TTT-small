```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
import math

# Gate & mask kernel
@triton.jit
def gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    # compute mask index by integer division
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

def apply_gate_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor, block_size: int = 1024):
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (math.ceil(total / block_size),)

    mask_flat = mask.view(-1)
    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out

# Batched GEMM kernel for TriMul core
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)  # batch dimension
    pid_m = tl.program_id(0)  # M tile
    pid_n = tl.program_id(1)  # N tile

    # Offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_base = a_ptr + pid_b * stride_a_batch
    b_base = b_ptr + pid_b * stride_b_batch
    c_base = c_ptr + pid_b * stride_c_batch

    # Masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block (M x K)
        a_offset = offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(
            a_base + a_offset,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B block transposed (K x N)
        b_offset = offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(
            b_base + b_offset,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c_offset = offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_base + c_offset, acc.to(tl.float16), mask=store_mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operation.

    Steps:
    1. Layer‑norm the input (float32 → fp16).
    2. Linear projections (left/right) and gate projections (left/right/out).
    3. Fuse mask, projection and gate with a small Triton kernel.
    4. Core O(N³·H) reduction implemented as a batched GEMM:
           out = left @ rightᵀ   with shape (B, N, N, H)
    5. Second layer‑norm, output‑gate and final linear projection.
    6. Cast back to the original dtype (typically float32).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (with sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask and fuse with gate (if mask provided)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction via batched GEMM (Triton)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # (B, N, N, H)

    # Rearrange to (B*H, N, N) where the hidden dimension is fused into the batch
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)

    batch = B * H
    M = N
    K = N
```

"""
Triton‑accelerated forward pass of the outgoing TriMul operation.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → fp16 for compute).
2. Linear projections (left/right) and gate projections (left/right/out) using
   fp16 weights.
3. Fuse mask, projection and gate with a small Triton pointwise kernel.
4. Core O(N³·H) reduction implemented as a batched GEMM:
       out = left @ rightᵀ   (shape (B, N, N, H)).
5. Second layer‑norm, output‑gate and final linear projection.
6. Cast back to the original dtype (typically float32) and return.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: elementwise proj * gate * mask  (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    proj_ptr,            # *[total_elems]  fp16
    gate_ptr,            # *[total_elems]  fp16
    mask_ptr,            # *[mask_elems]  fp16
    out_ptr,             # *[total_elems]  fp16
    total_elems,         # int
    hidden_dim,          # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # each mask element is repeated `hidden_dim` times in the flattened tensors
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mval = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mval
    tl.store(out_ptr + offs, out, mask=valid)


def apply_gate_mask(proj: torch.Tensor, gate: torch.Tensor,
                    mask: torch.Tensor, block_size: int = 1024) -> torch.Tensor:
    """
    proj : (B, N, N, H)   fp16
    gate : (B, N, N, H)   fp16
    mask : (B, N, N)      fp16
    returns proj * gate * mask (fp16)
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (math.ceil(total / block_size),)

    mask_flat = mask.view(-1)  # (B*N*N,)
    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: batched GEMM for the O(N³) core (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to flattened tensors
    batch, M, N, K,                     # integer sizes (batch = B*H)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch dimension
    pid_m = tl.program_id(0)   # output M tile
    pid_n = tl.program_id(1)   # output N tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_base = a_ptr + pid_b * stride_a_batch
    b_base = b_ptr + pid_b * stride_b_batch
    c_base = c_ptr + pid_b * stride_c_batch

    # boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A block: (M, K)
        a_off = offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(
            a_base + a_off,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # B block transposed: (K, N) because we need rightᵀ
        b_off = offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(
            b_base + b_off,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)   # FP32 accumulation

    # write back
    c_off = offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_base + c_off, acc.to(tl.float16), mask=store_mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict with the model weights
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the compute‑heavy path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections to hidden dimension (fp16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections with sigmoid (fp16)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask (if present) and fuse with gate using Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = apply_gate_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction – batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # Collapse the hidden dimension into the batch dimension for the kernel
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat   = torch.empty_like(left_flat)

    batch = B * H
    M = N
    K = N

    # Strides (in elements)
    stride_a_batch = left_flat.stride(0)
    stride_a_m     = left_flat.stride(1)
    stride_a_k     = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_n     = right_flat.stride(1)  # rows (j)
    stride_b_k     = right_flat.stride(2)  # cols (k)

    stride_c_batch = out_flat.stride(0)
    stride_c_m     = out_flat.stride(1)
    stride_c_n     = out_flat.stride(2)

    # Tuned block sizes (good for N up to 1024 on modern GPUs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        math.ceil(M / BLOCK_M),
        math.ceil(N / BLOCK_N),
        batch,
    )

    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype (float32) and return
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)
    return out
