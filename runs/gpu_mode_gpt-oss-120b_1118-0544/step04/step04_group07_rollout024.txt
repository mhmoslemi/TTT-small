def apply_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,    # optional [B,N,N] flatten
    left_out_ptr, right_out_ptr,
    total_elems, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # load
    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    # compute
    left = left_proj * left_gate
    right = right_proj * right_gate
    if HAS_MASK:
        # mask index is offset // hidden_dim
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * mask_val
        right = right * mask_val
    # store
    tl.store(left_out_ptr + offs, left, mask=mask)
    tl.store(right_out_ptr + offs, right, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # layernorm
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)
    # fused projection
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    # Fused weight
    fused_weight = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0)  # shape (5*hidden, dim)
    # flatten input
    x_flat = x.view(-1, dim)  # (B*N*N, dim)
    proj = torch.nn.functional.linear(x_flat, fused_weight) # (B*N*N, 5*hidden)
    # split slices
    H = hidden_dim
    left_proj = proj[:, :H].view(bs, N, N, H)
    right_proj = proj[:, H:2*H].view(bs, N, N, H)
    left_gate = proj[:, 2*H:3*H].view(bs, N, N, H)
    right_gate = proj[:, 3*H:4*H].view(bs, N, N, H)
    out_gate = proj[:, 4*H:].view(bs, N, N, H)
    # activations
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)
    # Apply mask and gates via kernel if mask needed
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        # flatten for kernel
        left_flat = left_proj.contiguous().view(-1)
        right_flat = right_proj.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat = mask_fp.squeeze(-1).contiguous().view(-1)  # (B*N*N,)
        total = left_flat.numel()

            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            H,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )

        right = right_flat.view_as(right_proj)
    else:

        right = right_proj * right_gate
    # Core matmul
    # permute to (B,H,N,N)
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    # matmul
    out_mat = torch.matmul(left_t, right_t.transpose(-1, -2))
    # back to (B,N,N,H)
    out = out_mat.permute(0, 2, 3, 1).contiguous()
    # to_out_norm
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    out_flat = out.view(-1, hidden_dim)

    if out_proj.dtype != input_tensor.dtype:

    return out_proj

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The implementation follows the PyTorch reference but reduces kernel launch
overhead by:
1. Layer‑norm in float32 (standard torch implementation).
2. A single fused half‑precision GEMM that computes all five linear
   projections (left/right value and three gates) at once.
3. A tiny Triton kernel that fuses gate multiplication and optional mask
   broadcasting in‑place.
4. Batched half‑precision matrix multiplication (B×H independent N×N
   products) performed with torch.matmul (cuBLAS) – the dominant O(N³·H)
   work.
5. Final layer‑norm, output‑gate and projection back to the input dimension.

All heavy arithmetic is done in float16 to exploit tensor‑cores; the
result is cast back to the input dtype (float32 by default).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr,                # [B,N,N,H] flattened, will be overwritten with left result
    right_ptr,               # [B,N,N,H] flattened, will be overwritten with right result
    left_gate_ptr,           # [B,N,N,H] flattened
    right_gate_ptr,          # [B,N,N,H] flattened
    mask_ptr,                # [B,N,N] flattened, broadcast over H
    total_elems,             # total number of elements in left/right tensors
    H: tl.constexpr,         # hidden dimension (compile‑time)
    HAS_MASK: tl.constexpr,  # whether mask is present
    BLOCK_SIZE: tl.constexpr # block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # load values
    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # optional mask (shared over H)
    if HAS_MASK:
        mask_idx = offs // H                     # each H‑wide slice shares the same mask entry
        mask_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        left = left * mask_val
        right = right * mask_val

    # store results in‑place
    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape (B, N, N, C)   (float32)
            - mask         : torch.Tensor of shape (B, N, N) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as input_tensor.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)   # [B,N,N,dim] float32

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the heavy linear work
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Fuse all five linear projections (left, right, three gates)
    # -----------------------------------------------------------------
    # gather weights and cast to fp16
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    fused_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape (5*H, dim)

    # fused GEMM: (B*N*N, dim) x (dim, 5*H)^T -> (B*N*N, 5*H)
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)
    proj = F.linear(x_flat, fused_weight)        # (B*N*N, 5*H)

    H = hidden_dim
    # split projections
    left_proj   = proj[:, :H].view(bs, N, N, H)
    right_proj  = proj[:, H:2 * H].view(bs, N, N, H)
    left_gate   = proj[:, 2 * H:3 * H].view(bs, N, N, H)
    right_gate  = proj[:, 3 * H:4 * H].view(bs, N, N, H)
    out_gate    = proj[:, 4 * H:].view(bs, N, N, H)

    # sigmoid gates
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # -----------------------------------------------------------------
    # 4. Apply mask and gates (fused Triton kernel when mask is present)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # mask → fp16 and flatten (no channel dim)
        mask_fp = mask.to(device).to(torch.float16).contiguous()
        mask_flat = mask_fp.view(-1)                     # (B*N*N,)

        # flatten tensors for kernel
        left_flat = left_proj.contiguous().view(-1)
        right_flat = right_proj.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

        _gate_mask_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            H,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )

        left = left_flat.view_as(left_proj)
        right = right_flat.view_as(right_proj)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5. Core O(N³·H) reduction – batched matmul over the sequence dimension
    # -----------------------------------------------------------------
    # reshape to (B, H, N, N) for batched matmul
    left_mat = left.permute(0, 3, 1, 2).contiguous()
    right_mat = right.permute(0, 3, 1, 2).contiguous()

    # (B, H, N, N) @ (B, H, N, N)^T  → (B, H, N, N)
    out_mat = torch.matmul(left_mat, right_mat.transpose(-1, -2))

    # back to (B, N, N, H)
    out = out_mat.permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm (half‑precision)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # -----------------------------------------------------------------
    # 7. Apply output gate
    # -----------------------------------------------------------------
    out = out * out_gate

    # -----------------------------------------------------------------
    # 8. Final linear projection back to the model dimension
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)   # shape (dim, hidden_dim)
    out_flat = out.view(-1, hidden_dim)      # (B*N*N, hidden_dim)
    out = F.linear(out_flat, to_out_w)       # (B*N*N, dim)
    out = out.view(bs, N, N, dim)

    # -----------------------------------------------------------------
    # 9. Cast back to the original dtype
    # -----------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
