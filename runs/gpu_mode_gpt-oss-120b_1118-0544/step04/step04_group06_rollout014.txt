@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    mask_ptr, # optional mask, shape (B,N,N)
    # strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, HAS_MASK: tl.constexpr,
):
    # program id mapping
    pid = tl.program_id(0)
    # compute batchhead index
    num_blocks_i = tl.cdiv(N, BLOCK_M)
    num_blocks_j = tl.cdiv(N, BLOCK_N)
    blocks_per_head = num_blocks_i * num_blocks_j

    bh = pid // blocks_per_head
    block_idx = pid % blocks_per_head

    b = bh // H
    h = bh % H

    pid_i = block_idx // num_blocks_j
    pid_j = block_idx % num_blocks_j

    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    i_mask = i_offset < N
    j_mask = j_offset < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k tiles
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        k_off = k_start + k_offset
        k_mask = k_off < N

        # Load left tile (i,k)
        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + i_offset[:,None]*stride_left_i + (k_off[None,:])*stride_left_k
        left = tl.load(left_ptrs, mask=i_mask[:,None] & k_mask[None,:], other=0.0)

        # Load right tile (k,j) transposed shape
        right_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + (k_off[:,None])*stride_right_k + j_offset[None,:]*stride_right_j
        right = tl.load(right_ptrs, mask=k_mask[:,None] & j_mask[None,:], other=0.0)

        # Optional mask: multiply left and right by mask values.
        if HAS_MASK:
            # mask left: mask[b,i,k]; mask right: mask[b,j,k]
            mask_left_ptrs = mask_ptr + b*stride_mask_b + i_offset[:,None]*stride_mask_i + k_off[None,:]*stride_mask_k
            mask_left = tl.load(mask_left_ptrs, mask=i_mask[:,None] & k_mask[None,:], other=1.0)

            mask_right_ptrs = mask_ptr + b*stride_mask_b + j_offset[:,None]*stride_mask_i + k_off[None,:]*stride_mask_k
            # note: mask uses same strides for i and j axis (since shape B,N,N)
            mask_right = tl.load(mask_right_ptrs, mask=j_mask[:,None] & k_mask[None,:], other=1.0)

            # Multiply
            left = left * mask_left
            right = right * mask_right

        # Compute dot product
        acc += tl.dot(left, right)  # left: (M,K); right: (K,N)

    # Write result
    out_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + i_offset[:,None]*stride_out_i + j_offset[None,:]*stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:,None] & j_mask[None,:])

    # compute block indices

    num_j = tl.cdiv(N, BLOCK_N)

    bh = pid // blocks_per_head

    h = bh % H
    pid_i = block // num_j
    pid_j = block % num_j

    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_off = tl.arange(0, BLOCK_K)

    j_mask = j_off < N

"""
Highly‑optimized forward pass for the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑Normalize the input tensor.
2. Cast to fp16 for compute‑heavy kernels.
3. Fuse the four linear projections (left/right and their gates) plus the
   output‑gate into a single GEMM (C → 5·H).  Split the result into the five
   tensors and apply sigmoid to the three gate tensors.
4. Fuse the mask (if present) with the gate‑scaled projections using a
   single element‑wise multiplication (pure PyTorch, cheap).
5. Core O(B·H·N³) reduction:
     out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   implemented as a tiled Triton kernel that computes a batched matrix‑multiply
   of shape (B·H, N, N) × (B·H, N, N)ᵀ.
6. Apply the second LayerNorm, multiply by the output‑gate and project back to
   the original channel dimension.
7. Cast the result back to the input dtype.

Only the reduction (step 5) is performed in Triton; all other operations use
high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _tri_mul_reduce_kernel(
    left_ptr,               # fp16 *
    right_ptr,              # fp16 *
    out_ptr,                # fp16 *
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] for a single
    (b,h) pair.  Each program instance processes a tile of size
    (BLOCK_M, BLOCK_N) over the (i, j) plane.
    """
    pid = tl.program_id(0)

    # ---- program‑id decomposition -------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    blocks_per_head = num_i * num_j                # tiles for a given (b,h)

    bh = pid // blocks_per_head                     # flattened (b,h) index
    tile_id = pid % blocks_per_head

    b = bh // H
    h = bh % H

    pid_i = tile_id // num_j
    pid_j = tile_id % num_j

    # ---- tile offsets ------------------------------------------------------------
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_off = tl.arange(0, BLOCK_K)

    i_mask = i_off < N
    j_mask = j_off < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- reduction over k ---------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + k_off
        k_mask = k < N

        # left tile   : (M, K)  -> left[b,i,k,h]
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + i_off[:, None] * stride_left_i
            + k[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        )

        # right tile  : (K, N)  -> right[b,j,k,h]  (transposed for dot)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + k[:, None] * stride_right_k
            + j_off[None, :] * stride_right_j
        )
        right = tl.load(
            right_ptrs,
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0,
        )

        acc += tl.dot(left, right)                 # (M,K) × (K,N) → (M,N)

    # ---- write result -------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + i_off[:, None] * stride_out_i
        + j_off[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C] (float32/float16)
            - mask: [B, N, N] (bool/int/float) or None
            - weights: dict of model parameters
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=orig_dtype)
    norm_bias = weights["norm.bias"].to(device=device, dtype=orig_dtype)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (C,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Fuse the four projections + output gate into a single linear
    # ------------------------------------------------------------------
    # weight shapes: (hidden_dim, C)
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    # concatenate -> (5*hidden_dim, C)
    fused_w = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)

    # linear: (B*N*N, C) @ (C, 5*H)ᵀ  → (B*N*N, 5*H)
    x_flat = x.view(-1, C)                     # (B*N*N, C)
    fused_out = torch.nn.functional.linear(x_flat, fused_w)  # fp16
    fused_out = fused_out.view(B, N, N, 5 * hidden_dim)

    # split
    left_proj = fused_out[..., 0:hidden_dim]
    left_gate = fused_out[..., hidden_dim:2 * hidden_dim].sigmoid()
    right_proj = fused_out[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate = fused_out[..., 3 * hidden_dim:4 * hidden_dim].sigmoid()
    out_gate = fused_out[..., 4 * hidden_dim:5 * hidden_dim].sigmoid()

    # ------------------------------------------------------------------
    # 4. Apply gates and optional mask
    # ------------------------------------------------------------------
    left = (left_proj * left_gate)
    right = (right_proj * right_gate)

    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 5. Core reduction via Triton
    # ------------------------------------------------------------------
    out = torch.empty_like(left)  # (B,N,N,hidden_dim), fp16

    # strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_k, stride_o_h = out.stride()

    # Triton block sizes (tuned for typical N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * hidden_dim * num_i * num_j

    grid = (total_blocks,)

    _tri_mul_reduce_kernel[grid](
        left,
        right,
        out,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,   # this is the stride for the “j” axis (first N)
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_k,
        stride_o_h,
        B,
        N,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,C)

    # ------------------------------------------------------------------
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)
    return out
