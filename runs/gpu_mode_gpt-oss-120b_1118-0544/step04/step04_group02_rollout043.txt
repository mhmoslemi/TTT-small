"""
custom_kernel: high‑performance outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension `dim`.
2. Cast to FP16 for the compute‑heavy path.
3. Linear projections (left/right) and three gating linear layers are performed
   with FP16 weights. Gate outputs are passed through a sigmoid.
4. Optional mask (broadcast over the hidden dimension) is fused with the
   projection and gate by simple element‑wise multiplication (still in PyTorch;
   the kernel requirement is satisfied by the GEMM kernel).
5. Core O(N³·H) reduction:
   For each batch `b` and hidden index `h` we compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   i.e. a batched matrix multiplication `left @ rightᵀ` where each matrix is
   (N×N).  This is implemented by a single Triton kernel that avoids any
   costly permute/copy; it works directly on the original `[B,N,N,H]` layout
   using the provided strides.
6. Convert the result to FP32, apply a second LayerNorm over `hidden_dim`,
   multiply by the output gate, and project back to the original channel
   dimension with a final linear layer.
7. Return a tensor of shape `[B, N, N, dim]` having the same dtype as the
   input.

The kernel uses FP16 arithmetic with FP32 accumulation and 128‑bit/32‑bit
tiling (BLOCK_M=64, BLOCK_N=64, BLOCK_K=32), which maps well to Tensor‑Core
execution on modern NVIDIA GPUs.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) core of TriMul
# Computes: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers to fp16 tensors of shape [B,N,N,H]
    B, N, H,                      # static sizes
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program IDs
    pid_batch = tl.program_id(2)   # combined (batch, hidden) index
    pid_m = tl.program_id(0)       # tile row (i)
    pid_n = tl.program_id(1)       # tile col (j)

    # Decompose combined index into batch and hidden indices
    b_idx = pid_batch // H
    h_idx = pid_batch % H

    # Compute base pointers for this (b,h) slice
    a_base = a_ptr + b_idx * stride_a_batch + h_idx * stride_a_h
    b_base = b_ptr + b_idx * stride_b_batch + h_idx * stride_b_h
    c_base = c_ptr + b_idx * stride_c_batch + h_idx * stride_c_h

    # Offsets for the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension (the reduction axis)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block: shape (BLOCK_M, BLOCK_K) -> left[b,i,k,h]
        a_off = offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(
            a_base + a_off,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B block transposed: we need right[b,j,k,h]ᵀ -> (k,j)
        # B layout: (batch, i, k, h) where i ↔ j in the formula
        b_off = offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i
        b = tl.load(
            b_base + b_off,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # Accumulate product
        acc += tl.dot(a, b)   # result cast to FP32 when added to acc

    # Write results back to C: shape (BLOCK_M, BLOCK_N)
    c_off = offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    mask = mask_i[:, None] & mask_j[None, :]
    tl.store(c_base + c_off, acc.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Helper to launch the Triton GEMM kernel
# ----------------------------------------------------------------------
def batched_gemm_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: [B, N, N, H]   (fp16, contiguous)
    returns out with same shape and dtype.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides in elements (not bytes)
    a_batch, a_i, a_k, a_h = left.stride()
    b_batch, b_i, b_k, b_h = right.stride()
    c_batch, c_i, c_j, c_h = out.stride()

    # Tile sizes (tuned for Tensor Cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
        B * H,
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        a_batch,
        a_i,
        a_k,
        a_h,
        b_batch,
        b_i,
        b_k,
        b_h,
        c_batch,
        c_i,
        c_j,
        c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive stages
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # 4. Linear projections (B,N,N,dim) -> (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)

    # ------------------------------------------------------------------

    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6. Apply mask (if provided) and combine with gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure tensors are contiguous for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction – batched GEMM via Triton
    # ------------------------------------------------------------------
    out = batched_gemm_tri_mul(left, right)   # (B,N,N,H), fp16

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, gate and final projection (FP32 path)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out_gated = out_norm * out_gate.to(torch.float32)
    out_final = F.linear(out_gated, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype of the input
    # ------------------------------------------------------------------
    out_final = out_final.to(input_tensor.dtype)
    return out_final
