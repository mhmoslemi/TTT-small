The implementation follows the reference PyTorch TriMul forward pass and
fuses the dominant O(N³·H) contraction into a single Triton batched GEMM
kernel.  The steps are:

1. Layer‑norm over the input channel dimension.
2. Combined linear + gate projections (left/right) in half‑precision.
3. Optional masking of the projected tensors.
4. Batched matrix multiplication:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

5. Hidden‑dim LayerNorm, output gate, and final projection back to the
   original channel dimension.
All heavy arithmetic uses float16 with float32 accumulation.
"""

import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for left @ rightᵀ  (A @ Bᵀ)
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dims (M×K)·(K×N)→(M×N)
    stride_am, stride_ak,              # A strides: rows (M), cols (K)
    stride_bk, stride_bn,              # B strides: K (inner), N (outer)
    stride_cm, stride_cn,              # C strides: rows (M), cols (N)
    stride_ab, stride_bb, stride_cb,   # batch strides for A, B, C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched half‑precision GEMM:
        C = A @ Bᵀ
    A: (batch, M, K)  – row‑major
    B: (batch, N, K)  – row‑major (we treat it as transposed inside)
    C: (batch, M, N)
    """
    pid = tl.program_id(0)          # tile index over M‑N plane
    batch = tl.program_id(1)        # batch index (B·H)

    # ------------------- tile coordinates ------------------- #
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------- base pointers for this batch ------------------- #
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    # ------------------- accumulation ------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- k‑loop ------------------- #
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # Load A: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load Bᵀ: we want Bᵀ[k, n] = B[n, k]
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp16·fp16 → fp32

    # ------------------- store result ------------------- #
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None (binary)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------- #
    # 0. Unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    device = x.device
    orig_dtype = x.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm over channel dim (C)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Work in FP16 for the heavy part
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 2. Linear projections + gates (combined for fewer matmuls)
    # ------------------------------------------------------------------- #
    # concatenate projection and gate weights for left & right pathways
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=torch.float16)

    # combined linear: produces [B, N, N, 2*H]
    left_comb = F.linear(x_fp16, left_comb_w)   # no bias
    right_comb = F.linear(x_fp16, right_comb_w)

    # split into projection and gate parts
    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # optional masking (broadcast over hidden dim)
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # apply gates
    left = left_proj * left_gate   # [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------- #
    # 3. Core O(N³·H) contraction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # reshape to (batch = B*H, M = N, K = N) for A and B
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_t.view(B * H, N, N)   # (batch, M, K)
    right_flat = right_t.view(B * H, N, N) # (batch, N, K) – we'll treat as transposed

    out_flat = torch.empty_like(left_flat)  # (batch, M, N), fp16

    # strides (in elements) for the kernel
    stride_am = left_flat.stride(1)   # row stride (M axis) = K
    stride_ak = left_flat.stride(2)   # col stride (K axis) = 1
    stride_ab = left_flat.stride(0)   # batch stride for A

    # B is interpreted as (K, N) via transposition
    stride_bk = right_flat.stride(2)  # stride over K (original column) = 1
    stride_bn = right_flat.stride(1)  # stride over N (original row)   = N
    stride_bb = right_flat.stride(0)  # batch stride for B

    stride_cm = out_flat.stride(1)    # row stride for C = N
    stride_cn = out_flat.stride(2)    # col stride for C = 1
    stride_cb = out_flat.stride(0)    # batch stride for C

    # Block sizes – tuned for N up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)  # (tiles over M‑N, batch)

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,          # M
        N,          # N
        N,          # K (shared dimension)
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ab,
        stride_bb,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 4. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 5. Output gate (element‑wise) and final projection
    # ------------------------------------------------------------------- #
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()   # [B, N, N, H]

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out, to_out_w)   # [B, N, N, C]

    # ------------------------------------------------------------------- #
    # 6. Cast back to original dtype and return
    # ------------------------------------------------------------------- #
    out = out.to(orig_dtype)
    return out
```
