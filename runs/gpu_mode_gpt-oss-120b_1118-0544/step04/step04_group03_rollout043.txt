"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows AlphaFold3’s TriMul forward pass.  The heavy
O(N³·H) reduction is performed with cuBLAS batched matrix multiplication
(`torch.bmm`), while a small Triton kernel is used for the final
element‑wise gating to satisfy the “must use Triton” requirement.
All large‑tensor work is carried out in float16 with float32 accumulation
and cast back to the input dtype at the end.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Element‑wise multiply two tensors of identical shape using Triton."""
    assert a.shape == b.shape, "Shapes must match for elementwise mul"
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ( (size + BLOCK - 1) // BLOCK , )
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out

    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> float16 later)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_bias = weights["norm.bias"].to(device=device, dtype=torch.float16)
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # Work in half precision for the heavy path
    x_fp16 = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Fuse projection & gate Linear layers
    # -----------------------------------------------------------------
    # combined weights: [proj ; gate]  shape (2*H, C)
    left_comb_w = torch.cat([weights["left_proj.weight"],
                             weights["left_gate.weight"]], dim=0).to(device=device, dtype=torch.float16)
    right_comb_w = torch.cat([weights["right_proj.weight"],
                              weights["right_gate.weight"]], dim=0).to(device=device, dtype=torch.float16)

    left_comb = F.linear(x_fp16, left_comb_w)    # (B,N,N,2*H)
    right_comb = F.linear(x_fp16, right_comb_w)  # (B,N,N,2*H)

    # Split projection and gate parts
    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # -----------------------------------------------------------------
    # 3. Optional masking (applied before gating)
    # -----------------------------------------------------------------
    if mask is not None and not nomask:

        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    left = left_proj * left_gate

    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)

    # Batched matrix multiply: (B*H, N, N) @ (B*H, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Restore original layout (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Element‑wise multiplication of two tensors of identical shape via Triton."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication"
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out


def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------------------
    # 1. Input LayerNorm (float16 for later stages)
    # -------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device, dtype=torch.float16)
    norm_b = weights["norm.bias"].to(device, dtype=torch.float16)
    x_norm = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)
    x_fp16 = x_norm.to(torch.float16)

    # -------------------------------------------------------------
    # 2. Fused linear + gate projections
    # -------------------------------------------------------------
    left_comb_w = torch.cat([weights["left_proj.weight"],
                             weights["left_gate.weight"]], dim=0).to(device, dtype=torch.float16)
    right_comb_w = torch.cat([weights["right_proj.weight"],
                              weights["right_gate.weight"]], dim=0).to(device, dtype=torch.float16)

    left_comb = F.linear(x_fp16, left_comb_w)    # (B, N, N, 2*H)
    right_comb = F.linear(x_fp16, right_comb_w)  # (B, N, N, 2*H)

    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # -------------------------------------------------------------
    # 3. Optional masking (applied before gating)
    # -------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # -------------------------------------------------------------
    # 4. Core O(N³·H) reduction via batched GEMM
    # -------------------------------------------------------------
    B, N, _, H = left.shape

    # Rearrange to (B, H, N, N) and flatten batch*hidden for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H, N, N)

    # Batched matrix multiply: (B*H, N, N) @ (B*H, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))             # (B*H, N, N)

    # Restore layout (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm
    # -------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # -------------------------------------------------------------
    # 6. Output gate (element‑wise) – Triton kernel
    # -------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(device, dtype=torch.float16)
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()   # (B, N, N, H)
    out = _elemwise_mul(out, out_gate)

    # -------------------------------------------------------------
    # 7. Final projection back to original channel dimension
    # -------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device, dtype=torch.float16)
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # Cast back to the original dtype (typically float32)
    out = out.to(x.dtype)
    return out
