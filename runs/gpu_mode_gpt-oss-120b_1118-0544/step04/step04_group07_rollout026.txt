        def _gate_fuse_kernel(l_ptr, r_ptr, l_gate_ptr, r_gate_ptr, mask_ptr,
                               total, hidden_dim,
                               HAS_MASK: tl.constexpr, BLOCK_SIZE: tl.constexpr):

            offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
            active = offs < total

            r = tl.load(r_ptr + offs, mask=active, other=0.0)
            l_g = tl.load(l_gate_ptr + offs, mask=active, other=0.0)
            r_g = tl.load(r_gate_ptr + offs, mask=active, other=0.0)
            if HAS_MASK:
                mask_idx = offs // hidden_dim
                m = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
                l = l * l_g * m
                r = r * r_g * m
            else:
                l = l * l_g
                r = r * r_g
            tl.store(l_ptr + offs, l, mask=active)
            tl.store(r_ptr + offs, r, mask=active)

            left_flat, right_flat, left_gate_flat, right_gate_flat, mask_flat,
            total, hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK
        )
        # reshape back
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # Compute out = left @ right^T via Triton matmul
    # reshape to (B*hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Transpose right for matmul
    right_mat_T = right_mat.transpose(1, 2).contiguous()

    # Prepare output
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Get strides
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_mat_T.stride(0)
    stride_b_k = right_mat_T.stride(1)
    stride_b_n = right_mat_T.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    # Determine block sizes (tuned values)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute grid dimensions
    num_batches = B * hidden_dim
    grid = (num_batches,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    @triton.jit
    def _matmul_kernel(A, B, C,
                       M, N, K,
                       stride_a_batch, stride_a_m, stride_a_k,
                       stride_b_batch, stride_b_k, stride_b_n,
                       stride_c_batch, stride_c_m, stride_c_n,
                       BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
        pid_batch = tl.program_id(0)
        pid_m = tl.program_id(1)
        pid_n = tl.program_id(2)

        # Compute offsets
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        offs_k = tl.arange(0, BLOCK_K)

        # Create masks for out-of-range
        mask_m = offs_m < M
        mask_n = offs_n < N

        # Pointers to A and B blocks
        a_ptrs = A + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = B + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over K dimension
        for k in range(0, tl.cdiv(K, BLOCK_K)):
            # Load A and B tiles
            a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
            b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)

            # Compute dot product  (matrix multiplication)
            acc += tl.dot(a, b)

            # Increment pointers
            a_ptrs += BLOCK_K * stride_a_k
            b_ptrs += BLOCK_K * stride_b_k

        # Convert accumulator to fp16 and store
        c = acc.to(tl.float16)
        c_ptrs = C + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
        mask = mask_m[:, None] & mask_n[None, :]
        tl.store(c_ptrs, c, mask=mask)

    # Launch kernel
    _matmul_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape out_mat back to (B, hidden_dim, N, N) then to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,),
                                        weight=to_out_norm_w,
                                        bias=to_out_norm_b)

    # Apply out gate
    out = out * out_gate

    # Final linear projection back to dim
    out_flat = out.view(-1, hidden_dim)
    out_proj = torch.nn.functional.linear(out_flat, to_out_w)
    out = out_proj.view(B, N, N, dim)

    # Cast back to original dtype (float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)
    return out

"""
Custom Triton implementation of the outgoing TriMul operator.
The kernel fuses:
  - element‑wise multiplication of projected tensors with their
    per‑position gates and optional attention mask,
  - a batched matrix multiplication performing the Einstein sum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
All heavy compute is performed in FP16, with accumulation in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernels
# --------------------------------------------------------------

@triton.jit
def _gate_fuse_kernel(
    l_ptr, r_ptr,          # [total] flattened left/right
    l_gate_ptr, r_gate_ptr,
    mask_ptr,              # [B*N*N] flattened mask, optional
    total, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr
):
    """
    In‑place: l = l * l_gate * mask (if present)
              r = r * r_gate * mask (if present)
    total      – total number of elements in each of the flattened tensors.
    hidden_dim – size of the hidden dimension (the number of values that share a mask entry).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total

    l = tl.load(l_ptr + offs, mask=active, other=0.0)
    r = tl.load(r_ptr + offs, mask=active, other=0.0)
    l_g = tl.load(l_gate_ptr + offs, mask=active, other=0.0)
    r_g = tl.load(r_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # each mask entry is broadcast over hidden_dim consecutive elements
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        l = l * l_g * m
        r = r * r_g * m
    else:
        l = l * l_g
        r = r * r_g

    tl.store(l_ptr + offs, l, mask=active)
    tl.store(r_ptr + offs, r, mask=active)

@triton.jit
def _matmul_kernel(
    A, B, C,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ B  where
        A : (batch, M, K)
        B : (batch, K, N)   (already transposed)
        C : (batch, M, N)
    Accumulation is performed in FP32.
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers for the current tile
    a_ptrs = A + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = B + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = C + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, c, mask=mask)

# Host entry point
# --------------------------------------------------------------

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as the input.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to fp16 for the heavy part of the model
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Load all linear weights in fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections and gates (batched via view)
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    # Flatten the first three dimensions for a single matmul
    x_flat = x.view(-1, dim)                      # (B*N*N, dim)

    left_flat    = F.linear(x_flat, left_proj_w)   # (B*N*N, hidden_dim)
    right_flat   = F.linear(x_flat, right_proj_w)
    left_gate_flat  = torch.sigmoid(F.linear(x_flat, left_gate_w))
    right_gate_flat = torch.sigmoid(F.linear(x_flat, right_gate_w))
    out_gate_flat   = torch.sigmoid(F.linear(x_flat, out_gate_w))

    # Reshape back to (B, N, N, hidden_dim)
    left    = left_flat.view(B, N, N, hidden_dim)
    right   = right_flat.view(B, N, N, hidden_dim)
    left_gate  = left_gate_flat.view(B, N, N, hidden_dim)
    right_gate = right_gate_flat.view(B, N, N, hidden_dim)
    out_gate   = out_gate_flat.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) and per‑position gates using a tiny Triton kernel
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device).to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        # Flatten tensors so that each contiguous block of size hidden_dim shares a mask entry
        left_flat    = left.contiguous().view(-1)
        right_flat   = right.contiguous().view(-1)
        left_gate_flat  = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat    = mask_fp16.squeeze(-1).contiguous().view(-1)   # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        # Re‑reshape
        left = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        # No mask – simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·H) reduction via a Triton batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Reshape to (B*hidden_dim, N, N) and transpose right for the dot product
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (B*H, N, K)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)  # (B*H, N, K)
    right_mat_T = right_mat.transpose(1, 2).contiguous()                           # (B*H, K, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Strides for the three tensors (batch, row, col)
    stride_a_batch = left_mat.stride(0)
    stride_a_m     = left_mat.stride(1)
    stride_a_k     = left_mat.stride(2)

    stride_b_batch = right_mat_T.stride(0)
    stride_b_k     = right_mat_T.stride(1)
    stride_b_n     = right_mat_T.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_m     = out_mat.stride(1)
    stride_c_n     = out_mat.stride(2)

    # Block sizes – tuned for H100 / B200. They work well for all tested N.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _matmul_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # Cast back to the original datatype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out

"""
Custom Triton implementation of the outgoing TriMul operator.

The kernel fuses:
  * element‑wise multiplication of the projected tensors with their
    per‑position gates and an optional mask,
  * a batched matrix multiplication that realizes the Einstein sum
    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d].

All heavy compute is performed in FP16 with FP32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Triton kernels
# ------------------------------------------------------------------

@triton.jit
def _gate_fuse_kernel(
    l_ptr, r_ptr,                # flattened left/right (size = total)
    l_gate_ptr, r_gate_ptr,      # flattened left/right gates
    mask_ptr,                    # flattened mask (size = B*N*N)
    total, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place fusion:
        l = l * l_gate * mask   (if mask is present)
        r = r * r_gate * mask   (if mask is present)
    Each mask entry is broadcast over `hidden_dim` consecutive elements.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total

    l = tl.load(l_ptr + offs, mask=active, other=0.0)
    r = tl.load(r_ptr + offs, mask=active, other=0.0)
    l_g = tl.load(l_gate_ptr + offs, mask=active, other=0.0)
    r_g = tl.load(r_gate_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        mask_idx = offs // hidden_dim               # integer division
        m = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        l = l * l_g * m
        r = r * r_g * m
    else:
        l = l * l_g
        r = r * r_g

    tl.store(l_ptr + offs, l, mask=active)
    tl.store(r_ptr + offs, r, mask=active)


@triton.jit
def _matmul_kernel(
    A, B, C,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ B
    where
        A : (batch, M, K)
        B : (batch, K, N)   (already transposed)
        C : (batch, M, N)
    Accumulation is in FP32.
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers to the current tile
    a_ptrs = A + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = B + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Store result
    c = acc.to(tl.float16)
    c_ptrs = C + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, c, mask=store_mask)


# ------------------------------------------------------------------
# Host entry point
# ------------------------------------------------------------------

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape (B, N, N, C)
            - mask: torch.Tensor of shape (B, N, N) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, C) with the same dtype as the input.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float32)
    norm_b = weights["norm.bias"].to(device).to(torch.float32)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # Cast to FP16 for the compute‑heavy part
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Load linear weights in FP16
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # --------------------------------------------------------------
    # 3. Projections and gates
    # --------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                     # (B*N*N, C)

    left_flat    = F.linear(x_flat, left_proj_w)          # (B*N*N, H)
    right_flat   = F.linear(x_flat, right_proj_w)
    left_gate_flat  = torch.sigmoid(F.linear(x_flat, left_gate_w))
    right_gate_flat = torch.sigmoid(F.linear(x_flat, right_gate_w))
    out_gate_flat   = torch.sigmoid(F.linear(x_flat, out_gate_w))

    left    = left_flat.view(B, N, N, hidden_dim)
    right   = right_flat.view(B, N, N, hidden_dim)
    left_gate  = left_gate_flat.view(B, N, N, hidden_dim)
    right_gate = right_gate_flat.view(B, N, N, hidden_dim)
    out_gate   = out_gate_flat.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 4. Fuse mask (if any) and per‑position gates via Triton
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device).to(torch.float16).unsqueeze(-1)      # (B, N, N, 1)

        # flatten everything – each mask entry covers `hidden_dim` consecutive values
        left_flat    = left.contiguous().view(-1)
        right_flat   = right.contiguous().view(-1)
        left_gate_flat  = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)
        mask_flat    = mask_fp16.squeeze(-1).contiguous().view(-1)       # (B*N*N,)

        total = left_flat.numel()
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        _gate_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            HAS_MASK=True,
            BLOCK_SIZE=BLOCK,
        )
        left  = left_flat.view_as(left)
        right = right_flat.view_as(right)
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core O(N³·H) reduction via a Triton batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (B*H, N, K)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)  # (B*H, N, K)
    right_mat_T = right_mat.transpose(1, 2).contiguous()                           # (B*H, K, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # stride information (batch, row, col)
    stride_a_batch = left_mat.stride(0)
    stride_a_m     = left_mat.stride(1)
    stride_a_k     = left_mat.stride(2)

    stride_b_batch = right_mat_T.stride(0)
    stride_b_k     = right_mat_T.stride(1)
    stride_b_n     = right_mat_T.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_m     = out_mat.stride(1)
    stride_c_n     = out_mat.stride(2)

    # Tuned block sizes – work well for the provided test configurations
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
