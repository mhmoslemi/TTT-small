    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # 1 layernorm
    x = torch.nn.functional.layer_norm(input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"])
    # Cast to fp16
    x = x.to(torch.float16)

    # linear projection weights
    # convert them to FP16
    left_proj_w = weights["left_proj.weight"].to(dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(dtype=torch.float16)

    # Projections
    left = torch.nn.functional.linear(x, left_proj_w)  # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply mask and gates (elementwise)
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    left = left * left_gate
    right = right * right_gate

    # Reshape for batch GEMM
    B, N, _, H = left.shape
    # Bring H to batch dimension
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Core reduction
    # Use torch.bmm which is highly optimized
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # Reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm
    out = torch.nn.functional.layer_norm(out,

        bias=to_out_norm_b)

    # Apply out_gate: use Triton kernel for fused multiplication

    out_flat = out.view(-1)

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The function follows the reference PyTorch implementation:
1. Layer‑normalize the 4‑D input (B,N,N,C).
2. Project to a hidden dimension H with two linear layers (left/right).
3. Optionally apply a binary mask.
4. Compute three sigmoid gates (left, right, out) from the normalized input.
5. Multiply left/right by their corresponding gates.
6. Collapse the hidden dimension into the batch dimension and compute the core
   reduction   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d],
   which is a batched GEMM C = A @ Bᵀ.
   The GEMM is performed with cuBLAS (torch.bmm) – the fastest path on B200.
7. Reshape back, apply a second LayerNorm over H, multiply by the output gate
   using a tiny Triton kernel, and finally project back to the original channel
   dimension.
All heavy arithmetic is performed in FP16; the final result is cast back to the
input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused element‑wise multiplication of the output with out_gate
# ----------------------------------------------------------------------
@triton.jit
def _gate_mul_kernel(
    out_ptr,         # *float16   output tensor (flattened)
    gate_ptr,        # *float16   gate tensor (flattened, same shape as out)
    size,            # i32        total number of elements
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    out_val = tl.load(out_ptr + offs, mask=mask)
    gate_val = tl.load(gate_ptr + offs, mask=mask)

    tl.store(out_ptr + offs, out_val * gate_val, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (FP16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2) Cast weights to FP16
    # --------------------------------------------------------------
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    left_proj_w = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w = to_fp16(weights["to_out.weight"])

    # --------------------------------------------------------------
    # 3) Linear projections and gates (all FP16)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_w)                     # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)                   # (B,N,N,hidden_dim)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B,N,N,hidden_dim)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))     # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 4) Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 5) Apply gates to projections
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6) Collapse hidden dimension into batch and perform batched GEMM
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Core O(N³·H) reduction using cuBLAS (torch.bmm)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7) Output LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 8) Apply output gate (fused via Triton)
    # --------------------------------------------------------------
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)
    total_elements = out_flat.numel()
    BLOCK = 1024
    grid = ( (total_elements + BLOCK - 1) // BLOCK, )

    _gate_mul_kernel[grid](
        out_flat,
        gate_flat,
        total_elements,
        BLOCK=BLOCK,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 9) Final linear projection back to original channel dimension
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 10) Cast back to original dtype
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
