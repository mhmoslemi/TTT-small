"""
Outgoing Triangle Multiplicative Update (TriMul) – custom Triton kernel.

Algorithm
---------
1. Layer‑normalize the input tensor (B,N,N,C).
2. Linear projections to hidden dimension H for *left* and *right* tensors.
3. (Optional) binary mask applied to the projected tensors.
4. Compute three sigmoid gates from the normalized input and multiply them
   into the projected tensors (left, right, out_gate).
5. Fold the hidden dimension into the batch dimension and perform a batched
   matrix multiplication  C = A @ Bᵀ  where A and B have shape (B*H,N,N).
   This is the O(N³·H) core and is implemented in Triton.
6. Reshape back to (B,N,N,H), apply a second LayerNorm over H,
   multiply by out_gate and finally project back to the original channel
   dimension C.
All heavy linear work is performed in fp16; the final result is cast back
to the input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all equal to seq_len)
    stride_ah, stride_am, stride_ak,   # A: batch, row, col strides
    stride_bh, stride_bk, stride_bn,   # B: batch, k, n strides (Bᵀ layout)
    stride_ch, stride_cm, stride_cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_b = tl.program_id(2)   # batch index (B*H)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base offsets for this batch
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # pointers to the start of the k‑tile (will be incremented inside the loop)
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        k_offset = k * BLOCK_K
        k_idx = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs + k_offset * stride_ak,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs + k_offset * stride_bk,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M,BLOCK_K) x (BLOCK_K,BLOCK_N) -> (BLOCK_M,BLOCK_N)

    # write result
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ Bᵀ for tensors of shape (batch, N, N)
    using a custom Triton kernel. All tensors must be half‑precision and on CUDA.
    """
    assert A.shape == B.shape
    batch, N, K = A.shape
    assert N == K, "TriMul requires square sequence dimension."

    # Heuristic block sizes
    if N <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
        num_warps = 2
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    elif N <= 768:
        BLOCK_M = BLOCK_N = 96
        BLOCK_K = 64
        num_warps = 6
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks in M dimension
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks in N dimension
        batch,                          # batch dimension (B*H)
    )

    C = torch.empty_like(A)

    _batched_matmul_kernel[grid](
        A, B, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),   # A: batch, row, col
        B.stride(0), B.stride(2), B.stride(1),   # B as Bᵀ: batch, k, n
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=3,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (fp32 or fp16)
            - mask          : torch.Tensor of shape [B, N, N] or None
            - weights       : dict of model parameters (fp32 tensors)
            - config        : dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – ignore mask if True

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # work in fp16 for the heavy path
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B,N,N,H)
    right = F.linear(x, right_proj_w)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4) Optional mask (binary, broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6) Apply gates to projections
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Fold hidden dimension into batch and compute core reduction
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for the batched GEMM
    left  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, i, k)
    right = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H, i, k)
    # compute A @ Bᵀ   (i,k) @ (j,k)ᵀ  -> (i,j) for each hidden slice
    out_flat = _batched_matmul_triton(left, right)                  # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8) Reshape back to (B,N,N,H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 10) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 12) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
