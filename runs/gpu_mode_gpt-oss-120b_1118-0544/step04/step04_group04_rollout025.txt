Pseudo:

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_aa, stride_ab, stride_ac,
    stride_ba, stride_bb, stride_bc,
    stride_ca, stride_cb, stride_cc,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # accumulate in fp32

    # Loop over K
    for k in range(0, K, BLOCK_K):
        # k offsets
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # compute pointers for A tile
        a_ptrs = a_ptr + batch_idx * stride_aa + (offs_m[:, None] * stride_ab) + (offs_k[None, :] * stride_ac)
        # similarly for B tile: we need B_T shape (K,N)
        b_ptrs = b_ptr + batch_idx * stride_ba + (offs_k[:, None] * stride_bb) + (offs_n[None, :] * stride_bc)
        # Load tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # Compute block multiplication
        acc += tl.dot(a, b)
    # Write back
    c_ptrs = c_ptr + batch_idx * stride_ca + (offs_m[:, None] * stride_cb) + (offs_n[None, :] * stride_cc)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device

    # LayerNorm weights
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    # Apply layer norm
    x = torch.nn.functional.layer_norm(
            input_tensor,
            normalized_shape=(dim,),
            weight=norm_weight,
            bias=norm_bias,
            eps=1e-5
        )
    # Convert to fp16
    x_fp16 = x.to(torch.float16)

    # Get weights
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # Linear projections
    left_proj = torch.nn.functional.linear(x_fp16, left_proj_weight)  # shape (B,N,N,hidden)
    right_proj = torch.nn.functional.linear(x_fp16, right_proj_weight)

    # Apply mask if needed
    if not nomask:
        mask_fp = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_weight))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Prepare for batched GEMM
    B, N, _, _ = left.shape  # B,seq,N,N, hidden
    # Permute to (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch*hidden
    total = B * hidden_dim
    left_mat = left.view(total, N, N)
    right_mat = right.view(total, N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Compute strides
    stride_a_batch = left_mat.stride(0)
    stride_a_row = left_mat.stride(1)
    stride_a_col = left_mat.stride(2)

    # For B_T view of right_mat
    stride_b_batch = right_mat.stride(0)
    stride_b_k = right_mat.stride(2)   # original col stride (K)
    stride_b_n = right_mat.stride(1)   # original row stride (N)

    stride_c_batch = out_mat.stride(0)
    stride_c_row = out_mat.stride(1)
    stride_c_col = out_mat.stride(2)

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (total, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_row, stride_c_col,
        M=N, N=N, K=N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape output back to (B,N,N,hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm (layer norm over hidden)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # out_gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_weight))
    out = out * out_gate

    # final linear projection
    out = torch.nn.functional.linear(out, to_out_weight)

    # Return as float32
    return out.to(torch.float32)

Outgoing Triangular Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm on the input (float32).
2. FP16 linear projections for left/right tensors and their gates.
3. Optional masking of the projected tensors.
4. Apply sigmoid gates element‑wise.
5. Batched GEMM: for each (batch, hidden) pair compute
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   performed by a custom Triton kernel that carries out the matrix
   multiplication left @ rightᵀ for all heads in parallel.
6. Layer‑norm over the hidden dimension.
7. Compute the output gate (linear + sigmoid) and multiply.
8. Final linear projection back to the original channel dimension.
9. Return FP32 tensor.

The heavy O(N³) contraction (step 5) is fully realized in Triton,
while the remaining inexpensive operations use PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_k,   stride_b_n,
    stride_c_batch, stride_c_row, stride_c_col,
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of matrices.

    A shape: (batch, M, K)  – row‑major
    B shape: (batch, N, K)  – row‑major, interpreted as (batch, K, N) inside the kernel
    C shape: (batch, M, N)
    All tensors are stored in FP16; accumulation is in FP32.
    """
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + batch_idx * stride_a_batch
            + (offs_m[:, None] * stride_a_row)
            + (offs_k[None, :] * stride_a_col)
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Bᵀ tile (BLOCK_K, BLOCK_N) – we read B as (K, N)
        b_ptrs = (
            b_ptr
            + batch_idx * stride_b_batch
            + (offs_k[:, None] * stride_b_k)
            + (offs_n[None, :] * stride_b_n)
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_N) accumulation in fp32

    c_ptrs = (
        c_ptr
        + batch_idx * stride_c_batch
        + (offs_m[:, None] * stride_c_row)
        + (offs_n[None, :] * stride_c_col)
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights: dict of model weights (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  float32

    # Cast to fp16 for the bulk of the computation
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (fp16)
    # --------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    left_proj = F.linear(x_fp16, left_proj_weight)   # (B, N, N, hidden)
    right_proj = F.linear(x_fp16, right_proj_weight)

    # --------------------------------------------------------------
    # 3. Optional mask
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))

    left = left_proj * left_gate    # (B, N, N, hidden)
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5. Batched GEMM via Triton: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) and collapse batch*hidden
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    total_heads = B * hidden_dim
    left_mat = left.view(total_heads, N, N)   # (batch*head, N, N)
    right_mat = right.view(total_heads, N, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Strides in element units (Triton works with element strides)
    stride_a_batch = left_mat.stride(0)
    stride_a_row = left_mat.stride(1)
    stride_a_col = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_k = right_mat.stride(2)   # original column stride → K dimension
    stride_b_n = right_mat.stride(1)   # original row stride    → N dimension

    stride_c_batch = out_mat.stride(0)
    stride_c_row = out_mat.stride(1)
    stride_c_col = out_mat.stride(2)

    # Tile sizes – chosen to fit Tensor Cores on B200
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        M=N,
        N=N,
        K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output LayerNorm over hidden dimension
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, hidden)  fp16

    # --------------------------------------------------------------
    # 7. Output gate and element‑wise multiplication
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))   # (B, N, N, hidden)
    out = out * out_gate

    # --------------------------------------------------------------
    # 8. Final linear projection back to original dimension
    # --------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_weight)   # (B, N, N, dim)  fp16

    # --------------------------------------------------------------
    # Return in float32 as required by the interface
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
