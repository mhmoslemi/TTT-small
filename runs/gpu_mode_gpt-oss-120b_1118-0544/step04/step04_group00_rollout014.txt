def _gate_mask_mul_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, total_elements, hidden_dim, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements
    # compute mask index

    # load
    proj = tl.load(proj_ptr + offs, mask=mask, other=tl.float16(0.0))

    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))
    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask)

4. Fuse mask, gate and projection in a single Triton kernel
   (mask broadcast over the hidden dimension H).  When no mask is
   provided a simple elementwise multiplication is used.
5. Core reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication performed with torch.bmm.
6. Output LayerNorm over the hidden dimension H.
7. Multiply by `out_gate`, apply the final linear projection back to `dim`.
8. Convert back to the original dtype (usually float32).

The implementation uses a Triton kernel for step 4 and relies on
high‑performance PyTorch primitives for the remaining operations.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: (proj * gate * mask) – mask broadcast over H
# --------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    proj_ptr,      # fp16 * (B*N*N*H,)
    gate_ptr,      # fp16 * (B*N*N*H,)
    mask_ptr,      # fp16 * (B*N*N,)
    out_ptr,       # fp16 * (B*N*N*H,)
    total_elems,   # int   = B*N*N*H
    hidden_dim,    # int   = H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Mask for the tail of the last block
    valid = offs < total_elems

    # Index of the corresponding mask element (broadcast over H)
    mask_idx = offs // hidden_dim

    # Load tensors
    proj = tl.load(proj_ptr + offs, mask=valid, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(1.0))

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_proj_gate_mask(proj: torch.Tensor,
                          gate: torch.Tensor,
                          mask: torch.Tensor,
                          block_size: int = 1024) -> torch.Tensor:
    """
    Fuse projection, gate and (optional) mask using Triton.
    All tensors must be fp16, contiguous.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N)  (binary 0/1, broadcasted over H)
    Returns proj * gate * mask (broadcast over H).
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _proj_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor of shape [B, N, N] (bool or 0/1) or None
            - weights      : dict of model parameters (float32)
            - config       : dict containing:
                * "dim" (int)        – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim), dtype = input dtype

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑heavy parts
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to fp16 (no bias in these layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x_fp16, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask + gate (fused via Triton when mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16, contiguous and flattened
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_proj_gate_mask(left_proj, left_gate, mask_fp16)
        right = _apply_proj_gate_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
