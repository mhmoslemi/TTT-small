@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # Program ID 0: combined batch_hidden tile index (BH * m_tiles)
    # Program ID 1: tile column index

    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    # Compute tile indices
    num_m_tiles = tl.cdiv(N, BLOCK_M)
    # Combined batch_hidden * m_tile => total number of blocks in dimension 0
    # Let bh_tile = pid0 // num_m_tiles, tile_m = pid0 % num_m_tiles
    bh_idx = pid0 // num_m_tiles
    tile_m = pid0 % num_m_tiles
    tile_n = pid1

    # Determine batch index (b) and hidden index (h) from bh_idx
    batch_idx = bh_idx // H
    hidden_idx = bh_idx % H

    # Compute offsets
    # Starting coordinates
    i_start = tile_m * BLOCK_M
    j_start = tile_n * BLOCK_N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute base offset for left/right/out
    # offset because of batch and hidden
    # left base = left_ptr + batch_idx * stride_left_b + hidden_idx * stride_left_h
    left_base = left_ptr + batch_idx * stride_left_b + hidden_idx * stride_left_h
    right_base = right_ptr + batch_idx * stride_right_b + hidden_idx * stride_right_h
    out_base = out_ptr + batch_idx * stride_out_b + hidden_idx * stride_out_h

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # offsets within tile
        k_offset = k * BLOCK_K
        # Load A tile
        i_offset = i_start + tl.arange(0, BLOCK_M)
        k_range = k_offset + tl.arange(0, BLOCK_K)
        # compute mask for bounds
        i_mask = i_offset < N
        k_mask = k_range < N
        # Load left tile
        a_ptrs = left_base + i_offset[:, None] * stride_left_i + k_range[None, :] * stride_left_k

        # We need B[j, k], where j dimension is rows of output tile.
        j_offset = j_start + tl.arange(0, BLOCK_N)
        j_mask = j_offset < N
        # B tile shape (BLOCK_K, BLOCK_N) i.e., k_offset as first dim.
        b_ptrs = right_base + j_offset[:, None] * stride_right_i + k_range[None, :] * stride_right_k

        # We'll load using b_ptrs = right_base + k_range[:, None] * stride_right_k + j_offset[None, :] * stride_right_i
        b_ptrs = right_base + k_range[:, None] * stride_right_k + j_offset[None, :] * stride_right_i

"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate and a batched
matrix‑multiply kernel for the O(N³·H) reduction.

Algorithm
---------
1. Layer‑norm the input (float32) and cast to fp16.
2. Four linear projections (left/right + their gates) are performed with
   fp16 weights; the three gate outputs are passed through sigmoid.
3. If a mask is present, a Triton kernel fuses `proj * gate * mask`
   in a single pass (otherwise simple elementwise multiplication is used).
4. Core reduction `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
   is implemented as a batched GEMM (`left @ rightᵀ`) using a custom
   Triton kernel that tiles the (M,N,K) dimensions.  The batch dimension
   is fused with the hidden dimension `H`, giving `B*H` independent GEMM
   problems.
5. Layer‑norm over the hidden dimension, multiplication by the output gate,
   and a final linear projection back to the original channel dimension.
6. Cast back to the input dtype (typically float32) and return.

The kernel works for arbitrary batch size (1‑2), sequence length up to 1024,
and hidden dimension 128 (as used in the test suite.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total #elements in proj/gate
    hidden_dim,        # i32 – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # mask index = element // H  (broadcast mask over hidden dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0).to(tl.float16)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0).to(tl.float16)
    msk  = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0).to(tl.float16)

    tl.store(out_ptr + offs, proj * gate * msk, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using Triton.
    All tensors must be on the same CUDA device, contiguous and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],               # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the core O(N³·H) reduction
# Computes, for each (b, h):
#   out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    left_ptr,      # fp16 *
    right_ptr,     # fp16 *
    out_ptr,       # fp16 *
    B, N, H,       # int32 dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)    # combines batch*H and tile-m
    pid1 = tl.program_id(1)    # tile-n

    # Number of tiles along the M dimension
    num_m_tiles = tl.cdiv(N, BLOCK_M)

    # Decompose pid0
    bh_idx = pid0 // num_m_tiles               # index over B*H groups
    tile_m = pid0 % num_m_tiles                 # tile row within the group
    tile_n = pid1                               # tile column

    # Recover batch and hidden indices
    batch_idx = bh_idx // H
    hidden_idx = bh_idx % H

    # Base pointers that already include batch and hidden offsets
    left_base  = left_ptr  + batch_idx * stride_left_b  + hidden_idx * stride_left_h
    right_base = right_ptr + batch_idx * stride_right_b + hidden_idx * stride_right_h
    out_base   = out_ptr   + batch_idx * stride_out_b    + hidden_idx * stride_out_h

    # Starting coordinates of the tile
    i_start = tile_m * BLOCK_M
    j_start = tile_n * BLOCK_N

    # Accumulator in fp32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K

        # Offsets within the current tile
        i_range = i_start + tl.arange(0, BLOCK_M)
        j_range = j_start + tl.arange(0, BLOCK_N)
        k_range = k_start + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds elements
        i_mask = i_range < N
        j_mask = j_range < N
        k_mask = k_range < N

        # ----- Load A = left[b, i, k, h]  (M×K) -----
        a_ptrs = left_base + (i_range[:, None] * stride_left_i) \
                             + (k_range[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=i_mask[:, None] & k_mask[None, :],
                    other=0.0).to(tl.float16)

        # ----- Load B = right[b, j, k, h]  (N×K) -----
        # We need Bᵀ (K×N) for the dot product; loading as (K,N)
        b_ptrs = right_base + (k_range[:, None] * stride_right_k) \
                               + (j_range[None, :] * stride_right_i)
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & j_mask[None, :],
                    other=0.0).to(tl.float16)

        # Multiply‑accumulate (A @ Bᵀ) → (M,N)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Write the result back (cast to fp16)
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)
    i_mask = i_range < N
    j_mask = j_range < N

    c_ptrs = out_base + (i_range[:, None] * stride_out_i) \
                         + (j_range[None, :] * stride_out_j)
    tl.store(c_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm (float32) and cast to fp16 for heavy work
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Cast all weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4. Gating signals
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5. Fuse mask, projection and gate (if mask present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core O(N³·H) reduction via Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # (B,N,N,H)

    out = torch.empty_like(left)  # fp16 output buffer

    # Strides for the tensors (all are contiguous)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Choose block sizes (tuned for B200)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_m_tiles = (N + BLOCK_M - 1) // BLOCK_M
    num_n_tiles = (N + BLOCK_N - 1) // BLOCK_N
    batch_hidden = B * hidden_dim

    grid = (batch_hidden * num_m_tiles, num_n_tiles)

    _tri_mul_matmul_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Second LayerNorm over hidden dimension, output gate, final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8. Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
