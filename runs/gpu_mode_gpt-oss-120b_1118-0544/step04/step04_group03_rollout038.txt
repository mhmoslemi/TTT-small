"""
TriMul (outgoing) forward pass with Triton acceleration.

The heavy O(N³·H) reduction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is performed by a batched GEMM kernel where each hidden head is treated as an
independent batch.  All other operations (LayerNorm, linear projections,
gating, output normalization and projection) are handled in PyTorch.  Mixed
precision (fp16 compute, fp32 accumulation) is used for the GEMM while the
final result is returned in the original input dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to A, B, C
    M, N, K,                           # matrix dimensions (M×K)·(K×N)→(M×N)
    stride_am, stride_ak,              # strides for A (M, K)
    stride_bk, stride_bn,              # strides for B (K, N) – accessed as transposed
    stride_cm, stride_cn,              # strides for C (M, N)
    stride_ab, stride_bb, stride_cb,   # strides for the batch dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched fp16 GEMM with fp32 accumulation: C = A @ B (B read transposed)."""
    pid = tl.program_id(0)            # output tile id
    batch = tl.program_id(1)          # batch index (B * hidden)

    # tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # per‑tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = kk * BLOCK_K + offs_k
        mask_k = cur_k < K

        # A: shape (M, K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B is stored as (K, N); we read it as‑is (acts as Bᵀ for the einsum)
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)  # fp16·fp16 → fp32

    # write back
    c = acc.to(tl.float16)
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_store,
    )

            - config: dict with keys:

                * "hidden_dim" (int) – hidden dimension H

    right_gate_w = to_fp16(weights["right_gate.weight"])

    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])

    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    right = right * right_gate

    # 4. Core O(N³·H) reduction via Triton batched GEMM
    # ------------------------------------------------------------------- #

    stride_a_batch = left_flat.stride(0)

    stride_b_batch = right_flat.stride(0)
    stride_bk = right_flat.stride(1)  # stride over K (rows of Bᵀ)
    stride_bn = right_flat.stride(2)  # stride over N (cols of Bᵀ)

    stride_c_batch = out_flat.stride(0)
    stride_cm = out_flat.stride(1)    # stride over M
    stride_cn = out_flat.stride(2)    # stride over N

    BLOCK_N = 64

        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 5. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------- #
    # 6. Output gating
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 7. Final projection to original dimension
    # ------------------------------------------------------------------- #
    out = F.linear(out, to_out_w)   # [B, N, N, C]

    # Return in the same dtype as the original input
    out = out.to(orig_dtype)
    return out
```
