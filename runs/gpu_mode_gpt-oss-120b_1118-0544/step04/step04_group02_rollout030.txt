"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

The operator processes a 4‑D tensor x ∈ ℝ^{B×N×N×C} as follows:
1. LayerNorm over the channel dimension C.
2. Linear projections to a hidden dimension H (left/right) and gated
   elementwise sigmoids (left_gate/right_gate/out_gate).
3. Optional mask multiplication (broadcast over H) fused with the
   projection‑gate product via a Triton kernel.
4. Core O(N³·H) reduction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM where for each (b,h) we multiply
   left (N×N) with rightᵀ (N×N).  The GEMM is performed in fp16 with
   fp32 accumulation using a custom Triton kernel.
5. LayerNorm over H, multiplication by out_gate, and a final linear
   projection back to C.

The implementation mixes high‑level torch for the light‑weight ops and
two Triton kernels for the heavy parts (gate‑mask fusion and the batched
matrix multiplication).
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: elementwise proj * gate * mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    proj_ptr,          # *[total_elems]  fp16
    gate_ptr,          # *[total_elems]  fp16
    mask_ptr,          # *[mask_elems]   fp16
    out_ptr,           # *[total_elems]  fp16
    total_elems,       # int
    hidden_dim,        # int
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask_valid = offs < total_elems

    # Each hidden dimension repeats the same mask entry
    mask_index = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask_valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask_valid, other=0.0)
    mask = tl.load(mask_ptr + mask_index, mask=mask_valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=mask_valid)


def apply_gate_and_mask(proj, gate, mask, block_size=1024):
    """
    Fuse projection, gate, and (optional) mask.
    proj, gate: (B, N, N, H) fp16
    mask:      (B, N, N)    fp16 (broadcast over H)
    Returns:   (B, N, N, H) fp16
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (math.ceil(total / block_size),)

    mask_flat = mask.contiguous().view(-1)  # (B*N*N,)
    gate_mask_kernel[grid](
        proj,
        gate,
        mask_flat,
        out,
        total,
        proj.shape[-1],
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³·H) core
# Computes out = left @ rightᵀ   for each (batch, hidden) pair.
# Shapes: left  (B*H, M, K)   where M=N, K=N
#         right (B*H, K, N)   (right already transposed)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to flattened tensors
    batch, M, N, K,                     # integer sizes (batch = B*H)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch dimension
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col

    # Offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for the current batch
    a_base = a_ptr + pid_b * stride_a_batch
    b_base = b_ptr + pid_b * stride_b_batch
    c_base = c_ptr + pid_b * stride_c_batch

    # Masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block (M, K)
        a_offset = offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(
            a_base + a_offset,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B block (K, N)
        b_offset = offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(
            b_base + b_offset,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)

    # Write back
    c_offset = offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_base + c_offset, acc.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, C] (float32 or float16)
        - mask         : torch.Tensor[B, N, N] (bool or numeric)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns
    -------
    torch.Tensor[B, N, N, C]  (same dtype as input_tensor)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑heavy stages
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast weights to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,dim) -> (B,N,N,hidden_dim)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Fuse mask, gates and projections
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask -> fp16, broadcast over hidden_dim
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # Prepare tensors for the GEMM:
    # left: (B, H, N_i, N_k) -> (batch, M, K)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # right: (B, H, N_k, N_j) -> (batch, K, N)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)  # fp16

    batch = B * H
    M = N
    N_ = N
    K = N

    # Strides in element units
    stride_l_batch = left_flat.stride(0)
    stride_l_m     = left_flat.stride(1)
    stride_l_k     = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_k     = right_flat.stride(1)   # K dimension
    stride_r_n     = right_flat.stride(2)   # N dimension

    stride_o_batch = out_flat.stride(0)
    stride_o_m     = out_flat.stride(1)
    stride_o_n     = out_flat.stride(2)

    # Tile sizes – tuned for fp16/TensorCore usage
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        math.ceil(M / BLOCK_M),
        math.ceil(N_ / BLOCK_N),
        batch,
    )

    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch,
        M,
        N_,
        K,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_n,   # note: order is batch, N, K in kernel signature
        stride_r_k,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    return out.to(input_tensor.dtype)
