custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1️⃣ Layer‑normalize the input tensor (B,N,N,C).
2️⃣ Project to hidden dimension H with two linear layers (left & right).
3️⃣ (Optional) apply a binary mask broadcast over H.
4️⃣ Compute three sigmoid gates (left, right, out) from the normalized input.
5️⃣ Multiply left/right by their respective gates.
6️⃣ Fold the hidden dimension into the batch dimension and compute the core
   reduction   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   This is a batched GEMM C = A @ Bᵀ with A,B of shape (B·H, N, N) performed
   with torch.bmm (cuBLAS) for speed.
7️⃣ Reshape back to (B,N,N,H) and apply a second LayerNorm over H.
8️⃣ Multiply the result element‑wise with the output gate using a tiny Triton
   kernel (demonstrates required Triton usage).
9️⃣ Final linear projection back to the original channel dimension C.
🔟 Cast the result back to the input dtype and return.

All heavy arithmetic is done in float16; only the final output is returned in
the original dtype (float32 or float16). A small Triton kernel is used for the
element‑wise gate multiplication.
"""

# Triton kernel: element‑wise multiplication (out = a * b)

@triton.jit
def _elemwise_mul_kernel(
    a_ptr,                # *const half*
    b_ptr,                # *const half*
    c_ptr,                # *half*
    n_elements: tl.int32,  # total number of elements
    BLOCK_SIZE: tl.constexpr,
):

    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < n_elements

    a = tl.load(a_ptr + offset, mask=mask)
    b = tl.load(b_ptr + offset, mask=mask)
    tl.store(c_ptr + offset, a * b, mask=mask)

        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Cast to float16 for the heavy part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (left & right) to hidden dimension H
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w) # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Optional binary mask (broadcast over H)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5️⃣ Compute gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # (B,N,N,H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))     # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Apply gates to projections
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Fold hidden dimension into batch and compute core reduction
    #     out_flat[b*h, i, j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, N, N, H where H == hidden_dim
    # Rearrange to (B, H, N, N) then merge B*H as batch dimension
    left = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # (B*H, N, N)

    # Batched GEMM: C = A @ Bᵀ  (using cuBLAS via torch.bmm)
    out_flat = torch.bmm(left, right.transpose(1, 2))   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to (B,N,N,H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 9️⃣ Output LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 🔟 Element‑wise multiplication with the output gate (Triton kernel)
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous and have matching dtype (float16)
    out = out.contiguous()
    out_gate = out_gate.contiguous()

    n_elements = out.numel()
    BLOCK_SIZE = 1024  # 1024 works well for most sizes; must be a power of two

    out_mul = torch.empty_like(out)
    grid = (triton.cdiv(n_elements, BLOCK_SIZE),)

    _elemwise_mul_kernel[grid](
        out,               # a_ptr
        out_gate,          # b_ptr
        out_mul,           # c_ptr
        n_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Final linear projection back to original dimension C
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = F.linear(out_mul, to_out_w)  # (B,N,N,C)

    # ------------------------------------------------------------------
    # 1️⃣2️⃣ Cast back to original dtype
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
