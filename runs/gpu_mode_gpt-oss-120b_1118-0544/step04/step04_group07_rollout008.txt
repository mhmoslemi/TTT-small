def custom_kernel(data):
    input_tensor, mask, weights, config = data

    B,N,N,C = input_tensor.shape

    dim = config["dim"]; hidden = config["hidden_dim"]; 

    # Load weights as half
    norm_w = weights["norm.weight"].to(input_tensor.device).to(torch.float16)
    norm_b = weights["norm.bias"].to(input_tensor.device).to(torch.float16)

    # Input to half
    x = input_tensor.to(torch.float16)

    # Apply LayerNorm (needs fp32 for precision? but we can use fp16? We'll cast to fp32 internally)
    # Use torch.nn.functional.layer_norm expects same dtype as input and weight, bias. We'll apply with fp32 weight, bias cast to half.
    # At fp16 might cause some underflow but likely fine.
    # Could compute LN in float32 and cast back.
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # Combined linear weights
    left_concat_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(x.device).to(torch.float16)  # shape (2*hidden, dim)
    right_concat_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(x.device).to(torch.float16)

    # out_gate weight
    out_gate_weight = weights["out_gate.weight"].to(x.device).to(torch.float16)

    # Compute left+gate projection: flatten input
    B,N,N,_ = x.shape

    # Reshape to (B*N*N, dim)
    flat_x = x.reshape(-1, dim)   # shape (B*N*N, dim)

    # Linear for left+gate
    left_gate_concat = torch.nn.functional.linear(flat_x, left_concat_weight)   # shape (B*N*N, 2*hidden)
    left_proj = left_gate_concat[:, :hidden]
    left_gate = torch.sigmoid(left_gate_concat[:, hidden:])   # shape (B*N*N, hidden)

    # Linear for right+gate
    right_gate_concat = torch.nn.functional.linear(flat_x, right_concat_weight)
    right_proj = right_gate_concat[:, :hidden]
    right_gate = torch.sigmoid(right_gate_concat[:, hidden:])

    # Linear out_gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(flat_x, out_gate_weight))

    # Reshape back to (B,N,N,hidden)
    left_proj = left_proj.view(B, N, N, hidden)
    left_gate = left_gate.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)
    right_gate = right_gate.view(B, N, N, hidden)
    out_gate = out_gate.view(B, N, N, hidden)

    # Apply mask if present: mask shape (B,N,N). Expand to (B,N,N,1) then broadcast.
    if mask is not None:
        mask_exp = mask.to(x.device).to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Core multiplication: left (B,N,N,hidden) -> (B*hidden, N, N), right -> (B*hidden, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # Multiply left_mat (B*H,N,N) with right_mat^T (swap dims 1 and 2). Use torch.bmm
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # shape (B*H, N, N)

    # Reshape back to (B,N,N,hidden)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm (to_out_norm): shape hidden -> apply weight/bias
    to_out_norm_w = weights["to_out_norm.weight"].to(x.device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(x.device).to(torch.float16)
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Apply out_gate: shape (B,N,N,hidden), broadcast.
    out = out * out_gate

    # Final linear to dim
    to_out_weight = weights["to_out.weight"].to(x.device).to(torch.float16)
    # Flatten and linear
    out_flat = out.view(-1, hidden)
    out = torch.nn.functional.linear(out_flat, to_out_weight).view(B, N, N, dim)

    # Cast back to input dtype
    out = out.to(input_tensor.dtype)

    return out

@triton.jit
def _apply_mask_gate_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
                            total_elems, hidden_dim,
                            HAS_MASK: tl.constexpr,
                            BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * left_gate * mask_val
        right = right * right_gate * mask_val
    else:
        left = left * left_gate
        right = right * right_gate

"""
custom_kernel: Optimized outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1. Layer‑norm the input tensor.
2. Fuse the two linear projections (proj + gate) for the left and right
   branches by concatenating their weight matrices, producing tensors of
   shape (B, N, N, 2·H).  The first H channels are the projections,
   the last H channels are the gate pre‑activations which are passed
   through a sigmoid.
3. Compute the output‑gate in a separate linear + sigmoid.
4. Fuse mask‑application, gate‑multiplication and projection‑masking for
   the left and right tensors with a tiny Triton kernel, producing the
   final left/right tensors of shape (B, N, N, H) ready for the O(N³·H)
   contraction.
5. Perform the contraction as a batched matrix multiplication:
      out[b, i, j, h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a batched GEMM of
   (B·H, N, N) @ (B·H, N, N)ᵀ.
6. Apply a second LayerNorm, the output‑gate and a final linear projection
   back to the original channel dimension.
7. Return the result in the original dtype (usually float32).

Only the mask/gate fusion is done in Triton (requirement of using a
kernel); all heavy linear algebra stays in cuBLAS which gives the
required sub‑millisecond runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,          # [B,N,N,H] flattened
    right_ptr,         # same
    left_gate_ptr,     # same
    right_gate_ptr,    # same
    mask_ptr,          # [B,N,N] flattened (broadcasted over H)
    total_elements,    # B*N*N*H
    hidden_dim,        # H
    HAS_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place fusion:
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
    The mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load data
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Each hidden entry shares the same mask value:
        #   mask_index = flat_index // hidden_dim
        idx = offs // hidden_dim
        m = tl.load(mask_ptr + idx, mask=mask, other=1.0)
        left = left * left_gate * m
        right = right * right_gate * m
    else:
        left = left * left_gate
        right = right * right_gate

    # Store results back in place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape (B, N, N, C)
        - mask       : torch.Tensor of shape (B, N, N) or None
        - weights    : dict of model parameters (float32)
        - config     : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape (B, N, N, C) with the same dtype as ``input_tensor``.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 0) Config & helpers
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float16)
    norm_b = weights["norm.bias"].to(device).to(torch.float16)
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2) Fuse proj + gate for left and right sides
    # ------------------------------------------------------------------
    # Concatenate the projection and gate weights (biases are absent)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device).to(torch.float16)  # (2*H, C)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device).to(torch.float16)  # (2*H, C)

    # Flatten x for a single GEMM
    B, N, _, _ = x.shape
    flat_x = x.reshape(-1, dim)  # (B*N*N, C)

    # Linear -> (B*N*N, 2*H)
    left_cat = F.linear(flat_x, left_concat_w)   # proj + gate
    right_cat = F.linear(flat_x, right_concat_w)

    # Split
    left_proj = left_cat[:, :hidden]
    left_gate = torch.sigmoid(left_cat[:, hidden:])
    right_proj = right_cat[:, :hidden]
    right_gate = torch.sigmoid(right_cat[:, hidden:])

    # Output gate (separate linear)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(flat_x, out_gate_w))

    # Reshape back to (B,N,N,H)
    left_proj = left_proj.view(B, N, N, hidden)
    left_gate = left_gate.view(B, N, N, hidden)
    right_proj = right_proj.view(B, N, N, hidden)
    right_gate = right_gate.view(B, N, N, hidden)
    out_gate = out_gate.view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 3) Fuse mask & gate multiplication using Triton
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp = mask.to(device).to(torch.float16)  # (B,N,N)
        mask_flat = mask_fp.reshape(-1)              # (B*N*N,)
    else:
        mask_fp = None
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)

    total = left_proj.numel()                      # B*N*N*H
    BLOCK = 2048
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    _apply_mask_gate_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask_flat,
        total,
        hidden,
        HAS_MASK=mask_fp is not None,
        BLOCK_SIZE=BLOCK,
    )
    # After the kernel, left_proj and right_proj already contain
    # the gated & masked values.
    del left_gate, right_gate  # free memory

    # ------------------------------------------------------------------
    # 4) Core O(N³·H) contraction via batched GEMM
    # ------------------------------------------------------------------
    # Reshape to (B*H, N, N)
    left_mat = left_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # left_mat @ right_matᵀ  -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, output‑gate and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)  # (C, H)
    out = F.linear(out.view(-1, hidden), to_out_w).view(B, N, N, dim)

    # Return in the original dtype (usually float32)
    if out.dtype != dtype_in:
        out = out.to(dtype_in)
    return out
