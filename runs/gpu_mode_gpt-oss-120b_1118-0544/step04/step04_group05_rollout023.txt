def _batched_matmul_triton(A, B):
    """
    Compute C = A @ B^T for shape (batch,N,N).
    A,B: half tensors.
    Returns C half.
    """
    batch, N, K = A.shape
    assert N == K
    # choose block sizes
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
        num_warps = 2
    elif N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             batch)
    C = torch.empty_like(A)
    _batched_matmul_kernel[grid](
        A, B, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),   # batch,row,col
        B.stride(0), B.stride(2), B.stride(1),   # treat as B^T: batch,k,n
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )
    return C
```

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The essential O(N³·H) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by reshaping the hidden dimension into the batch dimension and
executing a batched GEMM (A @ Bᵀ) with a custom Triton kernel.  All other
operations (LayerNorm, linear projections, gating) are carried out with
PyTorch in FP16 for maximal throughput.  The final result is returned in
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B shape: (batch, N, N))
# ---------------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all = seq_len)
    stride_ah, stride_am, stride_ak,   # A: batch, row, col strides
    stride_bh, stride_bk, stride_bn,   # B: batch, k, n strides (Bᵀ layout)
    stride_ch, stride_cm, stride_cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch index (B*H)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column

    # ------------------- block offsets -------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------- batch offsets -------------------
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # ------------------- pointers for the current tile -------------------
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------- write back -------------------
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for tensors of shape (batch, N, N) using Triton.
    A, B must be float16, on CUDA.
    """
    assert A.shape == B.shape, "A and B must have identical shapes"
    batch, N, K = A.shape
    assert N == K, "TriMul requires square sequence dimension"

    # Heuristic block sizes based on N
    if N <= 64:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
        num_warps = 2
    elif N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    C = torch.empty_like(A)

    _batched_matmul_kernel[grid](
        A, B, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),          # A: batch, row, col
        B.stride(0), B.stride(2), B.stride(1),          # B as Bᵀ: batch, k, n
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (binary) or None
            - weights: dict of model parameters
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – optional bool, ignore mask when True

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Work in FP16 for the heavy part
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast linear weights to FP16 (they are stored as float32)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (B,N,N,H)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # (B,N,N,H)
    right = F.linear(x, right_proj_w)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4) Optional mask (binary, broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6) Apply gates to projections
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Fold hidden dimension into batch for GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)
    left = left.view(B * H, N, N)                  # (B*H, N, N)
    right = right.view(B * H, N, N)                # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8) Core O(N³·H) reduction via batched GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------
    out_flat = _batched_matmul_triton(left, right)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 9) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 10) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 11) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 13) Cast back to float32 (or original dtype if it was float16)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
