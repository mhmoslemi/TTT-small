"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the 4‑D input tensor (B,N,N,C).
2. Project to hidden dimension H with two bias‑free linear layers (left & right).
3. Optionally apply a binary mask (broadcast over H).
4. Compute three sigmoid gates (left, right, out) from the normalized input.
5. Multiply left/right by their respective gates.
6. Fold the hidden dimension into the batch dimension and perform a batched
   matrix multiplication C = A @ Bᵀ where A and B have shape (B·H, N, N).
   This is the core ∑_k left[...,k,:] * right[...,k,:] reduction and is
   implemented with a custom Triton GEMM kernel.
7. Reshape back to (B,N,N,H), apply a second LayerNorm, multiply by the
   output gate and project back to the original channel dimension.
8. Cast the result back to the input dtype.

The Triton kernel is a conventional tiled GEMM (block‑wise dot products)
with FP16 inputs and FP32 accumulation, which gives high utilization
on Tensor‑Core hardware (e.g. Nvidia B200).  For very large sequence
lengths we fall back to torch.bmm which is faster through cuBLAS.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton batched GEMM kernel: C = A @ Bᵀ  (A,B : (batch,N,N))
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all equal to seq_len)
    stride_ah, stride_am, stride_ak,   # A: batch, row (M), col (K) strides
    stride_bh, stride_bk, stride_bn,   # B: batch, row (K), col (N) strides (Bᵀ layout)
    stride_ch, stride_cm, stride_cn,   # C: batch, row (M), col (N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch index (B*H)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column

    # --------------------- tile offsets -----------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------- batch offsets ----------------------
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # --------------------- pointers for the tile -------------
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------- K‑loop ----------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------- store result -----------------------
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------
# Wrapper that launches the Triton GEMM
# --------------------------------------------------------------
def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Computes C = A @ Bᵀ for tensors of shape (batch, N, N) using Triton.
    A and B must be half‑precision (float16) and reside on CUDA.
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    assert A.shape == B.shape
    batch, N, K = A.shape
    assert N == K, "TriMul requires square sequence dimension."

    # Choose block sizes heuristically based on N
    if N <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
        num_warps = 2
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks in M dimension
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks in N dimension
        batch,                          # batch (B * hidden_dim)
    )

    C = torch.empty_like(A)

    _batched_matmul_kernel[grid](
        A, B, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),   # A: batch, row, col
        B.stride(0), B.stride(2), B.stride(1),   # Bᵀ: batch, k, n (swap last two strides)
        C.stride(0), C.stride(1), C.stride(2),   # C: batch, row, col
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )
    return C

# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask          : torch.Tensor of shape [B, N, N] or None (binary)
            - weights       : dict of model parameters (float32 tensors)
            - config        : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, ignore mask when True (optional)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ----------------------- unpack ----------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------- LayerNorm (float32) -----------------
    norm_w = weights["norm.weight"].to(device=device, dtype=input_tensor.dtype)
    norm_b = weights["norm.bias"].to(device=device, dtype=input_tensor.dtype)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ---------------------------------------------------------
    # Cast to float16 for the heavy path
    # ---------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------- load linear weights (fp16) -----------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------- projections ---------------------------
    left  = F.linear(x, left_proj_w)    # (B, N, N, H)
    right = F.linear(x, right_proj_w)   # (B, N, N, H)

    # ------------------- optional mask -------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------- gates (sigmoid) ----------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # ------------------- apply gates -------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------- fold hidden dim into batch ----------
    B, N, _, H = left.shape
    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # ------------------- core reduction -----------------------
    # For very large N cuBLAS (torch.bmm) tends to be faster, otherwise Triton GEMM
    if N >= 512:
        out_flat = torch.bmm(left, right.transpose(-2, -1))
    else:
        out_flat = _batched_matmul_triton(left, right)   # (B*H, N, N)

    # ------------------- reshape back -------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------- output LayerNorm --------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------- apply output gate -------------------
    out = out * out_gate

    # ------------------- final projection --------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------- restore original dtype ---------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)
    return out
