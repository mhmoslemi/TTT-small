"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate & O(N³) reduction.

Algorithm
---------
1. Layer‑norm the input (dim‑wise) using the provided LN parameters.
2. Cast to FP16 for the heavy compute path.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) are performed with FP16 weights.
4. Fuse elementwise mask, projection and gate using a tiny Triton kernel.
5. Core reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   implemented as a tiled GEMM kernel that processes all (batch, hidden)
   pairs simultaneously (B*H independent N×N matmuls).
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and project back to the original channel dimension.
7. Cast the result back to the original dtype.

Only the mask‑gate fusion and the O(N³) reduction are written in Triton;
all other operations use highly‑optimized PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask (broadcast over hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i64 – total number of elements in proj/gate
    hidden_dim,        # i64 – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # bounds check
    mask = offs < total_elems

    # each mask element repeats `hidden_dim` times
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * mval
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(
    proj: torch.Tensor,   # (B,N,N,H) fp16
    gate: torch.Tensor,   # (B,N,N,H) fp16
    mask: torch.Tensor,   # (B,N,N) fp16
    block_size: int = 1024,
) -> torch.Tensor:
    """Fuse `proj * gate * mask` using Triton."""
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],               # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Triton kernel: batched N×N matrix multiplication for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,           # fp16 * (B,N,N,H)
    right_ptr,          # fp16 * (B,N,N,H)
    out_ptr,            # fp16 * (B,N,N,H)
    B, N, H,
    # strides for left
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for right (note ordering j,k)
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    # strides for out
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # block along i (rows)
    pid_j = tl.program_id(1)   # block along j (cols)
    pid_z = tl.program_id(2)   # combined batch*hidden index

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # decompose combined index
    b = pid_z // H
    h = pid_z % H

    # tile offsets
    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_tiles):
        k_start = kk * BLOCK_K

        # ---------- load left tile (i,k) ----------
        a_ptrs = (
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + (i_start + i_range[:, None]) * stride_l_i
            + (k_start + k_range[None, :]) * stride_l_k
        )
        a_mask = ((i_start + i_range[:, None]) < N) & ((k_start + k_range[None, :]) < N)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # ---------- load right tile (k,j) ----------
        b_ptrs = (
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + (k_start + k_range[:, None]) * stride_r_k
            + (j_start + j_range[None, :]) * stride_r_j
        )
        b_mask = ((k_start + k_range[:, None]) < N) & ((j_start + j_range[None, :]) < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # ---------- dot ----------
        acc += tl.dot(a, b)

    # ---------- store result ----------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + (i_start + i_range[:, None]) * stride_o_i
        + (j_start + j_range[None, :]) * stride_o_j
    )
    out_mask = ((i_start + i_range[:, None]) < N) & ((j_start + j_range[None, :]) < N)
    tl.store(out_ptrs, acc.to(tl.float16), mask=out_mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C] (float32 or float16)
            - mask: [B, N, N] (bool/int/float) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 -> fp16 later)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast weights to FP16 (no bias for linear layers)
    # ------------------------------------------------------------------
    def _to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    left_proj_w   = _to_fp16(weights["left_proj.weight"])
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])
    to_out_norm_w = _to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = _to_fp16(weights["to_out_norm.bias"])
    to_out_w      = _to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,C) -> (B,N,N,H)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (same shape) + sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Fuse mask, projection and gate (if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core O(N³·H) reduction with Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # (B,N,N,H) fp16

    # strides (contiguous tensors)
    stride_l = left.stride()   # (b,i,k,h)
    stride_r = right.stride()  # (b,i,k,h) – same layout, we will treat j as second N
    stride_o = out.stride()

    # launch kernel
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],   # j stride (second N)
        stride_r[2],   # k stride
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
