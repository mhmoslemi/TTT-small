@triton.jit
def tri_mul_bmm_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # compute offsets
    off_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    off_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # bounds
    mask_i = off_i < M
    mask_j = off_j < N

    # initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = off_k < K

        # load left block (M, K)
        left_ptrs = left_ptr + batch_id * stride_left_batch \
                    + off_i[:, None] * stride_left_m \
                    + off_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right block transposed (K, N)
        right_ptrs = right_ptr + batch_id * stride_right_batch \
                    + off_k[:, None] * stride_right_k \
                    + off_j[None, :] * stride_right_n
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # convert to fp32 for accumulate
        left = left.to(tl.float32)
        right = right.to(tl.float32)

        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + batch_id * stride_out_batch \
               + off_i[:, None] * stride_out_m \
               + off_j[None, :] * stride_out_n
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass – Triton‑accelerated batched GEMM.

Algorithm
---------
1. Layer‑norm the input tensor (B,N,N,C) with provided LN parameters.
2. Cast to float16 for compute‑intensive ops.
3. Linear projections (left/right) and three gates (left, right, out) are
   applied with float16 weights; gates use sigmoid.
4. If a mask is present (and not disabled) it is broadcast and fused with
   the projection‑gate multiplication using regular PyTorch (the heavy part
   is the GEMM, which is executed in Triton).
5. Core reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply for each hidden dimension d.
   Implemented as a custom Triton kernel that computes
   (B*H) independent matrix multiplications of shape (N,N) × (N,N)^T.
6. Output LayerNorm, element‑wise out‑gate, and final linear projection.
7. Cast back to the original dtype (usually float32).

The Triton kernel performs the O(N³·H) reduction in FP16 with FP32 accumulation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _tri_mul_bmm_kernel(
    left_ptr,           # fp16 *, shape (B*H, N, N)
    right_ptr,          # fp16 *, shape (B*H, N, N)
    out_ptr,            # fp16 *, shape (B*H, N, N)
    M, N, K,            # int, all equal to sequence length
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication:
       out[b, i, j] = Σ_k left[b, i, k] * right[b, j, k]   (right is transposed)
    """
    batch_id = tl.program_id(0)      # batch dimension (B*H)
    block_i = tl.program_id(1)       # tile over M
    block_j = tl.program_id(2)       # tile over N

    # Offsets for the current tile
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # -----------------------------------------------------------------
        # Load a block of left: shape (BLOCK_M, BLOCK_K)
        # -----------------------------------------------------------------
        left_ptrs = left_ptr + batch_id * stride_left_batch \
                    + offs_i[:, None] * stride_left_m \
                    + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # -----------------------------------------------------------------
        # Load a block of right transposed: we need right_T[k, j] = right[j, k]
        # right layout: (batch, row=j, col=k)
        # Therefore stride over k is stride_right_k, over j is stride_right_n
        # -----------------------------------------------------------------
        right_ptrs = right_ptr + batch_id * stride_right_batch \
                     + offs_k[:, None] * stride_right_k \
                     + offs_j[None, :] * stride_right_n
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # -----------------------------------------------------------------
        # Block‑wise dot product
        # -----------------------------------------------------------------
        acc += tl.dot(left, right)

    # -----------------------------------------------------------------
    # Write back the result (cast to FP16)
    # -----------------------------------------------------------------
    out_ptrs = out_ptr + batch_id * stride_out_batch \
               + offs_i[:, None] * stride_out_m \
               + offs_j[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to FP16 (ensure same device)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B,N,N,C) -> (B,N,N,H)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (same shape) and sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask and gates (mask broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * left_gate * mask
        right = right * right_gate * mask
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM (out = left @ rightᵀ)
    #    Prepare tensors as (B*H, N, N) contiguous blocks
    # ------------------------------------------------------------------
    # Change layout to (B, H, N, N) and flatten batch*hidden
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B, H, N, _ = left.shape
    left_flat = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)

    out_flat = torch.empty_like(left_flat)  # fp16 output buffer

    # Strides (batch, row, col)
    stride_l_batch, stride_l_m, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_m, stride_r_k = right_flat.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_flat.stride()

    # For the right matrix we need strides for (k, n) where
    # k corresponds to original column dimension and n to original row.
    stride_r_k = stride_r_k          # stride over original column (k)
    stride_r_n = stride_r_m          # stride over original row   (j)

    # Triton block sizes – tune for the target GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_bmm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_l_batch, stride_l_m, stride_l_k,
        stride_r_batch, stride_r_k, stride_r_n,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
