def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = input_tensor.device

    # Convert to fp32 for layernorm.
    # input shape [B,N,N,dim]; ensure it's contiguous.
    x = input_tensor

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)  # fp32

    # Cast to fp16 for linear ops.
    x_fp16 = x.to(torch.float16)

    # Projections
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    left_proj = F.linear(x_fp16, left_proj_weight)  # (B,N,N,hidden)
    right_proj = F.linear(x_fp16, right_proj_weight)

    if not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Rearrange for batched gemm
    B, N, _, _ = left.shape
    # left: (B,N,N,hidden)
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, hidden, N, N)

    # flatten batch*hidden
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # Allocate output tensor
    out_mat = torch.empty_like(left_mat)  # dtype fp16

    # compute batched gemm using triton
    _batched_gemm(left_mat, right_mat, out_mat, N, N, N)

    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # out_gate
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))
    out = out * out_gate  # elementwise multiply

    # final linear
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_weight)  # shape (B,N,N,dim), fp16

    return out.to(torch.float32)
```

"""
TriMul (outgoing) forward pass implemented with Triton.

The core of the operation is a batched matrix multiplication
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
for each hidden channel d.  This is performed by a custom Triton
kernel that computes many independent GEMMs (batch = B * hidden_dim)
in FP16 while accumulating in FP32.  The rest of the module
(layer‑norms, linear projections, gating and final projection) uses
standard PyTorch kernels; an element‑wise gating multiplication is
fused with a tiny Triton kernel for completeness.

Input:
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

Output:
    Tensor of shape [B, N, N, dim] in float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent matrices
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_ah, stride_am, stride_ak,
    stride_bh, stride_bk, stride_bn,
    stride_ch, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)               # batch * hidden index
    pid_m = tl.program_id(0)               # block row
    pid_n = tl.program_id(1)               # block col

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile (shape BLOCK_M x BLOCK_K)
        a_ptrs = a_ptr + pid_b * stride_ah \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        a_mask = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # Load B tile (shape BLOCK_K x BLOCK_N) – note the transposed layout
        b_ptrs = b_ptr + pid_b * stride_bh \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b_mask = (offs_k < K)[:, None] & (offs_n < N)[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Accumulate
        acc += tl.dot(a, b)   # a,b are fp16 → acc in fp32

    # Store result (convert back to fp16)
    c_ptrs = c_ptr + pid_b * stride_ch \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def batched_gemm_triton(A: torch.Tensor, B: torch.Tensor, C: torch.Tensor):
    """
    Compute C = A @ Bᵀ for a batch of independent matrices.
    All tensors must be FP16, contiguous, and have shapes:
        A: (batch, M, K)
        B: (batch, K, N)   (stored in row‑major order, we read it as transposed)
        C: (batch, M, N)
    """
    assert A.is_contiguous() and B.is_contiguous() and C.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    batched_gemm_kernel[grid](
        A, B, C,
        A.stride(0), A.stride(1), A.stride(2),               # A strides
        B.stride(0), B.stride(2), B.stride(1),               # B: batch, K (col), N (row) – note swapped strides
        C.stride(0), C.stride(1), C.stride(2),               # C strides
        M, N, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim]  (float32)
            - mask_tensor:  torch.Tensor [B, N, N]      (bool/float)
            - weights_dict: model weights (float32)
            - config_dict:  must contain "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, dim], float32

    # --------------------------------------------------------------
    # 2. Cast to FP16 for the heavy compute
    # --------------------------------------------------------------
    x_fp16 = x.to(dtype_fp16)

    # --------------------------------------------------------------
    # 3. Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(dtype_fp16)

    left_proj = F.linear(x_fp16, left_proj_w)    # (B, N, N, hidden)
    right_proj = F.linear(x_fp16, right_proj_w)

    # --------------------------------------------------------------
    # 4. Optional mask
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(device).to(dtype_fp16).unsqueeze(-1)   # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------------------
    # 5. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))

    left = left_proj * left_gate     # (B, N, N, hidden)
    right = right_proj * right_gate  # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 6. Batched GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    B, N, _, _ = left.shape

    # Rearrange to (B, hidden, N, N) and flatten batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    left_mat = left_perm.view(B * hidden_dim, N, N)   # (B*hidden, N, N)  fp16
    right_mat = right_perm.view(B * hidden_dim, N, N) # (B*hidden, N, N)  fp16

    out_mat = torch.empty_like(left_mat)  # (B*hidden, N, N)  fp16

    # Triton batched GEMM
    batched_gemm_triton(left_mat, right_mat, out_mat)

    # Reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 7. Output layer‑norm over the hidden dimension
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(dtype_fp16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # (B, N, N, hidden), fp16

    # --------------------------------------------------------------
    # 8. Out‑gate (linear + sigmoid) and element‑wise multiply
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # (B, N, N, hidden), fp16
    out = out * out_gate                                     # fused element‑wise

    # --------------------------------------------------------------
    # 9. Final linear projection back to original dimension
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).to(dtype_fp16)
    out = F.linear(out, to_out_w)   # (B, N, N, dim), fp16

    # --------------------------------------------------------------
    # 10. Return in float32 as required by the interface
    # --------------------------------------------------------------
    return out.to(torch.float32)
