@triton.jit

def triton_gemm_kernel(

    a_ptr, b_ptr, c_ptr,

    M, N, K,

    stride_am, stride_ak,

    stride_bk, stride_bn,

    stride_cm, stride_cn,

    stride_ab, stride_bb, stride_cb,

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr

):

    pid = tl.program_id(0)

    batch = tl.program_id(1)

    # compute tile coordinates

    num_pid_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_pid_n

    pid_n = pid % num_pid_n

    # compute offsets

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    offs_k = tl.arange(0, BLOCK_K)

    # mask within bounds

    mask_m = offs_m < M

    mask_n = offs_n < N

    # base pointers for batch

    a_ptr_batch = a_ptr + batch * stride_ab

    b_ptr_batch = b_ptr + batch * stride_bb

    c_ptr_batch = c_ptr + batch * stride_cb

    # accumulator

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K

    for k in range(0, tl.cdiv(K, BLOCK_K)):

        k_offset = k * BLOCK_K

        cur_k = k_offset + offs_k

        mask_k = cur_k < K

        # load A

        a = tl.load(

            a_ptr_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),

            mask=mask_m[:, None] & mask_k[None, :],

            other=0.0

        )

        # load B (as transposed)

        b = tl.load(

            b_ptr_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),

            mask=mask_k[:, None] & mask_n[None, :],

            other=0.0

        )

        acc += tl.dot(a, b)

    # store result

    c = acc.to(tl.float16)

    tl.store(

        c_ptr_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),

        c,

        mask=mask_m[:, None] & mask_n[None, :]

    )

    input_tensor, mask, weights, config = data

    bs, seq_len, _, dim = input_tensor.shape

    hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # LayerNorm on input (float32 maybe); apply using input dtype

    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]

    # assume dtype float32; we can cast to input dtype

    x = torch.nn.functional.layer_norm(

        input_tensor,

        normalized_shape=(dim,),

        weight=norm_weight,

        bias=norm_bias

    )

    # Cast to fp16 for compute (if input dtype is fp16 already, use that)

    compute_dtype = torch.float16

    x = x.to(compute_dtype)

    # Convert necessary weights to fp16

    device = x.device

    # Concatenate left_proj and left_gate weights for combined linear

    left_proj_w = weights["left_proj.weight"]  # shape [hidden_dim, dim]

    left_gate_w = weights["left_gate.weight"]

    right_proj_w = weights["right_proj.weight"]

    right_gate_w = weights["right_gate.weight"]

    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(device=device, dtype=compute_dtype)

    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(device=device, dtype=compute_dtype)

    # Linear: shape [B,N,N,2*H]

    left_comb = torch.nn.functional.linear(x, left_comb_w)  # uses bias? No bias

    right_comb = torch.nn.functional.linear(x, right_comb_w)

    # split

    left_proj = left_comb[..., :hidden_dim]   # shape B,N,N,H

    left_gate = left_comb[..., hidden_dim:].sigmoid()

    right_proj = right_comb[..., :hidden_dim]

    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # Mask

    if mask is not None and not nomask:

        mask_f = mask.to(compute_dtype).unsqueeze(-1)  # shape B,N,N,1

        left_proj = left_proj * mask_f

        right_proj = right_proj * mask_f

    left = left_proj * left_gate

    right = right_proj * right_gate

    # core gemm using Triton

    B, N, _, H = left.shape

    # permute to B, H, N, N

    left_perm = left.permute(0,3,1,2).contiguous()

    right_perm = right.permute(0,3,1,2).contiguous()

    # flatten batch*H

    left_flat = left_perm.view(B*H, N, N)

    right_flat = right_perm.view(B*H, N, N)

    # allocate output

    out_flat = torch.empty_like(left_flat)  # shape (B*H, N, N), dtype fp16

    # compute strides

    stride_a_batch = left_flat.stride(0)  # batch stride

    stride_a_m = left_flat.stride(1)      # row stride (M)

    stride_a_k = left_flat.stride(2)      # col stride (K)

    stride_b_batch = right_flat.stride(0)

    stride_b_n = right_flat.stride(2)  # N stride

    stride_c_m = out_flat.stride(1)

    stride_c_n = out_flat.stride(2)

    # block sizes

    BLOCK_M = 128

    BLOCK_N = 128

    BLOCK_K = 32

    # compute grid

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M

    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (num_blocks_m * num_blocks_n, B * H)

        N,  # N

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy O(N³·H) reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is re‑interpreted as a batched matrix multiplication
    out = left @ rightᵀ
for each (b,h) slice.  A fused Triton GEMM kernel computes all
(b·h) matmuls in one launch.  The remaining operations
(LayerNorm, linear projections, gating) are performed with
torch.nn.functional, using mixed‑precision (fp16) for the
compute‑intensive path and casting the final result back to
the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton batched GEMM kernel: C = A @ Bᵀ
#   A : (batch, M, K)  -> (B·H, N, N)
#   B : (batch, K, N)  -> (B·H, N, N) (interpreted as transposed)
#   C : (batch, M, N)  -> (B·H, N, N)
# --------------------------------------------------------------------------- #
@triton.jit
def _gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix dimensions (M×K)·(K×N)→(M×N)
    stride_am, stride_ak,               # strides for A (row, col)
    stride_bk, stride_bn,               # strides for B (k‑dim, n‑dim) – B is read as transposed
    stride_cm, stride_cn,               # strides for C (row, col)
    stride_ab, stride_bb, stride_cb,    # batch strides for A, B, C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # tile id in the output matrix
    batch = tl.program_id(1)        # batch index (B·H)

    # tile coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = k_offset + offs_k
        mask_k = cur_k < K

        # A is (M, K)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B is read as transposed: (K, N)
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp16·fp16 → fp32

    # write result
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------------------- #
# Entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # ------------------------------------------------------------------- #
    # unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device
    compute_dtype = torch.float16

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(device=device, dtype=compute_dtype),
        bias=weights["norm.bias"].to(device=device, dtype=compute_dtype),
    )
    x = x.to(compute_dtype)  # work in fp16 for the heavy path

    # ------------------------------------------------------------------- #
    # 2. Linear projections + gates (combined to reduce reads)
    # ------------------------------------------------------------------- #
    # concatenate proj and gate weights for a single matmul each
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=compute_dtype)        # (2*H, C)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=compute_dtype)        # (2*H, C)

    # linear transforms (no bias)
    left_comb = F.linear(x, left_comb_w)    # [B, N, N, 2*H]
    right_comb = F.linear(x, right_comb_w)  # [B, N, N, 2*H]

    # split into projection and gate parts
    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # optional masking (broadcast over hidden dim)
    if mask is not None and not nomask:
        mask_f = mask.to(compute_dtype).unsqueeze(-1)    # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # apply gates
    left = left_proj * left_gate      # [B, N, N, H]
    right = right_proj * right_gate   # [B, N, N, H]

    # ------------------------------------------------------------------- #
    # 3. Core O(N³·H) reduction as batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape

    # bring hidden dim to batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_perm.view(B * H, N, N)   # (batch, M, K) with M=K=N
    right_flat = right_perm.view(B * H, N, N) # (batch, K, N)

    out_flat = torch.empty_like(left_flat)    # (batch, M, N)

    # strides for the kernel
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)   # row stride (M)
    stride_a_k = left_flat.stride(2)   # col stride (K)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(1)   # K stride (rows of Bᵀ)
    stride_b_n = right_flat.stride(2)   # N stride (cols of Bᵀ)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # tuning parameters – work well for N up to 1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)   # (tiles, batch)

    _gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        stride_a_m,
        stride_a_k,
        stride_b_k,
        stride_b_n,
        stride_c_m,
        stride_c_n,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 4. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=compute_dtype),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=compute_dtype),
    )

    # ------------------------------------------------------------------- #
    # 5. Output gate (element‑wise)
    # ------------------------------------------------------------------- #
    out_gate = F.linear(
        x,
        weights["out_gate.weight"].to(device=device, dtype=compute_dtype),
    ).sigmoid()                     # [B, N, N, H]
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 6. Final projection back to original channel dimension
    # ------------------------------------------------------------------- #
    out = F.linear(
        out,
        weights["to_out.weight"].to(device=device, dtype=compute_dtype),
    )                               # [B, N, N, C]

    # ------------------------------------------------------------------- #
    # 7. Cast back to original dtype
    # ------------------------------------------------------------------- #
    out = out.to(x.dtype)          # input dtype (usually float32)
    return out
