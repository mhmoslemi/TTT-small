"""
TriMul (outgoing) forward pass – Triton‑accelerated full reduction.

Algorithm
---------
1. Layer‑norm the input tensor (B, N, N, C) with the provided LN parameters.
2. Cast to fp16 for compute‑intensive operations.
3. Linear projections:
      left_proj  = x @ W_left      (B,N,N,H)
      right_proj = x @ W_right
   Gate projections (sigmoid):
      left_gate  = σ(x @ W_left_gate)   (B,N,N,H)
      right_gate = σ(x @ W_right_gate)
      out_gate   = σ(x @ W_out_gate)    (B,N,N,H)
4. Fuse element‑wise multiplication of projection, gate and optional mask
   with a tiny Triton kernel (mask broadcast over H).
5. Core reduction: for each batch b and hidden d,
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply (B*H) of size N×N.
   Implemented as a tiled GEMM in Triton to avoid the overhead of
   torch.bmm and to keep data on‑chip.
6. Apply a second LayerNorm over the hidden dimension, multiply by out_gate,
   and a final linear projection back to the original channel dimension.
7. Cast the result back to the input dtype.

The heavy reduction (step 5) is the only part fully implemented in Triton;
all other linear / normalization steps use PyTorch which is already highly
optimized for those shapes.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i64 – total number of elements in proj/gate
    hidden_dim,        # i64 – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * mval
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse elementwise multiplication proj * gate * mask (broadcast over hidden_dim)
    using a Triton kernel. All tensors must be fp16 and contiguous.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out


@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 * (B*N*N*H,)
    right_ptr,         # fp16 * (B*N*N*H,)
    out_ptr,           # fp16 * (B*N*N*H,)
    N,                 # i64 – sequence length
    B,                 # i64 – batch size
    H,                 # i64 – hidden dimension
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_bh = tl.program_id(2)  # batch*hidden index

    # De‑compose batch and hidden indices
    b = pid_bh / H
    h = pid_bh % H

    # Base pointers for this (b, h) slice
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Tile coordinates
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left tile: (M, K)
        left_ptrs = left_base + (i[:, None] * stride_left_i) + (k[None, :] * stride_left_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile: we need (K, N) = right[k, j]  (right stored as (b, j, k, h))
        right_ptrs = right_base + (k[:, None] * stride_right_k) + (j[None, :] * stride_right_i)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right)   # (M,K)·(K,N) → (M,N)

    # Store the result
    out_ptrs = out_base + (i[:, None] * stride_out_i) + (j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Arguments:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C]  (float32 or float16)
            - mask:        [B, N, N] (bool/int/float) or None
            - weights: dict with all model parameters (float32)
            - config:  dict with keys:
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask" (optional) – if True, ignore mask
    Returns:
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for compute‑intensive ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Convert model weights to fp16 on the same device
    # ------------------------------------------------------------------
    def _to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    left_proj_w   = _to_fp16(weights["left_proj.weight"])
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])
    to_out_norm_w = _to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = _to_fp16(weights["to_out_norm.bias"])
    to_out_w      = _to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 4. Linear projections and gates (all in fp16)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)          # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5. Fuse projection, gate and (optional) mask
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Core TriMul reduction via Triton (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    assert H == hidden_dim, "Hidden dimension mismatch"

    # Allocate output tensor
    out = torch.empty_like(left)   # fp16, shape (B,N,N,H)

    # Strides (in number of elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Choose tile sizes (tuned for typical B200 GPU)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        N,
        B,
        H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 8. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
