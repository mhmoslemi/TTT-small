"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‐normalize the input over the feature dimension.
2. Cast to float16 and perform the four linear projections (left/right + gates)
   in two fused linear calls (proj + gate concatenated).  Split and apply
   sigmoid to the gate halves.
3. If a mask is provided (and not disabled) multiply the left/right
   projections by the mask (broadcast over the hidden dimension).
4. Apply the gates: left = left_proj * left_gate, right = right_proj * right_gate.
5. Core O(N³·H) reduction: for each hidden dimension d,
   out[..., i, j, d] = Σₖ left[..., i, k, d] * right[..., j, k, d].
   This is a batched matrix multiplication left @ rightᵀ.
   Implemented with cuBLAS via torch.bmm on tensors of shape
   (B·H, N, N) → (B·H, N, N).
6. Layer‑norm over the hidden dimension, multiply by an output gate
   (sigmoid of a linear projection) using a tiny Triton element‑wise kernel.
7. Final linear projection back to the original feature dimension.
8. Cast the result back to float32.

Only the element‑wise gate multiplication is performed with Triton,
satisfying the requirement to use a custom kernel while relying on
cuBLAS for the dominant GEMM, delivering substantial speed‑ups.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton element‑wise multiplication kernel (used for the output gate)
# --------------------------------------------------------------------------- #
@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    """Element‑wise product of two 1‑D contiguous tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton element‑wise kernel on two tensors of identical shape."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication."
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------- #
    # 2. Cast to half‑precision for the heavy path
    # ------------------------------------------------------------------- #
    x = x.to(dtype_fp16)

    # ------------------------------------------------------------------- #
    # 3. Prepare half‑precision weights (concatenated proj + gate)
    # ------------------------------------------------------------------- #
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype_fp16)          # [2*H, C]
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=dtype_fp16)          # [2*H, C]

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype_fp16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype_fp16)

    # ------------------------------------------------------------------- #
    # 4. Linear projections + gates (fused)
    # ------------------------------------------------------------------- #
    # Combined linear gives [B, N, N, 2*H]
    left_comb = F.linear(x, left_comb_w)   # proj | gate
    right_comb = F.linear(x, right_comb_w)

    # Split projection and gate parts
    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # Optional masking (broadcast over hidden dimension)
    if mask is not None and not nomask:
        mask_f = mask.to(dtype_fp16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------- #
    # 5. Core O(N³·H) reduction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------- #
    B, N, _, _ = left.shape  # (B, N, N, H)

    # Reshape to (B*H, N, N) for batched matrix multiplication
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_flat = left @ rightᵀ  (both are (B*H, N, N))
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 6. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------- #
    # 7. Output gate (sigmoid) and element‑wise multiplication (Triton)
    # ------------------------------------------------------------------- #
    out_gate = F.linear(x, out_gate_w).sigmoid()  # [B, N, N, H]
    out = _elemwise_mul(out, out_gate)

    # ------------------------------------------------------------------- #
    # 8. Final projection back to original channel dimension
    # ------------------------------------------------------------------- #
    out = F.linear(out, to_out_w)                 # [B, N, N, C] (half)

    # Cast back to float32 as required
    out = out.to(torch.float32)

    return out
