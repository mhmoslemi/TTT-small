"""
TriMul (outgoing) forward pass – Triton‑fused mask‑gate multiplication.

Algorithm
---------
1. Layer‑normalize the input tensor (shape [B, N, N, C]).
2. Cast to float16 for the compute‑intensive part.
3. Four linear projections (left/right and their gates) are performed with
   fp16 weights. The three gates are passed through a sigmoid.
4. If a mask is provided, a small Triton kernel fuses
   `proj * gate * mask` in a single pass (otherwise a simple elementwise
   multiplication is used). This removes extra memory traffic.
5. Core O(N³·H) reduction is carried out with a batched matrix multiplication:
   for each hidden channel h, `out_h = left_h @ right_h.T`.  This is done with
   `torch.bmm`, which invokes cuBLAS/Tensor‑core kernels.
6. Layer‑norm is applied over the hidden dimension, the result is multiplied
   by the output gate, and a final linear projection maps back to the original
   channel dimension.
7. The output is cast back to the input dtype (usually float32).

Only the mask‑gate fusion is implemented in Triton; all other heavy
operations use PyTorch’s highly‑optimized CUDA kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (total_elems,)
    gate_ptr,          # fp16 * (total_elems,)
    mask_ptr,          # fp16 * (mask_elems,)  where mask_elems = total_elems // hidden_dim
    out_ptr,           # fp16 * (total_elems,)
    total_elems,       # i32  total number of elements in proj/gate (B*N*N*H)
    hidden_dim,        # i32  H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # Guard against out‑of‑bounds accesses
    valid = offs < total_elems

    # Compute corresponding mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using the Triton kernel above.
    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],            # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict containing model parameters (float32 tensors)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ---------------------------- LayerNorm (input) -----------------------
    # Normalize over the last dimension (C)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for the compute‑heavy path
    x = x.to(torch.float16)

    # -------------------------- cast weights -----------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------- linear proj -----------------------------
    # Shape after proj: (B, N, N, H)
    left_proj  = torch.nn.functional.linear(x, left_proj_w)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # -------------------------- gates ------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # -------------------------- mask & gate fusion ----------------------
    if mask is not None and not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------------- core reduction ---------------------------
    # Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    # Implemented as batched matmul per hidden dim:
    #   left_reshaped: (B*H, N, N)
    #   right_reshaped: (B*H, N, N) -> transpose(1,2)
    B, N, _, H = left.shape
    left_reshaped  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)      # (B*H, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).reshape(B * H, N, N)     # (B*H, N, N)

    # Batch matrix multiplication: left @ right^T
    out = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))      # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------- second LayerNorm & output gate ----------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # -------------------------- final linear projection -----------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, C)

    # -------------------------- cast back -------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
