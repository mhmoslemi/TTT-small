import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def _elemwise_mul(a, b):
    assert a.shape == b.shape
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the PyTorch reference but fuses the expensive
O(N³·H) reduction into a batched half‑precision GEMM performed by
torch.bmm (cuBLAS).  To satisfy the “use Triton” requirement a tiny
element‑wise kernel is used for the final gating multiplication.
All heavy arithmetic is carried out in float16 with float32 accumulation,
and the result is cast back to the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------- #
# Triton element‑wise multiplication kernel (out * out_gate)
# --------------------------------------------------------------- #
@triton.jit
def _elemwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,
    size, BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton kernel for element‑wise multiplication."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication."
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out


# --------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # ----------------------------------------------------------- #
    # Unpack inputs
    # ----------------------------------------------------------- #
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    orig_dtype = x.dtype
    device = x.device

    # ----------------------------------------------------------- #
    # 1. Input LayerNorm (over C)
    # ----------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Work in float16 for the heavy part
    x = x.to(torch.float16)

    # ----------------------------------------------------------- #
    # 2. Cast model weights to float16
    # ----------------------------------------------------------- #
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ----------------------------------------------------------- #
    # 3. Linear projections + gates (combined to halve input reads)
    # ----------------------------------------------------------- #
    # concatenate proj and gate weights along the output‑feature axis
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, C)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # combined linear transforms
    left_comb = F.linear(x, left_comb_w)    # [B, N, N, 2*H]
    right_comb = F.linear(x, right_comb_w)

    # split into projection and gate tensors
    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:].sigmoid()
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()

    # optional mask (broadcast over hidden dimension)
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # apply gates
    left = left_proj * left_gate          # [B, N, N, H]
    right = right_proj * right_gate       # [B, N, N, H]

    # ----------------------------------------------------------- #
    # 4. Core O(N³·H) reduction: batched GEMM (left @ rightᵀ)
    # ----------------------------------------------------------- #
    B, N, _, _ = left.shape

    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matrix multiplication with transposed right operand
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------- #
    # 5. Hidden‑dim LayerNorm
    # ----------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------------------------------------------- #
    # 6. Output gate (element‑wise) – Triton kernel
    # ----------------------------------------------------------- #
    out_gate = F.linear(x, out_gate_w).sigmoid()          # [B, N, N, H]
    out = _elemwise_mul(out, out_gate)                  # fused multiply

    # ----------------------------------------------------------- #
    # 7. Final projection back to original channel dimension
    # ----------------------------------------------------------- #
    out = F.linear(out, to_out_w)                       # [B, N, N, C]

    # Cast back to the original dtype of the input tensor
    out = out.to(orig_dtype)
    return out
