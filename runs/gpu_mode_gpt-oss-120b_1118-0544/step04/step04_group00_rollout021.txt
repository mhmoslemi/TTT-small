def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # Norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Cast to fp16
    x = x.to(torch.float16)
    # weight cast
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    # Projections
    left_proj = torch.nn.functional.linear(x, left_proj_w)
    right_proj = torch.nn.functional.linear(x, right_proj_w)
    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()
    # Apply mask if present
    if mask is not None and not nomask:
        # mask shape (B,N,N); expand dim
        mask = mask.to(torch.float16).unsqueeze(-1)  # shape B,N,N,1
        left = left_proj * left_gate * mask
        right = right_proj * right_gate * mask
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core reduction via Triton GEMM
    B, N, _, H = left.shape
    # rearrange
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    # flatten batch*head
    left_flat = left_t.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)
    # transpose right for gemm
    right_flat_t = right_flat.transpose(1, 2).contiguous()
    # allocate output
    out_flat = torch.empty_like(left_flat)
    # launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (B * H, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    batched_gemm_kernel[grid](
        left_flat, right_flat_t, out_flat,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat_t.stride(0), right_flat_t.stride(1), right_flat_t.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape out
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # Output layer norm, gate, final projection
    out = torch.nn.functional.layer_norm(out, (H,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)
    # Cast back to original dtype
    out = out.to(input_tensor.dtype)
    return out

"""
Outgoing Triangle Multiplicative Update (TriMul) – forward only.

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel dimension (dim).
2. Cast to float16 for the heavy compute part.
3. Linear projections (left/right) and gate projections (left/right/out) are
   performed with torch.nn.functional.linear.
4. Fuse mask, projection and gate with an element‑wise multiplication
   (broadcast mask over the hidden dimension).
5. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication where each (b,h) pair performs
   an N×N @ (N×N)ᵀ multiply.  It is realized with a custom Triton kernel
   that tiles the operation (BLOCK_M×BLOCK_N) and accumulates in FP32.
6. Output LayerNorm, out‑gate multiplication and final linear projection
   back to the original channel dimension.
7. Cast the result back to the input dtype (usually float32).

The implementation keeps the projection & gating in PyTorch (highly optimized)
and only off‑loads the O(N³) reduction to Triton, which substantially cuts
memory traffic and improves runtime on large sequences.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    stride_A_batch, stride_A_m, stride_A_k,
    stride_B_batch, stride_B_k, stride_B_n,
    stride_C_batch, stride_C_m, stride_C_n,
    M, N, K,                           # matrix sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for many independent batches.
    A : (batch, M, K)   (row‑major)
    B : (batch, K, N)   (row‑major, already transposed)
    C : (batch, M, N)
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to the beginning of the output tile
    c_ptrs = (C_ptr + batch_id * stride_C_batch
              + offs_m[:, None] * stride_C_m
              + offs_n[None, :] * stride_C_n)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        # k‑range for this tile
        k_range = k + tl.arange(0, BLOCK_K)
        mask_k = k_range < K

        # Load A tile (M x K)
        a_ptrs = (A_ptr + batch_id * stride_A_batch
                  + offs_m[:, None] * stride_A_m
                  + k_range[None, :] * stride_A_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (K x N)
        b_ptrs = (B_ptr + batch_id * stride_B_batch
                  + k_range[:, None] * stride_B_k
                  + offs_n[None, :] * stride_B_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Cast to FP32 for accumulation
        a_f = tl.cast(a, tl.float32)
        b_f = tl.cast(b, tl.float32)

        # Rank‑K update
        acc += tl.dot(a_f, b_f)

    # Write back the result in FP16
    c = tl.cast(acc, tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] (or None)
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim), dtype = input dtype

    # ------------------------------------------------------------------
    # 2. Cast to float16 for the compute‑intensive path
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to float16 (biases are None for Linears)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask (if any) and fuse with gates
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: (B, N, N) -> broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left  = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM with a custom Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # Rearrange to (B, H, N, N) and flatten batch & head dimensions
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_flat  = left_t.view(B * H, N, N)               # (B*H, N, N)
    right_flat = right_t.view(B * H, N, N)              # (B*H, N, N)

    # Transpose RHS for the matmul (K dimension is the second axis)
    right_flat_t = right_flat.transpose(1, 2).contiguous()  # (B*H, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Tile sizes – chosen to align with Tensor Core dimensions
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: (batch*head, rows, cols)
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the Triton kernel
    _batched_gemm_kernel[grid](
        left_flat,
        right_flat_t,
        out_flat,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat_t.stride(0), right_flat_t.stride(1), right_flat_t.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, C)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)

    return out
