"""
TriMul (outgoing) forward pass – Triton‑fused mask & gate, FP16 heavy compute.

Algorithm
---------
1. Layer‑norm the 4‑D input (B,N,N,D).
2. Cast to float16 for the compute‑intensive part.
3. Perform all six linear projections (left/right, gates, out_gate, to_out)
   by flattening the (B,N,N) dimension and calling a single GEMM per projection.
4. Fuse element‑wise multiplication of projection * gate * mask (if a mask is present)
   with a small Triton kernel.
5. Compute the core O(N³·H) reduction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication L_h @ R_hᵀ for each hidden channel h.
6. Apply a second layer‑norm on the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original channel dimension.
7. Cast the result back to the input dtype (usually float32).

Only the mask‑gate fusion is done in Triton; the rest uses
high‑performance PyTorch GEMM/LN which already maps to Tensor‑cores.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate * mask) on a flattened view
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i32 – total number of elements in proj/gate
    hidden_dim,        # i32 – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask is broadcast over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    msk  = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * msk
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor, block_size: int = 1024
) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` using the Triton kernel above.
    All tensors must be contiguous, on the same CUDA device and dtype fp16.
    """
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


def _linear_flat(x: torch.Tensor, weight: torch.Tensor) -> torch.Tensor:
    """
    Perform a linear projection on a 4‑D tensor by flattening the leading
    dimensions, calling ``torch.nn.functional.linear`` once, and reshaping back.
    Shapes:
        x        : (B, N, N, D)
        weight   : (out_features, D)
    Returns:
        y        : (B, N, N, out_features)
    """
    B, N, _, D = x.shape
    x_flat = x.view(-1, D)                     # (B*N*N, D)
    y_flat = torch.nn.functional.linear(x_flat, weight)   # (B*N*N, out_features)
    y = y_flat.view(B, N, N, -1)
    return y


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, D]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension D
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, D] (same dtype as the input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device
    dtype_in = x.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) -> cast to fp16 for heavy work
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)  # now fp16

    # ------------------------------------------------------------------
    # 2. Cast weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (flattened for efficiency)
    # ------------------------------------------------------------------
    left_proj  = _linear_flat(x, left_proj_w)          # (B, N, N, H)
    right_proj = _linear_flat(x, right_proj_w)

    left_gate  = _linear_flat(x, left_gate_w).sigmoid()
    right_gate = _linear_flat(x, right_gate_w).sigmoid()
    out_gate   = _linear_flat(x, out_gate_w).sigmoid()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4. Apply mask & gate (fused in Triton when mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·H) reduction via batched GEMM:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    => for each h: out_h = left_h @ right_hᵀ
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) and make contiguous for view()
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_flat  = left_perm.view(B * H, N, N)          # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)         # (B*H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)ᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Second LayerNorm, output gate, and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # Final linear projection back to dim
    out_flat = out.view(-1, hidden_dim)                     # (B*N*N, H)
    out_proj_flat = torch.nn.functional.linear(out_flat, to_out_w)  # (B*N*N, D)
    out = out_proj_flat.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 7. Cast back to the original dtype (usually float32)
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
