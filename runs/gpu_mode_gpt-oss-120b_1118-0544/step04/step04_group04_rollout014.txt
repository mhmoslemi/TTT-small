import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(1)
    pid = tl.program_id(0)
    # Compute tile coordinates.
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    # Offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute pointer offsets to the start of the batch.
    a_ptrs = A + batch_id * stride_Ab + offs_m[:, None] * stride_Am + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = B + batch_id * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # Initialize accumulator.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute k range offsets
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Mask for valid k elements
        k_mask = k_offset < K

        # Load A and B blocks
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & k_mask[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & (offs_n[None, :] < N), other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

        # Increment pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # Cast result to fp16
    c = acc.to(tl.float16)

    # Store result
    c_ptrs = C + batch_id * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, c, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm:
1. Layer‑norm the 4‑D input.
2. Four linear projections (left/right + their gates) in FP16.
3. Optional masking of left/right projections.
4. Apply sigmoid gates and element‑wise multiply.
5. Batched GEMM: for each hidden channel d,
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].
   Implemented as a custom Triton kernel that computes many
   (batch*hidden) matrix multiplications in FP16 using tensor cores.
6. Layer‑norm over the hidden dimension.
7. Multiply by out_gate (sigmoid) using a simple Triton element‑wise kernel.
8. Final linear projection back to the original dimension.
All heavy arithmetic is performed in FP16; the final result is returned
as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B) where
# A.shape = (BATCH, M, K), B.shape = (BATCH, K, N), C.shape = (BATCH, M, N)
# All tensors are assumed to be contiguous in memory.
# --------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A, B, C,               # pointers to the tensors
    M, N, K,               # matrix dimensions
    stride_Ab, stride_Am, stride_Ak,   # strides for A (batch, M, K)
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (batch, K, N)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(1)
    pid = tl.program_id(0)

    # Compute 2‑D grid position.
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_blocks_n
    block_n = pid % num_blocks_n

    # Offsets for the current block.
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointer for this batch.
    batch_offset_A = batch_id * stride_Ab
    batch_offset_B = batch_id * stride_Bb
    batch_offset_C = batch_id * stride_Cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block (M x K)
        a = tl.load(
            A + batch_offset_A
                + (offs_m[:, None] * stride_Am)
                + (cur_k[None, :] * stride_Ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B block (K x N)
        b = tl.load(
            B + batch_offset_B
                + (cur_k[:, None] * stride_Bk)
                + (offs_n[None, :] * stride_Bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Matrix‑multiply the tiles.
        acc += tl.dot(a, b)

    # Cast to FP16 and store.
    c = acc.to(tl.float16)
    c_ptr = C + batch_offset_C + (offs_m[:, None] * stride_Cm) + (offs_n[None, :] * stride_Cn)
    tl.store(c_ptr, c, mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------
# Triton kernel: element‑wise multiplication (dst = dst * src)
# --------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(dst_ptr, src_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(dst_ptr + offs, mask=mask)
    b = tl.load(src_ptr + offs, mask=mask)
    tl.store(dst_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float); may be ignored
            - weights: dict of model weight tensors (float32)
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension H
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, C] float32

    # Switch to FP16 for the heavy compute.
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)

    left_proj = F.linear(x_fp16, left_proj_weight)   # (B, N, N, H)
    right_proj = F.linear(x_fp16, right_proj_weight)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))

    left = left_proj * left_gate    # (B, N, N, H)
    right = right_proj * right_gate  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5. Batched GEMM: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Move hidden dimension to batch axis for the GEMM.
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Collapse batch*hidden => batch_gemm
    batch_gemm = B * hidden_dim
    left_mat = left.view(batch_gemm, N, N)               # (batch_gemm, M=N, K=N)
    right_mat = right.view(batch_gemm, N, N).transpose(1, 2)  # (batch_gemm, K=N, N)

    # Allocate output tensor.
    out_mat = torch.empty((batch_gemm, N, N), dtype=torch.float16, device=device)

    # Strides (in elements).
    stride_Ab = left_mat.stride(0)
    stride_Am = left_mat.stride(1)
    stride_Ak = left_mat.stride(2)

    stride_Bb = right_mat.stride(0)
    stride_Bk = right_mat.stride(1)
    stride_Bn = right_mat.stride(2)

    stride_Cb = out_mat.stride(0)
    stride_Cm = out_mat.stride(1)
    stride_Cn = out_mat.stride(2)

    # Kernel launch configuration.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid_x = ( (N + BLOCK_M - 1) // BLOCK_M ) * ( (N + BLOCK_N - 1) // BLOCK_N )
    grid = (grid_x, batch_gemm)

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. to_out_norm (LayerNorm over hidden dimension)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 7. Multiply by out_gate using Triton
    # ------------------------------------------------------------------
    total_el = out.numel()
    BLOCK = 1024
    grid_el = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid_el](
        out,
        out_gate,
        total_el,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_weight)   # (B, N, N, C) fp16

    # Return float32 as required by the interface.
    return out.to(torch.float32)
