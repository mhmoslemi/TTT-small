"""
TriMul outgoing update (forward only) with Triton acceleration.
-------------------------------------------------------------------------------
Algorithm
---------
1. Layer‑normalize the 4‑D input (B, N, N, C).
2. Linear projections → left/right (hidden dim H) and three sigmoid gates.
3. (Optional) apply binary mask to left/right.
4. Multiply left/right by their respective gates.
5. Reshape left/right to (B·H, N, N) and compute
       out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]   (C = A @ Bᵀ)
   using a custom Triton batched GEMM kernel.
6. Reshape back to (B, N, N, H), LayerNorm over H,
   multiply by the output gate, and final linear projection back to C.
7. Cast to the original dtype and return.
-------------------------------------------------------------------------------
The heavy O(N³·H) reduction is performed entirely in fp16 with
tl.dot (Tensor‑core friendly) and accumulated in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM: C = A @ B   (both shapes (batch, M, K) and (batch, K, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,           # pointers
    M, N, K,                       # matrix dimensions (all int32)
    stride_ah, stride_am, stride_ak,   # A: batch, row, col strides
    stride_bh, stride_bk, stride_bn,   # B: batch, row(k), col(n) strides
    stride_ch, stride_cm, stride_cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ B  (no transpose)"""

    pid_b = tl.program_id(2)   # batch index (B*H)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column

    # -----------------------------------
    # Compute tile offsets
    # -----------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -----------------------------------
    # Pointers for this batch
    # -----------------------------------
    a_batch = pid_b * stride_ah
    b_batch = pid_b * stride_bh
    c_batch = pid_b * stride_ch

    # Offsets for the K dimension (will be iterated)
    a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_am
                               + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                               + offs_n[None, :] * stride_bn)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of tiles in K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # tl.dot performs fp16×fp16→fp32 accumulation
        acc += tl.dot(a, b)

        # Advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # -----------------------------------
    # Write back
    # -----------------------------------
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_cm
                               + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ B for tensors of shape (batch, N, N) in fp16 using Triton.
    A and B must be contiguous, half‑precision, and reside on the same CUDA device.
    """
    assert A.shape == B.shape, "A and B must have identical shapes"
    batch, M, K = A.shape
    _, _, N = B.shape
    assert M == K and K == N, "TriMul expects square sequence dimension"

    # Choose block sizes based on N
    if N <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
        num_warps = 2
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    else:  # up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,   # blocks along M
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along N
        batch,                          # batch dimension (B*H)
    )

    C = torch.empty_like(A)

    _batched_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),   # A: batch, row, col
        B.stride(0), B.stride(1), B.stride(2),   # B: batch, row, col
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )
    return C


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor of shape [B, N, N] or None (binary)
            - weights      : dict of model parameters (float32 tensors)
            - config       : dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – optional bool, ignore mask when True

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2) Promote to fp16 for the heavy path
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Cast linear weights to fp16
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w  = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (B, N, N, H)
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, H)
    right = F.linear(x, right_proj_w)  # (B, N, N, H)

    # --------------------------------------------------------------
    # 5) Optional mask (binary, broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 6) Compute gates (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B, N, N, H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B, N, N, H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B, N, N, H)

    # --------------------------------------------------------------
    # 7) Apply gates to projected tensors
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 8) Fold hidden dimension into batch dimension for GEMM
    # --------------------------------------------------------------
    B, N, _, H = left.shape   # B, seq_len, seq_len, hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # reshape to (B*H, N, N)
    left = left.view(B * H, N, N)
    right = right.view(B * H, N, N)

    # Transpose right for the A @ Bᵀ multiplication
    right_t = right.transpose(-2, -1).contiguous()

    # --------------------------------------------------------------
    # 9) Core reduction: out_flat = left @ rightᵀ   (batched GEMM)
    # --------------------------------------------------------------
    out_flat = _batched_gemm_triton(left, right_t)   # (B*H, N, N)

    # --------------------------------------------------------------
    # 10) Reshape back to (B, N, N, H)
    # --------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 11) Output LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 12) Apply output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 13) Final linear projection back to original channel dimension
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, C)

    # --------------------------------------------------------------
    # 14) Cast back to original dtype if needed
    # --------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
