"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the reference PyTorch TriMul forward pass:
1. Layer‑norm over the channel dimension.
2. Linear projections (left/right) and gated linear units (left/right gates).
   Projections and gates are fused into two linear calls to reduce memory traffic.
3. Optional binary mask applied to the projected values before gating.
4. Core O(N³·H) reduction computed as a batched GEMM:
      out[i,j,h] = Σ_k left[i,k,h] * right[j,k,h]
   which is equivalent to a matrix multiply of left (N×N) with the **transpose**
   of right (N×N) for each hidden dimension h.  A custom Triton kernel performs
   the batched GEMM in float16 with float32 accumulation.
5. Layer‑norm over the hidden dimension, multiplication by an output gate,
   and a final linear projection back to the original channel dimension.

All heavy arithmetic is performed in float16; the result is cast back to the
dtype of the original input tensor (typically float32).  The matmul kernel is
the only required Triton kernel, satisfying the “use at least part of the
operations in a kernel” rule.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernels
# --------------------------------------------------------------------------- #

@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix dimensions (M×K)·(K×N)→(M×N)
    stride_am, stride_ak,                    # strides for A: (M, K)
    stride_bk, stride_bn,                    # strides for B: (K, N)
    stride_cm, stride_cn,                    # strides for C: (M, N)
    stride_ab, stride_bb, stride_cb,         # strides for the batch dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B  (A shape: batch×M×K, B shape: batch×K×N)."""
    pid = tl.program_id(0)          # tile ID over (M,N)
    batch = tl.program_id(1)        # batch index (B*H)

    # Determine tile coordinates.
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # Offsets inside the tile.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds accesses.
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch.
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # Load a block of A (M×K).
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a block of B (K×N).
        b = tl.load(
            b_batch + (cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)  # fp16·fp16 → fp32

    # Write the result.
    c = acc.to(tl.float16)
    tl.store(
        c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


@triton.jit
def _elemwise_mul_kernel(
    a_ptr, b_ptr, out_ptr, size,
    BLOCK: tl.constexpr,
):
    """Element‑wise multiplication of two 1‑D contiguous tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton element‑wise multiplication kernel."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication."
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (bool or 0/1) or None
            - weights: dict mapping parameter names to torch.Tensors
            - config: dict with keys:
                * "dim" (int) – channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * optional "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device
    orig_dtype = x.dtype
    half_dtype = torch.float16

    # ------------------------------------------------------------------- #
    # 1. Initial LayerNorm over the channel dimension
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(device=device, dtype=half_dtype),
        bias=weights["norm.bias"].to(device=device, dtype=half_dtype),
    )

    # Work in float16 for the heavy computation.
    x = x.to(half_dtype)

    # ------------------------------------------------------------------- #
    # 2. Prepare weights (cast to half)
    # ------------------------------------------------------------------- #
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device=device, dtype=half_dtype)        # shape (2*H, C)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device=device, dtype=half_dtype)        # shape (2*H, C)

    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=half_dtype)   # (H, C)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=half_dtype)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=half_dtype)    # (H,)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=half_dtype)       # (C, H)

    # ------------------------------------------------------------------- #
    # 3. Linear projections + gates (combined)
    # ------------------------------------------------------------------- #
    # Combined linear maps produce projection and gate logits in one pass.
    left_comb = F.linear(x, left_comb_w)       # [B, N, N, 2*H]
    right_comb = F.linear(x, right_comb_w)     # [B, N, N, 2*H]

    # Split projections and gates.
    left_proj = left_comb[..., :hidden_dim]                # [B, N, N, H]
    left_gate = left_comb[..., hidden_dim:].sigmoid()      # [B, N, N, H]
    right_proj = right_comb[..., :hidden_dim]              # [B, N, N, H]
    right_gate = right_comb[..., hidden_dim:].sigmoid()    # [B, N, N, H]

    # Apply mask if present and not disabled.
    if mask is not None and not nomask:
        mask_f = mask.to(half_dtype).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Apply gates.
    left = left_proj * left_gate      # [B, N, N, H]
    right = right_proj * right_gate   # [B, N, N, H]

    # ------------------------------------------------------------------- #
    # 4. Core O(N³·H) reduction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape   # (B, N, N, H)

    # Rearrange tensors so that each hidden dimension becomes a separate batch.
    # left: (B, H, i, k)
    left_t = left.permute(0, 3, 1, 2).contiguous()
    # right needs to be transposed on its last two axes to get (B, H, k, j)
    right_t = right.permute(0, 3, 2, 1).contiguous()

    # Merge the batch and hidden dimensions.
    left_flat = left_t.view(B * H, N, N)    # (batch, M=N, K=N)
    right_flat = right_t.view(B * H, N, N)  # (batch, K=N, N)

    # Allocate output.
    out_flat = torch.empty_like(left_flat)   # (batch, M, N) in float16

    # Strides for the kernel (all tensors are contiguous in row‑major order).
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)   # stride for M dimension (N)
    stride_a_k = left_flat.stride(2)   # stride for K dimension (1)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(1)  # stride for K dimension (N)
    stride_b_n = right_flat.stride(2)  # stride for N dimension (1)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)    # N
    stride_c_n = out_flat.stride(2)    # 1

    # Block sizes – tuned for typical N ≤ 1024.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_m * num_blocks_n, B * H)

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        stride_a_m,
        stride_a_k,
        stride_b_k,
        stride_b_n,
        stride_c_m,
        stride_c_n,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H].
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 5. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------- #
    # 6. Output gate (element‑wise) using a Triton kernel
    # ------------------------------------------------------------------- #
    out_gate = F.linear(x, out_gate_w).sigmoid()   # [B, N, N, H]
    out = _elemwise_mul(out, out_gate)

    # ------------------------------------------------------------------- #
    # 7. Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------- #
    out = F.linear(out, to_out_w)   # [B, N, N, C]

    # Cast back to the original dtype of the input.
    out = out.to(orig_dtype)

    return out
