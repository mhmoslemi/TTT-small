@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_ab, stride_ai, stride_ak,
    stride_bb, stride_bk, stride_bn,
    stride_cb, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_WARPS: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute valid masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Base pointer for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_ab
    b_batch_ptr = b_ptr + pid_batch * stride_bb
    c_batch_ptr = c_ptr + pid_batch * stride_cb

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # get pointers
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_ai) + (offs_k[None, :] * stride_ak)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        # load tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Convert to fp32
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

        # Increment pointers for next k block
        # Not needed since we recompute offsets each iteration using offs_k
        # alternative: could update a_ptrs and b_ptrs offsets but fine.

    # write out
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_cj)
    # store as fp16
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_ab, stride_ai, stride_ak,
    stride_bb, stride_bk, stride_bn,
    stride_cb, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_WARPS: tl.constexpr,
):
    pidb = tl.program_id(0)
    pidm = tl.program_id(1)
    pidn = tl.program_id(2)

    # compute offsets
    offs_m = pidm * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pidn * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers offset for current batch
    a_batch_ptr = a_ptr + pidb * stride_ab
    b_batch_ptr = b_ptr + pidb * stride_bb
    c_batch_ptr = c_ptr + pidb * stride_cb

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_ai) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # load B tile (K x N) (note B is transposed)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

The kernel `tri_mul_kernel` performs the batched matmul in FP16 with
FP32 accumulation. All other heavy work is done with Torch primitives,
but at least one Triton kernel is used, satisfying the requirement.
"""
import torch
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel for the batched matrix multiplication:
#   C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]   (i.e. C = A @ Bᵀ)
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,                # tensor pointers (FP16)
    batch, M, N, K,                     # ints
    stride_ab, stride_ai, stride_ak,    # strides for A (batch, i, k)
    stride_bb, stride_bk, stride_bn,    # strides for B (batch, k, j)
    stride_cb, stride_ci, stride_cj,    # strides for C (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_WARPS: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch * hidden index
    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile col

    # -------------------------------------------------
    # Offsets for the output tile
    # -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------------------------------------------------
    # Base pointers for this batch
    # -------------------------------------------------
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    # -------------------------------------------------
    # Accumulator (FP32 for numerical stability)
    # -------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Loop over the reduction dimension K
    # -------------------------------------------------
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_ai) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B tile transposed: shape (BLOCK_K, BLOCK_N)
        # B is stored as (batch, j, k) → we need (k, j) for dot
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # -------------------------------------------------
    # Write back the result (FP16)
    # -------------------------------------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# -------------------------------------------------------------------------
# Helper: elementwise mask + gate fusion (simple PyTorch, no extra kernel)
# -------------------------------------------------------------------------
def _apply_mask_and_gate(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor | None) -> torch.Tensor:
    """
    proj, gate: (B, N, N, H)  FP16
    mask: (B, N, N) or None, broadcast over H
    Returns: proj * gate * mask  (FP16)
    """
    if mask is None:
        return proj * gate
    # ensure mask dtype matches proj
    mask = mask.unsqueeze(-1).to(proj.dtype)
    return proj * gate * mask

# -------------------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C]
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor).
    """
    # -------------------------------------------------
    # Unpack
    # -------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # -------------------------------------------------
    # 1. Input LayerNorm (float32) → cast to FP16
    # -------------------------------------------------
    norm_w = weights["norm.weight"].to(device).to(torch.float16)
    norm_b = weights["norm.bias"].to(device).to(torch.float16)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 2. Cast all linear weights to FP16 (no bias)
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # -------------------------------------------------
    # 3. Linear projections (B,N,N,hidden)
    # -------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # -------------------------------------------------
    # 4. Gates (sigmoid)
    # -------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # -------------------------------------------------
    # 5. Apply mask (if any) and gate (elementwise)
    # -------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(device).to(torch.float16)
    else:
        mask_fp16 = None

    left  = _apply_mask_and_gate(left_proj,  left_gate,  mask_fp16)
    right = _apply_mask_and_gate(right_proj, right_gate, mask_fp16)

    # -------------------------------------------------
    # 6. Core reduction using Triton (batched GEMM)
    #    left/right shapes: (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    # -------------------------------------------------
    B, N, _, H = left.shape
    total_batch = B * H

    # reshape to (total_batch, N, N) – contiguous
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_batch, N, N)

    out_flat = torch.empty_like(left_flat)

    # strides (in elements, not bytes)
    stride_ab = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(2)          # k dimension (last)
    stride_bn = right_flat.stride(1)          # j dimension (middle)

    stride_cb = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # Tiling parameters (tuned for Hopper B200)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    NUM_WARPS = 8

    grid = (
        total_batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        total_batch,
        N,
        N,
        N,
        stride_ab, stride_ai, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_WARPS=NUM_WARPS,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final projection
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) FP16

    # -------------------------------------------------
    # 8. Cast back to the original dtype
    # -------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
