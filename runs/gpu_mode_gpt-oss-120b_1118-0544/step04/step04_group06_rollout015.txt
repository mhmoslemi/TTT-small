    left_ptr, right_ptr, out_ptr,
    N,

    stride_r_batch, stride_r_k, stride_r_m,
    stride_o_batch, stride_o_m, stride_o_n,

"""
TriMul (outgoing) forward pass – Triton‑accelerated mask‑gate fusion and
batched GEMM.

Algorithm
---------
1. Layer‑norm the input (float32) with provided LN parameters.
2. Cast to fp16 for compute‑intensive operations.
3. Linear projections (left/right) and gate projections (left_gate,
   right_gate, out_gate) with fp16 weights.
4. Fuse element‑wise multiplication of projection, gate and optional mask
   using a Triton kernel (_gate_mask_kernel) to avoid extra memory passes.
5. Core reduction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched matrix‑multiply (left @ rightᵀ) using a custom
   Triton kernel (batched_matmul_kernel) that processes each (batch, head)
   independently, accumulating in fp32 and storing in fp16.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and project back to the original channel dimension.
7. Cast back to the original dtype (usually float32).

The implementation uses only Triton‑compatible operations for the mask‑gate
fusion and the heavy O(N³·H) reduction, while relying on PyTorch for
lightweight linear layers and LayerNorms.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _gate_mask_kernel(
    proj_ptr,          # fp16 * (B*N*N*H,)
    gate_ptr,          # fp16 * (B*N*N*H,)
    mask_ptr,          # fp16 * (B*N*N,)
    out_ptr,           # fp16 * (B*N*N*H,)
    total_elems,       # i64 – total number of elements in proj/gate
    hidden_dim,        # i64 – H (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # bounds check
    valid = offs < total_elems

    # broadcasted mask index: each mask element repeats hidden_dim times
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mval = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mval
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    block_size: int = 1024,
) -> torch.Tensor:
    """
    Fuse elementwise multiplication proj * gate * mask (broadcast over hidden_dim)
    using a Triton kernel. All tensors must be fp16 and contiguous.

    Args:
        proj: (B,N,N,H) tensor.
        gate: (B,N,N,H) tensor.
        mask: (B,N,N) tensor (fp16).
        block_size: kernel block size.

    Returns:
        Tensor of shape (B,N,N,H) = proj*gate*mask
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],  # hidden_dim = H
        BLOCK_SIZE=block_size,
    )
    return out


@triton.jit
def batched_matmul_kernel(
    left_ptr,            # fp16 * (B*H, N, N)
    right_ptr,           # fp16 * (B*H, N, N)
    out_ptr,             # fp16 * (B*H, N, N)
    N,                   # i64 – sequence length
    stride_l_batch,      # i64
    stride_l_m,          # i64
    stride_l_k,          # i64
    stride_r_batch,      # i64
    stride_r_k,          # i64  (row stride for transposed access)
    stride_r_m,          # i64  (col stride for transposed access)
    stride_o_batch,      # i64
    stride_o_m,          # i64
    stride_o_n,          # i64
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    batch_id = tl.program_id(2)
    pid_m = tl.program_id(0)  # tiles over output rows i
    pid_n = tl.program_id(1)  # tiles over output cols j

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch
    left_batch_ptr = left_ptr + batch_id * stride_l_batch
    right_batch_ptr = right_ptr + batch_id * stride_r_batch
    out_batch_ptr = out_ptr + batch_id * stride_o_batch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block (M, K)
        left = tl.load(
            left_batch_ptr + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right block transposed (K, N)
        # we need right_T[k, n] = right[n, k]
        right = tl.load(
            right_batch_ptr + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_m,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc += tl.dot(left, right)

    # write back result in fp16
    out = acc.to(tl.float16)
    tl.store(
        out_batch_ptr + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n,
        out,
        mask=mask_m[:, None] & mask_n[None, :],
    )

            - mask: [B, N, N] (bool/int/float) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        Tensor of shape [B, N, N, C] (same dtype as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16 (ensure same device)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (to hidden_dim)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)   # (B, N, N, H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Apply mask + gates (fused via Triton if mask is present)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM left @ rightᵀ
    #    Shape: (B, N, N, H) → (B*H, N, N) for each batch*head.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # Bring H dimension before N,N and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)

    # strides for the flattened tensors
    l_strides = left_flat.stride()   # (batch_stride, m_stride, k_stride)
    r_strides = right_flat.stride()
    o_strides = out_flat.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        l_strides[0],
        l_strides[1],
        l_strides[2],
        r_strides[0],
        r_strides[2],   # row stride for transposed access (original k stride)
        r_strides[1],   # col stride for transposed access (original i stride)
        o_strides[0],
        o_strides[1],
        o_strides[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to original dtype
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)

    return out
