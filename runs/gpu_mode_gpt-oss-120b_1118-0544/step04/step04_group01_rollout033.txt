@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize acc
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        mask_k = offs_k < K

        # Load A tile
        a_ptrs = a_ptr + batch_id * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # Load B tile
        b_ptrs = b_ptr + batch_id * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + batch_id * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass – Triton‑accelerated core.
Algorithm
---------
1. Layer‑norm the input (float32) and cast to fp16.
2. Linear projections (left/right) and their gates (all fp16).
3. Fuse element‑wise projection * gate * mask (if present) via a Triton kernel.
4. Core O(N³·H) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication over the hidden dimension.
   Implemented with a custom Triton kernel that processes each (batch*hidden)
   pair as an independent GEMM (fp16 → fp32 accumulation → fp16 store).
5. Layer‑norm on the hidden dimension, multiply by the output gate.
6. Final linear projection back to the original channel dimension.
7. Cast the result back to the input dtype (float32).

The kernel in step 3 and the batched GEMM in step 4 are the Triton‑powered
components. All heavy arithmetic uses fp16 for speed while accumulating in
fp32 for numerical stability.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Kernel 1: fuse projection * gate * mask (broadcast mask over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,      # fp16 * (total_elems,)
    gate_ptr,      # fp16 * (total_elems,)
    mask_ptr,      # fp16 * (mask_elems,)   # mask_elems = B*N*N
    out_ptr,       # fp16 * (total_elems,)
    total_elems,   # i32
    hidden_dim,    # i32
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # mask is broadcast over hidden_dim
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor,
                         block_size: int = 1024) -> torch.Tensor:
    """Fuse proj * gate * mask using a Triton kernel (fp16 tensors)."""
    assert proj.is_contiguous() and gate.is_contiguous() and mask.is_contiguous()
    total = proj.numel()
    out = torch.empty_like(proj)
    grid = (triton.cdiv(total, block_size),)
    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        proj.shape[-1],          # hidden_dim
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: batched matrix multiplication (fp16 A,B -> fp16 C, fp32 acc)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,               # A strides (row, col) within a batch
    stride_bk, stride_bn,               # B strides (row, col) within a batch
    stride_cm, stride_cn,               # C strides (row, col) within a batch
    stride_ab, stride_bb, stride_cb,    # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of this batch
    a_batch_ptr = a_ptr + batch_id * stride_ab
    b_batch_ptr = b_ptr + batch_id * stride_bb
    c_batch_ptr = c_ptr + batch_id * stride_cb

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # Load B tile
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul(a: torch.Tensor,
                    b: torch.Tensor,
                    BLOCK_M: int = 64,
                    BLOCK_N: int = 64,
                    BLOCK_K: int = 64) -> torch.Tensor:
    """
    Batched matrix multiplication using Triton.
    a: (B, M, K) fp16
    b: (B, K, N) fp16
    Returns c: (B, M, N) fp16
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Strides for row‑major (contiguous) tensors
    stride_am = K
    stride_ak = 1
    stride_ab = M * K

    stride_bk = N
    stride_bn = 1
    stride_bb = K * N

    stride_cm = N
    stride_cn = 1
    stride_cb = M * N

    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_ab,
        stride_bb,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (float32) or None
            - weights: dict of model parameters (float32)
            - config: dict containing "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor [B, N, N, C] (same dtype as input_tensor)
    """
    # -------------------- unpack --------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # -------------------- first LayerNorm (float32) --------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Cast to fp16 for the heavy part
    x = x.to(torch.float16)

    # -------------------- cast weights to fp16 --------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------- linear projections --------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # -------------------- gating (sigmoid) --------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()   # (B,N,N,hidden_dim)

    # -------------------- fuse projection * gate * mask --------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()
        left  = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -------------------- core O(N³·H) reduction via batched GEMM --------------------
    # Transform tensors to (B, H, N, N) layout
    # left: (B,N,N,H) -> (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: need to transpose the last two dims (j,k) -> (k,j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    # Flatten batch and hidden dim together for the Triton kernel
    left_flat  = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Perform batched matmul: (B*H, N, N) @ (B*H, N, N) -> (B*H, N, N)
    out_flat = _batched_matmul(left_flat, right_flat,
                               BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)

    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------- second LayerNorm on hidden dimension --------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------- output gate --------------------
    out = out * out_gate

    # -------------------- final linear projection --------------------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # -------------------- cast back to original dtype --------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
