Pseudo:

```
@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          BATCH, M, N, K,
                          stride_a_batch, stride_a_m, stride_a_k,
                          stride_b_batch, stride_b_n, stride_b_k,
                          stride_c_batch, stride_c_m, stride_c_n,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    # Determine batch index
    batch_id = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    # Determine block row col
    pid_m = pid // tl.cdiv(N, BLOCK_N)
    pid_n = pid % tl.cdiv(N, BLOCK_N)
    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Load initial accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # compute pointers for A and B
        a_ptrs = A_ptr + batch_id * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = B_ptr + batch_id * stride_b_batch + offs_n[None, :] * stride_b_n + offs_k[:, None] * stride_b_k
        a = tl.load(a_ptrs, mask=offs_m < M and offs_k < K, other=0.0)
        b = tl.load(b_ptrs, mask=offs_k < K and offs_n < N, other=0.0)
        # accumulate
        acc += tl.dot(a, b)
    # store result
    c_ptrs = C_ptr + batch_id * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated mask+gate fusion and
batched matrix multiplication.

Algorithm
---------
1. Layer‑norm the input over the channel dimension.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections to hidden_dim (left/right) and gate logits (left/right/out).
4. Fuse mask, sigmoid‑gate and projection with a Triton kernel
   (only when a mask is present).  Without a mask we fall back to a simple
   element‑wise multiplication.
5. Core reduction: for each hidden channel d compute
       out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
   This is a batched matrix multiplication (B*hidden_dim independent GEMMs)
   implemented with a custom Triton kernel.
6. Layer‑norm over the hidden dimension, multiply by out_gate and a final
   linear projection back to the original channel dimension.
7. Cast the result back to the input dtype.

The implementation uses two Triton kernels:
* `_apply_gate_mask_kernel` – element‑wise fuse of projection, gate and mask.
* `_batched_matmul_fp16_kernel` – strided‑batched FP16 GEMM with FP32
  accumulation.  Both kernels are launched from `custom_kernel`.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Kernel 1: fuse projection, sigmoid‑gate and mask (mask broadcast over H)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    proj_ptr,          # fp16*  (B*N*N*H,)
    gate_ptr,          # fp16*  (B*N*N*H,)
    mask_ptr,          # fp16*  (B*N*N,)
    out_ptr,           # fp16*  (B*N*N*H,)
    total_elems,       # int64, total elements in proj/gate (B*N*N*H)
    hidden_dim,        # int32, H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)          # [0, BLOCK_SIZE)

    valid = offs < total_elems

    # mask is broadcast over the hidden dimension
    mask_idx = offs // hidden_dim

    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(proj, gate, mask, block_size: int = 1024):
    """
    Fuse projection, sigmoid gate and optional mask using Triton.
    All tensors must be contiguous fp16.
    proj, gate : (B, N, N, H)
    mask       : (B, N, N) – broadcasted over H
    Returns: proj * gate * mask  (fp16)
    """
    B, N, _, H = proj.shape
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block_size),)

    _apply_gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        H,
        BLOCK_SIZE=block_size,
    )
    return out


# ----------------------------------------------------------------------
# Kernel 2: strided‑batched FP16 GEMM (A @ B) with FP32 accumulation
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_fp16_kernel(
    a_ptr,          # fp16* (batch, M, K)
    b_ptr,          # fp16* (batch, K, N)
    c_ptr,          # fp16* (batch, M, N)
    batch,          # int32
    M,              # int32
    N,              # int32
    K,              # int32
    stride_a_batch,  # int64
    stride_a_m,      # int64
    stride_a_k,      # int64
    stride_b_batch,  # int64
    stride_b_k,      # int64
    stride_b_n,      # int64
    stride_c_batch,  # int64
    stride_c_m,      # int64
    stride_c_n,      # int64
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # Compute block indices
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_m * num_n

    batch_id = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    pid_m = block_idx // num_n
    pid_n = block_idx % num_n

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for A and B tiles
        a_ptrs = (
            a_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            b_ptr
            + batch_id * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # tl.dot does FP16*FP16 -> FP32 accumulation
        acc += tl.dot(a, b)

    # Store result (cast back to FP16)
    c_ptrs = (
        c_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(A, B):
    """
    Compute C = A @ B (where B is already transposed as KxN) for
    a strided‑batched FP16 GEMM using Triton.
    A: (batch, M, K)  fp16, contiguous
    B: (batch, K, N)  fp16, contiguous
    Returns C: (batch, M, N) fp16.
    """
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in number of elements (not bytes)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    total_blocks = batch * tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N)
    grid = (total_blocks,)

    _batched_matmul_fp16_kernel[grid](
        A,
        B,
        C,
        batch,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation, fully fused where
    beneficial.  The function follows the same API as the reference
    PyTorch module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C]
            - mask         : torch.Tensor of shape [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict containing "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input_tensor)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype_in = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for compute‑intensive part
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast all linear weights to FP16 (biases are None)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gate projections (same shape) + sigmoid
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Fuse mask and gates (Triton) if a mask is present
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # Ensure mask is FP16 and contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left = _apply_gate_and_mask(left_proj, left_gate, mask_fp16)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_fp16)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Core reduction: batched GEMM over the sequence dimension
    #    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N_seq, _, H = left.shape  # H == hidden_dim
    batch_total = B * H

    # Reshape to (batch_total, N, N) where the hidden dimension is merged
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(batch_total, N_seq, N_seq)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(batch_total, N_seq, N_seq)

    # Transpose the second operand (right) as required by the einsum
    right_mat_T = right_mat.transpose(1, 2).contiguous()

    # Batched matmul with Triton
    out_mat = _batched_matmul_fp16(left_mat, right_mat_T)  # (batch_total, N, N) fp16

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, H, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 9. Cast back to the original dtype
    # ------------------------------------------------------------------
    if out.dtype != dtype_in:
        out = out.to(dtype_in)

    return out
