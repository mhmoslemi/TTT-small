"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‐norm the input tensor (B,N,N,C).
2. Linear projections to hidden dimension H (left/right) and three sigmoid gates.
3. Optional binary mask (broadcast over H).
4. Apply gates to the projected tensors.
5. Core O(N³·H) reduction:
   For each batch and hidden channel compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM C = A @ Bᵀ where A,B have shape (B·H, N, N).
   Implemented with a tiled Triton kernel (fp16 inputs, fp32 accumulation).
6. Layer‐norm over hidden dimension, apply output gate and final linear projection.
7. Return result in the original dtype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B shape: (batch, N, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dims (all = seq_len)
    stride_abatch, stride_am, stride_ak,   # A: batch, row, col strides
    stride_bbatch, stride_bk, stride_bn,   # B: batch, row(k), col strides (Bᵀ layout)
    stride_cbatch, stride_cm, stride_cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of square matrices."""
    batch_id = tl.program_id(0)   # batch index (B*H)
    pid_m = tl.program_id(1)      # block row
    pid_n = tl.program_id(2)      # block column

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the current tile of A and B
    a_ptrs = A_ptr + batch_id * stride_abatch + \
             offs_m[:, None] * stride_am + \
             tl.arange(0, BLOCK_K)[None, :] * stride_ak

    b_ptrs = B_ptr + batch_id * stride_bbatch + \
             tl.arange(0, BLOCK_K)[:, None] * stride_bk + \
             offs_n[None, :] * stride_bn

    # Accumulator in fp32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of tiles along K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        # K‑range for this tile
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load tiles with masking
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Matrix multiply‑accumulate
        acc += tl.dot(a, b)

        # Step pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back (convert to fp16)
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + batch_id * stride_cbatch + \
             offs_m[:, None] * stride_cm + \
             offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ Bᵀ using the Triton kernel above.
    A, B: half‑precision tensors of shape (batch, N, N) on CUDA.
    Returns C of the same shape and dtype.
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, N, K = A.shape
    assert batch == B.shape[0] and N == B.shape[1] and K == B.shape[2]

    # Choose block sizes heuristically
    if N <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
        num_warps = 2
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    else:  # N up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    # Grid dimensions: (batch, blocks_m, blocks_n)
    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Strides
    stride_abatch = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_bbatch = B.stride(0)
    stride_bk = B.stride(1)   # row (k) stride
    stride_bn = B.stride(2)   # col (n) stride

    C = torch.empty_like(A)

    stride_cbatch = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    _batched_matmul_kernel[grid](
        A, B, C,
        N, N, N,
        stride_abatch, stride_am, stride_ak,
        stride_bbatch, stride_bk, stride_bn,
        stride_cbatch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )
    return C

def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim"  – hidden dimension H
                * "nomask"      – bool, ignore mask when True (optional)

    Returns:
        torch.Tensor [B, N, N, C] with the same dtype as the input.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (torch implementation for simplicity)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Work in fp16 for the heavy computations
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Cast linear weights to fp16 (no bias in these layers)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections to hidden dimension (left/right)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B,N,N,H)
    right = F.linear(x, right_proj_w)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Gating (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6) Apply gates to projections
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Fold hidden dimension into batch for GEMM: (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left = left.view(B * H, N, N)   # (B*H, N, N)
    right = right.view(B * H, N, N)  # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8) Core reduction: batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # For very large N cuBLAS (torch.bmm) can be faster; however the kernel
    # works well across the whole range and satisfies the "must use Triton"
    # constraint.
    out_flat = _batched_matmul_triton(left, right)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 9) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 10) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 11) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 13) Cast back to original dtype if needed
    # ------------------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
```
