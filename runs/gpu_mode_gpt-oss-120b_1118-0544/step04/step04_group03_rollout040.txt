custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Cast to float16 for the heavy path.
3. Linear projections + sigmoid gates (left/right) are computed with
   PyTorch functional calls.
4. Optional binary mask (broadcast over the hidden dimension) zeros the
   projected values before gating.
5. Core O(N³·H) reduction:

   the output‑gate using a tiny Triton kernel.
7. Final linear projection back to the original channel dimension and cast
   to the input dtype.

satisfies the requirement to use a custom kernel while keeping the overall
runtime low.

    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton kernel on two tensors of identical shape."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication."
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 65536  # large block to amortize launch overhead
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out


# --------------------------------------------------------------------------- #
# Custom kernel entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] or None (binary mask)
            - weights: dict of model parameters (float32 tensors)
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as input_tensor.
    """
    # -------------------------- unpack inputs --------------------------- #
    x, mask, weights, config = data
    device = x.device
    orig_dtype = x.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------------------- 1. Input LayerNorm ------------------------ #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -------------------------- 2. FP16 cast -------------------------- #
    x = x.to(torch.float16)

    # -------------------------- 3. Project & gate ---------------------- #
    # Cast weights to half‑precision once
    dtype = torch.float16
    left_proj_w = weights["left_proj.weight"].to(device=device, dtype=dtype)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=dtype)
    left_gate_w = weights["left_gate.weight"].to(device=device, dtype=dtype)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=dtype)
    out_gate_w = weights["out_gate.weight"].to(device=device, dtype=dtype)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=dtype)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=dtype)

    # Linear projections
    left_proj = F.linear(x, left_proj_w)          # [B, N, N, H]
    right_proj = F.linear(x, right_proj_w)        # [B, N, N, H]

    # Optional mask (broadcast over hidden dim)
    if mask is not None and not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)    # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Gating
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # [B, N, N, H]

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------- 4. Core O(N³·H) reduction ------------------- #
    B, N, _, _ = left.shape  # batch, seq_len, seq_len, hidden_dim
    # reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication: C = A @ Bᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------- 5. Hidden‑dim LayerNorm -------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ---------------------- 6. Output gate (elemwise) --------------- #
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B, N, N, H]
    out = _elemwise_mul(out, out_gate)                  # Triton kernel

    # ------------------- 7. Final projection back to dim ------------ #
    out = F.linear(out, to_out_w)  # [B, N, N, C]

    # ------------------- 8. Cast back to original dtype ------------- #
    out = out.to(orig_dtype)

    return out
