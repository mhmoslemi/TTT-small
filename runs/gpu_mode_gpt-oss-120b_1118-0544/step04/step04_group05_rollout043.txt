def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)

    # prepare weight tensors
    left_proj_weight = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device, dtype=torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device, dtype=torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device, dtype=torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device, dtype=torch.float16)
    to_out_weight = weights["to_out.weight"].to(device, dtype=torch.float16)

    # fused left projection + gate
    left_concat_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0)  # (2*H, dim)
    left_concat = F.linear(x, left_concat_weight)  # shape (B,N,N,2H)
    left_proj = left_concat[..., :hidden_dim]
    left_gate = torch.sigmoid(left_concat[..., hidden_dim:])

    # fused right projection + gate
    right_concat_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0)
    right_concat = F.linear(x, right_concat_weight)
    right_proj = right_concat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_concat[..., hidden_dim:])

    # mask
    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # reshape for batched matmul
    B, N, _, H = left.shape
    left = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, N)
    right = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # core reduction
    if N >= 512:
        out_flat = torch.bmm(left, right.transpose(-2, -1))
    else:
        out_flat = batched_matmul_triton(left, right)   # (B*H,N,N)

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # output layer norm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # out gate
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # (B,N,N,H)
    out = out * out_gate

    # final projection
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    # cast back to original dtype
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
```

"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the 4‑D input tensor.
2. Fuse projection + gate linear layers by concatenating the weights and
   applying a single `F.linear` call for the left and right branches.
3. Optionally mask the projected tensors.
4. Apply the sigmoid gates and combine with the projected tensors.
5. Fold the hidden dimension into the batch dimension and compute the core
   reduction `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.
   This is a batched GEMM `C = A @ Bᵀ` with `A,B` of shape (B·H, N, N).
   For modest N we use a custom Triton kernel; for larger N we fall back
   to `torch.bmm` (cuBLAS) which is faster on the target B200.
6. Reshape the result back to (B,N,N,H), apply the output LayerNorm,
   multiply by the output gate and finally project back to the original
   channel dimension.
7. Return the result in the original dtype.

The heavy O(N³·H) work is performed by an optimized batched GEMM kernel.
All other inexpensive operations are handled by PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B shape: (batch, M, K) and (batch, K, N))
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix dimensions (M=N in TriMul)
    stride_aa, stride_am, stride_ak,    # A: batch, row, col strides
    stride_bb, stride_bk, stride_bn,    # B: batch, row(K), col(N) strides (B already transposed)
    stride_cc, stride_cm, stride_cn,    # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)   # batch index (B*H)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column

    # ---------------------------------
    # Tile offsets
    # ---------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---------------------------------
    # Pointers to the start of the tile
    # ---------------------------------
    a_ptrs = A_ptr + pid_b * stride_aa \
                     + offs_m[:, None] * stride_am \
                     + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    b_ptrs = B_ptr + pid_b * stride_bb \
                     + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                     + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_mask[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store result in FP16
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + pid_b * stride_cc \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------
# Wrapper for the Triton batched GEMM
# --------------------------------------------------------------
def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for tensors of shape (batch, N, N) using Triton.
    All tensors must be half‑precision and reside on CUDA.
    """
    assert A.is_cuda and B.is_cuda
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2 and M == N, "TriMul expects square N×N matrices."

    # Choose block sizes based on the sequence length
    if M <= 64:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
        num_warps = 2
    elif M <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif M <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    elif M <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,   # rows
        (N + BLOCK_N - 1) // BLOCK_N,   # columns
        batch,                          # batch dimension (B·H)
    )

    C = torch.empty(batch, M, N, dtype=A.dtype, device=A.device)

    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),   # A strides
        B.stride(0), B.stride(1), B.stride(2),   # B (already transposed) strides
        C.stride(0), C.stride(1), C.stride(2),   # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )
    return C

# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask         : torch.Tensor of shape [B, N, N] or None (binary)
            - weights      : dict of model parameters (float32 tensors)
            - config       : dict with keys
                * "dim"          – input channel dimension C
                * "hidden_dim"   – hidden dimension H
                * "nomask"       – bool, if True ignore the mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ----------------------------------------------------------
    # Unpack
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------
    # 1) Input LayerNorm
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Work in FP16 for the heavy part
    x = x.to(torch.float16)

    # ----------------------------------------------------------
    # 2) Cast linear weights to FP16 (biases are only for LayerNorms)
    # ----------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ----------------------------------------------------------
    # 3) Fused projection + gate for left branch
    # ----------------------------------------------------------
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0)          # (2*H, C)
    left_concat   = F.linear(x, left_concat_w)                            # (B,N,N,2H)
    left_proj = left_concat[..., :hidden_dim]
    left_gate = torch.sigmoid(left_concat[..., hidden_dim:])               # (B,N,N,H)

    # ----------------------------------------------------------
    # 4) Fused projection + gate for right branch
    # ----------------------------------------------------------
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)       # (2*H, C)
    right_concat   = F.linear(x, right_concat_w)                          # (B,N,N,2H)
    right_proj = right_concat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_concat[..., hidden_dim:])             # (B,N,N,H)

    # ----------------------------------------------------------
    # 5) Optional mask (binary, broadcast over hidden dimension)
    # ----------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ----------------------------------------------------------
    # 6) Apply gates
    # ----------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ----------------------------------------------------------
    # 7) Fold hidden dimension into batch for GEMM
    # ----------------------------------------------------------
    B, N, _, H = left.shape
    left = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)    # (B*H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H, N, N)

    # ----------------------------------------------------------
    # 8) Core O(N³·H) reduction
    #    Use Triton for modest N, cuBLAS (torch.bmm) for larger N.
    # ----------------------------------------------------------
    if N >= 512:
        # cuBLAS path – highly optimized for large matrices
        out_flat = torch.bmm(left, right.transpose(-2, -1))
    else:
        # Triton batched GEMM
        out_flat = _batched_matmul_triton(left, right)

    # ----------------------------------------------------------
    # 9) Reshape back to (B, N, N, H)
    # ----------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # ----------------------------------------------------------
    # 10) Output LayerNorm over hidden dimension
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------------------------------------------
    # 11) Output gate
    # ----------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,H)
    out = out * out_gate

    # ----------------------------------------------------------
    # 12) Final linear projection back to original channel dimension
    # ----------------------------------------------------------
    out = F.linear(out, to_out_w)                      # (B,N,N,dim)

    # ----------------------------------------------------------
    # 13) Cast back to original dtype if needed
    # ----------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
