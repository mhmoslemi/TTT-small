@triton.jit
def tri_mul_batched_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, N, 
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # Compute total blocks
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_blocks_m * num_blocks_n)
    pid_in_batch = pid % (num_blocks_m * num_blocks_n)
    block_i = pid_in_batch // num_blocks_n
    block_j = pid_in_batch % num_blocks_n

    # Compute offsets
    i_start = block_i * BLOCK_M
    j_start = block_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    # Mask for i/j exceed N
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + batch * stride_left_batch + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # Load right tile: (BLOCK_N, BLOCK_K) then transpose to (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + batch * stride_right_batch + offs_j[:, None] * stride_right_i + offs_k[None, :] * stride_right_k
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        b_T = tl.trans(b)  # shape (BLOCK_K, BLOCK_N)

        # cast to float32
        a = a.to(tl.float32)
        b_T = b_T.to(tl.float32)

        acc += tl.dot(a, b_T)

    # Write to output
    out_ptrs = out_ptr + batch * stride_out_batch + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated batch GEMM.

The core of the TriMul operation is:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
where `left` and `right` are the projected and gated pairwise tensors.
For every hidden dimension `h` (and batch `b`) this is a matrix multiplication
of shape (N,N) @ (N,N)^T. We fuse all these independent matmuls into a single
Triton kernel that treats the combined batch index `b*h` as the batch dimension.
The kernel tiles the `i` and `j` output dimensions (BLOCK_M×BLOCK_N) and loops
over `k` in BLOCK_K chunks, accumulating in fp32. All other operations (layer‑norm,
linear projections, gating, masking) stay in PyTorch (fp16 where possible)."""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 * (B*H, N, N)   – left tensor
    right_ptr,         # fp16 * (B*H, N, N)   – right tensor
    out_ptr,           # fp32 * (B*H, N, N)   – output tensor
    batch,             # total number of (batch * hidden) = B*H
    N: tl.constexpr,   # sequence length
    stride_l_batch: tl.constexpr,  # stride for batch dimension in left
    stride_l_i: tl.constexpr,      # stride for i (row) dimension in left
    stride_l_k: tl.constexpr,      # stride for k (col) dimension in left
    stride_r_batch: tl.constexpr,  # same for right
    stride_r_i: tl.constexpr,
    stride_r_k: tl.constexpr,
    stride_o_batch: tl.constexpr,  # stride for batch dimension in out
    stride_o_i: tl.constexpr,      # stride for i (row) in out
    stride_o_j: tl.constexpr,      # stride for j (col) in out
    BLOCK_M: tl.constexpr,         # tile size for output rows
    BLOCK_N: tl.constexpr,         # tile size for output cols
    BLOCK_K: tl.constexpr,         # tile size for reduction axis
):
    pid = tl.program_id(0)

    # Number of tiles in each dimension
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # Decode program ID into (batch_id, block_i, block_j)
    batch_id = pid // (num_blocks_m * num_blocks_n)
    block_index = pid - batch_id * (num_blocks_m * num_blocks_n)
    block_i = block_index // num_blocks_n
    block_j = block_index - block_i * num_blocks_n

    # Compute start offsets for the tile
    i_start = block_i * BLOCK_M
    j_start = block_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointer for this batch (combined batch*hidden)
    left_batch_ptr = left_ptr + batch_id * stride_l_batch
    right_batch_ptr = right_ptr + batch_id * stride_r_batch
    out_batch_ptr = out_ptr + batch_id * stride_o_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M x K)
        a_ptrs = left_batch_ptr + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load right tile (N x K) and transpose to (K x N)
        b_ptrs = right_batch_ptr + offs_j[:, None] * stride_r_i + offs_k[None, :] * stride_r_k
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        b = tl.trans(b).to(tl.float32)   # now shape (K, N)

        acc += tl.dot(a, b)

    # Store the result
    out_ptrs = out_batch_ptr + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32 or float16)
            - mask: torch.Tensor of shape [B, N, N] (bool/int/float) or None
            - weights: dict of parameters (float32)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype as input)
    """
    torch.cuda.empty_cache()  # ensure a clean start for timing
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for heavy compute
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (dim -> hidden_dim)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)      # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 6. Fuse projection, gate and optional mask (in‑place to save memory)
    # ------------------------------------------------------------------
    left = left_proj
    right = right_proj
    left.mul_(left_gate)
    right.mul_(right_gate)

    if mask is not None and not nomask:
        # broadcast mask over hidden dimension
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left.mul_(mask_fp16)
        right.mul_(mask_fp16)

    # ------------------------------------------------------------------
    # 7. Prepare tensors for the batched GEMM:
    #    bring hidden dim into batch dimension and make them contiguous.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # (B, H, N, N) contiguous
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    batch = B * H
    left_flat = left.view(batch, N, N)   # (B*H, N, N), fp16
    right_flat = right.view(batch, N, N)

    # ------------------------------------------------------------------
    # 8. Allocate output buffer (fp32 accumulation)
    # ------------------------------------------------------------------
    out_flat = torch.empty((batch, N, N), dtype=torch.float32, device=device)

    # Compute strides for the flattened tensors (they are contiguous)
    stride_l_batch = left_flat.stride(0)
    stride_l_i = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_i = right_flat.stride(1)
    stride_r_k = right_flat.stride(2)

    stride_o_batch = out_flat.stride(0)
    stride_o_i = out_flat.stride(1)
    stride_o_j = out_flat.stride(2)

    # ------------------------------------------------------------------
    # 9. Launch Triton kernel for the core O(N³·H) reduction
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = batch * num_blocks_m * num_blocks_n
    grid = (total_programs,)

    _tri_mul_kernel[grid](
        left_ptr=left_flat,
        right_ptr=right_flat,
        out_ptr=out_flat,
        batch=batch,
        N=N,
        stride_l_batch=stride_l_batch,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_r_batch=stride_r_batch,
        stride_r_i=stride_r_i,
        stride_r_k=stride_r_k,
        stride_o_batch=stride_o_batch,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 10. Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H), fp32

    # ------------------------------------------------------------------
    # 11. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim), fp32

    # ------------------------------------------------------------------
    # 12. Cast back to original dtype and return
    # ------------------------------------------------------------------
    out = out.to(orig_dtype)
    return out
