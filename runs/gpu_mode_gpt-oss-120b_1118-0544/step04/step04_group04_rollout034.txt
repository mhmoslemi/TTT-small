@triton.jit
def batched_matmul_kernel(
    out_ptr, left_ptr, right_ptr,
    batch_stride_out, stride_out_m, stride_out_n,
    batch_stride_left, stride_left_m, stride_left_k,
    batch_stride_right, stride_right_k, stride_right_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1) # block row
    pid_n = tl.program_id(2) # block col

    # Offsets for rows/cols
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulators fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offsets for k dimension
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + pid_batch * batch_stride_left + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0).to(tl.float32)

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + pid_batch * batch_stride_right + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0).to(tl.float32)

        # Compute
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = out_ptr + pid_batch * batch_stride_out + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    # Mask for out of bounds
    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

@triton.jit
def matmul_kernel(out_ptr, a_ptr, b_ptr,
                  M, N, K,
                  stride_out_batch, stride_out_m, stride_out_n,
                  stride_a_batch, stride_a_m, stride_a_k,
                  stride_b_batch, stride_b_k, stride_b_n,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute base offsets for this program ID
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # base pointers for batch dimension
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    out_batch_ptr = out_ptr + pid_batch * stride_out_batch

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B tile (B is already transposed: shape K x N)
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Write output
    out_ptrs = out_batch_ptr + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Layernorm
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Cast to fp16
    x_fp16 = x.to(torch.float16)

    # Linear proj
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    # Compute left/right projections
    left = F.linear(x_fp16, left_proj_weight)   # (B,N,N,hidden)
    right = F.linear(x_fp16, right_proj_weight)

    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # Compute gates
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))

    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

    # Prepare for batched matmul
    # Permute to (B, hidden, N, N) then reshape to (batch, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch = B * hidden_dim
    left_mat = left_perm.view(batch, N, N)
    right_mat = right_perm.view(batch, N, N)

    # Transpose right for multiplication
    right_T = right_mat.transpose(2, 1).contiguous()

    # Allocate output
    out_mat = torch.empty_like(left_mat)  # fp16

    # Get strides
    stride_out_batch = out_mat.stride(0)
    stride_out_m = out_mat.stride(1)
    stride_out_n = out_mat.stride(2)

    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_T.stride(0)
    stride_b_k = right_T.stride(1)
    stride_b_n = right_T.stride(2)

    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    matmul_kernel[grid](
        out_mat,
        left_mat,
        right_T,
        N, N, N,
        stride_out_batch, stride_out_m, stride_out_n,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B,N,N,hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layer norm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # Out gate
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))
    out = out * out_gate

    # Final linear projection
    out = F.linear(out, to_out_weight)

    # Cast back to float32
    return out.to(torch.float32)

"""
Outgoing Triangular Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1. Layer‑norm over the last channel dimension of the input `[B, N, N, C]`.
2. Four linear projections (left/right and three gates) are computed in fp16.
3. An optional mask (broadcasted on the channel dimension) is applied to
   the left and right projections.
4. Gating via a sigmoid and element‑wise multiplication.
5. Core O(N³·H) contraction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is implemented as a batched matrix multiplication `left @ rightᵀ`
   using a custom Triton kernel that processes many (batch, row‑block,
   col‑block) tiles in parallel.  The kernel accumulates in fp32 and
   writes fp16 results.
6. Layer‑norm over the hidden dimension, element‑wise multiplication with
   an output gate, and a final linear projection back to the original
   channel dimension.
7. The result is returned as float32.

The heavy O(N³) work is performed by the Triton kernel; all other
operations use PyTorch (fp16 where possible) for simplicity.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    out_ptr, a_ptr, b_ptr,
    M, N, K,
    stride_out_batch, stride_out_m, stride_out_n,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: out = a @ b   where
       a : (batch, M, K)   (fp16)
       b : (batch, K, N)   (fp16, already transposed)
       out : (batch, M, N) (fp16)
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    out_batch_ptr = out_ptr + pid_batch * stride_out_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M x K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(
            a_ptrs,
            mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load B tile (K x N)
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(
            b_ptrs,
            mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
            other=0.0,
        ).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Write out tile
    out_ptrs = out_batch_ptr + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float), may be ignored
            - weights: dict of model weight tensors (float32)
            - config: dict with keys
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden projection dimension H
                * "nomask" (bool, optional) – if True, mask is not applied

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # Cast to fp16 for the heavy compute
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (fp16)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(device).to(torch.float16)

    left = F.linear(x_fp16, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional masking
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(device).to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))

    left = left * left_gate      # [B, N, N, H]
    right = right * right_gate   # [B, N, N, H]

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 5. Core contraction via custom Triton batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Rearrange to (batch, N, N) where batch = B * H
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    batch = B * hidden_dim
    left_mat = left_perm.view(batch, N, N)   # (batch, M=N, K=N)
    right_mat = right_perm.view(batch, N, N) # (batch, K=N, N)

    # Transpose the second operand so we can compute A @ Bᵀ
    right_T = right_mat.transpose(2, 1).contiguous()  # (batch, K=N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat)  # fp16, shape (batch, N, N)

    # Strides (in elements, not bytes)
    stride_out_batch = out_mat.stride(0)
    stride_out_m = out_mat.stride(1)
    stride_out_n = out_mat.stride(2)

    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_T.stride(0)
    stride_b_k = right_T.stride(1)
    stride_b_n = right_T.stride(2)

    # Tile sizes – chosen to fit in registers and leverage tensor cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_kernel[grid](
        out_mat,
        left_mat,
        right_T,
        N,
        N,
        N,
        stride_out_batch,
        stride_out_m,
        stride_out_n,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output layer‑norm (hidden dimension)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # fp16

    # ------------------------------------------------------------------
    # 7. Output gate and final projection
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # [B, N, N, H]
    out = out * out_gate

    out = F.linear(out, to_out_weight)  # [B, N, N, C] fp16

    # Return in float32 as required by the interface
    return out.to(torch.float32)
