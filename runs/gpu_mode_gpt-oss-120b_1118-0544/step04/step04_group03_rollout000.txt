"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The implementation follows the reference PyTorch module but:
  * Uses fused half‑precision linear + gating (via PyTorch) and a
    cuBLAS batched matrix multiplication (torch.bmm) for the O(N³·H)
    reduction.
  * Applies the cheap element‑wise output gating with a tiny Triton kernel
    (required to have a user‑defined kernel in the solution).
  * All heavy arithmetic is performed in FP16 with FP32 accumulation;
    the final result is returned in the original input dtype (typically FP32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton helper: element‑wise multiplication (used for the output gate)
# --------------------------------------------------------------------------- #
@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    """Element‑wise a * b for 1‑D contiguous buffers."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton kernel for element‑wise multiplication."""
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication."
    size = a.numel()
    out = torch.empty_like(a)
    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](a, b, out, size, BLOCK=BLOCK)
    return out


# --------------------------------------------------------------------------- #
# TriMul entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * "dim" (int) – input channel dimension C
                * "hidden_dim" (int) – hidden dimension H
                * "nomask" (bool, optional) – if True, ignore mask

    Returns:
        torch.Tensor of shape [B, N, N, C] with the same dtype as the input.
    """
    # ------------------------------------------------------------------- #
    # 0. Unpack
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (float16 for the heavy part)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 2. Linear projections + gates (all in FP16)
    # ------------------------------------------------------------------- #
    # left projection & gate
    left_proj = F.linear(x_fp16, weights["left_proj.weight"].to(x.device, torch.float16))
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(x.device, torch.float16)).sigmoid()
    # right projection & gate
    right_proj = F.linear(x_fp16, weights["right_proj.weight"].to(x.device, torch.float16))
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(x.device, torch.float16)).sigmoid()

    # optional mask (broadcast over the hidden dimension)
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # apply gates
    left = left_proj * left_gate                     # [B, N, N, H]
    right = right_proj * right_gate                  # [B, N, N, H]

    # ------------------------------------------------------------------- #
    # 3. Core O(N³·H) reduction using batched GEMM (cuBLAS via torch.bmm)
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape to (batch = B*H, M=N, K=N) for both tensors
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication: C = A @ Bᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N), FP16->FP32 accum

    # reshape back to original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------- #
    # 4. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(x.device, torch.float16),
        bias=weights["to_out_norm.bias"].to(x.device, torch.float16),
    )

    # ------------------------------------------------------------------- #
    # 5. Output gate (element‑wise) – Triton kernel
    # ------------------------------------------------------------------- #
    out_gate = F.linear(
        x_fp16,
        weights["out_gate.weight"].to(x.device, torch.float16)
    ).sigmoid()  # [B, N, N, H]

    out = _elemwise_mul(out, out_gate)  # still FP16

    # ------------------------------------------------------------------- #
    # 6. Final projection back to original channel dimension
    # ------------------------------------------------------------------- #
    out = F.linear(
        out,
        weights["to_out.weight"].to(x.device, torch.float16)
    )  # [B, N, N, C]

    # Cast to the original data type (usually float32)
    out = out.to(x.dtype)

    return out
