"""
custom_kernel: Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the 4‑D input (B,N,N,C).
2. Project to hidden dimension H with two linear layers (left/right).
3. Optionally apply a binary mask (broadcast over H).
4. Compute three sigmoid gates (left, right, out) from the normalized input.
5. Multiply left/right by their gates.
6. Fold H into the batch dimension and compute the core reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM C = A @ Bᵀ where A,B have shape (B·H, N, N).
   For N ≤ 512 a custom Triton kernel is used; for larger N we fall back
   to torch.bmm (cuBLAS) which is faster on the target hardware.
7. Reshape back to (B,N,N,H), apply a second LayerNorm over H,
   multiply by the output gate and project back to the original channel dim.
8. Return the result in the same dtype as the input.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B shape: (batch, N, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,                     # all equal to seq_len
    stride_ah, stride_am, stride_ak,   # A: batch, row, col strides
    stride_bh, stride_bk, stride_bn,   # B: batch, k, n strides (Bᵀ layout)
    stride_ch, stride_cm, stride_cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch index (B*H)
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers for the A and B tiles (K dimension will be iterated)
    # ------------------------------------------------------------------
    a_ptrs = A_ptr + pid_b * stride_ah + (offs_m[:, None] * stride_am
                                          + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = B_ptr + pid_b * stride_bh + (tl.arange(0, BLOCK_K)[:, None] * stride_bk
                                          + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        k_mask = (tl.arange(0, BLOCK_K) + k * BLOCK_K) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_mask[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = C_ptr + pid_b * stride_ch + (offs_m[:, None] * stride_cm
                                          + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for tensors of shape (batch, N, N) using Triton.
    All tensors must be half‑precision and reside on CUDA.
    """
    assert A.shape == B.shape, "A and B must have the same shape."
    batch, N, K = A.shape
    assert N == K, "TriMul requires square sequence dimension."

    # Heuristic block sizes
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:  # N > 512
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along M
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along N
        batch,                          # batch dimension (B*H)
    )

    C = torch.empty_like(A)

    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),      # A: batch, row, col
        B.stride(0), B.stride(2), B.stride(1),      # B as Bᵀ: batch, k, n
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )
    return C

    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (fp32 or fp16)
            - mask: torch.Tensor [B, N, N] or None (binary)
            - weights: dict with model parameters (float32)
            - config: dict with keys:
                * "dim"         – input channel dimension C
                * "hidden_dim" – hidden dimension H
                * "nomask"      – bool, ignore mask when True (default False)

    Returns:
        torch.Tensor: [B, N, N, C] with the same dtype as `input_tensor`.
    """
    # ----------------------------------------------------------
    # Unpack
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    orig_dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----------------------------------------------------------
    # 2) Cast to fp16 for heavy computation
    # ----------------------------------------------------------
    x = x.to(torch.float16)

    # 3) Cast weights to fp16
    # ----------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ----------------------------------------------------------
    # 4) Linear projections (B,N,N,H)
    # ----------------------------------------------------------
    left  = F.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)     # (B,N,N,hidden_dim)

    # ----------------------------------------------------------
    # 5) Optional mask (binary, broadcast over hidden_dim)
    # ----------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------
    # 6) Gates (sigmoid of linear projections)
    # ----------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # (B,N,N,hidden_dim)

    # ----------------------------------------------------------
    # 7) Apply gates to projections
    # ----------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 8) Fold hidden dimension into batch for GEMM
    # ----------------------------------------------------------
    B, N, _, H = left.shape
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    left  = left.view(B * H, N, N)                  # (B*H, N, N)
    right = right.view(B * H, N, N)                 # (B*H, N, N)

    # ----------------------------------------------------------
    # 9) Core O(N³·H) reduction
    # ----------------------------------------------------------
    # Use Triton for moderate N, cuBLAS (torch.bmm) for very large N.
    if N <= 512:
        out_flat = _batched_gemm_triton(left, right)          # (B*H, N, N)
    else:
        out_flat = torch.bmm(left, right.transpose(-2, -1))

    # ----------------------------------------------------------
    # 10) Reshape back to (B, N, N, H)
    # ----------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ----------------------------------------------------------
    # 11) Output LayerNorm over hidden dimension
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------------------------------------------
    # 12) Apply output gate
    # ----------------------------------------------------------
    out = out * out_gate

    # ----------------------------------------------------------
    # 13) Final linear projection back to original channel dimension
    # ----------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)

    # ----------------------------------------------------------
    # 14) Cast back to original dtype
    # ----------------------------------------------------------
    if out.dtype != orig_dtype:
        out = out.to(orig_dtype)

    return out
```
