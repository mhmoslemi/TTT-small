"""
Outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm:
1. Layer‑norm the input over the channel dimension.
2. Cast to fp16 and compute four linear projections (left/right and their gates)
   using the provided weights (no bias). Optionally apply a pairwise mask.
3. Fuse the projections with their sigmoid gates.
4. Perform the core O(N³) contraction:
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM of shape (B*hidden, N, N) × (B*hidden, N, N)ᵀ.
   It is implemented in a single Triton kernel that processes all batches
   simultaneously, using tiling (BLOCK_M×BLOCK_N) and a K‑loop of size
   BLOCK_K. Accumulators are float32 while loads/stores are fp16.
5. Layer‑norm over the hidden dimension, compute the output gate, and fuse
   the element‑wise multiplication with a tiny Triton kernel.
6. Final linear projection back to the original channel dimension and cast
   the result to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------
# Tiny element‑wise kernel for out *= out_gate (fp16)
# ---------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(out_ptr, gate_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    out = tl.load(out_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out * gate, mask=mask)

# Batched GEMM kernel for the triangle multiplicative update
# ---------------------------------------------------------
@triton.jit
def gemm_tri_mul_kernel(
    left_ptr,          # fp16, shape (BH, N, N)
    right_ptr,         # fp16, shape (BH, N, N)
    out_ptr,           # fp16, shape (BH, N, N)
    N: tl.constexpr,   # sequence length
    # strides
    stride_left_bh, stride_left_m, stride_left_k,
    stride_right_bh, stride_right_m, stride_right_n,
    stride_out_bh, stride_out_m, stride_out_n,
    # tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: [M, K]
        left_ptrs = (left_ptr
                     + batch_idx * stride_left_bh
                     + offs_m[:, None] * stride_left_m
                     + offs_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile transposed: we need right[batch, k, j]
        right_ptrs = (right_ptr
                      + batch_idx * stride_right_bh
                      + offs_k[:, None] * stride_right_m
                      + offs_n[None, :] * stride_right_n)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        # accumulate
        acc += tl.dot(left, right)

    # store the result as fp16
    out_ptrs = (out_ptr
                + batch_idx * stride_out_bh
                + offs_m[:, None] * stride_out_m
                + offs_n[None, :] * stride_out_n)
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

    # 2. Cast to fp16 for the remaining compute
    # -------------------------------------------------
    x_fp16 = x.half()  # [B, N, N, dim]

    # -------------------------------------------------
    # 3. Linear projections (fp16)
    # -------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()
    left_gate_w  = weights["left_gate.weight"].to(device).half()
    right_gate_w = weights["right_gate.weight"].to(device).half()
    out_gate_w   = weights["out_gate.weight"].to(device).half()

    left_proj  = F.linear(x_fp16, left_proj_w)   # (B, N, N, hidden)
    right_proj = F.linear(x_fp16, right_proj_w)

    # -------------------------------------------------
    # 4. Optional mask
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(device).half().unsqueeze(-1)  # (B, N, N, 1)
        left_proj  = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 5. Gating (sigmoid) and combine
    # -------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))

    left  = left_proj * left_gate   # (B, N, N, hidden)
    right = right_proj * right_gate

    # -------------------------------------------------
    # 6. Core triangle multiplicative update (batched GEMM)
    # -------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (BH, N, N) where BH = B * hidden_dim
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim

    left_flat  = left_perm.view(BH, N, N)   # fp16
    right_flat = right_perm.view(BH, N, N)

    out_flat = torch.empty_like(left_flat)  # fp16

    # strides (contiguous layout)
    s_l_bh, s_l_m, s_l_k = left_flat.stride()
    s_r_bh, s_r_m, s_r_n = right_flat.stride()
    s_o_bh, s_o_m, s_o_n = out_flat.stride()

    # tile sizes (tuned for fp16 matmul)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    gemm_tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        s_l_bh, s_l_m, s_l_k,
        s_r_bh, s_r_m, s_r_n,
        s_o_bh, s_o_m, s_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # reshape back to (B, N, N, hidden)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 7. Output layer‑norm over hidden dimension
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).half()
    to_out_norm_b = weights["to_out_norm.bias"].to(device).half()
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # fp16

    # -------------------------------------------------
    # 8. Output gate (linear + sigmoid) and fused multiply
    # -------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))  # (B, N, N, hidden) fp16

    total_el = out.numel()
    BLOCK = 1024
    grid_elem = ((total_el + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid_elem](
        out, out_gate, total_el, BLOCK=BLOCK,
    )

    # -------------------------------------------------
    # 9. Final linear projection back to original dim
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(device).half()
    out = F.linear(out, to_out_w)   # (B, N, N, dim) fp16

    # Cast to float32 as required by the interface
    return out.float()
