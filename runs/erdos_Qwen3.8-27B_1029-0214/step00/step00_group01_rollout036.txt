import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def waterfill(v, target_sum):
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo = -v.max()
    hi = 1.0 - v.min()
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v + mid, 0.0, 1.0)) < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)

def lse_obj_grad(h, beta, dx):
    n = len(h)
    f = np.correlate(h, 1.0 - h, mode='full') * dx
    fmax = f.max()

    se = e.sum()

        else:
            r = -k
            grad[:L] += wm * dx * (1.0 - h[r:])
            grad[r:] -= wm * dx * h[:L]
    return obj, grad
