import time
import math

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def is_feasible(h, tol=1e-6):
    target = len(h) / 2.0
    return (np.all(h >= -tol) and np.all(h <= 1.0 + tol) 
            and abs(np.sum(h) - target) < tol * max(1, len(h))
            and np.all(np.isfinite(h)))

def gridify(h, M):
    """Convert float h to integer grid H with sum(H) = M*len(h)//2"""
    n = len(h)
    target = M * n // 2
    H = np.clip(np.floor(h * M + 0.5).astype(int), 0, M)
    err = target - int(H.sum())
    if err > 0:
        # Spread err units to positions with most room
        order = np.argsort(-(M - H))
        for i in range(err):
            idx = order[i % n]
            while H[idx] >= M:
                i += 1
                idx = order[i % n]
            H[idx] += 1
    elif err < 0:
        err = -err
        order = np.argsort(H)
        for i in range(err):
            idx = order[i % n]
            while H[idx] <= 0:
                i += 1
                idx = order[i % n]
            H[idx] -= 1
    return H

def make_binary(n, rng):

        H = np.zeros(n, dtype=int)

        return H  # sum = n//2, but target for M=1 would be n/2 which isn't integer for odd n
    H = np.zeros(n, dtype=int)

    H = H.copy()

            units = 1

            H[t] += units
            h_new = H.astype(float) / M

                improved = True
                if new_score < best_score:

- Decrease h[i] where h[i+k] is small (h[i+k] is 0-ish means (1-h[i+k]) is large)

So the active pair selection should be: for the active lag k, find pairs (i, i+k) where h[i]*(1-h[i+k]) is large, and transfer from i to i+k.

import numpy as np
import time
import math

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def is_feasible(h, tol=1e-6):
    n = len(h)
    target = n / 2.0
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(float(np.sum(h)) - target) > tol * max(1, n):
        return False
    return True

def gridify(h, M):
    n = len(h)
    target = M * n // 2
    H = np.clip(np.round(h * M).astype(int), 0, M)
    err = target - int(H.sum())
    if err > 0:
        # Need to distribute err additional units
        # Give to positions with most slack
        slack = M - H
        order = np.argsort(-slack, kind='stable')
        for i in range(err):
            idx = int(order[i % n])
            while H[idx] >= M:
                i += 1
                idx = int(order[i % n])
                if i > n * 2:
                    # Fallback: just add to first available
                    for j in range(n):
                        if H[j] < M:
                            H[j] += 1
                            err -= 1
                            break
                    break
            H[idx] += 1
    elif err < 0:
        err = -err
        order = np.argsort(H, kind='stable')
        for i in range(err):
            idx = int(order[i % n])
            while H[idx] <= 0:
                i += 1
                idx = int(order[i % n])
                if i > n * 2:
                    for j in range(n):
                        if H[j] > 0:
                            H[j] -= 1
                            err -= 1
                            break
                    break
            H[idx] -= 1
    return H

def resample(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def active_lag_descent(H, M, rng, max_iters=500, time_limit=None, t_start=None):
    n = len(H)
    H = H.copy()

    current_score = eval_c5(h)
    best = H.copy()
    best_score = current_score
    stalled = 0
    
    for it in range(max_iters):
        if time_limit is not None and (time.perf_counter() - t_start) > time_limit:
            break
        
        # Compute correlation and find active lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        active_lag = int(np.argmax(corr))
        d = active_lag - (n - 1)
        
        # Generate candidate moves focused on active lag
        moves = []
        if d >= 0:
            valid_i = np.arange(0, max(1, n - d))
            products = h[valid_i] * (1.0 - h[valid_i + d])
            top = np.argsort(-products)[:15]
            for idx in top:
                i = int(valid_i[idx])
                if products[idx] > 1e-6:
                    moves.append((i, i + d, 1))
        else:
            dp = -d
            valid_i = np.arange(dp, n)
            products = h[valid_i] * (1.0 - h[valid_i - dp])
            top = np.argsort(-products)[:15]
            for idx in top:
                i = int(valid_i[idx])
                if products[idx] > 1e-6:
                    moves.append((i, i - dp, 1))
        
        # Add some random moves
        for _ in range(5):

            t = int(rng.integers(n))
            if s != t:

                if amt > 0:
                    moves.append((s, t, amt))
        
        improved = False
        for s, t, amt in moves:
            if s == t or s < 0 or s >= n or t < 0 or t >= n:
                continue
            max_units = min(int(H[s]), int(M - H[t]))
            if max_units <= 0:
                continue
            amt = min(amt, max_units)
            H[s] -= amt
            H[t] += amt
            h_new = H.astype(np.float64) / M
            new_score = eval_c5(h_new)
            if new_score < current_score - 1e-12:
                current_score = new_score
                improved = True
                if new_score < best_score - 1e-12:
                    best = H.copy()
                    best_score = new_score
            else:
                H[s] += amt
                H[t] -= amt

            stalled = 0
        else:

                        continue

                        continue

def simulated_annealing(H, M, rng, n_steps=20000, T_max=0.03, T_min=0.0005,

            break

        T = T_max * (T_min / T_max) ** progress
        
        # Pick random pair and direction
        s = int(rng.integers(n))
        t = int(rng.integers(n))
        if s == t:
            continue
        
        # Decide direction
        if rng.random() < 0.5:
            src, dst = s, t
        else:
            src, dst = t, s
        
        max_units = min(int(H[src]), int(M - H[dst]))
        if max_units <= 0:
            # Try other direction
            src, dst = dst, src
            max_units = min(int(H[src]), int(M - H[dst]))
            if max_units <= 0:
                continue
        
        # Choose amount (mostly 1, sometimes more)
        r = rng.random()
        if r < 0.6:
            units = 1
        elif r < 0.85:
            units = min(2, max_units)
        else:
            units = min(int(rng.integers(2, max(3, max_units + 1))), max_units)
        
        units = max(1, min(units, max_units))
        
        # Apply
        H[src] -= units
        H[dst] += units
        h_new = H.astype(np.float64) / M
        new_score = eval_c5(h_new)
        delta = new_score - current_score
        
        if delta < 0 or (T > 1e-15 and rng.random() < math.exp(-delta / T)):
            current_score = new_score
            h = h_new
            if new_score < best_score - 1e-12:
                best = H.copy()
                best_score = new_score
        else:
            H[src] += units
            H[dst] -= units
    
    return best, best_score

def make_binary_seed(n, rng):
    h = np.zeros(n, dtype=np.float64)
    if n % 2 == 0:
        ones = rng.permutation(n)[:n // 2]
        h[ones] = 1.0
    else:
        ones = rng.permutation(n - 1)[:n // 2]
        h[ones] = 1.0
        h[n - 1] = 0.5
        # Shuffle
        perm = rng.permutation(n)
        h = h[perm]
    return h

def make_sin_seed(n, rng):
    freq = int(rng.integers(1, 8))
    phase = rng.random() * 2 * np.pi
    amp = rng.uniform(0.2, 0.45)
    x = np.linspace(0, 2, n, endpoint=False)
    v = 0.5 + amp * np.sin(np.pi * freq * x + phase)
    return np.clip(v, 0, 1)
