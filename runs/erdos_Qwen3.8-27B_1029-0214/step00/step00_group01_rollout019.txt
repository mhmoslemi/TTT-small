    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_box_simplex(z, target):
    n = len(z)
    z = np.array(z, dtype=float)
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, 0.5)
    y = np.clip(z, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):

        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        # Distribute correction to interior points
        mask = (result > 1e-10) & (result < 1 - 1e-10)
        if np.any(mask):

            result[mask] += diff / n_int
            result = np.clip(result, 0.0, 1.0)
        else:
            # Find any point we can adjust
            idx = np.argmax(np.minimum(result, 1.0 - result))
            result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

    n = len(h)

    k = j - (n - 1)
    grad = np.zeros(n)

        grad[0:L] -= dx * h[k:n]
    else:

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_box_simplex(z, target):
    n = len(z)
    z = np.array(z, dtype=float).copy()
    z[~np.isfinite(z)] = 0.5
    y = np.clip(z, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if np.any(mask):
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)

    n = len(h)

    else:
        r = -k
        L = n - r
        grad[:L] += dx * (1.0 - h[r:])
        grad[r:] -= dx * h[:L]
    return grad

            J[idx, :self.n] = -grad_corr_row(h, j)
            J[idx, self.n] = 1.0
        return J

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_box_simplex(z, target):
    n = len(z)
    z = np.array(z, dtype=float).copy()
    bad = ~np.isfinite(z)
    if np.any(bad):
        z[bad] = 0.5
    y = np.clip(z, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        nmask = np.sum(mask)
        if nmask > 0:
            result[mask] += diff / nmask
            result = np.clip(result, 0.0, 1.0)
    return result

def grad_corr_row(h, j):
    n = len(h)
    dx = 2.0 / n
    k = j - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        L = n - k
        grad[k:] += dx * (1.0 - h[:L])
        grad[:L] -= dx * h[k:]
    else:
        r = -k
        L = n - r
        grad[:L] += dx * (1.0 - h[r:])
        grad[r:] -= dx * h[:L]
    return grad

class NLPProblem:
    def __init__(self, n, active_indices, deadline):
        self.n = n
        self.dx = 2.0 / n
        self.target = n / 2.0
        self.active = active_indices
        self.m = len(active_indices)
        self.deadline = deadline
        self.call_count = 0
    
    def check_time(self):
        self.call_count += 1
        if self.call_count > 50000:
            raise TimeoutError()
        if time.time() > self.deadline:
            raise TimeoutError()
    
    def objective(self, x):
        self.check_time()
        return x[self.n]
    
    def objective_grad(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g
    
    def eq_constraint(self, x):
        return np.sum(x[:self.n]) - self.target
    
    def eq_constraint_grad(self, x):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g
    
    def ineq_constraint(self, x):
        self.check_time()
        h = x[:self.n]
        t = x[self.n]
        corrs = np.correlate(h, 1.0 - h, mode="full") * self.dx
        return t - corrs[self.active]
    
    def ineq_constraint_jac(self, x):
        self.check_time()
        h = x[:self.n]
        n = self.n
        m = self.m
        J = np.zeros((m, n + 1))
        for idx in range(m):
            j = self.active[idx]
            J[idx, :n] = -grad_corr_row(h, j)
            J[idx, n] = 1.0
        return J
