    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    h_init = project_feasible(h_init, n0)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init, n0)
    best_n = n0
    print(f"Start: n={n0}, C5={best_c5:.6f}", flush=True)

    for alpha in [5, 10, 20, 50, 100, 200, 500, 1000]:
        if time.time() - t0 > budget_s - 120:
            break
        h_new = slsqp_solve(best_h, best_n, alpha, max_iter=120)
        h_new = project_feasible(h_new, best_n)
        c5 = compute_c5(h_new, best_n)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5
            print(f"  P1 a={alpha}: {best_c5:.6f}", flush=True)

    for alpha in [100, 300, 1000]:
        if time.time() - t0 > budget_s - 200:
            break
        h_new = projected_grad_step(best_h, best_n, alpha, lr=0.01, n_steps=200)
        c5 = compute_c5(h_new, best_n)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5
            print(f"  P1b a={alpha}: {best_c5:.6f}", flush=True)

    cur_n = best_n
    cur_h = best_h.copy()
    for _ in range(4):
        if time.time() - t0 > budget_s - 250:
            break
        new_n = cur_n * 2
        if new_n > 3600:
            break
        xo = np.arange(cur_n) * (2.0 / cur_n)
        xn = np.arange(new_n) * (2.0 / new_n)
        h_interp = np.interp(xn, xo, cur_h)
        h_interp = project_feasible(h_interp, new_n)

        for alpha in [10, 50, 200, 500]:
            if time.time() - t0 > budget_s - 120:
                break
            h_new = slsqp_solve(h_interp, new_n, alpha, max_iter=80)
            h_new = project_feasible(h_new, new_n)
            c5 = compute_c5(h_new, new_n)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
                best_n = new_n
                cur_n = new_n
                cur_h = h_new
                print(f"  P2 n={new_n} a={alpha}: {best_c5:.6f}", flush=True)
            else:
                h_interp = h_new

    for alpha in [500, 1000, 3000, 10000]:
        if time.time() - t0 > budget_s - 20:
            break
        h_new = slsqp_solve(best_h, best_n, alpha, max_iter=100)
        h_new = project_feasible(h_new, best_n)
        c5 = compute_c5(h_new, best_n)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5
            print(f"  P3 a={alpha}: {best_c5:.6f}", flush=True)

    for lr_init in [0.05, 0.02, 0.005]:
        if time.time() - t0 > budget_s - 10:
            break
        h_new = projected_grad_step(best_h, best_n, 2000, lr=lr_init, n_steps=300)
        c5 = compute_c5(h_new, best_n)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5
            print(f"  P4 lr={lr_init}: {best_c5:.6f}", flush=True)
            best_h = slsqp_solve(best_h, best_n, 500, max_iter=100)
            best_h = project_feasible(best_h, best_n)
            c5 = compute_c5(best_h, best_n)
            if c5 < best_c5:
                best_c5 = c5

    best_c5 = compute_c5(best_h, best_n)
    print(f"Final: n={best_n}, C5={best_c5:.6f}", flush=True)
    return best_h, best_c5, best_n
