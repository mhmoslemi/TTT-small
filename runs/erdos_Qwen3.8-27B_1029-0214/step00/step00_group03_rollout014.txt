```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_correlate(h):
    q = 1.0 - h
    return np.correlate(h, q, mode='full')

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_correlate(h)) * dx

def pair_optimal_d(h, corr, i, j):
    """Find optimal d for pair exchange h[i]+d, h[j]-d"""
    n = len(h)
    k_vals = np.arange(-(n-1), n)  # lag values
    # L_m = (1-h)[i-k] - (1-h)[j-k] - h[i+k] + h[j+k]
    L = np.zeros(2*n-1)
    q = 1.0 - h
    for m in range(2*n-1):
        k = m - (n-1)
        val = 0.0
        if 0 <= i-k < n:
            val += q[i-k]
        if 0 <= j-k < n:
            val -= q[j-k]
        if 0 <= i+k < n:
            val -= h[i+k]
        if 0 <= j+k < n:
            val += h[j+k]
        L[m] = val
    
    # Q_m
    Q = np.zeros(2*n-1)
    r = i - j  # this is the k for one of the -1 terms
    Q[n-1] = 2.0  # k=0
    if 0 <= r < n:  # k = i-j, m = n-1 + (i-j)
        Q[n-1+r] -= 1.0
    if 0 <= -r < n:  # k = j-i = -r, m = n-1 + (j-i) = n-1-r
        Q[n-1-r] -= 1.0
    
    # Feasible range
    d_lo = max(-h[i], h[j]-1.0)
    d_hi = min(1.0-h[i], h[j])
    if d_lo >= d_hi:
        return 0.0, c5_value(h)
    
    # Evaluate objective at sample points
    n_samples = 200
    d_vals = np.linspace(d_lo, d_hi, n_samples)
    objectives = np.max(corr + d_vals[:,None]*L[None,:] - (d_vals**2)[:,None]*Q[None,:], axis=1)
    best_idx = np.argmin(objectives)
    best_d = d_vals[best_idx]
    
    # Refine
    if best_idx > 0 and best_idx < n_samples-1:
        from scipy.optimize import minimize_scalar
        def obj(d):
            return np.max(corr + d*L - d*d*Q)
        result = minimize_scalar(obj, bounds=(d_vals[best_idx-1], d_vals[best_idx+1]), method='bounded')
        if result.success and result.fun < objectives[best_idx]:
            best_d = result.x
    
    new_corr = corr + best_d*L - best_d**2*Q
    new_c5 = np.max(new_corr) * (2.0/n)
    return best_d, new_c5
```

Updating W after a pair exchange h[i] += d, h[j] -= d:
W[p, m] changes only if p-k_m = i or j, or p+k_m = i or j.
That is, for a given m, W[p,m] changes if p = i+k_m or p = j+k_m or p = i-k_m or p = j-k_m.

This is a small update. For each m:
- If i-k_m is valid: W[i-k_m, m] += d (because q[i-k_m] decreased by d)
  Wait, no. h[i] increased by d, so q[i] = 1-h[i] decreased by d.
  W[p, m] = q[p-k_m] - h[p+k_m]
  
  If p-k_m = i: W[p,m] includes q[i] which decreased by d. So W[p,m] -= d.
  If p-k_m = j: W[p,m] includes q[j] which increased by d. So W[p,m] += d.
  If p+k_m = i: W[p,m] includes -h[i] which changed by -d. So W[p,m] += d.
  If p+k_m = j: W[p,m] includes -h[j] which changed by +d. So W[p,m] -= d.

So for each m, at most 4 entries of W change, each by ±d.
Total updates: O(2n) per pair exchange. That's fine.

Similarly, corr updates:
new_corr[m] = corr[m] + d*L[m] - d²*Q[m]
This is O(2n).

```python
    q = 1.0 - h
    return np.correlate(h, q, mode='full')

def compute_W(h):
    n = len(h)
    q = 1.0 - h
    W = np.zeros((n, 2*n-1))
    for p in range(n):
        for m in range(2*n-1):
            k = m - (n-1)
            val = 0.0
            if 0 <= p - k < n:
                val += q[p - k]
            if 0 <= p + k < n:
                val -= h[p + k]
            W[p, m] = val
    return W

def make_Q(n, i, j):
    Q = np.zeros(2*n-1)
    Q[n-1] = 2.0
    r = i - j
    if 0 <= n-1+r < 2*n-1:
        Q[n-1+r] -= 1.0
    if 0 <= n-1-r < 2*n-1:
        Q[n-1-r] -= 1.0
    return Q

def find_optimal_d(corr, L, Q, d_lo, d_hi, n):
    dx = 2.0/n
    if d_lo >= d_hi:
        return 0.0
    # Quick check: is d=0 already optimal?
    # Sample
    n_samples = 300
    d_vals = np.linspace(d_lo, d_hi, n_samples)
    # Objective: max_m (corr[m] + d*L[m] - d²*Q[m])
    # Vectorized
    obj = np.max(corr[None,:] + d_vals[:,None]*L[None,:] - (d_vals**2)[:,None]*Q[None,:], axis=1)
    best_idx = np.argmin(obj)
    best_d = d_vals[best_idx]
    best_val = obj[best_idx]
    # Refine with golden section
    if best_idx > 0 and best_idx < n_samples-1:
        lo = d_vals[max(0, best_idx-3)]
        hi = d_vals[min(n_samples-1, best_idx+3)]
        def f(d):
            return np.max(corr + d*L - d*d*Q)
        res = minimize_scalar(f, bounds=(lo, hi), method='bounded', options={'xatol': 1e-10})
        if res.success and res.fun < best_val:
            best_d = res.x
            best_val = res.fun
    return best_d

def update_W(W, i, j, d, n):
    """Update W after h[i]+=d, h[j]-=d"""
    for m in range(2*n-1):
        k = m - (n-1)
        # p-k = i → p = i+k
        p = i + k

        # p-k = j → p = j+k

        if 0 <= p < n:

        p = i - k
        if 0 <= p < n:

        # p+k = j → p = j-k
        p = j - k
        if 0 <= p < n:
            W[p, m] -= d  # -h[j] changed by +d, so W -= d
```

```python
    q = 1.0 - h
    return np.correlate(h, q, mode='full')

def compute_W(h):
    n = len(h)
    q = 1.0 - h
    W = np.zeros((n, 2*n-1))
    for p in range(n):
        for m in range(2*n-1):
            k = m - (n-1)
            val = 0.0
            if 0 <= p - k < n:
                val += q[p - k]
            if 0 <= p + k < n:
                val -= h[p + k]
            W[p, m] = val
    return W

def make_Q(n, i, j):
    Q = np.zeros(2*n-1)
    Q[n-1] = 2.0
    r = i - j
    mp = n-1+r
    mm = n-1-r
    if 0 <= mp < 2*n-1:
        Q[mp] -= 1.0
    if 0 <= mm < 2*n-1:
        Q[mm] -= 1.0
    return Q

def find_optimal_d(corr, L, Q, d_lo, d_hi):
    n_s = 150
    ds = np.linspace(d_lo, d_hi, n_s)
    mat = corr[None, :] + ds[:, None] * L[None, :] - (ds**2)[:, None] * Q[None, :]
    obj = mat.max(axis=1)
    idx = obj.argmin()
    best_d = ds[idx]
    best_obj = obj[idx]

        lo = ds[max(0, idx-3)]
        hi = ds[min(n_s-1, idx+3)]
```
