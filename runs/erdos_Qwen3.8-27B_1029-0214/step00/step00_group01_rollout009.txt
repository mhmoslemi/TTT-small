def compute_corr_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    
    # Compute all lags
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    # corr has 2n-1 entries, index m corresponds to lag m-(n-1)
    
    # Compute weights for log-sum-exp
    max_c = np.max(corr)
    exp_vals = np.exp(beta * (corr - max_c))
    sum_exp = np.sum(exp_exp)
    weights = exp_vals / sum_exp  # softmax weights
    
    # Smooth objective
    F = (max_c + np.log(sum_exp)) / beta
    
    # Compute gradient
    grad = np.zeros(n)
    for m in range(2*n - 1):
        lag = m - (n - 1)
        w = weights[m] * dx  # weight includes dx scaling
        if w < 1e-15:
            continue
        L = n - abs(lag)
        if lag >= 0:
            # c = sum_{i=0}^{L-1} h[i+lag]*(1-h[i])
            # dc/dh[j] = (1-h[j-lag])*I(lag<=j<=n-1) - h[j+lag]*I(0<=j<=L-1)
            grad[lag:n] += w * one_minus_h[:L]  # (1-h[j-lag]) for j in [lag, n-1]
            grad[:L] -= w * h[lag:n]  # -h[j+lag] for j in [0, L-1]
        else:
            r = -lag
            # c = sum_{i=0}^{L-1} h[i]*(1-h[i+r])
            # dc/dh[j] = (1-h[j+r])*I(0<=j<=L-1) - h[j-r]*I(r<=j<=n-1)
            grad[:L] += w * one_minus_h[r:]  # (1-h[j+r]) for j in [0, L-1]
            grad[r:] -= w * h[:L]  # -h[j-r] for j in [r, n-1]
    
    return F, grad

import numpy as np
import time

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')
    return float(np.max(corr * dx))

def waterfill(v, target):
    n = len(v)
    v = v.astype(np.float64)
    # Binary search for shift
    lo = -2.0 - np.max(np.abs(v))
    hi = 2.0 + np.max(np.abs(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    return h

def safe_logit(h, eps=1e-7):
    h_clipped = np.clip(h, eps, 1.0 - eps)

    exp_vals = np.exp(beta * (corr - max_c))
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    
    F = (max_c + np.log(sum_exp)) / beta
    
    # Gradient of F w.r.t. h

            G[:L] -= w * h[lag:n]

            r = -lag
            G[:L] += w * one_minus_h[r:n]
            G[r:n] -= w * h[:L]

    h_clipped = np.clip(h, 1e-10, 1.0 - 1e-10)

        return F, np.zeros_like(y)
    G_bar = np.dot(G, w_i) / sum_w
    grad_y = w_i * (G - G_bar)
    
    return F, grad_y

import numpy as np
import time
import math
from scipy.optimize import minimize


def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')
    return float(np.max(corr * dx))


def waterfill(v, target):
    v = np.asarray(v, dtype=np.float64)
    lo = -3.0 - float(np.max(np.abs(v)))
    hi = 3.0 + float(np.max(np.abs(v)))
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)


def safe_logit(h, eps=1e-7):
    h_c = np.clip(h, eps, 1.0 - eps)
    return np.log(h_c / (1.0 - h_c))


def safe_sigmoid(x):
    x = np.clip(x, -40.0, 40.0)
    out = np.empty_like(x, dtype=np.float64)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out


def find_lambda(y, target):
    lo = -float(np.max(y)) - 25.0
    hi = -float(np.min(y)) + 25.0
    for _ in range(20):
        if np.sum(safe_sigmoid(y + lo)) >= target:
            break
        lo -= 15.0
    for _ in range(20):
        if np.sum(safe_sigmoid(y + hi)) <= target:
            break
        hi += 15.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(safe_sigmoid(y + mid)) < target:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0


def y_to_h(y, target):
    lam = find_lambda(y, target)
    h = safe_sigmoid(y + lam)
    return waterfill(h, target)


def smooth_obj_and_grad(y, target, beta):
    h = y_to_h(y, target)
    n = len(h)
    dx = 2.0 / n
    omdh = 1.0 - h
    corr = np.correlate(h, omdh, mode='full') * dx
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    exp_vals = np.exp(np.minimum(shifted, 0.0))
    sum_exp = np.sum(exp_vals)
    if sum_exp < 1e-300:
        return 1e10, np.zeros_like(y)
    weights = exp_vals / sum_exp
    F = max_c + np.log(sum_exp) / beta

    G = np.zeros(n, dtype=np.float64)
    for m in range(2 * n - 1):
        lag = m - (n - 1)
        w = weights[m] * dx
        if w < 1e-15:
            continue
        L = n - abs(lag)
        if lag >= 0:
            G[lag:n] += w * omdh[:L]
            G[:L] -= w * h[lag:n]
        else:
            r = -lag
            G[:L] += w * omdh[r:n]
            G[r:n] -= w * h[:L]

    h_c = np.clip(h, 1e-10, 1.0 - 1e-10)
    w_i = h_c * (1.0 - h_c)
    sum_w = float(np.sum(w_i))
    if sum_w < 1e-15:
        return F, np.zeros_like(y)
    G_bar = float(np.dot(G, w_i)) / sum_w
    grad_y = w_i * (G - G_bar)

    if not np.all(np.isfinite(grad_y)):
        grad_y = np.zeros_like(y)
    if not np.isfinite(F):
        return 1e10, np.zeros_like(y)
    return F, grad_y


def _wrapped_obj(y, target, beta, deadline):
    if time.monotonic() > deadline:
        raise RuntimeError("timeout")
    F, grad = smooth_obj_and_grad(y, target, beta)
    return F, grad

def transfer_resolution(h_src, n_target):
    n_src = len(h_src)
    if n_target == n_src:
        return h_src.copy()
    x_src = (np.arange(n_src, dtype=np.float64) + 0.5) * (2.0 / n_src)
    x_tgt = (np.arange(n_target, dtype=np.float64) + 0.5) * (2.0 / n_target)
    h_interp = np.interp(x_tgt, x_src, h_src)
    target = n_target / 2.0
    return waterfill(h_interp, target)


def construct_fallback(n=64):
    x = (np.arange(n, dtype=np.float64) + 0.5) * (2.0 / n)
    raw = 0.5 + 0.25 * np.sin(np.pi * x)

def local_search_pairs(h, target, n_evals_max, deadline):
    n = len(h)
    best_c = exact_c5(h)
    rng = np.random.RandomState(12345)
    n_evals = 0

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = 0.5 / n
        lo = max(0.0, h[i] - delta + h[j] - 1.0)
        hi = min(1.0, h[i] + delta + h[j])
        if hi <= lo:
            delta = -delta
            lo = max(0.0, h[i] - delta + h[j] - 1.0)
            hi = min(1.0, h[i] + delta + h[j])
            if hi <= lo:
                continue
        t = (lo + hi) / 2.0
        h_new = h.copy()
        h_new[i] = h[i] + t
        h_new[j] = h[j] - t
        if h_new[i] < -1e-15 or h_new[i] > 1.0 + 1e-15:
            continue
        if h_new[j] < -1e-15 or h_new[j] > 1.0 + 1e-15:
            continue
        c = exact_c5(h_new)
        n_evals += 1
        if c < best_c - 1e-12:
            best_c = c
            h = h_new
    return h, best_c
