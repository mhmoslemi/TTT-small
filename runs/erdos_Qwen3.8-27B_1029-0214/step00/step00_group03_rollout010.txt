        for idx in order:
            excess = v[idx]
            if excess <= 1e-15:
                continue
            rem = min(excess, remaining)
            v[idx] -= rem
            remaining -= rem
            if remaining < 1e-13:
                break
    return v

def _interpolate_resolution(h, new_n):
    n = len(h)
    x_old = np.arange(n, dtype=np.float64) * (2.0 / n) + 1.0 / n
    x_new = np.arange(new_n, dtype=np.float64) * (2.0 / new_n) + 1.0 / new_n
    h_new = np.interp(x_new, x_old, h)
    return _project_feasible(h_new, new_n)

def _eval_pair(h, i, j, d):
    n = len(h)
    h2 = h.copy()
    h2[i] += d
    h2[j] -= d
    q = 1.0 - h2
    return float(np.max(np.correlate(h2, q, mode='full'))) * (2.0 / n)

def _optimize_pair(h, i, j, n_iters=10):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return 0.0
    a, b = lo, hi
    for _ in range(n_iters):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = _eval_pair(h, i, j, m1)
        f2 = _eval_pair(h, i, j, m2)
        if f1 < f2:
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def _coordinate_descent(h, max_time, pair_iters=10):
    n = len(h)
    start = time.perf_counter()
    h = h.copy()
    current_c = _exact_c5(h)
    
    while time.perf_counter() - start < max_time:
        improved = False
        for i in range(n):
            if time.perf_counter() - start > max_time:
                break
            for j in range(i + 1, n):
                if time.perf_counter() - start > max_time:
                    break
                d = _optimize_pair(h, i, j, pair_iters)
                if abs(d) > 1e-10:
                    new_c = _eval_pair(h, i, j, d)
                    if new_c < current_c - 1e-12:
                        h[i] += d
                        h[j] -= d
                        current_c = new_c
                        improved = True
        if not improved:
            break
    
    return h, current_c

def _random_perturbation(h, n_perturbs, magnitude, rng):
    n = len(h)
    h2 = h.copy()
    for _ in range(n_perturbs):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        lo = max(-h2[i], h2[j] - 1.0)
        hi = min(1.0 - h2[i], h2[j])
        if hi - lo < 1e-12:
            continue
        d = float(rng.uniform(max(lo, -abs(hi-lo)*magnitude), min(hi, abs(hi-lo)*magnitude)))
        h2[i] += d
        h2[j] -= d
    return _project_feasible(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.perf_counter()
    rng = np.random.default_rng(seed)
    
    def elapsed():
        return time.perf_counter() - t0
    
    def time_left(frac=0.95):
        return budget_s * frac - elapsed()
    
    # Initialize
    try:
        h0 = np.array(initial_h_values, dtype=np.float64).flatten()
    except (NameError, UnboundLocalError):
        h0 = np.full(45, 0.5, dtype=np.float64)
    
    n0 = len(h0)
    h = _project_feasible(h0, n0)
    best_h = h.copy()
    best_c = _exact_c5(h)
    
    # Phase 1: Coordinate descent at initial resolution
    if time_left() > 60:
        t1 = min(120.0, time_left() * 0.35)
        h, c = _coordinate_descent(h, t1, pair_iters=12)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 2: Random restarts
    if time_left() > 60:
        n_restarts = 30
        for r in range(n_restarts):
            if time_left() < 60:
                break
            # Different perturbation strengths
            mag = 0.3 + 0.7 * (r % 4) / 3.0
            n_pert = 3 + (r % 5) * 2
            h_pert = _random_perturbation(best_h, n_pert, mag, rng)
            # Also try completely random start occasionally
            if r % 10 == 9:
                h_pert = _project_feasible(rng.random(len(best_h)), len(best_h))
            t_rem = time_left() - 60
            t_per = max(3.0, t_rem / max(1, n_restarts - r))
            h_r, c_r = _coordinate_descent(h_pert, t_per, pair_iters=8)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r.copy()
    
    # Phase 3: Resolution transfer
    if time_left() > 90:
        n_best = len(best_h)
        candidates = []
        for f in [1.5, 2, 2.5, 3]:
            tn = int(round(n_best * f))
            tn = tn if tn % 2 == 1 else tn + 1  # prefer odd
            if 63 <= tn <= 400 and tn != n_best:
                candidates.append(tn)
        for tn in [91, 137, 201, 271, 337]:
            if tn not in candidates and tn != n_best:
                candidates.append(tn)
        candidates = sorted(set(candidates))
        
        for tn in candidates:
            if time_left() < 90:
                break
            h_interp = _interpolate_resolution(best_h, tn)
            t_rem = time_left() - 90
            t_per = max(15.0, t_rem / max(1, len(candidates)))
            h_new, c_new = _coordinate_descent(h_interp, t_per, pair_iters=10)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    
    # Phase 4: Final polish
    if time_left() > 15:
        h_fin, c_fin = _coordinate_descent(best_h, time_left() - 5, pair_iters=12)
        if c_fin < best_c:
            best_c = c_fin
            best_h = h_fin.copy()
    
    # Final validation
    best_h = _project_feasible(best_h, len(best_h))
    best_c = _exact_c5(best_h)
    
    return best_h, best_c, len(best_h)
