def solve_lambda(y, target):
    # Find λ such that sum σ(y[i] + λ) = target
    # σ is sigmoid
    def f(lam):
        return np.sum(1.0 / (1.0 + np.exp(-(y + lam)))) - target
    
    # Bracket
    lo, hi = -50.0, 50.0
    flo, fhi = f(lo), f(hi)
    if flo * fhi > 0:
        # Try wider
        lo, hi = -200.0, 200.0
        flo, fhi = f(lo), f(hi)
    if flo * fhi > 0:
        return 0.0  # fallback
    
    for _ in range(100):
        mid = (lo + hi) / 2
        fm = f(mid)
        if abs(fm) < 1e-14:
            return mid
        if flo * fm <= 0:
            hi = mid
            fhi = fm
        else:
            lo = mid
            flo = fm
    return (lo + hi) / 2

    lam = solve_lambda(y, len(y) / 2.0)
    return 1.0 / (1.0 + np.exp(-(y + lam)))

    return np.log(h / (1 - h))

    C = np.dot(g, h * (1.0 - h)) / S
