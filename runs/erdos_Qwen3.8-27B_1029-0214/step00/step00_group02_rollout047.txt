        H_new[i] -= 1
        H_new[j] += 1
        new_score = score_h(H_new.astype(float) / M)
        delta = new_score - current_score
        if delta < 0 or np.random.random() < math.exp(-delta / (T + 1e-15)):
            H = H_new
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_H = H.copy()
        T *= cool
    return best_H, best_score

def multi_unit_polish(H, M, max_iters=200, time_budget=20.0):
    n = len(H)
    H = H.copy()
    t0 = time.perf_counter()
    current_score = score_h(H.astype(float) / M)
    
    for it in range(max_iters):
        if time.perf_counter() - t0 > time_budget:
            break
        h = H.astype(float) / M
        G = np.outer(h, 1.0 - h)
        flat_G = G.ravel()
        n_top = min(60, n * n)
        top_idx = np.argpartition(flat_G, -n_top)[-n_top:]
        
        improved = False
        for fi in top_idx:
            i = int(fi // n)
            j = int(fi % n)
            if i == j or H[i] <= 0 or H[j] >= M:
                continue
            max_units = min(H[i], M - H[j])
            for units in [max_units, max_units // 2, 2, 1]:
                if units <= 0 or units > max_units:
                    continue
                H_new = H.copy()
                H_new[i] -= units
                H_new[j] += units
                s = score_h(H_new.astype(float) / M)
                if s < current_score - 1e-14:
                    H = H_new
                    current_score = s
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return H, current_score

def try_resolution(h_start, n_start, n_try, M, rng, time_budget=20.0):
    if n_try % 2 != 0:
        n_try += 1
    x_old = np.linspace(0, 2, n_start)
    x_new = np.linspace(0, 2, n_try)
    h_res = np.interp(x_new, x_old, h_start, left=0.0, right=0.0)
    h_res = np.clip(h_res, 0, 1)
    H = grid_to_H(h_res, M)
    H_f = H.astype(float) / M
    s = score_h(H_f)
    H, s = grid_local_polish(H, M, max_iters=150, time_budget=time_budget * 0.6)
    H_f = H.astype(float) / M
    s2 = score_h(H_f)
    if s2 < s:
        s = s2
    H, s = multi_unit_polish(H, M, max_iters=100, time_budget=time_budget * 0.3)
    H_f = H.astype(float) / M
    s3 = score_h(H_f)
    if s3 < s:
        s = s3
    return H_f, s, n_try

def cosine_seed(n, rng, freq, phase, amp):
    x = np.linspace(0, 2, n)
    h = 0.5 + amp * np.cos(np.pi * freq * x + phase)
    return np.clip(h, 0, 1)

def binary_pattern_seed(n, rng):
    k = n // 2
    pos = np.zeros(n, dtype=int)
    chosen = set()
    while len(chosen) < k:
        i = int(rng.integers(0, n))
        if i not in chosen:
            chosen.add(i)
    for i in chosen:
        pos[i] = 1
    return pos.astype(float)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.perf_counter()
    reserve = 15.0
    
    def elapsed():
        return time.perf_counter() - t_start
    
    def remaining():
        return budget_s - elapsed() - reserve
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
        if n0 < 4:
            n0 = 64
            h0 = np.full(n0, 0.5)
    except (NameError, TypeError, Exception):
        n0 = 64
        h0 = np.full(n0, 0.5)
    
    h0 = np.clip(h0, 0, 1)
    s0 = np.sum(h0)
    if abs(s0 - n0 / 2.0) > 1e-10:
        diff = s0 - n0 / 2.0
        if diff > 0:
            order = np.argsort(h0)
            for i in order:
                if diff <= 1e-14:
                    break
                sub = min(diff, h0[i])
                h0[i] -= sub
                diff -= sub
        else:
            order = np.argsort(-h0)
            for i in order:
                if diff >= -1e-14:
                    break
                add = min(-diff, 1.0 - h0[i])
                h0[i] += add
                diff += add
    
    best_h = h0.copy()
    best_score = score_h(h0)
    best_n = n0
    
    M = 64
    H = grid_to_H(h0, M)
    H_f = H.astype(float) / M
    s_grid = score_h(H_f)
    if s_grid < best_score:
        best_score = s_grid
        best_h = H_f.copy()
        best_n = n0
    
    if remaining() > 20:
        H, s = grid_local_polish(H, M, max_iters=500, time_budget=min(80.0, remaining() * 0.35))
        H_f = H.astype(float) / M
        if s < best_score:
            best_score = s
            best_h = H_f.copy()
            best_n = n0
    
    if remaining() > 20:
        H, s = multi_unit_polish(H, M, max_iters=300, time_budget=min(40.0, remaining() * 0.25))
        H_f = H.astype(float) / M
        if s < best_score:
            best_score = s
            best_h = H_f.copy()
            best_n = n0
    
    if remaining() > 30:
        H_sa, s_sa = simulated_annealing(H, M, time_limit=min(80.0, remaining() * 0.35))
        H_sa_f = H_sa.astype(float) / M
        if s_sa < best_score:
            best_score = s_sa
            best_h = H_sa_f.copy()
            best_n = n0
            H = H_sa
        if remaining() > 15:
            H2, s2 = grid_local_polish(H_sa, M, max_iters=300, time_budget=min(30.0, remaining() * 0.25))
            H2_f = H2.astype(float) / M
            if s2 < best_score:
                best_score = s2
                best_h = H2_f.copy()
                best_n = n0
                H = H2
    
    if remaining() > 30:
        H3, s3 = simulated_annealing(H, M, time_limit=min(60.0, remaining() * 0.3))
        H3_f = H3.astype(float) / M
        if s3 < best_score:
            best_score = s3
            best_h = H3_f.copy()
            best_n = n0
            H = H3
            if remaining() > 15:
                H3, s3 = grid_local_polish(H, M, max_iters=200, time_budget=min(25.0, remaining() * 0.2))
                H3_f = H3.astype(float) / M
                if s3 < best_score:
                    best_score = s3
                    best_h = H3_f.copy()
                    best_n = n0
    
    if remaining() > 40:
        for n_try in [n0 + 16, n0 + 32, n0 * 2]:
            if n_try > 200 or remaining() < 25:
                break
            t_bud = min(40.0, remaining() * 0.3)
            h_res, s_res, nr = try_resolution(best_h, best_n, n_try, M, rng, time_budget=t_bud)
            if s_res < best_score:
                best_score = s_res
                best_h = h_res.copy()
                best_n = nr
    
    if remaining() > 25:
        for freq in [1.0, 1.5, 2.0, 2.5, 3.0]:
            if remaining() < 10:
                break
            h_cos = cosine_seed(n0, rng, freq, rng.uniform(0, 2 * np.pi), 0.25)
            Hc = grid_to_H(h_cos, M)
            Hc_f = Hc.astype(float) / M
            sc = score_h(Hc_f)
            if sc < best_score + 0.03:
                Hc, sc = grid_local_polish(Hc, M, max_iters=100, time_budget=min(15.0, remaining() * 0.2))
                Hc_f = Hc.astype(float) / M
                sc = score_h(Hc_f)
                if sc < best_score:
                    best_score = sc
                    best_h = Hc_f.copy()
                    best_n = n0
    
    if remaining() > 10:
        n_try = None
        if best_n <= 80:
            n_try = min(128, (best_n // 2 + 1) * 2 + 1)
            if n_try % 2 != 0:
                n_try += 1
        if n_try is not None and n_try != best_n:
            h_res, s_res, nr = try_resolution(best_h, best_n, n_try, 128, rng, time_budget=min(20.0, remaining() * 0.4))
            if s_res < best_score:
                best_score = s_res
                best_h = h_res.copy()
                best_n = nr
    
    best_h = np.clip(best_h, 0, 1.0)
    target_sum = best_n / 2.0
    current_sum = np.sum(best_h)
    if abs(current_sum - target_sum) > 1e-8:
        diff = current_sum - target_sum
        if diff > 0:
            order = np.argsort(best_h)
            for i in order:
                if diff <= 1e-14:
                    break
                sub = min(diff, best_h[i])
                best_h[i] -= sub
                diff -= sub
        else:
            order = np.argsort(-best_h)
            for i in order:
                if diff >= -1e-14:
                    break
                add = min(-diff, 1.0 - best_h[i])
                best_h[i] += add
                diff += add
    
    final_score = score_h(best_h)
    if final_score != best_score:
        best_score = final_score
    
    return (best_h, best_score, best_n)
