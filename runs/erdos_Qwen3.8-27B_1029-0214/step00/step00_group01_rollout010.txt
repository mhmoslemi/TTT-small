    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")

def project_to_feasible(x, target_sum):
    """Project x onto {h: 0 <= h_i <= 1, sum(h_i) = target_sum}"""
    n = len(x)
    # Bisection on lambda: sum(clip(x_i - lambda, 0, 1)) = target_sum
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction for exact sum
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        # Distribute among interior points
        interior = (result > 1e-12) & (result < 1 - 1e-12)
        if interior.any():
            result[interior] += diff / interior.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

    """Compute gradients of corr[i] for i in active_lags with respect to h."""
    n = len(h)
    grads = np.zeros((len(active_lags), n))
    for idx, i in enumerate(active_lags):

        g = np.zeros(n)
        if k >= 0:
            # corr[i] = sum_{m=0}^{n-1-k} h[m+k] * (1-h[m])
            for m in range(n - k):
                g[m + k] += (1 - h[m])
                g[m] -= h[m + k]

            # corr[i] = sum_{m=k_abs}^{n-1} h[m-k_abs] * (1-h[m])
            for m in range(-k, n):
                g[m + k] += (1 - h[m])
                g[m] -= h[m + k]
        grads[idx] = g

            best_h = h.copy()
            lr = min(lr * 1.1, 0.5)
        else:
            lr = max(lr * 0.8, 0.001)
    
    return best_h, best_c5

    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def project_feasible(x, target_sum):
    """Project onto {h: 0<=h_i<=1, sum(h_i)=target_sum}"""
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:

            g[m] -= h[m + k]

def project_feasible(x, target_sum):
    n = len(x)
    if n == 0:
        return x
    lo, hi = -3.0, 3.0
    for _ in range(120):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    result = np.clip(x - lam, 0.0, 1.0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-13:
        interior = (result > 1e-12) & (result < 1.0 - 1e-12)
        if interior.sum() > 0:

            result = np.clip(result, 0.0, 1.0)

def verify_gradient(h, lag_idx, idx):
    """Verify gradient of corr[lag_idx] w.r.t. h[idx] using finite differences."""
    n = len(h)
    eps = 1e-7
    h_plus = h.copy()
    h_plus[idx] += eps
    h_minus = h.copy()
    h_minus[idx] -= eps
    corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")[lag_idx]
    corr_minus = np.correlate(h_minus, 1 - h_minus, mode="full")[lag_idx]
    return (corr_plus - corr_minus) / (2 * eps)

def corr_gradient(h, lag_idx):
    n = len(h)
    k = lag_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:

        m_range = np.arange(n - k)
        g[m_range + k] += (1 - h[m_range])
        g[m_range] -= h[m_range + k]
    else:
        # k < 0, let ka = -k
        ka = -k
        m_range = np.arange(ka, n)
        # corr = sum_{m=ka}^{n-1} h[m-ka] * (1-h[m])
        g[m_range - ka] += (1 - h[m_range])

def smooth_gradient(h, beta):
    """Compute gradient of log-sum-exp smooth approximation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    # Softmax weights
    scaled = beta * corr
    scaled -= scaled.max()
    weights = np.exp(scaled)
    weights /= weights.sum()
    # Weighted average of gradients
    # grad = sum_k w_k * grad(corr_k) * dx (but we already have dx in corr)
    # Actually the smooth objective is (1/beta) * log(sum exp(beta * dx * corr_raw[k]))
    # gradient = dx * sum_k w_k * grad(corr_raw[k])

    scaled -= scaled.max()
    weights = np.exp(scaled)

    grad = np.zeros(n)
    for k_idx in range(len(corr_raw)):

    return grad

import numpy as np
from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(corr))

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -3.0, 3.0
    for _ in range(120):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    result = np.clip(x - lam, 0.0, 1.0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-13:
        interior = (result > 1e-12) & (result < 1.0 - 1e-12)
        if interior.sum() > 0:
            result[interior] += diff / interior.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def corr_gradient(h, lag_idx):
    n = len(h)
    k = lag_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        m_idx = np.arange(n - k)
        g[m_idx + k] += (1.0 - h[m_idx])
        g[m_idx] -= h[m_idx + k]
    else:
        ka = -k
        m_idx = np.arange(ka, n)
        g[m_idx - ka] += (1.0 - h[m_idx])
        g[m_idx] -= h[m_idx - ka]
    return g

def verify_gradients(h):
    n = len(h)
    for lag in [0, n - 1, 2 * n - 2, n - 2, 2 * n - 3]:
        if lag < 0 or lag >= 2 * n - 1:
            continue
        g_analytic = corr_gradient(h, lag)
        for j in [0, n // 2, n - 1]:
            eps = 1e-7
            hp = h.copy()
            hp[j] += eps
            hm = h.copy()
            hm[j] -= eps
            fd = (np.correlate(hp, 1.0 - hp, mode="full")[lag] -
                  np.correlate(hm, 1.0 - hm, mode="full")[lag]) / (2.0 * eps)
            if abs(fd - g_analytic[j]) > 1e-4:
                return False
    return True

def smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * dx * corr_raw
    scaled -= scaled.max()
    weights = np.exp(scaled)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights /= wsum
    grad = np.zeros(n)
    for k_idx in range(len(corr_raw)):
        if weights[k_idx] > 1e-14:
            grad += weights[k_idx] * corr_gradient(h, k_idx)
    grad *= dx
    return grad

def smooth_pg(h, target_sum, betas, iters_per_beta, time_budget):
    n = len(h)
    h = h.copy().astype(np.float64)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    t0 = time.time()
    for beta in betas:
        if time.time() - t0 > time_budget:
            break
        lr = 2.0 / (beta + 2.0)
        for it in range(iters_per_beta):
            if time.time() - t0 > time_budget:
                break
            grad = smooth_gradient(h, beta)
            grad = grad - np.mean(grad)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            step = h - lr * grad / gn * np.sqrt(gn)
            cand = project_feasible(step, target_sum)
            c5 = evaluate_c5(cand)
            if c5 < best_c5:
                best_c5 = c5
                best_h = cand.copy()
                h = cand.copy()
                lr = min(lr * 1.03, 1.0)
            else:
                lr *= 0.75
    return best_h, best_c5

def lp_direction(h, corr_raw, max_active, delta):
    n = len(h)
    sorted_idx = np.argsort(corr_raw)[::-1]
    active = sorted_idx[:min(max_active, len(sorted_idx))]
    grads = np.array([corr_gradient(h, idx) for idx in active])
    m = len(active)
    A_ub = np.zeros((m + 2 * n, n + 1))
    b_ub = np.zeros(m + 2 * n)
    A_ub[:m, :n] = grads
    A_ub[:m, n] = -1.0
    A_ub[m:m + n, :n] = np.eye(n)
    b_ub[m:m + n] = delta
    A_ub[m + n:m + 2 * n, :n] = -np.eye(n)
    b_ub[m + n:m + 2 * n] = delta
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None

def lp_refine(h, target_sum, time_budget, max_iters=150):
    n = len(h)
    h = h.copy().astype(np.float64)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    t0 = time.time()
    stall = 0
    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        if stall > 25:
            break
        corr_raw = np.correlate(h, 1.0 - h, mode="full")
        d = lp_direction(h, corr_raw, max_active=min(25, n), delta=0.5)
        if d is None:
            top = np.argmax(corr_raw)
            d = -corr_gradient(h, top)
        d = d - np.mean(d)
        dmax = np.max(np.abs(d))
        if dmax < 1e-13:
            break
        d = d / dmax
        improved = False
        for alpha in [0.12, 0.08, 0.05, 0.03, 0.015, 0.008, 0.004]:
            cand = h + alpha * d
            cand = project_feasible(cand, target_sum)
            c5 = evaluate_c5(cand)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = cand.copy()
                h = cand.copy()
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def random_perturb(h, target_sum, rng, scale, time_budget):
    n = len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    t0 = time.time()
    for _ in range(50):
        if time.time() - t0 > time_budget:
            break
        noise = rng.randn(n) * scale
        cand = h + noise
        cand = project_feasible(cand, target_sum)
        c5 = evaluate_c5(cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand.copy()
    return best_h, best_c5
