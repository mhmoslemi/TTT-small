def local_search(h, rng, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    c = np.correlate(h, one_minus_h, mode="full")
    current_c5 = np.max(c) * dx
    
    best_h = h.copy()
    best_c5 = current_c5
    
    # Temperature for simulated annealing
    T = 0.001
    T_min = 0.0001
    cooling = 0.999
    
    alpha = 0.01  # step size
    
    while time.time() - start_time < time_limit:
        # Choose a random move
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        sign = rng.choice([-1, 1])
        delta = alpha * rng.uniform(0.5, 1.5)
        
        new_i = h[i] + sign * delta
        new_j = h[j] - sign * delta
        
        if new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1:
            continue
        
        # Apply move
        h[i] += sign * delta
        h[j] -= sign * delta
        
        # Evaluate
        one_minus_h = 1.0 - h
        c = np.correlate(h, one_minus_h, mode="full")
        new_c5 = np.max(c) * dx
        
        if new_c5 < best_c5 or rng.random() < np.exp(-(new_c5 - best_c5) / T):
            # Accept (this is wrong for SA - should compare with current, not best)
            pass
        
        # Actually, for SA:
        # if new_c5 < current_c5: accept
        # else: accept with probability exp(-(new_c5 - current_c5) / T)
        
        if new_c5 >= current_c5 and rng.random() >= np.exp(-(new_c5 - current_c5) / T):
            # Reject
            h[i] -= sign * delta
            h[j] += sign * delta
        else:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        T = max(T * cooling, T_min)
    
    return best_h, best_c5
