            v = np.clip(v, 0.0, 1.0)
    return v


def build_A(q, n):
    """A[k, i] such that (A @ h)[k] = np.correlate(h, q, 'full')[k]"""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for p in range(2 * n - 1):
        s = n - 1 - p
        if s >= 0:
            idx = np.arange(0, n - s)
            A[p, idx] = q[idx + s]
        else:
            idx = np.arange(-s, n)
            A[p, idx] = q[idx + s]
    return A

            B[p, idx] = h[idx - s]
        else:
            idx = np.arange(0, n + s)
            B[p, idx] = h[idx - s]
    return B


def smooth_max_and_weights(y, beta):
    """Returns (smooth_max_value, softmax_weights)"""
    m = float(np.max(y))
    shifted = beta * (y - m)
    max_shifted = float(np.max(shifted))
    exps = np.exp(shifted - max_shifted)
    total = float(np.sum(exps))
    if total < 1e-300:
        w = np.ones_like(y) / len(y)
        return m + 1.0 / beta, w
    w = exps / total
    val = m + (1.0 / beta) * math.log(total)
    return val, w


def internal_objective(x, A, other, beta, gamma, target_ones):
    """Compute internal objective value and gradient for variable x"""
    y = A @ x
    val_sm, w = smooth_max_and_weights(y, beta)
    prox = gamma * (x + other - 1.0)
    grad = A.T @ w + prox
    val = val_sm + 0.5 * gamma * float(np.sum((x + other - 1.0) ** 2))
    return val, grad


def projected_grad_descent(x0, A, other, beta, gamma, target, max_iters, deadline, eval_func, best_ref):
    """
    Projected gradient descent on x to minimize smooth_max(A@x) + gamma/2*||x+other-1||^2
    subject to 0<=x<=1, sum(x)=target.
    Returns best x found (could be x0 if no improvement).
    """
    x = x0.copy()
    best_x = x0.copy()
    best_internal = internal_objective(x, A, other, beta, gamma, 1.0)[0]
    target = target
    step = 1.0 / (gamma + 10.0)
    
    for it in range(max_iters):
        if time.perf_counter() > deadline:
            break
        val_old, grad = internal_objective(x, A, other, beta, gamma, 1.0)
        
        # Backtracking line search
        accepted = False
        for ls in range(15):
            trial = x - step * grad
            trial = project_v(trial, target)
            val_new, _ = internal_objective(trial, A, other, beta, gamma, 1.0)
            if val_new < val_old - 1e-12:
                x = trial
                best_internal = val_new
                best_x = x.copy()
                accepted = True
                break
            step *= 0.5
        if not accepted:
            step *= 0.5
            if step < 1e-10:
                break
        
        # Update best based on exact evaluation (less frequently)
        if it % 5 == 0:
            c_exact = eval_func(x)
            if c_exact < best_ref[0] - 1e-10:
                best_ref[0] = c_exact
                best_x = x.copy()
    
    return best_x, best_internal

def exact_eval(h):
    h = np.asarray(h, dtype=np.float64).ravel()
    n = h.size

        return float('inf')

    remaining = deadline - time.perf_counter()
    if remaining > 5.0:
        n_best = len(best_h)
        h_final, c_final = pair_polish(best_h, n_best / 2.0, deadline, exact_eval)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()

    # Final verification
    best_h = project_v(best_h, len(best_h) / 2.0)
    best_c = exact_eval(best_h)
    if not np.isfinite(best_c):
        best_h = np.full(n0, 0.5)
        best_c = exact_eval(best_h)

    return best_h, float(best_c), int(len(best_h))
