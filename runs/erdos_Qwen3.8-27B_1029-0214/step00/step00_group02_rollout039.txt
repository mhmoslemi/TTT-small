import time

def evaluate_c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair_sum(h, target_sum):
    """Repair h to have the correct sum while staying in [0,1]"""
    h = h.copy()
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Add to coordinates with most room
        order = np.argsort(h)
        for idx in order:
            add = min(diff, 1.0 - h[idx])
            if add <= 0:
                continue
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    else:
        # Remove from coordinates with most value
        order = np.argsort(-h)
        for idx in order:
            rem = min(-diff, h[idx])
            if rem <= 0:
                continue
            h[idx] -= rem
            diff += rem
            if abs(diff) < 1e-12:
                break
    return h

def make_binary(n, rng):
    """Create a random binary vector with n/2 ones"""
    h = np.zeros(n)
    indices = rng.permutation(n)
    h[indices[:n//2]] = 1.0
    return h

def make_thue_morse(n):
    """Thue-Morse based balanced sequence"""
    h = np.zeros(n)
    for i in range(n):
        # Count ones in binary representation
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    # Adjust to have exactly n/2 ones
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        # Flip some 1s to 0s
        ones = np.where(h == 1.0)[0]
        for idx in ones[:n_ones - target]:
            h[idx] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0.0)[0]
        for idx in zeros[:target - n_ones]:
            h[idx] = 1.0
    return h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(h)
    else:
        order = np.argsort(-h)

            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            rem = min(-diff, h[idx])
            h[idx] -= rem
            diff += rem
        if abs(diff) < 1e-12:
            break
    return h
