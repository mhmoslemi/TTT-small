def project_feasible(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = np.sum(h) - target
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to reduce sum by diff
        # Reduce from elements that are largest (closest to 1)
        # Water-filling: reduce all by diff/n, but clip at 0
        # Better: sort and reduce from top
        order = np.argsort(h)[::-1]  # largest first
        remaining = diff
        for idx in order:
            reduce = min(h[idx], remaining)
            h[idx] -= reduce
            remaining -= reduce
            if remaining < 1e-12:
                break
    else:
        # Need to increase sum by -diff
        order = np.argsort(h)  # smallest first
        remaining = -diff
        for idx in order:
            increase = min(1.0 - h[idx], remaining)
            h[idx] += increase
            remaining -= increase
            if remaining < 1e-12:
                break
    return h

def eval_c5(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def is_feasible(h, tol=1e-8):
    n = len(h)
    return (np.all(h >= -tol) and np.all(h <= 1+tol) and abs(np.sum(h) - n/2.0) < tol * max(1, n))

def project_feasible(h, n):
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = np.sum(h) - target
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # reduce
        order = np.argsort(h)[::-1]
        remaining = diff
        for idx in order:
            red = min(h[idx], remaining)
            h[idx] -= red
            remaining -= red
            if remaining < 1e-14:
                break
    else:
        # increase
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            inc = min(1.0 - h[idx], remaining)
            h[idx] += inc
            remaining -= inc
            if remaining < 1e-14:
                break
    return h

def compute_grad(h, mu):
