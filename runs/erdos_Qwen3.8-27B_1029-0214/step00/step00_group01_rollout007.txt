import numpy as np
import time
from scipy.optimize import linprog

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_feasible(x, target_sum):
    n = len(x)
    # Bisection on lambda: sum clip(x_i - lambda, 0, 1) = target_sum
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    # Fine-tune
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def get_gradients(h, lags):
    n = len(h)
    one_minus_h = 1.0 - h
    m = len(lags)
    G = np.zeros((m, n))
    for idx, k in enumerate(lags):
        if k >= 0:
            G[idx, :n-k] += one_minus_h[k:]
            G[idx, k:] -= h[:n-k]
        else:
            m_ = -k
            G[idx, m_:] += one_minus_h[:n-m_]
            G[idx, :n-m_] -= h[m_:]
    return G

def solve_lp_direction(G, delta):
    n = G.shape[1]
    m = G.shape[0]
    # Variables: d_0,...,d_{n-1}, tau
    # Minimize tau
    # s.t. G_i dot d - tau <= 0 (i.e., G_i dot d <= tau)
    #      sum(d) = 0
    #      -delta <= d_i <= delta
    #      tau >= -infinity (but it will be bounded by the constraints)
    
    c = np.concatenate([np.zeros(n), [1.0]])
    A_ub = np.hstack([G, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.concatenate([np.ones(n), [0.0]])
    b_eq = [0.0]
    bounds = [(-delta, delta)] * n + [(-10.0, 10.0)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq.reshape(1,-1), b_eq=b_eq, bounds=bounds, method='highs')
    if result.success:
        return result.x[:n]
    return None

def local_optimize(h, max_iters, delta_init, budget_end):
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c, best_corr = eval_c5(best_h)
    
    delta = delta_init
    stall_count = 0
    
    for it in range(max_iters):
        if time.time() > budget_end:
            break
        
        # Get active lags (top correlations)
        corr_vals = best_corr
        corr_max = np.max(corr_vals)
        # Lags where corr is within 5% of max
        threshold = corr_max * 0.95
        active_mask = corr_vals >= threshold
        active_indices = np.where(active_mask)[0]
        # Cap the number of active constraints
        if len(active_indices) > 50:
            # Take top 50
            top_idx = np.argsort(corr_vals)[::-1][:50]
            active_indices = top_idx
        
        # Convert to lags (index r corresponds to lag r - (n-1))
        lags = active_indices - (n - 1)
        
        # Compute gradients
        G = get_gradients(best_h, lags)
        
        # Solve LP
        d = solve_lp_direction(G, delta)
        
        if d is None:
            # Fallback: use negative average gradient
            d = -G.mean(axis=0)
            d -= d.mean()  # zero sum
            d_norm = np.max(np.abs(d))
            if d_norm > 0:
                d = d / d_norm * delta
            else:
                delta *= 0.5
                continue
        
        # Ensure zero sum
        d -= d.mean()
        
        # Line search
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = best_h + alpha * d
            h_try = project_feasible(h_try, target_sum)
            c_try, corr_try = eval_c5(h_try)
            if c_try < best_c - 1e-8:
                best_h = h_try
                best_c = c_try
                best_corr = corr_try
                improved = True
                break
        
        if improved:
            stall_count = 0
            # Don't increase delta, keep same
        else:
            stall_count += 1
            delta *= 0.7
            if delta < 1e-6:
                break
        
        if stall_count > 20:
            break
    
    return best_h, best_c, best_corr

def upsample(h, factor):
    """Upsample by repeating each element 'factor' times"""
    return np.repeat(h, factor)
