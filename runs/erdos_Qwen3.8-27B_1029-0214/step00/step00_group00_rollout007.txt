def project_to_feasible(h, target_sum):

            hi = min(hi, (1.0 - h[idx]) / d[idx])
        elif d[idx] < -1e-14:
            lo = max(lo, (1.0 - h[idx]) / d[idx])
            hi = min(hi, -h[idx] / d[idx])
    if hi - lo < 1e-14:
        return None, None
    best_c5 = compute_c5(h)
    best_delta = 0.0
    deltas = np.linspace(lo, hi, n_evals)
    for delta in deltas:
        if abs(delta) < 1e-15:
            continue
        h_new = h + delta * d
        if h_new.min() < -1e-10 or h_new.max() > 1.0 + 1e-10:
            continue
        c5_val = compute_c5(h_new)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_delta = delta
    if best_delta == 0.0:
        return None, None
    h_new = h + best_delta * d
    return h_new, best_c5

def random_block_dir(n, rng):
    d = np.zeros(n)
    split = int(rng.integers(1, n))
    d[:split] = 1.0
    d[split:] = -float(split) / float(n - split)
    perm = rng.permutation(n)
    d = d[perm]
    d -= d.mean()
    return d

def random_sign_dir(n, rng):
    d = rng.choice([-1.0, 1.0], size=n)
    d -= d.mean()
    return d

def upsample_linear(h, new_n):
    old_n = len(h)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    return np.interp(new_x, old_x, h)

def do_pair_passes(h, n, c5, rng, max_passes, time_limit_s, target_c5):
    t0 = time.time()
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    for p in range(max_passes):
        if time.time() - t0 > time_limit_s:
            break
        if c5 <= target_c5:
            break
        rng.shuffle(pairs)
        improved = False
        for (i, j) in pairs:
            if time.time() - t0 > time_limit_s:
                break
            h_new, c5_new = pair_search(h, i, j)
            if h_new is not None:
                h = h_new
                c5 = c5_new
                improved = True
        if not improved:
            break
    return h, c5

def do_block_passes(h, n, c5, rng, max_iters, time_limit_s, target_c5):
    t0 = time.time()
    for it in range(max_iters):
        if time.time() - t0 > time_limit_s:
            break
        if c5 <= target_c5:
            break
        if it % 3 == 0:
            d = random_block_dir(n, rng)
        elif it % 3 == 1:
            d = random_sign_dir(n, rng)
        else:
            d = random_block_dir(n, rng)
            d = d * rng.choice([-1.0, 1.0])
        h_new, c5_new = block_dir_search(h, d)
        if h_new is not None:
            h = h_new
            c5 = c5_new
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent = np.array(initial_h_values, dtype=float).copy()
        pn = len(parent)
        parent = project_to_feasible(parent, pn / 2.0)
        parent_c5 = compute_c5(parent)
        best_h = parent.copy()
        best_c5 = parent_c5
        best_n = pn
    except Exception:
        best_n = 45
        best_h = np.full(best_n, 0.5)
        best_h = project_to_feasible(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    target = 0.38080

    # Phase 1: Intensive pair search at base resolution
    if remaining() > 60:
        h = best_h.copy()
        n = best_n
        c5 = best_c5
        tl = min(remaining() - 90, 250)
        h, c5 = do_pair_passes(h, n, c5, rng, max_passes=200, time_limit_s=tl, target_c5=target)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # Phase 2: Block direction search
    if remaining() > 60:
        h = best_h.copy()
        n = best_n
        c5 = best_c5
        tl = min(remaining() - 90, 60)
        h, c5 = do_block_passes(h, n, c5, rng, max_iters=5000, time_limit_s=tl, target_c5=target)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # Phase 3: Try complement and re-optimize
    if remaining() > 90:
        h_comp = 1.0 - best_h
        h_comp = project_to_feasible(h_comp, best_n / 2.0)
        c5_comp = compute_c5(h_comp)
        if c5_comp < best_c5 - 1e-6:
            h = do_pair_passes(h_comp, best_n, c5_comp, rng, max_passes=30, time_limit_s=min(remaining() - 60, 80), target_c5=target)
            if h[1] < best_c5:
                best_c5 = h[1]
                best_h = h[0].copy()
                best_n = best_n

    # Phase 4: Random restarts from perturbed copies
    if remaining() > 90:
        for restart in range(5):
            if remaining() < 40:
                break
            if best_c5 <= target:
                break
            perturbation = rng.normal(0, 0.05, size=best_n)
            h_pert = best_h + perturbation
            h_pert = project_to_feasible(h_pert, best_n / 2.0)
            c5_pert = compute_c5(h_pert)
            h2, c5_2 = do_pair_passes(h_pert, best_n, c5_pert, rng, max_passes=20, time_limit_s=min(remaining() - 30, 40), target_c5=target)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = best_n

    # Phase 5: Upsample to double resolution and optimize
    if remaining() > 100:
        n2 = 2 * best_n
        h2 = upsample_linear(best_h, n2)
        h2 = project_to_feasible(h2, n2 / 2.0)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2
        h2_opt = do_pair_passes(h2, n2, c5_2, rng, max_passes=30, time_limit_s=min(remaining() - 40, 150), target_c5=0.38078)
        if h2_opt[1] < best_c5:
            best_c5 = h2_opt[1]
            best_h = h2_opt[0].copy()
            best_n = n2

    # Phase 6: Final refinement
    if remaining() > 30:
        h = best_h.copy()
        n = best_n
        c5 = best_c5
        h, c5 = do_pair_passes(h, n, c5, rng, max_passes=20, time_limit_s=min(remaining() - 5, 60), target_c5=target)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # Final projection and validation
    best_h = project_to_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
