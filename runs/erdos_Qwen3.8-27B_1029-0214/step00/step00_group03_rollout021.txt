def exact_eval(h):
    """Evaluate C5 for a given h array."""
    h = np.asarray(h, dtype=np.float64).ravel()
    n = len(h)
    if n == 0:
        return None
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h onto feasible set: 0<=h[i]<=1, sum(h)=n/2."""
    h = np.asarray(h, dtype=np.float64).ravel().copy()
    if len(h) != n:
        h = h[:n] if len(h) >= n else np.pad(h, (0, n - len(h)))
    # Replace non-finite
    h[~np.isfinite(h)] = 0.5
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Adjust: need to add diff total mass
    if diff > 0:
        # Need to increase some elements
        slack = 1.0 - h  # max we can add to each
        order = np.argsort(-slack)  # most slack first
        for i in order:
            add = min(diff, slack[i])
            h[i] += add
            diff -= add
            if abs(diff) < 1e-14:
                break
    else:
        # Need to decrease some elements
        excess = h  # max we can remove from each
        order = np.argsort(-excess)  # most excess first
        for i in order:
            rem = min(-diff, excess[i])
            h[i] -= rem
            diff += rem
            if abs(diff) < 1e-14:
                break
    # Final correction for floating point
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        # Put residual in the element closest to 0.5
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h

    """Local pair optimization: try all pairs (i,j) and adjust."""
    best_c = exact_eval(h)

        passes += 1

            if elapsed > time_limit:
                break

            if time_limit is not None and (time.perf_counter() - start_time) > time_limit:
                break

                # Try moving mass from i to j or j to i
                # h[i] + delta, h[j] - delta (or reverse)
                # Range of delta:
                lo = max(-h[i], -(1.0 - h[j]))  # h[i]+d >= 0, h[j]-d <= 1
                hi = min(1.0 - h[i], h[j])  # h[i]+d <= 1, h[j]-d >= 0
                if hi - lo < 1e-12:
                    continue
                # Try a few values of d in [lo, hi]
                # The objective as function of d is piecewise (max of linear functions)
                # Try endpoints and some interior points
                candidates_d = [lo, hi, 0.0, (lo+hi)/2.0, lo + 0.25*(hi-lo), lo + 0.75*(hi-lo)]
                best_d = 0.0
                best_c_new = best_c
                for d in candidates_d:
                    if d <= lo - 1e-14 or d >= hi + 1e-14:
                        continue
                    h[i] += d
                    h[j] -= d

                    h[i] -= d

                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d

def exact_eval(h):
    h = np.asarray(h, dtype=np.float64).ravel()
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=np.float64).ravel().copy()
    if h.size != n:
        tmp = np.zeros(n)
        m = min(h.size, n)
        tmp[:m] = h[:m]
        h = tmp
    h[~np.isfinite(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        slack = 1.0 - h
        order = np.argsort(-slack)
        for idx in order:
            add = min(diff, slack[idx])
            if add <= 0:
                break
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-13:
                break
    else:
        excess = h.copy()
        order = np.argsort(-excess)
        for idx in order:
            rem = min(-diff, excess[idx])
            if rem <= 0:
                break
            h[idx] -= rem
            diff += rem
            if abs(diff) < 1e-13:
                break
    h = np.clip(h, 0.0, 1.0)
    resid = target - np.sum(h)
    if abs(resid) > 1e-13:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] += resid
        h = np.clip(h, 0.0, 1.0)
    return h

    while passes < max_passes:

        improved = False
        if time.perf_counter() >= deadline:
            break
        for i in range(n):
            if time.perf_counter() >= deadline:
                return h_work, best_c
            for j in range(i + 1, n):
                lo = max(-h_work[i], h_work[j] - 1.0)
                hi = min(1.0 - h_work[i], h_work[j])
                if hi - lo < 1e-12:

                best_d = 0.0

                    h_work[j] -= d

                    h_work[i] -= d
                    h_work[j] += d

                    h_work[j] -= best_d

                    improved = True

def solve_subproblem(A, target, n, gamma):
    x = cp.Variable(n)
    t = cp.Variable()
    # Objective: t + gamma/2 * ||x - target||^2
    obj = cp.Minimize(t + gamma * cp.sum_squares(x - target))

            return x.value

        pass

def exact_eval(h):
    h = np.asarray(h, dtype=np.float64).ravel()
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=np.float64).ravel().copy()
    if h.size != n:
        tmp = np.zeros(n)
        m = min(h.size, n)
        tmp[:m] = h[:m]
        h = tmp
    h[~np.isfinite(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        slack = 1.0 - h
        order = np.argsort(-slack)
        for idx in order:
            add = min(diff, slack[idx])
            if add <= 0:
                break
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-13:
                break
    else:

            diff += rem

                break
