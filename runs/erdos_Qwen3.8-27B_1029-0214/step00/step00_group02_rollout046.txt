                        break

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def gridify(h, M):
    n = len(h)
    target_int = M * n // 2
    Y = M * np.clip(h, 0.0, 1.0)
    base = np.floor(Y).astype(np.int64)
    base = np.clip(base, 0, M)
    err = target_int - int(np.sum(base))
    if err > 0:
        residuals = Y - base
        order = np.argsort(-residuals)
        for i in range(min(err, n * M)):
            idx = order[i % n]
            if base[idx] < M:
                base[idx] += 1
            elif err > n:
                # Force fit
                for j in range(n):
                    if base[j] < M:
                        base[j] += 1
                        break
    elif err < 0:
        residuals = Y - base
        order = np.argsort(residuals)
        for i in range(min(-err, n * M)):
            idx = order[i % n]
            if base[idx] > 0:
                base[idx] -= 1
            elif -err > n:
                for j in range(n):
                    if base[j] > 0:
                        base[j] -= 1
                        break
    return base

def structured_move(h, n, rng, scale):
    """Make a structured move: move mass from high-value positions to low-value positions."""
    s = rng.integers(0, n)
    t = rng.integers(0, n)
    if s == t:
        return None
    # Ensure s has more mass than t
    if h[s] < h[t]:
        s, t = t, s
    delta = scale * min(h[s], 1.0 - h[t])
    if delta < 1e-12:
        return None
    h_new = h.copy()
    h_new[s] -= delta
    h_new[t] += delta
    return h_new

def local_search(h, n, budget_s, start_time, max_iters=10000):
    best_h = h.copy()
    best_score = evaluate_c5(best_h)
    rng = np.random.default_rng(42)
    
    scales = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    it = 0
    no_improve = 0
    
    for scale in scales:
        while it < max_iters and no_improve < 2000:
            elapsed = time.perf_counter() - start_time
            if elapsed > budget_s - 2.0:
                return best_h, best_score
            
            # Choose move
            mode = rng.random()
            if mode < 0.4:
                # High to low
                thresh = np.median(best_h)
                high_idx = np.where(best_h > thresh)[0]
                low_idx = np.where(best_h < thresh)[0]
                if len(high_idx) == 0 or len(low_idx) == 0:
                    break
                s = high_idx[rng.integers(0, len(high_idx))]
                t = low_idx[rng.integers(0, len(low_idx))]
            elif mode < 0.7:
                # Random
                s = rng.integers(0, n)
                t = rng.integers(0, n)
                if s == t:
                    it += 1
                    continue
                if best_h[s] < best_h[t]:
                    s, t = t, s
            else:
                # Neighboring
                s = rng.integers(0, n)
                d = rng.integers(1, max(2, n // 4))
                t = (s + d) % n
                if best_h[s] < best_h[t]:
                    s, t = t, s
            
            delta = scale * rng.uniform(0.3, 1.0) * min(best_h[s], max(1e-6, 1.0 - best_h[t]))
            if delta < 1e-12:
                it += 1
                continue
            
            h_new = best_h.copy()
            h_new[s] -= delta
            h_new[t] += delta
            
            score_new = evaluate_c5(h_new)
            
            if score_new < best_score:
                best_h = h_new
                best_score = score_new
                no_improve = 0
            else:
                no_improve += 1
            
            it += 1
    
    return best_h, best_score

def sa_search(H, M, n, budget_s, start_time, init_T=0.005, end_T=0.0001, n_steps=20000):
    cur_H = H.copy()
    cur_h = cur_H.astype(float) / M
    cur_score = evaluate_c5(cur_h)
    
    best_H = cur_H.copy()
    best_h = cur_h.copy()
    best_score = cur_score
    
    rng = np.random.default_rng(7777)
    
    for step in range(n_steps):
        elapsed = time.perf_counter() - start_time
        if elapsed > budget_s - 2.0:
            break
        
        frac = step / max(n_steps - 1, 1)
        T = init_T * (end_T / init_T) ** frac
        
        # Choose move
        r = rng.random()
        if r < 0.3:
            # High to low
            vals = cur_H.astype(float)
            thresh = np.median(vals)
            high = np.where(vals > thresh)[0]
            low = np.where(vals < thresh)[0]
            if len(high) == 0 or len(low) == 0:
                s = rng.integers(0, n)
                t = rng.integers(0, n)
            else:
                s = high[rng.integers(0, len(high))]
                t = low[rng.integers(0, len(low))]
        elif r < 0.6:
            s = rng.integers(0, n)
            t = rng.integers(0, n)
        else:
            s = rng.integers(0, n)
            t = rng.integers(0, n)
        
        if s == t:
            continue
        
        units = int(rng.integers(1, 4))
        max_units = min(int(cur_H[s]), M - int(cur_H[t]))
        if max_units < 1:
            continue
        units = min(units, max_units)
        
        H_new = cur_H.copy()
        H_new[s] -= units
        H_new[t] += units
        
        h_new = H_new.astype(float) / M
        score_new = evaluate_c5(h_new)
        
        delta = score_new - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_H = H_new
            cur_h = h_new
            cur_score = score_new
            if cur_score < best_score:
                best_H = cur_H.copy()
                best_h = cur_h.copy()
                best_score = cur_score
    
    return best_H, best_h, best_score

def make_seed_balanced(n, M, rng):
    target_int = M * n // 2
    H = np.full(n, M // 2, dtype=np.int64)
    err = target_int - int(np.sum(H))
    if err != 0:
        idx = rng.permutation(n)
        if err > 0:
            for i in range(n):
                if err <= 0:
                    break
                j = idx[i]
                a = min(err, M - H[j])
                H[j] += a
                err -= a
        else:
            for i in range(n):
                if err >= 0:
                    break
                j = idx[i]
                r = min(-err, int(H[j]))
                H[j] -= r
                err += r
    return H

def make_seed_wave(n, M, rng):
    target_int = M * n // 2
    x = np.linspace(0, 2, n)
    n_freq = rng.integers(1, 4)
    h_raw = np.zeros(n)
    for k in range(1, n_freq + 1):
        amp = rng.uniform(0.05, 0.3) / k
        phase = rng.uniform(0, 2 * np.pi)
        h_raw += amp * np.sin(np.pi * k * x + phase)
    h_raw = 0.5 + h_raw
    return gridify(h_raw, M)
