            fc = evaluate_exact(hc)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            hd = base.copy(); hd[p] += d; hd[q] -= d
            fd = evaluate_exact(hd)
    # Best of c, d
    if fc < fd:
        return hc, fc
    return hd, fd

def compute_subgradient(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    k = k_star - (n - 1)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        r = k
        grad[r:] += (1.0 - h[:n - r])
        grad[:n - r] -= h[r:]
    else:
        r = -k
        grad[:n - r] += (1.0 - h[r:])
        grad[r:] -= h[:n - r]
    return grad, k

def gradient_informed_pairs(h, n_pairs=80):
    n = len(h)
    grad, k = compute_subgradient(h)
    # Also compute for top few lags
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_lags = np.argsort(corr)[-5:][::-1]
    combined_grad = np.zeros(n)
    for tl in top_lags:
        k_tl = int(tl) - (n - 1)
        g = np.zeros(n)
        if k_tl == 0:
            g[:] = 1.0 - 2.0 * h
        elif k_tl > 0:
            r = k_tl
            g[r:] += (1.0 - h[:n - r])
            g[:n - r] -= h[r:]
        else:
            r = -k_tl
            g[:n - r] += (1.0 - h[r:])
            g[r:] -= h[:n - r]
        weight = corr[tl] - corr[top_lags[-1]] + 0.1
        combined_grad += weight * g
    
    # Pairs: high grad paired with low grad
    indices = np.arange(n)
    # Sort by gradient
    sorted_idx = np.argsort(combined_grad)
    low_idx = sorted_idx[:n//2]
    high_idx = sorted_idx[n//2:]
    
    pairs = set()
    # Pair low with high
    for i in range(min(30, len(low_idx))):
        for j in range(min(3, len(high_idx))):
            p = int(low_idx[i])
            q = int(high_idx[j])
            if p != q:
                pairs.add((min(p,q), max(p,q)))
    
    # Adjacent pairs
    for i in range(n - 1):
        pairs.add((i, i + 1))
    
    # Some random
    return list(pairs)[:n_pairs]

def local_search(h, n_pairs_per_pass=100, max_passes=20, seed=42, deadline=None):
    n = len(h)
    rng = np.random.default_rng(seed)
    local_c = evaluate_exact(h)
    
    for pass_num in range(max_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        
        # Generate pairs
        if pass_num % 3 == 0:
            pairs = gradient_informed_pairs(h, n_pairs=n_pairs_per_pass)
        else:
            pairs = []
            for i in range(n - 1):
                pairs.append((i, i + 1))
            for _ in range(n_pairs_per_pass - len(pairs)):
                if _ < 0:
                    break
                p = int(rng.integers(n))
                q = int(rng.integers(n))
                if p != q:
                    pairs.append((p, q))
        
        # Shuffle for diversity
        if len(pairs) > 1:
            perm = rng.permutation(len(pairs))
            pairs = [pairs[i] for i in perm]
        
        for (p, q) in pairs:
            if deadline is not None and time.monotonic() > deadline:
                return h, local_c
            h_new, c_new = pair_search(h, p, q, n_gold=20)
            if c_new < local_c - 1e-11:
                h = h_new
                local_c = c_new
                improved = True
    
    return h, local_c
