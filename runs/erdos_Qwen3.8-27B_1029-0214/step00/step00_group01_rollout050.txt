                    pop_c[worst] = c_child

                if c_child < best_c - 1e-14:
                    best_h = h_child.copy()
                    best_c = c_child
                    best_n = n_res

            # Polish the best found by DE
            if _remaining() > 10.0:
                for _ in range(4):
                    if _remaining() < 5.0:
                        break
                    best_h, imp = _pair_pass(best_h, best_n / 2.0, max_pairs=40)
                    c2 = _eval(best_h)
                    if c2 < best_c - 1e-14:
                        best_c = c2
                    if not imp:
                        break

    # Final validation
    final_n = len(best_h)
    best_h = _project(best_h, final_n / 2.0)
    final_c = _eval(best_h)

    if not np.isfinite(final_c) or final_c > 0.5:
        best_h = np.full(64, 0.5)
        final_c = _eval(best_h)
        final_n = 64

    return best_h.copy(), final_c, final_n
