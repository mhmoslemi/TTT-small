def evaluate(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full')) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def is_feasible(h, n):
    """Check if h is feasible: 0<=h<=1, sum(h) = n/2"""
    if len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1 + 1e-10):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-8 * max(1, n):
        return False
    return True

def gridify(h, M):
    """Convert float h to integer grid H with sum(H) = M*n//2, 0<=H<=M"""
    n = len(h)
    target = (M * n) // 2
    H = np.clip(np.floor(h * M).astype(int), 0, M)
    err = target - int(np.sum(H))
    if err > 0:
        # Need to add err units
        # Sort by fractional part descending (those closest to next integer)
        frac = h * M - H
        order = np.argsort(-frac)
        for i in range(err):
            idx = order[i % n]
            if H[idx] < M:
                H[idx] += 1
        # If still not enough, spread
        remaining = target - int(np.sum(H))
        if remaining > 0:
            order2 = np.argsort(-H)  # prefer lower values
            for i in range(remaining):
                idx = order2[i % n]
                if H[idx] < M:
                    H[idx] += 1
    elif err < 0:
        # Need to remove |err| units
        frac = h * M - H
        order = np.argsort(frac)  # prefer those with smallest fractional part
        for i in range(-err):
            idx = order[i % n]
            if H[idx] > 0:
                H[idx] -= 1
        remaining = int(np.sum(H)) - target
        if remaining > 0:
            order2 = np.argsort(H)  # prefer higher values
            for i in range(remaining):
                idx = order2[i % n]
                if H[idx] > 0:
                    H[idx] -= 1
    return H

def local_grid_polish(H, M, rng, max_iters, time_limit):
    """Local improvement via pair transfers on integer grid"""
    n = len(H)
    H = H.copy()
    current_score = score_H(H, M)
    no_improve = 0
    t_start = time.perf_counter()

        it += 1

        top_s = np.argsort(-h)[:20]  # high h values (good sources)

                    continue

                H[ti] += 1

                H[ti] -= 1

            H[si] -= 1
            H[ti] += 1
            current_score -= best_improvement
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5:
                break
    
    return H, current_score

def simulated_annealing(H, M, rng, time_limit, initial_temp):
    """Simulated annealing on integer grid"""
    n = len(H)
    H_cur = H.copy()
    S_cur = score_H(H_cur, M)
    H_best = H_cur.copy()
    S_best = S_cur
    
    t_start = time.perf_counter()
    T = initial_temp
    cool_rate = 0.9995
    
    while (time.perf_counter() - t_start) < time_limit:
        # Choose a random move
        # Mix of: random pair, and active-pair
        h = H_cur.astype(np.float64) / M
        
        if rng.random() < 0.5:
            # Active pair: source from high h, target from low h
            src_candidates = np.where(H_cur > 0)[0]
            dst_candidates = np.where(H_cur < M)[0]
            if len(src_candidates) == 0 or len(dst_candidates) == 0:
                break
            # Weight by h value for source, by (1-h) for target
            s_weights = H_cur[src_candidates].astype(float)
            t_weights = (M - H_cur[dst_candidates]).astype(float)
            s_weights = s_weights / s_weights.sum()
            t_weights = t_weights / t_weights.sum()
            si = src_candidates[rng.choice(len(src_candidates), p=s_weights)]
            ti = dst_candidates[rng.choice(len(dst_candidates), p=t_weights)]
        else:
            # Random pair
            si = rng.integers(0, n)
            ti = rng.integers(0, n)
            if si == ti:
                continue
            if H_cur[si] <= 0 or H_cur[ti] >= M:
                continue
        
        # Random amount
        max_units = min(H_cur[si], M - H_cur[ti])
        units = rng.integers(1, max(2, max_units + 1))
        units = min(units, max_units)
        
        H_cur[si] -= units
        H_cur[ti] += units
        S_new = score_H(H_cur, M)
        delta = S_new - S_cur
        
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-10)):
            S_cur = S_new
            if S_cur < S_best:
                S_best = S_cur
                H_best = H_cur.copy()
        else:
            H_cur[si] += units
            H_cur[ti] -= units
        
        T = max(T * cool_rate, 1e-8)
    
    return H_best, S_best

def h_to_grid(h, M, n):
    target = M * n // 2
    H = np.clip(np.round(h * M).astype(int), 0, M)
    err = target - int(np.sum(H))
    if err > 0:
        order = np.argsort(-(M - H))  # prefer elements with most room (lowest H)
        # Actually prefer elements with highest fractional part
        H_float = np.clip(h * M, 0, M)
        frac = H_float - H
        order = np.argsort(-frac)
        for i in range(err):
            H[order[i % n]] += 1
    elif err < 0:
        H_float = np.clip(h * M, 0, M)
        frac = H_float - H
        order = np.argsort(frac)
        for i in range(-err):
            H[order[i % n]] -= 1
    return H

def grid_to_h(H, M):
    return H.astype(np.float64) / M

def local_opt(H, M, rng, max_time):
    n = len(H)
    H = H.copy()
    cur = eval_score(grid_to_h(H, M))
    t0 = time.perf_counter()
    stale = 0
    while time.perf_counter() - t0 < max_time:
        h = grid_to_h(H, M)
        # Get top sources (high h) and targets (low h)
        ns = min(25, n)
        top_s = np.argsort(-h)[:ns]
        top_t = np.argsort(h)[:ns]
        best_d = 0.0
        best_mv = None
        for si in top_s:
            if H[si] == 0:
                continue
            for ti in top_t:
                if si == ti or H[ti] == M:
                    continue
                H[si] -= 1
                H[ti] += 1
                ns2 = eval_score(grid_to_h(H, M))
                H[si] += 1
                H[ti] -= 1
                d = cur - ns2
                if d > best_d:
                    best_d = d
                    best_mv = (si, ti)
        if best_mv is not None:
            H[best_mv[0]] -= 1
            H[best_mv[1]] += 1
            cur -= best_d
            stale = 0
        else:
            stale += 1
            if stale > 50:
                break
    return H, cur

def sa_opt(H, M, rng, max_time, T0):
    n = len(H)
    Hc = H.copy()
    Sc = eval_score(grid_to_h(Hc, M))
    Hb = Hc.copy()
    Sb = Sc
    T = T0
    t0 = time.perf_counter()
    while time.perf_counter() - t0 < max_time:
        hc = grid_to_h(Hc, M)
        if rng.random() < 0.5:
            sm = Hc > 0
            dm = Hc < M
            if not np.any(sm) or not np.any(dm):
                break
            si = np.where(sm)[0][rng.integers(0, sm.sum())]
            ti = np.where(dm)[0][rng.integers(0, dm.sum())]
        else:
            si = rng.integers(0, n)
            ti = rng.integers(0, n)
        if si == ti or Hc[si] == 0 or Hc[ti] == M:
            continue
        mx = min(Hc[si], M - Hc[ti])
        u = int(rng.integers(1, min(mx, 5) + 1))
        Hc[si] -= u
        Hc[ti] += u
        Sn = eval_score(grid_to_h(Hc, M))
        dl = Sn - Sc
        if dl <= 0 or rng.random() < math.exp(-dl / max(T, 1e-15)):
            Sc = Sn
            if Sc < Sb:
                Sb = Sc
                Hb = Hc.copy()
        else:
            Hc[si] += u
            Hc[ti] -= u
        T = max(T * 0.9998, 1e-8)
    return Hb, Sb

def make_binary(n, rng):
    """Create random binary h with sum = n/2 (n must be even)"""
    H = np.zeros(n, dtype=int)
    idx = rng.permutation(n)[:n//2]
    H[idx] = 1  # M=1... but we need M*n//2 = sum. With M=1, sum should be n/2.
    # Wait, H values are 0 or 1, sum = n/2. That works for even n.
    return H  # This is for M=1

def make_spread_binary(n, rng):
    """Create a more spread-out binary pattern"""
    # Use a stride-based construction
    # Place n/2 ones as evenly as possible
    H = np.zeros(n, dtype=int)
    k = n // 2
    # Place at positions 0, n/k, 2n/k, ...
    positions = (np.arange(k) * n // k) % n
    H[positions] = 1
    return H
