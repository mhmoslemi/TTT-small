def project_box_sum(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target}"""
    h = np.clip(h, 0, 1)
    # Bisection on lambda: find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - 0.5*(lo+hi), 0, 1)
    # Fix any numerical issues
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-10:
        if diff > 0:

        else:
            idx = np.where(h_proj > 0.0)[0]

            h_proj[idx] += per

import numpy as np

def project_box_sum(h, target):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h_proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h_proj < 1.0 - 1e-15)[0]
        else:
            idx = np.where(h_proj > 1e-15)[0]
        if len(idx) > 0:
            per = diff / len(idx)
            h_proj[idx] = np.clip(h_proj[idx] + per, 0.0, 1.0)
    return h_proj

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

            return project_box_sum(u, target)
    except Exception:
        pass
    return None

def optimize_h(h, max_time, rng, current_best_tau):
    n = len(h)
    target = n / 2.0
    h_curr = h.copy()
    h_best = h.copy()
    tau_best = current_best_tau
    tau_curr = compute_c5(h_curr)
    if tau_curr < tau_best:
        tau_best = tau_curr
        h_best = h_curr.copy()
    start_time = time.monotonic()
    stall = 0
    max_stall = 25
    beta = 8.0
    trust = 0.25
    base_scale = 0.12
    it = 0
    while True:
        elapsed = time.monotonic() - start_time
        if elapsed > max_time - 3:
            break
        it += 1
        c_vals = compute_lag_values(h_curr)
        grad = smooth_gradient(h_curr, beta, c_vals)
        gnorm = float(np.linalg.norm(grad, ord=np.inf))
        if gnorm < 1e-14:
            break
        improved = False
        for alpha in [base_scale, base_scale*0.5, base_scale*0.25, base_scale*0.1]:
            step = alpha * grad
            step_norm = np.linalg.norm(step, ord=np.inf)
            if step_norm > 0.01:
                step = step * (0.01 / step_norm)
            h_trial = project_box_sum(h_curr - step, target)
            f_trial = compute_c5(h_trial)
            if f_trial < tau_curr - 1e-12:
                h_curr = h_trial
                tau_curr = f_trial
                improved = True
                if tau_curr < tau_best - 1e-12:
                    tau_best = tau_curr
                    h_best = h_curr.copy()
                    stall = 0
                break
        if not improved:
            c_vals2 = compute_lag_values(h_curr)
            max_lags = min(60, max(10, n // 4))
            u = local_lp_step(h_curr, c_vals2, trust, max_lags)
            if u is not None:
                f_lp = compute_c5(u)
                if f_lp < tau_curr - 1e-12:
                    h_curr = u
                    tau_curr = f_lp
                    improved = True
                    if tau_curr < tau_best - 1e-12:
                        tau_best = tau_curr
                        h_best = h_curr.copy()
                        stall = 0
        if improved:
            if stall > 0:
                stall = 0
        else:
            stall += 1
        if stall >= max_stall:
            h_curr = random_perturb(h_best, 0.04, rng)
            tau_curr = compute_c5(h_curr)
            if tau_curr < tau_best - 1e-12:
                tau_best = tau_curr
                h_best = h_curr.copy()
            stall = 0
            trust *= 0.7
            base_scale *= 0.85
        if it % 15 == 0:
            h_sym = symmetrize_h(h_curr)
            f_sym = compute_c5(h_sym)
            if f_sym < tau_curr - 1e-12:
                h_curr = h_sym
                tau_curr = f_sym
                if tau_curr < tau_best - 1e-12:
                    tau_best = tau_curr
                    h_best = h_curr.copy()
        if it % 40 == 0:
            beta *= 1.5
            if beta > 80.0:
                beta = 80.0
    return h_best, tau_best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n0 = 45
        h_init = np.zeros(n0)
        h_init[:n0//2] = 1.0
        h_init = project_box_sum(h_init, n0 / 2.0)

    h_init = project_box_sum(h_init, len(h_init) / 2.0)
    best_h = h_init.copy()
    best_tau = compute_c5(h_init)

    n_start = len(h_init)
    resolutions = sorted(set([n_start, min(n_start*2, 100), min(n_start*4, 200)]))
    if resolutions[0] < 45:
        resolutions = [r for r in resolutions if r >= 45]
        if not resolutions:
            resolutions = [45]

    h_work = h_init.copy()
    for i_res, n_res in enumerate(resolutions):
        remaining = budget_s - (time.monotonic() - t_start)
        if remaining < 25:
            break
        if n_res != len(h_work):
            h_work = interpolate_h(h_work, n_res)
        tau_work = compute_c5(h_work)
        if tau_work < best_tau - 1e-12:
            best_tau = tau_work
            best_h = h_work.copy()
        if i_res == len(resolutions) - 1:
            alloc = remaining - 10
        else:
            alloc = min(remaining * 0.35, 250)
        if alloc < 15:
            alloc = max(alloc, 10)
        h_opt, tau_opt = optimize_h(h_work, alloc, rng, best_tau)
        h_work = h_opt.copy()
        if tau_opt < best_tau - 1e-12:
            best_tau = tau_opt
            best_h = h_work.copy()

    remaining = budget_s - (time.monotonic() - t_start)
    if remaining > 20:
        n_best = len(best_h)
        for _ in range(min(8, int(remaining // 12))):
            remaining = budget_s - (time.monotonic() - t_start)
            if remaining < 8:
                break
            h_try = random_perturb(best_h, 0.02, rng)
            tau_try = compute_c5(h_try)
            if tau_try < best_tau - 1e-12:
                best_tau = tau_try
                best_h = h_try.copy()
    remaining = budget_s - (time.monotonic() - t_start)
    if remaining > 30:
        n_best = len(best_h)
        h_best_up = interpolate_h(best_h, min(n_best * 2, 240))
        h_opt2, tau_opt2 = optimize_h(h_best_up, remaining - 10, rng, best_tau)
        if tau_opt2 < best_tau - 1e-12:
            best_tau = tau_opt2
            best_h = h_opt2.copy()

    n_final = len(best_h)
    best_h = project_box_sum(best_h, n_final / 2.0)
    best_tau = compute_c5(best_h)
    return best_h, best_tau, n_final
