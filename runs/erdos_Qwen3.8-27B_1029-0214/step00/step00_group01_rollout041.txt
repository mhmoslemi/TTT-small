    n = len(h)
    if n < 1:

    return float(np.max(corr) * dx)

    n = len(h_raw)
    if n == 0:
        return h_raw.copy()
    h_raw = np.asarray(h_raw, dtype=np.float64)
    lo = -np.max(np.abs(h_raw)) - 2.0
    hi = 1.0 - np.min(h_raw) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h_test = np.clip(h_raw + mid, 0.0, 1.0)
        if np.sum(h_test) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h_raw + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h = h.copy()
            h[mask] += diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    expv = np.exp(shifted)
    sumexp = np.sum(expv)
    obj = cmax + math.log(sumexp) / beta
    w = expv / sumexp
    
    grad_h = np.zeros(n)
    nlags = 2 * n - 1
    for idx in range(nlags):
        wi = w[idx]
        if wi < 1e-15:
            continue
        s = idx - (n - 1)
        if s >= 0:
            L = n - s
            grad_h[s:s+L] += wi * dx * omh[:L]
            grad_h[:L] -= wi * dx * h[s:s+L]
        else:
            r = -s
            L = n - r
            grad_h[:L] += wi * dx * omh[r:r+L]
            grad_h[r:r+L] -= wi * dx * h[:L]
    
    return obj, grad_h

def lse_obj_and_grad_y(y, beta, target):
    h = make_feasible(y, target)
    obj, grad_h = lse_obj_and_grad_h(h, beta)
    s_arr = h * (1.0 - h)
    S = np.sum(s_arr)
    if S < 1e-14:
        return obj, np.zeros(len(y))
    wm = np.dot(grad_h, s_arr) / S
    grad_y = s_arr * (grad_h - wm)
    return obj, grad_y

    v, _ = lse_obj_and_grad_y(y, beta, target)
