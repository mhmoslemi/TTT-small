import numpy as np
import scipy.optimize
import time
import math

def exact_c5(h):
    n = len(h)
    if n < 2:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr))

def waterfill(v, target):
    n = len(v)
    if n == 0:
        return v
    lo = -np.max(v) - 2.0
    hi = 1.0 - np.min(v) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(v + lam, 0.0, 1.0)
    return result

def smooth_obj_and_grad(x, beta, n, mu):
    dx = 2.0 / n
    h = np.asarray(x, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -700.0, 700.0)
    exps = np.exp(shifted)
    sum_exps = np.sum(exps)
    if sum_exps < 1e-300 or not np.isfinite(sum_exps):
        val = max_c * 1e8 + mu * (np.sum(h) - n / 2.0) ** 2
        return val, 2.0 * mu * (np.sum(h) - n / 2.0) * np.ones(n)
    w = exps / sum_exps
    F = (max_c + np.log(sum_exps)) / beta
    s = np.sum(h) - n / 2.0
    F += mu * s * s
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        wm = w[m] * dx
        if abs(wm) < 1e-18:
            continue
        if k >= 0:
            if 0 < n - k:
                grad[k:n] += wm * (1.0 - h[0:n - k])
                grad[0:n - k] -= wm * h[k:n]
        else:
            r = -k
            if 0 < n - r:
                grad[0:n - r] += wm * (1.0 - h[r:n])
                grad[r:n] -= wm * h[0:n - r]
    grad += 2.0 * mu * s
    return F, grad

def obj_wrapper(x, beta, n, mu):
    return smooth_obj_and_grad(x, beta, n, mu)[0]

def grad_wrapper(x, beta, n, mu):
    return smooth_obj_and_grad(x, beta, n, mu)[1]

def upsample(h, n_target):
    n = len(h)
    if n == n_target:
        return h.copy()
    if n_target < n:
        x_old = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False) + 1.0 / n_target
        return np.interp(x_new, x_old, h)
    x_old = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False) + 1.0 / n_target
    return np.interp(x_new, x_old, h)

def make_fallback(n):
    if n % 2 == 1:
        n += 1
    h = np.zeros(n)
    h[0:n // 2] = 1.0
    return waterfill(h, n / 2.0)

def try_binary_construction(h_real, n):
    target = n // 2
    adjusted = list(h_real)
    order = list(range(n))
    order.sort(key=lambda i: -adjusted[i])
    binary = np.zeros(n)
    for i in range(target):
        binary[order[i]] = 1.0
    c5 = exact_c5(binary)
    if c5 < float('inf'):
        return binary, c5
    return None, float('inf')

def local_refine(h, n, budget_remaining, rng):
    target = n / 2.0
    best_h = h.copy()
    best_c5 = exact_c5(h)
    if budget_remaining < 1.0:
        return best_h, best_c5
    max_iter = min(5000, int(budget_remaining * 80))
    for it in range(max_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        trial = best_h.copy()
        trial[i], trial[j] = trial[j], trial[i]
        trial = waterfill(trial, target)
        c5 = exact_c5(trial)
        if c5 < best_c5 - 1e-12:
            best_h = trial
            best_c5 = c5
    return best_h, best_c5

def threshold_variants(h, n):
    target = n / 2.0
    sorted_idx = np.argsort(-h)
    variants = []
    num_ones = int(np.floor(target))
    for delta in [0, 1, -1, 2, -2]:
        k_ones = num_ones + delta
        if k_ones < 1 or k_ones >= n - 1:
            continue
        binary = np.zeros(n)
        binary[sorted_idx[:k_ones]] = 1.0
        half_idx = sorted_idx[k_ones]
        binary[half_idx] = 2.0 * (target - k_ones)
        if binary[half_idx] < 0 or binary[half_idx] > 1:
            continue
        c5 = exact_c5(binary)
        if c5 < float('inf'):
            variants.append((binary, c5))
    return variants

def optimize_resolution(h0, n, beta_list, mu, deadline, maxiter_per_beta):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = exact_c5(h)
    for beta in beta_list:
        if time.time() > deadline:
            break
        try:
            res = scipy.optimize.minimize(
                fun=obj_wrapper,
                x0=h,
                args=(beta, n, mu),
                jac=grad_wrapper,
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': maxiter_per_beta, 'maxfun': maxiter_per_beta * 3, 'ftol': 1e-12}
            )
            h_candidate = waterfill(res.x, n / 2.0)
            h_candidate = np.clip(h_candidate, 0.0, 1.0)
            c5_candidate = exact_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
                h = h_candidate.copy()
        except Exception:
            pass
    return best_h, best_c5
