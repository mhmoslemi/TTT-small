    target = n / 2.0
    
    h, lam = sigmoid_with_sum_constraint(y, target)
    omh = 1.0 - h

    sum_exp = np.sum(exp_shifted)

    w = exp_shifted / sum_exp
    
    grad_h = np.zeros(n)

        if abs(wk) < 1e-15:

def repair_h(h):
    h = np.asarray(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if n < 2:
        return h
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h.copy()
    lo = -1.0
    hi = 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    return result

    if not np.all(np.isfinite(h)):
        return np.inf
    n = len(h)
    if n < 4:
        return np.inf
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

    n = len(y)

    hi = 50.0

        if s < target:
            lo = mid
        else:
            hi = mid

    return scipy.special.expit(y + lam), lam

    sum_exp = np.sum(exp_s)
    if sum_exp < 1e-300:
        return mx / beta, np.zeros(n)
    F = (mx + np.log(sum_exp)) / beta

        wk = w[k] * dx
        if wk < 1e-18:

        L = n - abs(s)
        if s >= 0:
            grad_h[s:s + L] += wk * omh[:L]
            grad_h[:L] -= wk * h[s:s + L]
        else:
            r = -s
            grad_h[:L] += wk * omh[r:r + L]
            grad_h[r:r + L] -= wk * h[:L]
    jac = h * omh
    sum_jac = np.sum(jac)
    if sum_jac < 1e-15:
        return F, np.zeros(n)
    dF_dl = np.dot(grad_h, jac)
    grad_y = jac * (grad_h - dF_dl / sum_jac)
    if not np.all(np.isfinite(grad_y)):
        return F, np.zeros(n)
    return F, grad_y
