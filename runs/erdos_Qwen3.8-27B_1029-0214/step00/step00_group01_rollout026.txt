def project_capped_simplex(z, n, target):
    """Project z onto {h : 0 <= h_i <= 1, sum(h_i) = target}"""
    # Handle non-finite values
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, 0.5)
    z = np.clip(z, 0.0, 1.0)

    lo = z.min() - 1.0
    hi = z.max() + 1.0

import numpy as np

def project_capped_simplex(z, target):
    n = len(z)
    z = np.asarray(z, dtype=float).copy()
    z[~np.isfinite(z)] = 0.5
    z = np.clip(z, 0.0, 1.0)
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(z - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    err = target - float(h.sum())
    if abs(err) > 1e-13:
        interior = (h > 1e-15) & (h < 1.0 - 1e-15)
        ni = int(interior.sum())
        if ni > 0:
            h[interior] += err / ni
            h = np.clip(h, 0.0, 1.0)
    return h


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * float(np.max(corr)), corr


def compute_all_grads(h):
    n = len(h)
    grad = np.zeros((2 * n - 1, n))
    omh = 1.0 - h
    grad[n - 1] = 1.0 - 2.0 * h
    for k in range(1, n):
        m = k + n - 1
        grad[m, k:n] += omh[:n - k]
        grad[m, :n - k] -= h[k:n]
    for r in range(1, n):
        m = n - 1 - r
        grad[m, :n - r] += omh[r:]
        grad[m, r:] -= h[:n - r]
    return grad


def pair_golden(corr, g, lo, hi):
    if hi - lo < 1e-15:
        t = (lo + hi) * 0.5
        return t, float(np.max(corr + t * g))
    gr = (np.sqrt(5.0) - 1.0) * 0.5
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = float(np.max(corr + c * g))
    fd = float(np.max(corr + d * g))
    for _ in range(50):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = float(np.max(corr + c * g))
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = float(np.max(corr + d * g))
    t_opt = (a + b) * 0.5
    f_opt = float(np.max(corr + t_opt * g))
    return t_opt, f_opt

    grad = compute_all_grads(h)
    dx = 2.0 / n
    all_pairs = [(p, q) for p in range(n) for q in range(p + 1, n)]
    round_count = 0
    stall = 0
    while time.monotonic() < deadline and round_count < max_rounds:
        round_count += 1
        best_new_c = c5
        best_t = 0.0
        best_p = -1
        best_q = -1
        for p, q in all_pairs:
            g = grad[:, p] - grad[:, q]
            if np.all(g == 0.0):
                continue
            lo = max(-h[p], h[q] - 1.0)
            hi = min(1.0 - h[p], h[q])
            if hi - lo < 1e-14:

    sg = grad[active, :].mean(axis=0)
