def project_feasible(y, target):
    """Project y onto {x: 0 <= x_i <= 1, sum(x) = target}.
    Uses the fact that the projection is clip(y - lambda, 0, 1) for the right lambda.
    """
    y = np.asarray(y, dtype=float)
    n = len(y)
    if abs(y.sum() - target) < 1e-14 and y.min() >= -1e-14 and y.max() <= 1 + 1e-14:
        return np.clip(y, 0, 1)
    
    # Bisection on lambda
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    return result

```python
def project_feasible(y, target):
    y = np.asarray(y, dtype=float).copy()
    n = len(y)
    # If already feasible (within tight tolerance), just clip
    s = y.sum()
    if abs(s - target) < 1e-12 and y.min() >= -1e-12 and y.max() <= 1 + 1e-12:
        return np.clip(y, 0.0, 1.0)
    # Bisection on lambda: find lambda such that sum(clip(y - lambda, 0, 1)) = target
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    # Most exact possible:
    # After bisection, result.sum() should be very close to target
    # but might differ by last ulp. Let's do a tiny correction.
    # Actually, for the bisection to be exact, we'd need exact arithmetic.
    # The 100 iterations give about 2^-100 * (hi-lo) precision, which is way below machine epsilon.
    # So the result should satisfy sum to within ~n * eps * scale, which is fine.
    return result

def exact_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def interpolate_to_size(h, new_n):
    n_old = len(h)
    if n_old == new_n:
        return h.copy()
    # For step functions, the i-th sample represents interval [i*dx_old, (i+1)*dx_old)
    # When going to new_n samples, each new sample j represents [j*dx_new, (j+1)*dx_new)
    # The overlap of new sample j with old sample i determines the value.
    # Simple approach: use the center of each new bin and interpolate.
    # Old bin centers: (i + 0.5) / n_old (in [0,1] normalized)
    # New bin centers: (j + 0.5) / new_n (in [0,1] normalized)
    # Map: new_center -> old_index = new_center * n_old
    old_idx_float = (np.arange(new_n) + 0.5) * n_old / new_n
    idx_lo = np.floor(old_idx_float).astype(int)
    idx_hi = idx_lo + 1
    idx_lo = np.clip(idx_lo, 0, n_old - 1)
    idx_hi = np.clip(idx_hi, 0, n_old - 1)
    frac = old_idx_float - np.floor(old_idx_float)
    result = (1.0 - frac) * h[idx_lo] + frac * h[idx_hi]
    return result
```
