    grad = np.zeros(n)
    if s >= 0:
        # First part: (1-h[m+s]) for m in [0, n-s)
        grad[:n-s] += (1 - h[s:])
        # Second part: -h[m-s] for m in [s, n)
        grad[s:] -= h[:n-s]
    else:

        # m-s = m+ss must be in [0, n), so m+ss < n => m < n-ss, m+ss >= 0 (always)

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def project(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_shift(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        nv = n - s
        if nv > 0:
            g[:nv] += 1.0 - h[s:]
            g[s:] -= h[:nv]
    else:
        ss = -s
        nv = n - ss
        if nv > 0:
            g[ss:] += 1.0 - h[:nv]
            g[:nv] -= h[ss:]
    return g

def do_optimize(n, dx, target, h_init, t_limit, seed):
    h = project(h_init.copy(), target)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    tau = 10.0
    step = 0.5
    stall = 0

    while time.time() < t_limit:
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        c5 = np.max(corr)

        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 250:
            tau = min(tau * 2.0, 50000.0)
            step = min(step * 0.3, 0.1)
            stall = 0
            if tau > 50000.0:
                break

        cmax = np.max(corr)
        w = np.exp(tau * (corr - cmax))
        ws = np.sum(w)
        if ws < 1e-300:
            w = np.zeros(len(corr))
            w[np.argmax(corr)] = 1.0
            ws = 1.0
        w /= ws

        g = np.zeros(n)
        active = np.where(w > 1e-7)[0]
        for k in active:
            s = k - (n - 1)
            g += w[k] * grad_shift(h, s, n)
        g *= dx

        h_new = project(h - step * g, target)
        c5_new = compute_c5(h_new, dx)

        if c5_new < c5:
            h = h_new
            step = min(step * 1.04, 2.0)
        else:
            step *= 0.5
            if step < 1e-6:
                break

    return best_h, best_c5

def binary_local_search(h_bin, n, dx, target, t_limit):
    h = h_bin.copy()
    c5 = compute_c5(h, dx)
    improved = True
    while improved and time.time() < t_limit:
        improved = False
        for m in range(n):
            if h[m] > 0.5:
                h[m] = 0.0
            else:
                h[m] = 1.0
            h2 = project(h, target)
            c5_new = compute_c5(h2, dx)
            if c5_new < c5 - 1e-12:
                c5 = c5_new
                h = h2
                improved = True
            else:
                h[m] = 1.0 - h[m] if h[m] < 0.5 else h[m] - 1.0 + 0.0
                h = h_bin.copy() if not improved else h
                if h[m] < 0.5:
                    h[m] = 0.0
                else:
                    h[m] = 1.0
    return h, c5

def make_block_init(n, target, nblocks):
    hb = np.zeros(n)
    seg = n // nblocks
    wid = seg // 2
    for b in range(nblocks):
        start = b * seg
        end = min(start + wid, n)
        hb[start:end] = 1.0
    return project(hb, target)

def make_random_init(n, target, rng):
    hr = (rng.rand(n) < 0.5).astype(float)
    return project(hr, target)

def make_smooth_init(n, target, freq):
    x = np.arange(n, dtype=float) / n
    hs = (np.sin(2.0 * np.pi * freq * x) > 0.0).astype(float)
    return project(hs, target)
