def project_feasible(raw, n):
    target = n / 2.0
    h = np.clip(np.asarray(raw, dtype=float), 0.0, 1.0).copy()

        diff = target - h.sum()
        if abs(diff) < 1e-14:

    lo = 0.0

            lo = max(lo, -h[i] / d[i])

def compute_active_lags(h, tol=1e-10):
    """Return indices of lags achieving (near) the maximum"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = corr.max()
    active = np.where(corr >= mx - tol * max(1.0, mx))[0]
    return active, mx

def subgradient_descent_direction(h, top_k=5):
    """
    Compute a descent direction based on subgradient of max criterion.
    """
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    offset = n - 1
    
    # Get top k lags
    top_idx = np.argsort(corr)[-top_k:]
    
    grad = np.zeros(n)
    for lag_idx in top_idx:
        k = lag_idx - offset
        # Compute gradient of c_k w.r.t. h
        # First part: (1-h[m-k]) for valid m-k
        if k >= 0:
            grad[k:] += (1.0 - h[:n-k])
        else:
            grad[:n+k] += (1.0 - h[-k:])
        # Second part: -h[m+k] for valid m+k
        if k >= 0:
            grad[:n-k] += -h[k:]
        else:
            grad[-k:] += -h[:n+k]
    
    # Descent direction: negative gradient
    d = -grad
    # Make zero-sum
    d -= d.mean()
    # Normalize
    dn = np.linalg.norm(d)
    if dn < 1e-15:
        return None
    d /= dn
    return d

def random_block_direction(rng, n, block_size=None):
    """Generate a random block direction with zero sum."""
    if block_size is None:
        block_size = max(2, n // 10)
    
    d = np.zeros(n)
    # Pick two random blocks
    i = rng.integers(0, n - block_size)
    j = rng.integers(0, n - block_size)
    if j < i:
        i, j = j, i
    d[i:i+block_size] = 1.0
    d[j:j+block_size] = -1.0
    
    # Make zero sum
    dsum = d.sum()
    if abs(dsum) > 1e-15:
        d -= dsum / n
    # Renormalize
    dn = np.abs(d).sum()
    if dn < 1e-15:
        return None
    d /= dn
    return d

    offset = n - 1
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for lag_idx in top_idx:
        k = int(lag_idx - offset)
        if k >= 0:
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] += -h[k:]
        else:
           .;
            grad[:n + k] += (1.0 - h[-k:])
            grad[-k:] += -h[:n + k]
    d = -grad
    d -= d.mean()
    dn = np.linalg.norm(d)
    if dn < 1e-15:
        return None
    return d / dn


def random_block_direction(rng, n):
    bs = max(2, n // 15)
    d = np.zeros(n)
    i = int(rng.integers(0, max(1, n - bs)))
    j = int(rng.integers(0, max(1, n - bs)))
    if j < i:
        i, j = j, i
    if j - i < bs:
        j = min(n - 1, i + bs)
    d[i:i + bs] = 1.0
    if j + bs <= n:
        d[j:j + bs] = -1.0
    else:
        d[j:] = -1.0 * (bs / max(1, n - j))
    dsum = d.sum()
    d -= dsum / n
    dn = np.abs(d).sum()
    if dn < 1e-12:
        return None
    return d / dn


def upsample(h, new_n):
    old_n = len(h)
    old_x = (np.arange(old_n) + 0.5) * 2.0 / old_n
    new_x = (np.arange(new_n) + 0.5) * 2.0 / new_n
    interp = np.interp(new_x, old_x, h)
    return project_feasible(interp, new_n)


def optimize_local(h, val, rng, time_fn, min_time=15.0, max_iters=5000):
    n = len(h)
    best_h = h.copy()
    best_val = val
    iters = 0
    stale = 0
    while iters < max_iters and time_fn() > min_time:
        iters += 1
        mode = iters % 3
        if mode == 0:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
        elif mode == 1:
            d = subgradient_direction(h, top_k=3)
            if d is None:
                d = subgradient_direction(h, top_k=5)
            if d is None:
                continue
        else:
            d = random_block_direction(rng, n)
            if d is None:
                continue
        h_new, v_new = line_search(h, d, n_iter=40)
        if v_new < val - 1e-12:
            h = h_new
            val = v_new
            stale = 0
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            stale += 1
            if stale > 200:
                # Try a larger perturbation
                h_try = h + 0.02 * rng.standard_normal(n)
                h_try = project_feasible(h_try, n)
                v_try = evaluate_c5(h_try)
                if v_try < val:
                    h = h_try
                    val = v_try
                    if val < best_val:
                        best_val = val
                        best_h = h.copy()
                stale = 0
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    def time_left():
        return budget_s - 10.0 - (time.time() - start_time)

    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)

    n0 = len(h0)
    h0 = project_feasible(h0, n0)

    best_h = h0.copy()
    best_val = evaluate_c5(h0)

    # Phase 1: Full pair exchange at base resolution
    h_curr = h0.copy()
    val_curr = evaluate_c5(h_curr)
    n = n0

    max_passes = 80
    for p in range(max_passes):
        if time_left() < 60:
            break
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        perm = rng.permutation(len(pairs))
        improved_this_pass = 0
        for idx in perm:
            if time_left() < 30:
                break
            i, j = pairs[idx]
            d = np.zeros(n)
            d[i] = 1.0
            d[j] = -1.0
            h_new, v_new = line_search(h_curr, d, n_iter=35)
            if v_new < val_curr - 1e-11:
                h_curr = h_new
                val_curr = v_new
                improved_this_pass += 1
                if val_curr < best_val:
                    best_val = val_curr
                    best_h = h_curr.copy()
        if improved_this_pass == 0:
            break

    # Phase 2: Subgradient + random directions at base resolution
    if time_left() > 60:
        h_curr, val_curr = optimize_local(
            h_curr, val_curr, rng, time_left, min_time=40, max_iters=3000)
        if val_curr < best_val:
            best_val = val_curr
            best_h = h_curr.copy()

    # Phase 3: Try complement
    h_comp = 1.0 - best_h
    h_comp = project_feasible(h_comp, len(best_h))
    v_comp = evaluate_c5(h_comp)
    if v_comp < best_val:
        best_val = v_comp
        best_h = h_comp.copy()

    # Phase 4: Upsample and optimize
    if time_left() > 100:
        n2 = 2 * n0
        h2 = upsample(best_h, n2)
        v2 = evaluate_c5(h2)
        if v2 < best_val:
            best_val = v2
            best_h = h2.copy()

        h2_curr = h2.copy()
        val2_curr = v2 if v2 <= best_val else best_val
        # Re-optimize from best upsampled
        h2_curr = upsample(best_h, n2)
        val2_curr = evaluate_c5(h2_curr)

        # Full-ish pair scan at higher res
        for p in range(15):
            if time_left() < 60:
                break
            num_pairs = 2000
            for _ in range(num_pairs):
                if time_left() < 30:
                    break
                i = int(rng.integers(0, n2))
                j = int(rng.integers(0, n2))
                if i == j:
                    continue
                d = np.zeros(n2)
                d[i] = 1.0
                d[j] = -1.0
                h_new, v_new = line_search(h2_curr, d, n_iter=30)
                if v_new < val2_curr - 1e-11:
                    h2_curr = h_new
                    val2_curr = v_new
                    if val2_curr < best_val:
                        best_val = val2_curr
                        best_h = h2_curr.copy()

        # Subgradient at higher res
        if time_left() > 60:
            h2_curr, val2_curr = optimize_local(
                h2_curr, val2_curr, rng, time_left, min_time=30, max_iters=2000)
            if val2_curr < best_val:
                best_val = val2_curr
                best_h = h2_curr.copy()

    # Phase 5: Try n=120 or n=3*n0 if time allows
    if time_left() > 80:
        n3 = 3 * n0
        h3 = upsample(best_h, n3)
        v3 = evaluate_c5(h3)
        if v3 < best_val:
            best_val = v3
            best_h = h3.copy()
        if time_left() > 40:
            h3_curr, val3_curr = optimize_local(
                h3, v3, rng, time_left, min_time=20, max_iters=1500)
            if val3_curr < best_val:
                best_val = val3_curr
                best_h = h3_curr.copy()

    # Phase 6: Random restarts from best
    if time_left() > 40:
        for _ in range(30):
            if time_left() < 15:
                break
            sigma = 0.03
            h_try = best_h + sigma * rng.standard_normal(len(best_h))
            h_try = project_feasible(h_try, len(best_h))
            v_try = evaluate_c5(h_try)
            if v_try < best_val:
                # Quick local polish
                h_try, v_try = optimize_local(
                    h_try, v_try, rng, time_left, min_time=10, max_iters=200)
                if v_try < best_val:
                    best_val = v_try
                    best_h = h_try.copy()

    # Phase 7: Final polish
    if time_left() > 10:
        n_f = len(best_h)
        for _ in range(200):
            if time_left() < 5:
                break
            i = int(rng.integers(0, n_f))
            j = int(rng.integers(0, n_f))
            if i == j:
                continue
            d = np.zeros(n_f)
            d[i] = 1.0
            d[j] = -1.0
            h_new, v_new = line_search(best_h, d, n_iter=30)
            if v_new < best_val - 1e-12:
                best_h = h_new
                best_val = v_new

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, len(best_h))
    final_val = evaluate_c5(best_h)
    if final_val > best_val:
        best_val = final_val

    return (best_h, best_val, len(best_h))
