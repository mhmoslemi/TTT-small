def project_box_sum(y, target):
    """Project y onto {h: 0<=h_i<=1, sum(h)=target}"""
    n = len(y)
    # Bisection on lambda
    lo = -1e10
    hi = 1e10
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - hi, 0, 1)
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            # Need to increase some indices
            candidates = np.where(h < 1)[0]
            if len(candidates) > 0:
                h[candidates[0]] += min(diff, 1 - h[candidates[0]])
        else:
            candidates = np.where(h > 0)[0]
            if len(candidates) > 0:

import numpy as np

from scipy.optimize import linprog

def project_box_sum(y, target):
    """Project y onto {h: 0<=h_i<=1, sum(h_i)=target}"""
    n = len(y)
    # Bisection on threshold
    lo = y.min() - 1.5
    hi = y.max() + 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        h = np.clip(y - mid, 0.0, 1.0)
        if h.sum() >= target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - hi, 0.0, 1.0)
    # Fix residual (should be tiny)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                add = min(diff, 1.0 - h[idx[0]])
                h[idx[0]] += add
                diff -= add
                while diff > 1e-12 and len(idx) > 0:
                    idx = np.where(h < 1.0)[0]
                    if len(idx) == 0:
                        break
                    add = min(diff, 1.0 - h[idx[0]])
                    h[idx[0]] += add
                    diff -= add
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                sub = min(-diff, h[idx[0]])
                h[idx[0]] -= sub
                diff += sub
                while diff < -1e-12 and len(idx) > 0:
                    idx = np.where(h > 0.0)[0]
                    if len(idx) == 0:
                        break
                    sub = min(-diff, h[idx[0]])
                    h[idx[0]] -= sub
                    diff += sub
    return h

def get_correlations(h):
    """Compute np.correlate(h, 1-h, mode='full')"""
    return np.correlate(h, 1.0 - h, mode='full')

def get_c5(h):
    """Compute C5 = max(corr) * dx where dx = 2/n"""
    n = len(h)
    corr = get_correlations(h)
    return float(np.max(corr) * 2.0 / n)

def gradient_smoothed_max(h, corr, beta):
    """Gradient of (1/beta) * log(sum exp(beta * C_r))"""
    n = len(h)
    num_lags = len(corr)
    # Compute softmax weights
    c_max = corr.max()
    x = beta * (corr - c_max)
    x = np.clip(x, -700.0, 0.0)
    e = np.exp(x)
    Z = e.sum()
    if Z < 1e-300:
        # All mass on the max
        idx_max = np.argmax(corr)
        weights = np.zeros(num_lags)
        weights[idx_max] = 1.0
    else:
        weights = e / Z
    
    grad = np.zeros(n)
    lag_offsets = np.arange(num_lags) - (n - 1)
    
    for k in range(num_lags):
        w = weights[k]
        if w < 1e-14:
            continue
        r = lag_offsets[k]
        if r == 0:
            grad += w * (1.0 - 2.0 * h)
        elif r > 0:
            m = n - r
            grad[:m] += w * (1.0 - h[:m])
            grad[r:] -= w * h[:m]
        else:
            s = -r
            grad[s:] += w * (1.0 - h[s:])
            grad[:n-s] -= w * h[s:]
    
    return grad

def resample_to_n(h_old, n_new):
    """Resample h_old (of length n_old) to n_new points using nearest neighbor"""
    n_old = len(h_old)
    indices = (np.arange(n_new) * n_old / n_new).astype(int)
    indices = np.clip(indices, 0, n_old - 1)
    return h_old[indices]

def resample_linear(h_old, n_new):
    """Resample using linear interpolation"""
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, h_old)

def optimize_smoothed(h, n, budget_s, seed, initial_beta=5.0, beta_max=50.0):
    """Optimize using smoothed max gradient descent with projection"""
    target = n / 2.0
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    h = project_box_sum(h.copy(), target)
    best_h = h.copy()
    best_c5 = get_c5(h)
    
    beta = initial_beta
    step_size = 0.3
    no_improve_count = 0
    max_no_improve = 200
    
    corr = get_correlations(h)
    
    while time.time() - start_time < budget_s:
        # Compute gradient
        grad = gradient_smoothed_max(h, corr, beta)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        
        # Normalized gradient direction
        direction = -grad / grad_norm
        
        # Line search
        alpha = step_size
        found = False
        for _ in range(15):
            h_trial = project_box_sum(h + alpha * direction, target)
            c5_trial = get_c5(h_trial)
            if c5_trial < best_c5 - 1e-10:
                h = h_trial
                best_h = h_trial.copy()
                best_c5 = c5_trial
                step_size = min(alpha * 1.5, 2.0)
                no_improve_count = 0
                found = True
                corr = get_correlations(h)
                break
            alpha *= 0.5
        
        if not found:
            # Try smaller beta (smoother)
            beta = max(beta * 0.5, 1.0)
            no_improve_count += 1
            if no_improve_count > max_no_improve:
                # Random perturbation
                h = project_box_sum(h + 0.1 * rng.randn(n), target)
                corr = get_correlations(h)
                beta = initial_beta
                no_improve_count = 0
                step_size = 0.3
        
        # Slowly increase beta
        beta = min(beta * 1.005, beta_max)
    
    return best_h, best_c5

def optimize_lp(h, n, budget_s, trust_radius=0.1):
    """LP-based local optimization"""
    target = n / 2.0
    start_time = time.time()
    
    h = project_box_sum(h.copy(), target)
    best_h = h.copy()
    best_c5 = get_c5(h)
    
    while time.time() - start_time < budget_s:
        corr = get_correlations(h)
        t_curr = corr.max()
        active_tol = 0.05 * t_curr + 0.5 / n
        active = np.where(corr >= t_curr - active_tol)[0]
        
        # Build LP
        num_lags = len(corr)
        lags = np.arange(num_lags) - (n - 1)
        
        # Compute gradients for active lags
        grad_rows = []
        rhs = []
        for k in active:
            r = lags[k]
            g = np.zeros(n)
            if r == 0:
                g = 1.0 - 2.0 * h
            elif r > 0:
                m = n - r
                g[:m] = 1.0 - h[:m]
                g[r:] -= h[:m]
            else:
                s = -r
                g[s:] = 1.0 - h[s:]
                g[:n-s] -= h[s:]
            grad_rows.append(g)
            rhs.append(float(np.dot(g, h) - corr[k]))
        
        K = len(grad_rows)
        if K == 0:
            break
        
        # LP: minimize t
        # Variables: h_0, ..., h_{n-1}, t
        # Constraints:
        #   g_k · h - t <= rhs_k for each active k
        #   0 <= h_i <= 1
        #   sum h_i = target
        #   |h_i - h_bar_i| <= trust_radius (via bounds)
        
        A_ub = np.zeros((K, n + 1))
        for i in range(K):
            A_ub[i, :n] = grad_rows[i]
            A_ub[i, n] = -1.0
        b_ub = rhs
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds = []
        for i in range(n):
            lo = max(0.0, h[i] - trust_radius)
            hi = min(1.0, h[i] + trust_radius)
            bounds.append((lo, hi))
        bounds.append((t_curr - 1.0, t_curr + 1.0))
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs')
        
        if result.success:
            h_new = result.x[:n]
            h_new = project_box_sum(h_new, target)
            c5_new = get_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new.copy()
                best_c5 = c5_new
                h = h_new.copy()
                # Reduce trust radius
                trust_radius *= 0.7
            else:
                trust_radius *= 0.5
        else:
            trust_radius *= 0.5
        
        if trust_radius < 1e-6:
            break
    
    return best_h, best_c5

    n = len(h)
    num_lags = len(corr)
    c_max = corr.max()
    active = np.where(corr >= c_max - tolerance)[0]
    
    grad = np.zeros(n)
    lags = np.arange(num_lags) - (n - 1)
    
    for k in active:
        r = lags[k]
        g = np.zeros(n)
        if r == 0:
            g = 1.0 - 2.0 * h
        elif r > 0:
            m = n - r
            g[:m] = 1.0 - h[:m]
            g[r:] -= h[:m]
        else:
            s = -r
            g[s:] = 1.0 - h[s:]
            g[:n-s] -= h[s:]
        grad += g
    
    grad /= len(active)
    return grad
