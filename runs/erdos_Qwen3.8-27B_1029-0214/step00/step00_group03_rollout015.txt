import numpy as np
import time
import math

def exact_eval(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * float(np.max(corr))

def project_feasible(v, n):
    v = np.asarray(v, dtype=np.float64).copy()
    target = float(n) / 2.0
    v = np.clip(v, 0.0, 1.0)
    diff = target - float(np.sum(v))
    if abs(diff) < 1e-12:
        return v
    if diff > 0:
        slack = 1.0 - v
        total = float(np.sum(slack))
        if total < 1e-15:
            return v
        v = v + (diff / total) * slack
    else:
        total = float(np.sum(v))
        if total < 1e-15:
            return v
        v = v + (diff / total) * v
    v = np.clip(v, 0.0, 1.0)
    residual = target - float(np.sum(v))
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = int(np.argmax(1.0 - v))
        else:
            idx = int(np.argmin(v))
        v[idx] = v[idx] + residual
        v = np.clip(v, 0.0, 1.0)
    return v

def compute_coef(h, i, j, n):
    """Compute the coefficient vector for a pair move (i, j)."""
    n_corr = 2 * n - 1
    coef = np.zeros(n_corr)
    for k in range(n_corr):
        s = k - (n - 1)
        c_val = 0.0
        q1 = i - s
        if 0 <= q1 <= n - 1:
            c_val += (1.0 - h[q1])
        q2 = i + s
        if 0 <= q2 <= n - 1:
            c_val -= h[q2]
        q3 = j + s
        if 0 <= q3 <= n - 1:
            c_val += h[q3]
        q4 = j - s
        if 0 <= q4 <= n - 1:
            c_val -= (1.0 - h[q4])
        coef[k] = c_val
    return coef

def find_best_d(corr, coef, lo, hi, dx):
    """Find d in [lo, hi] that minimizes dx * max(corr + d * coef)."""
    n_samples = 80
    best_d = 0.0
    best_val = dx * float(np.max(corr))
    
    for m in range(n_samples):
        d = lo + (hi - lo) * (m + 0.5) / n_samples
        val = dx * float(np.max(corr + d * coef))
        if val < best_val:
            best_val = val
            best_d = d
    
    # Refine with golden section
    if best_d > lo and best_d < hi:
        # Find the interval around best_d
        idx = int((best_d - lo) / (hi - lo) * n_samples)
        lo_g = lo + (hi - lo) * max(0, idx - 1) / n_samples
        hi_g = lo + (hi - lo) * min(n_samples, idx + 1) / n_samples
        
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        a_g = lo_g
        b_g = hi_g
        c_g = b_g - gr * (b_g - a_g)
        d_g = a_g + gr * (b_g - a_g)
        fc = dx * float(np.max(corr + c_g * coef))
        fd = dx * float(np.max(corr + d_g * coef))
        
        for _ in range(30):
            if fc < fd:
                b_g = d_g
                d_g = c_g
                fd = fc
                c_g = b_g - gr * (b_g - a_g)
                fc = dx * float(np.max(corr + c_g * coef))
            else:
                a_g = c_g
                c_g = d_g
                fc = fd
                d_g = a_g + gr * (b_g - a_g)
                fd = dx * float(np.max(corr + d_g * coef))
        
        mid = (a_g + b_g) / 2.0
        val_mid = dx * float(np.max(corr + mid * coef))
        if val_mid < best_val:
            best_val = val_mid
            best_d = mid
    
    return best_d, best_val

            for j in range(i + 1, n):
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0
