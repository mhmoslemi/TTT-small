def make_feasible(h, n):
    """Project h to satisfy sum(h) = n/2 and 0 <= h <= 1."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

    # Use bisection on a shift

        return np.sum(np.clip(h + lam, 0.0, 1.0)) - target

def projected_gradient_step(h, n, beta, lr):
    # Compute gradient of log-sum-exp objective
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    vals = np.minimum(beta * (corr - m), 0.0)
    weights = np.exp(vals)
    weights = weights / np.sum(weights)
    
    # Gradient of each corr[k] wrt h[j]:
    # corr[k] = sum_i h[i] * (1-h[i-offset]) * dx where offset = k-(n-1)
    # d(corr[k])/d(h[j]) = dx * [(1-h[j-offset])*1_{0<=j-offset<n} - h[j+offset]*1_{0<=j+offset<n}]
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        offset = k - (n - 1)
        w = weights[k] * dx
        if w < 1e-15:
            continue
        # First term: (1-h[j-offset]) for j where 0 <= j-offset < n
        # j ranges from max(0, offset) to min(n-1, n-1+offset)
        j_start = max(0, offset) if offset > 0 else 0
        j_end = min(n - 1, n - 1 + offset) if offset < 0 else n - 1
        # Actually: 0 <= j - offset < n means offset <= j < n + offset
        # If offset >= 0: j ranges from offset to n-1 (since j < n and j >= offset)
        # If offset < 0: j ranges from 0 to n-1+offset (since j >= 0 and j < n+offset)
        
        if offset >= 0:
            # First term: for j = offset, offset+1, ..., n-1
            for j in range(offset, n):
                grad[j] += w * (1.0 - h[j - offset])
            # Second term: for j = 0, 1, ..., n-1-offset (where j+offset < n)
            for j in range(0, n - offset):
                grad[j] -= w * h[j + offset]
        else:
            r = -offset  # r > 0
            # First term: for j = 0, 1, ..., n-1-r (where j-offset = j+r < n)
            for j in range(0, n - r):
                grad[j] += w * (1.0 - h[j + r])
            # Second term: for j = r, r+1, ..., n-1 (where j+offset = j-r >= 0)
            for j in range(r, n):
                grad[j] -= w * h[j - r]
    
    # Project gradient onto the constraint manifold
    # The constraint is sum(h) = n/2, so we subtract the mean of the gradient
    grad_proj = grad - np.mean(grad)
    
    # Take a step
    h_new = h - lr * grad_proj
    
    # Project onto [0, 1]^n with sum constraint
    h_new = make_feasible(h_new, n)
    
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

_OPT_N = 45
_OPT_BETA = 100.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    lo = -1.0
    hi = 1.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    resid = target - np.sum(h)
    if resid > 1e-13:
        room = 1.0 - h
        idx = int(np.argmax(room))
        h[idx] = min(1.0, h[idx] + resid)
    elif resid < -1e-13:
        idx = int(np.argmax(h))
        h[idx] = max(0.0, h[idx] + resid)
    return h

def lse_value(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    vals = np.minimum(beta * (corr - m), 0.0)
    return m + np.log(np.sum(np.exp(vals))) / beta

def lse_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    vals = np.minimum(beta * (corr - m), 0.0)
    w = np.exp(vals)
    sw = np.sum(w)
    if sw < 1e-300:
        return np.zeros(n)
    w = w / sw
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        offset = k - (n - 1)
        wk = w[k] * dx
        if wk < 1e-18:
            continue
        if offset >= 0:
            grad[offset:] += wk * (1.0 - h[:n - offset])
            grad[:n - offset] -= wk * h[offset:]
        else:
            r = -offset
            grad[:n - r] += wk * (1.0 - h[r:])
            grad[r:] -= wk * h[:n - r]
    return grad

def _obj(h):
    return lse_value(h, _OPT_N, _OPT_BETA)

def _con_fun(h):
    return np.sum(h) - _OPT_N / 2.0

def _con_jac(h):
    return np.ones(_OPT_N)

def _build_bounds(n):
    return [(0.0, 1.0)] * n

def _slsqp(h_init, n, beta, maxiter, ftol):
    global _OPT_N, _OPT_BETA
    _OPT_N = n
    _OPT_BETA = float(beta)
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    bnds = _build_bounds(n)
    try:
        res = minimize(_obj, h_init, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return make_feasible(res.x, n)
    except Exception:
        pass
    return h_init

def _pg_step(h, n, beta, lr):
    g = lse_grad(h, n, beta)
    # Project gradient onto sum=0 subspace
    g = g - np.mean(g)
    h_new = h - lr * g
    return make_feasible(h_new, n)

def _pg_polish(h, n, beta, max_iter, base_lr):
    h_cur = h.copy()
    best_val = lse_value(h_cur, n, beta)
    for i in range(max_iter):

        h_try = _pg_step(h_cur, n, beta, lr)

        if val_try < best_val - 1e-16:
            h_cur = h_try
            best_val = val_try
        else:
            break
    return h_cur
