    """
    Solve: min t
    s.t. g_j . d <= t for j = 1..m
         sum(d) = 0
         -delta <= d[i] <= delta
    
    Variables: [d_0, ..., d_{n-1}, t]
    """
    m = g_matrix.shape[0]
    # c = [0, 0, ..., 0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # A_ub: g_j . d - t <= 0 for each j
    A_ub = np.hstack([g_matrix, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    # A_eq: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: -delta <= d[i] <= delta, t unbounded below (use -inf)
    bounds = [(-delta, delta)] * n + [(-np.inf, np.inf)]
    
    try:

                      bounds=bounds, method='highs')

            # Ensure sum is exactly 0
            d -= np.mean(d)
            return d

        pass
    return None

    max_stall = 60
    tol_active = 0.002
    max_active = 35

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        cmax = float(np.max(corr))

        thresh = cmax - tol_active * abs(cmax)
        if thresh < 0:
            thresh = 0.0
        active_idx = np.where(corr >= thresh)[0]
        if len(active_idx) > max_active:
            order = np.argsort(corr[active_idx])[::-1][:max_active]
            active_idx = active_idx[order]
        active_lags = (active_idx - (n - 1)).tolist()

        if len(active_lags) == 0:
            active_lags = [0]

        g_mat = np.array([grad_lag(cur_h, lag) for lag in active_lags])

        delta = 0.4 / np.sqrt(n)
        d, t_val = solve_lp_direction(g_mat, n, delta)

        if d is None or np.max(np.abs(d)) < 1e-15:
            avg_g = np.mean(g_mat, axis=0)
            d = -avg_g
            d -= d.mean()
            norm = np.max(np.abs(d))
            if norm < 1e-15:
                stall += 1
                if stall > max_stall:
                    break
                if stall % 5 == 0:
                    rng = np.random.RandomState(1000 + it)
                    pert = rng.standard_normal(n) * 0.02 / np.sqrt(n)
                    cur_h = project_feasible(cur_h + pert, target)
                continue
            d = d * (delta / norm)

        improved = False
        alphas = [1.0, 0.7, 0.4, 0.2, 0.1, 0.05, 0.02]
        for alpha in alphas:
            h_try = cur_h + alpha * d
            h_try = project_feasible(h_try, target)
            c5_try = evaluate_c5(h_try)
            if c5_try < best_c5:
                cur_h = h_try
                best_h = h_try.copy()
                best_c5 = c5_try
                improved = True
                break

        if not improved:
            for alpha in [0.3, 0.15, 0.07]:
                h_try = cur_h - alpha * d
                h_try = project_feasible(h_try, target)
                c5_try = evaluate_c5(h_try)
                if c5_try < best_c5:
                    cur_h = h_try
                    best_h = h_try.copy()
                    best_c5 = c5_try
                    improved = True
                    break

        if not improved:
            stall += 1
            if stall > max_stall:
                break
            if stall % 8 == 0:
                rng = np.random.RandomState(2000 + stall)
                amp = 0.03 / np.sqrt(n) * (1.0 + 0.1 * (stall // 8))
                pert = rng.standard_normal(n) * amp
                h_try = project_feasible(cur_h + pert, target)
                c5_try = evaluate_c5(h_try)
                if c5_try < best_c5:
                    cur_h = h_try
                    best_h = h_try.copy()
                    best_c5 = c5_try
                    stall = 0
                elif c5_try < best_c5 + 0.005:
                    cur_h = h_try
        else:
            stall = 0
            tol_active = max(0.0005, tol_active * 0.95)

    return best_h, best_c5


def try_symmetry(h, deadline):
    n = len(h)
    target = n / 2.0
    candidates = [h.copy()]
    candidates.append(h[::-1].copy())
    candidates.append(0.5 * (h + h[::-1]))
    candidates.append(1.0 - h)
    candidates.append(1.0 - h[::-1])
    candidates.append(0.5 * (h + 1.0 - h[::-1]))

    best = None
    best_c5 = float('inf')
    for c in candidates:
        c = project_feasible(c, target)
        if np.min(c) < -1e-10 or np.max(c) > 1.0 + 1e-10:
            continue
        c5 = evaluate_c5(c)
        if c5 < best_c5:
            best_c5 = c5
            best = c.copy()
    return best, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 25.0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = None
    if h0 is None or len(h0) == 0:
        n_init = 64
        h0 = np.ones(n_init) * 0.5

    n0 = len(h0)
    target0 = n0 / 2.0
    h0 = project_feasible(h0, target0)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    # Phase 1: Local optimization at original resolution
    phase_deadline = min(deadline, t0 + budget_s * 0.45)
    if time.time() < phase_deadline:
        h_opt, c5_opt = local_optimize(h0, phase_deadline)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            n0 = len(best_h)
            target0 = n0 / 2.0

    # Phase 2: Try symmetry variants
    if time.time() < deadline - 30:
        h_sym, c5_sym = try_symmetry(best_h, deadline)
        if h_sym is not None and c5_sym < best_c5:
            best_h = h_sym
            best_c5 = c5_sym
        sym_deadline = min(deadline, t0 + budget_s * 0.6)
        if time.time() < sym_deadline:
            h_opt, c5_opt = local_optimize(best_h, sym_deadline)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt

    # Phase 3: Upsample by 2 and optimize
    if time.time() < deadline - 40:
        h_up = np.repeat(best_h, 2)
        n_up = len(h_up)
        h_up = project_feasible(h_up, n_up / 2.0)
        c5_up = evaluate_c5(h_up)
        if c5_up < best_c5:
            best_h = h_up
            best_c5 = c5_up
        if time.time() < deadline:
            h_opt, c5_opt = local_optimize(h_up, deadline)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt

    # Phase 4: If still time, try upsampled by 4 from original
    if time.time() < deadline - 20:
        h0_q = np.repeat(h0, 4)
        n_q = len(h0_q)
        h0_q = project_feasible(h0_q, n_q / 2.0)
        c5_q = evaluate_c5(h0_q)
        if c5_q < best_c5:
            best_h = h0_q
            best_c5 = c5_q
        if time.time() < deadline:
            h_opt, c5_opt = local_optimize(h0_q, deadline)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt

    # Final verification and return
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h = project_feasible(best_h, target_final)
    final_c5 = evaluate_c5(best_h)

    return (best_h, final_c5, n_final)
