import numpy as np

import time

def compute_c5(h, dx):
    """Compute exact C5 value."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_raw(h):
    """Compute the raw max correlation (without dx scaling)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)), corr

def project_box_sum(h, target):
    """Project h onto {x : 0<=x[i]<=1, sum(x)=target}."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    
    # If sum is already correct, done
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    
    # Bisection for lambda
    # f(lambda) = sum(clamp(h[i] - lambda, 0, 1))
    # f is decreasing in lambda
    # We want f(lambda) = target
    
    lo = -1.0
    hi = 2.0
    # Ensure bracketing
    while np.sum(np.clip(h - lo, 0, 1)) < target:
        lo -= 1.0
    while np.sum(np.clip(h - hi, 0, 1)) > target:
        hi += 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    
    # Fix any residual
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        if diff > 0:
            # Need to add more; find indices where result < 1
            idx = np.where(result < 1.0 - 1e-10)[0]
        else:
            # Need to subtract; find indices where result > 0
            idx = np.where(result > 1e-10)[0]
        if len(idx) > 0:
            per = diff / len(idx)
            result[idx] += per
            result = np.clip(result, 0.0, 1.0)
    
    return result

        # Subgradient of max

def project_box_sum(h, target):

    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:

        else:
            hi = mid

def lag_grad(h, r):
    n = len(h)
    g = np.zeros(n)
    if r == 0:
        return 1.0 - 2.0 * h
    elif r > 0:
        m = n - r
        g[:m] += (1.0 - h[r:])
        g[r:] -= h[:m]
    else:
        s = -r
        m = n - s
        g[s:] += (1.0 - h[:m])
        g[:m] -= h[s:]
    return g

def smooth_max_grad(h, beta):
