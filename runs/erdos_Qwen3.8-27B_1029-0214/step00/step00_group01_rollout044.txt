import numpy as np
from scipy.optimize import minimize
import time

def evaluate_C5(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_box_simplex(z, target):
    n = len(z)
    z = np.array(z, dtype=float).copy()
    # Handle non-finite
    z[~np.isfinite(z)] = 0.5
    z = np.clip(z, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Find mu such that sum(clamp(z_i - mu, 0, 1)) = target
    lo = min(z) - 1.0
    hi = max(z) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    mu = 0.5 * (lo + hi)
    x = np.clip(z - mu, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(x)
    if abs(residual) > 1e-14:
        # Distribute residual
        interior = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if len(interior) > 0:
            idx = interior[0]
            adj = np.clip(residual, -(x[idx]), 1.0 - x[idx])
            x[idx] += adj
            residual -= adj
            for idx2 in interior[1:]:
                if abs(residual) < 1e-15:
                    break
                adj = np.clip(residual, -(x[idx2]), 1.0 - x[idx2])
                x[idx2] += adj
                residual -= adj
    return x

def corr_values_h(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def corr_values_q(h):
    # Same as corr_values_h but makes explicit
    n = len(h)
    q = 1.0 - h
    return np.correlate(h, q, mode="full")

def gradient_corr_pos(h, s):
    """Gradient of dx*corr(s) for s > 0. corr(s) = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])"""
    n = len(h)
    r = s
    grad = np.zeros(n)
    # h[j] appears as h[i+r] when i = j-r (valid: r <= j <= n-1): contrib (1-h[j-r])
    # h[j] appears as h[i] when i = j (valid: 0 <= j <= n-1-r): contrib -h[j+r]
    for j in range(n):
            g = 0.0
            if r <= j <= n - 1:
                g += (1.0 - h[j - r])
            if 0 <= j <= n - 1 - r:
                g += (-h[j + r])
            grad[j] = g
    return grad

def gradient_corr_neg(h, s):
    """Gradient of dx*corr(s) for s < 0. Let r=-s. corr(-r) = sum_{i=0}^{n-1-r} h[i]*(1-h[i+r])"""
    n = len(h)
    r = -s
    grad = np.zeros(n)
    # h[j] appears as h[i] when i=j (valid: 0 <= j <= n-1-r): contrib (1-h[j+r])
    # h[j] appears as h[i+r] when i=j-r (valid: r <= j <= n-1): contrib -h[j-r]
    for j in range(n):
        g = 0.0
        if 0 <= j <= n - 1 - r:
            g += (1.0 - h[j + r])
        if r <= j <= n - 1:
            g += (-h[j - r])
        grad[j] = g
    return grad

def gradient_corr_zero(h):
    """Gradient of corr(0) = sum h[i]*(1-h[i])"""
    n = len(h)
    return 1.0 - 2.0 * h

def gradient_corr(h, s):
    """Gradient of unscaled corr(s) with respect to h."""
    if s > 0:
        return gradient_corr_pos(h, s)
    elif s < 0:
        return gradient_corr_neg(h, s)
    else:
        return gradient_corr_zero(h)

def vec_corrs(h, s_list, dx):
    """Evaluate selected scaled correlation values."""
    c = corr_values_h(h)
    n = len(h)
    vals = np.zeros(len(s_list))
    for idx, s in enumerate(s_list):
        k = s + (n - 1)
        if 0 <= k < len(c):
            vals[idx] = c[k] * dx
    return vals

def vec_corr_grads(h, s_list, dx):
    """Evaluate gradients of selected scaled correlations. Returns array shape (len(s_list), n)."""
    m = len(s_list)
    n = len(h)
    grads = np.zeros((m, n))
    for idx, s in enumerate(s_list):
        if s > 0:
            grads[idx] = dx * gradient_corr_pos(h, s)
        elif s < 0:
            grads[idx] = dx * gradient_corr_neg(h, s)
        else:
            grads[idx] = dx * gradient_corr_zero(h)
    return grads

def select_active_rows(c_vals, dx, n, c_max, tol_rel, bundle_limit):
    """Select which correlation rows to use as constraints."""
    m = len(c_vals)
    if m <= bundle_limit:
        return list(range(m))
    threshold = c_max * dx - tol_rel * max(c_max * dx, 0.05)
    active = [k for k in range(m) if c_vals[k] * dx >= threshold]
    # Also include top K by value
    top_k = min(bundle_limit, m)
    top_idx = np.argsort(c_vals)[::-1][:top_k]
    combined = set(active) | set(int(x) for x in top_idx)
    # Always include zero lag (k = n-1)
    combined.add(n - 1)
    # Limit total
    if len(combined) > bundle_limit:
        order = sorted(combined, key=lambda k: -c_vals[k])
        combined = set(order[:bundle_limit])
    return sorted(combined)

def epi_fun(x, s_list, dx, n):
    return x[-1]

def epi_grad(x, s_list, dx, n):
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def epi_fun_eq(x, s_list, dx, n):
    return np.sum(x[:n]) - n / 2.0

def epi_jac_eq(x, s_list, dx, n):
    g = np.ones(n + 1)
    g[-1] = 0.0
    return g

def epi_fun_ineq(x, s_list, dx, n):
    h = x[:n]
    t = x[-1]
    c = corr_values_h(h)
    vals = np.zeros(len(s_list))
    for idx, k in enumerate(s_list):
        vals[idx] = t - c[k] * dx
    return vals

def epi_jac_ineq(x, s_list, dx, n):
    h = x[:n]
    m = len(s_list)
    jac = np.zeros((m, n + 1))
    for idx, k in enumerate(s_list):
        s = k - (n - 1)
        if s > 0:
            g = dx * gradient_corr_pos(h, s)
        elif s < 0:
            g = dx * gradient_corr_neg(h, s)
        else:
            g = dx * gradient_corr_zero(h)
        jac[idx, :n] = -g
        jac[idx, n] = 1.0
    return jac

def solve_epigraph(h_start, n, c_start, s_list, dx, t_init, maxiter, ftol):
    """Solve the epigraph NLP."""
    x0 = np.zeros(n + 1)
    x0[:n] = h_start
    x0[-1] = t_init
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    cons = [
        {
            'type': 'eq',
            'fun': epi_fun_eq,
            'jac': epi_jac_eq
        },
        {
            'type': 'ineq',
            'fun': epi_fun_ineq,
            'jac': epi_jac_ineq
        }
    ]
    
    args = (s_list, dx, n)
    
    try:
        res = minimize(
            epi_fun, x0, method='SLSQP', jac=epi_grad,
            bounds=bounds, constraints=cons, args=args,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        return res
    except Exception:
        return None

def try_candidate(h_cand, n, best_c, best_h):
    """Try a candidate, return (accepted, c, h) if valid and improving."""
    h_cand = np.asarray(h_cand, dtype=float).flatten()
    if len(h_cand) != n or not np.all(np.isfinite(h_cand)):
        return False, None, None
    # Check bounds
    if np.any(h_cand < -1e-10) or np.any(h_cand > 1.0 + 1e-10):
        h_cand = np.clip(h_cand, 0.0, 1.0)
    # Check sum
    target = n / 2.0
    if abs(np.sum(h_cand) - target) > 1e-8 * n:
        h_cand = project_box_simplex(h_cand, target)
    else:
        # Tiny adjustment if needed
        diff = target - np.sum(h_cand)
        if abs(diff) > 1e-14:
            interior = np.where((h_cand > 1e-10) & (h_cand < 1.0 - 1e-10))[0]
            if len(interior) > 0:
                h_cand[interior[0]] += diff
    c = evaluate_C5(h_cand)
    if not np.isfinite(c):
        return False, None, None
    if c < best_c - 1e-12:
        return True, c, h_cand
    return False, c, h_cand

def transfer_to_resolution(h_src, n_src, n_dst):
    """Transfer a solution from one resolution to another."""
    if n_src == n_dst:
        return h_src.copy()
    # Use interpolation
    x_src = (np.arange(n_src) + 0.5) / n_src * 2.0
    x_dst = (np.arange(n_dst) + 0.5) / n_dst * 2.0
    h_dst = np.interp(x_dst, x_src, h_src)
