def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    """Project h onto {x: 0 <= x_i <= 1, sum(x) = target}"""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)  # initial clip (not exact, just for bracketing)
    
    # Bisection on lambda
    lo = -1.0  # h - lo >= h >= 0, so clip gives h
    hi = 1.0   # h - hi <= h - 1 <= 0 (for h <= 1), so clip gives 0
    
    # Actually, we need sum(clip(h - lambda, 0, 1)) = target
    # As lambda -> -inf, clip(h - lambda, 0, 1) -> 1 for all, sum -> n
    # As lambda -> +inf, clip(h - lambda, 0, 1) -> 0 for all, sum -> 0
    # So we want lambda such that sum = target
    
    lo = min(h) - 1.0  # ensures h - lo > 1 for all, so clip = 1, sum = n
    hi = max(h) + 1.0  # ensures h - hi < 0 for all, so clip = 0, sum = 0
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    
    lambda_star = (lo + hi) / 2.0
    result = np.clip(h - lambda_star, 0.0, 1.0)
    
    # Final correction for numerical precision
    diff = target - result.sum()
    if abs(diff) > 1e-12:

def solve_direction_lp(grads, n, delta):
    """Solve LP to find descent direction.
    Minimize t subject to:
      grads[i] @ d <= t for all i
      sum(d) = 0
      -delta <= d_i <= delta
    """
    m = len(grads)
    # Variables: d_0, ..., d_{n-1}, t  (total n+1 variables)
    # Objective: minimize t -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active constraint i: grads[i] @ d - t <= 0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -delta <= d_i <= delta, t unbounded below (but >= -large)
    bounds = [(-delta, delta)] * n + [(-1e6, 1e6)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            # Zero out mean (should already be ~0)
            d -= d.mean()
            return d
        else:
            return None
    except Exception:
        return None

import time

def evaluate_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    n = len(h)
    h = np.asarray(h, dtype=np.float64)
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-13:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def compute_active_grads(h, n_top=30):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    L = len(corr)
    # Get top n_top correlations
    top_indices = np.argsort(corr)[::-1][:min(n_top, L)]
    max_corr = corr[top_indices[0]]
    
    grads = np.zeros((len(top_indices), n))
    for idx, j in enumerate(top_indices):
        k = j - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            for i in range(n - k):
                grad[i+k] += (1.0 - h[i])
                grad[i] -= h[i+k]
        else:
            m = -k
            for i in range(n - m):
                grad[i] += (1.0 - h[i+m])
                grad[i+m] -= h[i]
        grads[idx] = grad
    return grads, top_indices, max_corr

def solve_lp_direction(grads, n, delta):
    m = len(grads)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-delta, delta)] * n + [(-100.0, 100.0)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'time_limit': 5})
        if res.success and res.fun < -1e-10:
            d = res.x[:n].copy()
            d -= d.mean()
            return d
    except Exception:
        pass
    return None

def do_softmax_optimize(h, budget_s, max_iters=500, temp=0.01):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_c5 = evaluate_c5(h_cur)
    best_h = h_cur.copy()

    step = 0.1 / np.sqrt(n)

        w = np.exp(shifted)

        grad = np.zeros(n)

            else:

                    grad[i] += w[j] * (1.0 - h_cur[i+m])
