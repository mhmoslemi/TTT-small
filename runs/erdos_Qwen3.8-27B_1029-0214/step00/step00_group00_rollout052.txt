2. Evaluating f at each breakpoint and at 0 and 1.
3. Taking the minimum.

- Sort the lines by slope (d_k).
- The upper envelope is a convex polytope. The minimum of the upper envelope is at the point where the subgradient contains 0, i.e., where the slope changes from negative to positive.

    S = n / 2.0
    for _ in range(30):
        diff = S - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:

        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent_1d(h, i, n, corr):
    """Find optimal value of h[i] given all other coordinates fixed."""
    current = h[i]
    # Compute derivatives d_k for each output k
    # d_k = (1-h[m1]) if m1 valid + (-h[m2]) if m2 valid
    # where m1 = i - k + (n-1), m2 = i + k + (n-1)
    # Wait, m2 = i + k - (n-1)
    
    lines_a = []  # slopes d_k
    lines_b = []  # intercepts corr[k] - d_k * current (so that f(x) = max(lines_a * x + lines_b))
    
    for k in range(2 * n - 1):
        m1 = i - k + (n - 1)
        m2 = i + k - (n - 1)
        d = 0.0
        if 0 <= m1 <= n - 1:
            d += (1.0 - h[m1])
        if 0 <= m2 <= n - 1:
            d += (-h[m2])
        if abs(d) > 1e-15:
            # Line: corr[k] + d*(x - current) = d*x + (corr[k] - d*current)
            lines_a.append(d)
            lines_b.append(corr[k] - d * current)
    
    if not lines_a:
        return current
    
    lines_a = np.array(lines_a)
    lines_b = np.array(lines_b)
    
    # f(x) = max_j (lines_a[j] * x + lines_b[j])
    # Find min over x in [0,1]
    
    # Candidate points: 0, 1, and all intersection points within [0,1]
    candidates = [0.0, 1.0]
    
    # Find intersection of each pair
    for j in range(len(lines_a)):
        for l in range(j+1, len(lines_a)):
            if abs(lines_a[j] - lines_a[l]) < 1e-15:
                continue
            x_int = (lines_b[l] - lines_b[j]) / (lines_a[j] - lines_a[l])
            if 0.0 <= x_int <= 1.0:
                candidates.append(x_int)
    
    # Also add the current value as a candidate
    candidates.append(current)
    
    best_x = current
    best_val = np.inf
    
    for x in candidates:
        x = np.clip(x, 0.0, 1.0)
        val = np.max(lines_a * x + lines_b)
        if val < best_val:
            best_val = val
            best_x = x
    
    return best_x
