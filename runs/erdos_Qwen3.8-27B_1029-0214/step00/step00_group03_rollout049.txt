import numpy as np
import time
from scipy.optimize import minimize_scalar

def compute_G(h):
    return np.correlate(h, 1.0 - h, mode="full")

def c5_from_G(G):
    N = (len(G) + 1) // 2
    dx = 2.0 / N
    return np.max(G) * dx

def project_feasible(h):
    N = len(h)
    target = N / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            slack = 1.0 - h
            total = slack.sum()
            if total < 1e-14:
                break
            h += diff * slack / total
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += diff * h / total
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_D1D2(h, i, j):
    N = len(h)
    k_vals = np.arange(-(N - 1), N)
    i_mk = i - k_vals
    j_mk = j - k_vals
    i_pk = i + k_vals
    j_pk = j + k_vals
    D1 = np.zeros(2 * N - 1)
    m1 = (i_mk >= 0) & (i_mk < N)
    D1[m1] += (1.0 - h[i_mk[m1]])
    m2 = (j_mk >= 0) & (j_mk < N)
    D1[m2] -= (1.0 - h[j_mk[m2]])
    m3 = (i_pk >= 0) & (i_pk < N)
    D1[m3] -= h[i_pk[m3]]
    m4 = (j_pk >= 0) & (j_pk < N)
    D1[m4] += h[j_pk[m4]]
    D2 = np.zeros(2 * N - 1)
    D2[N - 1] = -2.0
    if i != j:
        D2[N - 1 + (i - j)] = 1.0
        D2[N - 1 + (j - i)] = 1.0
    return D1, D2

def sweep_pair_swaps(h, G0, deadline, rng):
    N = len(h)
    best_c5 = c5_from_G(G0)
    best_h = h.copy()
    best_G = G0.copy()
    no_improve_count = 0
    while time.time() < deadline:
        improved = False
        pairs = [(i, j) for i in range(N) for j in range(i + 1, N)]
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > deadline:
                return best_h, best_c5, best_G
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_hi - d_lo < 1e-12:
                continue
            D1, D2 = compute_D1D2(h, i, j)
            cur_max = np.max(G0)
            def obj(d):
                Gc = G0 + d * D1 + d * d * D2
                return np.max(Gc)
            res = minimize_scalar(obj, bounds=(d_lo, d_hi), method='bounded',
                                  options={'xatol': 1e-9, 'maxiter': 200})
            if res.fun < cur_max - 1e-9:
                h[i] += res.x
                h[j] -= res.x
                np.clip(h, 0.0, 1.0, out=h)
                h = project_feasible(h)
                G0 = compute_G(h)
                c5 = c5_from_G(G0)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_G = G0.copy()
                improved = True
        if not improved:
            no_improve_count += 1
            if no_improve_count >= 2:
                break
        else:
            no_improve_count = 0
    return best_h, best_c5, best_G

def perturb_h(h, rng, scale=0.05):
    N = len(h)
    h_new = h.copy()
    h_new += rng.uniform(-scale, scale, size=N)
    h_new = project_feasible(h_new)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()
    deadline_total = start + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(45) * 0.5

    N = len(h)
    h = project_feasible(h)
    G0 = compute_G(h)

    best_h = h.copy()
    best_c5 = c5_from_G(G0)
    best_N = N

    # Phase 1: Pair swap from initial
    deadline1 = min(start + budget_s * 0.35, deadline_total)
    h, c5, G0 = sweep_pair_swaps(h, G0, deadline1, rng)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_N = N

    # Phase 2: Perturbation + re-optimize (multiple restarts)
    n_restarts = 0
    while time.time() < start + budget_s * 0.75:
        h_try = perturb_h(best_h, rng, scale=0.03 + 0.02 * n_restarts)
        G_try = compute_G(h_try)
        deadline_r = min(time.time() + 30, start + budget_s * 0.75)
        h_try, c5_try, _ = sweep_pair_swaps(h_try, G_try, deadline_r, rng)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_N = N
        n_restarts += 1

    # Phase 3: Try other N values
    for N_try in [60, 90, 45, 75, 120]:
        if time.time() > deadline_total:
            break
        x_orig = np.linspace(0, 2, best_N)
        x_new = np.linspace(0, 2, N_try)
        if N_try == best_N:
            h_try = best_h.copy()
        else:
            h_try = np.interp(x_new, x_orig, best_h)
            h_try = project_feasible(h_try)
        G_try = compute_G(h_try)
        deadline_n = min(time.time() + 60, deadline_total)
        h_try, c5_try, _ = sweep_pair_swaps(h_try, G_try, deadline_n, rng)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_N = N_try

    # Phase 4: Final fine-tuning on best
    h_final = best_h.copy()
    G_final = compute_G(h_final)
    h_final, c5_final, _ = sweep_pair_swaps(h_final, G_final, deadline_total, rng)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()
        best_N = len(h_final)

    # Final check
    best_h = project_feasible(best_h)
    G_check = compute_G(best_h)
    c5_check = c5_from_G(G_check)
    if c5_check <= best_c5 + 1e-8:
        best_c5 = c5_check

    return best_h, best_c5, len(best_h)
