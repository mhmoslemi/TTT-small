def pairwise_descent(h, max_iter=100, timeout=None):
    N = len(h)
    dx = 2.0 / N
    q = 1.0 - h
    c = np.correlate(h, q, mode='full')  # unscaled
    current_max = np.max(c) * dx
    
    for iteration in range(max_iter):
        improved = False
        for i in range(N):
            for j in range(N):
                if i == j or h[i] <= h[j]:
                    continue
                # Try to move mass from i to j
                # h[i] += d, h[j] -= d
                lo = max(-h[i], h[j] - 1)  # d > lo
                hi = min(1 - h[i], h[j])   # d < hi
                if hi - lo < 1e-10:
                    continue
                # Find optimal d in (lo, hi)
                # C[k] + ΔC[k] for all k
                # ΔC[k] = d * D_k + Q_k * d^2
                # where D_k and Q_k are precomputed
                
                # For simplicity, do a few trial values of d
                # and pick the best one
                best_d = 0
                best_max = current_max
                # Try d values
                for d_trial in [0.1*hi, 0.5*hi, 0.9*hi, hi*0.3, hi*0.7]:

    term1 = np.where((idx1 >= 0) & (idx1 < N), q[m1], 0.0)
    idx2 = i + k
    m2 = np.clip(idx2, 0, N-1)
    term2 = np.where((idx2 >= 0) & (idx2 < N), h[m2], 0.0)
    idx3 = j - k
    m3 = np.clip(idx3, 0, N-1)
    term3 = np.where((idx3 >= 0) & (idx3 < N), q[m3], 0.0)
    idx4 = j + k
    m4 = np.clip(idx4, 0, N-1)
    term4 = np.where((idx4 >= 0) & (idx4 < N), h[m4], 0.0)
    D = term1 - term2 - term3 + term4
    Q = np.zeros(2*N - 1)
    Q[N-1] = -2.0
    if i != j:
        Q[N-1 + (i - j)] += 1.0
        Q[N-1 + (j - i)] += 1.0
    return D, Q
