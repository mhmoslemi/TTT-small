    lo = h.min() - 1.0
    hi = h.max() + 1.0

        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:

        else:
            hi = mid

def project_feasible(h, n):

    h = np.array(h, dtype=np.float64, copy=True)

    h = np.clip(h, 0.0, 1.0)
    
    lo = h.min() - 2.0
    hi = h.max() + 2.0

        if s > target:
            lo = mid
        else:
            hi = mid

    residual = target - result.sum()
    if abs(residual) > 1e-13:
