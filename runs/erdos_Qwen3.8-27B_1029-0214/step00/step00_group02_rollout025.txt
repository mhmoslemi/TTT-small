    corr = np.correlate(h, 1-h, mode='full') * dx
    # log-sum-exp for numerical stability
    max_corr = np.max(corr)
    log_sum_exp = max_corr + np.log(np.sum(np.exp(epsilon * (corr - max_corr))))
    F = log_sum_exp / epsilon
    # Sobolev-like weights
    p = np.exp(epsilon * (corr - max_corr)) / np.sum(np.exp(epsilon * (corr - max_corr)))
    # p has length 2n-1, indexed by correlation output index k = s + (n-1)

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using bisection."""
    lo = -np.max(h) - 0.5

def project_feasible(h, target_sum):
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_grad(h, epsilon):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = epsilon * (corr - max_c)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    p = exp_v / sum_exp
    F = (max_c + np.log(sum_exp)) / epsilon
    W = p
    W_flip = W[::-1]
    first = np.convolve(1.0 - h, W, mode='full')[n-1:2*n-1]
    second = np.convolve(h, W_flip, mode='full')[n-1:2*n-1]
    grad = dx * (first - second)
    return F, grad

def optimize_pg(h0, max_time, seed=0):
    n = len(h0)
    target_sum = n / 2.0
    start = time.time()
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    epsilon = 40.0
    lr = 0.03
    patience = 0
    best_F = lse_obj_grad(h, epsilon)[0]
    F_count = 0
    while time.time() - start < max_time:
        F, grad = lse_obj_grad(h, epsilon)
        F_count += 1
        if F_count % 30 == 0 and F < best_F:
            best_F = F
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            patience = 0
        else:
            patience += 1
            if patience >= 40:
                lr *= 0.75
                patience = 0
                if lr < 1e-9:
                    break
        if F_count % 50 == 0:
            epsilon = min(epsilon * 1.1, 800.0)
    return best_h, best_c5

def optimize_local_search(h0, max_time, seed=0):
    n = len(h0)
    start = time.time()
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    step_sizes = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for step in step_sizes:
        stagnation = 0
        while time.time() - start < max_time and stagnation < 300:
            i, j = rng.choice(n, 2, replace=False)
            deltas = [step, -step, 0.5*step, -0.5*step, 0.25*step, -0.25*step]
            rng.shuffle(deltas)
            found = False
            for delta in deltas:
                hi_new = h[i] - delta
                hj_new = h[j] + delta
                if hi_new < 0.0 or hi_new > 1.0 or hj_new < 0.0 or hj_new > 1.0:
                    continue
                h[i] = hi_new
                h[j] = hj_new
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    stagnation = 0
                    found = True
                    break
                else:
                    h[i] = hi_new + delta
                    h[j] = hj_new - delta
            if not found:
                stagnation += 1
    return best_h, best_c5

def make_start(n, mode, seed):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    rng = np.random.RandomState(seed)
    if mode == 0:
        h = np.full(n, 0.5)
    elif mode == 1:
        h = 0.5 + 0.3 * np.cos(np.pi * x)
    elif mode == 2:
        h = 0.5 - 0.3 * np.cos(np.pi * x)
    elif mode == 3:
        h = 0.5 + 0.25 * np.sin(2.0 * np.pi * x)
    elif mode == 4:
        h = 0.5 + 0.25 * np.cos(2.0 * np.pi * x)
    elif mode == 5:
        c1 = rng.uniform(0.1, 0.3)
        c2 = rng.uniform(0.05, 0.15)
        ph = rng.uniform(0, 2*np.pi)
        h = 0.5 + c1 * np.sin(2.0*np.pi*x + ph) + c2 * np.cos(4.0*np.pi*x)
    elif mode == 6:
        h = 0.5 + 0.25 * np.cos(2.0*np.pi*x) + 0.15 * np.cos(4.0*np.pi*x + 0.5)
    elif mode == 7:
        h = 0.5 + 0.2 * np.cos(3.0*np.pi*x) + 0.1 * np.sin(5.0*np.pi*x)
    else:
        h = 0.5 + rng.randn(n) * 0.08
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n / 2.0)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = make_start(47, 0, seed)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init)
    best_n = n_init

    def rem():
        return budget_s - (time.time() - t0)

    if rem() > 120:
        alloc = min(300, rem() * 0.3)
        h_opt, c5_opt = optimize_pg(h_init, alloc, seed)
        if c5_opt < best_c5:
            best_c5, best_h, best_n = c5_opt, h_opt.copy(), n_init

    if rem() > 60:
        alloc = min(150, rem() * 0.2)
        h_opt, c5_opt = optimize_local_search(best_h, alloc, seed + 1)
        if c5_opt < best_c5:
            best_c5, best_h, best_n = c5_opt, h_opt.copy(), best_n

    modes_to_try = [1, 2, 3, 4, 5, 6, 7]
    for m in modes_to_try:
        if rem() < 80:
            break
        h_start = make_start(n_init, m, seed + m * 7)
        alloc = min(100, rem() * 0.15)
        h_opt, c5_opt = optimize_pg(h_start, alloc, seed + m)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_init
        if rem() < 60:
            break
        alloc_ls = min(50, rem() * 0.1)
        h_opt2, c5_opt2 = optimize_local_search(best_h, alloc_ls, seed + m * 3)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2.copy()
            best_n = best_n

    for n_try in [93, 187]:
        if rem() < 150:
            break
        h_up = upsample(best_h, n_try)
        alloc = min(200, rem() * 0.3)
        h_opt, c5_opt = optimize_pg(h_up, alloc, seed + 100)
        if rem() > 40:
            alloc_ls = min(80, rem() * 0.2)
            h_opt2, c5_opt2 = optimize_local_search(h_opt, alloc_ls, seed + 200)
            if c5_opt2 < c5_opt:
                h_opt, c5_opt = h_opt2, c5_opt2
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_try

    if rem() > 30:
        alloc = min(80, rem() * 0.4)
        h_opt, c5_opt = optimize_local_search(best_h, alloc, seed + 500)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = best_n

    return (best_h, best_c5, best_n)
