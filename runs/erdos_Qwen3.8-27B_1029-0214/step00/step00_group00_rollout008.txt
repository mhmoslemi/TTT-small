def project_feasible(y, n):
    target = n / 2.0
    h = np.clip(np.array(y, dtype=float).copy(), 0.0, 1.0)
    for _ in range(100):

        if abs(diff) < 1e-14:

    dx = 2.0 / n
    grad = np.zeros(n)
    if r >= 0:
        # C_+(r) = dx * sum_{i=0}^{n-1-r} h[i+r]*(1-h[i])

        else:

            grad[r:] += (1.0 - h[:n-r])
            grad[:n-r] -= h[r:]
    else:
        s = -r
        # C_-(s) = dx * sum_{j=0}^{n-1-s} h[j]*(1-h[j+s])
        # d/dh[m] = (1-h[m+s]) for m in [0, n-1-s] minus h[m-s] for m in [s, n-1]
        grad[:n-s] += (1.0 - h[s:])
        grad[s:] -= h[:n-s]
    return dx * grad

    max_val = corr.max()

        else:
            h += residual / n
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_lag(h, r, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    if r >= 0:
        if r == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[r:] += (1.0 - h[:n-r])
            grad[:n-r] -= h[r:]
    else:
        s = -r
        grad[:n-s] += (1.0 - h[s:])
        grad[s:] -= h[:n-s]
    return dx * grad

def slsqp_solve(h0, n, maxiter=300):
    dx = 2.0 / n
    target = float(n) / 2.0
    
    def obj(x):
        return x[n]
    
    def obj_g(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_f(x):
        h = x[:n]
        return x[n] - np.correlate(h, 1.0 - h, mode="full") * dx
    
    def ineq_j(x):
        h = x[:n]
        nc = 2 * n - 1
        J = np.zeros((nc, n + 1))
        for k in range(nc):
            r = k - (n - 1)
            J[k, :n] = -grad_lag(h, r, n)
            J[k, n] = 1.0
        return J
    
    def eq_f(x):
        return np.array([x[:n].sum() - target])
    
    def eq_j(x):
        J = np.zeros((1, n + 1))
        J[0, :n] = 1.0
        return J
    
    t0 = evaluate_c5(h0) + 0.05
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_g, bounds=bounds,
                       constraints=[{'type': 'ineq', 'fun': ineq_f, 'jac': ineq_j},
                                    {'type': 'eq', 'fun': eq_f, 'jac': eq_j}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project_feasible(res.x[:n], n)
        return h_new
    except:
        return h0

def subgradient_optimize(h0, n, iters=5000, lr0=0.01, top_k=5):
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(h0)
    for it in range(iters):
        corr = corr_values(h)
        max_val = corr.max()
        if max_val < best_c - 1e-12:
            best_c = max_val
            best_h = h.copy()
        
        # Get top-k lags
        sorted_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for idx in sorted_idx:
            r = int(idx) - (n - 1)
            grad += grad_lag(h, r, n)
        
        # Adaptive learning rate
        lr = lr0 / (1.0 + it * 0.01)
        h_new = project_feasible(h - lr * grad, n)
        if not np.all(np.isfinite(h_new)):
            break
        h = h_new
    
    return best_h, best_c

def binary_search(h0, n, iters=10000):
    """Try to find a near-binary solution"""
    # Start from h0, push towards binary
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(h0)
    for it in range(iters):
        corr = corr_values(h)
        max_val = corr.max()
        if max_val < best_c - 1e-12:
            best_c = max_val
            best_h = h.copy()
        
        # Gradient of top active constraints
        sorted_idx = np.argsort(corr)[-5:]
        grad = np.zeros(n)
        for idx in sorted_idx:
            r = int(idx) - (n - 1)
            grad += grad_lag(h, r, n)
        
        # Binary pressure
        binary_pressure = 10.0 * h * (1.0 - h)  # pushes towards 0 or 1
        h_new = project_feasible(h - 0.005 * grad - 0.01 * binary_pressure, n)
        h = h_new
    
    return best_h, best_c

def make_variants(h, n):
    """Generate variants of h"""
    variants = [h.copy()]
    variants.append(h[::-1].copy())
    variants.append(1.0 - h)
    variants.append((1.0 - h)[::-1].copy())
    variants.append(0.5 * (h + h[::-1]))
    return variants

def refine_grid(h_coarse, n_fine):
    """Transfer a coarse solution to a finer grid via interpolation"""
    n_coarse = len(h_coarse)
    # Map indices
    x_coarse = np.linspace(0, 1, n_coarse, endpoint=False) + 0.5/n_coarse
    x_fine = np.linspace(0, 1, n_fine, endpoint=False) + 0.5/n_fine
    # Actually, for step functions, we should map by interval
    # Coarse interval i covers [i/n_coarse, (i+1)/n_coarse)
    # Fine interval j covers [j/n_fine, (j+1)/n_fine)
    # For each fine interval, find which coarse interval it corresponds to
    centers_fine = (np.arange(n_fine) + 0.5) / n_fine
    coarse_idx = np.minimum((centers_fine * n_coarse).astype(int), n_coarse - 1)
    h_fine = h_coarse[coarse_idx].copy()
    return project_feasible(h_fine, n_fine)
