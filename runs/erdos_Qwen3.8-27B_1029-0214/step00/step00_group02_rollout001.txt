```python
import time

def evaluate_c5(h):
    """Compute the exact C5 value for a given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def project(h, n):
    """Project h onto the feasible set {0 <= h <= 1, sum(h) = n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Now fix the sum using water-filling
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Distribute the difference
    if diff > 0:
        # Need to increase sum; add to entries that are < 1
        capacity = 1.0 - h
        total_cap = capacity.sum()
        if total_cap < 1e-15:
            return h  # can't increase further
        h += diff / total_cap * capacity
    else:
        # Need to decrease sum; subtract from entries that are > 0
        capacity = h
        total_cap = capacity.sum()
        if total_cap < 1e-15:
            return h
        h += diff / total_cap * capacity  # diff is negative
    # Clip again and repeat if needed
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if diff > 0:
        # Add to the smallest entries
        idx = np.argsort(h)[:max(1, int(abs(diff)))]
        for i in idx:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    elif diff < 0:
        idx = np.argsort(-h)[:max(1, int(abs(-diff)))]
        for i in idx:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def smooth_objective(h, tau):
    """Compute the smooth approximation of C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Log-sum-exp for stability
    max_c = np.max(corr)
    log_sum = max_c + np.log(np.sum(np.exp(tau * (corr - max_c))))
    return (1.0 / tau) * log_sum
```

```python
import time

def evaluate_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def project_feasible(y, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Find the entry closest to the boundary where we can adjust
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, tau):
    """Compute smooth objective and numerical gradient."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    # Log-sum-exp
    shifted = tau * (corr - max_c)
    log_sum = np.log(np.sum(np.exp(shifted)))
    f = (max_c * tau + log_sum) / tau
    
    # Numerical gradient (central differences)
    grad = np.zeros(n)
    delta = 1e-7
    for j in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[j] += delta
        h_minus[j] -= delta
        # Compute objective at h_plus
        cp = np.correlate(h_plus, 1.0 - h_plus, mode='full') * dx
        max_cp = np.max(cp)
        fp = (max_cp * tau + np.log(np.sum(np.exp(tau * (cp - max_cp))))) / tau
        # Compute objective at h_minus
        cm = np.correlate(h_minus, 1.0 - h_minus, mode='full') * dx
        max_cm = np.max(cm)
        fm = (max_cm * tau + np.log(np.exp(tau * (cm - max_cm)).sum())) / tau
        grad[j] = (fp - fm) / (2.0 * delta)
    
    return f, grad
```
