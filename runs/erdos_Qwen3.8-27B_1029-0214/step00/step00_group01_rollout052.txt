                res = minimize(surrogate_value, y.copy(),
                               jac=surrogate_grad,
                               method='L-BFGS-B',
                               options={'maxiter': iter_list[bi],
                                        'gtol': 1e-6, 'iprint': 0})
                if hasattr(res, 'x') and res.x is not None:
                    y = res.x
            except Exception:
                pass
            p = y_to_h(y)
            cv = exact_C5(p)
            if np.isfinite(cv) and cv < _best_c5:
                _best_c5 = cv
                _best_h = p.copy()
        return y

    bl = [30, 100, 300, 800, 2000]
    itl = [80, 60, 50, 40, 30]
    opt_one_resolution(_best_h, bl, itl)

    if time_left() > 150:
        n_high = n_parent * 2
        x_old = (np.arange(n_parent) + 0.5) * (2.0 / n_parent)
        x_new = (np.arange(n_high) + 0.5) * (2.0 / n_high)
        h_high = np.interp(x_new, x_old, _best_h)
        h_high = np.clip(h_high, 0.0, 1.0)
        h_high = waterfill(h_high, n_high / 2.0)
        ch = exact_C5(h_high)
        if np.isfinite(ch) and ch < _best_c5:
            _best_c5 = ch
            _best_h = h_high.copy()
        bl_h = [30, 100, 300, 800]
        itl_h = [60, 50, 40, 30]
        opt_one_resolution(_best_h, bl_h, itl_h)

    if time_left() > 80:
        n_best = len(_best_h)
        for attempt in range(8):
            if time_left() < 30.0:
                break
            scale = 0.15
            h_pert = _best_h + scale * rng.randn(n_best)
            h_pert = waterfill(h_pert, n_best / 2.0)
            opt_one_resolution(h_pert, [200, 600], [40, 30])

    if time_left() > 40:
        h_rev = 1.0 - _best_h[::-1].copy()
        h_rev = waterfill(h_rev, len(h_rev) / 2.0)
        cr = exact_C5(h_rev)
        if cr < _best_c5:
            _best_c5 = cr
            _best_h = h_rev.copy()

    final_h = np.asarray(_best_h, dtype=np.float64).copy()
    final_h = np.clip(final_h, 0.0, 1.0)
    final_h = waterfill(final_h, len(final_h) / 2.0)
    final_c = exact_C5(final_h)

    if not np.isfinite(final_c):
        final_h = np.zeros(64)
        final_h[0::2] = 1.0
        final_h = waterfill(final_h, 32.0)
        final_c = exact_C5(final_h)

    return (final_h, final_c, len(final_h))
