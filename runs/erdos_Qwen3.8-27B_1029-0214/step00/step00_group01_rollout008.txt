def project_to_feasible(h, target):
    """Project h onto {x: 0 <= x_i <= 1, sum(x) = target} using bisection on lambda."""
    def sum_clipped(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0))
    
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if sum_clipped(mid) > target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def evaluate_c5(h):
    """Compute C5 = max_corr * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def lse_gradient(h, lam_factor=20.0):
    """
    Compute gradient of (1/lambda)*log(sum exp(lambda * dx * corr_r))
    which approximates dx * max(corr_r) = C5.
    
    Returns the gradient vector and the current C5 value.
    """
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Softmax weights (shift for numerical stability)
    scaled = lam_factor * dx * corr
    scaled -= np.max(scaled)
    weights = np.exp(scaled)
    weights /= np.sum(weights)
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        w = weights[r]
        if w < 1e-10:
            continue
        s = r - (n - 1)
        if s >= 0:
            g = np.zeros(n)
            g[s:] += (1.0 - h[:n-s])
            g[:n-s] -= h[s:]
        else:
            r_abs = -s
            g = np.zeros(n)
            g[:n-r_abs] += (1.0 - h[r_abs:])
            g[r_abs:] -= h[:n-r_abs]
        grad += w * g
    
    c5 = float(dx * np.max(corr))
    return grad, c5

def solve_lp_direction(h, top_k=10, delta=0.1):

    Solve LP to find descent direction.
    min t s.t. grad_r . d <= t for top-k correlations
              sum(d) = 0, -delta <= d <= delta

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

    # Variables: d_0, ..., d_{n-1}, t (total n+1 variables)
    # Constraints:
    #   For each r in top_idx: sum_m grad_r[m] * d[m] - t <= 0
    #   sum_m d[m] = 0 (as two inequalities)
    #   -delta <= d[m] <= delta (box, handled by bounds)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    b_eq = np.zeros(1)
    
    A_ub = np.zeros((len(top_idx), n + 1))
    for i, r in enumerate(top_idx):
        s = r - (n - 1)
        if s >= 0:
            g = np.zeros(n)
            g[s:] += (1.0 - h[:n-s])
            g[:n-s] -= h[s:]
        else:
            r_abs = -s
            g = np.zeros(n)
            g[:n-r_abs] += (1.0 - h[r_abs:])
            g[r_abs:] -= h[:n-r_abs]
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0  # grad . d - t <= 0
    
    b_ub = np.zeros(len(top_idx))
    
    bounds = [(None, None)] * n + [(None, None)]
    # Actually, we want -delta <= d[m] <= delta, and t is unbounded
    bounds = [(-delta, delta)] * n + [(None, None)]
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    try:

                      bounds=bounds, method='highs')

            d = d - np.mean(d)
            if np.max(np.abs(d)) > 1e-12:

    except:
        pass
    return None

def coordinate_step(h, i, j, target, n):
    """
    For fixed i and j, find optimal delta that minimizes C5
    when h[i] += delta and h[j] -= delta.
    Returns (delta, new_c5) or (0, current_c5) if no improvement.
    """
    h_curr = h
    corr_curr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
    c5_curr = float(2.0 / n * np.max(corr_curr))
    
    # Compute coefficients for each lag
    # For lag s >= 0: z_s = sum_{k=0}^{n-1-s} h[k+s]*(1-h[k])
    # coeff of h[m] in z_s:
    #   from h[k+s] part: (1-h[k]) where k = m-s, valid if s <= m <= n-1
    #   from (1-h[k]) part: -h[k+s] where k = m, valid if 0 <= m <= n-1-s
    
    # For the update h[i] += d, h[j] -= d:
    # delta_z_s = coeff_i_s * d - coeff_j_s * d = (coeff_i_s - coeff_j_s) * d
    # new_z_s = z_s + (coeff_i_s - coeff_j_s) * d
    
    # Compute coeff_i_s and coeff_j_s for all s
    # This is just the gradient of z_s w.r.t. h[i] and h[j]
    
    ci = np.zeros(2*n-1)
    cj = np.zeros(2*n-1)
    
    for r in range(2*n-1):
        s = r - (n-1)
        if s >= 0:
            # coeff of h[m] in z_s = [s<=m<=n-1]*(1-h[m-s]) - [0<=m<=n-1-s]*h[m+s]
            ci[r] = 0.0
            cj[r] = 0.0
            if s <= i <= n-1:
                ci[r] += (1.0 - h[i-s])
            if 0 <= i <= n-1-s:
                ci[r] -= h[i+s]
            if s <= j <= n-1:
                cj[r] += (1.0 - h[j-s])
            if 0 <= j <= n-1-s:
                cj[r] -= h[j+s]
        else:
            r_abs = -s
            # coeff of h[m] in z_{-r} = [0<=m<=n-1-r]*(1-h[m+r]) - [r<=m<=n-1]*h[m-r]
            ci[r] = 0.0
            cj[r] = 0.0
            if 0 <= i <= n-1-r_abs:
                ci[r] += (1.0 - h[i+r_abs])
            if r_abs <= i <= n-1:
                ci[r] -= h[i-r_abs]
            if 0 <= j <= n-1-r_abs:
                cj[r] += (1.0 - h[j+r_abs])
            if r_abs <= j <= n-1:
                cj[r] -= h[j-r_abs]
    
    m = ci - cj  # coefficient of delta in each z_s
    c = corr_curr  # current z_s values (these are the constants)
    
    # New z_s(d) = c + m * d
    # Minimize max_s (c_s + m_s * d) over d in [d_min, d_max]
    d_min = max(-h[i], -(1-h[j]))  # h[i]+d >= 0, h[j]-d <= 1
    d_max = min(1-h[i], h[j])      # h[i]+d <= 1, h[j]-d >= 0
    
    if d_min >= d_max:
        return 0.0, c5_curr
    
    # Use bisection on the derivative
    # f(d) = max_s (c_s + m_s * d)
    # f'(d) = m_{argmax} (subgradient)
    # Minimum where 0 is in the subgradient
    
    def f(d):
        return np.max(c + m * d)
    
    # Finding the minimum of a convex piecewise-linear function
    # by finding where the slope changes from negative to positive
    lo, hi = d_min, d_max
    
    # Binary search: find d where the "active" slope changes sign
    for _ in range(50):
        mid = (lo + hi) / 2.0
        # Compute subgradient at mid
        vals = c + m * mid
        max_val = np.max(vals)
        active = (vals >= max_val - 1e-12)
        slope = np.sum(m[active])  # subgradient is average of active slopes... 
        # Actually, for min max, the subgradient at the optimum contains 0
        # If all active slopes are > 0, we need to decrease d
        # If all active slopes are < 0, we need to increase d
        # If there are both positive and negative, we're at the optimum
        
        if np.all(m[active] > 0):
            hi = mid  # need to decrease
        elif np.all(m[active] < 0):
            lo = mid  # need to increase
        else:
            break  # at optimum
    
    d_opt = (lo + hi) / 2.0
    d_opt = max(d_min, min(d_max, d_opt))
    
    # Evaluate
    new_c5 = float(2.0 / n * np.max(c + m * d_opt))
    return d_opt, new_c5

Wait, the subgradient of f(d) = max_s (c_s + m_s * d) at a point d is the convex hull of {m_s : s is active}. So the minimum is where 0 belongs to this convex hull, i.e., where there exist active s with m_s ≤ 0 and m_s ≥ 0.

My binary search logic is correct:
- If all active slopes > 0: the function is increasing, so we want to go left (decrease d)
- If all active slopes < 0: the function is decreasing, so we want to go right (increase d)
- If mixed: we're at the minimum

This should work! And it's much more efficient than the LP approach because it's solving a 1D problem.

def project_to_feasible(h, target):
    """Project h onto {x: 0 <= x_i <= 1, sum(x) = target}"""
    if abs(np.sum(h) - target) < 1e-10:
        return np.clip(h, 0.0, 1.0)

        return np.sum(np.clip(h - lam, 0.0, 1.0))
