        g[:n-s] += -h[s:]
    return g

def smooth_grad(h, c_vals, beta, active_idx):
    n = len(h)
    dx = 2.0 / n
    ac = c_vals[active_idx]
    sh = beta * ac - np.max(beta * ac)
    w = np.exp(sh)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(active_idx)) / len(active_idx)
    else:
        w /= ws
    grad = np.zeros(n)
    for i, idx in enumerate(active_idx):
        if w[i] > 1e-10:
            grad += w[i] * lag_grad(h, idx - (n-1))
    return dx * grad

def opt_smooth(h, iters, tlimit, t0):
    n = len(h)
    dx = 2.0 / n
    ts = n / 2.0
    hc = h.copy()
    bh = hc.copy()
    bo = eval_c5(hc)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    stall = 0
    for it in range(iters):
        if time.monotonic() - t0 > tlimit:
            break
        cv = np.correlate(hc, 1.0 - hc, mode="full")
        tau = cv.max() * dx
        thr = tau - 0.08 * max(tau, 0.01)
        am = cv * dx >= thr
        if am.sum() < 5:
            am = np.zeros(len(cv), dtype=bool)
            am[np.argsort(cv)[-10:]] = True
        ai = np.where(am)[0]
        prog = it / max(1, iters-1)
        beta = 15.0 + 185.0 * prog
        gr = smooth_grad(hc, cv, beta, ai)
        gr -= gr.mean()
        t += 1
        m = b1*m + (1-b1)*gr
        v = b2*v + (1-b2)*gr**2
        mh = m/(1-b1**t)
        vh = v/(1-b2**t)
        step = 0.4 * mh / (np.sqrt(vh) + eps)
        best_tr = hc
        best_to = bo
        for f in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            tr = project_box_sum(hc - f*step, ts)
            to = eval_c5(tr)
            if to < best_to - 1e-12:
                best_to = to
                best_tr = tr
                break
        if best_to < bo - 1e-12:
            hc = best_tr
            bo = best_to
            bh = hc.copy()
            stall = 0
        else:
            hc = best_tr  # Still move
            stall += 1
            if stall > 15:
                break
    return bh, bo

def opt_lp(h, iters, tlimit, t0):
    n = len(h)
    dx = 2.0 / n
    ts = n / 2.0
    hc = h.copy()
    bh = hc.copy()
    bo = eval_c5(hc)
    trust = 0.25
    kc = max(30, min(80, n))
    for it in range(iters):
        if time.monotonic() - t0 > tlimit:
            break
        cv = np.correlate(hc, 1.0 - hc, mode="full")
        tau = cv.max() * dx
        thr = tau - 0.02 * (tau + 1e-10)
        act = np.where(cv * dx >= thr)[0]
        if len(act) > kc:
            act = act[np.argsort(cv[act])[::-1][:kc]]
        if len(act) == 0:
            act = np.array([int(cv.argmax())])
        c_obj = np.zeros(n+1)
        c_obj[-1] = 1.0
        Arows = []
        brows = []
        for idx in act:
            g = lag_grad(hc, idx-(n-1))
            row = np.zeros(n+1)
            row[:n] = g
            row[n] = -1.0
            Arows.append(row)
            brows.append(cv[idx] - g @ hc)
        A_ub = np.array(Arows)
        b_ub = np.array(brows)
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([ts])
        xlo = np.maximum(0.0, hc - trust)
        xhi = np.minimum(1.0, hc + trust)
        bounds = [(xlo[i], xhi[i]) for i in range(n)]
        bounds.append((max(0.0, tau/dx - trust*20), tau/dx + trust*20))
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.status == 0:
            xn = project_box_sum(np.clip(res.x[:n], 0, 1), ts)
            on = eval_c5(xn)
            if on < bo - 1e-12:
                bo = on
                bh = xn.copy()
                hc = xn
                trust = min(0.4, trust*1.05)
            else:
                trust *= 0.8
        else:
            trust *= 0.7
    return bh, bo

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    rng = np.random.RandomState(seed)
    
    # Get parent
    try:
        par = np.array(initial_h_values, dtype=float).copy()
        par = np.clip(par, 0, 1)
        par = project_box_sum(par, len(par)/2.0)
        bh = par.copy()
        bo = eval_c5(bh)
    except Exception:
        n0 = 45
        h0 = np.zeros(n0)
        h0[:n0//2] = 1.0
        if n0 % 2 == 1:
            h0[n0//2] = 0.5
        bh = project_box_sum(h0, n0/2.0)
        bo = eval_c5(bh)
    
    deadline = t0 + budget_s * 0.93
    
    # Phase 1: Optimize at native resolution
    if time.monotonic() < deadline:
        tl = min(80.0, (deadline - time.monotonic()) * 0.2)
        h, o = opt_smooth(bh, 40, tl, time.monotonic())
        if o < bo:
            bo, bh = o, h
