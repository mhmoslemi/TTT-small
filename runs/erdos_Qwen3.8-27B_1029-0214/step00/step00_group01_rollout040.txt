def exact_c5(h):
    """Compute the exact C5 value for a valid h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def validate_h(h, n):
    """Check if h is valid."""
    if len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-9) or np.any(h > 1 + 1e-9):
        return False
    target = n / 2.0
    if abs(np.sum(h) - target) > 1e-6:
        return False
    return True

def project_to_feasible(h, n):
    """Project h to the feasible set [0,1]^n intersect {sum = n/2}."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = np.sum(h) - target
    if abs(diff) > 1e-10:
        # Simple approach: adjust uniformly, then re-clip
        # This might not work perfectly, so use water-filling
        h = h - diff / n
        h = np.clip(h, 0.0, 1.0)

        for _ in range(100):

            if diff > 0:

                mask = h > 0
                if not np.any(mask):

                h[mask] -= adj

                if not np.any(mask):
                    break
                adj = -diff / np.sum(mask)
                h[mask] += adj
                h = np.clip(h, 0.0, 1.0)

    h = np.clip(h.copy(), 0.0, 1.0)

                break
