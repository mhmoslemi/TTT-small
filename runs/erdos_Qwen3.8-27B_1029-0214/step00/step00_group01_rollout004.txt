import numpy as np
import time
from scipy.optimize import minimize

def exact_c5(h):
    """Compute exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def waterfill(y, target):
    """Project y onto {h: 0 <= h <= 1, sum(h) = target}."""
    h = np.clip(y.copy(), 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        # Need to decrease sum by diff
        # Subtract from entries that are < 1
        # Binary search on tau
        lo = 0.0
        hi = diff  # maximum we might need to subtract from any entry

        # h_i - tau is clipped to [0, 1], but since h_i <= 1, clipping to [0,1] means clip(h_i - tau, 0, 1) = max(0, h_i - tau) (since h_i - tau <= 1 always if tau >= 0)
        # So we need sum(max(0, h_i - tau)) = target
        # This is a decreasing function of tau

            s = np.sum(np.maximum(0.0, h - mid))
            if s > target:
                lo = mid
            else:
                hi = mid
        h = np.maximum(0.0, h - (lo + hi) / 2)
    else:
        # Need to increase sum by -diff
        # Add to entries that are > 0
        # sum(min(1, h_i + tau)) = target
        lo = 0.0
        hi = -diff  # maximum we might need to add

            else:

    h = np.clip(h, 0.0, 1.0)
    return h

def sigmoid(x):
    """Numerically stable sigmoid."""
    x = np.clip(x, -50, 50)
    return 1.0 / (1.0 + np.exp(-x))

def solve_lambda(y, target):

    lo = -50.0
    hi = 50.0

    return (lo + hi) / 2

    # Numerically correct the sum
    diff = np.sum(h) - target

        h = waterfill(h, target)
    return h

    dx = 2.0 / n
    h = h_from_y(y, target)
    
    # Compute all c_k * dx
    # c_k for k = -(n-1), ..., n-1

    corr = np.correlate(h, 1.0 - h, mode="full") * dx

        shift = max_c
    else:

    n_src = len(h)

        return h.copy()
    # Use linear interpolation

    x_tgt = np.linspace(0, 2, n_target, endpoint=False) + 1.0/n_target
    h_new = np.interp(x_tgt, x_src, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    # Project to correct sum
    target = n_target / 2.0
    h_new = waterfill(h_new, target)
    return h_new

def make_fallback(n):

    x = np.linspace(0, 2, n, endpoint=False) + 1.0/n
    h = 0.5 + 0.3 * np.sin(np.pi * x)
    h = np.clip(h, 0, 1)
    h = waterfill(h, target)
    return h

```python
import numpy as np
import time
from scipy.optimize import minimize

_opt_state = {}

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def waterfill(y, target):
    h = np.clip(y.copy(), 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) < 1e-14 * max(1.0, target):
        return h
    if diff > 0:
        lo, hi = 0.0, diff
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.maximum(0.0, h - mid))
            if s > target:
                lo = mid
            else:
                hi = mid
        h = np.maximum(0.0, h - (lo + hi) / 2.0)
    else:
        lo, hi = 0.0, -diff
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.minimum(1.0, h + mid))
            if s < target:
                lo = mid
            else:
                hi = mid
        h = np.minimum(1.0, h + (lo + hi) / 2.0)
    return np.clip(h, 0.0, 1.0)

def sigmoid(x):
    x = np.clip(x, -50.0, 50.0)
    return 1.0 / (1.0 + np.exp(-x))

def solve_lambda(y, target, n):
    lo, hi = -50.0, 50.0
    if np.sum(sigmoid(y + hi)) < target:
        hi = 200.0
    if np.sum(sigmoid(y + lo)) > target:
        lo = -200.0
    for _ in range(150):
        mid = (lo + hi) / 2.0
        s = np.sum(sigmoid(y + mid))
        if s > target:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2.0

def h_from_y(y, target, n):
    lam = solve_lambda(y, target, n)
    h = sigmoid(y + lam)
    diff = np.sum(h) - target
    if abs(diff) > 1e-8:
        h = waterfill(h, target)
    return h

def smooth_obj_grad(y, beta, target, n):
    dx = 2.0 / n
    h = h_from_y(y, target, n)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    
    max_c = np.max(corr)
    z = beta * (corr - max_c)
    exp_z = np.exp(z)
    sum_exp = np.sum(exp_z)
    obj = (1.0 / beta) * (np.log(sum_exp) + beta * max_c)
    w = exp_z / sum_exp
    
    grad_h = np.zeros(n)
    n_corr = len(corr)
    for idx in range(n_corr):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        L = n - abs(k)
        if k >= 0:
            if L > 0:
                grad_h[0:L] += wi * dx * (1.0 - h[k:k+L])
            if k < n:
                grad_h[k:n] -= wi * dx * h[0:L]
        else:
            r = -k
            if r < n:
                grad_h[r:n] += wi * dx * (1.0 - h[0:L])
            if L > 0:
                grad_h[0:L] -= wi * dx * h[r:r+L]
    
    p = h
    W = p * (1.0 - p)
    S_W = np.sum(W)
    if S_W < 1e-15:
        return obj, np.zeros(n)
    weighted_mean = np.sum(grad_h * W) / S_W
    grad_y = W * (grad_h - weighted_mean)
    
    return obj, grad_y

def smooth_obj_wrapper(y):
    return smooth_obj_grad(y, _opt_state['beta'], _opt_state['target'], _opt_state['n'])[0]

def smooth_obj_grad_wrapper(y):
    return smooth_obj_grad(y, _opt_state['beta'], _opt_state['target'], _opt_state['n'])

def transfer_resolution(h, n_target):
    n_src = len(h)
    if n_src == n_target:
        return h.copy()
    x_src = (np.arange(n_src) + 0.5) / n_src * 2.0
    x_tgt = (np.arange(n_target) + 0.5) / n_target * 2.0
    h_new = np.interp(x_tgt, x_src, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    return waterfill(h_new, n_target / 2.0)

def make_fallback(n):
    x = (np.arange(n) + 0.5) / n * 2.0
    h = 0.5 + 0.4 * np.sin(np.pi * x)
    h = np.clip(h, 0.0, 1.0)
    return waterfill(h, n / 2.0)

def make_binary_variant(h, rng):
    n = len(h)
    target = n / 2.0
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:int(target)]] = 1.0
    if n % 2 == 1:
        h_bin[sorted_idx[int(target)]] = 1.0
    # Wait, target = n/2. If n is odd, target is x.5
    # We need exactly target sum. For n=63, target=31.5
    # So we need 31 ones and one 0.5
    if abs(target - int(target)) > 0.1:
        h_bin = np.zeros(n)
        h_bin[sorted_idx[:int(target)]] = 1.0
        h_bin[sorted_idx[int(target)]] = target - int(target)
    return h_bin

def optimize_at_resolution(h_start, n, deadline, rng):
    target = n / 2.0
    h_start = waterfill(h_start.copy(), target)
    best_c = exact_c5(h_start)
    best_h = h_start.copy()
    
    eps = 1e-6
    h_clip = np.clip(h_start, eps, 1.0 - eps)
    y = np.log(h_clip / (1.0 - h_clip))
    y = y - np.mean(y)  # center
    
    betas = [15, 40, 100, 250, 600, 1200, 2500, 5000]
    max_iters = [80, 60, 50, 40, 30, 25, 20, 15]
    
    _opt_state['target'] = target
    _opt_state['n'] = n
    
    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        _opt_state['beta'] = beta
        max_it = max_iters[bi]
        try:
            res = minimize(smooth_obj_grad_wrapper, y, method='L-BFGS-B',
                          options={'maxiter': max_it, 'maxfun': max_it * 15,
                                   'ftol': 1e-14, 'gtol': 1e-10})
            if res.x is not None:
                y = res.x
        except Exception:
            break
        
        h_new = h_from_y(y, target, n)
        h_new = waterfill(h_new, target)
        c_new = exact_c5(h_new)
        if c_new < best_c:
