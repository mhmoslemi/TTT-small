def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_all_c5_terms(h, n):
    """Compute all 2n-1 correlation terms (scaled by dx)."""
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    return corr * dx

    if residual > 0:

        can_increase = 1.0 - h
        room = np.sum(can_increase)

    # For stability

    val = m + np.log(np.sum(np.exp(beta * (terms - m)))) / beta
    return val

def violation_objective(h, n, target):
    """Sum of squared violations above target."""
    terms = compute_all_c5_terms(h, n)
    violations = np.maximum(terms - target, 0.0)
    return np.sum(violations ** 2)

def violation_grad(h, n, target):

    target = n / 2.0

    if abs(residual) < 1e-12:
        return h
    if residual > 0:

        total_room = np.sum(room)

        else:

            excess = np.sum(h) - target
            cap = np.sum(h)
            if cap > 1e-10:
                h -= excess * h / cap
    else:
