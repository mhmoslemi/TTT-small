    J[:, n] = 1.0
    for r in range(n):
        e = n - r
        k_idx = r + n - 1
        J[k_idx, 0:e] += dx * (1.0 - h[r:r+e])
        J[k_idx, r:n] += -dx * h[0:e]
    for r in range(1, n):
        e = n - r
        k_idx = n - 1 - r
        J[k_idx, r:n] += dx * (1.0 - h[0:e])
        J[k_idx, 0:e] += -dx * h[r:n]
    return J

def _eq_fun(x):
    return float(np.sum(x[:_GN])) - _GN / 2.0

def _eq_jac(x):
    j = np.zeros(_GN + 1)
    j[:_GN] = 1.0
    return j

def solve_epigraph(h_start, n, maxiter=300):
    global _GN, _GDX
    _GN = n
    _GDX = 2.0 / n
    target = n / 2.0
    h_start = exact_project(h_start, target)
    c5_start = compute_c5(h_start)
    x0 = np.zeros(n + 1)
    x0[:n] = h_start
    x0[n] = c5_start + 0.005
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [
        {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac},
        {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}
    ]
    best_h = h_start.copy()
    best_c5 = c5_start
    try:
        res = minimize(_obj, x0, method='SLSQP', jac=_obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if np.all(np.isfinite(res.x)):
            h_opt = exact_project(res.x[:n], target)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    except Exception:
        pass
    return best_h, best_c5
