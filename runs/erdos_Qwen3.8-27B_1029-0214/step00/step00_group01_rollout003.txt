def solve_direction(G, n, delta):
    """Solve LP: minimize t s.t. G @ d <= t, sum(d) = 0, -delta <= d <= delta"""
    m = G.shape[0]

    # Minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: G @ d <= t  =>  G @ d - t <= 0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -delta <= d_i <= delta, t unbounded below (but... t can be any real)
    bounds = [(-delta, delta)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')

            if c5_up3 < best_c5 + 0.005:
                h_opt4, c5_opt4 = local_optimize(h_up3, deadline, max_iters=50)
                if c5_opt4 < best_c5:
                    best_c5 = c5_opt4
                    best_h = h_opt4.copy()
    
    # Phase 5: Final polish
    if time.time() < deadline - 5:
        n_final = len(best_h)
        h_f, c5_f = local_optimize(best_h, deadline, max_iters=50)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    # Ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = evaluate_c5(best_h)
    best_c5 = min(best_c5, final_c5)
    
    return best_h, best_c5, len(best_h)
