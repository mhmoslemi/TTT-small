def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx

def project_to_box_sum(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum using bisection."""
    n = len(h)
    h = h.copy()
    # Bisection on lambda: clip(h - lambda, 0, 1) has sum = target_sum
    lo = np.min(h) - 1.0
    hi = np.max(h) + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target_sum - h_proj.sum()
    if abs(residual) > 1e-12:
        # Distribute residual
        if residual > 0:
            room = 1.0 - h_proj
            if room.sum() > 1e-15:
                frac = residual / room.sum()
                h_proj = np.minimum(1.0, h_proj + frac * room)
        else:
            room = h_proj.copy()
            if room.sum() > 1e-15:
                frac = -residual / room.sum()
                h_proj = np.maximum(0.0, h_proj - frac * room)
    # Final exact fix
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-15:
        # Find an index that can absorb the residual
        if diff > 0:
            idx = np.argmax(1.0 - h_proj)
        else:
            idx = np.argmin(h_proj)
        h_proj[idx] += diff
        h_proj[idx] = np.clip(h_proj[idx], 0.0, 1.0)
    return h_proj

        t_max = min(t_max, np.min(h[neg_mask] / (-d[neg_mask])))

    margin = 1e-12
    t_min += margin
    t_max -= margin

    lo = t_min
    hi = t_max

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_to_sum(h, target_sum):
    n = len(h)
    h = np.array(h, dtype=float)
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h_p = np.clip(h - lam, 0.0, 1.0)
    res = target_sum - float(h_p.sum())
    if abs(res) > 1e-12:
        if res > 0:
            room = 1.0 - h_p
            tot = float(room.sum())
            if tot > 1e-15:
                h_p = np.minimum(1.0, h_p + (res / tot) * room)
        else:
            tot = float(h_p.sum())
            if tot > 1e-15:
                h_p = np.maximum(0.0, h_p + (res / tot) * h_p)

    if abs(diff) > 1e-14:
        idx = int(np.argmax(1.0 - h_p)) if diff > 0 else int(np.argmin(h_p))
