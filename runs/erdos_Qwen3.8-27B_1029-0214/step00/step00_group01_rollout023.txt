def evaluate_c5(h):
    """Exact evaluation of C5 objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def project_to_feasible(h, n):
    """Project h onto the feasible set: 0 <= h[i] <= 1, sum(h) = n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Adjust to meet sum constraint using water-filling
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    # Need to adjust: if s > target, decrease; if s < target, increase
    diff = target - s
    # Find elements with room to adjust
    if diff > 0:
        # Need to increase total by diff
        # Prioritize elements closest to 1 (they have least room, actually we want those with most room)
        # Actually we want to increase elements, prioritizing those with most room (lowest values)
        room = 1.0 - h
        # Scale up proportionally to room
        total_room = np.sum(room)
        if total_room > 1e-15:
            h = h + diff / total_room * room
        else:
            # All at 1, distribute evenly
            h = h + diff / n
    else:
        # Need to decrease total by |diff|
        room = h  # room to decrease
        total_room = np.sum(room)
        if total_room > 1e-15:
            h = h + diff / total_room * room  # diff is negative
        else:
            h = h + diff / n
    # Final clip and correction
    h = np.clip(h, 0.0, 1.0)
    # Correct any remaining error
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:

        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(interior) > 0:

            h = np.clip(h, 0.0, 1.0)

            idx = np.argmin(np.abs(h - 0.5))
