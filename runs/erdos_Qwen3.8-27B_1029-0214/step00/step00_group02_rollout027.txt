            # m ranges from lo1 to hi1-1, m-s ranges from lo1-s to hi1-1-s
            # h[m-s] = h[lo1-s : hi1-s]

        lo2 = max(0, -s)

        if lo2 < hi2:
            # m ranges from lo2 to hi2-1, m+s ranges from lo2+s to hi2-1+s
            # h[m+s] = h[lo2+s : hi2+s]
            grad[lo2:hi2] -= p[j] * dx * h[lo2+s:hi2+s]
    
    return grad

    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)


def _proj(y, tgt):
    lo = -2.0
    hi = 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)


def _solve(h0, n, taus, nit, bnd):
    bh = h0.copy()
    bv = _exact(h0)
    _PN[0] = n
    CONS = [{'type': 'eq', 'fun': _eqf, 'jac': _eqj}]
    tgt = n / 2.0
    for tau in taus:
        _PT[0] = tau
        try:
            r = minimize(_obj, bh, jac=_grad, method='SLSQP',
                         bounds=bnd, constraints=CONS,
                         options={'maxiter': nit, 'ftol': 1e-10})
            cand = _proj(r.x, tgt)
            v = _exact(cand)
            if v < bv - 1e-13:
                bh = cand
                bv = v
        except Exception:
            pass
    return bh, bv


def _tleft(t0, bud, res):
    return bud - res - (time.time() - t0)


def _local_search(h0, n, seed_rng, max_time_s):
    bh = h0.copy()
    bv = _exact(h0)
    t_start = time.time()
    for st in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:
        while time.time() - t_start < max_time_s:
            for _ in range(2000):
                if time.time() - t_start > max_time_s:
                    break
                i = int(rng_int(seed_rng, 0, n)[0])
                j = int(rng_int(seed_rng, 0, n)[1])
                if i == j:
                    continue
                d = st * (2.0 * seed_rng.random() - 1.0)
                ni = bh[i] + d
                nj = bh[j] - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    bh[i] = ni
                    bh[j] = nj
                    v = _exact(bh)
                    if v < bv - 1e-14:
                        bv = v
                    else:
                        bh[i] -= d
                        bh[j] += d
            if _exact(bh) < bv:
                bv = _exact(bh)
    return bh, bv


def rng_int(rng, lo, hi):
    return rng.integers(lo, hi, size=2)
