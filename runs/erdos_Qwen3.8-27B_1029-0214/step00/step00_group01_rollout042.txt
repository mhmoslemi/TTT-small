        else:
            step_size = max(0.01, step_size * 0.98)

import numpy as np
import math
import time
from scipy.optimize import minimize

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(h, target_sum=None):
    n = len(h)
    if target_sum is None:
        target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s0 = np.sum(h)
    if abs(s0 - target_sum) < 1e-14:
        return h
    if s0 < target_sum:
        lo, hi = -1.0, 2.0
    else:
        lo, hi = -2.0, 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        ss = np.sum(np.clip(h + mid, 0.0, 1.0))
        if ss < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def sigmoid_safe(x):
    x = np.clip(x, -40.0, 40.0)
    out = np.empty_like(x, dtype=np.float64)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out

def y_to_h(y, n, lam_prev=0.0):
    target = n / 2.0
    lam = lam_prev
    for _ in range(50):
        h = sigmoid_safe(y + lam)
        p = h * (1.0 - h)
        S = np.sum(p)
        f = np.sum(h) - target
        if abs(f) < 1e-13:
            break
        if S < 1e-12:
            lo = lam - 20.0
            hi = lam + 20.0
            for _ in range(100):
                mid = (lo + hi) / 2.0
                if np.sum(sigmoid_safe(y + mid)) < target:
                    lo = mid
                else:
                    hi = mid
            lam = (lo + hi) / 2.0
            h = sigmoid_safe(y + lam)
            break
        lam -= f / S
        lam = float(np.clip(lam, -40.0, 40.0))
    h = project_feasible(h, target)
    return h, lam

def lse_val_grad_h(h, beta):
    n = len(h)
    dx = 2.0 / n
    carr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(carr))
    shifted = beta * (carr - m)
    exp_vals = np.exp(np.clip(shifted, -700.0, 700.0))
    Z = float(np.sum(exp_vals))
    val = (m + math.log(Z)) / beta
    w = exp_vals / Z
    G = np.zeros(n, dtype=np.float64)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-18:
            continue
        k = idx - (n - 1)
        L = n - abs(k)
        if k >= 0:
            G[k:k+L] += wi * dx * (1.0 - h[:L])
            G[:L] += wi * dx * (-h[k:k+L])
        else:
            r = -k
            G[:L] += wi * dx * (1.0 - h[r:r+L])
            G[r:r+L] += wi * dx * (-h[:L])
    return val, G

def make_smooth_fun(n, beta):
    state = [0.0, 0.0]  # [lam, last_val]
    def fun(y):
        h, state[0] =
