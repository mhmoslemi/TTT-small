    return float(np.max(cor)) * dx

def project(h, n):
    """Project to {0<=h<=1, sum=h=n/2}"""
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        sr = np.sum(room)
        if sr > 1e-14:
            h += (diff / sr) * room
        else:
            h[:] = target / n
    else:
        sh = np.sum(h)
        if sh > 1e-14:
            h += (diff / sh) * h
        else:
            h[:] = target / n
    return np.clip(h, 0.0, 1.0)

def binary_round(h, n, rng):
    """Round h towards binary while maintaining sum constraint"""
    target = n * 0.5
    h = np.asarray(h, dtype=float).copy()
    # Sort by distance to nearest integer
    dist0 = h  # distance to 0
    dist1 = 1.0 - h  # distance to 1
    dist = np.minimum(dist0, dist1)
    order = np.argsort(dist)[:n//2]  # entries closest to binary
    # Try to round these
    h_new = h.copy()
    for i in order:
        if h[i] >= 0.5:
            h_new[i] = 1.0
        else:
            h_new[i] = 0.0
    h_new = project(h_new, n)
    return h_new
