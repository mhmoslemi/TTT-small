def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_box_simplex(z, target):
    n = len(z)
    z = np.asarray(z, dtype=float).copy()
    bad = ~np.isfinite(z)
    if np.any(bad):
        z[bad] = 0.5
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-13:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
        else:
            idx = int(np.argmax(result)) if diff > 0 else int(np.argmin(result))
            result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result


def corr_lag_value(h, k):
    n = len(h)
    if k >= 0:
        if k >= n:
            return 0.0
        return float(np.dot(h[:n - k], 1.0 - h[k:]))
    else:
        r = -k
        if r >= n:
            return 0.0
        return float(np.dot(h[r:], 1.0 - h[:n - r]))


def corr_lag_grad(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        if k >= n:
            return grad
        grad[k:] += (1.0 - h[:n - k])
        grad[:n - k] -= h[k:]
    else:
        r = -k
        if r >= n:
            return grad
        grad[:n - r] += (1.0 - h[r:])
        grad[r:] -= h[:n - r]
    return grad


class NLPProblem:
    def __init__(self, n, lags, dx, target, deadline):
        self.n = n
        self.lags = lags
        self.dx = dx
        self.target = target
        self.deadline = deadline

    def _check(self):
        if time.time() > self.deadline:
            raise ValueError("timeout")

    def fun(self, x):
        self._check()
        return x[-1]

    def grad(self, x):
        self._check()
        g = np.zeros(self.n + 1)
        g[-1] = 1.0
        return g

    def con_eq(self, x):
        self._check()
        return np.sum(x[:self.n]) - self.target

    def con_eq_jac(self, x):
        self._check()
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g

    def con_ineq(self, x):
        self._check()
        h = x[:self.n]
        t = x[-1]
        out = np.empty(len(self.lags))
        for i in range(len(self.lags)):
            out[i] = t - self.dx * corr_lag_value(h, self.lags[i])
        return out

    def con_ineq_jac(self, x):
        self._check()
        h = x[:self.n]
        n = self.n
        M = len(self.lags)
        J = np.zeros((M, n + 1))
        for i in range(M):
            J[i, :n] = -self.dx * corr_lag_grad(h, self.lags[i])
            J[i, n] = 1.0
        return J

    deadline = t_start + budget_s - reserve
    rng = np.random.default_rng(seed)

    # Get parent
    try:
        parent = np.asarray(initial_h_values, dtype=float)
    except (NameError, TypeError):
        parent = None

    if parent is not None and len(parent) >= 8:
        n_base = len(parent)
        h_best = project_box_simplex(parent, n_base / 2.0)
        c_best = exact_c5(h_best)
    else:
        n_base = 64
        h_best = np.full(n_base, 0.5)
        c_best = exact_c5(h_best)

    # Phase 1: Optimize at base resolution with multiple strategies
    if time.time() < deadline:
        h_new, c_new = optimize_at_resolution(
            h_best, n_base, deadline, rng,
            max_solves=min(40, int((deadline - time.time()) / 3.0)))
        if c_new < c_best:
            h_best = h_new
            c_best = c_new

    # Phase 2: Try transforms and re-optimize
    if time.time() < deadline and c_best > 0.38079:
        h_tr, c_tr = try_transforms(h_best, n_base, deadline, rng)
        if c_tr < c_best:
            h_best = h_tr
            c_best = c_tr

        h_push = binarize_push(h_best, n_base, 0.5)
        c_push = exact_c5(h_push)
        if c_push < c_best:
            h_best = h_push
            c_best = c_push

        # Re-optimize from improved start
        h_new, c_new = optimize_at_resolution(
            h_best, n_base, deadline, rng, max_solves=20)
        if c_new < c_best:
            h_best = h_new
            c_best = c_new

    # Phase 3: Try at 2x resolution
    if time.time() < deadline - 30:
        n2 = n_base * 2
        h2 = np.repeat(h_best, 2)
        h2 = project_box_simplex(h2, n2 / 2.0)
        c2 = exact_c5(h2)
        if c2 < c_best:
            h_best = h2
            c_best = c2

        h_new, c_new = optimize_at_resolution(
            h2, n2, deadline, rng, max_solves=15)
        if c_new < c_best:
            h_best = h_new
            c_best = c_new

    # Phase 4: Try at 4x resolution if time permits
    if time.time() < deadline - 30 and budget_s > 400:
        n3 = n_base * 4
        h3 = np.repeat(h_best, max(1, n3 // len(h_best)))
        h3 = project_box_simplex(h3, len(h3) / 2.0)
        c3 = exact_c5(h3)
        if c3 < c_best:
            h_best = h3
            c_best = c3

        h_new, c_new = optimize_at_resolution(
            h3, len(h3), deadline, rng, max_solves=10)
        if c_new < c_best:
            h_best = h_new
            c_best = c_new

    # Phase 5: Final polish at best resolution
    if time.time() < deadline - 5:
        n_final = len(h_best)
        h_new, c_new = optimize_at_resolution(
            h_best, n_final, deadline, rng, max_solves=10)
        if c_new < c_best:
            h_best = h_new
            c_best = c_new

    # Final validation and repair
    n_final = len(h_best)
    h_final = project_box_simplex(h_best, n_final / 2.0)
    h_final = np.clip(h_final, 0.0, 1.0)
    c_final = exact_c5(h_final)

    if not np.isfinite(c_final) or c_final > 1.0:
        h_final = project_box_simplex(h_best, n_final / 2.0)
        c_final = exact_c5(h_final)

    # Ensure sum constraint is satisfied within tolerance
    sum_err = np.sum(h_final) - n_final / 2.0
    if abs(sum_err) > 1e-10:
        h_final = project_box_simplex(h_final, n_final / 2.0)
        c_final = exact_c5(h_final)

    return (h_final, c_final, n_final)
