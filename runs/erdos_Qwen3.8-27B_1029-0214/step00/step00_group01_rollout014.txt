    if tau >= 0:
        L = n - tau
        if L <= 0:
            return grad
        grad[:L] += (1.0 - h[tau:])
        grad[tau:] -= h[:L]
    else:
        r = -tau
        L = n - r
        if L <= 0:
            return grad
        grad[r:] += (1.0 - h[:L])
        grad[:L] -= h[r:]
    return grad
