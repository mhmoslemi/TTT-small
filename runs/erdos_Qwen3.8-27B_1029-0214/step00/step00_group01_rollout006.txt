def _logsumexp_obj_grad(y, beta, target=None):
    n = len(y)
    if target is None:
        target = n / 2.0
    dx = 2.0 / n
    
    h = _make_feasible(y, target)
    
    # Compute c values
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    num_lags = 2 * n - 1
    
    # Log-sum-exp
    c_max = c.max()
    exp_vals = np.exp(beta * (c - c_max))
    Z = exp_vals.sum()
    
    if Z < 1e-300:
        w = np.ones(num_lags) / num_lags
    else:
        w = exp_vals / Z
    
    F = c_max + np.log(Z) / beta
    
    # Gradient of F with respect to h
    G = np.zeros(n)
    for idx in range(num_lags):
        ws = w[idx]
        if ws < 1e-14:
            continue
        s = idx - (n - 1)
        if s >= 0:
            L = n - s
            G[s:] += ws * dx * (1.0 - h[:L])
            G[:L] -= ws * dx * h[s:]
        else:
            r = -s
            L = n - r
            G[:L] += ws * dx * (1.0 - h[r:])
            G[r:] -= ws * dx * h[:L]
    
    # Convert to y-space gradient
    W = h * (1.0 - h)
    S = W.sum()
    if S < 1e-12:
        return F, np.zeros(n)
    A = (G * W).sum()
    grad_y = W * (G - A / S)
    
    # Numerical cleanup
    if not np.isfinite(F):
        return 1.0, np.zeros(n)
    grad_y = np.nan_to_num(grad_y, nan=0.0, posinf=0.0, neginf=0.0)
    
    return F, grad_y

        mode = rng.integers(0, 3)
        if mode == 0:
            i = rng.integers(0, n)
            h_t[i] = np.clip(h_t[i] + rng.uniform(-0.4, 0.4), 0.0, 1.0)
        elif mode == 1:
            i, j = rng.choice(n, size=2, replace=False)
            h_t[i], h_t[j] = h_t[j], h_t[i]
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                d = h_t[i] - h_t[j]
                h_t[i] = np.clip(h_t[i] - d * 0.5, 0.0, 1.0)
                h_t[j] = np.clip(h_t[j] + d * 0.5, 0.0, 1.0)
        h_t = _waterfill(h_t, target)
        c5_t = _compute_c5(h_t)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_h = h_t.copy()
    return best_h, best_c5

def _try_quantize(h):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    for thresh in np.linspace(0.3, 0.7, 9):
        h_q = np.where(h >= thresh, 1.0, 0.0)
        s = h_q.sum()
        if abs(s - target) < 0.5:
            if s < target:
                idx = np.argmax(np.where(h >= thresh, 0, h))
                h_q[idx] = 1.0
            elif s > target:
                idx = np.argmin(np.where(h >= thresh, h, 1.0))
                h_q[idx] = 0.0
            h_q = _waterfill(h_q, target)
            c5_q = _compute_c5(h_q)
            if c5_q < best_c5:
                best_c5 = c5_q
                best_h = h_q.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + budget_s * 0.90
    rng = np.random.default_rng(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        n_fb = 126
        x = (np.arange(n_fb) + 0.5) / n_fb * 2.0
        h_parent = 0.5 + 0.3 * np.sin(np.pi * x)
        h_parent = _waterfill(h_parent, n_fb / 2.0)

    n_parent = len(h_parent)
    h_parent = np.clip(h_parent, 0.0, 1.0)
    h_parent = _waterfill(h_parent, n_parent / 2.0)

    best_h = h_parent.copy()
    best_c5 = _compute_c5(h_parent)

    beta_sched = [5, 20, 80, 300, 1000, 3000, 8000, 20000]

    if time.monotonic() < deadline:
        h_opt, c5_opt = _optimize_smooth(h_parent, beta_sched, deadline, maxiter=100)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if time.monotonic() < deadline:
        h_q, c5_q = _try_quantize(best_h)
        if c5_q < best_c5:
            best_c5 = c5_q
            best_h = h_q.copy()

    if time.monotonic() < deadline:
        h_ls, c5_ls = _local_search(best_h, 3000, deadline, seed=seed+1)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    if time.monotonic() < deadline and n_parent <= 128:
        h_up = _upsample(h_parent, 2)
        n_up = len(h_up)
        h_up = _waterfill(h_up, n_up / 2.0)
        beta_up = [10, 50, 200, 800, 3000, 10000]
        h_opt, c5_opt = _optimize_smooth(h_up, beta_up, deadline, maxiter=70)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if time.monotonic() < deadline:
            h_q, c5_q = _try_quantize(best_h)
            if c5_q < best_c5:
                best_c5 = c5_q
                best_h = h_q.copy()
        if time.monotonic() < deadline:
            h_ls, c5_ls = _local_search(best_h, 5000, deadline, seed=seed+2)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()

    if time.monotonic() < deadline:
        h_best_up = _upsample(best_h, max(1, 126 // n_parent)) if n_parent < 126 else _upsample(best_h, 2)
        h_best_up = _waterfill(h_best_up, len(h_best_up) / 2.0)
        beta_r = [50, 300, 1500, 5000]
        h_opt, c5_opt = _optimize_smooth(h_best_up, beta_r, deadline, maxiter=60)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    for restart in range(4):
        if time.monotonic() > deadline:
            break
        scale = 0.05 + 0.05 * restart
        h_pert = best_h + rng.normal(0, scale, size=len(best_h))
        h_pert = _waterfill(h_pert, len(best_h) / 2.0)
        beta_r = [30, 200, 1000, 5000]
        h_opt, c5_opt = _optimize_smooth(h_pert, beta_r, deadline, maxiter=50)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if time.monotonic() < deadline:
        h_ls, c5_ls = _local_search(best_h, 5000, deadline, seed=seed+100)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    if time.monotonic() < deadline:
        h_q, c5_q = _try_quantize(best_h)
        if c5_q < best_c5:
            best_c5 = c5_q
            best_h = h_q.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    best_h = _waterfill(best_h, n_final / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _compute_c5(best_h)

    if not np.isfinite(final_c5) or final_c5 > 1.0:
        best_h = h_parent.copy()
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = _waterfill(best_h, n_parent / 2.0)
        final_c5 = _compute_c5(best_h)

    return best_h, final_c5, len(best_h)
