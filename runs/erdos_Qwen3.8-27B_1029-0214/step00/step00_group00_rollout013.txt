    return float(np.max(c) * dx)

def is_feasible(h, tol=1e-9):
    n = len(h)
    if not np.all(h >= -tol) or not np.all(h <= 1.0 + tol):
        return False
    if abs(h.sum() - n/2.0) > tol * n:
        return False
    return True

def project(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Distribute the difference
    if diff > 0:
        rooms = 1.0 - h
        total_room = rooms.sum()
        if total_room < 1e-15:
            return h
        h += (diff / total_room) * rooms
        h = np.clip(h, 0.0, 1.0)
    else:
        rooms = h.copy()
        total_room = rooms.sum()
        if total_room < 1e-15:
            return h
        h += (diff / total_room) * rooms
        h = np.clip(h, 0.0, 1.0)

Instead of ternary search (which does 70 evaluations), I can use the fact that the function is piecewise linear and find the minimum in O(number of kinks) evaluations. The number of kinks is at most 2n-1 (the number of correlation terms). But finding them requires knowing which term is active at each point, which requires evaluation...

Actually, for a max of L linear functions, the minimum can be found in O(L) evaluations using a "walk" method:
1. Start at some point
2. Find the active term(s)
3. If the subgradient contains 0, we're done
4. Else, move in the negative subgradient direction to the next kink
5. Repeat

Each kink is at a point where two terms are equal. The number of kinks crossed is at most L = 2n-1. So the total is O(L) evaluations.

For n=45, L = 89. So at most 89 evaluations per pair (vs 70 for ternary search). Not a clear win.

But the key advantage is that the walk method is exact (finds the true minimum of the piecewise linear function), while ternary search approximates it. For our purpose (improving by 5×10^-5), both should work.

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project(h, n):
    target = n / 2.0
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        rooms = 1.0 - h
        total = rooms.sum()
        if total > 1e-15:
            h = h + (diff / total) * rooms
    else:
        total = h.sum()
        if total > 1e-15:
            h = h * (1.0 + diff / total)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        for i in range(n):
            if abs(diff) < 1e-14:
                break
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h

def upsample_step(h, new_n):
    old_n = len(h)
    idx = np.minimum((np.arange(new_n) * old_n) // new_n, old_n - 1)
    return project(h[idx].copy(), new_n)

def line_search_pair(h, i, j):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return 0.0, evaluate_c5(h)
    a, b = lo, hi
    for _ in range(25):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        h1 = h.copy()
        h1[i] += m1
        h1[j] -= m1
        v1 = evaluate_c5(h1)
        h2 = h.copy()
        h2[i] += m2
        h2[j] -= m2
        v2 = evaluate_c5(h2)
        if v1 <= v2:
            b = m2
        else:
            a = m1
        if b - a < 1e-13:
            break
    opt = (a + b) / 2.0
    h_opt = h.copy()
    h_opt[i] += opt
    h_opt[j] -= opt
    return opt, evaluate_c5(h_opt)
