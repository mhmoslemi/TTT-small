```python
import numpy as np
from scipy.optimize import minimize
from scipy import correlate as sp_correlate
import time

def project_to_feasible(y, n):
    """Project y onto {h in [0,1]^n : sum(h) = n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -1.0, 2.0  # lambda range
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo+hi)/2.0, 0.0, 1.0)
    # Fine adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Distribute among non-bound entries
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    """Compute exact C5 = max_k correlate(h, 1-h, full) * dx"""
    n = len(h)
    dx = 2.0 / n
    complement = 1.0 - h
    corr = np.correlate(h, complement, mode='full')
    return np.max(corr) * dx

def soft_objective_and_grad(h, tau):
    """Compute soft objective F_tau(h) and its gradient."""
    n = len(h)
    dx = 2.0 / n
    complement = 1.0 - h
    corr = np.correlate(h, complement, mode='full') * dx  # shape (2n-1,)
    
    # Softmax weights
    max_c = np.max(corr)
    exp_vals = np.exp(tau * (corr - max_c))
    sum_exp = np.sum(exp_vals)
    p = exp_vals / sum_exp  # shape (2n-1,)
    
    F = (1.0/tau) * (np.log(sum_exp) + tau * max_c)
    
    # Gradient: ∇F[j] = (dx/tau) * sum_k p[k] * ((1-h[j+d_k]) - h[j-d_k])
    # where d_k = n-1-k
    # Build gradient vector
    grad = np.zeros(n)
    for k in range(2*n - 1):
        d = n - 1 - k
        # (1-h[j+d]) term: j+d in [0, n-1] => j in [-d, n-1-d]
        # h[j-d] term: j-d in [0, n-1] => j in [d, n-1+d]
        for j in range(n):
            term1 = 0.0
            if 0 <= j + d <= n - 1:
                term1 = 1.0 - h[j + d]
            term2 = 0.0
            if 0 <= j - d <= n - 1:
                term2 = h[j - d]
            grad[j] += p[k] * (term1 - term2)
    
    grad *= dx / tau
    return F, grad
```

```python
def build_grad_matrix(h, n):
    """Build matrix G of shape (2n-1, n) where G[k,j] = (1-h[j+d_k]) - h[j-d_k] (0 for OOB)"""
    d = np.arange(n-1, -(n-1), -1)  # d_k = n-1-k for k=0,...,2n-2
    # d = [n-1, n-2, ..., 0, -1, ..., -(n-1)]
    grad_mat = np.zeros((2*n-1, n))
    for k, dk in enumerate(d):
        # term1: 1-h[j+dk] for valid j
        j_vals = np.arange(n)
        idx1 = j_vals + dk
        valid1 = (idx1 >= 0) & (idx1 < n)
        grad_mat[k, valid1] += 1.0 - h[idx1[valid1]]
        # term2: -h[j-dk] for valid j
        idx2 = j_vals - dk
        valid2 = (idx2 >= 0) & (idx2 < n)
        grad_mat[k, valid2] -= h[idx2[valid2]]
    return grad_mat
```

```python
    # Handle potential overflow (shouldn't happen with shift)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return (max_c + log_sum / tau)

        if mask2.any():
            contrib[mask2] -= h[idx2[mask2]]

        return soft_gradient(h, tau)
```

```python
def compute_swap_delta(i, j, h, n):
    """Compute the change in each c_k when h[i] goes 1->0 and h[j] goes 0->1."""
    dx = 2.0 / n
    # For lag k, d = n-1-k
    # delta_c_k = dx * (h[i-d] - h[i+d] - h[j-d] + h[j+d]) with OOB = 0
    d = np.arange(n-1, -(n-1), -1)  # shape (2n-1,)
    
    def safe_h(idx):
        result = np.zeros(len(idx))
        valid = (idx >= 0) & (idx < n)
        result[valid] = h[idx[valid]]
        return result
    
    di_plus = safe_h(i + d)
    di_minus = safe_h(i - d)
    dj_plus = safe_h(j + d)
    dj_minus = safe_h(j - d)
    
    delta = dx * (di_minus - di_plus - dj_minus + dj_plus)
    return delta
```

```python
import time

def project_to_feasible(y, n):
    target = n / 2.0
    y = np.asarray(y, dtype=float)
    lo, hi = -5.0, 6.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(y - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nfree = free.sum()
        if nfree > 0:
            h[free] += diff / nfree
            h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_objective(h, tau):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = tau * (corr - max_c)
    return max_c + np.log(np.sum(np.exp(shifted))) / tau

def soft_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = tau * (corr - max_c)
    exp_vals = np.exp(shifted)
    p = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    j_idx = np.arange(n)
    for k in range(2 * n - 1):
        dk = n - 1 - k
        wk = p[k]
        if wk < 1e-15:
            continue
        idx1 = j_idx + dk
        mask1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j_idx - dk
        mask2 = (idx2 >= 0) & (idx2 < n)
        if mask1.any():
            grad[mask1] += wk * (1.0 - h[idx1[mask1]])
        if mask2.any():
            grad[mask2] -= wk * h[idx2[mask2]]
    grad *= dx / tau
    return grad

                   jac=soft_gradient, method='SLSQP',
                   bounds=bounds, constraints=[constraint],
                   options={'maxiter': maxiter, 'ftol': 1e-10})
    if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
        return np.array(res.x, dtype=float)
    return None

def to_binary(h, n):
    target = n // 2
    order = np.argsort(-h)

    binary[order[:target]] = 1.0

            h[best_j] = 1.0
            c = best_new_c
```
