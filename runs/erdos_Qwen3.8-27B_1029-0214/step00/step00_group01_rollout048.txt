import numpy as np
import time
from scipy.optimize import linprog

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    n = len(h)
    # Project onto {x : 0 <= x_i <= 1, sum(x) = target}
    # Binary search on lambda: sum(clip(x_i - lambda, 0, 1)) = target
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def get_tau(k, n):
    return k - (n - 1)

def compute_gradient_row(h, k):
    n = len(h)
    tau = get_tau(k, n)
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        # First term: (1-h[i-tau]) if 0 <= i-tau <= n-1
        j1 = i - tau
        if 0 <= j1 <= n - 1:
            g += (1.0 - h[j1])
        # Second term: -h[i+tau] if 0 <= i+tau <= n-1
        j2 = i + tau
        if 0 <= j2 <= n - 1:
            g -= h[j2]
        grad[i] = g
    return grad

def solve_lp_direction(G, delta):
    # Minimize t subject to G @ d <= t, sum(d) = 0, -delta <= d_i <= delta
    n, m = G.shape
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: G @ d - t <= 0
    A_ub = np.hstack([G, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    # Equality constraint: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] * 0  # will set below
    d_bounds = [(-delta, delta)] * n
    t_bounds = [(None, None)]
    bounds = d_bounds + t_bounds
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n]
    return None

def local_optimize(h, budget_s, deadline, max_iter=200):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    h = project_feasible(h, target)
    best_c = evaluate_c5(h)
    best_h = h.copy()
    
    delta = 0.5 / np.sqrt(n)
    active_frac = 0.15
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c_min = np.min(corr)
        
        # Active constraints: top fraction
        n_active = max(3, int(len(corr) * active_frac))
        # Also always include the max
        sorted_idx = np.argsort(corr)[::-1]
        active_idx = sorted_idx[:n_active]
        
        # Build gradient matrix
        G = np.array([compute_gradient_row(h, k) for k in active_idx])
        
        # Solve LP
        d = solve_lp_direction(G, delta)
        
        if d is not None:
            # Normalize
            d_norm = np.linalg.norm(d)
            if d_norm > 1e-10:
                d = d / d_norm * delta
                # Make sum zero
                d -= np.mean(d)
                
                # Line search
                best_alpha = 0
                best_c_step = best_c
                for alpha in [0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
                    h_new = h + alpha * d
                    h_new = project_feasible(h_new, target)
                    c_new = evaluate_c5(h_new)
                    if c_new < best_c_step:
                        best_c_step = c_new
                        best_alpha = alpha
                    if alpha < 0.01:
                        break
                
                if best_alpha > 0 and best_c_step < best_c:
                    h = project_feasible(h + best_alpha * d, target)
                    best_c = best_c_step
                    best_h = h.copy()
                else:
                    # Try smaller delta
                    delta *= 0.7
        else:
            break
        
        # Occasionally try random perturbation
        if it % 10 == 5:
            rng = np.random.RandomState(it * 7 + n)
            noise = rng.randn(n) * 0.1
            h_rand = project_feasible(h + noise, target)
            c_rand = evaluate_c5(h_rand)
            if c_rand < best_c:
                best_c = c_rand
                best_h = h_rand.copy()
                h = h_rand.copy()
    
    return best_h, best_c

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    n = len(h)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def all_gradients(h):
    n = len(h)
    num_corr = 2 * n - 1
    G = np.zeros((num_corr, n))
    omh = 1.0 - h
    for k in range(num_corr):
        tau = k - (n - 1)
        if tau >= 0:
            G[k, tau:] += omh[:n - tau]
            G[k, :n - tau] -= h[tau:]
        else:
            G[k, :n + tau] += omh[-tau:]
            G[k, -tau:] -= h[:n + tau]
    return G

def lse_value(h, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = corr - corr.max()
    return float(np.log(np.sum(np.exp(beta * shifted))) / beta + corr.max())

def lse_gradient(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = beta * (corr - corr.max())
    weights = np.exp(shifted)
    weights /= weights.sum()
    G = all_gradients(h)
    return weights @ G

def projected_gradient_step(h, grad, target, lr):
    h_new = h - lr * grad
    return project_feasible(h_new, target)

def line_search(h, grad, target, lr_init, max_iters=20):
    best_h = h.copy()
    best_c = evaluate_c5(h)
    lr = lr_init
    for _ in range(max_iters):
        h_new = projected_gradient_step(h, grad, target, lr)
        c_new = evaluate_c5(h_new)
        if c_new < best_c:
            best_h = h_new
            best_c = c_new
        lr *= 0.5
        if lr < 1e-8:
            break
    return best_h, best_c
