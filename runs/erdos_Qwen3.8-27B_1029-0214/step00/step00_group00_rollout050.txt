    """Compute gradient of sum_k w_k * T_k w.r.t. h"""
    n_terms = len(terms_weights)
    grad = np.zeros(n)
    dx = 2.0 / n
    for k in range(n_terms):
        w = terms_weights[k]
        if w < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            # j ranges from 0 to n-1-s
            # d(d*T_S)/dh[i] = dx * [(1-h[i+s]) for i in [0, n-1-s] - h[i-s] for i in [s, n-1]]
            grad[:n-s] += w * dx * (1.0 - h[s:])
            grad[s:] += w * dx * (-h[:n-s])
        else:
            t = -s
            # j ranges from t to n-1
            # d/dh[i] = dx * [(1-h[i-t]) for i in [t, n-1] - h[i+t] for i in [0, n-1-t]]
            grad[t:] += w * dx * (1.0 - h[:n-t])
            grad[:n-t] += w * dx * (-h[t:])
    return grad

    return grad

def project_to_sum(h, target_sum):

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total_room = np.sum(room)

            h = h + scale * room
    else:
        total_mass = np.sum(h)
        if total_mass > 1e-12:
            scale = -diff / total_mass
            h = h - scale * h
    return np.clip(h, 0.0, 1.0)

    terms = compute_all_terms(h, n)
    m = np.max(terms)
    exp_shifted = np.exp(alpha * (terms - m))
    weights = exp_shifted / np.sum(exp_shifted)
    grad = compute_grad_weighted(weights, h, n)
    grad = grad - np.mean(grad)
    return grad

def sum_fun(x, ts):
    return np.sum(x) - ts

def sum_jac(x, ts):
    return np.ones(len(x))

def obj_violation(x, n, target_t, active):
    ts = n / 2.0
    h = project_to_sum(x, ts)
    terms = compute_all_terms(h, n)
    v = terms[active] - target_t
    v = np.maximum(v, 0.0)
    return np.sum(v * v)

def grad_violation(x, n, target_t, active):
    ts = n / 2.0
    h = project_to_sum(x, ts)
    terms = compute_all_terms(h, n)
    v = terms[active] - target_t
    pos = v > 0
    if not np.any(pos):
        return np.zeros(n)
    w_eff = 2.0 * v[pos]
    a_pos = active[pos]
    grad = np.zeros(n)
    dx = 2.0 / n
    for idx in range(len(a_pos)):
        w = w_eff[idx]
        k = a_pos[idx]
        s = k - (n - 1)
        if s >= 0:
            e = n - s
            grad[:e] += w * dx * (1.0 - h[s:])
            grad[s:] += w * dx * (-h[:e])
        else:
            t2 = -s
            e = n - t2
            grad[t2:] += w * dx * (1.0 - h[:e])
            grad[:e] += w * dx * (-h[t2:])
    grad = grad - np.mean(grad)
    return grad

def upsample(h, n_from, n_to):
    x_old = np.arange(n_from, dtype=float) / n_from
    x_new = np.arange(n_to, dtype=float) / n_to
    h_up = np.interp(x_new, x_old, h, period=1.0)
    return h_up

def local_perturbation(h, n, ts, rng, step):
    h_new = h.copy()
    mask = rng.random(n) < 0.3
    perturb = rng.normal(0, step, n) * mask
    h_new = project_to_sum(h_new + perturb, ts)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    rng = np.random.RandomState(seed)
    
    try:

        h0 = np.full(45, 0.5)

    alphas_p1 = [20, 50, 100, 200, 500, 1000, 2000, 5000]

        if time.time() - start_time > budget_s * 0.35:
            break
        bounds = [(0.0, 1.0)] * cur_n
        cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (cur_ts,)}]
        res = minimize(obj_smoothed, cur_h, method='SLSQP',
                       jac=grad_smoothed, args=(cur_n, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-14})
        cand = project_to_sum(res.x, cur_ts)
        c5 = compute_c5(cand, cur_n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand.copy()
            cur_h = cand.copy()
    
    for rnd in range(25):
        if time.time() - start_time > budget_s * 0.65:
            break
        h = project_to_sum(cur_h, cur_ts)

        c5_cur = float(np.max(terms))
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        
        sorted_idx = np.argsort(terms)[::-1]
        n_active = min(20, 2 * cur_n - 1)

        target_t = c5_cur - 0.00002
        if target_t < 0.01:

        bounds = [(0.0, 1.0)] * cur_n

                       jac=grad_violation, args=(cur_n, target_t, active),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-14})
        cand = project_to_sum(res.x, cur_ts)
        c5 = compute_c5(cand, cur_n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand.copy()
            cur_h = cand.copy()
    
    for target_n in [90, 180, 360]:
        if time.time() - start_time > budget_s * 0.55:
            break
        h_up = upsample(best_h, len(best_h), target_n)
        ts_new = target_n / 2.0
        h_up = project_to_sum(h_up, ts_new)
        cur_h = h_up.copy()
        cur_n = target_n
        cur_ts = ts_new
        
        for alpha in [50, 100, 300, 1000]:
            if time.time() - start_time > budget_s * 0.45:
                break
            bounds = [(0.0, 1.0)] * cur_n
            cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (cur_ts,)}]
            res = minimize(obj_smoothed, cur_h, method='SLSQP',
                           jac=grad_smoothed, args=(cur_n, alpha),
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-14})
            cand = project_to_sum(res.x, cur_ts)
            c5 = compute_c5(cand, cur_n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = cand.copy()
                cur_h = cand.copy()
        
        for rnd in range(5):
            if time.time() - start_time > budget_s * 0.4:

            terms = compute_all_terms(cur_h, cur_n)
            c5_cur = float(np.max(terms))
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = cur_h.copy()
            sorted_idx = np.argsort(terms)[::-1]

            target_t = c5_cur - 0.00001

            if c5 < best_c5:

        if time.time() - start_time > budget_s - 30:

        cur_n = len(h_pert)

            cand = project_to_sum(res.x, cur_ts)

    final_h = project_to_sum(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h, len(final_h))
    return final_h, final_c5, len(final_h)
