def eval_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n):
    """Ensure h is feasible: 0<=h<=1, sum(h)=n/2."""
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
        if not np.any(mask):
            return h
        frac = (1.0 - h)
        frac[~mask] = 0.0
        total = frac.sum()
        if total < diff:
            h[mask] = 1.0
        else:
            h += diff * frac / total
    else:
        mask = h > 1e-12
        if not np.any(mask):
            return h
        frac = h.copy()
        frac[~mask] = 0.0
        total = frac.sum()
        if total < -diff:
            h[mask] = 0.0
        else:
            h += diff * frac / total
    h = np.clip(h, 0.0, 1.0)
    return h

    """Return (func, jac) for SLSQP."""
    dx = 2.0 / n
    target = n / 2.0
    ncorr = 2 * n - 1
    
    def con_fun(x):
        h = x[:n]
        t = x[n]
        corr = np.correlate(h, 1.0 - h, mode="full")
        ineq = t - corr * dx  # each should be >= 0
        eq = np.sum(h) - target
        return np.append(ineq, eq)
    
    def con_jac(x):
        h = x[:n]
        t = x[n]
        jac = np.zeros((ncorr + 1, n + 1))

np.correlate with mode="full" doesn't wrap around. It just computes the cross-correlation over the valid overlap. So for a shift of k positions, the sum is over the overlapping region only.

So the integral is effectively:
For shift k (in units of dx): ∫₀^{2-kdx} h(x)(1-h(x+kdx)) dx ≈ dx * sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])

This is exactly what np.correlate(h, 1-h, mode="full") computes.

So there's no periodic wrapping. The function is defined on [0,2] and shifts that go outside are just not included.

1. Map each old sub-interval to the new grid
2. For each new sub-interval, determine what fraction of it is covered by old sub-intervals with value 1 (or whatever value)

Actually, the simplest correct approach is:

- New grid: points at y_j = j * 2/n_new for j = 0, ..., n_new-1, each representing interval [y_j, y_j + 2/n_new)
- h_new[j] = (1/(2/n_new)) * integral of h_old over [y_j, y_j + 2/n_new)

```python
def resample_h(h_old, n_old, n_new):
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    h_new = np.zeros(n_new)
    for j in range(n_new):
        y_start = j * dx_new
        y_end = (j + 1) * dx_new
        # Find overlap with old intervals
        # Old interval i covers [i*dx_old, (i+1)*dx_old)
        i_start = int(y_start // dx_old)
        i_end = int(np.ceil(y_end // dx_old))
        i_start = max(0, i_start)
        i_end = min(n_old, i_end)
        for i in range(i_start, i_end):
            overlap_start = max(y_start, i * dx_old)
            overlap_end = min(y_end, (i + 1) * dx_old)
            h_new[j] += h_old[i] * (overlap_end - overlap_start)
        h_new[j] /= dx_new
    return h_new
```
