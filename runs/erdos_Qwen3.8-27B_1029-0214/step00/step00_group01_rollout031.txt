def _raise_if_deadline():
    global _G_EVALS
    if time.time() > _G_DEADLINE:
        raise RuntimeError("timeout")
    _G_EVALS += 1
    if _G_EVALS > _G_EVAL_LIMIT:
        raise RuntimeError("evaluation limit")

def _value_one_tau(h, tau, n):
    tau = int(tau)
    if tau >= 0:
        L = n - tau
        if L <= 0:
            return 0.0
        return float(np.dot(h[:L], 1.0 - h[tau:tau + L]))
    r = -tau
    L = n - r
    if L <= 0:
        return 0.0
    return float(np.dot(h[r:r + L], 1.0 - h[:L]))

def derive_tau_map(n):
    m = 2 * int(n) - 1
    rng = np.random.default_rng(1234567)
    h = 0.25 + 0.5 * rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    y = np.correlate(h, 1.0 - h, mode="full")
    yA = np.empty(m, dtype=float)
    yB = np.empty(m, dtype=float)
    for j in range(m):
        yA[j] = _value_one_tau(h, j - (n - 1), n)
        yB[j] = _value_one_tau(h, (n - 1) - j, n)
    errA = float(np.linalg.norm(y - yA))
    errB = float(np.linalg.norm(y - yB))
    tol = 1e-9 * max(1.0, float(n))
    if errA < tol and errA <= errB:
        return (np.arange(m, dtype=int) - (n - 1)), True
    if errB < tol and errB < errA:
        return ((n - 1) - np.arange(m, dtype=int)).astype(int), True
    if errA < 1e-5 and errA <= errB:
        return (np.arange(m, dtype=int) - (n - 1)), True
    if errB < 1e-5 and errB < errA:
        return ((n - 1) - np.arange(m, dtype=int)).astype(int), True
    return None, False

def exact_c5(h):

def project_box_simplex(z, target):
    z = np.asarray(z, dtype=float).ravel().copy()
    n = z.size
    if n == 0:
        return z
    if not np.isfinite(target):
        target = 0.5 * float(n)
    if not np.all(np.isfinite(z)):

    if target <= 0.0:
        return np.zeros(n, dtype=float)

        return np.ones(n, dtype=float)
    y = np.clip(z, 0.0, 1.0)

def select_rows(h, n, c_ref, active_tol, max_cons):
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.size
    if not np.isfinite(c_ref):
        c_ref = float(np.max(corr)) if m > 0 else 0.0
    if m <= max_cons:
        return np.arange(m, dtype=int)
    order = np.argsort(corr)[::-1]
    threshold = max(c_ref, 1e-6) - active_tol * max(c_ref, 0.05)
    active = order[corr[order] >= threshold]
    top_count = min(max_cons, m)
    top = order[:top_count]
    rows = np.unique(np.concatenate((active, top)))
    if rows.size > max_cons:
        rows = rows[np.argsort(corr[rows])[::-1]][:max_cons]
    if not np.any(rows == (n - 1)):
        if rows.size >= max_cons:
            drop_order = np.argsort(corr[rows])
            drop = int(drop_order[0])
            rows = np.delete(rows, drop)
        rows = np.append(rows, n - 1)
    return rows.astype(int)

def _epi_ineq_jac(x):
    _raise_if_deadline()
    h = x[:-1]
    n = _G_N
    dx = _G_DX
    rows = _G_ROWS
    m = rows.size
    J = np.zeros((m, n + 1), dtype=float)
    for ii in range(m):
        tau = int(_G_TAU_MAP[int(rows[ii])])
        J[ii, n] = 1.0
        if tau >= 0:
            L = n - tau
            if L > 0:
                J[ii, 0:L] -= dx * (1.0 - h[tau:tau + L])
                J[ii, tau:tau + L] += dx * h[0:L]
        else:
            r = -tau
            L = n - r
            if L > 0:
                J[ii, r:r + L] -= dx * (1.0 - h[:L])
                J[ii, 0:L] += dx * h[r:r + L]
    return J

def solve_epigraph(h0, n, tau_map, rows, deadline, maxiter, ftol):

    if tau_map is None:
        return None
    rows = np.asarray(rows, dtype=int).ravel()
    if rows.size == 0:
        return None
    rows = rows[(rows >= 0) & (rows < 2 * n - 1)]
    if rows.size == 0:
        return None
    target = 0.5 * float(n)
    h = project_box_simplex(np.asarray(h0, dtype=float), target)
    h = repair_sum(h, n)
    c0 = exact_c5(h)
    if not np.isfinite(c0) or c0 >= 1.0 - 1e-12:
        return None
    dx = 2.0 / float(n)
    _G_N = n
    _G_DX = dx
    _G_TARGET = target
    _G_ROWS = rows
    _G_TAU_MAP = tau_map
    _G_DEADLINE = float(deadline)
    _G_EVALS = 0
    _G_EVAL_LIMIT = max(3000, 30 * (n + int(rows.size)))
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h
    slack = max(1e-8, 1e-9 * max(1.0, float(n)))
    t0 = c0 + slack
    t_upper = 1.0 + 1e-7
    if t0 >= t_upper:
        t0 = t_upper - 1e-9
    if t0 < 0.0:
        t0 = 0.0
    x0[n] = t0
    bounds = [(0.0, 1.0)] * n
    bounds.append((0.0, t_upper))
    cons = (
        {"type": "eq", "fun": _epi_eq, "jac": _epi_eq_jac},
        {"type": "ineq", "fun": _epi_ineq, "jac": _epi_ineq_jac},
    )
    try:
        res = minimize(
            _epi_obj,
            x0,
            jac=_epi_jac,
            bounds=bounds,
            constraints=cons,

def _tau_grad(h, tau, n, dx):
    g = np.zeros(n, dtype=float)
    if tau >= 0:
        L = n - tau
        if L > 0:
            g[0:L] += dx * (1.0 - h[tau:tau + L])
            g[tau:tau + L] -= dx * h[0:L]
    else:
        r = -tau
        L = n - r
        if L > 0:
            g[r:r + L] += dx * (1.0 - h[:L])
            g[0:L] -= dx * h[r:r + L]
    return g

        return True, best_h, c3
    return False, None, current_best


def select_rows(h, n, c_ref, active_tol, max_cons):
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.size
    if not np.isfinite(c_ref):
        c_ref = float(np.max(corr)) if m > 0 else 0.0
    if m <= max_cons:
        return np.arange(m, dtype=int)
    order = np.argsort(corr)[::-1]
    threshold = max(c_ref, 1e-6) - active_tol * max(c_ref, 0.05)
    active = order[corr[order] >= threshold]
    top_count = min(max_cons, m)
    top = order[:top_count]
    rows = np.unique(np.concatenate((active, top)))
    if rows.size > max_cons:
        rows = rows[np.argsort(corr[rows])[::-1]][:max_cons]
    if not np.any(rows == (n - 1)):
        if rows.size >= max_cons:
            drop_order = np.argsort(corr[rows])
            drop = int(drop_order[0])
            rows = np.delete(rows, drop)
        rows = np.append(rows, n - 1)
    return rows.astype(int)


def _epi_obj(x):
    _raise_if_deadline()
    return float(x[-1])


def _epi_jac(x):
    _raise_if_deadline()
    g = np.zeros(x.size, dtype=float)
    g[-1] = 1.0
    return g


def _epi_eq(x):
    _raise_if_deadline()
    return np.array([float(np.sum(x[:-1]) - _G_TARGET)], dtype=float)


def _epi_eq_jac(x):
    _raise_if_deadline()
    g = np.zeros((1, x.size), dtype=float)
    g[0, :-1] = 1.0
    return g


def _epi_ineq(x):
    _raise_if_deadline()
    h = x[:-1]
    n = _G_N
    dx = _G_DX
    rows = _G_ROWS
    vals = np.empty(rows.size, dtype=float)
    for ii in range(rows.size):
        tau = int(_G_TAU_MAP[int(rows[ii])])
        if tau >= 0:
            L = n - tau
            if L > 0:
                vals[ii] = dx * float(np.dot(h[:L], 1.0 - h[tau:tau + L]))
            else:
                vals[ii] = 0.0
        else:
            r = -tau
            L = n - r
            if L > 0:
                vals[ii] = dx * float(np.dot(h[r:r + L], 1.0 - h[:L]))
            else:
                vals[ii] = 0.0
    return x[-1] - vals


def _epi_ineq_jac(x):
    _raise_if_deadline()
    h = x[:-1]
    n = _G_N
    dx = _G_DX
    rows = _G_ROWS
    m = rows.size
    J = np.zeros((m, n + 1), dtype=float)
    for ii in range(m):
        tau = int(_G_TAU_MAP[int(rows[ii])])
        J[ii, n] = 1.0
        if tau >= 0:
            L = n - tau
            if L > 0:
                J[ii, 0:L] -= dx * (1.0 - h[tau:tau + L])
                J[ii, tau:tau + L] += dx * h[0:L]
        else:
            r = -tau
            L = n - r
            if L > 0:
                J[ii, r:r + L] -= dx * (1.0 - h[:L])
                J[ii, 0:L] += dx * h[r:r + L]
    return J


def solve_epigraph(h0, n, tau_map, rows, deadline, maxiter, ftol):
    global _G_N, _G_DX, _G_TARGET, _G_ROWS, _G_TAU_MAP, _G_DEADLINE, _G_EVALS, _G_EVAL_LIMIT
    if tau_map is None:
        return None
    rows = np.asarray(rows, dtype=int).ravel()
    if rows.size == 0:
        return None
    rows = rows[(rows >= 0) & (rows < 2 * n - 1)]
    if rows.size == 0:
        return None
    target = 0.5 * float(n)
    h = project_box_simplex(np.asarray(h0, dtype=float), target)
    h = repair_sum(h, n)
    c0 = exact_c5(h)
    if not np.isfinite(c0) or c0 >= 1.0 - 1e-12:
        return None
    dx = 2.0 / float(n)
    _G_N = n
    _G_DX = dx
    _G_TARGET = target
    _G_ROWS = rows
    _G_TAU_MAP = tau_map
    _G_DEADLINE = float(deadline)
    _G_EVALS = 0
    _G_EVAL_LIMIT = max(3000, 30 * (n + int(rows.size)))
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h
    slack = max(1e-8, 1e-9 * max(1.0, float(n)))
    t0 = c0 + slack
    t_upper = 1.0 + 1e-7
    if t0 >= t_upper:
        t0 = t_upper - 1e-9
    if t0 < 0.0:
        t0 = 0.0
    x0[n] = t0
    bounds = [(0.0, 1.0)] * n
    bounds.append((0.0, t_upper))
    cons = (
        {"type": "eq", "fun": _epi_eq, "jac": _epi_eq_jac},
        {"type": "ineq", "fun": _epi_ineq, "jac": _epi_ineq_jac},
    )
    try:
        res = minimize(
            _epi_obj,
            x0,
            jac=_epi_jac,
            bounds=bounds,
            constraints=cons,
            method="SLSQP",
            options={"maxiter": int(maxiter), "ftol": float(ftol)},
        )
    except Exception:
        return None
    if res is None or not hasattr(res, "x") or res.x is None:
        return None
    x = np.asarray(res.x, dtype=float).ravel()
    if x.size != n + 1:
        return None
    if not np.all(np.isfinite(x)):
        return None
    return x[:-1].copy()


def transfer_to_resolution(src, n_src, n_tgt):
    src = np.asarray(src, dtype=float).ravel()
    if n_src == n_tgt:
        return project_box_simplex(src, 0.5 * float(n_tgt))
    if n_src > 0 and n_tgt % n_src == 0:
        k = n_tgt // n_src
        return project_box_simplex(np.repeat(src, k), 0.5 * float(n_tgt))
    x_old = (np.arange(n_src) + 0.5) * (2.0 / float(n_src))
    x_new = (np.arange(n_tgt) + 0.5) * (2.0 / float(n_tgt))
    z = np.interp(x_new, x_old, src)
    return project_box_simplex(z, 0.5 * float(n_tgt))


def obtain_parent():
    parent = None
    try:
        parent = initial_h_values
    except Exception:
        parent = None
    if parent is None:
        return np.full(64, 0.5, dtype=float)
    h = np.asarray(parent, dtype=float).ravel()
    if h.size < 8:
        h = np.full(64, 0.5, dtype=float)
    if not np.all(np.isfinite(h)):
        h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(h.size)
    if abs(float(np.sum(h)) - target) > 1e-6:
        h = project_box_simplex(h, target)
    else:
        h = repair_sum(h, h.size)
    return h


def rank_quantize(h, n, target):
    h = np.asarray(h, dtype=float).ravel()
    target = 0.5 * float(n)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)
    k = int(np.floor(target + 1e-12))
    if k > n:
        k = n
    rem = target - float(k)
    q = np.zeros(n, dtype=float)
    order = np.argsort(h)[::-1]
    for i in range(k):
        q[int(order[i])] = 1.0
    if rem > 1e-12 and k < n:
        q[int(order[k])] = rem
    return repair_sum(q, n)


def perturb_h(h, rng, scale):
    z = h + scale * rng.standard_normal(h.size)
    return project_box_simplex(z, 0.5 * float(h.size))


def _tau_grad(h, tau, n, dx):
    g = np.zeros(n, dtype=float)
    if tau >= 0:
        L = n - tau
        if L > 0:
            g[0:L] += dx * (1.0 - h[tau:tau + L])
            g[tau:tau + L] -= dx * h[0:L]
    else:
        r = -tau
        L = n - r
        if L > 0:
            g[r:r + L] += dx * (1.0 - h[:L])
            g[0:L] -= dx * h[r:r + L]
    return g


def active_descent_step(h, c, n, tau_map, deadline, max_rows=3, steps=(0.4, 0.2, 0.1, 0.05)):
    best_h = np.asarray(h, dtype=float).copy()
    best_c = float(c)
    if tau_map is None or time.time() >= deadline:
        return best_h, best_c
    dx = 2.0 / float(n)
    corr = np.correlate(best_h, 1.0 - best_h, mode="full") * dx
    if corr.size == 0:
        return best_h, best_c
    top = np.argsort(corr)[::-1][:max_rows]
    for row in top:
        if time.time() >= deadline:
            break
        tau = int(tau_map[int(row)])
        g = _tau_grad(best_h, tau, n, dx)
        tangent = -g + float(np.mean(g))
        for s in steps:
            cand = best_h + (s * (-g))
            ok, bh, bc = try_candidate(cand, n, best_c, deadline)
            if ok:
                best_h, best_c = bh, bc
            cand = best_h + (s * tangent)
            ok, bh, bc = try_candidate(cand, n, best_c, deadline)
            if ok:
                best_h, best_c = bh, bc
    return best_h, best_c


def apply_deterministic_improvements(h, c, n, deadline):
    best_h = np.asarray(h, dtype=float).copy()
    best_c = float(c)
    if time.time() >= deadline:
        return best_h, best_c
    flip = best_h[::-1]
    cand = flip
    ok, bh, bc = try_candidate(cand, n, best_c, deadline)
    if ok:
        best_h, best_c = bh, bc
    cand = 0.5 * (best_h + flip)
    ok, bh, bc = try_candidate(cand, n, best_c, deadline)
    if ok:
        best_h, best_c = bh, bc
    cand = 0.5 * (best_h + 1.0 - flip)
    ok, bh, bc = try_candidate(cand, n, best_c, deadline)
    if ok:
        best_h, best_c = bh, bc
    cand = 1.0 - best_h
    ok, bh, bc = try_candidate(cand, n, best_c, deadline)
    if ok:
        best_h, best_c = bh, bc
    cand = 1.0 - flip
    ok, bh, bc = try_candidate(cand, n, best_c, deadline)
    if ok:
        best_h, best_c = bh, bc
    smooth = 0.5 * best_h + 0.25 * (np.roll(best_h, 1) + np.roll(best_h, -1))
    ok, bh, bc = try_candidate(smooth, n, best_c, deadline)
    if ok:
        best_h, best_c = bh, bc
    cand = rank_quantize(best_h, n, 0.5 * float(n))
    ok, bh, bc = try_candidate(cand, n, best_c, deadline)
    if ok:
        best_h, best_c = bh, bc
    for k in (1, 2, 3, 4, 5):
        if time.time() >= deadline:
            break
        r = np.roll(best_h, k)
        ok, bh, bc = try_candidate(r, n, best_c, deadline)
        if ok:
            best_h, best_c = bh, bc
        cand = 0.5 * (best_h + 1.0 - r)
        ok, bh, bc = try_candidate(cand, n, best_c, deadline)
        if ok:
            best_h, best_c = bh, bc
    return best_h, best_c


def improve_local(h0, n, c0, deadline, rng):
    target = 0.5 * float(n)
    h = project_box_simplex(np.asarray(h0, dtype=float), target)
    h = repair_sum(h, n)
    c = exact_c5(h)
    if not np.isfinite(c):
        return h, c
    tau_map, ok_grad = derive_tau_map(n)
    attempts = 0
    max_attempts = 50
    no_improve = 0
    tol_seq = (1e-3, 7e-4, 5e-4, 3.5e-4, 2.5e-4, 1.8e-4, 1.2e-4, 8e-5, 5e-5, 3e-5)
    if n <= 90:
        max_cons = 2 * n - 1
    elif n <= 150:
        max_cons = 200
    else:
        max_cons = 160
    max_cons_cur = max_cons
    if n <= 90:
        maxiter_seq = (90, 75, 65, 55, 50, 45, 40, 35, 30, 25)
    elif n <= 150:
        maxiter_seq = (70, 60, 55, 50, 45, 40, 35, 30, 25, 20)
    else:
        maxiter_seq = (55, 50, 45, 40, 35, 30, 25, 20)
    call_time = 35.0 if n <= 90 else (45.0 if n <= 150 else 60.0)
    ftol = 1e-8
    scale_seq = (0.10, 0.05, 0.025, 0.012, 0.006, 0.003)
    while time.time() < deadline - 5.0 and attempts < max_attempts:
        c_det_before = c
        h, c = apply_deterministic_improvements(h, c, n, deadline)
        if c < c_det_before - 1e-12:
            no_improve = 0
        start = h
        if no_improve >= 2:
            scale = scale_seq[min(no_improve // 2, len(scale_seq) - 1)]
            hp = perturb_h(h, rng, scale)
            cp = exact_c5(hp)
            if np.isfinite(cp) and cp < c - 1e-12:
                h = hp
                c = cp
                no_improve = 0
            elif np.isfinite(cp) and cp < c + 0.05:
                start = hp
        if not ok_grad:
            if no_improve >= 6:
                break
            h, c = active_descent_step(h, c, n, tau_map, deadline)
            h, c = apply_deterministic_improvements(h, c, n, deadline)
            no_improve += 1
            attempts += 1
            continue
        c_start = exact_c5(start)
        tol = tol_seq[min(attempts, len(tol_seq) - 1)]
        if no_improve >= 4 and max_cons_cur > 40:
            max_cons_cur = max(40, int(max_cons_cur * 0.8))
        rows = select_rows(start, n, c_start, tol, max_cons_cur)
        maxiter = maxiter_seq[min(attempts, len(maxiter_seq) - 1)]
        solve_deadline = min(deadline, time.time() + call_time)
        h_try = solve_epigraph(start, n, tau_map, rows, solve_deadline, maxiter, ftol)
        if h_try is not None:
            ok, h_new, c_new = try_candidate(h_try, n, c, deadline)
            if ok:
                h = h_new
                c = c_new
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve >= 2:
                h, c = active_descent_step(h, c, n, tau_map, deadline)
        attempts += 1
    h, c = apply_deterministic_improvements(h, c, n, deadline)
    return h, c


def make_stage_list(n_base, budget):
    base = int(n_base) if n_base >= 8 else 64
    if budget < 120:
        return [base]
    multipliers = (1.0, 2.0, 2.032, 3.0, 3.032, 4.0)
    max_n = 320 if budget > 500 else 240
    stages = []
    for mm in multipliers:
        nn = int(round(base * mm))
        if 8 <= nn <= max_n:
            found = False
            for xx in stages:
                if xx == nn:
                    found = True
                    break
            if not found:
                stages.append(nn)
    remaining = []
    for ss in stages:
        if ss != base:
            remaining.append(ss)
    sorted_rem = []
    for ss in remaining:
        pos = 0
        for i in range(len(sorted_rem)):
            d1 = abs(ss - base)
            d2 = abs(sorted_rem[i] - base)
            if d1 < d2 or (d1 == d2 and ss < sorted_rem[i]):
                pos = i
                break
        sorted_rem.insert(pos, ss)
    return [base] + sorted_rem


def make_stage_ends(t0, deadline, stages):
    total = deadline - t0
    if total <= 0.0:
        ends = []
        for _ in stages:
            ends.append(deadline)
        return ends
    weights = (1.8, 1.4, 1.2, 1.0, 0.9, 0.8)
    used = []
    for i in range(len(stages)):
        if i < len(weights):
            used.append(weights[i])
        else:
            used.append(0.5)
    total_w = 0.0
    for ww in used:
        total_w += ww
    if total_w <= 0.0:
        ends = []
        for _ in stages:
            ends.append(deadline)
        return ends
    acc = 0.0
    ends = []
    for ww in used:
        acc += ww
        end = t0 + total * (acc / total_w)
        if end > deadline:
            end = deadline
        if ends and end < ends[-1]:
            end = ends[-1]
        ends.append(end)
    return ends


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1100.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1100.0
    reserve = max(20.0, 0.06 * budget)
    deadline = t0 + max(0.0, budget - reserve)
    if deadline <= t0 + 5.0:
        deadline = t0 + max(1e-3, 0.98 * budget)
    rng = np.random.default_rng(seed)

    h_best = obtain_parent()
    n_best = h_best.size
    c_best = exact_c5(h_best)
    if not np.isfinite(c_best):
        h_best = np.full(64, 0.5, dtype=float)
        c_best = exact_c5(h_best)
        n_best = h_best.size

    h_best, c_best = apply_deterministic_improvements(h_best, c_best, n_best, deadline)

    stages = make_stage_list(n_best, budget)
    stage_ends = make_stage_ends(t0, deadline, stages)
    stage_best = {}
    stage_best[n_best] = (h_best.copy(), c_best)

    for idx in range(len(stages)):
        nstage = stages[idx]
        stage_deadline = stage_ends[idx]
        if time.time() >= stage_deadline - 2.0:
            continue
        if nstage == n_best:
            src_h = stage_best[n_best][0]
            src_c = stage_best[n_best][1]
        else:
            best_src_h = None
            best_src_c = None
            for sn, pair in stage_best.items():
                if sn < nstage:
                    cc = pair[1]
                    hh = pair[0]
                    if best_src_c is None or cc < best_src_c:
                        best_src_c = cc
                        best_src_h = hh
            if best_src_h is None:
                best_src_h = h_best
                best_src_c = c_best
            src_h = best_src_h
            src_c = best_src_c
        start_h = np.asarray(src_h, dtype=float).copy()
        if start_h.size != nstage:
            start_h = transfer_to_resolution(start_h, start_h.size, nstage)
        start_c = exact_c5(start_h)
        if not np.isfinite(start_c):
            start_h = transfer_to_resolution(h_best, h_best.size, nstage)
            start_c = exact_c5(start_h)
        start_h, start_c = apply_deterministic_improvements(start_h, start_c, nstage, stage_deadline)
        if start_c < c_best - 1e-12:
            c_best = start_c
            h_best = start_h.copy()
        if time.time() < stage_deadline - 10.0:
            hloc, cloc = improve_local(start_h, nstage, start_c, stage_deadline, rng)
        else:
            hloc, cloc = start_h, start_c
        if np.isfinite(cloc) and cloc < c_best - 1e-12:
            c_best = cloc
            h_best = hloc.copy()
        if nstage not in stage_best or cloc < stage_best[nstage][1]:
            stage_best[nstage] = (hloc.copy(), cloc)
        if time.time() >= deadline - 15.0:
            break

    best_final = None
    best_final_c = float("inf")
    for hh in stage_best.values():
        ff = np.asarray(hh[0], dtype=float).copy()
        nn = ff.size
        if nn < 1:
            continue
        ff = force_sum_exact(ff, nn)
        cc = exact_c5(ff)
        if np.isfinite(cc) and cc < best_final_c:
            best_final_c = cc
            best_final = ff
    ff = force_sum_exact(h_best, h_best.size)
    cc = exact_c5(ff)
    if np.isfinite(cc) and cc < best_final_c:
        best_final_c = cc
        best_final = ff

    if best_final is None or not np.isfinite(best_final_c):
        best_final = np.full(64, 0.5, dtype=float)
        best_final_c = exact_c5(best_final)

    final_c = exact_c5(best_final)
    if not np.isfinite(final_c):
        final_h = np.full(64, 0.5, dtype=float)
        final_c = exact_c5(final_h)
    else:
        final_h = best_final
    return final_h, float(final_c), int(final_h.size)
