def evaluate_c5(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    n = len(h)

    corr = np.correlate(h, 1-h, mode='full')

    # A[i-s] valid when 0 <= i-s < N, i.e., i-N < s <= i

    # B[j+s] valid when 0 <= j+s < N, i.e., -j <= s < N-j
