def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def get_correlation(h):
    """Get the full correlation vector (before scaling by dx)."""
    return np.correlate(h, 1-h, mode='full')

def project_feasible(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    # Bisection
    lo, hi = -max(h), 1 - min(h)
    for _ in range(80):
        mid = (lo + hi) / 2
        s_mid = np.sum(np.clip(h + mid, 0, 1))
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2, 0, 1)
    # Fix residual
    diff = target - np.sum(h)
    if diff > 1e-10:
        # Add to coordinates with most room
        order = np.argsort(1 - h)
        for idx in order:
            add = min(diff, 1 - h[idx])
            h[idx] += add
            diff -= add
            if diff <= 1e-10:
                break
    elif diff < -1e-10:

        for idx in order:
            rem = min(-diff, h[idx])
            h[idx] -= rem
            diff += rem
            if diff >= -1e-10:
                break
    return h

    rng = np.random.RandomState(seed)
    n = len(h)

    G0 = get_correlation(h)
    current_max = np.max(G0)
    
    for it in range(n_iterations):
        best_improvement = 0
        best_d = 0
        best_pair = None

            i = rng.randint(0, n)
            j = rng.randint(0, n)

                if 0 <= i-k < n: val -= h[i-k]
                if 0 <= i+k < n: val -= h[i+k]

            max_vals = G_new.max(axis=1)

def project_feasible(h):
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Binary search for uniform shift
    lo, hi = -1.0, 1.0
    if diff > 0:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix tiny residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-9:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
            else:
                rem = min(-residual, h[idx])
                h[idx] -= rem
                residual += rem
            if abs(residual) < 1e-12:
                break
    return h

def safe_index(arr, idx):
    """Vectorized safe indexing."""
    n = len(arr)
    result = np.zeros(len(idx), dtype=float)
    mask = (idx >= 0) & (idx < n)
    result[mask] = arr[idx[mask]]
    return result

def pair_exchange_step(h, G0, rng, n_trials=200, n_d_samples=30):
    """Try n_trials random pairs, return best update."""
    n = len(h)
    current_max = np.max(G0)
    best_val = current_max
    best_i = -1
    best_j = -1
    best_d = 0.0
    
    for _ in range(n_trials):
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:
            continue
        d_min = max(-h[i], h[j] - 1.0)
        d_max = min(1.0 - h[i], h[j])
        if d_max - d_min < 1e-10:
            continue
        
        r = j - i
        ks = np.arange(-(n-1), n)
        
        # Compute A vector
        A = (safe_index(h, j - ks) - safe_index(h, i - ks) 
             - safe_index(h, i + ks) + safe_index(h, j + ks))
        
        # Compute B vector
        B = np.zeros(2*n - 1)
        B[n-1] = 2.0
        B_pos = n - 1 + r
        B_neg = n - 1 - r
        if 0 <= B_pos < 2*n - 1:
            B[B_pos] = -1.0
        if 0 <= B_neg < 2*n - 1:
            B[B_neg] = -1.0
        
        # Sample d values
        d_vals = np.linspace(d_min, d_max, n_d_samples)
        # Compute max G for each d
        # G_new[d_idx, k] = G0[k] + d*d_vals * A[k] - d^2 * B[k]
        G_new = G0[np.newaxis, :] + d_vals[:, np.newaxis] * A[np.newaxis, :] - (d_vals**2)[:, np.newaxis] * B[np.newaxis, :]
        max_vals = G_new.max(axis=1)
        min_idx = np.argmin(max_vals)
        
        if max_vals[min_idx] < best_val:
            best_val = max_vals[min_idx]
            best_i = i
            best_j = j
            best_d = d_vals[min_idx]
    
    if best_i >= 0:
        h[best_i] += best_d
        h[best_j] -= best_d
        G0 = np.correlate(h, 1.0 - h, mode='full')
    
    return h, G0, best_val, (best_i, best_j, best_d)
