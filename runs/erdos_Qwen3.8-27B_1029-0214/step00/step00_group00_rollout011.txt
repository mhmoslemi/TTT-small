from scipy.optimize import minimize

# --- Objective and gradient ---

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_active = int(np.argmax(corr))
    k = k_active - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:n] += (1.0 - h[0:n-k])
        grad[0:(n-k)] -= h[k:n]
    else:
        e = -k
        grad[0:(n-e)] += (1.0 - h[e:n])
        grad[e:n] -= h[0:(n-e)]
    return dx * grad

# --- Feasibility repair ---

def repair_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            s = np.sum(room)
            if s < 1e-15:
                break
            h = np.clip(h + diff * room / s, 0.0, 1.0)
        else:
            room = h
            s = np.sum(room)
            if s < 1e-15:
