def evaluate_c5(h):

def is_feasible(h):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -1e-8 or np.max(h) > 1.0 + 1e-8:
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-6 * n:
        return False
    return True


def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.array(h, dtype=np.float64).copy()

            break
        if diff > 0:

            cnt = int(np.sum(mask))

            step = min(diff / cnt, (1.0 - h[mask]).min() * 0.999)
            h[mask] += step

def smooth_obj_only(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c = np.max(corr)
    if not np.isfinite(c):
        return 1e10, np.zeros(n)
    shifted = tau * (corr - c)
    mx = np.max(shifted)
    if mx > 500:
        shifted = shifted - mx

    s = np.sum(exp_vals)
    if s <= 0 or not np.isfinite(s):
        return 1e10, np.zeros(n)
    weights = exp_vals / s
    obj = c + np.log(s) / tau
    grad = np.zeros(n)
    offsets = np.arange(2 * n - 1) - (n - 1)
    for m in range(2 * n - 1):

                if val < best_val - 1e-12:
                    best_h = result.copy()
                    best_val = val
        if time_left() > 50:
            h_loc, val_loc = local_exact_search(best_h, rng, n_steps=500, base_step=0.03)
            if val_loc < best_val - 1e-13:
                best_h = h_loc.copy()
                best_val = val_loc

    # Phase 4: Multiple restarts with structured starts
    if time_left() > 80:
        n_try = len(best_h)
        for restart in range(4):
            if time_left() < 40:
                break
            if restart == 0:
                h_r = best_h + rng.normal(0, 0.08, n_try)
            elif restart == 1:
                h_r = make_wave_h(n_try, seed=seed + restart * 100)
            elif restart == 2:
                h_r = best_h + rng.normal(0, 0.15, n_try)
            else:
                h_r = project_feasible(np.sort(best_h))
            h_r = project_feasible(h_r)
            for tau in [30, 80, 150]:
                if time_left() < 20:
                    break
                result = run_slsqp(h_r, tau, maxiter=100)
                if result is not None and is_feasible(result):
                    val = evaluate_c5(result)
                    if val < best_val - 1e-12:
                        best_h = result.copy()
                        best_val = val
                        h_r = best_h.copy()

    # Phase 5: Final local polish with smaller steps
    if time_left() > 20:
        h_loc, val_loc = local_exact_search(best_h, rng, n_steps=800, base_step=0.015)
        if val_loc < best_val - 1e-13:
            best_h = h_loc.copy()
            best_val = val_loc

    # Final: project and validate
    best_h = project_feasible(best_h)
    best_val = evaluate_c5(best_h)
    n_points = len(best_h)

    return (best_h, best_val, n_points)
