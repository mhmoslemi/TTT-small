def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(z, n):
    target = n / 2.0
    z = np.asarray(z, dtype=np.float64).copy()
    bad = ~np.isfinite(z)
    if np.any(bad):
        z[bad] = 0.5
    lo = float(np.min(z)) - 2.0
    hi = float(np.max(z)) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)

        cnt = int(np.sum(interior))
        if cnt > 0:
            result[interior] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result
