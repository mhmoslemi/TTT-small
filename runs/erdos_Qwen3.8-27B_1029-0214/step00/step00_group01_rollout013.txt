def compute_logsumexp(c_vals, beta):
    """Stable log-sum-exp of beta*c_vals."""
    max_val = np.max(c_vals)
    if not np.isfinite(max_val):
        return np.inf, np.zeros_like(c_vals)
    shifted = beta * (c_vals - max_val)
    # For very negative values, exp underflows to 0 which is fine
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    if sum_exp == 0:
        # All underflowed, use uniform
        weights = np.ones_like(c_vals) / len(c_vals)
        lse = max_val
    else:
        weights = exp_vals / sum_exp
        lse = max_val + np.log(sum_exp) / beta
    return lse, weights

def sigmoid_fn(x):
    x = np.clip(x, -40.0, 40.0)
    out = np.empty_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    exp_x = np.exp(x[~pos])
    out[~pos] = exp_x / (1.0 + exp_x)
    return out

def solve_lambda_fn(y, n):
    target = n * 0.5
    lo = -40.0
    hi = 40.0
    # Ensure bracket
    s_lo = np.sum(sigmoid_fn(y + lo))
    s_hi = np.sum(sigmoid_fn(y + hi))
    if s_lo >= target:
        return lo
    if s_hi <= target:
        return hi
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s_mid = np.sum(sigmoid_fn(y + mid))
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)

def logsumexp_fn(c_vals, beta):
    max_c = np.max(c_vals)
    if not np.isfinite(max_c):
        return max_c, np.ones(len(c_vals)) / len(c_vals)
    shifted = beta * (c_vals - max_c)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    if s < 1e-300:
        weights = np.ones(len(c_vals)) / len(c_vals)
    else:
        weights = exp_vals / s
    val = max_c + np.log(max(s, 1e-300)) / beta
    return val, weights

def grad_h_fn(h, weights, dx, n):
    G = np.zeros(n)
    nm1 = n - 1
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-14:
            continue
        lag = idx - nm1
        if lag >= 0:

            if L <= 0:

                G[:lag] -= w * dx * h[lag:]
        else:
            r = -lag

            if L <= 0:
                continue
            G[:L] += w * dx * (1.0 - h[r:])
            if r > 0:

    return G

def project_feasible_fn(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Waterfill
    if diff > 0:
        # Need to increase sum by diff
        # Add to elements that are < 1
        mask = h < 1.0
        if not np.any(mask):
            return h
        capacity = np.sum(1.0 - h[mask])
        if capacity <= 0:
            return h
        frac = diff / capacity
        if frac >= 1.0:
            h[mask] = 1.0
            # Still need more? shouldn't happen if target is feasible
        else:
            h[mask] += frac * (1.0 - h[mask])

        mask = h > 0.0
        if not np.any(mask):
            return h
        h[mask] += diff  # diff is negative

            s = np.sum(h)

import numpy as np
from scipy.optimize import minimize
import time


def sigmoid_fn(x):
    x = np.clip(x, -35.0, 35.0)
    out = np.empty_like(x)
    pos = x >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    ex = np.exp(x[~pos])
    out[~pos] = ex / (1.0 + ex)
    return out


def solve_lambda_fn(y, n):
    target = n * 0.5
    lo = -35.0
    hi = 35.0
    if np.sum(sigmoid_fn(y + lo)) >= target:
        return lo
    if np.sum(sigmoid_fn(y + hi)) <= target:
        return hi
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(sigmoid_fn(y + mid)) < target:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)


def make_h_fn(y, n):
    lam = solve_lambda_fn(y, n)
    return sigmoid_fn(y + lam)


def compute_c_fn(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx


def logsumexp_fn(c_vals, beta):
    max_c = np.max(c_vals)
    if not np.isfinite(max_c):
        return max_c, np.ones(len(c_vals)) / len(c_vals)
    shifted = beta * (c_vals - max_c)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    if s < 1e-300:
        weights = np.ones(len(c_vals)) / len(c_vals)
    else:
        weights = exp_vals / s
    val = max_c + np.log(max(s, 1e-300)) / beta
    return val, weights


def grad_h_fn(h, weights, dx, n):
    G = np.zeros(n)
    nm1 = n - 1
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-14:
            continue
        lag = idx - nm1
        if lag >= 0:
            L = n - lag
            if L <= 0:
                continue
            G[lag:] += w * dx * (1.0 - h[:L])
            if lag > 0:
                G[:lag] -= w * dx * h[lag:]
        else:
            r = -lag
            L = n - r
            if L <= 0:
                continue
            G[:L] += w * dx * (1.0 - h[r:])
            if r > 0:
                G[n - r:] -= w * dx * h[:L]
    return G


def grad_y_fn(h, G):
    W = h * (1.0 - h)
    S = np.sum(W)
    if S < 1e-12:
        return np.zeros(len(h))
    mean_G = np.dot(G, W) / S
    return W * (G - mean_G)


def exact_c5_fn(h):
    dx = 2.0 / len(h)
    c_vals = compute_c_fn(h, dx)
    return float(np.max(c_vals))


def project_feasible_fn(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        if not np.any(mask):
            return h
        cap = np.sum(1.0 - h[mask])
        if cap <= 0:
            return h
        t = min(diff / cap, 1.0)
        h[mask] += t * (1.0 - h[mask])
    else:
        mask = h > 0.0
        if not np.any(mask):
            return h
        need = -diff
        avail = np.sum(h[mask])
        if avail <= 0:
            return h
        t = min(need / avail, 1.0)
        h[mask] *= (1.0 - t)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            per = diff2 / len(idx)
            h[idx] += per
    h = np.clip(h, 0.0, 1.0)
    return h


class SmoothOpt:
    def __init__(self, beta, dx, n, deadline):
        self.beta = beta
        self.dx = dx
        self.n = n
        self.deadline = deadline
        self._y_cache = None
        self._h_cache = None
        self._c_cache = None
        self._w_cache = None

    def _compute(self, y):
        if (self._y_cache is not None and
                len(y) == len(self._y_cache) and
                np.array_equal(y, self._y_cache)):
            return self._h_cache, self._c_cache, self._w_cache
        h = make_h_fn(y, self.n)
        c_vals = compute_c_fn(h, self.dx)
        _, weights = logsumexp_fn(c_vals, self.beta)
        self._y_cache = y.copy()
        self._h_cache = h
        self._c_cache = c_vals
        self._w_cache = weights
        return h, c_vals, weights

    def fun(self, y):
        if time.monotonic() > self.deadline:
            raise TimeoutError()
        h, c_vals, weights = self._compute(y)
        val, _ = logsumexp_fn(c_vals, self.beta)
        return val

    def jac(self, y):
        if time.monotonic() > self.deadline:
            raise TimeoutError()
        h, c_vals, weights = self._compute(y)
        G = grad_h_fn(h, weights, self.dx, self.n)
        return grad_y_fn(h, G)


def optimize_at_resolution(h_start, n, deadline):
    h_start = project_feasible_fn(h_start)
    c_start = exact_c5_fn(h_start)
    best_h = h_start.copy()
    best_c = c_start

    eps = 1e-5
    h_clip = np.clip(h_start, eps, 1.0 - eps)
    y = np.log(h_clip / (1.0 - h_clip))

    betas = [20, 50, 150, 400, 1000, 2500, 5000]
    max_iters = [80, 60, 50, 40, 35, 30, 25]

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 5.0:
            break
        iters = max_iters[i] if i < len(max_iters) else 25
        dx = 2.0 / n
        opt = SmoothOpt(beta, dx, n, deadline)

        try:
            res = minimize(opt.fun, y, jac=opt.jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-14, 'gtol': 1e-10})
            y_new = res.x
        except (TimeoutError, Exception):
            y_new = y

        h_new = make_h_fn(y_new, n)
        h_new = project_feasible_fn(h_new)
        c_new = exact_c5_fn(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h_clip = np.clip(h_new, eps, 1.0 - eps)
            y = np.log(h_clip / (1.0 - h_clip))

    return best_h, best_c

def get_start_at_resolution(h_parent, n_target):
    n_parent = len(h_parent)
    if n_target == n_parent:
        return h_parent.copy()
    if n_target % n_parent == 0:
        k = n_target // n_parent
        return np.repeat(h_parent, k)
    x_old = np.linspace(0.0, 2.0, n_parent + 2)[1:-1]
    x_new = np.linspace(0.0, 2.0, n_target + 2)[1:-1]
    return np.interp(x_new, x_old, h_parent)


def fallback_construction(n=64):
    target = n * 0.5
    h = np.zeros(n)
    n_ones = int(target)
    frac = target - n_ones
    if n_ones > 0:
        positions = np.unique((np.arange(n_ones) * n // n_ones).astype(int))
        positions = positions[positions < n]
        h[positions[:n_ones]] = 1.0
    if abs(frac - 0.5) < 1e-10 and n_ones < n:
        pos = (n_ones * n // max(n_ones + 1, 1)) % n
        h[pos] = 0.5
    return project_feasible_fn(h)
