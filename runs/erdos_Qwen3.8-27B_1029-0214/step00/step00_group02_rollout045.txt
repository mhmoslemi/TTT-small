The score is: max of np.correlate(h, 1-h, mode="full") * dx

The correlate gives for each lag the sum of h[i] * (1-h[i+lag]) (with appropriate boundaries). We want to minimize the maximum over all lags.

Key insight: The product h[i]*(1-h[j]) appears in the correlation when j = i + lag (or i = j + lag). So large h at position i and small h (i.e., 1-h large) at position j=i+lag creates a large contribution.

Strategies:
1. For each pair (i,j), compute G[i,j] = h[i]*(1-h[j]). The top entries in G correspond to positions where we might want to reduce h[i] or increase h[j] (or both via transfer).
2. A transfer from i to j (reduce h[i], increase h[j]) by amount δ changes the score. We can try various (i,j) pairs and amounts.

            fG = G.ravel()

            top_k = np.argpartition(fG, -K)[-K:]
            idx = top_k[rng.integers(K)]
            i = int(idx) // n
            j = int(idx) % n
        else:
            i = int(rng.integers(n))
            j = int(rng.integers(n))

        if i == j:
            continue
        h_i = cur[i] / M
        h_j = cur[j] / M
        if h_i >= h_j:
            if cur[i] <= 0 or cur[j] >= M:
                continue
            src, dst = i, j
        else:
            if cur[j] <= 0 or cur[i] >= M:
                continue
            src, dst = j, i

        H_new = cur.copy()
        H_new[src] -= 1
        H_new[dst] += 1
        ns = grid_score(H_new, M)
        delta = ns - cur_s

        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            cur = H_new
            cur_s = ns
            if cur_s < best_sa_s:
                best_sa_s = cur_s
                best_sa = cur.copy()
            if cur_s < global_best_s:
                global_best_s = cur_s
                global_best_h = cur.astype(np.float64) / M

        T *= cool_rate

    return best_sa, best_sa_s


def try_alt_resolution(h_best, n_best, M, t_deadline, rng, global_best_h, global_best_s):
    for n_alt in [n_best + 4, n_best + 8, n_best + 12, n_best + 20]:
        if time.perf_counter() > t_deadline:
            break
        x_orig = (np.arange(n_best) + 0.5) * (2.0 / n_best)
        x_new = (np.arange(n_alt) + 0.5) * (2.0 / n_alt)
        h_alt = np.interp(x_new, x_orig, h_best)
        h_alt = repair_feasible(h_alt)
        s_alt = compute_score(h_alt)
        if s_alt < global_best_s:
            global_best_s = s_alt
            global_best_h = h_alt.copy()
        h_alt, s_alt = continuous_local_search(h_alt, 500, time.perf_counter() + 20, rng)
        if s_alt < global_best_s:
            global_best_s = s_alt
            global_best_h = h_alt.copy()
    return global_best_h, global_best_s


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.perf_counter() - t_start

    def remaining():
        return budget_s - elapsed()

    try:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
        n = len(h0)
    except Exception:
        n = 47
        h0 = 0.5 * np.ones(n)

    h0 = repair_feasible(h0)
    best_h = h0.copy()
    best_s = compute_score(best_h)

    # Phase 1: Continuous local search
    t1 = t_start + budget_s * 0.20
    if remaining() > 10:
        best_h, best_s = continuous_local_search(best_h, 8000, t1, rng)

    # Phase 2: Grid local search
    M = 64
    H = gridify(best_h, M, (M * n) // 2)
    gs = grid_score(H, M)
    if gs < best_s:
        best_s = gs
        best_h = H.astype(np.float64) / M

    t2 = t_start + budget_s * 0.40
    if remaining() > 10:
        H, gs = grid_local_search(H, M, 8000, t2, rng)
        if gs < best_s:
            best_s = gs
            best_h = H.astype(np.float64) / M

    # Phase 3: Simulated annealing on grid
    t3 = t_start + budget_s * 0.70
    if remaining() > 20:
        H = gridify(best_h, M, (M * n) // 2)
        gb_h, gb_s = best_h.copy(), best_s
        H, gs = grid_sa(H, M, t3, 40000, rng, gb_h, gb_s)
        if gs < best_s:
            best_s = gs
            best_h = H.astype(np.float64) / M
        # Final polish after SA
        H = gridify(best_h, M, (M * n) // 2)
        H, gs = grid_local_search(H, M, 3000, t_start + budget_s * 0.78, rng)
        if gs < best_s:
            best_s = gs
            best_h = H.astype(np.float64) / M

    # Phase 4: Continuous fine-tuning
    t4 = t_start + budget_s * 0.85
    if remaining() > 15:
        best_h, best_s = continuous_local_search(best_h, 5000, t4, rng)

    # Phase 5: Alternative resolutions
    t5 = t_start + budget_s * 0.93
    if remaining() > 30:
        best_h, best_s = try_alt_resolution(best_h, n, M, t5, rng, best_h, best_s)
        # Polish the alt result
        best_h, best_s = continuous_local_search(best_h, 3000, t_start + budget_s * 0.96, rng)

    # Final: ensure feasibility and compute exact score
    n_final = len(best_h)
    best_h = repair_feasible(best_h)
    if not check_feasible(best_h, 1e-8):
        H_f = gridify(best_h, M, (M * n_final) // 2)
        best_h = H_f.astype(np.float64) / M

    final_score = compute_score(best_h)
    return best_h, final_score, n_final
