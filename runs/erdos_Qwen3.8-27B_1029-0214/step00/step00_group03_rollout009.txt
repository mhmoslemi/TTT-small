    n_restarts = int(time_left(start_time, budget_s, 80) / 80)
    for r in range(max(1, n_restarts)):
        if time_left(start_time, budget_s, 70) < 0:
            break
        scale = 0.15 + 0.1 * r
        h_pert = best_h + scale * rng.standard_normal(n)
        h_pert = project_feasible(h_pert, target_sum)
        try:
            res = minimize(smooth_max_obj, h_pert, args=(50.0,), method='SLSQP',
                           bounds=bounds, constraints=[con],
                           options={'maxiter': 2000, 'ftol': 1e-15})
            res_h = project_feasible(res.x, target_sum)
            c5 = compute_c5(res_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = res_h
        except Exception:
            pass
        if time_left(start_time, budget_s, 70) < 0:
            break
        try:
            res2 = minimize(raw_max_penalized, best_h, args=(target_sum, 500.0),
                            method='Nelder-Mead',
                            options={'maxiter': 10000, 'xatol': 1e-12,
                                     'fatol': 1e-14, 'adaptive': True})
            res_h2 = project_feasible(res2.x, target_sum)
            c5_2 = compute_c5(res_h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = res_h2
        except Exception:
            pass

    if time_left(start_time, budget_s, 300) > 0:
        for n_high in [2 * n, 3 * n]:
            if time_left(start_time, budget_s, 200) < 0:
                break
            target_high = n_high / 2.0
            xs_old = np.linspace(0.0, 2.0, n, endpoint=False)
            xs_new = np.linspace(0.0, 2.0, n_high, endpoint=False)
            h_high = np.interp(xs_new, xs_old, best_h)
            h_high = project_feasible(h_high, target_high)

            best_hh = h_high.copy()
            best_c5h = compute_c5(h_high)

            bounds_high = [(0.0, 1.0)] * n_high
            con_high = {'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (target_high,)}

            for alpha_h in [20.0, 50.0, 100.0, 200.0]:
                if time_left(start_time, budget_s, 150) < 0:
                    break
                try:
                    res = minimize(smooth_max_obj, best_hh, args=(alpha_h,), method='SLSQP',
                                   bounds=bounds_high, constraints=[con_high],
                                   options={'maxiter': 3000, 'ftol': 1e-16})
                    res_h = project_feasible(res.x, target_high)
                    c5 = compute_c5(res_h)
                    if c5 < best_c5h:
                        best_c5h = c5
                        best_hh = res_h
                except Exception:
                    pass

            if time_left(start_time, budget_s, 100) > 0:
                try:
                    res_nm = minimize(raw_max_penalized, best_hh,
                                      args=(target_high, 500.0),
                                      method='Nelder-Mead',
                                      options={'maxiter': 20000, 'xatol': 1e-12,
                                               'fatol': 1e-14, 'adaptive': True})
                    res_h = project_feasible(res_nm.x, target_high)
                    c5 = compute_c5(res_h)
                    if c5 < best_c5h:
                        best_c5h = c5
                        best_hh = res_h
                except Exception:
                    pass

            if best_c5h < best_c5:
                best_c5 = best_c5h
                best_h = best_hh
                n = n_high

    if time_left(start_time, budget_s, 40) > 0:
        for alpha_final in [300.0, 500.0]:
            if time_left(start_time, budget_s, 40) < 0:
                break
            n_cur = len(best_h)
            target_cur = n_cur / 2.0
            bounds_cur = [(0.0, 1.0)] * n_cur
            con_cur = {'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (target_cur,)}
            try:
                res = minimize(smooth_max_obj, best_h, args=(alpha_final,), method='SLSQP',
                               bounds=bounds_cur, constraints=[con_cur],
                               options={'maxiter': 5000, 'ftol': 1e-18})
                res_h = project_feasible(res.x, target_cur)
                c5 = compute_c5(res_h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = res_h
            except Exception:
                pass

    diff = target_sum - np.sum(best_h) if len(best_h) == n else len(best_h) / 2.0 - np.sum(best_h)
    if abs(diff) > 1e-10:
        best_h = project_feasible(best_h, len(best_h) / 2.0)

    return best_h, compute_c5(best_h), len(best_h)
