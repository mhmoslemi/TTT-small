import numpy as np
import time
from scipy.optimize import minimize

def exact_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr) * dx)
    return val if np.isfinite(val) else np.inf

def project_feasible(v, target_sum):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        if float(np.sum(np.clip(v - lo, 0.0, 1.0))) <= target_sum:
            break
        lo -= 1.0
    for _ in range(60):
        if float(np.sum(np.clip(v - hi, 0.0, 1.0))) >= target_sum:
            break
        hi += 1.0
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5*(lo+hi), 0.0, 1.0)

def smooth_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    scaled = corr * dx
    mx = float(np.max(scaled))
    if not np.isfinite(mx):
        return 1e10, np.zeros(n)
    exp_arg = beta * (scaled - mx)
    ev = np.exp(exp_arg)
    total = float(np.sum(ev))
    if total < 1e-300:
        return 1e10, np.zeros(n)
    w = ev / total
    obj = mx + float(np.log(total)) / beta
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = float(w[k]) * dx
        if abs(wk) < 1e-18:
            continue
        if s >= 0:
            L = n - s
            grad[s:] += wk * omh[:L]
            grad[:L] -= wk * h[s:]
        else:
            r = -s
            L = n - r
            grad[:L] += wk * omh[r:r+L]
            grad[r:] -= wk * h[:L]
    return obj, grad
