        target = n / 2.0

            return h

            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-15:
                h += (residual / total_room) * room
        else:
            total_mass = np.sum(h)
            if total_mass > 1e-15:
                h += (residual / total_mass) * h
        h = np.clip(h, 0.0, 1.0)

        if abs(residual) > 1e-12:

                h[idx] = min(h[idx] + residual, 1.0)
            else:

                h[idx] = max(h[idx] + residual, 0.0)

def _eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _project(h, n):
    h = np.clip(h, 0.0, 1.0).astype(np.float64)
    target = n * 0.5
    diff = target - h.sum()
    if diff > 1e-15:
        room = (1.0 - h).sum()
        if room > 1e-15:
            h = h + diff / room * (1.0 - h)
    elif diff < -1e-15:
        mass = h.sum()
        if mass > 1e-15:
            h = h + diff / mass * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            i = int(np.argmin(h))
            h[i] = min(h[i] + diff, 1.0)
        else:
            i = int(np.argmax(h))
            h[i] = max(h[i] + diff, 0.0)
    return h

def _lse_vg(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = c.max()
    s = np.clip(beta * (c - mx), -80.0, 80.0)
    e = np.exp(s)
    se = e.sum()
    p = e / se
    val = mx + np.log(se) / beta
    idx = np.arange(n)
    AI = idx[None, :] - idx[:, None] + (n - 1)
    A = p[AI]
    g = A @ (1.0 - h) - A.T @ h
    return val, g
