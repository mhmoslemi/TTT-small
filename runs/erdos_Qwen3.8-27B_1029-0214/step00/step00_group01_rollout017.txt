    lo = -1.0

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_to_feasible(h, target_sum):
    n = len(h)
    h = np.asarray(h, dtype=float).copy()
    # Handle non-finite
    h[~np.isfinite(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)
    # Bisection for mu
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix floating point
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-14:
        idx = np.where((result > 1e-12) & (result < 1 - 1e-12))[0]
        if len(idx) > 0:
            i = idx[0]
            result[i] += diff
            result[i] = np.clip(result[i], 0.0, 1.0)

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

    n = len(h)
    dx = 2.0 / n
    tau = row_idx - (n - 1)
    grad = np.zeros(n)
    if tau >= 0:
        L = n - tau
        if L <= 0:
            return grad
        grad[:L] += dx * (1.0 - h[tau:tau+L])
        grad[tau:tau+L] -= dx * h[:L]
    else:
        r = -tau
        L = n - r
        if L <= 0:
            return grad
        grad[r:r+L] += dx * (1.0 - h[:L])
        grad[:L] -= dx * h[r:r+L]

```python
class ErdosNLP:
    def __init__(self, n, active_rows, deadline):
        self.n = n
        self.dx = 2.0 / n
        self.target = n / 2.0
        self.active_rows = np.asarray(active_rows)
        self.deadline = deadline
    
    def _check(self):
        if time.time() > self.deadline:
            raise RuntimeError("Time limit exceeded")
    
    def fun(self, x):
        self._check()
        return float(x[self.n])
    
    def fun_grad(self, x):
        self._check()
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g
    
    def eq_fun(self, x):
        self._check()
        return np.sum(x[:self.n]) - self.target
    
    def eq_jac(self, x):
        self._check()
        j = np.zeros(self.n + 1)
        j[:self.n] = 1.0
        return j
    
    def ineq_fun(self, x):
        self._check()
        h = x[:self.n]
        t = x[self.n]
        n = self.n
        m = len(self.active_rows)
        out = np.empty(m)
        for i in range(m):
            row = self.active_rows[i]
            tau = row - (n - 1)
            if tau >= 0:
                L = n - tau
                out[i] = t - (self.dx * np.dot(h[:L], (1.0-h)[tau:tau+L]) if L > 0 else 0.0)
            else:
                r = -tau
                L = n - r
                out[i] = t - (self.dx * np.dot(h[r:], (1.0-h)[:L]) if L > 0 else 0.0)
        return out
    
    def ineq_jac(self, x):
        self._check()
        h = x[:self.n]
        n = self.n
        m = len(self.active_rows)
        J = np.zeros((m, n + 1))
        for i in range(m):
            row = self.active_rows[i]
            tau = row - (n - 1)
            if tau >= 0:
                L = n - tau
                if L > 0:
                    J[i, :L] += self.dx * (1.0 - h[tau:tau+L])
                    J[i, tau:tau+L] -= self.dx * h[:L]
            else:
                r = -tau
                L = n - r
                if L > 0:
                    J[i, r:r+L] += self.dx * (1.0 - h[:L])
                    J[i, :L] -= self.dx * h[r:r+L]
            J[i, n] = 1.0
        return J
```
