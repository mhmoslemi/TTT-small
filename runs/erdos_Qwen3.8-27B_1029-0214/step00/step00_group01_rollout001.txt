    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h):
    """Project h to [0,1]^n with sum = n/2 using water-filling."""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Water-filling: find shift τ such that sum(clip(h + τ, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def compute_correlations(h):
    """Compute all c_s values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr

def surrogate_and_grad(h, beta):
    """Compute log-sum-exp surrogate and its gradient w.r.t. h."""
    c = compute_correlations(h)
    # Stable log-sum-exp
    cmax = np.max(c)
    exp_c = np.exp(beta * (c - cmax))
    log_sum = cmax + np.log(np.sum(exp_c)) / beta
    w = exp_c / np.sum(exp_c)
    
    # Gradient w.r.t. h
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # For each lag s (corresponding to index s + n - 1 in corr array)
    for idx in range(2 * n - 1):
        s = idx - (n - 1)  # lag
        if w[idx] < 1e-15:
            continue
        wi = w[idx] * dx
        if s >= 0:
            L = n - s
            # h[i] for i in [0, L-1]: contrib += wi * (1 - h[i+s])
            grad[:L] += wi * (1.0 - h[s:])
            # h[i] for i in [s, n-1]: contrib -= wi * h[i-s]
            grad[s:] -= wi * h[:L]
        else:
            r = -s
            L = n - r
            # h[i] for i in [0, L-1]: contrib -= wi * h[i+r]
            grad[:L] -= wi * h[r:]
            # h[i] for i in [r, n-1]: contrib += wi * (1 - h[i-r])
            grad[r:] += wi * (1.0 - h[:L])

    n = len(y)
    target = n / 2.0

    h, lam = make_h_from_y(y)

    if S < 1e-15:

    inner_product = np.dot(grad_h, h * (1.0 - h)) / S
    grad_y = h * (1.0 - h) * (grad_h - inner_product)
