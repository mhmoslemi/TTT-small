import time

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    """Make h feasible: 0<=h<=1, sum=h = n/2"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if diff > 0:
        # Need to add mass; add to positions with most room
        room = 1.0 - h
        order = np.argsort(-room)
        add = diff
        for i in order:
            take = min(add, room[i])
            h[i] += take
            add -= take
            if add <= 1e-15:
                break
    elif diff < 0:
        # Need to remove mass
        order = np.argsort(-h)
        rem = -diff
        for i in order:
            take = min(rem, h[i])
            h[i] -= take
            rem -= take
            if rem <= 1e-15:
                break
    return h

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)

            else:
                idx = np.argmax(h)
