import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h[~np.isfinite(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if diff > 1e-14:
        room = 1.0 - h
        total_room = np.sum(room)
        if total_room > 1e-14:
            h = h + diff * room / total_room
        else:
            h[:] = 0.5
    elif diff < -1e-14:
        total = np.sum(h)
        if total > 1e-14:
            h = h + diff * h / total
        else:
            h[:] = 0.5
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + residual)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + residual)
    return h

def lse_objective(h, beta, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    scaled = beta * dx * corr
    max_s = np.max(scaled)
    if max_s > 600:
        return max_s / beta
    return (max_s + np.log(np.sum(np.exp(scaled - max_s)))) / beta

def lse_gradient(h, beta, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    scaled = beta * dx * corr
    max_s = np.max(scaled)
    exps = np.exp(scaled - max_s)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            r = lag
            e = n - r
            grad[0:e] += w * (-h[r:])
            grad[r:] += w * (1.0 - h[0:e])
        else:
            r = -lag
            e = n - r
            grad[0:e] += w * (1.0 - h[r:])
            grad[r:] += w * (-h[0:e])
    return grad

def penalty_obj(h, beta, n, pen):
    val = lse_objective(h, beta, n)
    sv = np.sum(h) - n / 2.0
    return val + pen * sv * sv

def penalty_grad(h, beta, n, pen):
    g = lse_gradient(h, beta, n)
    sv = np.sum(h) - n / 2.0
    g += 2.0 * pen * sv / n
    return g

def violation_obj(h, t, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    scaled = dx * corr
    violations = np.maximum(scaled - t, 0.0)
    val = np.sum(violations * violations)
    sv = np.sum(h) - n / 2.0
    return val + 1e6 * sv * sv

def violation_grad(h, t, n):

    corr = np.correlate(h, 1 - h, mode="full")
    scaled = dx * corr

            grad[r:] += w * (-h[0:e])
    sv = np.sum(h) - n / 2.0

        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_to_feasible(h_interp, n)
        bounds_n = [(0.0, 1.0)] * n

        for beta in [10, 50, 100, 200, 500]:
            if time_left() < 90:
                break
            res = minimize(
                penalty_obj, h_interp, args=(beta, n, pen),
                jac=penalty_grad, method='SLSQP',
                bounds=bounds_n,
                options={'maxiter': 400, 'ftol': 1e-16}
            )
            if res.x is not None:
                h_new = project_to_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_interp = h_new.copy()

    # Phase 3: Violation-based optimization at best resolution
    n = len(best_h)
    bounds_n = [(0.0, 1.0)] * n
    current_max = compute_c5(best_h)

    for shrink in [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]:
        if time_left() < 60:
            break
        t_target = best_c5 - shrink
        if t_target < 0.2:
            continue
        h_start = best_h.copy()
        res = minimize(
            violation_obj, h_start, args=(t_target, n),
            jac=violation_grad, method='SLSQP',
            bounds=bounds_n,
            options={'maxiter': 500, 'ftol': 1e-16}
        )
        if res.x is not None:
            h_new = project_to_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 4: Random restarts with SLSQP
    n = len(best_h)
    bounds_n = [(0.0, 1.0)] * n
    attempts = 0
    while time_left() > 40 and attempts < 50:
        attempts += 1
        scale = 0.003 + 0.01 * rng.rand()
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_to_feasible(h_pert, n)

        beta = 500 if rng.rand() < 0.5 else 2000
        res = minimize(
            penalty_obj, h_pert, args=(beta, n, pen),
            jac=penalty_grad, method='SLSQP',
            bounds=bounds_n,
            options={'maxiter': 300, 'ftol': 1e-16}
        )
        if res.x is not None:
            h_new = project_to_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_nw = c5_new
                best_h = h_new.copy()

    # Phase 5: Final fine-tuning with very high beta
    if time_left() > 30:
        n = len(best_h)
        bounds_n = [(0.0, 1.0)] * n
        for beta in [5000, 10000]:
            if time_left() < 20:
                break
            res = minimize(
                penalty_obj, best_h, args=(beta, n, pen),
                jac=penalty_grad, method='SLSQP',
                bounds=bounds_n,
                options={'maxiter': 500, 'ftol': 1e-16}
            )
            if res.x is not None:
                h_new = project_to_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    return (best_h, best_c5, len(best_h))
