def optimal_delta(R, c, d_min, d_max):
    """Find delta in [d_min, d_max] minimizing max(R + delta*c)"""
    if d_max - d_min < 1e-14:
        return 0.0
    
    # Check if the minimum is at a boundary
    f_min = np.max(R + d_min * c)
    f_max = np.max(R + d_max * c)
    
    # Binary search for the point where subdifferential contains 0
    lo = d_min
    hi = d_max
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = R + mid * c
        mx = np.max(vals)
        tol = 1e-12 * max(1.0, mx)
        active = vals >= (mx - tol)
        c_act = c[active]
        if np.min(c_act) > 1e-12:
            # All active slopes positive → f increasing → go left
            hi = mid
        elif np.max(c_act) < -1e-12:
            # All active slopes negative → f decreasing → go right  
            lo = mid
        else:
            # 0 in subdifferential
            lo = hi = mid
            break
    
    d_opt = (lo + hi) / 2.0
    return d_opt

def pair_swap_sweep(h, R, n, rng, max_pairs=None):
    """One sweep of pair swaps. Returns updated h, R, and improvement."""
    m = 2 * n - 1
    c_vec = np.zeros(m)
    
    # Generate pair order
    idx = np.arange(n)
    pairs = []
    if max_pairs is None:
        for i in range(n):
            for j in range(i+1, n):
                pairs.append((i, j))
        rng.shuffle(pairs)
    else:
        all_i = rng.integers(0, n, size=max_pairs)
        all_j = rng.integers(0, n, size=max_pairs)
        # Ensure i != j
        mask = all_i != all_j
        all_i = all_i[mask][:max_pairs]
        all_j = all_j[mask][:max_pairs]
        pairs = list(zip(all_i, all_j))
    
    total_improvement = 0.0
    current_max = np.max(R)
    
    for (i, j) in pairs:
        a = h[i]
        b = h[j]
        d_min = max(-a, b - 1.0)
        d_max = min(1.0 - a, b)
        if d_max - d_min < 1e-13:
            continue
        
        # Compute c vector
        np.fill_(c_vec, 0.0)
        for s in range(-(n-1), n):
            ip = i + s
            im = i - s
            jp = j + s
            jm = j - s
            hp = h[ip] if 0 <= ip < n else 0.0
            hh = h[im] if 0 <= im < n else 0.0
            jp_val = h[jp] if 0 <= jp < n else 0.0
            jm_val = h[jm] if 0 <= jm < n else 0.0
            c_vec[s + n - 1] = jp_val + jm_val - hp - hh
        
        d_opt = optimal_delta(R, c_vec, d_min, d_max)
        
        if abs(d_opt) < 1e-14:
            continue
        
        # Apply
        h[i] += d_opt
        h[j] -= d_opt
        R += d_opt * c_vec
        
        # Clamp for numerical safety
        h[i] = np.clip(h[i], 0.0, 1.0)
        h[j] = np.clip(h[j], 0.0, 1.0)
    
    return h, R

def upsample(h, new_n):

    h_new = np.interp(x_new, x_old, h)
