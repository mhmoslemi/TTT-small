```python
import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(x, n):
    """Project x onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    x = np.clip(x, 0, 1)
    # Use bisection to find shift
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(x + mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(x + (lo + hi) / 2, 0, 1)
    # Fix residual
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        # Distribute among elements with room
        if residual > 0:
            room = 1.0 - x
            idx = np.argsort(-room)[:max(1, int(abs(residual)) + 1)]
        else:
            room = x
            idx = np.argsort(room)[:max(1, int(abs(-residual)) + 1)]
        # Simple distribution
        per = residual / len(idx)
        x[idx] = np.clip(x[idx] + per, 0, 1)
    return x

def compute_c5(h):
    """Compute the Erdos C5 value"""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def lse_objective(x, alpha, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(c)
    shifted = alpha * (c - max_c)
    return max_c + np.log(np.sum(np.exp(shifted))) / alpha

def lse_gradient(x, alpha, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(c)
    shifted = alpha * (c - max_c)
    e = np.exp(shifted)
    p = e / np.sum(e)  # softmax weights
    
    grad = np.zeros(n)
    for r in range(-(n-1), n):
        k = r + n - 1
        pk = p[k]
        if pk < 1e-15:
            continue
        if r == 0:
            grad += pk * dx * (1 - 2 * x)
        elif r > 0:
            grad[r:] += pk * dx * (1 - x[:n-r])
            grad[:n-r] -= pk * dx * x[r:]
        else:
            s = -r
            grad[:n-s] += pk * dx * (1 - x[s:])
            grad[s:] -= pk * dx * x[:n-s]
    return grad

def optimize_slsqp(h, alpha, n, time_limit=60):
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    start = time.time()
    result = minimize(
        fun=lambda x: lse_objective(x, alpha, n),
        x0=h.copy(),
        jac=lambda x: lse_gradient(x, alpha, n),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 200, 'ftol': 1e-12, 'disp': False}
    )
    return result.x if result.success or compute_c5(result.x) < compute_c5(h) else h

def subgradient_step(h, step_size, n, top_m=5):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Find top m
    top_idx = np.argsort(c)[-top_m:]
    # Average gradient of top terms
    grad = np.zeros(n)
    for k in top_idx:
        r = k - (n - 1)
        if r == 0:
            grad += dx * (1 - 2 * h)
        elif r > 0:
            grad[r:] += dx * (1 - h[:n-r])
            grad[:n-r] -= dx * h[r:]
        else:
            s = -r
            grad[:n-s] += dx * (1 - h[s:])
            grad[s:] -= dx * h[:n-s]
    grad /= top_m
```

```python
def project_feasible(x, n):
    target = n / 2.0
    x = np.asarray(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    # Bisection for shift
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            room = 1.0 - x
            order = np.argsort(-room)
        else:
            room = x
            order = np.argsort(room)
        # Distribute
        for idx in order:
            if abs(residual) < 1e-12:
                break
            if residual > 0:
                add = min(residual, room[idx])
                x[idx] += add
                residual -= add
            else:
                sub = min(-residual, room[idx])
                x[idx] -= sub
                residual += sub
    return x

    e = np.exp(sh)
    p = e / np.sum(e)

        return lse_grad(x, alpha, n)

        x0=x0,
```

```python
import numpy as np
import time
from scipy.optimize import minimize


def project_feasible(x, n):
    target = n / 2.0
    x = np.asarray(x, dtype=float).copy()
    x[~np.isfinite(x)] = 0.5
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(x)
    if abs(residual) > 1e-13:
        if residual > 0:
            room = 1.0 - x
            order = np.argsort(-room)
        else:
            room = x
            order = np.argsort(room)
        for idx in order:
            if abs(residual) < 1e-14:
                break
            if residual > 0:
                add = min(residual, room[idx])
            else:
                add = max(residual, -room[idx])
            x[idx] += add
            residual -= add
    return x


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def lse_obj(x, alpha, n):
    c = corr_scaled(x)
    mc = np.max(c)
    sh = alpha * (c - mc)
    return mc + np.log(np.sum(np.exp(sh))) / alpha


def lse_grad(x, alpha, n):
    c = corr_scaled(x)
    mc = np.max(c)
    sh = alpha * (c - mc)
    e = np.exp(sh)
    p = e / np.sum(e)
    dx = 2.0 / n
    grad = np.zeros(n)
    for r in range(-(n - 1), n):
        k = r + n - 1
        pk = p[k]
        if pk < 1e-16:
            continue
        if r == 0:
            grad += pk * dx * (1.0 - 2.0 * x)
        elif r > 0:
            grad[r:] += pk * dx * (1.0 - x[:n - r])
            grad[:n - r] -= pk * dx * x[r:]
        else:
            s = -r
            grad[:n - s] += pk * dx * (1.0 - x[s:])
            grad[s:] -= pk * dx * x[:n - s]
    return grad

            aborted[0] = True
        return lse_grad(x, alpha, n)

    def con_fun(x):
        return np.sum(x) - n / 2.0

        return np.ones(n)

            fun=fun, x0=h.copy(), jac=jac, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
```
