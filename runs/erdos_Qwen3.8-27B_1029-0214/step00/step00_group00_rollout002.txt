    h0 = project_box_sum(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = exact_c5(h0, dx0)
    best_n = n0

    target_c5 = 0.38080
    grid_sizes = sorted(set([n0 * 2, n0 * 3, n0 * 4, 100, 120, 150, 180, 200]))
    grid_sizes = [g for g in grid_sizes if g > n0 and g <= 240]

    for n_new in grid_sizes:
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed
        if remaining < 20:
            break
        if best_c5 <= target_c5:
            break
        dx = 2.0 / n_new
        target = n_new / 2.0
        h_work = interp_h(best_h, n_new)
        h_work = project_box_sum(h_work, target)
        tau_work = exact_c5(h_work, dx)
        time_alloc = min(remaining - 15, 400)
        if n_new > 150:
            time_alloc = min(time_alloc, 200)
        h_work, tau_work = optimize_local(h_work, dx, target, time_alloc, t_start, seed=seed + n_new)
        if tau_work < best_c5 - 1e-13:
            best_h = h_work.copy()
            best_c5 = tau_work
            best_n = n_new

    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 30 and best_c5 > target_c5:
        dx_best = 2.0 / best_n
        target_best = best_n / 2.0
        h_refine = best_h.copy()
        h_refine, tau_refine = optimize_local(h_refine, dx_best, target_best,
                                              remaining - 10, t_start, seed=seed + 999)
        if tau_refine < best_c5 - 1e-13:
            best_h = h_refine.copy()
            best_c5 = tau_refine

    final_n = best_n
    final_dx = 2.0 / final_n
    final_h = project_box_sum(best_h.copy(), final_n / 2.0)
    final_c5 = exact_c5(final_h, final_dx)

    return (final_h, final_c5, final_n)
