def compute_terms_and_derivatives(h, d, n):
    """
    Compute A_k, B_k, C_k for each correlation term k.
    T_k(δ) = A_k + B_k*δ + C_k*δ²
    Returns arrays A, B, C of length 2n-1.
    """
    dx = 2.0 / n
    A = np.zeros(2*n-1)
    B = np.zeros(2*n-1)
    C = np.zeros(2*n-1)
    
    one_minus_h = 1.0 - h
    
    for k in range(2*n-1):
        lag = k - (n-1)
        # T_k = dx * sum_{i: 0<=i<n, 0<=i+lag<n} h[i+lag] * (1-h[i])
        if lag >= 0:
            i_start = 0
            i_end = n - lag
        else:
            i_start = -lag
            i_end = n
        m = i_end - i_start
        if m <= 0:
            continue
        i_idx = np.arange(i_start, i_end)
        h_lag = h[i_idx + lag]
        h_base = h[i_idx]
        d_lag = d[i_idx + lag]
        d_base = d[i_idx]
        omh_base = one_minus_h[i_idx]
        
        A[k] = dx * np.sum(h_lag * omh_base)
        B[k] = dx * np.sum(d_lag * omh_base - d_base * h_lag)
        C[k] = -dx * np.sum(d_lag * d_base)
    
    return A, B, C

            active.append(lag)
    # Compute subgradient (average of active term gradients)
    grad = np.zeros(n)
    for lag in active:
        for m in range(n):
            g = 0.0
            if 0 <= m - lag < n:
                g += (1.0 - h[m - lag])
            if 0 <= m + lag < n:
                g -= h[m + lag]
            grad[m] += dx * g
    grad /= max(len(active), 1)
    # Project to zero-mean
    grad -= np.mean(grad)
    norm = np.linalg.norm(grad)
    if norm < 1e-15:
        return None, tau
    grad /= norm
    return grad, tau


def make_pair_direction(n, i, j):
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    return d


def make_random_direction(n, rng):

    d -= np.mean(d)
    norm = np.linalg.norm(d)
    if norm < 1e-15:

def make_block_direction(n, rng, block_len):
    if block_len < 2 or block_len > n:
        return make_random_direction(n, rng)
    d = np.zeros(n)
    start = rng.integers(0, n - block_len + 1)
    mid = block_len // 2
    d[start:start + mid] = 1.0
    d[start + mid:start + block_len] = -1.0
    # Ensure zero mean
    d -= np.mean(d)
    norm = np.linalg.norm(d)
    if norm < 1e-15:
        return None
    d /= norm
    return d
