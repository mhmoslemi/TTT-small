    dx = 2.0 / n

            g[:] = dx * (1 - 2 * h)
        elif k > 0:
            e = n - k
            g[k:] = dx * (1 - h[:e])
            g[:e] -= dx * h[k:]
        else:

            e = n - kp
            g[:e] = dx * (1 - h[kp:])

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        residual = target - np.sum(h)
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr > 1e-14:
                h += residual * room / tr
            else:
                break
        else:
            mass = h.sum()
            if mass > 1e-14:
                h += residual * h / mass
            else:
                break
    h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-13:
        if residual > 0:
            for i in np.argsort(1.0 - h):
                add = min(residual, 1.0 - h[i])
                if add > 0:
                    h[i] += add
                    residual -= add
                if abs(residual) < 1e-14:
                    break
        else:

                rem = min(-residual, h[i])
                if rem > 0:
                    h[i] -= rem
                    residual += rem
