def project_to_feasible(x, target_sum):
    # Project x onto {h: 0 <= h[i] <= 1, sum(h) = target_sum}
    # Using bisection on the threshold
    h = x.copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo+hi)/2, 0, 1)
    # Fix any tiny numerical issues
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        idx = np.argsort(x - (lo+hi)/2)
        if diff > 0:
            for i in range(len(idx)):
                if h[idx[i]] < 1:
                    adjust = min(diff, 1 - h[idx[i]])
                    h[idx[i]] += adjust
                    diff -= adjust
                    if abs(diff) < 1e-10:
                        break
        else:
            for i in range(len(idx)):
                if h[idx[i]] > 0:
                    adjust = min(-diff, h[idx[i]])
                    h[idx[i]] -= adjust
                    diff += adjust
                    if abs(diff) < 1e-10:
                        break
    return h

def binary_local_search(h, n_points, max_iter=10000):
    # Local search for binary solutions (requires even n)
    dx = 2.0 / n_points
    target_sum = n_points // 2
    h = np.round(h).clip(0, 1)
    # Ensure sum is correct
    ones = np.sum(h)
    if ones > target_sum:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        for i in range(int(ones - target_sum)):
            h[idx[i]] = 0
    elif ones < target_sum:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        for i in range(int(target_sum - ones)):
            h[idx[i]] = 1
    
    def get_c5(h):
        g = 1.0 - h
        corr = np.correlate(h, g, mode="full")
        return np.max(corr) * dx
    
    best_c5 = get_c5(h)

        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        best_improve = 0

        for i in ones_idx:

        h[j] = 1

        h[j] = 1

def grad_smooth_c5(h, n_points, tau=30.0):
    dx = 2.0 / n_points
    n = n_points
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full") * dx
    c = np.max(corr)
    weights = np.exp(corr - c) / tau
    weights /= weights.sum()
    
    # For each k, C[k] = sum_{valid i} h[i] * (1-h[i-k+n-1])
    # grad[C[k]]/dh[j] = (1-h[j-k+n-1])*1[j in domain_k] - h[j+k-n+1]*1[j+k-n+1 in domain_k]
    # where domain_k = {i: max(0,k-n+1) <= i <= min(n-1,k)}
    
    obj = c + np.log(weights.sum()) / tau  # wait, this is just c + log(1)/tau = c
    # Actually obj = c + (1/tau) * log(sum exp(corr-c)) = c + (1/tau) * log(exp(0) + ...) 
    # = c + (1/tau) * log(sum exp(corr - c))
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k]
        if w < 1e-12:
            continue
        S_lo = max(0, k - n + 1)
        S_hi = min(n - 1, k)
        for i in range(S_lo, S_hi + 1):
            j = i - k + n - 1
            # C[k] has term h[i] * (1-h[j])
            grad[i] += w * (1.0 - h[j])
            grad[j] += w * (-h[i])
    
    return obj, grad

Now, for efficiency: for each m (0 to 2n-2), the gradient ∇C[m] has the form:
- At position j: (1-h[j-m+n-1]) if 0 ≤ j-m+n-1 ≤ n-1
- At position j: -h[j+m-n+1] if 0 ≤ j+m-n+1 ≤ n-1

The first part is: for j in range [m-n+1, m-n+1+n-1] = [m-n+1, m], clip to [0, n-1]:
  grad[j] += (1-h[j-m+n-1])

The second part is: for j in range [n-1-m, n-1-m+n-1] = [n-1-m, 2n-2-m], clip to [0, n-1]:
  grad[j] += -h[j+m-n+1]

So for each m, the gradient contributions are two contiguous segments (possibly empty). This can be computed with numpy slicing!

```python
def grad_smooth_c5(h, n_points, tau=30.0):
    dx = 2.0 / n_points
    n = n_points
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")  # C[m] for m=0,...,2n-2
    corr_scaled = corr * dx
    c = np.max(corr_scaled)
    exp_vals = np.exp(tau * corr_scaled - tau * c)
    weights = exp_vals / exp_vals.sum()
    
    obj = c + np.log(exp_vals.sum()) / tau  # = c + log(sum exp(tau*(C-c)))/tau
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        w = weights[m]
        if w < 1e-12:
            continue
        # First part: (1-h[j-m+n-1]) for j in [max(0, m-n+1), min(n-1, m)]
        j_lo = max(0, m - n + 1)
        j_hi = min(n - 1, m)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi+1)
            h_idx = idx - m + n - 1
            grad[j_lo:j_hi+1] += w * (1.0 - h[h_idx])
        
        # Second part: -h[j+m-n+1] for j in [max(0, n-1-m), min(n-1, 2n-2-m)]
        j_lo2 = max(0, n - 1 - m)
        j_hi2 = min(n - 1, 2*n - 2 - m)
        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2, j_hi2+1)
            h_idx2 = idx2 + m - n + 1
            grad[j_lo2:j_hi2+1] += w * (-h[h_idx2])
    
    grad *= dx
    return obj, grad
```
