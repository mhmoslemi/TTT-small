def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def is_feasible(h, tol=1e-9):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    target = n / 2.0
    if abs(np.sum(h) - target) > 1e-8 * max(1, n):
        return False
    return True

def make_feasible(h):
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    return np.clip(h, 0.0, 1.0)

def gridify(h, M):
    n = len(h)
    target_int = M * n // 2
    Y = M * np
    np.clip(h, 0.0, 1.0)
    Y = M * np.clip(h, 0.0, 1.0)
    base = np.floor(Y).astype(np.int64)
    base = np.clip(base, 0, M)
    err = target_int - int(np.sum(base))
    if err > 0:
        frac = Y - np.floor(Y)
        order = np.argsort(-frac)
        count = 0
        i = 0
        while count < err:
            idx = int(order[i % n])
            if base[idx] < M:
                base[idx] += 1
                count += 1
            i += 1
            if i > n * (err + 2):
                for j in range(n):

                        break
                    if base[j] < M:

                        count += 1

            if base[idx] > 0:

                count += 1
