        h_new = project_to_box_sum(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        # Always update h (even if c5 got worse) to allow exploration
        h = h_new
    return best_h, best_c5

def binary_local_search(h, n_iters, rng, max_time):
    n = len(h)
    h = np.round(h).astype(float)
    h = project_to_box_sum(h, n / 2.0)
    h = (h > 0.5).astype(float)  # Ensure binary
    # Fix sum if needed
    s = h.sum()
    if s != n / 2.0:
        # Adjust
        if s > n / 2.0:
            excess = int(s - n / 2.0)
            ones = np.where(h == 1)[0]
            for _ in range(excess):
                idx = rng.randint(len(ones))
                h[ones[idx]] = 0
                ones = np.delete(ones, idx)
        else:
            deficit = int(n / 2.0 - s)
            zeros = np.where(h == 0)[0]
            for _ in range(deficit):
                idx = rng.randint(len(zeros))
                h[zeros[idx]] = 1
                zeros = np.delete(zeros, idx)
    
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    start = time.time()
    
    for it in range(n_iters):
        if it % 50 == 0 and time.time() - start > max_time:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        if new_c5 <= current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_c5
