    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    return np.max(corr) * dx

def constraint_values(h):
    """Return all lag values (before scaling by dx)."""
    n = len(h)
    v = 1.0 - h
    return np.correlate(h, v, mode='full')

def constraint_jac_row(r, h):
    """Return the gradient of C_r (before dx scaling) with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if r == 0:
        g[:] = 1.0 - 2.0 * h
    elif r > 0:
        e = n - r
        # dC_r/dh[j] for j in [0, e): includes (1 - h[j+r])
        g[0:e] += (1.0 - h[r:n])

        g[r:n] += (-h[0:e])
    else:
        s = -r
        e = n - s
        # dC_{-s}/dh[j] for j in [0, e): includes (-h[j+s])
        g[0:e] += (-h[s:n])
        # dC_{-s}/dh[j] for j in [s, n): includes (1 - h[j-s])
        g[s:n] += (1.0 - h[0:e])
    return g

def objective_and_constraints(x, n):
    """x = [h_0, ..., h_{n-1}, t]"""
    h = x[:n]
    t = x[n]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Objective: minimize t
    obj = t
    
    # Equality: sum(h) - n/2
    eq_val = np.sum(h) - n / 2.0
    
    # Inequalities: t - corr[i] * dx >= 0
    ineq_vals = t - corr * dx
    
    # Jacobian
    # eq jac: [1, 1, ..., 1, 0]
    eq_jac = np.zeros(n + 1)
    eq_jac[:n] = 1.0
    
    # ineq jacs: for each of the 2n-1 constraints
    # grad of (t - C_r * dx) w.r.t. h = -dx * dC_r/dh

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_feasible(y, n):
    """Project y to feasible set: 0<=h<=1, sum(h)=n/2."""
    target = n / 2.0
    h = np.clip(y.copy(), 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to add: add to elements not at 1
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            additions = 1.0 - h[mask]
            total_add = np.sum(additions)
            if total_add < 1e-15:
                break
            scale = min(diff / total_add, 1.0)
            h[mask] += additions * scale
        else:
            # Need to subtract: subtract from elements not at 0
            mask = h > 1e-12
            if not np.any(mask):
                break
            subtractions = h[mask]
            total_sub = np.sum(subtractions)
            if total_sub < 1e-15:
                break
            scale = min(-diff / total_sub, 1.0)
            h[mask] -= subtractions * scale
    # Final exact adjustment
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-14:
        # Distribute evenly among non-bound elements
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def constraint_jac_row(r, h):
    n = len(h)
    g = np.zeros(n)
    if r == 0:
        g[:] = 1.0 - 2.0 * h
    elif r > 0:
        e = n - r
        g[0:e] += (1.0 - h[r:n])
        g[r:n] += (-h[0:e])
    else:
        s = -r
        e = n - s
        g[0:e] += (-h[s:n])
        g[s:n] += (1.0 - h[0:e])
    return g

def solve_epigraph(h0, n, maxiter=300, ftol=1e-12):
    """Solve the epigraph problem starting from h0."""
    t0 = eval_c5(h0) + 1e-6
    x0 = np.concatenate([h0, [t0]])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    dx = 2.0 / n
    n_corr = 2 * n - 1
    
    def objective(x):
        return x[n]
    
    def objective_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def eq_fun(x):
        return np.sum(x[:n]) - n / 2.0
    
    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    def ineq_fun(x):
        h = x[:n]
        t = x[n]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr * dx
    
    def ineq_jac(x):
        h = x[:n]
        jac = np.zeros((n_corr, n + 1))
        jac[:, n] = 1.0
        for r in range(-(n-1), n):
            idx = r + (n - 1)
            g = constraint_jac_row(r, h)
            jac[idx, :n] = -dx * g
        return jac
    
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
    ]
    
    res = minimize(objective, x0, method='SLSQP', jac=objective_jac,
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': ftol})
    
    if res.success or res.x is not None:
        h_opt = proj_feasible(res.x[:n], n)
        c5_opt = eval_c5(h_opt)
        if np.isfinite(c5_opt):
            return h_opt, c5_opt
    return h0, t0

def interpolate_to(h, target_n):
    """Interpolate h from its current length to target_n."""
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n)
    x_new = np.linspace(0, n - 1, target_n)
    h_new = np.interp(x_new, x_old, h)
    return proj_feasible(h_new, target_n)

def block_replicate(h, target_n):
    """Replicate h to target_n by block repetition."""
    n = len(h)
    if n == target_n:
        return h.copy()
    h_new = np.zeros(target_n)
    for i in range(target_n):
        old_idx = int(i * n / target_n)
        h_new[i] = h[old_idx]
    return proj_feasible(h_new, target_n)

import numpy as np
from scipy.optimize import minimize
import time


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def proj_feasible(y, n):
    target = 0.5 * n
    h = np.clip(np.asarray(y, dtype=float).copy(), 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not mask.any():
                break
            room = 1.0 - h[mask]
            total = room.sum()
            if total < 1e-15:
                break
            frac = min(diff / total, 1.0)
            h[mask] += room * frac
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            head = h[mask]
            total = head.sum()
            if total < 1e-15:
                break
            frac = min(-diff / total, 1.0)
            h[mask] -= head * frac
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def lag_grad(r, h):
    n = len(h)
    g = np.zeros(n)
    if r == 0:
        g[:] = 1.0 - 2.0 * h
    elif r > 0:
        e = n - r
        g[:e] += 1.0 - h[r:]
        g[r:] += -h[:e]
    else:
        s = -r
        e = n - s
        g[:e] += -h[s:]
        g[s:] += 1.0 - h[:e]
    return g


def interp_to(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x0 = np.arange(n, dtype=float)
    x1 = np.linspace(0.0, float(n - 1), target_n)
    h_new = np.interp(x1, x0, h)
    return proj_feasible(h_new, target_n)


def solve_slqp(h0, n, maxiter=300):
    t0 = eval_c5(h0) + 1e-6
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * (n + 1)
    dx = 2.0 / n
    nc = 2 * n - 1

    def obj(x):
        return x[n]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def ceq(x):
        return x[:n].sum() - 0.5 * n

    def ceq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    def cineq(x):
        h = x[:n]
        t = x[n]
        corr = np.correlate(h, 1.0 - h, mode='full')
        return t - corr * dx

    def cineq_jac(x):
        h = x[:n]
        jac = np.zeros((nc, n + 1))
        jac[:, n] = 1.0
        for r in range(-(n - 1), n):
            idx = r + (n - 1)
            jac[idx, :n] = -dx * lag_grad(r, h)
        return jac

    cons = [
        {'type': 'eq', 'fun': ceq, 'jac': ceq_jac},
        {'type': 'ineq', 'fun': cineq, 'jac': cineq_jac},
    ]

    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10, 'iprint': 0})
        if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
            h_opt = proj_feasible(res.x[:n], n)
            c5 = eval_c5(h_opt)
            if np.isfinite(c5) and c5 < eval_c5(h0) + 0.01:
                return h_opt, c5
    except Exception:
        pass
    return h0.copy(), eval_c5(h0)
