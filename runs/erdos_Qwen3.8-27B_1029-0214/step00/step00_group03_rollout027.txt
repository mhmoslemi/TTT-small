import numpy as np
import time

def compute_correlation(h):
    q = 1.0 - h
    return np.correlate(h, q, mode='full')

def compute_score(h):
    N = len(h)
    return np.max(compute_correlation(h)) * (2.0 / N)

def eval_quad_max(g, D, q_idx, q_val, d_val):
    new_g = g + d_val * D
    new_g[q_idx] += d_val * d_val * q_val
    return np.max(new_g)

def golden_search_1d(g, D, q_idx, q_val, lo, hi):
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - invphi * (b - a)
    d_pt = a + invphi * (b - a)
    fc = eval_quad_max(g, D, q_idx, q_val, c)
    fd = eval_quad_max(g, D, q_idx, q_val, d_pt)
    for _ in range(40):
        if fc < fd:
            b = d_pt
            d_pt = c
            fd = fc
            c = b - invphi * (b - a)
            fc = eval_quad_max(g, D, q_idx, q_val, c)
        else:
            a = c
            c = d_pt
            fc = fd
            d_pt = a + invphi * (b - a)
            fd = eval_quad_max(g, D, q_idx, q_val, d_pt)
    return 0.5 * (a + b)

def pairwise_step(h, g, i, j):
    N = len(h)
    q = 1.0 - h
    M = 2 * N - 1
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-10:
        return h, g, False
    D = np.zeros(M)
    D[i:i+N] += q[::-1]
    D[j:j+N] -= q[::-1]
    D[N-1-i:2*N-1-i] += -h
    D[N-1-j:2*N-1-j] += h
    q_idx_list = [N - 1]
    q_val_list = [-2.0]
    if i != j:
        ib = N - 1 + i - j
        ic = N - 1 + j - i
        if 0 <= ib < M:
            q_idx_list.append(ib)
            q_val_list.append(1.0)
        if 0 <= ic < M:
            q_idx_list.append(ic)
            q_val_list.append(1.0)
    q_idx = np.array(q_idx_list, dtype=np.int64)
    q_val = np.array(q_val_list, dtype=np.float64)
    d_opt = golden_search_1d(g, D, q_idx, q_val, lo, hi)
    h_new = h.copy()
    h_new[i] += d_opt
    h_new[j] -= d_opt
    g_new = compute_correlation(h_new)
    if np.max(g_new) < np.max(g) - 1e-12:
        return h_new, g_new, True
    return h, g, False

def get_active_gradient(h, g, N):
    M = 2 * N - 1
    m_star = int(np.argmax(g))
    L = m_star - (N - 1)
    q = 1.0 - h
    grad = np.zeros(N)
    ks1 = max(0, L)
    ke1 = min(N, N + L)
    if ks1 < ke1:
        grad[ks1:ke1] += q[ks1 - L:ke1 - L]
    ks2 = max(0, -L)
    ke2 = min(N, N - L)
    if ks2 < ke2:
        grad[ks2:ke2] -= h[ks2 + L:ke2 + L]
    return grad

def refine_copy(h):
    N = len(h)
    h_new = np.empty(2 * N)
    h_new[0::2] = h
    h_new[1::2] = h
    return h_new

def fix_feasibility(h):
    N = len(h)
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = N / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(np.sum(mask))
        if cnt > 0:
            h = h.copy()
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def perturb_h(h, sigma, rng):
    N = len(h)
    h_new = h + rng.randn(N) * sigma
    return fix_feasibility(h_new)
