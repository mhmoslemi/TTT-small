import numpy as np
import time

# Global state for tracking best
_best_h = None
_best_c5 = np.inf
_deadline = 0.0

def exact_c5(h):
    n = len(h)
    if n < 2:
        return np.inf
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    val = np.max(corr) * dx
    if not np.isfinite(val):
        return np.inf
    return float(val)

def validate_h(h):
    if not np.all(np.isfinite(h)):
        return False
    if h.ndim != 1 or len(h) < 4:
        return False
    if np.any(h < -1e-12) or np.any(h > 1 + 1e-12):
        return False
    target = len(h) / 2.0
    if abs(np.sum(h) - target) > 1e-8 * max(1, len(h)):
        return False
    return True

def waterfill(y, target):
    """Given y, find lambda such that sum(clip(y+lambda, 0, 1)) = target.
    Return the clipped values."""
    n = len(y)
    # Binary search for lambda
    lo = -1.0 - np.max(np.abs(y)) - 1.0
    hi = 1.0 + np.max(np.abs(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(y + mid, 0.0, 1.0)
        s = np.sum(h)
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)
    # Fine correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Adjust interior points
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] = np.clip(h[interior] + diff / n_int, 0.0, 1.0)
    return h

def get_interior_mask(h):
    return (h > 1e-8) & (h < 1.0 - 1e-8)

def smooth_obj_grad(y, beta):
    """Compute smooth objective value and gradient w.r.t. y."""
    global _deadline
    h = waterfill(y, len(y) / 2.0)
    n = len(h)
    dx = 2.0 / n
    
    # Compute all correlation values
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    # corr has length 2n-1
    scaled = corr * dx  # These are the actual C5 candidates
    
    # Log-sum-exp (numerically stable)
    mx = np.max(scaled)
    if mx < 0:
        mx = 0.0
    beta_scaled = beta * scaled
    beta_scaled_shifted = beta_scaled - beta * mx
    exp_vals = np.exp(beta_scaled_shifted)
    log_sum = np.log(np.sum(exp_vals))
    F = mx + log_sum / beta
    
    # Weights
    Z = np.sum(exp_vals)
    if Z < 1e-300:
        weights = np.ones(len(exp_vals)) / len(exp_vals)
    else:
        weights = exp_vals / Z
    
    # Gradient w.r.t. h
    grad_h = np.zeros(n)
    # For each lag index in corr array (0 to 2n-2), compute contribution
    # corr[k] corresponds to lag s = k - (n-1)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            # c(s) = sum_{j=s}^{n-1} h[j] * (1-h[j-s])
            # dc/dh[k] = (1-h[k-s]) for k in [s, n-1] - h[k+s] for k in [0, n-1-s]
            # First term: indices s to n-1
            idx1 = np.arange(s, n)
            grad_h[idx1] += w * (1.0 - h[idx1 - s])
            # Second term: indices 0 to n-1-s
            if s > 0:
                idx2 = np.arange(0, n - s)
                grad_h[idx2] -= w * h[idx2 + s]
            else:
                # s=0: c(0) = sum h[j]*(1-h[j]), dc/dh[j] = 1-2h[j]
                # The formula gives: (1-h[j]) - h[j] = 1-2h[j] ✓
                # Actually for s=0: idx1 = 0..n-1, first term adds (1-h[j])
                # idx2 = 0..n-1, second term subtracts h[j]
                # Total: 1-2h[j] ✓ (handled above already)
                idx2 = np.arange(0, n)
                grad_h[idx2] -= w * h[idx2 + 0]
        else:
            r = -s
            # c(-r) = sum_{j=0}^{n-1-r} h[j] * (1-h[j+r])
            # dc/dh[k] = (1-h[k+r]) for k in [0, n-1-r] - h[k-r] for k in [r, n-1]
            idx1 = np.arange(0, n - r)
            grad_h[idx1] += w * (1.0 - h[idx1 + r])
            idx2 = np.arange(r, n)
            grad_h[idx2] -= w * h[idx2 - r]
    
    grad_h *= dx
    
    # Project gradient to y-space
    interior = get_interior_mask(h)
    n_int = np.sum(interior)
    grad_y = np.zeros(n)
    if n_int > 0:
        mean_grad = np.mean(grad_h[interior])
        grad_y[interior] = grad_h[interior] - mean_grad
    # Boundary points get zero gradient (can't move)
    
    return F, grad_y

def transfer_up(h, n_target):
    """Transfer h from its current resolution to n_target using linear interpolation."""
    h_current = h
    n_cur = len(h_current)
    if n_cur == n_target:
        return h_current.copy()
    x_old = np.linspace(0, 2, n_cur, endpoint=False) + 1.0/n_cur
    x_new = np.linspace(0, 2, n_target, endpoint=False) + 1.0/n_target
    h_new = np.interp(x_new, x_old, h_current, period=2)
    # Hmm, period might not make sense here. Let me just use linear interp with extension
    h_new = np.interp(x_new, x_old, h_current)
    h_new = np.clip(h_new, 0.0, 1.0)
    return waterfill(h_new, n_target / 2.0)

    variants = []
    # Direct transfer
    variants.append(transfer_up(h, n_target))
    
    # Repeated pattern (if n_target is a multiple of len(h))

        h_rep = np.tile(h, reps)

        h_sm = np.clip(h_sm, 0.0, 1.0)
        h_sm = waterfill(h_sm, n_target / 2.0)

        if time.monotonic() > deadline:
            break
        
        max_it = max_iters_per_stage[stage_idx]
        
        # Reduce iterations if time is tight
        elapsed = time.monotonic() - (deadline - 1100)  # rough

            method='L-BFGS-B',

        if 'timeout' in str(res) or not np.all(np.isfinite(res.x)):

def make_fallback(n):
    """Create a simple valid h with sum = n/2."""
    h = np.zeros(n)
    target = int(n // 2)
    # Put 1s at evenly spaced positions
    if target > 0:
        idx = np.linspace(0, n-1, target, dtype=int)
        idx = np.unique(idx)
        # Adjust if we don't have exactly target

            if new_idx not in idx:
                idx = np.append(idx, new_idx)
        while len(idx) > target:
            idx = np.delete(idx, -1)
        h[idx] = 1.0
    if n % 2 == 1:
        # Need fractional
        pass
    return waterfill(h, n / 2.0)

For s < 0, r = -s: out[k] = sum_{m=0}^{n-1-r} h[m+r] * (1-h[m])

d(out[k])/dh[j]:
- j appears in h[m+r] when m+r = j, i.e., m = j-r, provided 0 <= j-r <= n-1-r, i.e., r <= j <= n-1
  Contribution: (1-h[j-r])
- j appears in h[m] when m = j, provided 0 <= j <= n-1-r
  Contribution: -h[j+r]

So: d(out/k)/dh[j] = (1-h[j-r])*[j >= r] - h[j+r]*[j <= n-1-r]

In array operations:
`grad[r:n] += w * (1-h[0:n-r])`  (since j-r ranges from 0 to n-1-r)
`grad[0:n-r] -= w * h[r:n]`        (since j+r ranges from r to n-1)
