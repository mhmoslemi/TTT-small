    corr = np.correlate(h, 1.0 - h, mode="full")

            g[idx1] += (1.0 - h[idx1 + kp])
            idx2 = np.arange(kp, N)
            g[idx2] -= h[idx2 - kp]

import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, N):
    target = N / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / len(h)
    return float(corr.max() * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_c_k(h, k, N):
    """Gradient of c_k (output index k in the full correlation) wrt h."""
    lag = k - (N - 1)
    g = np.zeros(N)
    if lag >= 0:
        idx1 = np.arange(lag, N)
        g[idx1] += (1.0 - h[idx1 - lag])
        idx2 = np.arange(0, N - lag)
        g[idx2] -= h[idx2 + lag]
    else:
        p = -lag
        idx1 = np.arange(0, N - p)
        g[idx1] += (1.0 - h[idx1 + p])
        idx2 = np.arange(p, N)
        g[idx2] -= h[idx2 - p]
    return g

def subgradient_topm(h, N, m=3):
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-m:]
    g = np.zeros(N)
    for j in top_idx:
        g += grad_c_k(h, j, N)
    return g / m, corr

def projected_gradient_descent(h, N, max_iter=3000, time_limit=None, t_start=None):
    alpha = 1.0
    best_local = float('inf')
    best_h_local = h.copy()
    
    for it in range(max_iter):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        
        g, corr = subgradient_topm(h, N, m=3)
        c5_curr = float(corr.max() * (2.0 / N))
        
        if c5_curr < best_local:
            best_local = c5_curr
            best_h_local = h.copy()
        
        # Backtracking line search
        for _ in range(30):
            h_try = project_feasible(h - alpha * g, N)
            c5_try = float(compute_corr(h_try).max() * (2.0 / N))
            if c5_try < c5_curr + 1e-14:
                break
            alpha *= 0.5
        else:
            break
        
        h = h_try
        alpha = min(alpha * 1.1, 10.0)
    
    return best_h_local, best_local

    best_h = None

    best_n = 45

    time_per_n = budget_s * 0.7 / len(n_list)

                best_c5 = c5_res2
                best_h = h_res2
                best_n = n

import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, N):
    target = N / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / len(h)
    return float(corr.max() * dx)

def grad_c_index(h, j, N):
    lag = j - (N - 1)
    g = np.zeros(N)
    if lag >= 0:
        idx1 = np.arange(lag, N)
        g[idx1] += (1.0 - h[idx1 - lag])
        idx2 = np.arange(0, N - lag)
        g[idx2] -= h[idx2 + lag]
    else:
        p = -lag
        idx1 = np.arange(0, N - p)
        g[idx1] += (1.0 - h[idx1 + p])
        idx2 = np.arange(p, N)
        g[idx2] -= h[idx2 - p]
    return g

def smooth_grad(h, beta):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / N
    vals = beta * corr * dx
    vals = vals - vals.max()
    exp_vals = np.exp(vals)
    p = exp_vals / exp_vals.sum()
    g = np.zeros(N)
    for j in range(len(corr)):
        if p[j] > 1e-8:
            g += p[j] * grad_c_index(h, j, N)
    g *= dx
    return g

def optimize_smooth(h, N, beta_start, beta_end, max_iter, deadline):
    beta = beta_start
    alpha = 0.05
    best_local = compute_c5(h)
    best_h_local = h.copy()
    c5_curr = best_local
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        frac = it / max(1, max_iter)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        g = smooth_grad(h, beta)
        for _ in range(25):
            h_try = project_feasible(h - alpha * g, N)
            c5_try = compute_c5(h_try)
            if c5_try <= c5_curr + 1e-14:
                break
            alpha *= 0.5
        
        h = h_try
        c5_try = compute_c5(h)
        if c5_try < c5_curr:
            c5_curr = c5_try
            alpha = min(alpha * 1.15, 2.0)
            if c5_curr < best_local:
                best_local = c5_curr
                best_h_local = h.copy()
        else:
            alpha *= 0.6
            if alpha < 1e-12:
                break
    
    return best_h_local, best_local

    deadline = t_start + budget_s * 0.88
