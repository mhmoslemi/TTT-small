        active_idx = active_idx[top_order]
    
    A_ub_list = []
    b_ub_list = []
    
    for idx in active_idx:
        k = idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[0:n-k] += (1.0 - h[k:n])
            g[k:n] += (-h[0:n-k])
        else:
            m = -k
            g[m:n] += (1.0 - h[0:n-m])
            g[0:n-m] += (-h[m:n])
        
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        rhs = np.dot(g, h) - corr[idx]
        A_ub_list.append(row)
        b_ub_list.append(rhs)
    
    if not A_ub_list:
        return h
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    lb = np.maximum(0.0, h - trust_radius)
    ub = np.minimum(1.0, h + trust_radius)
    bounds = list(zip(lb, ub)) + [(None, None)]
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', options={'presolve': True})
        if result.success:
            h_new = project_box_sum(result.x[:n], target)
            return h_new
    except Exception:
        pass
    return h

def make_random_feasible(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    num_blocks = rng.randint(2, 5)
    edges = np.sort(rng.uniform(0, 2, num_blocks))
    h = np.zeros(n)
    for i in range(num_blocks):
        lo = edges[i] if i > 0 else 0.0
        hi = edges[i + 1] if i + 1 < num_blocks else 2.0
        idx_lo = int(lo * n / 2.0)
        idx_hi = int(hi * n / 2.0)
        idx_hi = max(idx_hi, idx_lo + 1)
        h[idx_lo:idx_hi] = rng.uniform(0.4, 1.0)
    return project_box_sum(h, n / 2.0)

def make_blend_feasible(h_best, n, rng, alpha=0.3):
    x = np.linspace(0, 2, n, endpoint=False)
    freq = rng.uniform(1, 4)
    phase = rng.uniform(0, 2 * np.pi)
    wave = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x + phase)
    h_best_resized = upsample_linear(h_best, n)
    blend = (1 - alpha) * h_best_resized + alpha * wave
    return project_box_sum(blend, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    
    def tleft():
        return budget_s - (time.time() - t0)
    
    rng = np.random.RandomState(seed)
    
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass
    
    if h_init is None or len(h_init) < 10:
        n_init = 45
        x = np.linspace(0, 2, n_init, endpoint=False)
        h_init = np.where(x < 1.0, 1.0, 0.0).astype(float)
        h_init = project_box_sum(h_init, n_init / 2.0)
    
    best_h = h_init.copy()
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = exact_eval(best_h)
    
    n = len(best_h)
    
    target_ns = []
    for nn in [60, 90, 120, 160, 200, 256]:
        if nn > n and nn <= 300:
            target_ns.append(nn)
    if not target_ns:
        target_ns = [n]
    
    for n_target in target_ns:
        if tleft() < 25:
            break
        
        h_curr = upsample_linear(best_h, n_target)
        c5_curr = exact_eval(h_curr)
        if c5_curr < best_c5:
            best_h = h_curr.copy()
            best_c5 = c5_curr
        
        h_work = h_curr.copy()
        betas_coarse = [30.0, 15.0, 8.0]
        betas_fine = [80.0, 40.0, 20.0]
        
        if tleft() > 60:
            h_work = gradient_optimize(h_work, betas_coarse, max_iter_per_beta=80)
            c5_w = exact_eval(h_work)
            if c5_w < best_c5:
                best_h = h_work.copy()
                best_c5 = c5_w
        
        if tleft() > 40:
            h_work = gradient_optimize(h_work, betas_fine, max_iter_per_beta=100)
            c5_w = exact_eval(h_work)
            if c5_w < best_c5:
                best_h = h_work.copy()
                best_c5 = c5_w
        
        if tleft() > 30:
            h_best_at_n = upsample_linear(best_h, n_target) if len(best_h) != n_target else best_h.copy()
            h_lp = lp_refine(h_work, max_active=80, trust_radius=0.25)
            c5_lp = exact_eval(h_lp)
            if c5_lp < best_c5:
                best_h = h_lp.copy()
                best_c5 = c5_lp
            h_lp2 = lp_refine(h_best_at_n, max_active=80, trust_radius=0.15)
            c5_lp2 = exact_eval(h_lp2)
            if c5_lp2 < best_c5:
                best_h = h_lp2.copy()
                best_c5 = c5_lp2
        
        if tleft() > 20:
            h_work = gradient_optimize(h_best_at_n if len(best_h) != n_target else best_h.copy(),
                                       [100.0, 50.0], max_iter_per_beta=100)
            c5_w = exact_eval(h_work)
            if c5_w < best_c5:
                best_h = h_work.copy()
                best_c5 = c5_w
    
    n_final = len(best_h)
    
    if tleft() > 50:
        n_explore = max(n_final, 160)
        n_restarts = max(2, int(min(8, tleft() / 25)))
        for r in range(n_restarts):
            if tleft() < 10:
                break
            h_rand = make_blend_feasible(best_h, n_explore, rng, alpha=0.25)
            h_opt = gradient_optimize(h_rand, [20.0, 50.0], max_iter_per_beta=60)
            c5_opt = exact_eval(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                n_final = len(best_h)
            if tleft() > 15:
                h_rand2 = make_random_feasible(n_explore, rng)
                h_blend = project_box_sum(0.6 * upsample_linear(best_h, n_explore) + 0.4 * h_rand2, n_explore / 2.0)
                h_opt2 = gradient_optimize(h_blend, [30.0], max_iter_per_beta=40)
                c5_opt2 = exact_eval(h_opt2)
                if c5_opt2 < best_c5:
                    best_h = h_opt2.copy()
                    best_c5 = c5_opt2
                    n_final = len(best_h)
    
    if tleft() > 15:
        h_polish = best_h.copy()
        h_polish = gradient_optimize(h_polish, [200.0, 100.0], max_iter_per_beta=150)
        c5_p = exact_eval(h_polish)
        if c5_p < best_c5:
            best_h = h_polish.copy()
            best_c5 = c5_p
            n_final = len(best_h)
    
    if tleft() > 10:
        h_lp_final = lp_refine(best_h, max_active=60, trust_radius=0.1)
        c5_lp = exact_eval(h_lp_final)
        if c5_lp < best_c5:
            best_h = h_lp_final.copy()
            best_c5 = c5_lp
            n_final = len(best_h)
    
    if tleft() > 8:
        h_polish2 = best_h.copy()
        h_polish2 = gradient_optimize(h_polish2, [300.0], max_iter_per_beta=200)
        c5_p2 = exact_eval(h_polish2)
        if c5_p2 < best_c5:
            best_h = h_polish2.copy()
            best_c5 = c5_p2
            n_final = len(best_h)
    
    n_final = len(best_h)
    best_h = project_box_sum(best_h, n_final / 2.0)
    best_c5 = exact_eval(best_h)
    
    return (best_h, best_c5, n_final)
