def project_to_feasible(x, target_sum):
    """Project x onto {0 <= h <= 1, sum(h) = target_sum} using bisection on the water level."""
    x = x.copy()
    # We want to find t such that sum(clip(x + t, 0, 1)) = target_sum
    def sum_at_level(t):
        return np.sum(np.clip(x + t, 0, 1))
    
    # Bisection
    lo, hi = -max(x) - 1, max(1 - x) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        if sum_at_level(mid) < target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    return np.clip(x + t, 0, 1)

def project_to_feasible(x, target_sum):
    n = len(x)
    x = x.copy()
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(x + t, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def smoothed_subgradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    shifted = c - c_max
    weights = np.exp(beta * shifted)
    Z = np.sum(weights)
    if Z < 1e-300:
        return np.zeros(n), c_max, 0.0
    weights /= Z
    grad = np.zeros(n)
    for idx in range(len(c)):
        w = weights[idx]
        if w < 1e-12:
            continue
        s = idx - (n - 1)
        if s >= 0:
            if n - s > 0:
                grad[s:] += w * (1.0 - h[:n-s])
                grad[:n-s] += -w * h[s:]
        else:
            sp = -s
            if n - sp > 0:
                grad[:n-sp] += w * (1.0 - h[sp:])
                grad[sp:] += -w * h[:n-sp]
    grad *= dx
    return grad, c_max, float((1.0/beta) * np.log(Z) + c_max)

    if elapsed_ok(30):
        h_opt, c5_opt = optimize_phase(best_h, n_final, budget_s, start_time, max_iter=3000, beta_init=200.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    best_c5 = evaluate_erdos_solution(best_h)
    return best_h, best_c5, len(best_h)
