```python
def project_to_feasible(h, N):
    """Project h to feasible set: 0<=h[i]<=1, sum(h)=N/2"""
    h = np.clip(h.copy(), 0.0, 1.0)
    target = N / 2.0
    current_sum = h.sum()
    diff = target - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Try to distribute the difference
    # If need to add: add to entries < 1
    # If need to remove: remove from entries > 0
    if diff > 0:
        mask = h < 1.0
        if mask.sum() == 0:
            return h
        amt = diff / mask.sum()
        h[mask] = np.minimum(h[mask] + amt, 1.0)
    else:
        mask = h > 0.0
        if mask.sum() == 0:
            return h
        amt = abs(diff) / mask.sum()
        h[mask] = np.maximum(h[mask] - amt, 0.0)

    for _ in range(100):

        if abs(resid) < 1e-12:

        if mask.sum() == 0:
            break

    return h
```

    = δ_{k,0} - δ_{k,j-i} - δ_{k,i-j} + δ_{k,0}
    = 2*δ_{k,0} - δ_{k,j-i} - δ_{k,i-j}

So c_k'(d) = G_k + d*L_k - d²*Q_k

And f(d) = max_k c_k'(d) = max_k [G_k + d*L_k - d²*Q_k]

Now, Q_k is zero for all k except k=0 (Q=2), k=j-i (Q=-1), k=i-j (Q=-1).

So for most k, c_k'(d) is linear in d. For k=0 it's concave quadratic, for k=±(j-i) it's convex quadratic.

The max of these 2N-1 functions of d is what we want to minimize.

For the grid evaluation: I can precompute G (the base correlation, shape (2N-1,)), then for each pair compute L (shape (2N-1,)) and Q (shape (2N-1,)), and evaluate:

vals = G + d_vals[:, None] * L[None, :] - d_vals[:, None]**2 * Q[None, :]
max_c = np.max(vals, axis=1)

This is much more efficient! The cost per pair is:
- Computing L: O(N) (just a few lookups)
- Computing Q: O(1) (just 3 non-zero entries)
- Evaluating vals: O(n_d * N) (matrix operations)
- Total: O(n_d * N) per pair

For N=45, n_d=51: 2300 per pair. With 1000 pairs: 2.3M per round. Very fast!

Let me implement this approach. I need to map between the "lag k" convention and the np.correlate output index.

np.correlate(h, q, mode='full') output:
- Index m = 0 corresponds to lag k = -(N-1)
- Index m = N-1 corresponds to lag k = 0
- Index m = 2N-2 corresponds to lag k = N-1

So G[m] = c_{m-(N-1)} where c_k is the correlation at shift k.

In my formula, k is the shift. So:
- G array: G[m] = c_{m-(N-1)} for m = 0, ..., 2N-2

For the linear coefficient L_k, I need to evaluate at each lag k:
L_k = q[i-k] (if 0≤i-k<N) - q[j-k] (if 0≤j-k<N) - h[i+k] (if 0≤i+k<N) + h[j+k] (if 0≤j+k<N)

For the quadratic Q_k:
Q_k = 2 if k==0 else (-1 if k==j-i or k==i-j else 0)

Let me encode this:
- k = m - (N-1) for m in range(2N-1)

So for each m:
k = m - (N-1)
L[m] = (q[i-k] if 0<=i-k<N else 0) - (q[j-k] if 0<=j-k<N else 0) - (h[i+k] if 0<=i+k<N else 0) + (h[j+k] if 0<=j+k<N else 0)

This can be computed with numpy fancy indexing with bounds checking.

```python
def pair_cd_fast(h, N, n_d=51, max_rounds=20, time_budget=None, t0=None):
    h = h.astype(float).copy()
    q = 1.0 - h
    G = np.correlate(h, q, mode='full')  # shape (2N-1,)
    current_max = float(np.max(G))
    
    num_lags = 2 * N - 1
    lags = np.arange(-(N-1), N)  # k values
    
    for rnd in range(max_rounds):
        if time_budget is not None and t0 is not None:
            if time.time() - t0 > time_budget:
                break
        perm = np.random.permutation(N)
        any_improvement = False
        
        for a in range(N):
            i = int(perm[a])
            for b in range(a + 1, N):
                j = int(perm[b])
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-9:
                    continue
                
                # Compute L and Q for this pair
                # L_k = q[i-k] - q[j-k] - h[i+k] + h[j+k] (with 0 for OOB)
                L = np.zeros(num_lags)
                for idx, k in enumerate(lags):
                    ik = i - k
                    jk = j - k
                    i_k = i + k
                    j_k = j + k
                    val = 0.0
                    if 0 <= ik < N:
                        val += q[ik]
                    if 0 <= jk < N:
                        val -= q[jk]
                    if 0 <= i_k < N:
                        val -= h[i_k]
                    if 0 <= j_k < N:
                        val += h[j_k]
                    L[idx] = val
                
                # Q_k
                Q = np.zeros(num_lags)
                for idx, k in enumerate(lags):
                    if k == 0:
                        Q[idx] = 2.0
                    elif k == (j - i):
                        Q[idx] = -1.0
                    elif k == (i - j):
                        Q[idx] = -1.0
                
                # Evaluate f(d) = max_k (G_k + d*L_k - d^2*Q_k)
                d_vals = np.linspace(d_lo, d_hi, n_d)
                vals = G[None, :] + d_vals[:, None] * L[None, :] - (d_vals[:, None])**2 * Q[None, :]
                max_c = np.max(vals, axis=1)
                
                best_idx = int(np.argmin(max_c))
                if max_c[best_idx] < current_max - 1e-12:
                    h[i] += d_vals[best_idx]
                    h[j] -= d_vals[best_idx]
                    np.clip(h, 0.0, 1.0, out=h)
                    current_max = float(max_c[best_idx])
                    # Update G and q
                    q = 1.0 - h
                    G = np.correlate(h, q, mode='full')
                    any_improvement = True
        
        if not any_improvement:
            break
    
    return h
```

Wait, I'm recomputing G after every accepted update. That's O(N²) per update (for the correlation). If I accept 100 updates per round, that's 100 * N² = 100*2025 = 200K per round. Negligible.

But actually, I can update G incrementally. When h changes at positions i and j, the correlation changes only at lags where i or j is in the relevant range. Specifically, all lags k where (i-k) ∈ [0,N) or (j-k) ∈ [0,N) or (i+k) ∈ [0,N) or (j+k) ∈ [0,N). Since k ranges over [-(N-1), N-1], and i,j ∈ [0,N), all lags are potentially affected. So I might as well recompute G.

Actually wait, the change in G at lag k when we change h[i] by d and h[j] by -d is:
ΔG_k = d * L_k - d² * Q_k

This is exactly what we computed! So I can update G in place:
G += d_best * L - d_best² * Q

And similarly for q: q[i] -= d_best, q[j] += d_best.

This avoids the O(N²) recompute. Let me use this.

But I need to be careful: after updating G and q, the L and Q for the next pair will be different (because h and q have changed). So I need to recompute L for each pair anyway (which is O(N) per pair).

```python
                break
        perm = np.random.permutation(N)
        any_improvement = False
        
        for a in range(N):
            i = int(perm[a])
            for b in range(a + 1, N):
                j = int(perm[b])
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-9:
                    continue
                
                # Compute L for this pair
                L = np.zeros(num_lags)
                for idx, k in enumerate(lags):
                    ik = i - int(k)
                    jk = j - int(k)
                    ipk = i + int(k)
                    jpk = j + int(k)
                    val = 0.0
                    if 0 <= ik < N: val += q[ik]
                    if 0 <= jk < N: val -= q[jk]
                    if 0 <= ipk < N: val -= h[ipk]
                    if 0 <= jpk < N: val += h[jpk]
                    L[idx] = val
                
                # Q
                Q = np.zeros(num_lags)
                # k=0 is at index N-1
                Q[N-1] = 2.0
                # k = j-i
                r = j - i
                if r != 0:
                    Q[(j-i) + N-1] = -1.0  # k = j-i
                    Q[(i-j) + N-1] = -1.0  # k = i-j = -r
                
                d_vals = np.linspace(d_lo, d_hi, n_d)
                vals = G[None, :] + d_vals[:, None] * L[None, :] - (d_vals[:, None])**2 * Q[None, :]
                max_c = np.max(vals, axis=1)
                
                best_idx = int(np.argmin(max_c))
                if max_c[best_idx] < current_max - 1e-12:
                    d_best = float(d_vals[best_idx])
                    h[i] += d_best
                    h[j] -= d_best
                    q[i] -= d_best
                    q[j] += d_best
                    G += d_best * L - d_best**2 * Q
                    current_max = float(max_c[best_idx])
                    any_improvement = True
        
        if not any_improvement:
            break
    
    # Final cleanup
    h = np.clip(h, 0.0, 1.0)
    return h
```

```python
import numpy as np
import time

def project_to_feasible(h, N):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = float(N) / 2.0
    for _ in range(2000):
        resid = target - h.sum()
        if abs(resid) < 1e-14:
            break
        if resid > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += resid / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def eval_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def pair_cd_fast(h, N, n_d=51, max_rounds=30, time_budget=None, t0=None, seed=42):
    h = h.astype(float).copy()
    q = 1.0 - h
    G = np.correlate(h, q, mode='full')
    current_max = float(np.max(G))
    num_lags = 2 * N - 1
    lags = np.arange(-(N - 1), N, dtype=int)
    rng = np.random.RandomState(seed)
    
    for rnd in range(max_rounds):
        if time_budget is not None and t0 is not None:
            if time.time() - t0 > time_budget:
                break
        perm = rng.permutation(N)
        any_improvement = False
        
        for a in range(N):
            i = int(perm[a])
            for b in range(a + 1, N):
                j = int(perm[b])
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-9:
                    continue
                
                ik = i - lags
                jk = j - lags
                ipk = i + lags
                jpk = j + lags
                
                L = np.zeros(num_lags)
                m1 = (ik >= 0) & (ik < N)
                m2 = (jk >= 0) & (jk < N)
                m3 = (ipk >= 0) & (ipk < N)
                m4 = (jpk >= 0) & (jpk < N)
                L[m1] += q[ik[m1]]
                L[m2] -= q[jk[m2]]
                L[m3] -= h[ipk[m3]]
                L[m4] += h[jpk[m4]]
                
                Q = np.zeros(num_lags)
                r = j - i
                Q[N - 1] = 2.0
                Q[r + N - 1] = -1.0
                Q[-r + N - 1] = -1.0
                
                d_vals = np.linspace(d_lo, d_hi, n_d)
                dv = d_vals[:, None]
                vals = G[None, :] + dv * L[None, :] - (dv ** 2) * Q[None, :]
                max_c = np.max(vals, axis=1)
                
                best_idx = int(np.argmin(max_c))
                if max_c[best_idx] < current_max - 1e-12:
                    d_best = float(d_vals[best_idx])
                    h[i] += d_best
                    h[j] -= d_best
                    q[i] -= d_best
                    q[j] += d_best
                    G += d_best * L - d_best ** 2 * Q
                    current_max = float(max_c[best_idx])
                    any_improvement = True
        
        if not any_improvement:
            break
    
    h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_h(h_old, N_old, N_new):
    dx_old = 2.0 / N_old
    x_old = (np.arange(N_old) + 0.5) * dx_old
    dx_new = 2.0 / N_new
    x_new = (np.arange(N_new) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new

def multi_start_pair_cd(h, N, n_d, max_rounds, time_budget, t0, n_starts=3):
    best_h = h.copy()
    best_c5 = eval_c5(h)
    rng = np.random.RandomState(123)
    
    for s in range(n_starts):
        if time.time() - t0 > time_budget:

        if s == 0:

        else:

        h_opt = project_to_feasible(h_opt, N)
        c5 = eval_c5(h_opt)
        if c5 < best_c5:

            best_h = h_opt.copy()
    
    return best_h, best_c5
```
