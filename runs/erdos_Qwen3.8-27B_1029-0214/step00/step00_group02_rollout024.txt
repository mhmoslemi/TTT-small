import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to increase sum
        mask = h < 1.0
        if np.sum(mask) == 0:
            return h
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    else:
        # Need to decrease sum
        mask = h > 0.0
        if np.sum(mask) == 0:
            return h
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_max_corr(h, active_k_indices):
    n = len(h)
    grad = np.zeros(n)
    for k_idx in active_k_indices:
        k = k_idx - (n - 1)
        if k >= 0:
            nk = n - k
            if nk > 0 and nk <= n:
                grad[0:nk] += (1.0 - h[k:n])
                grad[k:n] += (-h[0:nk])
        else:
            kp = -k
            two_kp = 2 * kp
            if two_kp < n:
                grad[kp:n-kp] += (1.0 - h[two_kp:n])
            if kp < n:
                grad[0:n-kp] += (-h[kp:n])
    grad /= max(1, len(active_k_indices))
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    # Start with initial
    best_h = np.array(initial_h_values, dtype=np.float64)
    best_c5 = compute_c5(best_h)
    
    n = len(best_h)
    target_sum = n / 2.0
    
    # Phase 1: Projected gradient descent with momentum
    h = best_h.copy()
    momentum = np.zeros(n)
    lr = 0.1
    mu = 0.9  # momentum coefficient
    
    for it in range(20000):
        if time.time() - t_start > budget_s - 60:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        # Active lags: those within a small epsilon of max
        tol = max(1e-4, 0.01 * max_val)
        active_idx = np.where(corr >= max_val - tol)[0]
        
        grad = gradient_max_corr(h, active_idx)
        
        # Momentum
        momentum = mu * momentum + grad
        h_new = h - lr * momentum
        h_new = project_to_constraint(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:

def project_exact(y, target_sum):
    """Project y onto {h : 0 <= h <= 1, sum(h) = target_sum} exactly."""
    n = len(y)
    if target_sum <= 0 or target_sum >= n:
        return np.clip(y, 0, 1)
    
    # Binary search for lambda
    lo, hi = y.max() - target_sum, y.min() - target_sum
    # Actually: h[i] = clip(y[i] - lambda, 0, 1)
    # sum(h) is a piecewise linear decreasing function of lambda
    # We want sum(h) = target_sum
    
    def sum_h(lam):
        return np.sum(np.clip(y - lam, 0, 1))
    
    # Find lambda such that sum_h(lam) = target_sum
    lo = y.min() - 1  # definitely gives sum >= n (all ones)
    hi = y.max()      # definitely gives sum <= 0 (all zeros)
    # Actually: when lam is very negative, all h[i] = 1, sum = n
    # When lam is very positive, all h[i] = 0, sum = 0
    lo = -1.0  # will give sum close to n
    hi = 2.0   # will give sum close to 0
    
    # Binary search
    for _ in range(100):
        mid = (lo + hi) / 2
        s = sum_h(mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2
    h = np.clip(y - lam, 0, 1)
    
    # Final exact adjustment for floating point
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        # Adjust the least-clipped elements
        if diff > 0:
            mask = (h > 0) & (h < 1)
        else:
            mask = (h > 0) & (h < 1)
        if not np.any(mask):
            mask = h < 1 if diff > 0 else h > 0
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_exact(y, target_sum):
    n = len(y)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(y - lam, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h > 0.0) & (h < 1.0)
        if not np.any(mask):
            mask = h < 1.0 if diff > 0 else h > 0.0
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        return 1.0 - 2.0 * h
    if k > 0:
        nk = n - k
        if nk <= 0:
            return grad
        grad[:nk] += 1.0 - h[k:]
        grad[k:] += -h[:nk]
    else:
        kp = -k
        two_kp = 2 * kp
        if two_kp < n:
            grad[kp:n-kp] += 1.0 - h[two_kp:]
        if kp < n:
            grad[:n-kp] += -h[kp:]
    return grad

def smooth_optimize(h0, target, time_limit, alpha_init=12.0):
    n = len(h0)
    h = project_exact(h0.copy(), target)
    best = h.copy()
    best_c5 = compute_c5(h)
    mom = np.zeros(n)
    lr = 0.04
    mu = 0.9
    alpha = alpha_init
    t_start = time.time()
    stuck = 0
    for it in range(500000):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        shifted = (corr - np.max(corr)) / alpha
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-30:
            break
        w /= ws
        grad = np.zeros(n)
        for idx in range(len(corr)):
            if w[idx] < 1e-8:
                continue
            k = idx - (n - 1)
            grad += w[idx] * grad_lag(h, k)
        mom = mu * mom + grad
        h_new = project_exact(h - lr * mom, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best = h_new.copy()
            best_c5 = c5_new
            h = h_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 50:
                lr *= 0.9
            if stuck > 500:
                lr = 0.04
                mom = np.zeros(n)
                stuck = 0
        if it % 3000 == 2999:
            alpha *= 0.6
            if alpha < 0.3:
                alpha = 0.3
    return best, best_c5

def subgrad_optimize(h0, target, time_limit):
    n = len(h0)
    h = project_exact(h0.copy(), target)
    best = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    lr = 0.03
    for it in range(100000):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        tol = max(0.02 * max_val, 0.02)
        active_idx = np.where(corr >= max_val - tol)[0]
        grad = np.zeros(n)
        for k_idx in active_idx:
            k = k_idx - (n - 1)
            grad += grad_lag(h, k)
        grad /= max(1, len(active_idx))
        h_new = project_exact(h - lr * grad, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best = h_new.copy()
            best_c5 = c5_new
            h = h_new
        else:
            lr *= 0.95
            if lr < 0.002:
                lr = 0.03
    return best, best_c5

def coordinate_polish(h, target, time_limit):
    n = len(h)
    h = project_exact(h.copy(), target)
    best = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    for rnd in range(2000):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - t_start > time_limit:
                break
            old = h[j]
            for new_val in (0.0, 1.0, 0.5):
                if abs(new_val - old) < 1e-12:
                    continue
                h[j] = new_val
                h = project_exact(h, target)
                c5_try = compute_c5(h)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    best = h.copy()
                    improved = True
                    break
                else:
                    h[j] = old
            if improved:
                h = best.copy()
        if not improved:
            break
    return best, best_c5
