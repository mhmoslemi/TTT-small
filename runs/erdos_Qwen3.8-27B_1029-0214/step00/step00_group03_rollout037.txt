```python
    grad = np.zeros(N)
    if k >= 0:
        # ∂c[k]/∂h[i] = (1-h[i-k])·[k ≤ i ≤ N-1] - h[i+k]·[0 ≤ i ≤ N-1-k]
        for i in range(k, N):
            grad[i] += (1 - h[i-k])
        for i in range(0, N-k):
            grad[i] -= h[i+k]
    else:
        k_prime = -k
        # ∂c[-k']/∂h[i] = (1-h[i+k'])·[0 ≤ i ≤ N-1-k'] - h[i-k']·[k' ≤ i ≤ N-1]
        for i in range(0, N-k_prime):
            grad[i] += (1 - h[i+k_prime])
        for i in range(k_prime, N):
            grad[i] -= h[i-k_prime]
    return grad
```

```python
def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_values(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_onto_sum(h, S):
    lo = -2.0
    hi = 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_shift_gradient(h, shift, N):
    grad = np.zeros(N)
    if 0 <= shift < N:
        idx1 = np.arange(shift, N)
        grad[idx1] += (1.0 - h[idx1 - shift])
        idx2 = np.arange(0, N - shift)
        grad[idx2] -= h[idx2 + shift]
    elif -N < shift < 0:
        p = -shift
        idx1 = np.arange(0, N - p)
        grad[idx1] += (1.0 - h[idx1 + p])
        idx2 = np.arange(p, N)
        grad[idx2] -= h[idx2 - p]
    return grad

        kernel_size = 3

        h_smooth = project_onto_sum(h_smooth, S)
        t_alloc = min(60, remaining() - 60)
        if t_alloc > 10:
            h_s, c5_s = optimize_subgrad(h_smooth, t_alloc, seed + 3000, top_k=5)
            if c5_s < best_c5:
                best_h, best_c5 = h_s, c5_s
        
        # Random restarts
        n_restarts = min(10, int(remaining() / 50))
        for i in range(n_restarts):
            if remaining() < 55:
                break
            scale = 0.1 + 0.2 * (i / max(n_restarts, 1))
            h_pert = best_h + rng.normal(0, scale, N)
            h_pert = project_onto_sum(h_pert, S)
            t_alloc = min(45, remaining() - 50)
            h_r, c5_r = optimize_subgrad(h_pert, t_alloc, seed + 4000 + i, top_k=5)
            if c5_r < best_c5:
                best_h, best_c5 = h_r, c5_r
    
    # Phase 4: Final polish with pair swaps
    if remaining() > 30:
        t_alloc = min(remaining() - 20, 120)
        h_f, c5_f = pair_swap_improve(best_h, t_alloc, seed + 9999)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f
    
    # Phase 5: One more subgradient pass
    if remaining() > 30:
        t_alloc = min(remaining() - 15, 100)
        h_f2, c5_f2 = optimize_subgrad(best_h, t_alloc, seed + 8888, top_k=3)
        if c5_f2 < best_c5:
            best_h, best_c5 = h_f2, c5_f2
    
    return best_h, best_c5, len(best_h)
```
