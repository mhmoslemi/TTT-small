```python
def compute_G(h, n):
    n_lags = 2 * n - 1
    G = np.zeros((n, n_lags))
    for idx in range(n_lags):
        lag = idx - (n - 1)
        if lag == 0:
            G[:, idx] = 1.0 - 2.0 * h
        elif lag > 0:
            # G[i, idx] = (1-h[i+lag]) for i < n-lag, 0 otherwise
            #           - h[i-lag] for i >= lag, 0 otherwise
            term1 = np.zeros(n)
            term1[:n-lag] = 1.0 - h[lag:]
            term2 = np.zeros(n)
            term2[lag:] = h[:n-lag]
            G[:, idx] = term1 - term2
        else:
            nl = -lag
            term1 = np.zeros(n)
            term1[nl:] = 1.0 - h[:nl]
            term2 = np.zeros(n)
            term2[:n-nl] = h[nl:]
            G[:, idx] = term1 - term2
    return G
```

        k = idx - (n - 1)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            term1 = np.zeros(n)
            term1[k:] = 1.0 - h[:n-k]
            term2 = np.zeros(n)
            term2[:n-k] = h[k:]
            grad += w * (term1 - term2)
        else:
            nk = -k
            term1 = np.zeros(n)
            term1[:n-nk] = 1.0 - h[nk:]
            term2 = np.zeros(n)
            term2[nk:] = h[:n-nk]
            grad += w * (term1 - term2)
    
    return grad

def project(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def subgrad_opt(h, n, time_limit=60, beta=50.0):
    h = h.copy()
    h = project(h, n)
    best_h = h.copy()
    best_val = compute_c5(h, n)
    start = time.time()
    base_step = 0.5
    
    while time.time() - start < time_limit:
        grad = compute_grad(h, n, beta=beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        step = base_step / (1.0 + 0.005 * (time.time() - start))
        h_new = h - step * grad
        h_new = project(h_new, n)
        val = compute_c5(h_new, n)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
            h = h_new
        else:
            base_step *= 0.95
            if base_step < 0.001:
                break
    
    return best_h, best_val

def pair_swap(h, n, time_limit=30):
    h = h.copy()
    dx = 2.0 / n
    c_vals = np.correlate(h, 1.0 - h, mode='full') * dx
    best_val = np.max(c_vals)
    best_h = h.copy()
    start = time.time()
    
    def compute_G(h):
        G = np.zeros((n, 2 * n - 1))
        for idx in range(2 * n - 1):
            k = idx - (n - 1)
            if k == 0:
                G[:, idx] = 1.0 - 2.0 * h
            elif k > 0:
                t1 = np.zeros(n)
                t1[k:] = 1.0 - h[:n-k]
                t2 = np.zeros(n)
                t2[:n-k] = h[k:]
                G[:, idx] = t1 - t2
            else:
                nk = -k
                t1 = np.zeros(n)
                t1[:n-nk] = 1.0 - h[nk:]
                t2 = np.zeros(n)
                t2[nk:] = h[:n-nk]
                G[:, idx] = t1 - t2
        return G
    
    G = compute_G(h)
    n_grid = 15
    
    for sweep in range(200):
        if time.time() - start > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - start > time_limit:
                break
            for j in range(i + 1, n):
                if time.time() - start > time_limit:
                    break
                d = G[i] - G[j]
                dmin = -h[i]
                dmax = h[j]
                if dmax <= dmin + 1e-12:
                    continue
                # Find best delta
                deltas = np.linspace(dmin, dmax, n_grid)
                best_delta = 0.0
                best_obj = best_val
                for delta in deltas:
                    obj = np.max(c_vals + delta * d)
                    if obj < best_obj:
                        best_obj = obj
                        best_delta = delta
                if abs(best_delta) > 1e-14 and best_obj < best_val - 1e-14:
                    h[i] += best_delta
                    h[j] -= best_delta
                    best_val = best_obj
                    best_h = h.copy()
                    improved = True
                    c_vals = c_vals + best_delta * d
                    G = compute_G(h)
        if not improved:
            break
    
    return best_h, best_val
