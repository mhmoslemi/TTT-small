    dx = 2.0 / n

    corr = np.correlate(h, 1 - h, mode="full")

    max_val = np.max(corr)

    argmax_indices = np.where(corr == max_val)[0]

        _add_lag_gradient(grad, h, lag, n, dx)

    if lag >= 0:
        # f_lag = dx * sum_{j=0}^{n-lag-1} h[j] * (1-h[j+lag])
        # d/dh[i] = dx * [(1-h[i+lag]) if i < n-lag, minus h[i-lag] if i >= lag]

        #   If i < n-m: h[i] appears in (1-h[i]) term -> derivative is -dx*h[i+m]

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(y, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_gradient(h, beta):
    """Compute gradient of smooth max of correlation values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    # Numerical stability
    corr_shifted = corr - np.max(corr)
    weights = np.exp(beta * corr_shifted)
    weights /= np.sum(weights)
    # corr has length 2n-1
    # Index i in corr corresponds to: sum_j h[j] * (1-h)[j + i - (n-1)]
    # The "lag" is i - (n-1)
    grad = np.zeros(n)
    for i in range(len(corr)):
        w = weights[i]
        if w < 1e-10:
            continue
        lag = i - (n - 1)
        if lag >= 0:
            # contrib = w * dx * grad of lag
            m = lag
            if m == 0:
                grad += w * dx * (1 - 2 * h)
            else:
                grad[:n-m] += w * dx * (1 - h[m:])
                grad[m:] -= w * dx * h[:n-m]
        else:
            m = -lag
            grad[m:] += w * dx * (1 - h[:n-m])
            grad[:n-m] -= w * dx * h[m:]
    return grad

    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    # Project to ensure constraint is satisfied
    h = project_onto_constraint(h, n)
    
    # Optimized: projected gradient descent with smooth max
    beta = 50.0  # sharpness of max
    alpha = 0.1  # step size
    n_iter = 500
    
    for it in range(n_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 60:
            break
        
        grad = smooth_max_gradient(h, beta)
        h_new = h - alpha * grad
        h_new = project_onto_constraint(h_new, n)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            # Reduce step size when improving
            alpha = max(0.01, alpha * 0.999)
        else:
            # Reject and reduce step size
            alpha *= 0.9
            if alpha < 0.001:
                break
        h = h_new
    
    # Now try to increase resolution
    current_n = len(best_h)
    target_resolutions = [90, 180, 360, 720]
    
    for target_n in target_resolutions:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 120:

        # Upsample by repetition
        upsample_factor = target_n // current_n
        if upsample_factor < 1:

        h_up = np.repeat(best_h, upsample_factor)
        current_n = target_n
        h_up = project_onto_constraint(h_up, current_n)

        alpha = 0.05
        for it in range(300):
            elapsed = time.time() - start_time
            if elapsed > budget_s - 60:

            grad = smooth_max_gradient(h_up, beta)
            h_new = h_up - alpha * grad
            h_new = project_onto_constraint(h_new, current_n)

            else:
                alpha *= 0.9
                if alpha < 0.0005:

            h_up = h_new

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(y, n):
    """Euclidean projection onto {x in [0,1]^n : sum(x) = n/2}."""
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient(h, weights, threshold=1e-12):
    """Compute gradient of weighted sum of correlation values."""
    n = len(h)
    grad = np.zeros(n)
    sig_indices = np.where(weights > threshold)[0]
    for idx in sig_indices:
        lag = int(idx) - (n - 1)
        w = weights[idx]
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            m = lag
            grad[:n-m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n-m]
        else:
            m = -lag
            grad[m:] += w * (1.0 - h[:n-m])
            grad[:n-m] -= w * h[m:]
    return grad

    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    alpha = alpha_start
    beta = beta_start

Actually, let me think about this more carefully. If I change h[i] by δ and h[j] by -δ, how does c_m change?

c_m = sum_{k=0}^{n-m-1} h[k]*(1-h[k+m]) for positive m.

The change in c_m from changing h[i] by δ and h[j] by -δ:
- h[i] appears in c_m if: i < n-m (as first factor) or i >= m (as second factor, in 1-h[i-m])
  - As first factor: contributes δ*(1-h[i+m])
  - As second factor: contributes -δ*h[i-m] (since 1-h[i-m] changes by -δ)
- h[j] appears similarly with -δ

So Δc_m = δ*(1-h[i+m])*1_{i<n-m} - δ*h[i-m]*1_{i>=m} - δ*(1-h[j+m])*1_{j<n-m} + δ*h[j-m]*1_{j>=m}

= δ * [(1-h[i+m])*1_{i<n-m} - h[i-m]*1_{i>=m} - (1-h[j+m])*1_{j<n-m} + h[j-m]*1_{j>=m}]

Similarly for negative lags.

So for a given pair (i,j) and a given set of significant lags, I can compute the change in each correlation value as a linear function of δ. Then I need to find the δ that minimizes the max over lags.

This is a piecewise-linear minimization problem in 1D, which can be solved exactly.

This is the "pairs" local search approach. It's more powerful than gradient descent because it can make exact moves (not approximate).

def pair_local_search(h, max_passes=10, start_time=None, budget_s=None):
    """Local search by trying all pairs (i, j)."""
    n = len(h)
    dx = 2.0 / n
    best_c5 = evaluate_c5(h)
    h = h.copy()
    
    for pass_idx in range(max_passes):
        if start_time is not None and time.time() > budget_s - 30:
            break
        
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                # Try transferring mass from j to i (or vice versa)
                # h[i] increases by δ, h[j] decreases by δ
                # Need: 0 <= h[i]+δ <= 1 and 0 <= h[j]-δ <= 1
                # So δ in [max(0, h[i]-1+1), min(1-h[i], h[j])] = [0, min(1-h[i], h[j])]
                # And also the negative direction: δ in [-min(h[i], 1-h[j]), 0]
                
                # Compute current correlations
                corr = np.correlate(h, 1.0 - h, mode="full")
                
                # Compute Δc for each lag as a function of δ
                # For simplicity, just try δ = ±0.01, ±0.05, ±0.1
                for delta in [0.01, 0.05, 0.1, 0.2]:
                    if h[i] + delta <= 1.0 + 1e-9 and h[j] - delta >= -1e-9:
                        h[i] += delta
                        h[j] -= delta
                        h = np.clip(h, 0, 1)
                        new_c5 = evaluate_c5(h)
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
                
                    if h[j] + delta <= 1.0 + 1e-9 and h[i] - delta >= -1e-9:
                        h[j] += delta

                        h = np.clip(h, 0, 1)

                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            improved = True
                        else:
                            h[j] -= delta
                            h[i] += delta
        
        if not improved:

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(y, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-12:
            continue
        lag = idx - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            m = lag
            grad[:n-m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n-m]
        else:
            m = -lag
            grad[m:] += w * (1.0 - h[:n-m])
            grad[:n-m] -= w * h[m:]
    return grad

def optimize_pgd(h, max_iters, start_time, deadline, beta0=30.0, alpha0=0.3):
    n = len(h)
    h = project_onto_constraint(h, n).copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    alpha = alpha0
    beta = beta0
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        
        # Softmax weights
        shifted = corr - np.max(corr)
        weights = np.exp(beta * shifted)
        weight_sum = np.sum(weights)
        if weight_sum < 1e-300:
            break
        weights /= weight_sum
        
        grad = compute_gradient(h, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Step
        h_new = h - alpha * grad
        h_new = project_onto_constraint(h_new, n)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            alpha = min(0.5, alpha * 1.02)
            beta = min(200.0, beta * 1.01)
            stall = 0
        else:
            alpha *= 0.85
            stall += 1
            if stall > 50:
                # Random restart perturbation
                h = project_onto_constraint(h + 0.05 * np.random.randn(n), n)
                alpha = alpha0 * 0.5
                stall = 0
            if alpha < 0.0003:
                break
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    deadline = start_time + budget_s - 10
    np.random.seed(seed)
    
    # Get initial values

    if 'initial_h_values' in dir():

        except (NameError, TypeError):

    h_opt, c5_opt = optimize_pgd(best_h, 500, start_time, deadline, beta0=30, alpha0=0.3)
    if c5_opt < best_c5:

        best_c5 = c5_opt

        best_h = np.repeat(best_h, up_factor)
        n_curr = len(best_h)
        best_h = project_onto_constraint(best_h, n_curr)

        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    return best_h, best_c5, len(best_h)

import numpy as np
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraint(y, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)


def compute_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-14:
            continue
        lag = idx - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            m = lag
            grad[:n - m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n - m]
        else:
            m = -lag
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
    return grad


def optimize_pgd(h, max_iters, deadline, beta, alpha, seed_val):
    n = len(h)
    rng = np.random.RandomState(seed_val)
    h = project_onto_constraint(h, n).copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        shifted = corr - np.max(corr)
        weights = np.exp(beta * shifted)
        ws = np.sum(weights)
        if ws < 1e-300:
            break
        weights /= ws
        grad = compute_gradient(h, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        h_new = project_onto_constraint(h - alpha * grad, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            alpha = min(0.5, alpha * 1.01)
            beta = min(300.0, beta * 1.005)
            stall = 0
        else:
            alpha *= 0.88
            stall += 1
            if stall > 60:
                h = project_onto_constraint(h + 0.04 * rng.randn(n), n)
                alpha = 0.25
                stall = 0
            if alpha < 0.0004:
                break
    return best_h, best_c5


def make_alternative_init(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = 0.5 + 0.3 * rng.randn(n)
    return project_onto_constraint(h, n)


def make_thue_morse_init(n):
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    h = np.where(t > 0.5, 1.0, 0.0)
    target = n / 2.0
    n_ones = int(np.sum(h))
    if n_ones < target:
        zeros = np.where(h == 0.0)[0]
        need = int(target) - n_ones
        if need > 0 and len(zeros) >= need:
            h[zeros[:need]] = 1.0
    elif n_ones > target:
        ones = np.where(h == 1.0)[0]
        need = n_ones - int(target)
        if need > 0 and len(ones) >= need:
            h[ones[-need:]] = 0.0
    return project_onto_constraint(h, n)
