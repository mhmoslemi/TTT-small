def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def log_sum_exp(corr_scaled, beta):
    """Compute (1/beta) * log(sum(exp(beta * x))) in a numerically stable way."""
    max_val = np.max(corr_scaled)
    if max_val == 0:
        return 0.0
    shifted = corr_scaled - max_val
    return max_val + np.log(np.sum(np.exp(beta * shifted))) / beta

def log_sum_exp_grad(corr_scaled, beta):
    """Compute (w_idx) weights = softmax(beta * corr_scaled)."""
    max_val = np.max(corr_scaled)
    shifted = beta * (corr_scaled - max_val)
    exp_shifted = np.exp(shifted)
    return exp_shifted / np.sum(exp_shifted)

def obj_and_grad(h, n, dx, beta):
    """Objective and gradient for log-sum-exp of correlation terms."""
    corr = np.correlate(h, 1 - h, mode="full")
    corr_scaled = corr * dx  # length 2n-1
    
    # Objective
    obj = log_sum_exp(corr_scaled, beta)
    
    # Weights (softmax)
    w = log_sum_exp_grad(corr_scaled, beta)
    
    # Gradient: sum over idx of w[idx] * dx * d/dh[m] g[idx]
    # where g[idx] is the unscaled correlation
    grad = np.zeros(n)
    
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        k = idx - (n - 1)  # the lag
        n_lo = max(0, -k)
        n_hi = n - 1 - max(0, k)
        
        # d/dh[m] g[idx] = (1-h[m-k]) * [n_lo <= m-k <= n_hi] - h[m] * [n_lo <= m <= n_hi]
