```python
                # Simple approach: ternary search on [lo_d, hi_d]
                # f is convex (max of convex functions), so ternary search works.
                
                def f(d):
                    vals = g + d * D
                    if Q[m1] > 0:  # m1 in range
                        vals = vals.copy()
                        vals[m1] += d * d
                    if Q[m2] > 0 and m2 != m1:
                        vals[m2] += d * d
                    return np.max(vals)
                
                # But making a copy each time is expensive. Let me optimize.
                # Actually for N=45, M=89, this is fine.
                
                # Ternary search
                a = lo_d
                b = hi_d
                for _ in range(60):  # 60 iterations gives precision ~ (b-a)/phi^60 ≈ (b-a)*1e-13
                    m1t = a + (b - a) / 3.0
                    m2t = b - (b - a) / 3.0
                    fa = f(m1t)
                    fb = f(m2t)
                    if fa < fb:
                        b = m2t
                    else:
                        a = m1t
                
                d_opt = (a + b) / 2.0
                f_opt = f(d_opt)
                
                # Check if this improves
                if f_opt < best_max - tol:
                    # Apply the update
                    h[i] += d_opt
                    h[j] -= d_opt
                    q[i] -= d_opt
                    q[j] += d_opt
                    # Update g
                    g = g + d_opt * D + d_opt * d_opt * Q
                    # Clip small negatives
                    g = np.maximum(g, 0)  # g should be non-negative
                    best_max = f_opt
                    improved = True
        
        if not improved:
            break
    
    return h, best_max
```

```python
def _pairwise_optimize(h, max_sweeps=200, tol=1e-12):
    """Perform pairwise descent on h. Returns (h, best_c5)."""

    dx = 2.0 / n
    M = 2 * n - 1
    q = 1.0 - h
    g = np.correlate(h, q, mode='full').copy()

    # D[i:i+n] += q[::-1]
    # D[n-1-i:2*n-1-i] -= h
    # D[j:j+n] -= q[::-1]
    # D[n-1-j:2*n-1-j] += h

                m2 = j - i + (n - 1)

                for _ in range(50):

                    m_b = b - (b - a) / 3.0

                    vals = g + m_a * D
                    if 0 <= m1 < M:
                        vals[m1] += m_a * m_a
                    if m2 != m1 and 0 <= m2 < M:
                        vals[m2] += m_a * m_a
                    fa = np.max(vals)
                    
                    # Evaluate f at m_b
                    vals = g + m_b * D
                    if 0 <= m1 < M:
                        vals[m1] += m_b * m_b
                    if m2 != m1 and 0 <= m2 < M:
                        vals[m2] += m_b * m_b
                    fb = np.max(vals)
                    
                    if fa < fb:
                        b = m_b
                    else:
                        a = m_a

                # Evaluate at d_opt
                vals = g + d_opt * D
                if 0 <= m1 < M:
                    vals[m1] += d_opt * d_opt

                    vals[m2] += d_opt * d_opt
                f_opt = np.max(vals)

                    g = vals  # already computed

                    improved = True
        
        if not improved:

    return h, best_max
```

```python
import numpy as np
import time

def _pairwise_sweep(h, max_sweeps=100, tol=1e-11, time_limit=None, start_time=None):
    """Perform pairwise descent. Returns (h, best_c5, num_sweeps_done)."""
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    q = 1.0 - h
    g = np.correlate(h, q, mode='full').copy()
    best_max_unscaled = np.max(g)
    
    q_rev = q[::-1].copy()
    best_c5 = best_max_unscaled * dx
    sweeps_done = 0
    
    for sweep in range(max_sweeps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 5:
                break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo_d = max(-h[i], h[j] - 1.0)
                hi_d = min(1.0 - h[i], h[j])
                if hi_d - lo_d < 1e-14:
                    continue
                
                # Compute D vector (linear coefficients)
                D = np.zeros(M)
                D[i:i+n] += q_rev
                D[n-1-i:2*n-1-i] -= h
                D[j:j+n] -= q_rev
                D[n-1-j:2*n-1-j] += h
                
                # Quadratic indices
                m1 = i - j + (n - 1)
                m2 = j - i + (n - 1)
                
                # Check if quadratic terms are in range
                has_q1 = (0 <= m1 < M)
                has_q2 = (0 <= m2 < M and m2 != m1)
                
                # Ternary search for optimal d
                a, b = lo_d, hi_d
                # Reduce iterations for speed; 40 gives ~1e-8 precision on [0,1]
                for _ in range(40):
                    m_a = a + (b - a) / 3.0
                    m_b = b - (b - a) / 3.0
                    
                    vals = g + m_a * D
                    if has_q1:
                        vals[m1] += m_a * m_a
                    if has_q2:
                        vals[m2] += m_a * m_a
                    fa = np.max(vals)
                    
                    vals = g + m_b * D
                    if has_q1:
                        vals[m1] += m_b * m_b
                    if has_q2:
                        vals[m2] += m_b * m_b
                    fb = np.max(vals)
                    
                    if fa < fb:
                        b = m_b
                    else:
                        a = m_a
                
                d_opt = (a + b) / 2.0
                
                # Final evaluation
                vals = g + d_opt * D
                if has_q1:
                    vals[m1] += d_opt * d_opt
                if has_q2:
                    vals[m2] += d_opt * d_opt
                f_opt = np.max(vals)
                
                if f_opt < best_max_unscaled - tol:
                    h[i] += d_opt
                    h[j] -= d_opt
                    q[i] -= d_opt
                    q[j] += d_opt
                    q_rev[n-1-i] -= d_opt
                    q_rev[n-1-j] += d_opt
                    g = vals
                    best_max_unscaled = f_opt
                    best_c5 = f_opt * dx
                    improved = True
        
        sweeps_done += 1
        if not improved:
            break
    
    return h, best_c5, sweeps_done


def _upsample(h, factor=2):
    """Upsample h by a factor using nearest neighbor, preserving sum."""
    n = len(h)
    new_n = n * factor
    h_new = np.repeat(h, factor)
    # Check sum
    # sum(h_new) = factor * sum(h) = factor * n/2 = new_n/2 ✓
    return h_new


def _perturb_and_descent(h, n_perturb=50, max_sweeps=50, tol=1e-11, rng=None):
    """Perturb h with random pair moves, then do pairwise descent."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(h)
    h_pert = h.copy()
    
    for _ in range(n_perturb):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        lo_d = max(-h_pert[i], h_pert[j] - 1.0)
        hi_d = min(1.0 - h_pert[i], h_pert[j])
        if hi_d - lo_d < 1e-14:
            continue
        d = rng.uniform(lo_d, hi_d) * 0.3  # small perturbation
        h_pert[i] += d
        h_pert[j] -= d
    
    # Ensure constraints
    h_pert = np.clip(h_pert, 0, 1)
    # Adjust sum
    current_sum = h_pert.sum()
    target_sum = n / 2.0
    diff = target_sum - current_sum
    if abs(diff) > 1e-14:
        # Distribute the difference
        n_pos = np.sum(h_pert < 1.0)
        n_neg = np.sum(h_pert > 0.0)
        n_avail = n_pos + n_neg
        if n_avail > 0:
            h_pert += diff / n_avail
            h_pert = np.clip(h_pert, 0, 1)
            # Re-adjust (iteration may not be exact)
            # For simplicity, just accept small error
    
    # Do descent
    h_result, c5, _ = _pairwise_sweep(h_pert, max_sweeps=max_sweeps, tol=tol)
    return h_result, c5
```
