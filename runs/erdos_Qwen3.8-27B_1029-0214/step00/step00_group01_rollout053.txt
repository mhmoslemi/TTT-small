def forward_model(y, beta):

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx

    # Gradient w.r.t. h

        w = weights[k + n - 1]

        elif k > 0:

            G[r:] += w * dx * (1 - h[:n-r])
            G[:n-r] += w * dx * (-h[r:])
        else:
            r = -k
            G[:n-r] += w * dx * (1 - h[r:])
            G[r:] += w * dx * (-h[:n-r])

    # h = waterfill(expit(y), n/2)
    # For interior points of waterfill (0 < h < 1):
    #   h_i = expit(y_i) - tau, where tau is the waterfill level
    #   d tau / d y_j = expit'(y_j) / sum_{i in S_int} expit'(y_i) if j in S_int

    # Actually, let me just use finite differences for the gradient.

    #   The gradient of F w.r.t. y at an interior point is the same as the gradient
    #   projected onto the affine subspace {sum = n/2}, restricted to interior coordinates.
    #   This is: for j in S_int: dF/dy_j = G_j - mean_{i in S_int}(G_i)

    # Let me just use a simpler parameterization: h = waterfill(y, n/2) directly.

    # dF/dy_j = G_j - mean(G[S_int])
    # For boundary j: dF/dy_j = 0 (we'll let the optimizer's bound handling take care of it)
    
    return log_sum, G

import numpy as np
import time
from scipy.optimize import minimize

def exact_C5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(c.max())

def waterfill(h, target):
    n = len(h)
    if n == 0:
        return h
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def compute_lse(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    F = c_max + np.log(exp_c.sum()) / beta
    weights = exp_c / exp_c.sum()
    return F, weights

def compute_grad_h(h, weights, beta):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros(n)
    for k in range(-(n - 1), n):
        w = weights[k + n - 1]
        if w < 1e-14:
            continue
        if k == 0:
            G += w * dx * (1.0 - 2.0 * h)
        elif k > 0:
            r = k
            G[r:] += w * dx * (1.0 - h[:n - r])
            G[:n - r] += w * dx * (-h[r:])
        else:
            r = -k
            G[:n - r] += w * dx * (1.0 - h[r:])
            G[r:] += w * dx * (-h[:n - r])
    return G

def projected_grad_step(h, beta, alpha, deadline):
    n = len(h)
    F_old, weights = compute_lse(h, beta)
    G = compute_grad_h(h, weights, beta)
    h_new = waterfill(h - alpha * G, n / 2.0)
    F_new, _ = compute_lse(h_new, beta)
    backtrack = 0
    while F_new >= F_old and backtrack < 40:
        alpha *= 0.5
        h_new = waterfill(h - alpha * G, n / 2.0)
        F_new, _ = compute_lse(h_new, beta)
        backtrack += 1
    if F_new >= F_old:
        return h, alpha, False
    return h_new, alpha, True

def optimize_continuation(h_start, beta_list, max_iter_per_beta, deadline):
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_c = exact_C5(h)
    alpha = 1.0

    for beta in beta_list:
        alpha = 2.0 / (beta * (2.0 / n) * n)
        alpha = min(alpha, 5.0)
        for it in range(max_iter_per_beta):
            if time.monotonic() > deadline:
                break
            h, alpha, improved = projected_grad_step(h, beta, alpha, deadline)
            if not improved:
                break
            if it % 10 == 0:
                c_exact = exact_C5(h)
                if c_exact < best_c:
                    best_c = c_exact
                    best_h = h.copy()
        c_exact = exact_C5(h)
        if c_exact < best_c:
            best_c = c_exact
            best_h = h.copy()
        if time.monotonic() > deadline:
            break

    c_exact = exact_C5(h)
    if c_exact < best_c:
        best_c = c_exact
        best_h = h.copy()
    return best_h, best_c, h

def transfer_resolution(h, n_target):
    n_src = len(h)
    if n_src == n_target:
        return waterfill(h.copy(), n_target / 2.0)
    if n_target % n_src == 0:
        ratio = n_target // n_src
        h_new = np.repeat(h, ratio)
    else:
        indices = (np.arange(n_target) * n_src / n_target).astype(int)
        indices = np.clip(indices, 0, n_src - 1)
        h_new = h[indices]
    return waterfill(h_new, n_target / 2.0)

def local_search_perturb(h, n_perturb, deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_c = exact_C5(h)

        if c_try < best_c - 1e-10:
            best_h = h_try.copy()
            best_c = c_try
    return best_h, best_c
