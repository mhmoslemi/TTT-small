from scipy.optimize import minimize
import time

def compute_c5_array(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1 - h, mode="full") * dx

def c5_value(h):
    return float(np.max(compute_c5_array(h)))

def project_box_sum(x, target_sum):
    n = len(x)
    lo = -x.max() - 2.0
    hi = 2.0 - x.min()
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = compute_c5_array(h)
    c_max = c.max()
    logits = beta * (c - c_max)
    weights = np.exp(logits)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k_idx in range(len(c)):
        w = weights[k_idx]
        if w < 1e-8:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            grad[0:n-k] += w * (1.0 - h[k:n])
            grad[k:n] += w * (-h[0:n-k])
        else:
            grad[-k:n] += w * (1.0 - h[0:n+k])
            grad[0:n+k] += w * (-h[-k:n])
    return grad * dx

def optimize_smooth(h_init, n, max_iter=3000, beta_start=10, beta_end=200):
    """Optimize using projected gradient with increasing beta (sharpness)."""
    h = h_init.copy()
    target_sum = n / 2.0
    best_h = h.copy()
    best_val = c5_value(h)
    
    for it in range(max_iter):
        progress = it / max(max_iter - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        g = smooth_max_grad(h, beta)
        
        # Adaptive step size
        # Use a diminishing step size
        alpha = 1.0 / (10.0 + 0.1 * it)
        
        h_new = h - alpha * g
        h = project_box_sum(h_new, target_sum)
        
        val = c5_value(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    return best_h, best_val

    def obj(x):
        return float(np.max(np.correlate(x, 1 - x, mode="full"))) * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0, 1.0)] * n
    
    res = minimize(obj, h_init, method='SLSQP', bounds=bounds,

    h_out = np.clip(res.x, 0.0, 1.0)
