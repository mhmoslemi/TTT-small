    h0 *= (n0 / 2.0) / s0
    np.clip(h0, 0.0, 1.0, out=h0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    best_n = n0

    h_opt, c5_opt = pairwise_descent(h0, max_passes=200, tol=1e-10, n_grid=51,
                                      time_limit=time_limit, start_time=start_time)
    if c5_opt < best_c5:
        best_h = h_opt.copy()
        best_c5 = c5_opt

    n_restarts = 5
    for rs in range(n_restarts):
        if time.time() - start_time > time_limit:
            break
        h_pert = random_perturb(h_opt, 20, rng)
        h_opt2, c5_opt2 = pairwise_descent(h_pert, max_passes=80, tol=1e-10, n_grid=51,
                                            time_limit=time_limit, start_time=start_time)
        if c5_opt2 < best_c5:
            best_h = h_opt2.copy()
            best_c5 = c5_opt2
            h_opt = h_opt2.copy()

    for scale_pass in range(3):
        if time.time() - start_time > time_limit:
            break
        h_up = upsample(best_h if scale_pass == 0 else h_opt, 2)
        n_up = len(h_up)
        sg = 41 if n_up <= 100 else 31
        mp = 50 if n_up <= 100 else (30 if n_up <= 200 else 15)
        h_opt, c5_opt = pairwise_descent(h_up, max_passes=mp, tol=1e-9, n_grid=sg,
                                          time_limit=time_limit, start_time=start_time)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n = n_up
        else:
            h_opt = best_h.copy()

    if time.time() - start_time < time_limit:
        n_final = len(best_h)
        if n_final <= 180:
            h_up = upsample(best_h, 2)
            n_up = len(h_up)
            sg = 31
            h_opt2, c5_opt2 = pairwise_descent(h_up, max_passes=10, tol=1e-9, n_grid=sg,
                                                time_limit=time_limit, start_time=start_time)
            if c5_opt2 < best_c5:
                best_h = h_opt2.copy()
                best_c5 = c5_opt2
                best_n = n_up

    s = np.sum(best_h)
    target = best_n / 2.0
    diff = s - target
    if abs(diff) > 1e-15:
        interior = np.where((best_h > 1e-6) & (best_h < 1.0 - 1e-6))[0]
        if len(interior) > 0:
            best_h[interior[0]] -= diff
        else:
            best_h[0] -= diff
        np.clip(best_h, 0.0, 1.0, out=best_h)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
