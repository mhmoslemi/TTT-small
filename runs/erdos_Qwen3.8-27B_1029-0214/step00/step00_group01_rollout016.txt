import numpy as np
from scipy.optimize import minimize
from scipy.special import expit
import time

def waterfill_project(h, target_sum):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum}"""
    n = len(h)
    # Bisection on tau
    lo = h.min() - target_sum / n - 1
    hi = h.max() + target_sum / n + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)
    return proj

def exact_C5(h):
    """Compute exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def compute_c_values(h):
    """Compute c_s for all lags s."""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return corr * dx  # array of length 2n-1

def smooth_objective_and_gradient(h, beta):
    """Compute log-sum-exp surrogate and its gradient."""
    n = len(h)
    dx = 2.0 / n
    c_vals = compute_c_values(h)
    
    # Stable log-sum-exp
    c_max = c_vals.max()
    exp_vals = np.exp(beta * (c_vals - c_max))
    sum_exp = exp_vals.sum()
    
    obj = c_max + np.log(sum_exp) / beta
    
    # Weights
    w = exp_vals / sum_exp
    
    # Gradient
    G = np.zeros(n)
    one_minus_h = 1.0 - h
    for p in range(2 * n - 1):
        s = p - (n - 1)
        wp = w[p]
        if wp < 1e-15:
            continue
        if s >= 0:
            L = n - s
            # c_s = dx * sum_{i=0}^{L-1} h[i] * (1-h[i+s])
            G[:L] += wp * dx * one_minus_h[:L]
            G[s:] -= wp * dx * h[:L]
        else:
            r = -s
            L = n - r
            # c_s = dx * sum_{i=0}^{L-1} h[i+r] * (1-h[i])
            G[r:] += wp * dx * one_minus_h[:L]
            G[:L] -= wp * dx * h[r:]
    
    return obj, G

def optimize_smooth(h_start, beta_schedule, max_iters_per_beta=50, deadline=None):
    """Optimize using smooth surrogate with projected gradient descent."""
    n = len(h_start)
    target_sum = n / 2.0
    best_h = h_start.copy()
    best_c = exact_C5(h_start)
    h = h_start.copy()
    
    for beta in beta_schedule:
        for it in range(max_iters_per_beta):
            if deadline is not None and time.monotonic() > deadline:
                return best_h, best_c
            
            obj, G = smooth_objective_and_gradient(h, beta)
            
            # Project gradient onto constraint (remove mean component)
            G_proj = G - G.mean()
            
            # Projected gradient descent with backtracking
            alpha = 1.0 / beta  # initial step size
            for _ in range(30):
                h_new = h - alpha * G_proj
                h_new = waterfill_project(h_new, target_sum)
                obj_new, _ = smooth_objective_and_gradient(h_new, beta)
                if obj_new < obj:
                    break
                alpha *= 0.5
            
            h = h_new
            
            # Evaluate exact C5 periodically
            if it % 5 == 0 or it == max_iters_per_beta - 1:
                c_exact = exact_C5(h)
                if c_exact < best_c:
                    best_c = c_exact
                    best_h = h.copy()
        
        # After each beta, evaluate exact
        c_exact = exact_C5(h)
        if c_exact < best_c:
            best_c = c_exact
            best_h = h.copy()
    
    return best_h, best_c

def waterfill_project(h, target_sum):
    n = len(h)
    lo = -10.0
    hi = 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def exact_C5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    e = np.exp(np.clip(beta * (c - cmax), -500.0, 500.0))
    se = e.sum()
    obj = cmax + np.log(se) / beta
    w = e / se
    G = np.zeros(n)
    for p in range(2 * n - 1):
        wp = w[p]
        if wp < 1e-14:
            continue
        s = p - (n - 1)
        if s >= 0:
            L = n - s
            G[:L] += wp * dx * (1.0 - h[s:s+L])
            G[s:s+L] -= wp * dx * h[:L]
        else:
            r = -s
            L = n - r
            G[r:r+L] += wp * dx * (1.0 - h[:L])
            G[:L] -= wp * dx * h[r:r+L]
    return obj, G
