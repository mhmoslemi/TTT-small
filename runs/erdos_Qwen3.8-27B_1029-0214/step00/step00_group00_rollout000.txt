    corr = np.correlate(h, 1.0 - h, mode="full")

    """Project h onto {0 <= x <= 1, sum(x) = n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection on shift lambda
    for _ in range(100):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        if s > target:
            # Need to decrease
            above = h > 0
            if not np.any(above):
                break
            h[above] -= (s - target) / np.sum(above)
        else:
            # Need to increase
            below = h < 1
            if not np.any(below):
                break
            h[below] += (target - s) / np.sum(below)
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

    hi = 0.0

        val = evaluate_c5(h_new)
        if val < best_local_val + 1e-9:

- d[i] = +1 for i in S, -1 for i not in S
- Adjust to make sum(d) = 0: d <- d - mean(d)
- Clip to [-1, 1] (or don't, since the feasible interval will handle it)

For the line search with a general direction, I'll sample the function at several points and use ternary search for refinement. The function along the line is the max of quadratics, which is not necessarily unimodal, but in practice for small intervals it often is.

def _evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13 * n:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            count = np.sum(mask)
            if count == 0:
                break
            h[mask] = np.clip(h[mask] + diff / count, 0.0, 1.0)
        else:
            mask = h > 1e-15
            count = np.sum(mask)
            if count == 0:
                break
            h[mask] = np.clip(h[mask] + diff / count, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            candidates = np.where(h < 1.0)[0]
        else:
            candidates = np.where(h > 0.0)[0]
        if len(candidates) > 0:
            h[candidates[0]] = np.clip(h[candidates[0]] + diff, 0.0, 1.0)
    return h

def _line_search_general(h, d, current_c5, max_iter=30):
    """General line search along direction d (mean-zero, sum=0)"""
    lo = 0.0
    hi = 0.0
    for i in range(len(h)):
        if d[i] > 1e-14:
            lo = max(lo, -h[i] / d[i])
            hi = min(hi, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-14:
            lo = max(lo, (h[i] - 1.0) / d[i])
            hi = min(hi, -h[i] / d[i])
    if hi <= lo + 1e-12:
        return None, current_c5
    
    h_orig = h.copy()
    lo += 1e-12
    hi -= 1e-12
    if hi <= lo:
        return None, current_c5
    
    best_val = current_c5
    best_delta = 0.0
    n_samples = 20
    for k in range(n_samples):
        delta = lo + (hi - lo) * k / (n_samples - 1)
        h_new = h_orig + delta * d
        if np.any(h_new < -1e-10) or np.any(h_new > 1.0 + 1e-10):
            continue
        h_new = np.clip(h_new, 0.0, 1.0)

            best_delta = delta

        h1 = np.clip(h_orig + m1 * d, 0.0, 1.0)

        v1 = _evaluate_c5(h1)
        v2 = _evaluate_c5(h2)
