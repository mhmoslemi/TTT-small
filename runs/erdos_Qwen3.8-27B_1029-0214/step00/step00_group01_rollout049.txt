import numpy as np
import time
from scipy.optimize import minimize

_G = {}

def _sigmoid(x):
    x = np.clip(x, -35.0, 35.0)
    return 1.0 / (1.0 + np.exp(-x))

def exact_c5(h):
    n = len(h)
    if n < 2:
        return np.inf
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(corr)):
        return np.inf
    return float(np.max(corr))

def waterfill(v, target_sum):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v = np.clip(v, 0.0, 1.0)
    lo = -max(1.0, v.max()) - target_sum / max(n, 1)
    hi = 1.0 + max(1.0, v.max()) + target_sum / max(n, 1)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - result)
            for i in range(min(len(idx), max(1, int(abs(diff)) + 1))):
                add = min(diff, 1.0 - result[idx[i]])
                result[idx[i]] += add
                diff -= add
        else:
            idx = np.argsort(result)
            for i in range(min(len(idx), max(1, int(abs(diff)) + 1))):
                sub = min(-diff, result[idx[i]])
                result[idx[i]] -= sub
                diff += sub
    return result

def _solve_lambda(y, target):
    def f(lam):
        return np.sum(_sigmoid(y + lam)) - target
    lo = -50.0
    hi = 50.0
    if f(lo) > 0:
        lo = -200.0
    if f(hi) < 0:
        hi = 200.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if f(mid) < 0:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def _y_to_h(y, n):
    target = n / 2.0
    lam = _solve_lambda(y, target)
    h = _sigmoid(y + lam)
    if abs(np.sum(h) - target) > 1e-10:
        h = waterfill(h, target)
    return h

def _surrogate_and_grad(y, n, beta):
    target = n / 2.0
    lam = _solve_lambda(y, target)
    h = _sigmoid(y + lam)
    if abs(np.sum(h) - target) > 1e-10:
        h = waterfill(h, target)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    max_corr = np.max(corr)
    exp_vals = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_vals)
    if sum_exp < 1e-300:
        sum_exp = 1.0
    log_sum = np.log(sum_exp) + beta * max_corr
    F = log_sum / beta
    w = exp_vals / sum_exp
    G = np.zeros(n, dtype=np.float64)
    for j in range(2 * n - 1):
        s = j - (n - 1)
        a = w[j] * dx
        if a < 1e-15:
            continue
        if s >= 0:
            L = n - s
            if L > 0:
                G[s:] += a * (1.0 - h[:L])
                G[:L] -= a * h[s:]
        else:
            r = -s
            L = n - r
            if L > 0:
                G[:L] += a * (1.0 - h[r:])
                G[r:] -= a * h[:L]
    W = h * (1.0 - h)
    W_sum = np.sum(W)
    if W_sum < 1e-15:
        grad_y = np.zeros(n)
    else:
        center = np.sum(G * W) / W_sum
        grad_y = W * (G - center)
    return F, grad_y, h

        raise RuntimeError("timeout")
    F, grad, h = _surrogate_and_grad(y, _G['n'], _G['beta'])

    if not np.isfinite(F):
        return 1e10, np.zeros_like(y)
    if not np.all(np.isfinite(grad)):

    return F, grad

def _upscale(h, factor):
    return np.repeat(h, factor)

def _make_fallback(n):
    x = (np.arange(n) + 0.5) / n * 2.0
    h = 0.5 + 0.25 * np.cos(np.pi * (x - 1.0))

                best_h = h_try.copy()
            step = max(0.01, step * 1.05)
        else:
            step = max(0.005, step * 0.9)

    target = n / 2.0
    n_one = int(np.floor(target))
    n_frac = target - n_one
    sorted_idx = np.argsort(-h)

    h_thresh = waterfill(h_thresh, target)
