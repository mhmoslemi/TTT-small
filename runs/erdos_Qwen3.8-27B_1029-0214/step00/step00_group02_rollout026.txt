    lo = -2.0
    hi = 2.0

        mid = (lo + hi) / 2.0

            lo = mid
        else:

def gradient_refine(h, n, time_limit, rng):
    dx = 2.0 / n
    best_c5 = evaluate_c5(h)
    h_best = h.copy()
    alpha = 0.01

        max_val = np.max(corr)
        tol = max_val * 1e-6 + 1e-12

        if len(active) > 50:
            active = active[:50]  # Limit to top 50
        
        # Compute subgradient
        g = np.zeros(n)
        for j in active:

            i_hi = min(n, n + s)

            best_c5 = c5_new
            h_best = h.copy()
        else:

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr)) * dx

def project_to_slice(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def local_search(h, n, time_limit, rng):
    dx = 2.0 / n
    s_arr = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode="full").astype(np.float64)
    best_c5 = float(np.max(corr)) * dx
    h_best = h.copy()
    start = time.time()
    delta_max = 0.15
    n_acc = 0
    n_rej = 0
    it = 0
    while time.time() - start < time_limit:
        it += 1
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        delta = rng.uniform(0.001, delta_max)
        if rng.random() < 0.5:
            delta = -delta
        hi_val = h[i] + delta
        hj_val = h[j] - delta
        if hi_val < -1e-12 or hi_val > 1.0 + 1e-12 or hj_val < -1e-12 or hj_val > 1.0 + 1e-12:
            continue
        hi_val = max(0.0, min(1.0, hi_val))
        hj_val = max(0.0, min(1.0, hj_val))
        adj = (hi_val - (h[i] + delta)) + (hj_val - (h[j] - delta))
        if abs(adj) > 1e-15:
            continue
        js = j - s_arr
        jp = j + s_arr
        is_idx = i - s_arr
        ip = i + s_arr
        h_is = np.where((is_idx >= 0) & (is_idx < n), h[is_idx], 0.0)
        h_js = np.where((js >= 0) & (js < n), h[js], 0.0)
        h_ip = np.where((ip >= 0) & (ip < n), h[ip], 0.0)
        h_jp = np.where((jp >= 0) & (jp < n), h[jp], 0.0)
        dc = delta * (h_js - h_is - h_ip + h_jp)
        new_corr = corr + dc
        new_max = float(np.max(new_corr)) * dx
        if new_max < best_c5 - 1e-13:
            h[i] = hi_val
            h[j] = hj_val
            corr = new_corr
            best_c5 = new_max
            h_best = h.copy()
            n_acc += 1
        else:
            n_rej += 1
        if it % 2000 == 0:
            if n_acc > n_rej * 3 and delta_max < 0.4:
                delta_max *= 1.3
            elif n_rej > n_acc * 8 and delta_max > 0.005:
                delta_max *= 0.6
            n_acc = 0
            n_rej = 0
        if it % 20000 == 0:
            corr = np.correlate(h, 1 - h, mode="full").astype(np.float64)
            exact_c5 = float(np.max(corr)) * dx
            if exact_c5 < best_c5:
                best_c5 = exact_c5
            h_best = h.copy()
    best_c5 = evaluate_c5(h_best)
    return h_best, best_c5

def gradient_refine(h, n, time_limit, rng):
    dx = 2.0 / n
    best_c5 = evaluate_c5(h)
    h_best = h.copy()
    alpha = 0.005
    start = time.time()
    stall = 0
    while time.time() - start < time_limit and alpha > 1e-7:
        corr = np.correlate(h, 1 - h, mode="full")
        max_val = np.max(corr)
        tol = max_val * 1e-7 + 1e-14
        active = np.where(corr >= max_val - tol)[0]
        if len(active) > 30:
            idx = np.argsort(-corr[active])[:30]
            active = active[idx]
        g = np.zeros(n)
        for jj in active:
            s = jj - (n - 1)
            i_lo = max(0, s)
            i_hi = min(n, n + s)
            if i_lo < i_hi:
                g[i_lo:i_hi] += (1.0 - h[i_lo - s:i_hi - s])
            i_lo = max(0, -s)
            i_hi = min(n, n - s)
            if i_lo < i_hi:
                g[i_lo:i_hi] -= h[i_lo + s:i_hi + s]
        g *= dx / len(active)
        h_new = h - alpha * g
        h_new = project_to_slice(h_new, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            h = h_new
            best_c5 = c5_new
            h_best = h.copy()
            stall = 0
            alpha = min(alpha * 1.1, 0.05)
        else:
            stall += 1
            alpha *= 0.9
        if stall > 50:
            break
    return h_best, best_c5

def try_pattern(n, pattern_type, rng):
    if pattern_type == 0:
        h = np.full(n, 0.5)
    elif pattern_type == 1:
        freq = rng.uniform(1.0, 5.0)
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x)
        h = np.clip(h, 0, 1)
        h = project_to_slice(h, n)
    elif pattern_type == 2:
        h = np.full(n, 0.5)
        for _ in range(n // 4):
            center = int(rng.integers(n))
            width = int(rng.integers(2, max(3, n // 8)))
            lo = max(0, center - width)
            hi = min(n, center + width)
            h[lo:hi] += rng.uniform(0.1, 0.3)
        h = np.clip(h, 0, 1)
        h = project_to_slice(h, n)
    elif pattern_type == 3:
        h = np.zeros(n)
        frac = 0.5
        k = int(n * frac)
        positions = rng.choice(n, size=k, replace=False)
        h[positions] = 1.0
        h = project_to_slice(h, n)
    else:
        x = np.linspace(0, 2, n)
        f1 = rng.uniform(1, 4)
        f2 = rng.uniform(5, 12)
        h = 0.5 + 0.2 * np.sin(2 * np.pi * f1 * x + rng.uniform(0, 6)) + 0.1 * np.sin(2 * np.pi * f2 * x)
        h = np.clip(h, 0, 1)
        h = project_to_slice(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        h = np.full(100, 0.5)

    n = len(h)
    best_c5 = evaluate_c5(h)
    h_best = h.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > 15:
        h, c5 = local_search(h, n, min(15, budget_s - elapsed), rng)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h.copy()

    resolutions = [94, 188, 376, 752, 1504]
    for target_n in resolutions:
        elapsed = time.time() - start_time
        if budget_s - elapsed < 40:
            break
        if target_n <= n:
            continue
        old_x = np.linspace(0, 2, n)
        new_x = np.linspace(0, 2, target_n)
        h_new = np.interp(new_x, old_x, h)
        h_new = project_to_slice(h_new, target_n)
        time_left = budget_s * 0.55 - (time.time() - start_time)
        if time_left <= 5:
            break
        h_new, c5_new = local_search(h_new, target_n, min(time_left, 50), rng)
        if c5_new < best_c5:
            h = h_new.copy()
            n = target_n
            best_c5 = c5_new
            h_best = h.copy()

    for _ in range(4):
        elapsed = time.time() - start_time
        if budget_s - elapsed < 30:
            break
        target_n = n
        h_try = try_pattern(target_n, int(rng.integers(5)), rng)
        h_try, c5_try = local_search(h_try, target_n, min(10, budget_s - elapsed - 5), rng)
        if c5_try < best_c5:
            h = h_try.copy()
            n = target_n
            best_c5 = c5_try
            h_best = h.copy()

    elapsed = time.time() - start_time
    time_left = budget_s * 0.8 - elapsed
    if time_left > 10:
        h, c5 = gradient_refine(h, n, time_left, rng)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h.copy()

    elapsed = time.time() - start_time
    time_left = budget_s * 0.95 - elapsed
    if time_left > 5:
        h, c5 = local_search(h, n, time_left, rng)
        if c5 < best_c5:
            best_c5 = c5
            h_best = h.copy()

    h_best = project_to_slice(h_best, len(h_best))
    best_c5 = evaluate_c5(h_best)

    return h_best, best_c5, len(h_best)
