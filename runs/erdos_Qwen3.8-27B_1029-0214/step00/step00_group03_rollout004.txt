    rng = np.random.RandomState(seed)
    start_time = time.time()

    def elapsed():
        return time.time() - start_time

    h0 = None
    if 'initial_h_values' in globals():
        iv = initial_h_values
        if iv is not None:
            h0 = np.array(iv, dtype=float).copy()
    if h0 is None:
        n0 = 45
        h0 = np.full(n0, 0.5)

    n_init = len(h0)
    best_h = h0.copy()
    best_c5 = float(np.max(compute_c5_array(best_h)))
    best_n = n_init

    # Phase 1: Subgradient descent
    h = h0.copy()
    n = len(h)
    step_size = 0.01
    for it in range(50000):
        if elapsed() > budget_s * 0.15:
            break
        corr = compute_c5_array(h)
        c5 = float(np.max(corr))
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        grad = subgradient_step(h, n, corr)
        h = h - step_size * grad
        h = project_constraint(h, n)
        if it % 2000 == 1999:
            step_size *= 0.85

    # Phase 2: Simulated annealing
    h = best_h.copy()
    n = len(h)
    T = 0.004
    cur_c5 = best_c5
    no_improve = 0
    for it in range(300000):
        if elapsed() > budget_s * 0.55:
            break
        if T < 1e-8:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.12, 0.12)
        ni = h[i] + delta
        nj = h[j] - delta
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        h[i] = ni
        h[j] = nj
        new_c5 = float(np.max(compute_c5_array(h)))
        diff = new_c5 - cur_c5
        if diff <= 0 or rng.random() < np.exp(-diff / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] -= delta
            h[j] += delta
        T = max(1e-8, T * 0.99997)
        if no_improve > 50000 and it % 10000 == 0:
            T = 0.003
            no_improve = 0

    # Phase 3: Try different n and starting points
    n_candidates = []
    for n_try in [n_init, n_init + 5, n_init + 10, 50, 60, 72, 80, 90, 100, 120, 144, 180]:
        if n_try > 200 or n_try < 20:
            continue
        if n_try not in n_candidates:
            n_candidates.append(n_try)

    for n_try in n_candidates:
        if elapsed() > budget_s * 0.88:
            break

        for trial in range(2):
            if elapsed() > budget_s * 0.89:
                break

            if trial == 0:
                h_try = np.full(n_try, 0.5)
                freqs = [1.0, 2.0, 3.0, 5.0]
                for f in freqs:
                    h_try += 0.15 * rng.normal() * np.sin(2.0 * np.pi * f * np.arange(n_try) / n_try + rng.uniform(0, 2 * np.pi))
            else:
                t = np.linspace(0, 1, n_try)
                h_try = np.interp(t, np.linspace(0, 1, len(best_h)), best_h)
                h_try += rng.normal(0, 0.05, n_try)

            h_try = project_constraint(h_try, n_try)
            c5_try = float(np.max(compute_c5_array(h_try)))

            T_try = 0.002
            for it2 in range(8000):
                if elapsed() > budget_s * 0.90:
                    break
                i = rng.randint(n_try)
                j = rng.randint(n_try)
                if i == j:
                    continue
                delta = rng.uniform(-0.1, 0.1)
                ni = h_try[i] + delta
                nj = h_try[j] - delta
                if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                    continue
                h_try[i] = ni
                h_try[j] = nj
                new_c5 = float(np.max(compute_c5_array(h_try)))
                diff2 = new_c5 - c5_try
                if diff2 <= 0 or rng.random() < np.exp(-diff2 / T_try):
                    c5_try = new_c5
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        best_n = n_try
                else:
                    h_try[i] -= delta
                    h_try[j] += delta
                T_try = max(1e-8, T_try * 0.9998)

    # Phase 4: Final polish with binary search on active constraints
    h = best_h.copy()
    n = len(h)
    for it in range(20000):
        if elapsed() > budget_s * 0.97:
            break
        corr = compute_c5_array(h)
        c5 = float(np.max(corr))
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.03, 0.03)
        ni = h[i] + delta
        nj = h[j] - delta
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        h[i] = ni
        h[j] = nj
        new_c5 = float(np.max(compute_c5_array(h)))
        if new_c5 <= c5:
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta

    # Ensure constraint is met
    final_h = project_constraint(best_h, len(best_h))
    final_c5 = float(np.max(compute_c5_array(final_h)))
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h

    return best_h, best_c5, len(best_h)
