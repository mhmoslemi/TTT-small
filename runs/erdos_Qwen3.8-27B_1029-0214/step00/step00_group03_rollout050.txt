import numpy as np
from numpy import correlate
import time

def project_h(h, target_sum):
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = weights[m]
        if wm < 1e-12:
            continue
        k = m - (n - 1)
        w = wm * dx
        if k >= 0:
            nk = n - k
            if nk > 0:
                grad[k:] += w * (1.0 - h[:nk])
                grad[:nk] -= w * h[k:]
        else:
            kp = -k
            nk = n - kp
            if nk > 0:
                grad[:nk] += w * (1.0 - h[kp:])
                grad[kp:] -= w * h[:nk]
    return grad

import time

def project_h(h, target_sum):
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):

        s = np.clip(h - mid, 0.0, 1.0).sum()
