import numpy as np
import time

def _project_feasible(z, n):
    """Project z onto {h: 0 <= h_i <= 1, sum(h) = n/2}"""
    target = n / 2.0
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.5)
    z = np.clip(z, 0.0, 1.0)
    
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    h = np.clip(z - lam, 0.0, 1.0)
    
    # Final correction
    err = target - h.sum()
    if abs(err) > 1e-13:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.any():
            h[interior] += err / interior.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            # Distribute among boundary points
            if err > 0:
                idx = np.argmin(h)
                h[idx] = min(h[idx] + err, 1.0)
            else:
                idx = np.argmax(h)
                h[idx] = max(h[idx] + err, 0.0)
    
    return h

def _c5(h):
    """Compute C5 = max_k correlate(h, 1-h, full)[k] * 2/n"""
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    return float(2.0 / n * raw.max())

def _corr_values(h):
    """Return all lag values scaled by dx"""
    n = len(h)
    dx = 2.0 / n
    return dx * np.correlate(h, 1.0 - h, mode='full')

    """Evaluate C5 after pair move h[p]+=t, h[q]-=t"""
    h2 = h.copy()
    h2[p] += t
    h2[q] -= t
    return _c5(h2)

def _pair_optimize(h, p, q):
    """Find best t for pair move (p, q). Returns (best_h, best_c)."""
    n = len(h)
    t_lo = max(-h[p], h[q] - 1.0) + 1e-15
    t_hi = min(1.0 - h[p], h[q]) - 1e-15
    
    if t_hi - t_lo < 1e-12:
        return h.copy(), _c5(h)
    
    current_c = _c5(h)
    
    # Coarse grid
    ts = np.linspace(t_lo, t_hi, 40)
    best_t = 0.0
    best_c = current_c
    for t in ts:
        h2 = h.copy()
        h2[p] += t
        h2[q] -= t
        c = _c5(h2)
        if c < best_c:
            best_c = c
            best_t = t
    
    # Fine grid around best
    if best_c < current_c - 1e-14:
        span = t_hi - t_lo
        margin = 0.15 * span
        lo2 = max(t_lo, best_t - margin)
        hi2 = min(t_hi, best_t + margin)
        ts2 = np.linspace(lo2, hi2, 20)
        for t in ts2:

            h2[p] += t
            h2[q] -= t
            c = _c5(h2)

                best_t = t

    h_new[p] += best_t
    h_new[q] -= best_t
    return h_new, best_c

def _subgradient_direction(h, active_idx, n):
    """Compute subgradient of C5 at h using active lags, projected to zero-sum."""
    grad = np.zeros(n)
    for m in active_idx:
        k = m - (n - 1)
        if k >= 0:

                grad[k:] += (1.0 - h[:n - k])
            if n - k > 0:
                grad[:n - k] -= h[k:]

            r = -k
            if n - r > 0:
                grad[:n - r] += (1.0 - h[r:])
            if r < n:
                grad[r:] -= h[:n - r]
    
    # Project to zero-sum
    d = -(grad - grad.mean())

import numpy as np
import time


def _project_feasible(z, n):
    target = n / 2.0
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.5)
    z = np.clip(z, 0.0, 1.0)
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(z - lam, 0.0, 1.0)
    err = target - h.sum()
    if abs(err) > 1e-13:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.any():
            h[interior] += err / interior.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def _c5(h):
    n = len(h)
    raw = np.correlate(h, 1.0 - h, mode='full')
    return float(2.0 / n * raw.max())


def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return dx * np.correlate(h, 1.0 - h, mode='full')


def _pair_optimize(h, p, q):
    n = len(h)
    t_lo = max(-h[p], h[q] - 1.0)
    t_hi = min(1.0 - h[p], h[q])
    if t_hi - t_lo < 1e-13:
        return h.copy(), _c5(h)
    cur_c = _c5(h)
    best_t = 0.0
    best_c = cur_c
    ts = np.linspace(t_lo, t_hi, 32)
    for t in ts:
        h2 = h.copy()
        h2[p] += t
        h2[q] -= t
        c = _c5(h2)
        if c < best_c:
            best_c = c
            best_t = t
    if best_c < cur_c - 1e-14:
        span = t_hi - t_lo
        margin = 0.12 * span
        lo2 = max(t_lo, best_t - margin)

                best_c = c
                best_t = t
    h_new = h.copy()
    h_new[p] += best_t
    h_new[q] -= best_t
    return h_new, best_c


def _full_pair_sweep(h, c_best, n, deadline, random_order, rng):

    if random_order:

        rng.shuffle(order)

    else:
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    for p, q in pairs:
        if time.monotonic() > deadline:
            break
        h_new, c_new = _pair_optimize(h, p, q)
        if c_new < c_cur - 1e-13:
            h = h_new
            c_cur = c_new
            improved = True
    return h, c_cur, improved


def _subgrad_step(h, n, active_frac, alpha):
    vals = _corr_vals(h)
    c_max = vals.max()
    thresh = c_max - active_frac * c_max
    active = np.where(vals >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(vals)])
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)
        if k >= 0:
            if k < n:
                grad[k:] += (1.0 - h[:n - k])
            if n - k > 0:
                grad[:n - k] -= h[k:]
        else:
            r = -k
            if n - r > 0:
                grad[:n - r] += (1.0 - h[r:])
            if r < n:
                grad[r:] -= h[:n - r]
    d = -(grad - grad.mean())
    d_norm = np.linalg.norm(d)
    if d_norm < 1e-14:
        return h, _c5(h)
    d = d / d_norm
    h_try = _project_feasible(h + alpha * d, n)
    c_try = _c5(h_try)
    return h_try, c_try


def _random_perturb(h, n, rng, scale):
    noise = rng.standard_normal(n) * scale
    h_try = _project_feasible(h + noise, n)
    return h_try, _c5(h_try)


def _try_upsample(h, n, factor, deadline):
    n2 = n * factor
    h2 = np.repeat(h, factor)
    h2 = _project_feasible(h2, n2)
    c2 = _c5(h2)
    return h2, c2, n2
