def project_to_constraint(v, S):
    """Project v onto {h: 0<=h<=1, sum(h)=S} in L2 norm."""
    n = len(v)
    # Bisection on lambda
    lo = -v.max() - 1
    hi = 1 - v.min()
    for _ in range(200):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0, 1)
        s = h.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0, 1)

    alpha = 8.0
    stall = 0
    stall_threshold = 300
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if it % 1000 == 0 and alpha < 80.0:
            alpha *= 1.3
        
        grad = compute_grad(h, alpha, 40)
        h_new = h - lr * grad
        h_new = project_to_constraint(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            h = h_new
            stall += 1
            if stall > stall_threshold:
                lr *= 0.6
                stall = 0
                if lr < 0.0005:
                    break
    
    return best_h, best_c5

def make_thue_morse(n):
    tm = np.zeros(n)
    for i in range(n):
        c = i
        cnt = 0
        while c > 0:
            cnt += c & 1
            c >>= 1
        tm[i] = cnt % 2
    return tm

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    def try_solution(h):
        nonlocal best_h, best_c5, best_n
        n = len(h)
        dx = 2.0 / n
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
    
    # Start from initial_h_values if available
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        try:
            h_init = np.array(initial_h_values, dtype=float)
            try_solution(h_init)
            if remaining() > 60:
                n0 = len(h_init)
                dx0 = 2.0 / n0
                target0 = n0 / 2.0
                h_opt, c5_opt = optimize_h(h_init.copy(), dx0, target0, rng, 15000, max(0, remaining() - 60))
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n0
        except Exception:
            pass
    
    # Try various n values with different initializations
    n_candidates = [44, 45, 46, 48, 50, 52, 54, 56, 58, 60, 64, 68, 72, 80, 90, 100, 120, 144, 160]
    
    for n in n_candidates:
        if remaining() < 40:
            break
        dx = 2.0 / n
        target_sum = n / 2.0
        time_per_run = max(5.0, min(30.0, (remaining() - 40) / max(1, len(n_candidates) + 10)))
        
        # Initialization strategies
        inits = []
        
        # 1. Constant 0.5
        inits.append(np.full(n, 0.5))
        
        # 2. Thue-Morse with noise
        tm = make_thue_morse(n)
        h_tm = tm.astype(float) + rng.uniform(-0.3, 0.3, n)
        inits.append(project_to_constraint(h_tm, target_sum))
        
        # 3. Random
        h_rand = rng.uniform(0, 1, n)
        inits.append(project_to_constraint(h_rand, target_sum))
        
        # 4. Block pattern
        h_block = np.zeros(n)
        h_block[:n//2] = 1.0
        h_block += rng.uniform(-0.15, 0.15, n)
        inits.append(project_to_constraint(h_block, target_sum))
        
        # 5. Interpolate from best if available
        if best_h is not None and len(best_h) >= 2:
            x_new = np.linspace(0, 1, n)
            x_old = np.linspace(0, 1, len(best_h))
            h_interp = np.interp(x_new, x_old, best_h)
            inits.append(project_to_constraint(h_interp, target_sum))
        
        # 6. Cosine pattern
        x = np.linspace(0, 2, n, endpoint=False)
        h_cos = 0.5 + 0.3 * np.cos(np.pi * x) + 0.1 * np.cos(2 * np.pi * x)
        inits.append(project_to_constraint(h_cos, target_sum))
        
        for h_init in inits:
            if remaining() < 15:
                break
            h_opt, c5_opt = optimize_h(h_init.copy(), dx, target_sum, rng, 10000, max(2.0, min(time_per_run, remaining() - 15)))
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
    
    # If we still have time, do more refined optimization on the best
    if best_h is not None and remaining() > 20:
        n_best = len(best_h)
        dx_best = 2.0 / n_best
        target_best = n_best / 2.0
        h_final, c5_final = optimize_h(best_h.copy(), dx_best, target_best, rng, 30000, remaining() - 10)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
            best_n = n_best
    
    if best_h is None:
        n = 44
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h, 2.0/n)
        best_n = n
    
    return best_h, best_c5, best_n
