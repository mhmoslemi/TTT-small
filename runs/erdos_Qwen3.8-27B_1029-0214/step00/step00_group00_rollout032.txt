def terms_and_grad(h, n):

    one_minus_h = 1 - h
    grad = np.zeros((2 * n - 1, n))
    # k=0
    grad[0] = dx * (1 - 2 * h)
    # k=2r (even, r>=1)
    for r in range(1, n):
        k = 2 * r
        e = n - r
        grad[k, 0:e] = dx * one_minus_h[r:n]
        grad[k, r:n] = -dx * h[0:e]
    # k=2r+1 (odd, r>=1)
    for r in range(1, n):
        k = 2 * r + 1
        e = n - r
        grad[k, r:n] = dx * one_minus_h[0:e]
        grad[k, 0:e] = -dx * h[r:n]

    shifted = alpha * (terms - m)
    log_sum = np.log(np.sum(np.exp(shifted)))

def project_feasible(y, n):
    target = n / 2.0
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 2.0

        if np.sum(np.clip(y - lo, 0.0, 1.0)) < target:
            break

            break
        hi += abs(hi) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    residual = target - np.sum(proj)
    if abs(residual) > 1e-11:

            mask = proj < 1.0 - 1e-14
        else:
            mask = proj > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += residual / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

    grad = np.zeros(n)
    w0 = weights[0]

        grad += w0 * dx * (1.0 - 2.0 * h)

        ek = 2 * r

    return obj, grad

def slsqp_step(h0, n, alpha, maxiter=200):
    def obj(x):
        v, _ = lse_obj_and_grad(np.clip(x, 0.0, 1.0), n, alpha)
        return v
    def grad(x):
        _, g = lse_obj_and_grad(np.clip(x, 0.0, 1.0), n, alpha)
        return g
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    cons = [{'type': 'eq', 'fun': eq_fun}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def coordinate_refine(h, n, max_passes=50):
    h = h.copy()
    best = eval_c5(h, n)
    for p in range(max_passes):
        improved = False
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                for delta in [0.001, 0.0005, -0.001, -0.0005, 0.002, -0.002]:
                    h_new = h.copy()
                    h_new[i] += delta
                    h_new[j] -= delta
                    if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
                        continue
                    c5_new = eval_c5(h_new, n)
                    if c5_new < best:
                        h = h_new
                        best = c5_new
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best
