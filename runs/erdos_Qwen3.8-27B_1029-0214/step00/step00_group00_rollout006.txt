                best_v = h.copy()
    
    # Check final midpoint
    dm = 0.5 * (lo2 + hi2)
    h[i] = bi + dm
    h[j] = bj - dm
    tm = eval_h(h)
    h[i] = bi
    h[j] = bj
    if tm < best_t:
        best_t = tm
        dm_best = dm
        best_v = h.copy()
    
    if best_t < tau_cur - 1e-14:
        return best_v, best_t
    return None, tau_cur
