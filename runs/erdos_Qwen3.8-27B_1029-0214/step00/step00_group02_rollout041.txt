import time
import math

def eval_score(h):
    """Evaluate C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")

    """Check if h is feasible."""
    n = len(h)
    if not np.all(np.isfinite(h)):

    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):

        return False

def gridify(h, M):
    """Convert continuous h to integer grid H where h ≈ H/M."""
    n = len(h)
    target_int = (M * n) // 2
    H = np.floor(h * M + 1e-12).astype(np.int64)

    err = target_int - int(np.sum(H))
    if err > 0:
        residuals = h * M - H.astype(np.float64)
        idx = np.argsort(-residuals)

            H[idx[i % n]] += 1
    elif err < 0:

        idx = np.argsort(residuals)
        for i in range(min(-err, n * M)):
            H[idx[i % n]] -= 1

            if H[s] == 0:

import math

    if not np.all(np.isfinite(h)):

import numpy as np
import time
import math

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def gridify(h, M):
    n = len(h)
    target = (M * n) // 2
    H = np.floor(h * M + 1e-12).astype(np.int64)
    np.clip(H, 0, M, out=H)
    err = target - int(np.sum(H))
    if err > 0:
        res = h * M - H.astype(float)
        idx = np.argsort(-res, kind='stable')
        for i in range(err):
            H[idx[i % n]] += 1
    elif err < 0:
        res = h * M - H.astype(float)
        idx = np.argsort(res, kind='stable')
        for i in range(-err):
            H[idx[i % n]] -= 1
    return H

def local_search(H, M, t_end):
    n = len(H)
    H = H.copy()
    inv_M = 1.0 / M
    h_f = H.astype(float) * inv_M
    cur = eval_score(h_f)
    while time.perf_counter() < t_end:
        best_impr = 1e-14
        best_s = -1
        best_t = -1
        for s in range(n):
            if H[s] == 0:
                continue
            H[s] -= 1
            h_f[s] -= inv_M
            for t in range(n):
                if t == s or H[t] == M:
                    continue
                H[t] += 1
                h_f[t] += inv_M
                sc = eval_score(h_f)
                impr = cur - sc
                if impr > best_impr:
                    best_impr = impr
                    best_s = s
                    best_t = t
                h_f[t] -= inv_M
                H[t] -= 1
            h_f[s] += inv_M
            H[s] += 1
        if best_s >= 0:
            H[best_s] -= 1
            H[best_t] += 1
            h_f[best_s] -= inv_M
            h_f[best_t] += inv_M
            cur -= best_impr
        else:
            break
    return H, cur

def sa_step(H, s, t, direction, M):
    if direction == 0:
        if H[s] == 0:
            return False
        H[s] -= 1
        H[t] += 1
    else:
        if H[t] == 0:
            return False
        H[t] -= 1
        H[s] += 1
    return True

def undo_step(H, s, t, direction):
    if direction == 0:
        H[s] += 1
        H[t] -= 1
    else:
        H[t] += 1
        H[s] -= 1

def simulated_annealing(H_start, M, t_end, rng):
    n = len(H_start)
    H = H_start.copy()
    inv_M = 1.0 / M
    h_f = H.astype(float) * inv_M
    cur = eval_score(h_f)
    best = cur
    best_H = H.copy()
    T = 0.003
    T_min = 1e-5
    cooling = 0.9998
    while T > T_min and time.perf_counter() < t_end:
        s = int(rng.integers(n))
        t = int(rng.integers(n))
        if s == t:
            continue
        d = int(rng.integers(2))
        if not sa_step(H, s, t, d, M):
            continue
        h_f[s] = H[s] * inv_M
        h_f[t] = H[t] * inv_M
        new_sc = eval_score(h_f)
        delta = new_sc - cur
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur = new_sc
            if cur < best:
                best = cur
                best_H = H.copy()
        else:
            undo_step(H, s, t, d)
            h_f[s] = H[s] * inv_M
            h_f[t] = H[t] * inv_M
        T *= cooling
    return best_H, best

def basin_hopping(H_init, M, t_end, rng, n_kicks=200):
    n = len(H_init)
    best_H, best_sc = H_init.copy(), eval_score(H_init.astype(float) / M)
    cur_H, cur_sc = local_search(H_init, M, min(t_end, time.perf_counter() + 60))
    if cur_sc < best_sc:
        best_sc = cur_sc
        best_H = cur_H.copy()
    for _ in range(n_kicks):
        if time.perf_counter() >= t_end:
            break
        kick_H = best_H.copy()
        k = int(rng.integers(2, 8))
        for _ in range(k):
            s = int(rng.integers(n))
            t = int(rng.integers(n))
            if s == t:
                continue
            d = int(rng.integers(2))
            if d == 0:
                if kick_H[s] > 0:
                    kick_H[s] -= 1
                    kick_H[t] += 1
            else:
                if kick_H[t] > 0:
                    kick_H[t] -= 1
                    kick_H[s] += 1
        t_rem = max(5.0, t_end - time.perf_counter())
        new_H, new_sc = local_search(kick_H, M, time.perf_counter() + t_rem)
        if new_sc < best_sc:
            best_sc = new_sc
            best_H = new_H.copy()
        t_rem2 = max(3.0, t_end - time.perf_counter())
        if time.perf_counter() < t_end - 20:
            sa_H, sa_sc = simulated_annealing(best_H, M, time.perf_counter() + min(t_rem2, 30), rng)
            if sa_sc < best_sc:
                best_sc = sa_sc
                best_H = sa_H.copy()
            fin_H, fin_sc = local_search(best_H, M, time.perf_counter() + min(t_rem2, 15))
            if fin_sc < best_sc:
                best_sc = fin_sc
                best_H = fin_H.copy()
    return best_H, best_sc

def resample_to_n(h_src, n_new):
    n_src = len(h_src)
    if n_src == n_new:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_src, h_src)
    h_new = np.clip(h_new, 0.0, 1.0)
    target = n_new / 2.0
    diff = target - float(np.sum(h_new))
    if abs(diff) > 1e-12:
        h_new += diff / n_new
        h_new = np.clip(h_new, 0.0, 1.0)
        diff2 = target - float(np.sum(h_new))
        if abs(diff2) > 1e-12:
            order = np.argsort(1.0 - h_new) if diff2 > 0 else np.argsort(h_new)
            rem = diff2
            for i in range(n_new):
                if abs(rem) < 1e-14:
                    break
                if rem > 0:
                    add = min(rem, 1.0 - h_new[order[i]])
                    h_new[order[i]] += add
                    rem -= add
                else:
                    sub = min(-rem, h_new[order[i]])
                    h_new[order[i]] -= sub
                    rem += sub
    return h_new

def random_binary(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    positions = rng.choice(n, size=n_ones, replace=False)
    h[positions] = 1.0
    if n % 2 == 1:
        remaining = np.setdiff1d(np.arange(n), positions)
        if len(remaining) > 0:
            h[remaining[0]] = 0.5
    return h

def make_cosine_h(n, freq, phase, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(freq * x + phase)
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - float(np.sum(h))
    h += diff / n
    h = np.clip(h, 0, 1)
    return h
