    q = 1 - h
    G = np.zeros(2*N-1)
    for s in range(-(N-1), N):
        m = s + N - 1
        if s >= 0:
            G[m] = np.dot(h[:N-s], q[s:])
        else:
            G[m] = np.dot(h[-s:], q[:N+s])
    return G

def compute_D1(h, i, j, N):
    """Compute linear coefficient D1_s for pair (i,j) for all lags s"""
    q = 1 - h
    D1 = np.zeros(2*N-1)
    for s in range(-(N-1), N):
        m = s + N - 1
        d1 = 0.0
        # (1-h(i+s)) if i+s valid
        if 0 <= i+s < N:
            d1 += 1 - h[i+s]
        # -(1-h(j+s)) if j+s valid
        if 0 <= j+s < N:
            d1 -= 1 - h[j+s]
        # -h(i-s) if i-s valid
        if 0 <= i-s < N:
            d1 -= h[i-s]
        # +h(j-s) if j-s valid
        if 0 <= j-s < N:
            d1 += h[j-s]
        D1[m] = d1
    return D1

def compute_Q(i, j, N):
    """Compute quadratic coefficient Q_s for pair (i,j)"""
    Q = np.zeros(2*N-1)
    # s=0: Q = -2
    Q[N-1] = -2.0
    r = j - i
    if r != 0:
        Q[N-1 + r] = 1.0
        Q[N-1 - r] = 1.0
    return Q

def find_best_d(G, D1, Q, d_min, d_max, n_samples=200):
    d_vals = np.linspace(d_min, d_max, n_samples)
    # Shape: (n_samples, 2N-1)
    obj = G[np.newaxis, :] + d_vals[:, np.newaxis] * D1[np.newaxis, :] + d_vals[:, np.newaxis]**2 * Q[np.newaxis, :]
    max_obj = obj.max(axis=1)
    best_idx = np.argmin(max_obj)
    best_d = d_vals[best_idx]
    return best_d, max_obj[best_idx]

def compute_G_direct(h, N):
    q = 1 - h
    G = np.zeros(2*N-1)
    for s in range(-(N-1), N):
        m = s + N - 1
        if s >= 0:
            G[m] = np.dot(h[:N-s], q[s:])
        else:
            G[m] = np.dot(h[-s:], q[:N+s])
    return G

def compute_D1(h, i, j, N):
    D1 = np.zeros(2*N-1)
    for s in range(-(N-1), N):
        m = s + N - 1
        d1 = 0.0
        if 0 <= i+s < N:
            d1 += 1.0 - h[i+s]
        if 0 <= j+s < N:
            d1 -= 1.0 - h[j+s]
        if 0 <= i-s < N:
            d1 -= h[i-s]
        if 0 <= j-s < N:
            d1 += h[j-s]
        D1[m] = d1
    return D1

def compute_Q(i, j, N):

    Q[N-1] = -2.0
    r = j - i
    if r != 0:
        Q[N-1 + r] = 1.0
        Q[N-1 - r] = 1.0

        return 0.0, G.max()

    max_obj = obj.max(axis=1)
    best_idx = np.argmin(max_obj)

        d_hi = d_vals[best_idx + 1]

    h = np.clip(h, 0, 1)

        return h

    # If diff > 0, we need to increase some entries
    # If diff < 0, we need to decrease some entries
    if diff > 0:

        room = 1.0 - h
        total_room = room.sum()
        if total_room < 1e-10:
            return h  # can't fix
        h += diff / total_room * room
    else:
        # Find entries with room to shrink
        room = h
        total_room = room.sum()
        if total_room < 1e-10:
            return h
        h += diff / total_room * room

    diff = target - h.sum()
    if abs(diff) > 1e-10:

            per = diff / len(idx)
            h[idx] += per
            h = np.clip(h, 0, 1)

import numpy as np

def compute_G_direct(h, N):
    q = 1.0 - h
    G = np.zeros(2 * N - 1)
    for s in range(-(N - 1), N):
        m = s + N - 1
        if s >= 0:
            G[m] = np.dot(h[:N - s], q[s:])
        else:
            G[m] = np.dot(h[-s:], q[:N + s])
    return G

def precompute_W(h, N):
    W = np.zeros((N, 2 * N - 1))
    for i in range(N):
        for s in range(-(N - 1), N):
            m = s + N - 1
            w = 1.0
            if 0 <= i + s < N:
                w -= h[i + s]
            if 0 <= i - s < N:
                w -= h[i - s]
            W[i, m] = w
    return W

def make_Q(i, j, N):
    Q = np.zeros(2 * N - 1)
    Q[N - 1] = -2.0
    r = j - i
    if r != 0:
        Q[N - 1 + r] = 1.0
        Q[N - 1 - r] = 1.0
    return Q

def find_best_d(G, D1, Q, d_min, d_max, n_coarse=256, n_fine=256):
    if d_min >= d_max - 1e-12:
        return 0.0
    d_vals = np.linspace(d_min, d_max, n_coarse)
    obj = G[None, :] + d_vals[:, None] * D1[None, :] + (d_vals[:, None] ** 2) * Q[None, :]
    max_obj = obj.max(axis=1)
    best_idx = int(np.argmin(max_obj))
    best_d = float(d_vals[best_idx])
    if 0 < best_idx < n_coarse - 1:
        d_lo = d_vals[best_idx - 1]

        obj_fine = G[None, :] + d_fine[:, None] * D1[None, :] + (d_fine[:, None] ** 2) * Q[None, :]
        max_fine = obj_fine.max(axis=1)
        best_d = float(d_fine[int(np.argmin(max_fine))])

def project_to_box_sum(h, N):

            h = np.clip(h, 0.0, 1.0)
    return h
