def compute_lag_values(h):
    """Compute all 2n-1 lag values (before multiplying by dx)."""
    n = len(h)
    one_minus_h = 1.0 - h
    vals = np.correlate(h, one_minus_h, mode='full')
    return vals  # length 2n-1

def compute_lag_values_and_grad(h):
    """Compute all 2n-1 lag values and their gradients dC/dh (shape (2n-1, n))."""
    n = len(h)
    dx = 1.0  # We'll multiply by dx later; here we compute the raw sums
    vals = np.zeros(2*n - 1)
    grad = np.zeros((2*n - 1, n))
    
    # Lag 0: C_0 = sum_i h[i]*(1-h[i]) = sum h - sum h^2
    # In full correlation indexing: c[n-1]
    vals[n-1] = np.sum(h) - np.sum(h**2)
    grad[n-1] = 1.0 - 2.0*h
    
    # Positive lag s (s=1..n-1): c[n-1+s] = sum_{j=0}^{n-1-s} h[j+s]*(1-h[j])
    for s in range(1, n):
        e = n - s
        j = np.arange(e)
        vals[n-1+s] = np.sum(h[j+s] * (1.0 - h[j]))
        # d/dh[j+s] = 1 - h[j] for j=0..e-1
        # d/dh[j] = -h[j+s] for j=0..e-1
        grad[n-1+s, j+s] += (1.0 - h[j])
        grad[n-1+s, j] += (-h[j+s])
    
    # Negative lag -s (s=1..n-1): c[n-1-s] = sum_{j=0}^{n-1-s} h[j]*(1-h[j+s])
    for s in range(1, n):

        # d/dh[j] = 1 - h[j+s]

        grad[n-1-s, j] += (1.0 - h[j+s])
        grad[n-1-s, j+s] += (-h[j])
    
    return vals, grad
