So c_m = dx * result[n-1+m] for m = -(n-1), ..., n-1.

def waterfill(v, target):

        else:

    h = y_to_h(y)
    val, grad_h = surrogate_value_and_grad(h, beta)
    if not np.isfinite(val):
        return val, np.zeros_like(y)
    
    # Chain rule: dy/dy_i through h = sigmoid(y + lambda)
    # dh_i/dy_j = h_i(1-h_i) * (delta_ij - h_j(1-h_j)/S)

import numpy as np
import time
from scipy.optimize import minimize


def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())


def waterfill(v, target):
    n = len(v)
    if n == 0:
        return v
    def f(tau):
        return np.sum(np.clip(v + tau, 0.0, 1.0)) - target
    lo = float(v.min()) - 1.0 - target / n
    hi = float(v.max()) + 1.0
    for _ in range(200):
        flo = f(lo)
        fhi = f(hi)
        if flo <= 0.0 and fhi >= 0.0:
            break
        if flo > 0.0:
            lo -= 1.0
        if fhi < 0.0:
            hi += 1.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        if f(mid) > 0.0:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-16:
            break
    return np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)


def make_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - target) > 1e-12:
        h = waterfill(h, target)
    return h


def sigmoid(x):
    x = np.clip(x, -500.0, 500.0)
    return 1.0 / (1.0 + np.exp(-x))


def solve_lambda(y, target):
    n = len(y)
    def f(lam):
        return np.sum(sigmoid(y + lam)) - target
    lo = -200.0
    hi = 200.0
    for _ in range(300):
        if f(lo) <= 0.0 and f(hi) >= 0.0:
            break
        if f(lo) > 0.0:
            lo -= 1.0
        if f(hi) < 0.0:
            hi += 1.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        if f(mid) > 0.0:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    return 0.5 * (lo + hi)


def y_to_h(y):
    n = len(y)
    target = n / 2.0
    lam = solve_lambda(y, target)
    h = sigmoid(y + lam)
    h = make_feasible(h)
    return h


def h_to_y(h):
    eps = 1e-7
    h = np.clip(h, eps, 1.0 - eps)
    return np.log(h / (1.0 - h))


def surrogate_value_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    max_c = float(corr.max())
    if not np.isfinite(max_c):
        return np.inf, np.zeros(n)
    shifted = beta * (corr - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    if sum_exp < 1e-300:
        return np.inf, np.zeros(n)
    weights = exp_shifted / sum_exp
    value = max_c + np.log(sum_exp) / beta
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            L = n - m
            grad[:L] += w * dx * omh[m:]
            grad[m:] += w * dx * (-h[:L])
        else:
            r = -m
            L = n - r
            grad[r:] += w * dx * omh[:L]
            grad[:L] += w * dx * (-h[r:])
    return value, grad


def y_objective(y, beta):
    h = y_to_h(y)
    val, grad_h = surrogate_value_and_grad(h, beta)
    if not np.isfinite(val):
        return val, np.zeros_like(y)
    p = h * (1.0 - h)
    S = float(p.sum())
    if S < 1e-15:
        return val, np.zeros_like(y)
    grad_y = p * (grad_h - np.dot(grad_h, p) / S)
    return val, grad_y


def obj_wrapper(y, beta):
    return y_objective(y, beta)


def transfer_resolution(h, n_target):
    n_old = len(h)
    if n_old == n_target:
        return make_feasible(h.copy())
    x_old = np.arange(n_old, dtype=np.float64) + 0.5
    x_new = np.arange(n_target, dtype=np.float64) + 0.5
    scale = float(n_target) / float(n_old)
    h_new = np.interp(x_new * scale, x_old + 0.5, h)
    return make_feasible(h_new)


def local_search(h, n_trials, rng):
    n = len(h)
    best_c5 = exact_c5(h)
    best_h = h.copy()
    for _ in range(n_trials):
        idx = rng.integers(0, n, size=2)
        if idx[0] == idx[1]:
            continue
        h_trial = best_h.copy()
        delta = (h_trial[idx[1]] - h_trial[idx[0]]) * 0.5
        h_trial[idx[0]] += delta
        h_trial[idx[1]] -= delta
        h_trial = make_feasible(h_trial)
        c5_trial = exact_c5(h_trial)
        if c5_trial < best_c5 - 1e-14:
            best_c5 = c5_trial
            best_h = h_trial
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s * 0.90

    best_c5 = np.inf
    best_h = None

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        n_fb = 64
        x_fb = np.linspace(0.5 / n_fb, 2.0 - 0.5 / n_fb, n_fb)
        h_init = 0.5 + 0.3 * np.sin(np.pi * (x_fb - 1.0))
        h_init = make_feasible(h_init)

    h_init = make_feasible(h_init)
    c5_init = exact_c5(h_init)
    best_c5 = c5_init
    best_h = h_init.copy()

    n_orig = len(h_init)
    beta_list = [15, 50, 150, 500, 1500, 4000]

    resolutions = [n_orig]
    n_double = n_orig * 2
    if n_double % 2 == 0:
        resolutions.append(n_double)
    elif n_double + 1 > n_orig:
        resolutions.append(n_double + 1)

    for res_n in resolutions:
        if time.monotonic() > deadline:
            break
        if res_n == n_orig:
            h_cur = best_h.copy()
        else:
            h_cur = transfer_resolution(best_h, res_n)
            c5_cur = exact_c5(h_cur)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()

        for beta in beta_list:
            if time.monotonic() > deadline:
                break
            if res_n <= 100:
                max_it = 100 if beta < 100 else 60
            else:
                max_it = 70 if beta < 100 else 40
            try:
                y0 = h_to_y(h_cur)
                res = minimize(
                    fun=obj_wrapper,
                    x0=y0,
                    args=(beta,),
                    method='L-BFGS-B',
                    jac=True,
                    options={'maxiter': max_it, 'maxfun': max_it * 4, 'gtol': 1e-6}
                )
                if res.x is not None and np.all(np.isfinite(res.x)):
                    h_new = y_to_h(res.x)
                    c5_new = exact_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                    h_cur = h_new
            except Exception:
                pass

        if time.monotonic() < deadline - 20:
            h_cur, c5_cur = local_search(h_cur, 200, rng)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()

    if time.monotonic() < deadline - 5:
        h_best = best_h.copy()
        for trial in range(100):
            if time.monotonic() > deadline:
                break
            eps = 0.03 + 0.02 * trial / 100.0
            h_pert = h_best + rng.randn(len(h_best)) * eps
            h_pert = make_feasible(h_pert)
            c5_pert = exact_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()

    h_final = make_feasible(best_h)
    c5_final = exact_c5(h_final)
    n_final = len(h_final)

    return h_final, c5_final, n_final
