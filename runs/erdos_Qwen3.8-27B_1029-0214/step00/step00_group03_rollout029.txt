import numpy as np

def compute_G(h):
    """Compute correlation G_s for all lags s."""
    N = len(h)
    q = 1.0 - h
    # np.correlate(h, q, 'full') gives c[m] = sum_n h[n] * q[n - (m - (N-1))]
    # which is our G_s with s = m - (N-1)
    G = np.correlate(h, q, mode='full')
    return G

def compute_W(h):
    """Compute W_p(s) for all p and s.
    W_p(s) = (1-h[p+s]) if p+s in [0,N) else 0) - (h[p-s] if p-s in [0,N) else 0)
    Stored as matrix W[p, m] where m = s + N - 1.
    """
    N = len(h)
    q = 1.0 - h
    W = np.zeros((N, 2*N-1))
    for p in range(N):
        # First term: (1-h[p+s]) for s where 0 <= p+s < N
        # s ranges from -p to N-1-p, m = s+N-1 ranges from N-1-p to N-1+p
        s_start = -p
        s_end = N-1-p
        m_start = N-1-p
        m_end = N-1+p
        W[p, m_start:m_end+1] = q[p+s_start:p+s_end+1]
        
        # Second term: -h[p-s] for s where 0 <= p-s < N
        # s ranges from p-N+1 to p, m = s+N-1 ranges from p to 2N-2-p
        s_start2 = p-N+1
        s_end2 = p
        m_start2 = p
        m_end2 = 2*N-2-p
        W[p, m_start2:m_end2+1] -= h[p-s_start2:p-s_end2+1]
    return W

def compute_quad(N, i, j):
    """Compute quad coefficients for pair (i,j)."""
    quad = np.zeros(2*N-1)
    r = j - i
    quad[N-1] = -2.0  # lag 0
    quad[N-1+r] = 1.0  # lag +r
    quad[N-1-r] = 1.0  # lag -r
    return quad

def find_best_d(G, D1, quad, d_min, d_max, n_grid=200):
    """Find d in [d_min, d_max] minimizing max_m (G[m] + d*D1[m] + d²*quad[m])."""
    if d_min >= d_max:
        return 0.0, None
    
    # If range is very small, skip
    if d_max - d_min < 1e-12:
        return 0.0, None
    
    d_vals = np.linspace(d_min, d_max, n_grid)
    # Vectorized: shape (n_grid, 2N-1)
    vals = G[None, :] + d_vals[:, None] * D1[None, :] + (d_vals[:, None] ** 2) * quad[None, :]
    f_vals = vals.max(axis=1)
    best_idx = f_vals.argmin()
    best_d = d_vals[best_idx]
    best_f = f_vals[best_idx]
    
    # Parabolic refinement
    if 0 < best_idx < n_grid - 1:
        d_minus = d_vals[best_idx - 1]
        d_plus = d_vals[best_idx + 1]
        f_minus = f_vals[best_idx - 1]
        f_plus = f_vals[best_idx + 1]
        # Fit parabola
        denom = (f_minus - 2*f_vals[best_idx] + f_plus)
        if abs(denom) > 1e-15:
            offset = 0.5 * (f_minus - f_plus) / denom
            refined_d = best_d + offset * (d_plus - d_minus) / 2
            if d_min < refined_d < d_max:
                # Evaluate at refined point
                f_refined = (G + refined_d * D1 + refined_d**2 * quad).max()
                if f_refined < best_f:
                    best_d = refined_d
                    best_f = f_refined
    
    return best_d, best_f

def evaluate_c5(h):
    """Evaluate C5 using the authoritative formula."""
    N = len(h)
    dx = 2.0 / N
    G = np.correlate(h, 1.0 - h, mode='full')
    return (G.max() * dx)
