    n_old = len(h)

    return np.clip(np.interp(x_new, x_old, h), 0.0, 1.0)

def _enforce_sum(h, n):
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            total = room.sum()
            if total < 1e-14:
                break
            h[mask] += min(diff / total, 1.0) * room
        else:
