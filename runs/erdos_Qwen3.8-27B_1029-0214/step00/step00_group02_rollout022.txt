    for nt in [65, 97, 130, 195, 257]:
        if time.time() >= tl:
            break
        if nt <= bn:
            continue
        hb2 = _interpolate(bh, bn, nt)
        h, c = _anneal2(hb2, 0.008, 3e-6, rng, tl)
        if c < bc:
            bc, bh, bn = c, h, nt

    for r in range(40):
        if time.time() >= tl:
            break
        ht = _perturb(bh, bn, rng, 0.2)
        h, c = _anneal2(ht, 0.005 + 0.01 * rng.random(), 2e-6, rng, tl)
        if c < bc:
            bc, bh, bn = c, h, bn

    return bh, bc, bn
