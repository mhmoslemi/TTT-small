def compute_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    c_vals = corr * dx
    
    # Softmax weights
    x = tau * c_vals
    x_max = np.max(x)
    w = np.exp(x - x_max)
    w /= np.sum(w)
    
    # Gradient: grad_i = sum_k w_k * dx * [(1-h[i+k]) - h[i-k]]
    # k ranges from -(n-1) to (n-1), corresponding to corr index k + (n-1)
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        k_idx = k + (n - 1)
        w_k = w[k_idx] * dx
        if w_k < 1e-15:
            continue
        for i in range(n):
            # (1-h[i+k]) term if 0 <= i+k < n
            term = 0
            if 0 <= i + k < n:
                term += (1 - h[i + k])
            # -h[i-k] term if 0 <= i-k < n
            if 0 <= i - k < n:
                term -= h[i - k]
            grad[i] += w_k * term
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def smooth_max(values, tau):
    x = tau * values
    x_max = np.max(x)
    return (x_max + np.log(np.sum(np.exp(x - x_max)))) / tau

def fix_sum(h, n):
    target = n * 0.5
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def compute_smooth_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    c_vals = corr * dx
    x = tau * c_vals
    x_max = np.max(x)
    w = np.exp(x - x_max)
    w /= np.sum(w)
    
    # grad_i = sum_k w_k * dx * dC_k/dh_i
    # dC_k/dh_i = (1-h[i+k])*I(0<=i+k<n) - h[i-k]*I(0<=i-k<n)
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        k_idx = k + n - 1
        wk = w[k_idx] * dx
        if wk < 1e-16:
            continue
        # (1-h[i+k]) for i in [0, n-1-k] if k >= 0, or [0, n-1] if k < 0
        # h[i-k] for i in [k, n-1] if k >= 0, or [0, n-1+k] if k < 0
        if k >= 0:
            # (1-h[i+k]) valid for i in [0, n-1-k]
            grad[:n-k] += wk * (1 - h[k:])
            # -h[i-k] valid for i in [k, n-1]
            grad[k:] -= wk * h[:n-k]
        else:
            # (1-h[i+k]) valid for i in [-k, n-1]
            grad[-k:] += wk * (1 - h[:n+k])
            # -h[i-k] valid for i in [0, n-1+k]
            grad[:n+k] -= wk * h[-k:]
    return grad

def project_onto_feasible(v, n):
    """Project v onto {0<=h<=1, sum(h)=n/2}."""
    target = n * 0.5
    lo, hi = -max(abs(v.min()), abs(1-v.max())) - 1, max(abs(v.min()), abs(1-v.max())) + 1
    for _ in range(80):
