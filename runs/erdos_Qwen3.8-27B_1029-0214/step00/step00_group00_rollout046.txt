    obj = exact_evaluate(h_new)
    return h_new, obj

import numpy as np
import time
import math

def exact_evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    x = np.clip(y.astype(np.float64), 0.0, 1.0).copy()
    residual = target - np.sum(x)
    if abs(residual) < 1e-12:
        return x
    for _ in range(30):
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            room = 1.0 - x
            total_room = np.sum(room)
            if total_room < 1e-15:
                break
            ratio = min(1.0, residual / total_room)
            x += ratio * room
        else:
            room = x
            total_room = np.sum(room)
            if total_room < 1e-15:
                break
            ratio = min(1.0, -residual / total_room)
            x -= ratio * room
        x = np.clip(x, 0.0, 1.0)
        residual = target - np.sum(x)
    # Final nudge
    residual = target - np.sum(x)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argmax(1.0 - x)
            x[idx] = min(1.0, x[idx] + residual)
        else:
            idx = np.argmin(x)
            x[idx] = max(0.0, x[idx] + residual)
    return x

def compute_terms_and_slopes(h, d):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    nterms = 2 * n - 1
    values = np.empty(nterms)
    slopes = np.empty(nterms)
    
    # Lag 0 (index n-1)
    values[n-1] = dx * np.dot(h, one_minus_h)
    slopes[n-1] = dx * np.dot(d, one_minus_h - h)
    
    # Positive lags r=1..n-1 (indices n-1+r)
    for r in range(1, n):
        e = n - r
        values[n-1+r] = dx * np.dot(h[r:r+e], one_minus_h[:e])
        slopes[n-1+r] = dx * np.dot(d[r:r+e], one_minus_h[:e]) - dx * np.dot(d[:e], h[r:r+e])
    
    # Negative lags r=1..n-1 (indices n-1-r)
    for r in range(1, n):
        e = n - r
        values[n-1-r] = dx * np.dot(h[:e], one_minus_h[r:r+e])
        slopes[n-1-r] = dx * np.dot(d[:e], one_minus_h[r:r+e]) - dx * np.dot(d[r:r+e], h[:e])
    
    return values, slopes

def line_minimize(h, d):
    n = len(h)
    dmax = np.max(np.abs(d))
    if dmax < 1e-15:
        return h.copy(), exact_evaluate(h)
    
    values, slopes = compute_terms_and_slopes(h, d)
    current_val = float(np.max(values))
    
    # Feasible interval
    eps = 1e-14
    lo_bound = 0.0
    hi_bound = 0.0
    for i in range(n):
        if d[i] > eps:
            lo_bound = max(lo_bound, -h[i] / d[i])
            hi_bound = min(hi_bound, (1.0 - h[i]) / d[i])
        elif d[i] < -eps:
            lo_bound = max(lo_bound, (1.0 - h[i]) / d[i])
            hi_bound = min(hi_bound, -h[i] / d[i])
    
    lo_bound = max(lo_bound, -5.0) + 1e-12
    hi_bound = min(hi_bound, 5.0) - 1e-12
    
    if hi_bound - lo_bound < 1e-12:
        return h.copy(), current_val
    
    # Determine search direction
    active_tol = max(1e-10, 1e-12 * max(1.0, current_val))
    active_mask = values >= (current_val - active_tol)
    min_slope = float(np.min(slopes[active_mask]))
    max_slope = float(np.max(slopes[active_mask]))
    slope_tol = 1e-8
    
    if max_slope < -slope_tol:
        a, b = 0.0, hi_bound
    elif min_slope > slope_tol:
        a, b = lo_bound, 0.0
    else:
        # Check if truly near-optimal
        return h.copy(), current_val
    
    # Bisection
    best_delta = 0.0
    best_val = current_val
    
    for _ in range(60):
        if b - a < 1e-15:
            break
        mid = 0.5 * (a + b)
        term_vals = values + mid * slopes
        mid_val = float(np.max(term_vals))
        if mid_val < best_val - 1e-15:
            best_val = mid_val
            best_delta = mid
        
        act_tol = max(1e-10, 1e-12 * max(1.0, mid_val))
        act = term_vals >= (mid_val - act_tol)
        s_lo = float(np.min(slopes[act]))
        s_hi = float(np.max(slopes[act]))
        
        if s_hi < -slope_tol:
            a = mid
        elif s_lo > slope_tol:
            b = mid
        else:
            break
    
    # Check a few extra points
    for frac in [0.0, 0.25, 0.5, 0.75, 1.0]:
        c = a + frac * (b - a)
        if lo_bound <= c <= hi_bound:
            cv = float(np.max(values + c * slopes))
            if cv < best_val:
                best_val = cv
                best_delta = c
    
    # Also check endpoints of full interval
    for c in [lo_bound, hi_bound]:
        cv = float(np.max(values + c * slopes))
        if cv < best_val:
            best_val = cv
            best_delta = c
    
    h_new = h + best_delta * d
    h_new = np.clip(h_new, 0.0, 1.0)
    
    # Quick feasibility check
    s = np.sum(h_new)
    if abs(s - n/2.0) < 1e-10:
        obj = exact_evaluate(h_new)
    else:
        h_new = project_feasible(h_new, n)
        obj = exact_evaluate(h_new)
    
    return h_new, obj

def gen_pair_dir(i, j, n):
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    return d

def gen_random_dir(n, rng, kind=0):
    if kind == 0:
        d = rng.choice([-1.0, 1.0], size=n)
    elif kind == 1:
        freq = rng.randint(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        d = np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
    else:
        d = rng.normal(0, 1, n)
    d -= np.mean(d)
    mx = np.max(np.abs(d))
    if mx > 1e-15:
        d /= mx
    return d

def gen_block_dir(n, rng):
    # Pick two blocks of equal length
    max_len = n // 4
    if max_len < 1:
        max_len = 1
    blen = rng.randint(1, max_len + 1)
    a = rng.randint(0, n - blen)
    c = rng.randint(0, n - blen)
    if a == c:
        c = (c + blen) % (n - blen)
        if a == c:
            return gen_random_dir(n, rng, 2)
    d = np.zeros(n)
    d[a:a+blen] = 1.0
    d[c:c+blen] = -1.0
    d -= np.mean(d)
    mx = np.max(np.abs(d))
    if mx > 1e-15:
        d /= mx
    return d

def upscale_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(new_x, old_x, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, new_n)
    return h_new
