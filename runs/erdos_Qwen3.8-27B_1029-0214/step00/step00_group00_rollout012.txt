import numpy as np
import time

def evaluate_erdos(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def repair_h(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n * 0.5
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr > 1e-15:
                h += (res / tr) * room
        else:
            tr = h.sum()
            if tr > 1e-15:
                h += (res / tr) * h
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_pair(h, i, j, t):
    h2 = h.copy()
    h2[i] -= t
    h2[j] += t
    return evaluate_erdos(h2)

def eval_dir(h, d, t):
    h2 = h + t * d
    if h2.min() < -1e-10 or h2.max() > 1.0 + 1e-10:
        return 1e10
    return evaluate_erdos(h2)

def golden_pair(h, i, j, a, b, n_iter=25):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    cp = b - gr * (b - a)
    dp = a + gr * (b - a)
    fc = eval_pair(h, i, j, cp)
    fd = eval_pair(h, i, j, dp)
    for _ in range(n_iter):
        if fc < fd:
            b = dp
            dp = cp
            fd = fc
            cp = b - gr * (b - a)
            fc = eval_pair(h, i, j, cp)
        else:
            a = cp
            cp = dp
            fc = fd
            dp = a + gr * (b - a)
            fd = eval_pair(h, i, j, dp)
    t_opt = 0.5 * (a + b)
    return t_opt, eval_pair(h, i, j, t_opt)

def golden_dir(h, d, a, b, n_iter=25):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    cp = b - gr * (b - a)
    dp = a + gr * (b - a)
    fc = eval_dir(h, d, cp)
    fd = eval_dir(h, d, dp)
    for _ in range(n_iter):
        if fc < fd:
            b = dp
            dp = cp
            fd = fc
            cp = b - gr * (b - a)
            fc = eval_dir(h, d, cp)
        else:
            a = cp
            cp = dp
            fc = fd
            dp = a + gr * (b - a)
            fd = eval_dir(h, d, dp)
    t_opt = 0.5 * (a + b)
    return t_opt, eval_dir(h, d, t_opt)

def pair_line_search(h, i, j, n_samples):
    hi_t = min(h[i], 1.0 - h[j])
    if hi_t < 1e-12:
        return h, evaluate_erdos(h), 0.0
    cur_val = evaluate_erdos(h)
    best_val = cur_val
    best_t = 0.0
    ts = np.linspace(0.0, hi_t, n_samples)
    for t in ts:
        v = eval_pair(h, i, j, t)
        if v < best_val:
            best_val = v
            best_t = t
    if best_t == 0.0:
        return h, cur_val, 0.0
    step = hi_t / (n_samples - 1) if n_samples > 1 else hi_t
    lo_r = max(0.0, best_t - step)
    hi_r = min(hi_t, best_t + step)
    if hi_r - lo_r > 1e-12:
        t_opt, v_opt = golden_pair(h, i, j, lo_r, hi_r, n_iter=20)
        if v_opt < best_val:
            best_val = v_opt
            best_t = t_opt
    h[i] -= best_t
    h[j] += best_t
    return h, best_val, best_t

def pair_optimize_pass(h, n, rng, n_samples, deadline_fn):
    bound = evaluate_erdos(h)
    n_pairs = n * (n - 1) // 2
    if n_pairs <= 5000:
        order = np.arange(n_pairs)
        rng.shuffle(order)
    else:
        order = rng.choice(n_pairs, size=5000, replace=False)
    improved = 0
    for pi in order:
        if deadline_fn():
            break
        i = pi // (n - 1)
        j = pi % (n - 1)
        if j >= i:
            j += 1
        h_new, v, t = pair_line_search(h, i, j, n_samples)
        if t != 0.0 and v < bound - 1e-14:
            h = h_new
            bound = v
            improved += 1
    return h, bound, improved
