```python
    """Compute all scaled correlation values (2n-1 of them)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def unique_lag_values(h, n, dx):
    """Compute the n unique lag values (lags 0 to n-1)."""
    vals = np.zeros(n)
    corr_full = np.correlate(h, 1.0 - h, mode="full")
    # lag tau corresponds to index tau + (n-1) in the full correlation
    for tau in range(n):
        vals[tau] = corr_full[tau + n - 1] * dx
    return vals

def corr_grad(h, tau, n, dx):
    """Gradient of dx * G_tau with respect to h (length n vector)."""
    L = n - tau
    grad = np.zeros(n)
    if tau == 0:

    else:
        grad[tau:] += (1.0 - h[:L]) * dx
```

```python
class EpiSolver:
    def __init__(self, h0, n, dx, target, taus, deadline):
        self.h0 = h0.copy()
        self.n = n
        self.dx = dx
        self.target = target
        self.taus = np.array(taus)
        self.deadline = deadline
        self.nt = len(self.taus)
        self.h = self.h0
        self.t = 0.0
        self._timeout = False

    def _check_time(self):
        if time.time() > self.deadline:
            self._timeout = True
            raise RuntimeError("timeout")

    def fun(self, x):
        self._check_time()
        return x[self.n]  # t

    def jac(self, x):
        self._check_time()
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g

    def con_eq_fun(self, x):
        self._check_time()
        return x[:self.n].sum() - self.target

    def con_eq_jac(self, x):
        self._check_time()
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g

    def con_ineq_fun(self, x):
        self._check_time()
        h = x[:self.n]
        t = x[self.n]
        vals = np.zeros(self.nt)
        for i, tau in enumerate(self.taus):
            vals[i] = t - np.sum(h[tau:] * (1.0 - h[:self.n - tau])) * self.dx
        return vals

    def con_ineq_jac(self, x):
        self._check_time()
        h = x[:self.n]
        n = self.n
        nt = self.nt
        jac = np.zeros((nt, n + 1))
        for i, tau in enumerate(self.taus):
            L = n - tau
            g = corr_grad(h, tau, n, self.dx)
            jac[i, :n] = -g
            jac[i, n] = 1.0
        return jac

    def solve(self, maxiter, ftol):
        n = self.n
        x0 = np.zeros(n + 1)
        x0[:n] = self.h0
        x0[n] = exact_C5(self.h0) + 1e-8

        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        
        cons = [
            {'type': 'eq', 'fun': self.con_eq_fun, 'jac': self.con_eq_jac},
            {'type': 'ineq', 'fun': self.con_ineq_fun, 'jac': self.con_ineq_jac},
        ]

        try:
            res = minimize(
                self.fun, x0, jac=self.jac, bounds=bounds,
                constraints=cons, method='SLSQP',
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            return res
        except RuntimeError:
            return None
        except Exception:
            return None
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def exact_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_simplex(z, target):
    n = len(z)
    z = np.array(z, dtype=np.float64)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - result.sum()
    if abs(residual) > 1e-14:
        idx = np.where((result > 1e-12) & (result < 1.0 - 1e-12))[0]
        if len(idx) > 0:
            result[idx[0]] = np.clip(result[idx[0]] + residual, 0.0, 1.0)
    return result

def corr_lag_value(h, tau, n, dx):
    L = n - tau
    return float(np.sum(h[tau:tau+L] * (1.0 - h[:L])) * dx)

def corr_lag_grad(h, tau, n, dx):
    L = n - tau
    grad = np.zeros(n)
    if tau == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    else:
        grad[tau:] += (1.0 - h[:L]) * dx
        grad[:L] -= h[tau:] * dx
    return grad

class EpiSolver:
    def __init__(self, h0, n, dx, target, taus, deadline):
        self.h0 = np.array(h0, dtype=np.float64)
        self.n = n
        self.dx = dx
        self.target = target
        self.taus = np.array(taus, dtype=np.int64)
        self.nt = len(self.taus)
        self.deadline = deadline

    def _check(self):
        if time.time() > self.deadline:
            raise RuntimeError("timeout")

    def obj(self, x):
        self._check()
        return x[self.n]

    def obj_grad(self, x):
        self._check()
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g

    def eq_fun(self, x):
        self._check()
        return x[:self.n].sum() - self.target

    def eq_jac(self, x):
        self._check()
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g

    def ineq_fun(self, x):
        self._check()
        h = x[:self.n]
        t = x[self.n]
        out = np.empty(self.nt)
        for i in range(self.nt):
            tau = int(self.taus[i])
            L = self.n - tau
            out[i] = t - np.sum(h[tau:tau+L] * (1.0 - h[:L])) * self.dx
        return out

    def ineq_jac(self, x):
        self._check()
        h = x[:self.n]
        n = self.n
        nt = self.nt
        jac = np.zeros((nt, n + 1))
        for i in range(nt):
            tau = int(self.taus[i])
            L = n - tau
            if tau == 0:
                g = (1.0 - 2.0 * h) * self.dx
            else:
                g = np.zeros(n)
                g[tau:] += (1.0 - h[:L]) * self.dx
                g[:L] -= h[tau:] * self.dx
            jac[i, :n] = -g
            jac[i, n] = 1.0
        return jac

    def solve(self, maxiter, ftol):
        n = self.n
        x0 = np.zeros(n + 1)
        x0[:n] = self.h0
        x0[n] = exact_C5(self.h0) + 1e-6

        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'eq', 'fun': self.eq_fun, 'jac': self.eq_jac},
            {'type': 'ineq', 'fun': self.ineq_fun, 'jac': self.ineq_jac},
        ]

        try:
            res = minimize(self.obj, x0, jac=self.obj_grad, bounds=bounds,
                          constraints=cons, method='SLSQP',
                          options={'maxiter': maxiter, 'ftol': ftol})
            return res
        except RuntimeError:
            return None
        except Exception:
            return None


def select_taus(vals, max_count, active_tol):
    """Select constraint lags: those within active_tol of max, plus fill to max_count."""
    n = len(vals)
    max_val = np.max(vals)
    active = np.where(vals >= max_val - active_tol)[0]
    
    if len(active) >= max_count:
        # Take the top max_count
        idx = active[np.argsort(vals[active])[::-1][:max_count]]
    else:
        # Add more by value
        remaining = np.setdiff1d(np.arange(n), active)
        need = max_count - len(active)
        if need > 0:
            extra = remaining[np.argsort(vals[remaining])[::-1][:need]]
            idx = np.concatenate([active, extra])
        else:
            idx = active
    
    return np.sort(idx)
```

import numpy as np

def exact_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_box_simplex(z, target):
    n = len(z)
    z = np.array(z, dtype=np.float64)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    z = np.clip(z, 0.0, 1.0)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - result.sum()
    if abs(residual) > 1e-14:
        idx = np.where((result > 1e-12) & (result < 1.0 - 1e-12))[0]
        if len(idx) > 0:
            result[idx[0]] = np.clip(result[idx[0]] + residual, 0.0, 1.0)
    return result


def compute_lag_values(h, n, dx):
    vals = np.zeros(n)
    corr_full = np.correlate(h, 1.0 - h, mode="full")
    for tau in range(n):
        vals[tau] = corr_full[tau + n - 1] * dx
    return vals


def select_taus(vals, max_count, active_tol):
    n = len(vals)
    max_val = np.max(vals)
    active = np.where(vals >= max_val - active_tol)[0]
    if len(active) >= max_count:
        idx = active[np.argsort(vals[active])[::-1][:max_count]]
    else:
        remaining = np.setdiff1d(np.arange(n), active)
        need = max_count - len(active)
        if need > 0 and len(remaining) > 0:
            extra = remaining[np.argsort(vals[remaining])[::-1][:need]]
            idx = np.concatenate([active, extra])
        else:
            idx = active
    return np.sort(idx)


class EpiSolver:
    def __init__(self, h0, n, dx, target, taus, deadline):
        self.h0 = np.array(h0, dtype=np.float64)
        self.n = n
        self.dx = dx
        self.target = target
        self.taus = np.array(taus, dtype=np.int64)
        self.nt = len(self.taus)
        self.deadline = deadline

    def _check(self):
        if time.time() > self.deadline:
            raise RuntimeError("timeout")

    def obj(self, x):
        self._check()
        return x[self.n]

    def obj_grad(self, x):
        self._check()
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g

    def eq_fun(self, x):
        self._check()
        return x[:self.n].sum() - self.target

    def eq_jac(self, x):
        self._check()
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g

    def ineq_fun(self, x):
        self._check()
        h = x[:self.n]
        t = x[self.n]
        dx = self.dx
        n = self.n
        out = np.empty(self.nt)
        for i in range(self.nt):
            tau = int(self.taus[i])
            L = n - tau
            out[i] = t - np.sum(h[tau:tau + L] * (1.0 - h[:L])) * dx
        return out

    def ineq_jac(self, x):
        self._check()
        h = x[:self.n]
        n = self.n
        nt = self.nt
        dx = self.dx
        jac = np.zeros((nt, n + 1))
        for i in range(nt):
            tau = int(self.taus[i])

            else:
                g = np.zeros(n)
                g[tau:] += (1.0 - h[:L]) * dx
                g[:L] -= h[tau:] * dx
            jac[i, :n] = -g
            jac[i, n] = 1.0
        return jac
