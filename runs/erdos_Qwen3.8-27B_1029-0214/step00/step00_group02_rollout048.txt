def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_logsumexp(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = np.max(corr)
    shifted = beta * (corr - max_c) * dx
    exp_terms = np.exp(shifted)
    p = exp_terms / np.sum(exp_terms)
    grad = np.zeros(n)
    shifts = np.arange(2 * n - 1) - (n - 1)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        s = shifts[k]
        if s == 0:
            grad += pk * (1.0 - 2.0 * h)
        elif s > 0:
            first = np.zeros(n)
            first[0:n - s] = 1.0 - h[s:n]
            second = np.zeros(n)
            second[s:n] = h[0:n - s]
            grad += pk * (first - second)

            second = np.zeros(n)
            second[0:n - sn] = h[sn:n]
            grad += pk * (first - second)
