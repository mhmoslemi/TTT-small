    Returns array shape (2n-1, n).
    Due to symmetry, we compute n terms and mirror.
    """
    n = len(h)
    m = 2 * n - 1
    grad = np.zeros((m, n))

    # C_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])

        if k > 0:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def get_corr_first_half(h):
    """Get the first n correlation values (non-negative lags)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr[n-1:]  # n values, lags 0 to n-1

def grad_corr_first_half(h):
    """Gradients of the n non-negative lag correlation terms.
    Returns array of shape (n, n).
    Row k is gradient of R_k (correlation at lag k).
    """
    n = len(h)
    grad = np.zeros((n, n))
    
    for k in range(n):
        e = n - k
        if k == 0:
            grad[k] = 1.0 - 2.0 * h
        else:
            grad[k, k:n] += (1.0 - h[0:e])
            grad[k, 0:e] -= h[k:n]
    
    return grad

def project_to_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Redistribute
    if diff > 0:
        mask = h < 1.0
        room = np.sum(1.0 - h[mask])
        if room > 1e-12:
            h[mask] += diff / room * (1.0 - h[mask])
    else:
        mask = h > 0.0
        excess = np.sum(h[mask])
        if excess > 1e-12:
            h[mask] += diff / excess * h[mask]
    h = np.clip(h, 0.0, 1.0)
    # Fine adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Add to a single element
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

    n = len(h)
    R = get_corr_first_half(h)
    # Shift for numerical stability
    R_max = np.max(R)
    val = R_max + np.log(np.sum(np.exp(alpha * (R - R_max)))) / alpha
    return val

def _pair_swap_local_search(h, n, n_iter=10000):
    """Try pairwise swaps to reduce max correlation."""
    h = h.copy()
    best_c5 = _compute_c5(h)
    R = _get_R(h)
    max_R = np.max(R)
    
    for it in range(n_iter):
        # Find the lag with max correlation
        k_max = np.argmax(R)
        G_k = _grad_R(h)[k_max]
        
        # Try to find a pair (i, j) where moving mass from i to j reduces R_{k_max}
        # We need G_k[i] > G_k[j] (so moving from i to j reduces R_k)
        # And h[i] > 0, h[j] < 1
        
        # Find best swap
        best_improvement = 0
        best_i, best_j = -1, -1
        
        for i in range(n):
            if h[i] < 1e-12:
                continue
            for j in range(n):
                if i == j or h[j] > 1.0 - 1e-12:
                    continue
                # Moving delta from i to j: change in R_k ≈ delta * (G_k[j] - G_k[i])
                # We want G_k[j] - G_k[i] < 0, i.e., G_k[i] > G_k[j]
                improvement = G_k[i] - G_k[j]
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_i, best_j = i, j
        
        if best_i < 0 or best_improvement < 1e-10:
            break
        
        # Determine max delta
        delta = min(h[best_i], 1.0 - h[best_j])
        # Try a fraction
        for frac in [1.0, 0.5, 0.25, 0.1]:
            d = frac * delta
            h_new = h.copy()
            h_new[best_i] -= d
            h_new[best_j] += d
            c5_new = _compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                h = h_new
                best_c5 = c5_new
                R = _get_R(h)
                break
    
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_state = {'alpha': 200, 'n': 45, 'target_raw': 0.0}


def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _get_R(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr[n - 1:]


def _grad_R(h):
    n = len(h)
    G = np.zeros((n, n))
    for k in range(n):
        e = n - k
        if k == 0:
            G[k] = 1.0 - 2.0 * h
        else:
            G[k, k:n] += (1.0 - h[0:e])
            G[k, 0:e] -= h[k:n]
    return G


def _project(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            scale = diff / np.sum(room)
            if scale >= 1.0:
                h[mask] = 1.0
            else:
                h[mask] += scale * room
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            tot = np.sum(h[mask])
            scale = -diff / tot
            if scale >= 1.0:
                h[mask] = 0.0
            else:
                h[mask] *= (1.0 - scale)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h


def _lse_val(R, alpha):
    Rmax = np.max(R)
    shifted = alpha * (R - Rmax)
    shifted = np.clip(shifted, -700.0, 0.0)
    return Rmax + np.log(np.sum(np.exp(shifted))) / alpha


def _obj_lse(x):
    return _lse_val(_get_R(x), _state['alpha'])


def _jac_lse(x):
    R = _get_R(x)
    G = _grad_R(x)
    Rmax = np.max(R)
    shifted = _state['alpha'] * (R - Rmax)
    shifted = np.clip(shifted, -700.0, 0.0)
    w = np.exp(shifted)
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(len(R))
        w[np.argmax(R)] = 1.0
    else:
        w /= sw
    return w @ G


def _cons_fun(x):
    return np.sum(x) - _state['n'] / 2.0


def _cons_jac(x):
    return np.ones(len(x))


def _obj_tlv(x):
    R = _get_R(x)
    v = np.maximum(R - _state['target_raw'], 0.0)
    return np.sum(v * v)


def _jac_tlv(x):
    R = _get_R(x)
    G = _grad_R(x)
    a = R - _state['target_raw']
    m = a > 0
    if not np.any(m):
        return np.zeros(len(x))
    return 2.0 * (a[m] @ G[m])


def _run_lse_slsqp(h, alpha, n, maxiter=300, ftol=1e-14):
    _state['alpha'] = alpha
    _state['n'] = n
    constraints = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_lse, h, jac=_jac_lse, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x


def _run_tlv_slsqp(h, target_raw, n, maxiter=500, ftol=1e-15):
    _state['target_raw'] = target_raw
    _state['n'] = n
    constraints = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_tlv, h, jac=_jac_tlv, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x


def _pair_swap_refine(h, n, max_iters=5000, time_limit=30.0):
    h = h.copy()
    best_c5 = _compute_c5(h)
    t_start = time.time()
    for _ in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        R = _get_R(h)
        k_max = int(np.argmax(R))
        G_all = _grad_R(h)
        G_k = G_all[k_max]
        best_improve = 1e-12
        best_i = -1
        best_j = -1
        for i in range(n):
            if h[i] < 1e-12:
                continue
            for j in range(n):
                if i == j or h[j] > 1.0 - 1e-12:
                    continue
                imp = G_k[i] - G_k[j]
                if imp > best_improve:
                    best_improve = imp
                    best_i = i
                    best_j = j
        if best_i < 0:
            break
        delta = min(h[best_i], 1.0 - h[best_j])
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            d = frac * delta
            h_trial = h.copy()
            h_trial[best_i] -= d
            h_trial[best_j] += d
            c5_trial = _compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-13:
                h = h_trial
                best_c5 = c5_trial
                break
    return h, best_c5
