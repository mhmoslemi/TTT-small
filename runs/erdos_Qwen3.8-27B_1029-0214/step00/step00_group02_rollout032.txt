import numpy as np
import time
import math

def eval_score(h):
    n = len(h)
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cor)) * dx

def make_feasible(h, target_sum):
    h = np.array(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def transfer(h, i, j, delta):
    """Transfer delta from position i to position j."""
    if delta <= 1e-15:
        return None
    if h[i] < delta or h[j] > 1.0 - delta:
        return None
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def get_active_lag(h):
    n = len(h)
    cor = np.correlate(h, 1.0 - h, mode="full")
    return int(np.argmax(cor)), cor

def greedy_lag_descent(h, n, target_sum, start_time, time_limit, rng):
    """Focus on reducing the active lag."""
    current_score = eval_score(h)
    
    while time.time() - start_time < time_limit:
        active_k, cor = get_active_lag(h)
        max_val = cor[active_k]
        
        # For the active lag, compute which positions contribute most
        # out[k] = sum_i h[i] * (1-h[i-k+n-1]) for valid i
        # We want to reduce the large terms
        
        # Compute marginal contributions
        # For lag k (0-indexed in the 2n-1 output):
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_lo = max(0, active_k - n + 1)
        i_hi = min(n - 1, active_k)
        
        if i_hi < i_lo:
            break
        
        # For each contributing i, the term is h[i] * (1-h[i-k+n-1])
        # partner index: j = i - k + n - 1
        partners = np.arange(i_lo, i_hi + 1) - active_k + n - 1
        terms = h[i_lo:i_hi+1] * (1.0 - h[partners])
        
        # Find the position with the largest contribution
        if len(terms) == 0:
            break
        max_local = np.argmax(terms)
        i_best = i_lo + max_local
        j_best = int(partners[max_local])
        
        # Try transferring from i_best to j_best (reduces h[i_best] and increases h[j_best])
        # This reduces the term h[i_best]*(1-h[j_best])
        # But it might increase other terms!
        
        accepted = False
        for frac in [0.5, 0.25, 0.1, 0.05, 0.01]:
            delta = frac * min(h[i_best], 1.0 - h[j_best])
            if delta < 1e-6:
                continue
            new_h = transfer(h, i_best, j_best, delta)
            if new_h is None:
                continue
            new_score = eval_score(new_h)
            if new_score < current_score - 1e-12:
                h = new_h
                current_score = new_score
                accepted = True
                break
        
        if not accepted:
            # Try random pairs among the top contributors
            top_n = min(10, len(terms))
            top_idx = np.argsort(terms)[-top_n:]
            for _ in range(20):
                pi = top_idx[rng.randint(len(top_idx))]
                ri = i_lo + pi
                rj = int(partners[pi])
                
                # Try transferring from ri to rj
                delta = rng.uniform(0.01, 0.1) * min(h[ri], 1.0 - h[rj])
                if delta < 1e-6:
                    continue
                new_h = transfer(h, ri, rj, delta)
                if new_h is None:
                    continue
                new_score = eval_score(new_h)
                if new_score < current_score - 1e-12:
                    h = new_h
                    current_score = new_score
                    accepted = True
                    break
            
            if not accepted:
                # No improvement found, break
                break
    
    return h, current_score

def sa_descent(h, n, target_sum, start_time, time_limit, rng):
    """Simulated annealing with pair transfers."""
    current_score = eval_score(h)
    best_h = h.copy()
    best_score = current_score
    T = 0.01
    
    step = 0
    while time.time() - start_time < time_limit and T > 1e-8:
        step += 1
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(0.005, 0.05)
        
        new_h = transfer(h, i, j, delta)
        if new_h is None:
            new_h = transfer(h, j, i, delta)
            if new_h is None:
                continue
        
        new_score = eval_score(new_h)
        diff = new_score - current_score
        
        if diff < 0 or (T > 1e-10 and rng.random() < math.exp(-diff / T)):
            h = new_h
            current_score = new_score
            if current_score < best_score:
                best_h = h.copy()
                best_score = current_score
        
        if step % 100 == 0:
            T *= 0.99
    
    return best_h, best_score
