```python
def project_to_feasible(y, n):
    """Project y to satisfy 0<=h<=1 and sum(h)=n/2"""
    target = n / 2.0
    h = np.clip(y, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    # Bisect on lambda: sum(clip(y - lambda, 0, 1)) = target
    lo = -n - 1.0
    hi = n + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine correction
    res = target - np.sum(h)
    if abs(res) > 1e-12:
        # Distribute proportionally
        if res > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-12:
                h += res / total_room * room
        else:
            room = h
            total_room = np.sum(room)
            if total_room > 1e-12:
                h += res / total_room * room
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    res = target - np.sum(h)
    if abs(res) > 1e-13:
        # Find an index to adjust
        if res > 0:
            idx = np.argmax(1.0 - h)
            adjust = min(res, 1.0 - h[idx])
            h[idx] += adjust
        else:
            idx = np.argmax(h)
            adjust = min(-res, h[idx])
            h[idx] -= adjust
    return h
```

```python
def compute_terms_and_slopes(h, d):
    """Compute all 2n-1 correlation term values and their slopes along direction d"""
    n = len(h)
    num_terms = 2 * n - 1
    values = np.zeros(num_terms)
    slopes = np.zeros(num_terms)
    one_minus_h = 1.0 - h
    one_minus_d = 1.0  # not needed directly
    
    for m in range(num_terms):
        r = m - (n - 1)  # lag
        i_lo = max(0, -r)
        i_hi = min(n - 1, n - 1 - r)
        if i_lo > i_hi:
            continue
        i_range = np.arange(i_lo, i_hi + 1)
        values[m] = np.sum(h[i_range] * one_minus_h[i_range + r])
        slopes[m] = np.sum(d[i_range] * one_minus_h[i_range + r] - h[i_range] * d[i_range + r])
    
    return values, slopes
```

```python
import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def project_to_feasible(y, n):
    target = n / 2.0
    h = np.clip(np.asarray(y, dtype=float), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo_b = -2.0 * n
    hi_b = 2.0 * n
    for _ in range(100):
        mid = (lo_b + hi_b) / 2.0
        s_mid = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s_mid > target:
            lo_b = mid
        else:
            hi_b = mid
    lam = (lo_b + hi_b) / 2.0
    h = np.clip(y - lam, 0.0, 1.0)
    res = target - np.sum(h)
    if abs(res) > 1e-12:
        if res > 0:
            room = 1.0 - h
            tot = np.sum(room)
            if tot > 1e-14:
                h += (res / tot) * room
        else:
            tot = np.sum(h)
            if tot > 1e-14:
                h += (res / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - np.sum(h)
    if abs(res) > 1e-14:
        if res > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + res)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + res)
    return h

def compute_terms_slopes(h, d):
    n = len(h)
    num_terms = 2 * n - 1
    values = np.zeros(num_terms)
    slopes = np.zeros(num_terms)
    omh = 1.0 - h
    for m in range(num_terms):
        r = m - (n - 1)
        i_lo = max(0, -r)
        i_hi = min(n - 1, n - 1 - r)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        values[m] = np.dot(h[idx], omh[idx + r])
        slopes[m] = np.dot(d[idx], omh[idx + r]) - np.dot(h[idx], d[idx + r])
    return values, slopes

def line_search(h, d, current_tau, dx, max_iter=50):
    n = len(h)
    values, slopes = compute_terms_slopes(h, d)
    
    # Check internal consistency
    # max(values)*dx should equal current_tau
    internal_tau = np.max(values) * dx
    if abs(internal_tau - current_tau) > 1e-8:
        return None  # numeric issue, skip
    
    # Feasible interval for delta
    eps_d = 1e-15
    lo = 0.0
    hi = 0.0
    for i in range(n):
        di = d[i]
        if di > eps_d:
            lo = max(lo, -h[i] / di)
            hi = min(hi, (1.0 - h[i]) / di)
        elif di < -eps_d:
            lo = max(lo, (1.0 - h[i]) / di)
            hi = min(hi, -h[i] / di)
    
    # Extend to include 0 (it should be feasible)
    lo = min(lo, 0.0)
    hi = max(hi, 0.0)
    
    if hi - lo < 1e-13:
        return None
    
    def f_at(delta):
        return float(np.max(values + delta * slopes) * dx)
    
    active_tol = 1e-9
    slope_tol = 1e-9
    
    # Check if delta=0 is already optimal
    max_v = np.max(values)
    act = values >= max_v - active_tol
    ld0 = float(np.min(slopes[act]))
    rd0 = float(np.max(slopes[act]))
    
    if ld0 <= slope_tol and rd0 >= -slope_tol:
        return None  # already at local min along this line
    
    # Determine search direction
    if rd0 < -slope_tol:
        # Decreasing to the right
        a, b = 0.0, hi
    elif ld0 > slope_tol:
        # Decreasing to the left
        a, b = lo, 0.0
    else:
        # Check both sides
        f0 = f_at(0.0)
        fr = f_at(hi * 0.5)
        fl = f_at(lo * 0.5)
        if fr < fl:
            a, b = 0.0, hi
        else:
            a, b = lo, 0.0
    
    # Bisection
    for _ in range(max_iter):
        if b - a < 1e-13:
            break
        mid = (a + b) / 2.0
        vm = values + mid * slopes
        max_vm = np.max(vm)
        act_m = vm >= max_vm - active_tol
        ld = float(np.min(slopes[act_m]))
        rd = float(np.max(slopes[act_m]))
        if ld <= slope_tol and rd >= -slope_tol:
            a = b = mid
            break
        if rd < -slope_tol:
            a = mid
        else:
            b = mid
    
    delta_opt = (a + b) / 2.0
    f_opt = f_at(delta_opt)
    
    # Check endpoints too
    f_lo = f_at(lo)
    f_hi = f_at(hi)
    f_0 = f_at(0.0)
    
    best_delta = delta_opt
    best_f = f_opt
    if f_lo < best_f:
        best_f = f_lo
        best_delta = lo
    if f_hi < best_f:
        best_f = f_hi
        best_delta = hi
    if f_0 < best_f:
        best_f = f_0
        best_delta = 0.0
    
    if best_f < current_tau - 1e-10:
        return best_delta, best_f
    return None

def make_pair_direction(n, i, j):
    d = np.zeros(n)
    d[i] = 1.0
    d[j] = -1.0
    return d

def make_random_direction(n, rng):
    # Random signs
    d = rng.choice([-1.0, 1.0], size=n)
    # Zero mean
    d -= np.mean(d)
    # Normalize
    norm = np.max(np.abs(d))
    if norm > 1e-15:
        d /= norm
    # Re-zero mean after normalization
    d -= np.mean(d)
    return d

def make_block_direction(n, rng):
    """Create a direction that moves a block up and another block down"""
    n_blocks = n // 3
```

def project_to_feasible(y, n):
    target = n / 2.0
    h = np.clip(np.asarray(y, dtype=float), 0.0, 1.0)

    if abs(s - target) < 1e-14:

    lo_b = -2.0 * n
    hi_b = 2.0 * n

        s_mid = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s_mid > target:
            lo_b = mid
        else:
            hi_b = mid
    lam = (lo_b + hi_b) / 2.0
    h = np.clip(y - lam, 0.0, 1.0)

        if res > 0:
            room = 1.0 - h
            tot = np.sum(room)
            if tot > 1e-14:
                h += (res / tot) * room

            tot = np.sum(h)
            if tot > 1e-14:
