                T *= cooling
            continue

        new_s = score_fn(new_h)

                    new_h = new_H.astype(np.float64) / M
                    s_new = score_fn(new_h)
                    if s_new < best_s - 1e-13:
                        best_H = new_H
                        best_h = new_h
                        best_s = s_new
                        improved = True
        if not improved:
            for _ in range(50):
                s = int(rng.integers(0, n))
                t = int(rng.integers(0, n))
                if s == t:
                    continue
                units = int(rng.integers(1, 17))
                if best_H[s] >= units and best_H[t] + units <= M:
                    new_H = best_H.copy()
                    new_H[s] -= units
                    new_H[t] += units
                    new_h = new_H.astype(np.float64) / M
                    s_new = score_fn(new_h)
                    if s_new < best_s - 1e-13:
                        best_H = new_H
                        best_h = new_h
                        best_s = s_new
                        improved = True
                elif best_H[t] >= units and best_H[s] + units <= M:
                    new_H = best_H.copy()
                    new_H[t] -= units
                    new_H[s] += units
                    new_h = new_H.astype(np.float64) / M
                    s_new = score_fn(new_h)
                    if s_new < best_s - 1e-13:
                        best_H = new_H
                        best_h = new_h
                        best_s = s_new
                        improved = True
            if not improved:
                break
    return best_h, best_s

def simulated_annealing(H, M, rng, t_ref, time_limit):
    n = len(H)
    cur_H = H.copy()
    cur_h = cur_H.astype(np.float64) / M
    cur_s = score_fn(cur_h)
    best_H = cur_H.copy()
    best_h = cur_h.copy()
    best_s = cur_s
    T = 0.04
    T_min = 0.0005
    cooling = 0.996
    end_time = t_ref + time_limit
    step = 0
    while time.perf_counter() < end_time and T > T_min:
        s = int(rng.integers(0, n))
        t = int(rng.integers(0, n))
        if s == t:
            step += 1
            if step % 100 == 0:
                T *= cooling
            continue
        units = int(rng.integers(1, 6))
        if cur_H[s] >= units and cur_H[t] + units <= M:
            new_H = cur_H.copy()
            new_H[s] -= units
            new_H[t] += units
        elif cur_H[t] >= units and cur_H[s] + units <= M:
            new_H = cur_H.copy()
            new_H[t] -= units
            new_H[s] += units
        else:
            step += 1
            if step % 100 == 0:
                T *= cooling
            continue
        new_h = new_H.astype(np.float64) / M
        new_s = score_fn(new_h)
        delta = new_s - cur_s
        if delta < 0:
            cur_H = new_H
            cur_h = new_h
            cur_s = new_s
            if new_s < best_s:
                best_H = new_H.copy()
                best_h = new_h.copy()
                best_s = new_s
        else:
            if rng.random() < math.exp(-delta / T):
                cur_H = new_H
                cur_h = new_h
                cur_s = new_s
        step += 1
        if step % 50 == 0:
            T *= cooling
    return best_h, best_s

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return h_dst

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    reserve = 5.0
    rng = np.random.default_rng(seed)

    n = 47
    h_init = None
    try:
        if 'initial_h_values' in globals():
            h_arr = np.array(initial_h_values, dtype=np.float64).copy()
            if len(h_arr) > 0 and np.all(np.isfinite(h_arr)):
                h_init = h_arr
                n = len(h_init)
    except Exception:
        pass

    if h_init is None:
        h_init = np.full(n, 0.5)

    if n % 2 != 0:
        n = n + 1
        if len(h_init) < n:
            h_init = np.concatenate([h_init, np.array([0.5])])
        else:
            h_init = h_init[:n]

    best_h = repair_feasible(h_init, n)
    best_s = score_fn(best_h)

    elapsed = time.perf_counter() - t_start
    remaining = budget_s - elapsed - reserve
    if remaining > 15:
        phase_time = min(remaining * 0.12, 45.0)
        best_h, best_s = continuous_descent(best_h, n, rng, 200, phase_time, time.perf_counter())

    elapsed = time.perf_counter() - t_start
    remaining = budget_s - elapsed - reserve
    if remaining > 20:
        n_seeds = 6
        seed_time = max(5.0, (remaining * 0.18) / n_seeds)
        for i in range(n_seeds):
            if time.perf_counter() - t_start > budget_s - reserve:
                break
            r = rng.random()
            if r < 0.3:
                h_seed = make_seed_cosine(n, rng)
            elif r < 0.5:
                h_seed = make_seed_binary(n, rng)
            elif r < 0.7:
                h_seed = best_h + rng.normal(0.0, 0.05, size=n)
                h_seed = repair_feasible(h_seed, n)
            else:
                h_seed = best_h.copy()
                idx_perm = rng.permutation(n)
                for k in range(min(5, n // 2)):
                    a = int(idx_perm[2 * k])
                    b = int(idx_perm[2 * k + 1])
                    if a < n and b < n and a != b:
                        tmp = best_h[a] * 0.5
                        best_h[a] = 0.5 * (best_h[a] + best_h[b])
                        best_h[b] = tmp
                h_seed = repair_feasible(h_seed, n)
            h_cand, s_cand = continuous_descent(h_seed, n, rng, 150, seed_time, time.perf_counter())
            if s_cand < best_s:
                best_h = h_cand
                best_s = s_cand

    elapsed = time.perf_counter() - t_start
    remaining = budget_s - elapsed - reserve
    if remaining > 15:
        M = 64
        H_grid = gridify(best_h, M)
        grid_time = min(remaining * 0.15, 90.0)
        h_grid, s_grid = grid_descent(H_grid, M, rng, 400, grid_time, time.perf_counter())
        if s_grid < best_s:
            best_h = h_grid
            best_s = s_grid

    elapsed = time.perf_counter() - t_start
    remaining = budget_s - elapsed - reserve
    if remaining > 30:
        M_sa = 64
        H_sa = gridify(best_h, M_sa)
        sa_time = min(remaining * 0.30, 250.0)
        h_sa, s_sa = simulated_annealing(H_sa, M_sa, rng, time.perf_counter(), sa_time)
        if s_sa < best_s:
            best_h = h_sa
            best_s = s_sa

    elapsed = time.perf_counter() - t_start
    remaining = budget_s - elapsed - reserve
    if remaining > 10:
        fine_time = min(remaining * 0.35, 120.0)
        best_h, best_s = continuous_descent(best_h, n, rng, 400, fine_time, time.perf_counter())

    elapsed = time.perf_counter() - t_start
    remaining = budget_s - elapsed - reserve
    if remaining > 20:
        M2 = 128
        H_fine = gridify(best_h, M2)
        fine_grid_time = min(remaining * 0.3, 90.0)
        h_fine, s_fine = grid_descent(H_fine, M2, rng, 300, fine_grid_time, time.perf_counter())
        if s_fine < best_s:
            best_h = h_fine
            best_s = s_fine

    best_h = np.clip(np.array(best_h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - best_h
            room_sum = room.sum()
            if room_sum > 1e-15:
                best_h = best_h + (room / room_sum) * diff
        else:
            h_sum = best_h.sum()
            if h_sum > 1e-15:
                best_h = best_h + (best_h / h_sum) * diff
        best_h = np.clip(best_h, 0.0, 1.0)

    best_s = score_fn(best_h)
    return best_h, best_s, n
