        hi = 2.0

            return np.sum(np.clip(h - lam, 0, 1))

            lo -= 1.0

        n = len(h)
        dx = 2.0 / n
        c_vals = compute_all_c(h)

        tol = 1e-8
        active = c_vals >= (cmax - tol)
        n_active = np.sum(active)
        if n_active == 0:

import numpy as np
from scipy.optimize import minimize
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_scaled_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    lo = -2.0
    hi = 2.0
    while np.sum(np.clip(h - lo, 0.0, 1.0)) < target:
        lo -= 1.0
    while np.sum(np.clip(h - hi, 0.0, 1.0)) > target:
        hi += 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(hp < 1.0 - 1e-12)[0]
        else:
            idx = np.where(hp > 1e-12)[0]
        if len(idx) > 0:
            hp[idx[0]] = np.clip(hp[idx[0]] + diff, 0.0, 1.0)
    return hp


def lse_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c_max = np.max(c)
    exp_c = np.exp(c - c_max)
    return (c_max + np.log(np.sum(exp_c))) / beta


def lse_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c_max = np.max(c)
    exp_c = np.exp(c - c_max)
    total = np.sum(exp_c)
    w = exp_c / total
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-15:
            continue
        rk = k - (n - 1)
        wk = w[k] / beta
        if rk == 0:
            grad += wk * dx * (1.0 - 2.0 * x)
        elif rk > 0:
            grad[0:n - rk] += wk * dx * (1.0 - x[rk:n])
            grad[rk:n] -= wk * dx * x[0:n - rk]
        else:
            rn = -rk
            grad[rn:n] += wk * dx * (1.0 - x[0:n - rn])
            grad[0:n - rn] -= wk * dx * x[rn:n]
    return grad


def sum_eq(x):
    return np.sum(x) - len(x) / 2.0


def sum_eq_jac(x):
    return np.ones(len(x))


def topk_subgradient(h, top_k, sharp):
    n = len(h)
    dx = 2.0 / n
    c = compute_scaled_corr(h)
    c_max = np.max(c)
    if top_k < len(c):
        top_idx = np.argsort(c)[-top_k:]
    else:
        top_idx = np.arange(len(c))
    sel_c = c[top_idx]
    w = np.exp((sel_c - c_max) * sharp)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        k = int(idx)
        r = k - (n - 1)
        if r == 0:
            grad += wi * dx * (1.0 - 2.0 * h)
        elif r > 0:
            grad[0:n - r] += wi * dx * (1.0 - h[r:n])
            grad[r:n] -= wi * dx * h[0:n - r]
        else:
            rn = -r
            grad[rn:n] += wi * dx * (1.0 - h[0:n - rn])
            grad[0:n - rn] -= wi * dx * h[rn:n]
    return grad, c_max


def slsqp_optimize(h_start, n, beta, maxiter):
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    res = minimize(lse_obj, h_start, args=(beta,), jac=lse_grad,
                   method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_feasible(res.x, n)

    best_local = cur_val

            v_trial = evaluate_c5(h_trial)
            if v_trial < cur_val - 1e-15:

                cur_val = v_trial
                improved = True
                if v_trial < best_local:
                    best_local = v_trial
                    best_h_local = h.copy()
                break
            alpha *= 0.5
        if not improved:
            for trial_alpha in [0.01, 0.005, 0.001, 0.0005]:

                v_trial = evaluate_c5(h_trial)
                if v_trial < cur_val:
                    h = h_trial
                    cur_val = v_trial

                if cur_val > best_local + 0.001:
                    h = best_h_local.copy()
                    cur_val = best_local
    return best_h_local, best_local


def coordinate_descent(h, n, rng, max_iters, time_limit_fn):

    best_local = cur_val
    best_h_local = h.copy()
    for it in range(max_iters):
        if time_limit_fn():
            break
        i = int(rng.randint(n))

            h_trial = h.copy()
            h_trial[i] = np.clip(h_trial[i] + delta, 0.0, 1.0)
            diff = n / 2.0 - np.sum(h_trial)

            v_trial = evaluate_c5(h_trial)
            if v_trial < cur_val:

                cur_val = v_trial
