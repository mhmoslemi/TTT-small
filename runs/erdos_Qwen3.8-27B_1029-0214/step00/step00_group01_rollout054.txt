        return np.ones(n)

        else:

def select_active_ks(corr_values, dx, c_max, tol, max_count):
    """Select lag indices k whose correlation is close to c_max."""

def exact_c5(h):

def project_box_simplex(z, target):

        self.n = n
        self.ks = ks
        self.dx = dx
        self.deadline = deadline

    def check_time(self):
        if time.time() > self.deadline:
            raise TimeoutError()

    def fun(self, x):
        self.check_time()
        return x[self.n]

    def jac_fun(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g

    def cin(self, x):
        self.check_time()
        h = x[:self.n]
        t = x[self.n]
        n = self.n
        vals = np.zeros(len(self.ks))
        for idx, k in enumerate(self.ks):
            d = abs(k)
            L = n - d

            else:
                vals[idx] = t - self.dx * np.dot(h[:L], 1.0 - h[d:])
        return vals

        self.check_time()
        h = x[:self.n]
