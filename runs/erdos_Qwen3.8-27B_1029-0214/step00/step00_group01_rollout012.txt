import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def project_to_feasible(v, n):
    # Project v onto {x: 0 <= x[i] <= 1, sum(x) = n/2}
    target = n / 2.0
    # Binary search on lambda
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Small correction
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        x = x + diff / n
        x = np.clip(x, 0.0, 1.0)
    return x

def get_active_lags(corr, n_active):
    # Return indices of the top n_active correlation values
    idx = np.argsort(corr)[::-1][:n_active]
    return idx

def compute_gradients(h, active_indices, n):
    # Compute gradients of corr at active_indices w.r.t. h
    # corr = np.correlate(h, 1-h, mode='full')
    # For output index r, k = r - (n-1)
    grads = np.zeros((len(active_indices), n))
    for ii, r in enumerate(active_indices):
        k = r - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            # c_k = sum_{n=0}^{n-1-k} h[n+k] * (1-h[n])
            # ∂c_k/∂h[i] = (1-h[i-k]) * [k <= i <= n-1] - h[i+k] * [0 <= i <= n-1-k]
            valid1 = np.arange(k, n)
            g[valid1] += (1.0 - h[valid1 - k])
            valid2 = np.arange(0, n - k)
            g[valid2] -= h[valid2 + k]
        else:
            m = -k
            # c_{-m} = sum_{n=m}^{n-1} h[n-m] * (1-h[n])
            # ∂c_{-m}/∂h[i] = (1-h[i+m]) * [0 <= i <= n-1-m] - h[i-m] * [m <= i <= n-1]
            valid1 = np.arange(0, n - m)
            g[valid1] += (1.0 - h[valid1 + m])
            valid2 = np.arange(m, n)
            g[valid2] -= h[valid2 - m]
        grads[ii] = g
    return grads

def solve_direction_lp(grads, n):
    # Solve: min t s.t. g_i · d <= t, sum(d) = 0, |d_i| <= 1
    m = len(grads)
    # Variables: d[0..n-1], t
    # c = [0, ..., 0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: g_i · d - t <= 0
    # A_ub shape (m, n+1)
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(m)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -1 <= d_i <= 1, t unbounded below
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1], res.fun
        else:
            # Fallback: negative mean gradient of active constraints
            g_mean = np.mean(grads, axis=0)
            d = -g_mean
            d -= d.mean()  # enforce sum = 0
            d = np.clip(d, -1, 1)
            return d, None, None
    except:
        g_mean = np.mean(grads, axis=0)
        d = -g_mean
        d -= d.mean()
        d = np.clip(d, -1, 1)
        return d, None, None

def line_search(h, d, c_current, dx, n):
    # Try to find alpha that improves
    best_alpha = 0
    best_c = c_current
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

            best_c = c_new
            best_alpha = alpha
            h_best = h_new.copy()
    if best_alpha > 0:
        return h_best, best_c
    return None, c_current
