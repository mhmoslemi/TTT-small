from scipy.optimize import minimize

def exact_objective(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def is_feasible(h, tol=1e-8):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -tol or np.max(h) > 1 + tol:
        return False
    target = n * 0.5
    if abs(np.sum(h) - target) > tol * max(1, n):
        return False
    return True

def project_feasible(y, n):
    target = n * 0.5
    y = np.asarray(y, dtype=np.float64)
    y = np.clip(y, 0, 1)
    # Water filling to adjust sum
    # h[i] = clip(y[i] - lam, 0, 1)
    # Find lam such that sum(h) = target
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo+hi)/2, 0, 1)
    # Fix any remaining tiny error
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute diff among non-bound entries
        free = (h > 1e-10) & (h < 1 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0, 1)
    return h

def smooth_objective_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    L = corr  # L_k values
    maxL = np.max(L)
    shifted = tau * (L - maxL)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    F = maxL + np.log(sum_exp) / tau
    p = exp_shifted / sum_exp
    # Gradient
    grad = np.zeros(n)
    one_minus_h = 1 - h
    for j in range(n):
        idx_start = j
        idx_end = min(2*n-2, j+n-1)
        if idx_start <= idx_end:
            idxs = np.arange(idx_start, idx_end+1)
            m1s = j + n - 1 - idxs
            s1 = np.dot(p[idxs], one_minus_h[m1s])
        else:
            s1 = 0.0
        idx_start2 = max(0, n-1-j)
        idx_end2 = 2*n-2-j
        if idx_start2 <= idx_end2:
            idxs2 = np.arange(idx_start2, idx_end2+1)
            m2s = j + idxs2 - n + 1
            s2 = np.dot(p[idxs2], h[m2s])
        else:
            s2 = 0.0
        grad[j] = dx * (s1 - s2)
    return F, grad

def make_optimization_args(n):
    bounds = [(0, 1)] * n
    constraints = [{'type': 'eq', 
                    'fun': lambda x: np.sum(x) - n*0.5,
                    'jac': lambda x: np.ones(n)}]
    return bounds, constraints

def run_slsqp(h0, tau, n, maxiter=200, ftol=1e-8):
    bounds = [(0, 1)] * n
    constraint_fun = lambda x: np.sum(x) - n*0.5
    constraint_jac = lambda x: np.ones(n)
    constraints = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    def obj(x):
        val, _ = smooth_objective_and_grad(x, tau)
        return val
    
    def grad(x):
        _, g = smooth_objective_and_grad(x, tau)
        return g
    
    try:
        result = minimize(obj, h0, jac=grad, method='SLSQP',
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': maxiter, 'ftol': ftol})
        return result.x
    except:
        return h0

def local_search(h, n, rng, n_steps=1000):
    best_h = h.copy()
    best_c5 = exact_objective(best_h)
    dx = 2.0 / n
    target = n * 0.5

        j = rng.integers(0, n)

            delta = direction * amount
            new_h[i] += delta
            new_h[j] -= delta
            if new_h[i] < 0 or new_h[i] > 1 or new_h[j] < 0 or new_h[j] > 1:

            if new_c5 < best_c5 - 1e-12:

def resample(h_src, n_src, n_dst):
    """Resample h from n_src points to n_dst points, preserving integral."""
    # h_src is a step function on [0, 2] with n_src steps
    # We want n_dst steps
    src_centers = (np.arange(n_src) + 0.5) * 2.0 / n_src
    dst_centers = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    # Interpolate (step function -> use nearest or linear)
    h_dst = np.interp(dst_centers, src_centers, h_src)
    h_dst = np.clip(h_dst, 0, 1)
    # Project to satisfy sum constraint
    h_dst = project_feasible(h_dst, n_dst)
    return h_dst

    return starts
