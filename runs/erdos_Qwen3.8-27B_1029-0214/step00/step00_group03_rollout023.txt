import numpy as np

def exact_evaluator(h):
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    return dx * np.max(corr)

def project_feasible(v, n):
    """Project v onto the feasible set: 0 <= v_i <= 1, sum(v) = n/2"""
    target = n / 2.0
    v = np.clip(v, 0.0, 1.0)
    diff = target - np.sum(v)
    if abs(diff) < 1e-12:
        return v
    if diff > 0:
        # Need to add mass; add to coordinates with most room (closest to 1 but not at 1)
        # Actually, add to coordinates farthest from 1
        order = np.argsort(1.0 - v)  # most room first
        for idx in order:
            if diff <= 0:
                break
            add = min(diff, 1.0 - v[idx])
            v[idx] += add
            diff -= add
    else:
        # Need to remove mass from coordinates with most mass
        order = np.argsort(v)  # least mass first (remove from highest? No, we need to remove, so go from highest)
        order = np.argsort(-v)  # most mass first
        diff = -diff
        for idx in order:
            if diff <= 0:
                break
            rem = min(diff, v[idx])
            v[idx] -= rem
            diff -= rem
    # Final exact fix
    residual = target - np.sum(v)
    if abs(residual) > 1e-15:
        # Adjust one coordinate
        if residual > 0:
            idx = np.argmin(v)
        else:
            idx = np.argmax(v)
        v[idx] += residual
    v = np.clip(v, 0.0, 1.0)
    return v

def compute_pair_coefficients(h, q, i, j, n):
    """Compute coefficient vector c_lag for pair (i,j) perturbation h[i]+=d, h[j]-=d"""
    lags = np.arange(-(n-1), n)  # 2n-1 values
    # c_lag = q[i-lag] (if valid) - q[j-lag] (if valid) - h[i+lag] (if valid) + h[j+lag] (if valid)
    c = np.zeros(2*n-1)
    for m, lag in enumerate(lags):
        val = 0.0
        il = i - lag
        if 0 <= il < n:
            val += q[il]
        jl = j - lag
        if 0 <= jl < n:
            val -= q[jl]
        ip = i + lag
        if 0 <= ip < n:
            val -= h[ip]
        jp = j + lag
        if 0 <= jp < n:
            val += h[jp]
        c[m] = val
    return c

import numpy as np
import time

def exact_evaluator(h):
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    return float(dx * np.max(corr))

def project_feasible(v, n):
    target = n / 2.0
    v = np.array(v, dtype=np.float64).copy()
    v = np.clip(v, 0.0, 1.0)
    diff = target - np.sum(v)
    if abs(diff) < 1e-13:
        return v
    if diff > 0:
        # Add mass where there's most room
        order = np.argsort(1.0 - v)
        for idx in order:
            if diff <= 1e-15:
                break
            add = min(diff, 1.0 - v[idx])
            v[idx] += add
            diff -= add
    else:
        # Remove mass from where there's most
        order = np.argsort(v)
        diff = -diff
        for idx in order[::-1]:
            if diff <= 1e-15:
                break
            rem = min(diff, v[idx])
            v[idx] -= rem
            diff -= rem
    # Exact final fix
    residual = target - np.sum(v)
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = int(np.argmin(v))
        else:
            idx = int(np.argmax(v))
        v[idx] += residual
    return np.clip(v, 0.0, 1.0)

def compute_coefficients(q_rev, h, i, j, n):
    c = np.zeros(2 * n - 1)
    c[i:i+n] += q_rev
    c[j:j+n] -= q_rev
    c[n-1-i:2*n-1-i] -= h
    c[n-1-j:2*n-1-j] += h
    return c

def compute_quadratic(i, j, n):
    quad = np.zeros(2 * n - 1)
    quad[n-1] = -2.0
    quad[i - j + n - 1] += 1.0
    quad[j - i + n - 1] += 1.0
    return quad

def line_search(R, c, quad, lo, hi):
    if hi - lo < 1e-12:
        return 0.0
    # Coarse search
    n_coarse = 12
    best_d = 0.0
    best_val = np.inf
    ds = np.linspace(lo, hi, n_coarse)
    for d in ds:
        val = np.max(R + d * c + d * d * quad)
        if val < best_val:
            best_val = val
            best_d = d
    # Fine search
    width = (hi - lo) / n_coarse
    lo2 = max(lo, best_d - width)
    hi2 = min(hi, best_d + width)
    if hi2 - lo2 < 1e-12:
        return best_d
    n_fine = 12
    ds2 = np.linspace(lo2, hi2, n_fine)
    for d in ds2:
        val = np.max(R + d * c + d * d * quad)
        if val < best_val:
            best_val = val
            best_d = d
    return best_d

def optimize_pairs(h, time_budget, start_time):
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    R = np.correlate(h, q, mode='full')
    current_c = dx * np.max(R)
    q_rev = q[::-1].copy()
    
    deadline = start_time + time_budget
    pairs_idx = [(i, j) for i in range(n) for j in range(i+1, n)]
    
    improved_any = True
    pass_count = 0
    while improved_any and time.time() < deadline:
        improved_any = False
        pass_count += 1
        # Shuffle pairs for diversity
        rng_order = np.random.RandomState(pass_count)
        order = rng_order.permutation(len(pairs_idx))
        for pidx in order:
            if time.time() >= deadline:
                return h, current_c
            i, j = pairs_idx[pidx]
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-12:
                continue

            quad = compute_quadratic(i, j, n)
            d_opt = line_search(R, c, quad, lo, hi)
            if abs(d_opt) < 1e-14:
                continue
            # Exact evaluation
            h[i] += d_opt
            h[j] -= d_opt

            c_new = dx * np.max(R_new)
            if c_new < current_c - 1e-13:

                R = R_new

                improved_any = True

                # Revert

                h[j] += d_opt
                q = 1.0 - h
                # R stays the same since we reverted
    
    return h, current_c
