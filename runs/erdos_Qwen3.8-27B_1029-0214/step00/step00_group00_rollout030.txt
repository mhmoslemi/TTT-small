def project_feasible(h, n):
    target = n / 2.0
    def F(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0)) - target
    lo, hi = -1.0, 1.0
    while F(lo) < 0:
        lo -= 1.0
    while F(hi) > 0:
        hi += 1.0

def lse_objective(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(c)
    shifted = beta * (c - max_c)
    # For numerical stability
    max_shift = np.max(shifted)
    if max_shift > 0:
        shifted = shifted - max_shift
    log_sum = max_c + (max_shift + np.log(np.sum(np.exp(shifted)))) / beta

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    if abs(current_sum - target) < 1e-12:
        return h
    def F(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0)) - target
    lo, hi = -1.0, 1.0
    max_iter = 100
    for _ in range(max_iter):
        if F(lo) > 0 and F(hi) < 0:
            break
        if F(lo) < 0:
            lo -= 1.0
        if F(hi) > 0:
            hi += 1.0
    lam = brentq(F, lo, hi, xtol=1e-15)
    result = np.clip(h - lam, 0.0, 1.0)
    return result

def lse_value(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(c)
    shifted = beta * (c - max_c)
    max_s = np.max(shifted)
    log_total = max_s + np.log(np.sum(np.exp(shifted - max_s)))
    return max_c + log_total / beta

def lse_grad(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(c)
    shifted = beta * (c - max_c)
    max_s = np.max(shifted)
    exps = np.exp(shifted - max_s)
    w = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        m = j - (n - 1)
        if m >= 0:
            i1 = np.arange(0, n - m)
            i2 = np.arange(m, n)
            grad[i1] += w[j] * dx * (1.0 - h[i1 + m])
            grad[i2] += w[j] * dx * (-h[i2 - m])
        else:
            p = -m
            i1 = np.arange(p, n)
            i2 = np.arange(0, n - p)
            grad[i1] += w[j] * dx * (1.0 - h[i1 - p])
            grad[i2] += w[j] * dx * (-h[i2 + p])
    return grad

        if F(lo) < 0.0 and F(hi) > 0.0:
            break
        if F(lo) < 0.0:
            lo -= 1.0
        if F(hi) > 0.0:
            hi += 1.0
    lam = brentq(F, lo, hi, xtol=1e-15, maxiter=200)
    return np.clip(h - lam, 0.0, 1.0)

def lse_value(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(c)
    shifted = beta * (c - max_c)
    ms = np.max(shifted)
    log_total = ms + np.log(np.sum(np.exp(shifted - ms)))
    return max_c + log_total / beta

def lse_grad(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(c)
    shifted = beta * (c - max_c)
    ms = np.max(shifted)
    exps = np.exp(shifted - ms)
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        m = j - (n - 1)
        if m >= 0:
            i1 = np.arange(0, n - m)
            i2 = np.arange(m, n)
            grad[i1] += w[j] * dx * (1.0 - h[i1 + m])
            grad[i2] += w[j] * dx * (-h[i2 - m])
        else:
            p = -m
            i1 = np.arange(p, n)
            i2 = np.arange(0, n - p)
            grad[i1] += w[j] * dx * (1.0 - h[i1 - p])
            grad[i2] += w[j] * dx * (-h[i2 + p])
    return grad

def _slsqp_fun(h):
    return lse_value(h, _g_n, _g_beta)

def _slsqp_grad(h):
    return lse_grad(h, _g_n, _g_beta)

def _slsqp_con(h):
    return np.sum(h) - _g_n / 2.0

def _slsqp_con_jac(h):
    return np.ones(_g_n)

def slsqp_optimize(h_init, n, beta, time_limit=90):
    global _g_n, _g_beta
    _g_n = n
    _g_beta = beta
    constraints = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_con_jac}]
    bounds = [(0.0, 1.0)] * n
    t0 = time.time()
    try:
        result = minimize(_slsqp_fun, h_init, jac=_slsqp_grad, method='SLSQP',
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 800, 'ftol': 1e-18})
        h_out = np.clip(result.x, 0.0, 1.0)
        h_out = project_feasible(h_out, n)
        c5_out = compute_c5(h_out, n)
        c5_in = compute_c5(h_init, n)
        if c5_out <= c5_in + 1e-12:
            return h_out
    except Exception:
        pass
    return h_init

def pg_optimize(h_init, n, beta, max_iter=200000, time_limit=90):
    h = h_init.copy()
    f_curr = lse_value(h, n, beta)
    best_h = h.copy()
    best_f = f_curr
    alpha = 0.05 / max(1.0, beta)
    prev_grad = lse_grad(h, n, beta)
    t0 = time.time()
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        grad = lse_grad(h, n, beta)
        alpha_new = alpha * 1.0
        improved = False
        for ls in range(30):
            h_try = h - alpha_new * grad
            h_try = project_feasible(h_try, n)
            f_try = lse_value(h_try, n, beta)
            if f_try < f_curr - 1e-16:
                h = h_try
                f_curr = f_try
                improved = True
                if f_try < best_f:
                    best_f = f_try
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                alpha = min(alpha_new * 1.05, 0.5 / beta)
                break
            alpha_new *= 0.5
        if not improved:
            stall += 1
            alpha *= 0.5
            if stall > 500:
                h = best_h + 0.001 * np.random.randn(n)
                h = project_feasible(h, n)
                f_curr = lse_value(h, n, beta)
                stall = 0
        prev_grad = grad
    return best_h

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    f = interp1d(x_old, h_old, kind='linear', bounds_error=False, fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new)

def make_random_h(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    start_time = time.time()
    np.random.seed(seed)
    deadline = start_time + budget_s

    def remaining():
        return deadline - time.time()

    h0 = np.array(initial_h_values, dtype=np.float64)
    n0 = len(h0)
    h0 = project_feasible(h0, n0)

    c5_best = compute_c5(h0, n0)
    h_best = h0.copy()
    n_best = n0

    # Phase 1: Optimize at n0 with SLSQP and PG, increasing beta
    h_curr = h0.copy()
    for beta in [5, 10, 20, 50, 100, 200]:
        if remaining() < 120:
            break
        time_for_this = min(remaining() - 100, 120)
        if time_for_this < 20:
            break

        h_slsqp = slsqp_optimize(h_curr, n0, beta, time_limit=max(20, time_for_this * 0.5))
        c5_s = compute_c5(h_slsqp, n0)
        if c5_s < c5_best:
            c5_best = c5_s
            h_best = h_slsqp.copy()
            h_curr = h_slsqp.copy()

        h_pg = pg_optimize(h_curr, n0, beta, max_iter=100000, time_limit=max(10, time_for_this * 0.5))
        c5_p = compute_c5(h_pg, n0)
        if c5_p < c5_best:
            c5_best = c5_p
            h_best = h_pg.copy()
            h_curr = h_pg.copy()

    # Phase 2: Random restarts at n0
    for trial in range(8):
        if remaining() < 100:
            break
        h_r = h_best + 0.005 * np.random.randn(n0)
        h_r = project_feasible(h_r, n0)
        h_r = slsqp_optimize(h_r, n0, 50, time_limit=min(40, max(15, remaining() - 80)))
        c5_r = compute_c5(h_r, n0)
        if c5_r < c5_best:
            c5_best = c5_r
            h_best = h_r.copy()
            n_best = n0

    # Phase 3: Try higher n values
    for n_try in [48, 50, 55, 60, 80, 100]:
        if remaining() < 80:
            break
        h_interp = interpolate_h(h_best, n_best, n_try)
        c5_i = compute_c5(h_interp, n_try)
        if c5_i < c5_best:
            c5_best = c5_i
            h_best = h_interp.copy()
            n_best = n_try

        if remaining() < 60:
            break

        time_budget = min(remaining() - 50, 150)
        if time_budget < 30:
            continue

        h_opt = slsqp_optimize(h_best, n_best, 20, time_limit=max(20, time_budget * 0.4))
        c5_o = compute_c5(h_opt, n_best)
        if c5_o < c5_best:
            c5_best = c5_o
            h_best = h_opt.copy()
            n_best = n_try

        if remaining() > 60:
            h_opt2 = slsqp_optimize(h_best, n_best, 50, time_limit=max(15, min(60, remaining() - 40)))
            c5_o2 = compute_c5(h_opt2, n_best)
            if c5_o2 < c5_best:
                c5_best = c5_o2
                h_best = h_opt2.copy()
                n_best = n_best

        if remaining() > 40:
            h_opt3 = pg_optimize(h_best, n_best, 50, max_iter=50000, time_limit=min(40, remaining() - 30))
            c5_o3 = compute_c5(h_opt3, n_best)
            if c5_o3 < c5_best:
                c5_best = c5_o3
                h_best = h_opt3.copy()
                n_best = n_best

    # Final: ensure feasibility and compute exact C5
    h_best = project_feasible(h_best, n_best)
    c5_final = compute_c5(h_best, n_best)

    return h_best, c5_final, n_best
