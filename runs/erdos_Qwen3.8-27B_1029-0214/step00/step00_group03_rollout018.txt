def exact_eval(h):
    """Compute C5 for a given h array."""
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    return dx * np.max(corr)

    """Project v onto the feasible set: 0 <= v[i] <= 1, sum(v) = n/2."""
    v = np.array(v, dtype=np.float64)
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(v)
    if abs(diff) < 1e-12:
        return v
    if diff > 0:
        # Need to add mass; add to entries with most room
        order = np.argsort(1.0 - v)  # most room first
        remaining = diff
        for idx in order:
            add = min(remaining, 1.0 - v[idx])
            v[idx] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    else:
        # Need to remove mass
        order = np.argsort(v)  # least value first (remove from small ones? No, remove from large)
        order = np.argsort(-v)  # most value first
        remaining = -diff
        for idx in order:
            rem = min(remaining, v[idx])
            v[idx] -= rem
            remaining -= rem
            if remaining <= 1e-12:
                break

                a = lo_val
                b = hi_val

                c_pt = b - golden * (b - a)

                    h[i] -= pt

                        d_pt = c_pt
                        f_d = f_c
                        c_pt = b - golden * (b - a)
                        f_c = eval_d(c_pt)

                        a = c_pt
                        c_pt = d_pt
                        f_c = f_d
                        d_pt = a + golden * (b - a)
                        f_d = eval_d(d_pt)

                best_d = (a + b) / 2.0
                if abs(best_d) < 1e-12:

                h[j] -= best_d
                best_c = exact_eval(h)

                h[j] += best_d

import numpy as np
import time

def exact_eval(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def project_feasible(v, n):
    v = np.array(v, dtype=np.float64).copy()
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(v)
    if abs(diff) < 1e-12:
        return v
    if diff > 0:
        order = np.argsort(1.0 - v)
        for idx in order:
            add = min(diff, 1.0 - v[idx])
            v[idx] += add
            diff -= add
            if diff <= 1e-14:
                break
    else:
        order = np.argsort(-v)
        for idx in order:
            rem = min(-diff, v[idx])
            v[idx] -= rem
            diff += rem
            if diff >= -1e-14:
                break
    return v

def pair_descent(h, n, passes, n_samples, deadline, rng=None):
    h = np.array(h, dtype=np.float64).copy()
    current_c = exact_eval(h)
    for p in range(passes):
        if deadline is not None and time.perf_counter() > deadline:
            break
        pass_improvement = 0.0
        if rng is not None:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            rng.shuffle(pairs)
        else:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        for i, j in pairs:
            hi_val = min(1.0 - h[i], h[j])
            lo_val = max(-h[i], h[j] - 1.0)
            if hi_val - lo_val < 1e-10:
                continue
            best_d = 0.0
            best_c = current_c
            for s in range(n_samples + 1):
                d = lo_val + (hi_val - lo_val) * s / n_samples
                if abs(d) < 1e-14:
                    continue
                h[i] += d
                h[j] -= d
                c = exact_eval(h)
                h[i] -= d
                h[j] += d
                if c < best_c:
                    best_c = c
                    best_d = d
            if best_d != 0.0 and best_c < current_c - 1e-15:
                h[i] += best_d
                h[j] -= best_d
                current_c = best_c
                pass_improvement += pass_improvement
        if pass_improvement < 1e-12:
            break
    h = project_feasible(h, n)
    current_c = exact_eval(h)
    return h, current_c

def upsample_linear(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old, dtype=np.float64)
    x_new = np.linspace(0.0, float(n_old - 1), n_new)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_feasible(h_new, n_new)
    return h_new
