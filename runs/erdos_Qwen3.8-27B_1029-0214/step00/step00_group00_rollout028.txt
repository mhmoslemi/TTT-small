    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def fix_sum(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    residual = target - np.sum(h)
    tol = max(1e-12, 1e-14 * n)
    if abs(residual) < tol:
        return h
    if residual > 0:
        room = 1.0 - h
        total_room = room.sum()
        if total_room > 1e-14:
            h = h + residual * room / total_room
        else:
            # Can't add more, just clip
            pass
    else:
        total_mass = h.sum()
        if total_mass > 1e-14:
            h = h + residual * h / total_mass
        else:
            pass

                res = minimize(lse_obj, h_cur, args=(n, beta), method='SLSQP',
                             bounds=make_bounds(n),
                             constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n,)}],
                             options={'maxiter': 200, 'ftol': 1e-14})
                h_cur = fix_sum(res.x, n)
                c5_new = evaluate_c5(h_cur)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_cur.copy()
            except Exception:
                pass

    for n_try in [40, 50, 55]:
        if (time.time() - t0) > budget_s * 0.94:
            break
        n_cur = len(best_h)
        x_old = np.linspace(0, 2, n_cur, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = fix_sum(h_interp, n_try)
        c5_interp = evaluate_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
        h_opt = h_interp.copy()
        for beta in [30, 100, 500, 2000, 10000]:
            if (time.time() - t0) > budget_s * 0.96:
                break
            try:
                res = minimize(lse_obj, h_opt, args=(n_try, beta), method='SLSQP',
                             bounds=make_bounds(n_try),
                             constraints=[{'type': 'eq', 'fun': sum_con, 'args': (n_try,)}],
                             options={'maxiter': 300, 'ftol': 1e-14})
                h_opt = fix_sum(res.x, n_try)
                c5_new = evaluate_c5(h_opt)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_opt.copy()
            except Exception:
                pass

    if (time.time() - t0) < budget_s * 0.98:
        h = best_h.copy()
        n_cur = len(h)
        for _ in range(100):
            if (time.time() - t0) > budget_s * 0.985:
                break
            best_imp = 0.0
            best_move = None
            for _trial in range(200):
                i = rng.randint(0, n_cur)
                j = rng.randint(0, n_cur)
                if i == j:
                    continue
                delta = rng.choice([0.001, 0.003, 0.01, 0.03]) * rng.choice([-1.0, 1.0])
                hi = h[i] + delta
                hj = h[j] - delta
                if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
                    continue
                h[i] = hi
                h[j] = hj
                c5_try = evaluate_c5(h)
                h[i] = hi - delta
                h[j] = hj + delta
                imp = best_c5 - c5_try
                if imp > best_imp:
                    best_imp = imp
                    best_move = (i, j, delta)
            if best_move is not None:
                i_m, j_m, d_m = best_move
                h[i_m] += d_m
                h[j_m] -= d_m
                best_c5 = evaluate_c5(h)
                best_h = h.copy()

    return best_h, best_c5, len(best_h)
