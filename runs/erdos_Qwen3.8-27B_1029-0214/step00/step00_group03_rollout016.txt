def exact_eval(h):
    """Compute C5 = dx * max(correlate(h, 1-h, 'full'))"""
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode='full')
    return float(dx * np.max(corr))

def project_feasible(h, target=None):

    h = np.asarray(h, dtype=np.float64).copy()
    n = len(h)

    # Clip

def exact_eval(h):
    """Compute C5 = dx * max(correlate(h, 1-h, 'full'))"""
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def project_feasible(h, target=None):
    """Project to [0,1]^n with exact sum = target"""
    h = np.asarray(h, dtype=np.float64).copy()
    n = len(h)
    if target is None:
        target = n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    # Distribute the difference
    if diff > 0:
        room = 1.0 - h
        total_room = np.sum(room)
        if total_room < diff:
            # Shouldn't happen for valid target
            h[:] = 1.0
            h[:] = min(h, 1.0)
        # Proportional distribution
        h += diff * room / max(total_room, 1e-15)
        np.clip(h, 0.0, 1.0, out=h)
    else:
        mass = h.copy()
        total_mass = np.sum(mass)
        if total_mass < -diff:
            h[:] = 0.0
        h += diff * mass / max(total_mass, 1e-15)
        np.clip(h, 0.0, 1.0, out=h)
    
    # Iterative fix for any remaining error
    for _ in range(10):

        h[mask] += per

        for k in active:
            wk = w[k]
            if wk < 1e-8:
                continue
            s = k - (n - 1)
            for i in range(n):
                idx1 = i - s
                if 0 <= idx1 < n:
                    g[i] += wk * (1.0 - h[idx1])
                idx2 = i + s
                if 0 <= idx2 < n:
                    g[i] -= wk * h[idx2]
        
        # Step size
        alpha = 1.0 / (0.1 + 0.5 * np.sqrt(it + 1))
        
        # Projected step
        h_new = project_feasible(h - alpha * g)
        c_new = exact_eval(h_new)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    return best_h, best_c

def pair_descent(h, n_passes, time_limit_fn):
    """Pair-coordinate descent"""
    h = project_feasible(h.copy())
    n = len(h)
    best_c = exact_eval(h)
    
    for p in range(n_passes):
        if time_limit_fn():
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if hi - lo < 1e-12:
                    continue
                # Try candidate deltas
                candidates = [lo, hi, (lo + hi) * 0.5]
                # Add the point that equalizes top two
                best_d = 0.0
                best_v = best_c
                for d in candidates:
                    h_trial = h.copy()
                    h_trial[i] += d
                    h_trial[j] -= d
                    c_trial = exact_eval(h_trial)
                    if c_trial < best_v:
                        best_v = c_trial
                        best_d = d
                if best_v < best_c - 1e-12 and best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    h = project_feasible(h)
                    best_c = exact_eval(h)
                    improved = True
        if not improved:
            break
    return h, best_c

def upsample(h, new_n):
    """Upsample via linear interpolation"""
    old_n = len(h)
    if new_n == old_n:
        return project_feasible(h.copy())
    x_old = np.arange(old_n, dtype=np.float64)
    x_new = np.linspace(0, old_n - 1, new_n)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, target=new_n / 2.0)

One optimization: instead of evaluating the full correlation for each candidate δ, I can compute only the changed correlation values. Let me also think about whether the subgradient method is actually making progress. The issue is that the problem is non-convex (bilinear), so the subgradient method on a linearized approximation might not converge to a good solution. But it can still make progress by moving away from high-overlap regions.

A better approach might be to combine:
- Use the subgradient to determine a direction
- Then do a pair-descent in the projected direction (decompose the direction into pair moves)

Or alternatively:
- Use the subgradient to identify which coordinates should increase and which should decrease
- Do targeted pair moves in that direction

Let me think about a hybrid approach:

1. Compute the subgradient g (direction to decrease the objective)
2. Identify coordinates that should increase (g[i] > 0 means decreasing h[i] helps, so we want h[i] to decrease; wait no, we move in direction -g, so if g[i] > 0, h[i] should decrease)
   Actually: we set h_new = h - αg. If g[i] > 0, h[i] decreases.
3. Find pairs (i,j) where g[i] > 0 and g[j] < 0 (i should decrease, j should increase)
4. For each such pair, do a pair move
