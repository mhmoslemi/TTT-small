def project_box_sum(x, S):

            lo = mid
        else:

import numpy as np
import time

def project_box_sum(x, S):
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def compute_ov(h):
    return np.correlate(h, 1.0 - h, mode='full')

def eval_c5(h):
    return float(np.max(compute_ov(h)) * (2.0 / len(h)))

def grad_ov(h, k):
    n = len(h)
    L = k - n + 1
    g = np.zeros(n)
    if L >= 0:
        idx = np.arange(n - L)
        g[idx] += 1.0 - h[idx + L]
        g[idx + L] -= h[idx]
    else:
        M = -L
        idx = np.arange(M, n)
        g[idx] += 1.0 - h[idx - M]
        g[idx - M] -= h[idx]
    return g
