import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def optimize_pair(h, i, j):
    c = h[i] + h[j]
    lo = max(0.0, c - 1.0)
    hi = min(1.0, c)
    if hi - lo < 1e-14:
        return
    for _ in range(40):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        h[i], h[j] = t1, c - t1
        f1 = evaluate_c5(h)
        h[i], h[j] = t2, c - t2
        f2 = evaluate_c5(h)
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    h[i] = (lo + hi) / 2.0
    h[j] = c - h[i]

def coordinate_descent_stochastic(h, n, max_time, rng):
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    deadline = time.time() + max_time
    stall = 0
    while time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        optimize_pair(h, i, j)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                for _ in range(5):
                    a = rng.randint(n)

                    d = rng.uniform(-0.1, 0.1)

                    h[b] = np.clip(h[b] + d, 0.0, 1.0)
                project_fix(h, n)
                stall = 0
    return best_h, best_c5

def coordinate_descent_systematic(h, n, max_time):
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    deadline = time.time() + max_time
    while time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                old_i, old_j = h[i], h[j]
                optimize_pair(h, i, j)
                if h[i] != old_i or h[j] != old_j:
                    improved = True
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        if not improved:
            break
    return best_h, best_c5

def project_fix(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() > target:
            hi = mid
        else:
            lo = mid
    h[:] = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
