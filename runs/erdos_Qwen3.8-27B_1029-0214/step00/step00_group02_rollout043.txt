def score(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    if not np.all(np.isfinite(h)):

    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(np.sum(h) - n / 2.0) > tol * n:

            g += (1.0 - h[m - shift])
        if 0 <= m + shift < n:
            g -= h[m + shift]
