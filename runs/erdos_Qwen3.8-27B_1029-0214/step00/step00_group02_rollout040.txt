    new_x = np.linspace(0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    h_new = np.interp(new_x, old_x, h, period=2.0) if False else np.interp(new_x, old_x, h)

    current_sum = np.sum(h_new)
    if current_sum > 1e-10:
        h_new = h_new * (target_sum / current_sum)

    # Fix sum by adjusting
    diff = target_sum - np.sum(h_new)

    return h_new

import numpy as np

def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_correlation(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(proj))
        else:
            idx = int(np.argmax(proj))
        proj[idx] = np.clip(proj[idx] + diff, 0.0, 1.0)
    return proj

def lis_score_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = compute_correlation(h)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    max_shifted = float(shifted.max())
    exp_shifted = np.exp(shifted - max_shifted)
    sum_exp = float(exp_shifted.sum())
    log_sum = max_shifted + np.log(sum_exp)
    score = c_max + log_sum / beta
    w = exp_shifted / sum_exp
    A = np.zeros(n)
    B = np.zeros(n)
    for m in range(n):
        a_val = 0.0
        b_val = 0.0
        for j in range(n):
            k1 = m - j + n - 1
            if 0 <= k1 <= 2 * n - 2:
                a_val += (1.0 - h[j]) * w[k1]
            k2 = j - m + n - 1
            if 0 <= k2 <= 2 * n - 2:
                b_val += h[j] * w[k2]
        A[m] = a_val
        B[m] = b_val
    grad = dx * (A - B)
    return score, grad

def lis_optimize(h0, beta_start, beta_end, max_iters, time_limit, rng):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_score(h)
    start_time = time.perf_counter()
    lr = 0.5
    for it in range(max_iters):
        if time.perf_counter() - start_time > time_limit:
            break
        progress = it / max(1, max_iters)
        beta = beta_start * (beta_end / beta_start) ** progress
        score_lis, grad = lis_score_grad(h, beta)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-10:
            break
        grad_dir = -grad / grad_norm
        step = lr
        for _ in range(20):
            h_try = h + step * grad * (step / max(step, 1e-15))
            h_try = h - step * grad
            h_try = project_feasible(h_try, target_sum)
            s_try = compute_score(h_try)
            exact_lis_try, _ = lis_score_grad(h_try, beta)
            if exact_lis_try < score_lis + 1e-12:
                h = h_try
                score_lis = exact_lis_try
                if s_try < best_score:
                    best_score = s_try
                    best_h = h.copy()
                break
            step *= 0.5
            if step < 1e-8:
                break
        if it % 50 == 0:
            s_cur = compute_score(h)
            if s_cur < best_score:
                best_score = s_cur
                best_h = h.copy()
    return best_h, best_score

def grid_search(H, M, max_rounds, time_limit, rng):

        G = np.outer(h_cur, 1.0 - h_cur)
        flat = G.ravel()
        top_idx = np.argsort(flat)[::-1][:100]
        improved = False
        for fi in top_idx:
            if time.perf_counter() - start_time > time_limit:
                break
            s = int(fi // n)
            t = int(fi % n)
            if s == t:
                continue
            max_units = min(int(best_H[s]), M - int(best_H[t]))
            if max_units <= 0:
                continue

                if units > max_units:
                    continue
                cand_H = best_H.copy()
                cand_H[s] -= units
                cand_H[t] += units
                cand_h = cand_H.astype(np.float64) / M

                if s_new < best_score - 1e-13:

def anneal_grid(H, M, time_limit, rng, t_init=0.01, t_final=1e-5):
    n = len(H)
    H_cur = H.copy()
    h_cur = H_cur.astype(np.float64) / M
    score_cur = compute_score(h_cur)
    H_best = H_cur.copy()
    score_best = score_cur
    start_time = time.perf_counter()
    step_count = 0
    while time.perf_counter() - start_time < time_limit:
        step_count += 1
        progress = min(1.0, step_count / max(1, int(time_limit / 0.001)))
        temp = t_init * (t_final / t_init) ** progress
        h_cur = H_cur.astype(np.float64) / M
        G = np.outer(h_cur, 1.0 - h_cur)
        if rng.random() < 0.5:
            flat = G.ravel() + rng.random(G.ravel().shape) * 0.001
            fi = int(np.argmax(flat))
            s = fi // n
            t = fi % n
        else:
            s = int(rng.integers(0, n))
            t = int(rng.integers(0, n))
        if s == t:
            continue
        max_units = min(int(H_cur[s]), M - int(H_cur[t]))
        if max_units <= 0:
            continue
        max_step = max(1, max_units // 4)
        units = int(rng.integers(1, max_step + 1))
        H_new = H_cur.copy()
        H_new[s] -= units
        H_new[t] += units
        h_new = H_new.astype(np.float64) / M
        score_new = compute_score(h_new)
        delta = score_new - score_cur
        if delta < 0 or rng.random() < math.exp(-delta / max(temp, 1e-12)):
            H_cur = H_new
            score_cur = score_new
            if score_new < score_best:
                score_best = score_new
                H_best = H_new.copy()
    return H_best, score_best

def make_grid_from_h(h, M):
    n = len(h)
    target = M * n // 2
    H = np.floor(np.clip(h, 0.0, 1.0) * M).astype(np.int64)
    H = np.clip(H, 0, M)
    err = target - int(H.sum())
    if err > 0:
        order = np.argsort(np.clip(h, 0.0, 1.0) * M - H)
        for i in range(err):
            H[order[i % n]] += 1
    elif err < 0:
        order = np.argsort(-(np.clip(h, 0.0, 1.0) * M - H))
        for i in range(-err):
            H[order[i % n]] -= 1
    H = np.clip(H, 0, M)
    diff = target - int(H.sum())
    if diff > 0:
        for i in range(diff):
            idx = int(np.argmax(M - H))
            H[idx] += 1
    elif diff < 0:
        for i in range(-diff):
            idx = int(np.argmax(H))
            H[idx] -= 1
    return H

def resample_to_n(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n + 1)
    h_ext = np.concatenate([h, [h[0]]])
    new_x = np.linspace(0.0, 2.0 / new_n, new_n) + np.linspace(0.0, 2.0, new_n) * 0 + np.arange(new_n) * (2.0 / new_n) + 1.0 / new_n
    h_new = np.interp(new_x, old_x, h_ext)
    target_sum = new_n / 2.0
    current_sum = h_new.sum()
    if current_sum > 1e-10:
        h_new = h_new * (target_sum / current_sum)
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target_sum - h_new.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmin(h_new))
        else:
            idx = int(np.argmax(h_new))
        h_new[idx] = np.clip(h_new[idx] + diff, 0.0, 1.0)
    return h_new
