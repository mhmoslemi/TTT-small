def subgradient_step(h, dx, learning_rate):
    """One subgradient descent step with projection."""
    g = 1 - h
    corr = np.correlate(h, g, mode='full')
    k_star = np.argmax(corr)

    # Linearization: c_k(h) ≈ c_k(h*) + sum_i [∂c_k/∂h[i]]_{h*} * (h[i] - h*[i])

    # => sum_i grad_i * h[i] + t <= c_k(h*) + sum_i grad_i * h*[i]

    b_ub = []

    # Simple approach: add a constant to make the sum work, then clip
    # This doesn't perfectly maintain the sum, but it's a good approximation
    # For exact projection, use bisection
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(h_p) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_top_lags(h, n, k=15):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argpartition(corr, -k)[-k:]
    top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
    lags = top_idx - (n - 1)
    return lags

def solve_lp(h, lags, n, max_t=None):
    dx = 2.0 / n
    target = n / 2.0
    nv = n + 1
    c_obj = np.zeros(nv)
    c_obj[-1] = 1.0
    
    A_rows = []
    b_vals = []
    
    for lag in lags:
        # Compute grad and c_val efficiently
        grad = np.zeros(n)
        c_val = 0.0
        if lag >= 0:
            valid = slice(0, n - lag)
            h_i = h[valid]
            h_j = h[lag:lag + (n - lag)]
            c_val = np.sum(h_i * (1.0 - h_j)) * dx
            grad[valid] += (1.0 - h_j) * dx
            grad[lag:lag + (n - lag)] += (-h_i) * dx
        else:
            lag_p = -lag
            valid = slice(0, n - lag_p)
            h_i = h[valid]
            h_j = h[lag_p:lag_p + (n - lag_p)]
            c_val = np.sum(h_i * (1.0 - h_j)) * dx
            # For negative lag, h[i+lag] = h[i-lag_p]
            # grad[i] gets contribution from h[i] as first factor: (1-h[i-lag_p])*dx for i >= lag_p
            # grad[i] gets contribution from h[i] as second factor: -h[i+lag_p]*dx for i <= n-lag_p-1
            # Wait, let me re-derive for negative lag
            # c_lag = sum_i h[i] * (1-h[i+lag]) * dx where lag < 0
            # = sum_i h[i] * (1-h[i-|lag|]) * dx for i >= |lag|
            # Let i' = i - |lag|, then = sum_{i'=0}^{n-|lag|-1} h[i'+|lag|] * (1-h[i']) * dx
            # grad wrt h[i']: 
            #   h[i'] appears as (1-h[i']) in the term where i = i'+|lag|: contributes -h[i'+|lag|]*dx
            #   h[i'] appears as h[i] in the term where i = i': contributes (1-h[i+lag])*dx = (1-h[i-|lag|])*dx if i >= |lag|
            
            # Let me just use the general formula
            # c_lag = sum over valid (i, j) pairs where j = i+lag
            # For lag < 0: valid pairs are (i, i+lag) where i+lag >= 0, i.e., i >= -lag = lag_p
            # c_lag = sum_{i=lag_p}^{n-1} h[i] * (1-h[i-lag_p]) * dx
            # grad[i]:
            #   If i >= lag_p: h[i] appears as first factor: (1-h[i-lag_p])*dx
            #   If i <= n-1-lag_p: h[i] appears as second factor (h[i+lag_p]): -h[i+lag_p]*dx
            #   Wait: h[i-lag_p] is the second factor. h[i] is the first factor when i >= lag_p.
            #   h[i] is the second factor when i = j - lag_p where j >= lag_p, i.e., when i + lag_p >= lag_p, i.e., always for i >= 0.
            #   But we need j = i + lag_p < n, i.e., i < n - lag_p.
            
            for i in range(lag_p):
                # h[i] only appears as second factor (h[i+lag_p])
                grad[i] += (-h[i + lag_p]) * dx
            for i in range(lag_p, n - lag_p):
                # h[i] appears as both first and second factor
                grad[i] += (1.0 - h[i - lag_p]) * dx + (-h[i + lag_p]) * dx
            for i in range(n - lag_p, n):
                # h[i] only appears as first factor
                grad[i] += (1.0 - h[i - lag_p]) * dx
            
            # Recompute c_val for negative lag
            c_val = np.sum(h[lag_p:] * (1.0 - h[:-lag_p])) * dx
        
        row = np.zeros(nv)
        row[:n] = grad
        row[n] = -1.0
        A_rows.append(row)
        b_vals.append(c_val - np.dot(grad, h))
    
    if not A_rows:
        return h.copy(), c_obj @ h  # No constraints
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n].copy(), float(res.x[n])
    except:
        pass
    
    return None, None

def optimize_at_n(h, n, time_limit):
    """Run successive linearization at fixed n."""
    start = time.time()
    best_c5 = compute_c5(h)
    h_best = h.copy()
    stagnation = 0
    top_k = 10
    
    while time.time() - start < time_limit:
        if best_c5 <= 0.3808:
            break
        
        lags = get_top_lags(h, n, k=top_k)
        h_new, t_new = solve_lp(h, lags, n)
        
        if h_new is None:
            break
        
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-12:
            h = h_new
            best_c5 = c5_new
            h_best = h.copy()
            stagnation = 0
        else:
            # Mix with current for stability
            alpha = 0.5
            h_trial = (1 - alpha) * h + alpha * h_new
            h_trial = project_sum(h_trial, n / 2.0)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-12:
                h = h_trial
                best_c5 = c5_trial
                h_best = h.copy()
                stagnation = 0
            else:
                stagnation += 1
        
        if stagnation > 20:
            # Increase top_k for more constraints
            top_k = min(top_k + 5, 2 * n - 1)
            if top_k > 2 * n - 3:
                # Perturb
                h_pert = h + 0.02 * np.random.randn(n)
                h = project_sum(h_pert, n / 2.0)
            top_k = max(5, top_k // 2)  # Reset
            stagnation = 0
    
    return h_best, best_c5
