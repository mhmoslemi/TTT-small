def exact_score(h):

    dx = 2.0 / n

def is_feasible(h, tol=1e-9):
    n = len(h)
    if n == 0:
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    target = n / 2.0
    if abs(np.sum(h) - target) > tol * n:
        return False
    return True

def gridify(h, M):
    n = len(h)
    target = M * n // 2
    H = np.clip(np.round(h * M), 0, M).astype(int)
    s = int(H.sum())
    err = target - s
    if err > 0:
        # Need to add err units
        # Sort by fractional part (descending) to decide where to add
        frac = h * M - np.floor(h * M)
        order = np.argsort(-frac)
        for idx in order:
            if err <= 0:
                break
            if H[idx] < M:
                H[idx] += 1
                err -= 1
        # If still err > 0, add to any position with capacity
        if err > 0:
            for idx in range(n):
                if err <= 0:
                    break
                if H[idx] < M:
                    H[idx] += 1
                    err -= 1
    elif err < 0:
        # Need to remove -err units
        frac = h * M - np.floor(h * M)
        order = np.argsort(frac)

                break

                err += 1

import numpy as np
import time
import math

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def gridify(h, M):
    n = len(h)
    target = M * n // 2
    h_clip = np.clip(h, 0.0, 1.0)
    H = np.clip(np.round(h_clip * M), 0, M).astype(np.int32)
    s = int(H.sum())
    err = target - s
    if err > 0:
        frac = h_clip * M - np.floor(h_clip * M)
        order = np.argsort(-frac, kind='stable')
        for idx in order:
            if err <= 0:
                break
            if H[idx] < M:
                H[idx] += 1
                err -= 1
        if err > 0:
            for idx in range(n):
                if err <= 0:
                    break
                if H[idx] < M:
                    H[idx] += 1
                    err -= 1
    elif err < 0:
        frac = h_clip * M - np.floor(h_clip * M)
        order = np.argsort(frac, kind='stable')
        for idx in order:
            if err >= 0:
                break
            if H[idx] > 0:
                H[idx] -= 1
                err += 1
        if err < 0:
            for idx in range(n):
                if err >= 0:
                    break
                if H[idx] > 0:
                    H[idx] -= 1
                    err += 1
    return H

def fix_sum(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    iters = 0
    while abs(diff) > 1e-10 and iters < n * 2:
        iters += 1
        if diff > 0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            add = min(diff, room[idx])
            h[idx] += add
            diff -= add
        else:
            idx = int(np.argmax(h))
            rem = min(-diff, h[idx])
            h[idx] -= rem
            diff += rem
    return h

def resample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def local_polish(H, M, max_steps=5000, time_limit=60.0, start_time=None):
    n = len(H)
    inv_M = 1.0 / M
    h = H.astype(np.float64) * inv_M
    curr_score = exact_score(h)
    best_score = curr_score
    best_H = H.copy()
    no_improve = 0

    for step in range(max_steps):
        if start_time is not None and time.perf_counter() - start_time > time_limit:
            break

        improved = False

        # Lag-targeted approach: find dominant lag and its top contributing pair
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_lags = np.argsort(corr)[::-1][:3]

        for k in top_lags:
            k = int(k)
            off = k - (n - 1)
            if off >= 0:
                if off >= n:

            else:
                o = -off
                if o >= n:

                terms = h[:n - o] * (1.0 - h[o:])

            if len(terms) == 0:

            max_i = int(np.argmax(terms))
            if off >= 0:

                t = max_i
            else:
                o = -off
                s = max_i

            if H[s] >= 1 and H[t] < M:
