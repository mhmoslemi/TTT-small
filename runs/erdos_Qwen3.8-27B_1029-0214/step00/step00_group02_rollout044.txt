def evaluate_score(h):
    """Exact score computation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def is_feasible(h, tol=1e-9):

def gridify(h, M):
    """Convert float h to integer representation on dyadic grid [0, M]."""
    n = len(h)
    target_int = M * n // 2
    y = np.clip(h, 0, 1) * M
    base = np.floor(y).astype(np.int64)
    base = np.clip(base, 0, M)
    current_sum = int(np.sum(base))
    err = target_int - current_sum
    # If err > 0, need to add units
    if err > 0:
        # Prefer indices with most room
        room = M - base
        # Sort by room descending, tie-break by index
        order = np.argsort(-room, kind='stable')
        added = 0
        for idx in order:
            if added >= err:
                break
            can_add = min(int(M - base[idx]), err - added)
            base[idx] += can_add
            added += can_add
    elif err < 0:
        need_remove = -err
        # Prefer indices with most value
        order = np.argsort(-base, kind='stable')
        removed = 0
        for idx in order:
            if removed >= need_remove:
                break
            can_remove = min(int(base[idx]), need_remove - removed)
            base[idx] -= can_remove
            removed += can_remove

def make_balanced_seed(n, rng):
    """Create a balanced feasible seed."""
    target = n / 2.0
    # Start with 0.5 everywhere
    h = np.full(n, 0.5)
    # Add some structure
    phase = rng.uniform(0, 2 * np.pi)
    freq = rng.uniform(1, 3)
    amplitude = rng.uniform(0.2, 0.45)
    h += amplitude * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint using water-filling
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Proportional adjustment
        if diff > 0:
            # Add to elements with most room
            room = 1.0 - h
            order = np.argsort(-room)
            added = 0
            for idx in order:
                if added >= diff:
                    break
                can = min(1.0 - h[idx], diff - added)
                h[idx] += can
                added += can
        else:
            # Remove from elements with most value
            order = np.argsort(-h)
            removed = 0
            for idx in order:
                if removed >= -diff:
                    break
                can = min(h[idx], -diff + removed)
                h[idx] -= can
                removed += can
    return h
