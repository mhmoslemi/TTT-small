def evaluate_erdos(h):
    """Compute C5 = max(correlate(h, 1-h, full)) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_gradients(h, lag_positions):
    """
    Compute gradients of corr[lag_positions] with respect to h.
    lag_positions are indices into the full correlation array (0 to 2n-2).
    Returns array of shape (len(lag_positions), n).
    """
    n = len(h)
    offset = n - 1

    for idx, pos in enumerate(lag_positions):
        k = pos - offset  # actual lag
        if k >= 0:
            # corr at lag k = sum_{j=0}^{n-1-k} h[j+k]*(1-h[j])
            # d/d(h[i]) = (1-h[i-k]) for i >= k  (from h[j+k] term)
            #            -h[i+k]   for i <= n-1-k (from (1-h[j]) term)

            else:
                # First part: (1-h[i-k]) for i in [k, n-1]
                part1 = np.zeros(n)
                part1[k:] = 1.0 - h[:n-k]
                # Second part: -h[i+k] for i in [0, n-1-k]
                part2 = np.zeros(n)
                part2[:n-k] = -h[k:]
                grads[idx, :] = part1 + part2
        else:
            m = -k
            # corr at lag -m = sum_{j=0}^{n-1-m} h[j]*(1-h[j+m])
            # d/d(h[i]) = (1-h[i+m]) for i <= n-1-m (from h[j] term)
            #            -h[i-m]   for i >= m   (from (1-h[j+m]) term)

                part2 = np.zeros(n)

    return grads
