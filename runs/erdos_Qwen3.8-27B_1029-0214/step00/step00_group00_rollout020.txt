import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h[~np.isfinite(h)] = 0.5
    target = n / 2.0
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(120):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0.0, 1.0)
    residual = target - h_proj.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            room = 1.0 - h_proj
            if room.sum() > 1e-12:
                h_proj = h_proj + residual * room / room.sum()
        else:
            if h_proj.sum() > 1e-12:
                h_proj = h_proj + residual * h_proj / h_proj.sum()
    return np.clip(h_proj, 0.0, 1.0)


def lse_objective(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(logsumexp(beta * corr) / beta)


def sum_constraint(h, n):
    return np.sum(h) - n / 2.0


def run_slsqp_optimize(h_start, n, beta, maxiter):
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(lse_objective, h_start, args=(n, beta), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return res

                h_try[i] += delta
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break

            break
