```python
import numpy as np
from scipy.optimize import minimize
import time

def project_sum(h, n):
    """Project h to satisfy sum(h) = n/2 while staying in [0,1]."""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    residual = target - np.sum(h)
    if abs(residual) < 1e-12:
        return h
    if residual > 0:
        # Need to increase sum
        room = 1.0 - h
        if np.sum(room) < residual:
            # Can't increase enough
            h = np.where(h < 1.0, 1.0, h)
            # Still not enough, just accept
            return np.clip(h, 0, 1)
        # Distribute proportionally
        scale = residual / np.sum(room)
        h = h + scale * room
    else:
        # Need to decrease sum
        capacity = h
        if np.sum(capacity) < -residual:
            h = np.where(h > 0.0, 0.0, h)
            return np.clip(h, 0, 1)
        scale = -residual / np.sum(capacity)
        h = h - scale * capacity
    return np.clip(h, 0, 1)

    dx = 2.0 / n

    return np.max(corr) * dx

    return corr * dx

    Compute gradient of T_k w.r.t. h.
    k is the index in np.correlate(h, 1-h, 'full') output (0 to 2n-2).
    T_k = dx * corr[k]

    dx = 2.0 / n

    #   = sum_{j=0}^{n-1-offset} h[j+offset] * (1-h)[j]
    # If offset < 0: let r = -offset
    #   corr[k] = sum_{i=0}^{n-1-r} h[i] * (1-h)[i+r]
    
    offset = k - (n - 1)

        g[offset:n] += (1 - h[0:e])

        g[0:e] += (1 - h[r:n])
        g[r:n] += -h[0:e]

    max_scaled = np.max(scaled)

        logsum = max_scaled + np.log(np.sum(np.exp(shifted)))
    else:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def project_sum(h, n):
    h = np.clip(h, 0, 1).copy()
    target = n / 2.0
    residual = target - np.sum(h)
    if abs(residual) < 1e-12:
        return h
    if residual > 0:
        room = 1.0 - h
        total_room = np.sum(room)
        if total_room <= 0:
            return h
        if total_room >= residual:
            h = h + (residual / total_room) * room
        else:
            h = np.minimum(h + room, 1.0)
    else:
        total_cap = np.sum(h)
        if total_cap <= 0:
            return h
        if total_cap >= -residual:
            h = h + (residual / total_cap) * h
        else:
            h = np.maximum(h + h, 0.0)
    # Final projection
    h = np.clip(h, 0, 1)
    # Fix any remaining rounding
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        idx = np.argmax(1.0 - h) if residual > 0 else np.argmin(h)
        h[idx] += residual
        h = np.clip(h, 0, 1)
    return h

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def correlation_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr * dx

def logsumexp_hf(h, alpha):
    T = correlation_values(h)
    m = np.max(T)
    if m > 200 / alpha:
        return m + (1.0/alpha) * np.log(np.sum(np.exp(alpha * (T - m))))
    return (1.0/alpha) * np.log(np.sum(np.exp(alpha * T)))

def logsumexp_grad(h, alpha):
    T = correlation_values(h)
    aT = alpha * T
    m = np.max(aT)
    e = np.exp(aT - m)
    p = e / np.sum(e)
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        offset = k - (n - 1)
        e_len = n - abs(offset)
        if offset >= 0:
            g[offset:n] += p[k] * (1 - h[0:e_len])
            g[0:e_len] += -p[k] * h[offset:n]
        else:
            r = -offset
            g[0:e_len] += p[k] * (1 - h[r:n])
            g[r:n] += -p[k] * h[0:e_len]
    return dx * g

def violation_hf(h, t, active_set=None):
    T = correlation_values(h)
    if active_set is not None:
        T = T[active_set]
    viol = np.maximum(T - t, 0)
    return np.sum(viol ** 2)

def violation_grad(h, t, active_set=None):
    T = correlation_values(h)
    if active_set is not None:
        T_full = T
        T = T[active_set]
        indices = active_set
    else:
        T_full = T
        indices = np.arange(len(T))
    viol = np.maximum(T - t, 0)
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    mask = viol > 1e-12
    for idx_i in np.where(mask)[0]:
        k = indices[idx_i]
        coeff = 2.0 * viol[idx_i]
        offset = k - (n - 1)
        e_len = n - abs(offset)
        if offset >= 0:
            g[offset:n] += coeff * (1 - h[0:e_len])
            g[0:e_len] += -coeff * h[offset:n]
        else:
            r = -offset
            g[0:e_len] += coeff * (1 - h[r:n])
            g[r:n] += -coeff * h[0:e_len]
    return dx * g

def make_binary_initial(n, rng):
    """Create a random binary (or near-binary) initial guess."""
    h = np.zeros(n)
    n_ones = n // 2
    indices = rng.choice(n, size=n_ones, replace=False)
    h[indices] = 1.0
    if n % 2 == 1:
        h[rng.randint(n)] += 0.5
    return h

def optimize_logsumexp(h0, n, alpha, max_iter=200, max_time=60):
    """Optimize using log-sum-exp penalty with SLSQP."""
    target_sum = n / 2.0
    
    def obj(x):
        return logsumexp_hf(x, alpha)
    
    def obj_grad(x):
        return logsumexp_grad(x, alpha)
    
    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
         'jac': lambda x: np.ones(n)}
    ]
    bounds = [(0, 1)] * n
    
    result = minimize(obj, h0, method='SLSQP', jac=obj_grad,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': 1e-12})
    
    if result.success or result.fun < obj(h0):
        h_opt = project_sum(result.x, n)
        c5 = eval_c5(h_opt)
        return h_opt, c5
    return None

def optimize_violation(h0, n, t_target, active_set=None, max_iter=200):
    """Optimize using violation penalty with SLSQP."""
    target_sum = n / 2.0
    
    def obj(x):
        return violation_hf(x, t_target, active_set)
    
    def obj_grad(x):
        return violation_grad(x, t_target, active_set)
    
    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
         'jac': lambda x: np.ones(n)}
    ]
    bounds = [(0, 1)] * n
    
    result = minimize(obj, h0, method='SLSQP', jac=obj_grad,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': 1e-14})
    
    if result.x is not None:
        h_opt = project_sum(result.x, n)
        c5 = eval_c5(h_opt)
        return h_opt, c5
    return None
```

```python
            r = -offset
            g[0:e_len] += w * (1.0 - h[r:n])
            g[r:n] += -w * h[0:e_len]
    return dx * g


def lse_obj(x, alpha):
    T = get_corr_values(x)
    m = np.max(T)
    if m * alpha > 200:
        return m + (1.0 / alpha) * np.log(np.sum(np.exp(alpha * (T - m))))
    return (1.0 / alpha) * np.log(np.sum(np.exp(alpha * T)))


def lse_weights(x, alpha):
    T = get_corr_values(x)
    aT = alpha * T
    m = np.max(aT)
    e = np.exp(aT - m)
    s = np.sum(e)
    if s < 1e-300:
        return np.ones(len(T)) / len(T)
    return e / s


def lse_grad(x, alpha):
    w = lse_weights(x, alpha)
    return max_weighted_grad(x, w)


def vio_obj(x, t_target, active):
    T = get_corr_values(x)
    Ta = T[active]
    v = np.maximum(Ta - t_target, 0.0)
    return float(np.sum(v * v))


def vio_grad(x, t_target, active):
    T = get_corr_values(x)[active]
    v = np.maximum(T - t_target, 0.0)
    n = len(x)
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(len(active)):
        vi = v[i]
        if vi < 1e-14:
            continue
        coeff = 2.0 * vi
        k = active[i]
        offset = k - (n - 1)
        e_len = n - abs(offset)
        if offset >= 0:
            g[offset:n] += coeff * (1.0 - x[0:e_len])
            g[0:e_len] += -coeff * x[offset:n]
        else:
            r = -offset
            g[0:e_len] += coeff * (1.0 - x[r:n])
            g[r:n] += -coeff * x[0:e_len]
    return dx * g


def sum_eq_fun(x):
    return np.sum(x) - len(x) / 2.0


def sum_eq_jac(x):
    return np.ones(len(x))


def make_binary_initial(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    indices = rng.choice(n, size=n_ones, replace=False)
    h[indices] = 1.0
    if n % 2 == 1:
        h[rng.randint(n)] += 0.5
    return project_sum(h, n)


def do_slqp_lse(h0, n, alpha, max_iter, bounds, constraint):
    res = minimize(lse_obj, h0, method='SLSQP', jac=lse_grad,
                   args=(alpha,),
                   bounds=bounds, constraints=[constraint],
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    if res.x is not None:
        h_cand = project_sum(res.x, n)
        c5_cand = eval_c5(h_cand)
        return h_cand, c5_cand
    return None


def do_slqp_vio(h0, n, t_target, active, max_iter, bounds, constraint):
    res = minimize(vio_obj, h0, method='SLSQP', jac=vio_grad,
                   args=(t_target, active),
                   bounds=bounds, constraints=[constraint],
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    if res.x is not None:
        h_cand = project_sum(res.x, n)
        c5_cand = eval_c5(h_cand)
        return h_cand, c5_cand
    return None


def projected_gradient_step(h, n, lr):
    T = get_corr_values(h)
    max_val = np.max(T)
    mask = T >= max_val - 1e-9
    weights = np.zeros(2 * n - 1)
    weights[mask] = 1.0 / np.sum(mask)
    g = max_weighted_grad(h, weights)
    h_try = project_sum(h - lr * g, n)
    return h_try


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    best_h = None
    best_c5 = float('inf')

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_c5 = eval_c5(best_h)

    if best_h is None:
        best_h = make_binary_initial(45, rng)
        best_c5 = eval_c5(best_h)

    n = len(best_h)
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}

    # Phase 1: Log-sum-exp optimization with increasing alpha
    alphas = [20, 40, 80, 150, 300, 600, 1200]
    for alpha in alphas:
        if time.time() - start_time > budget_s - 180:
            break
        result = do_slqp_lse(best_h, n, alpha, 500, bounds, constraint)
        if result is not None:
            h_cand, c5_cand = result
            if c5_cand < best_c5 - 1e-13:
                best_h = h_cand
                best_c5 = c5_cand

    # Phase 2: Violation minimization - push max down iteratively
    for it in range(50):
        if time.time() - start_time > budget_s - 120:
            break
        current_c5 = eval_c5(best_h)
        T = get_corr_values(best_h)
        active = np.argsort(T)[::-1][:25]
        margin = max(2e-5, (current_c5 - 0.38070) * 0.15)
        t_target = current_c5 - margin
        result = do_slqp_vio(best_h, n, t_target, active, 400, bounds, constraint)
        if result is not None:
            h_cand, c5_cand = result
            if c5_cand < best_c5 - 1e-13:
                best_h = h_cand
                best_c5 = c5_cand
                if best_c5 <= 0.38080:
                    break

    # Phase 3: Projected gradient polish
    if time.time() - start_time < budget_s - 60:
        n = len(best_h)
        bounds = [(0.0, 1.0)] * n
        constraint = {'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}
        for step in range(2000):
            if time.time() - start_time > budget_s - 60:
                break
            current_c5 = eval_c5(best_h)
            improved = False
            for lr in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
                h_try = projected_gradient_step(best_h, n, lr)
                c5_try = eval_c5(h_try)
                if c5_try < current_c5 - 1e-13:
                    best_h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
            if not improved:
                break

    # Phase 4: Try nearby resolutions
    if time.time() - start_time < budget_s - 40:
        n_base = len(best_h)
        for n2 in [n_base - 2, n_base - 1, n_base + 1, n_base + 2, n_base + 5, n_base + 10]:
            if time.time() - start_time > budget_s - 20:
                break
            if n2 < 20:
                continue
            x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
            x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
            h2 = np.interp(x_new, x_old, best_h)
            h2 = project_sum(h2, n2)
            c5_2 = eval_c5(h2)
            if c5_2 < best_c5:
                best_h = h2
                best_c5 = c5_2
                continue
            bounds2 = [(0.0, 1.0)] * n2
            constraint2 = {'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}
            for alpha in [50, 150, 400]:
                if time.time() - start_time > budget_s - 20:
                    break
                result = do_slqp_lse(h2, n2, alpha, 300, bounds2, constraint2)
                if result is not None:
                    h_cand, c5_cand = result
                    if c5_cand < best_c5:
                        best_h = h_cand
                        best_c5 = c5_cand
                        h2 = h_cand
            for it2 in range(10):
                if time.time() - start_time > budget_s - 20:
                    break
                current_c5 = eval_c5(h2)
                T2 = get_corr_values(h2)
                active2 = np.argsort(T2)[::-1][:20]
                t2 = current_c5 - max(1e-5, (current_c5 - 0.38070) * 0.1)
                result = do_slqp_vio(h2, n2, t2, active2, 200, bounds2, constraint2)
                if result is not None:
                    h_cand, c5_cand = result
                    if c5_cand < best_c5:
                        best_h = h_cand
                        best_c5 = c5_cand
                        h2 = h_cand

    # Phase 5: Final random restarts at best resolution
    if time.time() - start_time < budget_s - 10:
        n = len(best_h)
        bounds = [(0.0, 1.0)] * n
        constraint = {'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}
        for restart in range(5):
            if time.time() - start_time > budget_s - 10:
                break
            h_pert = best_h + 0.02 * rng.randn(n)
            h_pert = project_sum(h_pert, n)
            c5_pert = eval_c5(h_pert)
            if c5_pert > best_c5 + 0.01:
                continue
            for alpha in [100, 300]:
                if time.time() - start_time > budget_s - 10:
                    break
                result = do_slqp_lse(h_pert, n, alpha, 300, bounds, constraint)
                if result is not None:
                    h_cand, c5_cand = result
                    if c5_cand < best_c5 - 1e-13:
                        best_h = h_cand
                        best_c5 = c5_cand
                        h_pert = h_cand
            # Violation push on perturbed
            for it3 in range(10):
                if time.time() - start_time > budget_s - 10:
                    break
                current_c5 = eval_c5(h_pert)
                T3 = get_corr_values(h_pert)
                active3 = np.argsort(T3)[::-1][:20]
                t3 = current_c5 - max(1e-5, (current_c5 - 0.38070) * 0.1)
                result = do_slqp_vio(h_pert, n, t3, active3, 200, bounds, constraint)
                if result is not None:
                    h_cand, c5_cand = result
                    if c5_cand < best_c5 - 1e-13:
                        best_h = h_cand
                        best_c5 = c5_cand
                        h_pert = h_cand

    # Final projection and evaluation
    best_h = project_sum(best_h, len(best_h))
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
```
