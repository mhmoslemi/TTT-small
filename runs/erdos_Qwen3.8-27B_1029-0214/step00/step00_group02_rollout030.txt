        H_f_loc, s_f = _local_descent(H_f, M_f, max_iter=3000, t0=t0, time_limit=max(tl_f, 1.0), top_pairs=300, max_units=1)
        try_update(H_f_loc.astype(np.float64) / M_f, s_f)

    best_h = np.clip(best_h, 0.0, 1.0)
    target = len(best_h) / 2.0
    diff = target - float(np.sum(best_h))
    if abs(diff) > 1e-10:
        if diff > 0:
            for i in range(len(best_h)):
                if diff < 1e-12:
                    break
                a = min(diff, 1.0 - best_h[i])
                best_h[i] += a
                diff -= a
        else:
            for i in range(len(best_h)):
                if -diff < 1e-12:
                    break
                r = min(-diff, best_h[i])
                best_h[i] -= r
                diff += r

    final_score = _score(best_h)
    return (best_h, final_score, len(best_h))
