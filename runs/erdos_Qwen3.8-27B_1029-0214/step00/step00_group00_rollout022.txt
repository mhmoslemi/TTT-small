def project_box_sum(h, target):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target}"""
    n = len(h)
    if abs(np.sum(h) - target) < 1e-12:
        return np.clip(h, 0, 1)
    
    # Bisection on lambda
    lo = np.min(h) - 2.0
    hi = np.max(h) + 2.0
    
    def g(lam):
        return np.sum(np.clip(h - lam, 0, 1)) - target
    
    # Ensure bracket

    n = len(h)
