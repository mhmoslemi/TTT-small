```python
def project_to_feasible(raw, n):

    h = np.clip(raw.astype(float), 0.0, 1.0)
```

```python
                    continue
                new_corr = np.correlate(h_new, 1.0 - h_new, mode="full")
                new_tau = float(np.max(new_corr) * dx)
                if new_tau < tau - 1e-13:
                    h = h_new
                    tau = new_tau
                    raw_corr = new_corr
                    improved = True
        if not improved:
            break
    return h, tau
```
