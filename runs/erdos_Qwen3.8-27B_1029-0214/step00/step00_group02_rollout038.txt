def compute_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_to_feasible(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        caps = 1.0 - h
        order = np.argsort(caps)[::-1]
        rem = diff
        for idx in order:
            add = min(rem, caps[idx])
            h[idx] += add
            rem -= add
            if rem < 1e-14:
                break
    else:
        order = np.argsort(h)
        rem = -diff
        for idx in order:
            remv = min(rem, h[idx])
            h[idx] -= remv
            rem -= remv
            if rem < 1e-14:
                break
