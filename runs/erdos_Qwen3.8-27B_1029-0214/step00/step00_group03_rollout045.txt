import numpy as np

def exact_evaluate(h):
    h = np.asarray(h, dtype=np.float64)
    N = len(h)
    dx = 2.0 / N
    q = 1.0 - h
    R = np.correlate(h, q, mode="full")
    c = dx * np.max(R)
    return c if np.isfinite(c) else float('inf')

def project_feasible(h):
    N = len(h)
    h = np.array(h, dtype=np.float64)
    h[~np.isfinite(h)] = 0.5
    np.clip(h, 0.0, 1.0, out=h)
    target = N / 2.0
    diff = target - np.sum(h)
    if diff > 1e-12:
        order = np.argsort(1.0 - h)
        rem = diff
        for idx in order:
            if rem < 1e-14:
                break
            room = 1.0 - h[idx]
            add = min(room, rem)
            h[idx] += add
            rem -= add
    elif diff < -1e-12:
        order = np.argsort(h)
        rem = -diff
        for idx in order:
            if rem < 1e-14:
                break
            h[idx] -= min(h[idx], rem)
            rem -= min(h[idx], rem)
    # Final precision
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            mask = h < 0.999
        else:
            mask = h > 1e-6
        if np.any(mask):
            h[mask] += residual / np.sum(mask)
            np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_correlation(h):
    q = 1.0 - h

def find_optimal_d(R, C, d_lo, d_hi):
    if d_hi - d_lo < 1e-14:
        return 0.0
    for _ in range(64):
        m1 = d_lo + (d_hi - d_lo) / 3.0
        m2 = d_hi - (d_hi - d_lo) / 3.0
        f1 = np.max(R + m1 * C)
        f2 = np.max(R + m2 * C)
        if f1 <= f2:
            d_hi = m2
        else:
            d_lo = m1
    return (d_lo + d_hi) / 2.0

def pair_sweep(h, N, F1, F2, R):
    """One full sweep. Returns (h, R, current_max, improved)."""
    current_max = np.max(R)
    improved = False
    for i in range(N):
        for j in range(i + 1, N):
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_hi - d_lo < 1e-12:
                continue
            C = F1[i] - F1[j] - F2[i] + F2[j]
            d_opt = find_optimal_d(R, C, d_lo, d_hi)
            if abs(d_opt) < 1e-14:
                continue
            new_max = np.max(R + d_opt * C)
            if new_max < current_max - 1e-10:
                h[i] += d_opt
                h[j] -= d_opt
                np.clip(h, 0.0, 1.0, out=h)
                R = R + d_opt * C
                current_max = new_max
                # Update F1 and F2 for changed indices
                _update_F(F1, F2, h, 1.0 - h, N, i)
                _update_F(F1, F2, h, 1.0 - h, N, j)
                improved = True
    return h, R, current_max, improved

    # Update F1[idx]
    F1[idx, :] = 0.0
    start = N - 1 - idx

    start = max(start, 0)
    if start < end:

    start2 = max(start2, 0)
    if start2 < end2:

        F2[idx, start2:end2] = h[N-1 : N-1-length : -1]

def upsample(h, N_new):
    N_old = len(h)
    if N_new == N_old:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_feasible(h_new)

def random_kick(h, N, rng, kick_size=0.05):
    """Apply a small random perturbation."""
    h2 = h.copy()
    for _ in range(10):
        i = rng.integers(N)
        j = rng.integers(N)
        if i == j:
            continue
        d = rng.uniform(-kick_size, kick_size)
        d_lo = max(-h2[i], h2[j] - 1.0)
        d_hi = min(1.0 - h2[i], h2[j])
        d = np.clip(d, d_lo, d_hi)
        if abs(d) > 1e-10:
            h2[i] += d
            h2[j] -= d
    np.clip(h2, 0.0, 1.0, out=h2)
    return project_feasible(h2)
