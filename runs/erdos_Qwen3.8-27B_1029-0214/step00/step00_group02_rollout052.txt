def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * dx

def _project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
