def targeted_move(h, n):
    """Try to reduce the max correlation by adjusting coordinates that contribute to it."""
    # Compute correlations
    c = np.correlate(h, 1-h, mode="full")
    # Find the lag with max correlation
    k_star = np.argmax(c)
    # The offset in the "full" correlation array
    # full correlation has indices 0 to 2n-2, representing lags -(n-1) to (n-1)
    lag = k_star - (n - 1)
    
    # Which pairs (i, i+lag) contribute to c[k_star]?
    # h[i] * (1-h[i+lag]) for valid i
    if lag >= 0:
        valid_i = range(0, n - lag)
    else:
        valid_i = range(0, n + lag)
    
    # Find the i's that contribute most to c[k_star]
    # h[i] * (1-h[i+lag]) is the contribution
    contributions = np.array([h[i] * (1 - h[i + lag]) for i in valid_i])
    
    # Pick the i with largest contribution and try to reduce it
    # Option 1: decrease h[i] (reduces h[i]*(1-h[i+lag]))
    # Option 2: increase h[i+lag] (reduces (1-h[i+lag]))
    # But these affect other lags too
    
    i = np.argmax(contributions) + (0 if lag >= 0 else 0)  # need to handle indexing
    # Actually valid_i is already the right range
    best_i = list(valid_i)[np.argmax(contributions)]
    
    # Try to move mass from h[best_i] to h[best_i + lag] (if lag > 0)
    # or from h[best_i - lag] to h[best_i] (if lag < 0)
    
    # This is a 2-coordinate move
    if lag > 0:
        j = best_i + lag
        delta = min(h[best_i], 1 - h[j]) * np.random.uniform(0.1, 1.0)

        j = best_i
        best_i = i2
        delta = min(h[best_i], 1 - h[j]) * np.random.uniform(0.1, 1.0)
    
    return best_i, j, delta

The term h'[i]*(1-h'[i+k]) = (h[i]+d)*(1-h[i+k]) if i+k ≠ j, or (h[i]+d)*(1-h[j]+d) if i+k=j
Similarly for the other terms.
