    """Evaluate the exact C5 objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    """Project y onto the feasible set {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(np.array(y, dtype=float), 0.0, 1.0)
    # Iterate to fix the sum
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            h[:] = target / n
            break
        step = diff / mask.sum()
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_all_lags(h):
    """Compute all 2n-1 lag values of correlate(h, 1-h, full) * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    """Compute gradients of all 2n-1 lag values with respect to h.
    Returns array of shape (2n-1, n)."""
    n = len(h)

    grads = np.zeros((2 * n - 1, n))
    
    # r ranges from -(n-1) to (n-1), mapped to index k = r + (n-1)
    for r in range(-(n-1), n):
        k = r + (n - 1)
        if r >= 0:

import numpy as np
from scipy.optimize import minimize
import time


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(y, n):
    target = n / 2.0
    h = np.clip(np.asarray(y, dtype=float), 0.0, 1.0)
    for _ in range(2000):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h[:] = target / n
            return h
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def ineq_fun(x):
    n = len(x) - 1
    h = x[:n]
    t = x[n]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return t - corr * dx


def ineq_jac(x):
    n = len(x) - 1
    h = x[:n]
    dx = 2.0 / n
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n + 1))
    J[:, n] = 1.0
    for r in range(-(n - 1), n):
        k = r + (n - 1)
        if r >= 0:
            e = n - r
            J[k, 0:e] -= dx * (1.0 - h[r:r + e])
            J[k, r:n] += dx * h[0:e]
        else:
            s = -r
            e = n - s
            J[k, s:n] -= dx * (1.0 - h[0:e])
            J[k, 0:e] += dx * h[s:s + e]
    return J


def eq_fun(x):
    n = len(x) - 1
    return x[:n].sum() - n / 2.0


def eq_jac(x):
    n = len(x) - 1
    J = np.zeros(n + 1)
    J[:n] = 1.0
    return J


def obj_fun(x):
    return x[-1]


def obj_grad(x):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g


def solve_epigraph(h0, n, maxiter=300, ftol=1e-12):
    h_start = project_feasible(h0, n)
    c5_start = eval_c5(h_start)
    t_start = c5_start + 1e-6
    x0 = np.concatenate([h_start, [t_start]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
    ]
    best_h = h_start
    best_c5 = c5_start
    try:
        result = minimize(
            obj_fun, x0, method='SLSQP',
            jac=obj_grad,
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'ftol': ftol, 'disp': False}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            h_opt = project_feasible(result.x[:n], n)
            c5_opt = eval_c5(h_opt)
            if c5_opt < best_c5 - 1e-12:
                best_h = h_opt
                best_c5 = c5_opt
    except Exception:
        pass
    return best_h, best_c5
