from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(y):
    n = len(y)
    target = n * 0.5
    lo = float(y.min()) - 2.0
    hi = float(y.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(y - mid, 0.0, 1.0))
        if val > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.count_nonzero(mask) > 0:
            h[mask] += residual / np.count_nonzero(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_lags(h, tau):
    """Compute c_vals and softmax weights p for all lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    c_vals = corr * dx
    max_c = np.max(c_vals)
    shifted = tau * (c_vals - max_c)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / np.sum(exp_shifted)
    return c_vals, p

    c_vals, p = compute_lags(h, tau)
    n = len(h)
    dx = 2.0 / n
    max_c = np.max(c_vals)
    log_sum = max_c + np.log(np.sum(np.exp(tau * (c_vals - max_c))))
    obj = log_sum / tau

        lag = k_idx - (n - 1)

        if pk < 1e-16:
            continue

            grad[lag:] += pk * dx * (1 - h[:n - lag])
            # Second term: -h[m+lag] for m in [0, n-1-lag]

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(y):
    n = len(y)
    target = n * 0.5
    lo = float(y.min()) - 2.0
    hi = float(y.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        val = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if val > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - float(h.sum())
    if abs(residual) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            h[mask] += residual / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def _get_lags_and_weights(h, tau):
    n = len(h)
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_vals = corr * dx
    max_c = float(np.max(c_vals))
    shifted = tau * (c_vals - max_c)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / float(np.sum(exp_shifted))
    return c_vals, p, max_c

    log_sum = max_c + float(np.log(float(np.sum(np.exp(tau * (c_vals - max_c))))))
    return log_sum / tau

        if lag >= 0:

            grad[:n - lag] -= pk * dx * h[lag:]
        else:
            ll = -lag
            grad[:n - ll] += pk * dx * (1.0 - h[ll:])

from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_OPT = {'n': 0, 'tau': 1.0}


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_feasible(y):
    n = len(y)
    target = n * 0.5
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        val = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if val > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - float(np.sum(h))
    if abs(residual) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            h[mask] += residual / float(cnt)
            h = np.clip(h, 0.0, 1.0)
    return h


def _lws(h, tau):
    n = len(h)
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    mc = float(np.max(c))
    sh = tau * (c - mc)
    e = np.exp(sh)
    s = float(np.sum(e))
    p = e / s
    return c, p, mc


def _sobj(x):
    c, p, mc = _lws(x, _OPT['tau'])
    tau = _OPT['tau']
    mc2 = float(np.max(c))
    return (mc2 + float(np.log(float(np.sum(np.exp(tau * (c - mc2))))))) / tau


def _sjac(x):
    n = len(x)
    tau = _OPT['tau']
    dx = 2.0 / float(n)
    c, p, mc = _lws(x, tau)
    g = np.zeros(n)
    for ki in range(2 * n - 1):

        pk = p[ki]
        if pk < 1e-15:

            ll = -lag
            g[:n - ll] += pk * dx * (1.0 - x[ll:])
            g[ll:] -= pk * dx * x[:n - ll]
    return g


def _efun(x):
    return float(np.sum(x)) - _OPT['n'] * 0.5


def _ejac(x):

def resample_h(h0, n_t):
    n0 = len(h0)
    if n0 == n_t:
        return h0.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n_t) + 0.5) * (2.0 / float(n_t))
    f = interp1d(x0, h0, kind='linear', fill_value='extrapolate', bounds_error=False)
    h1 = f(x1)
    return project_feasible(h1)


def make_starts(n, parent_h, best_h, rng):

    starts.append(resample_h(best_h, n))

    _OPT['n'] = n
    _OPT['tau'] = tau
    try:
        res = minimize(
            fun=_sobj,
            x0=h0.copy(),
            jac=_sjac,

    return None


def _nonsmooth_polish(h0, n, rng, n_iter):
    best = h0.copy()
    best_c5 = evaluate_c5(best)

        nh[i2] += d
        c5 = evaluate_c5(nh)
        if c5 < best_c5 - 1e-12:

            best_c5 = c5
