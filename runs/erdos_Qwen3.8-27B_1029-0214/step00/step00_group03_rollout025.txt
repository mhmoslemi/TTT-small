def evaluate_score(h):
    N = len(h)
    dx = 2.0 / N
    q = 1.0 - h
    c = np.correlate(h, q, mode='full')
    return np.max(c) * dx, c

def compute_DQ(h, q, i, j):
    N = len(h)
    M = 2 * N - 1
    l = np.arange(M) - (N - 1)
    
    D = np.zeros(M)
    Q = np.zeros(M)
    
    # D[m] = -h[i+l] + h[j+l] + q[i-l] - q[j-l]
    # i+l in [0, N-1] means l in [-i, N-1-i]
    mask = (l >= -i) & (l <= N - 1 - i)
    D[mask] -= h[i + l[mask]]
    
    mask = (l >= -j) & (l <= N - 1 - j)
    D[mask] += h[j + l[mask]]
    
    # i-l in [0, N-1] means l in [i-(N-1), i]
    mask = (l >= i - (N - 1)) & (l <= i)
    D[mask] += q[i - l[mask]]
    
    mask = (l >= j - (N - 1)) & (l <= j)
    D[mask] -= q[j - l[mask]]
    
    # Q: non-zero at l=0 (value -2), l=j-i (value +1), l=i-j (value +1)
    Q[N - 1] -= 2
    r = j - i
    idx = N - 1 + r
    if 0 <= idx < M:
        Q[idx] += 1
    idx = N - 1 - r
    if 0 <= idx < M:
        Q[idx] += 1
    
    return D, Q

def evaluate_score(h):
    N = len(h)
    dx = 2.0 / N
    q = 1.0 - h
    c = np.correlate(h, q, mode='full')
    return float(np.max(c) * dx)

def compute_DQ(h, q, i, j):
    N = len(h)
    M = 2 * N - 1
    l = np.arange(M) - (N - 1)
    
    D = np.zeros(M)
    Q = np.zeros(M)
    
    mask = (l >= -i) & (l <= N - 1 - i)
    D[mask] -= h[i + l[mask]]
    
    mask = (l >= -j) & (l <= N - 1 - j)
    D[mask] += h[j + l[mask]]
    
    mask = (l >= i - (N - 1)) & (l <= i)
    D[mask] += q[i - l[mask]]
