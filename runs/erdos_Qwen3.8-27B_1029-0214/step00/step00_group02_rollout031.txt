def project_feasible(y, n):
    target = n / 2.0
    # Bisection on lambda
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        val = np.clip(y - mid, 0, 1).sum()
        if val > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo+hi)/2, 0, 1)
    # Fix any remaining error (should be tiny)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Distribute to non-boundary entries
        free = (h > 1e-10) & (h < 1-1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0, 1)
    return h

def smooth_objective(h, tau, n):
    lags = compute_lags(h, n)  # array of length 2n-1
    # Log-sum-exp
    max_lag = lags.max()
    if np.isinf(max_lag):
        return max_lag
    logsum = max_lag + np.log(np.sum(np.exp(tau * (lags - max_lag))))
    return logsum / tau

def smooth_gradient(h, tau, n):
    lags = compute_lags(h, n)
    max_lag = lags.max()
    exp_arg = tau * (lags - max_lag)
    p = np.exp(exp_arg)
    p /= p.sum()
    # Compute gradient
    dx = 2.0 / n
    # Build gradient
    grad = np.zeros(n)
    for k_idx in range(2*n - 1):
        if p[k_idx] < 1e-15:
            continue
        actual_k = k_idx - (n - 1)
        # ∂L_k/∂h[j] for all j
        # Contribution from h[j] as h[i+k]: (1-h[j-k]) if 0<=j-k<n
        # Contribution from h[j] as h[i]: -h[j+k] if 0<=j+k<n
        for j in range(n):
            idk = j - actual_k
            idh = j + actual_k
            term = 0.0
            if 0 <= idk < n:
                term += (1 - h[idk])
            if 0 <= idh < n:
                term -= h[idh]
            grad[j] += dx * p[k_idx] * term
    return grad
