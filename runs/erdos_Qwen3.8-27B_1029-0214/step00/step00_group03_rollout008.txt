def gradient_step(h, N, lr=0.01):
    """One step of projected subgradient descent on C5."""
    G = compute_G(h)  # length 2N-1
    G_max = np.max(G)
    # Subgradient: the gradient of G_k with respect to h, for k achieving the max
    # G_k = sum_p h[p]*(1-h[p+s]) where s is the lag
    # dG_k/dh[p] = (1-h[p+s]) - h[p-s] (with boundary convention h[x]=0 if x out of range)
    
    # Find which lags achieve the max
    max_lags = np.where(G >= G_max - 1e-10)[0]
    
    # Compute subgradient
    grad = np.zeros(N)
    for m in max_lags:
        s = m - (N - 1)  # lag
        for p in range(N):
            p_plus = p + s
            p_minus = p - s
            g = 0.0
            if 0 <= p_plus < N:
                g += (1.0 - h[p_plus])
            if 0 <= p_minus < N:
                g -= h[p_minus]
            grad[p] += g / len(max_lags)
    
    # Projected gradient step
    h_new = h - lr * grad
    h_new = project_h(h_new, N)
    return h_new

def compute_G(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, N):
    h = np.array(h, dtype=float)

    h = np.clip(h, 0.0, 1.0)

    if abs(s - target) < 1e-12:

    diff = target - s

        mask = h < 1.0
        cnt = np.sum(mask)
    else:
        mask = h > 0.0
        cnt = np.sum(mask)
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)
