    n = len(h)
    dx = 2.0 / n
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    # For each correlation index i and each h index m:
    # J[i, m] = dx * (1-h[m-lags[i]]) * (lags[i] <= m) * (m-lags[i] < n)
    #         - dx * h[m+lags[i]] * (m+lags[i] >= 0) * (m+lags[i] < n)
    
    # Create index arrays
    i_idx = np.arange(2*n-1)[:, None]  # (2n-1, 1)
    m_idx = np.arange(n)[None, :]  # (1, n)
    
    # First term: (1-h[m-lag]) where 0 <= m-lag < n
    m_minus_lag = m_idx - lags[:, None]  # m - lag, shape (2n-1, n)
    valid1 = (m_minus_lag >= 0) & (m_minus_lag < n)
    # h[m-lag] where valid
    h_m_minus_lag = np.where(valid1, h[m_minus_lag.clip(0, n-1)], 0)
    term1 = (1 - h_m_minus_lag) * valid1
    
    # Second term: h[m+lag] where 0 <= m+lag < n
    m_plus_lag = m_idx + lags[:, None]  # m + lag, shape (2n-1, n)

    h_m_plus_lag = np.where(valid2, h[m_plus_lag.clip(0, n-1)], 0)
    term2 = h_m_plus_lag * valid2
    
    J = dx * (term1 - term2)
