    """Project z onto {x : 0 <= x_i <= 1, sum(x) = target}"""
    n = len(z)
    target = float(target)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    # Bisection on mu
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mu = (lo + hi) / 2.0

        if val > target:
            lo = mu
        else:
            hi = mu
    mu = (lo + hi) / 2.0

    # Fix rounding

        else:
            # More careful adjustment
            result = np.clip(result, 0.0, 1.0)
    return result

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

    n = len(h)
    if tau >= 0:
        L = n - tau
        if L <= 0:
            return 0.0
        return dx * np.dot(h[:L], 1.0 - h[tau:tau+L])
    else:
        r = -tau
        L = n - r
        if L <= 0:
            return 0.0
        return dx * np.dot(h[r:r+L], 1.0 - h[:L])

def corr_row_gradient(h, tau, dx):
    """Compute gradient of the scaled correlation for lag tau."""
    n = len(h)
    grad = np.zeros(n)
    if tau >= 0:
        L = n - tau
        if L <= 0:
            return grad
        # For i in 0..L-1: grad[i] += dx*(1-h[i+tau]), grad[i+tau] -= dx*h[i]

        grad[tau:tau+L] -= dx * h[:L]
    else:
        r = -tau
        L = n - r
        if L <= 0:
            return grad
        # For i in 0..L-1: grad[i+r] += dx*(1-h[i]), grad[i] -= dx*h[i+r]
        grad[r:r+L] += dx * (1.0 - h[:L])
        grad[:L] -= dx * h[r:r+L]
    return grad

        return x[self.n]  # t

        g[n] = 1.0
        return g

def run_epigraph_solve(h0, n, dx, target, rows, t0, maxiter, ftol, deadline):
    """Run SLSQP on the epigraph problem."""
    _solver_deadline[0] = deadline
    x0 = np.zeros(n + 1)
    x0[:n] = h0
    x0[n] = t0
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    prob = EpigraphProblem(h0, n, dx, target, rows, t0)
    
    constraints = [
        {'type': 'eq', 'fun': prob.con_eq, 'jac': prob.jac_eq},
        {'type': 'ineq', 'fun': prob.con_ineq, 'jac': prob.jac_ineq},
    ]
    
    try:
        res = minimize(prob.fun, x0, method='SLSQP', jac=prob.grad,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return res.x[:n], res.x[n]
    except (TimeoutError, Exception):
        pass
    return None, None

    solver = EpigraphSolver(h0, n, dx, target, rows, t0)

        {'type': 'eq', 'fun': solver.con_eq, 'jac': solver.jac_eq},
        {'type': 'ineq', 'fun': solver.con_ineq, 'jac': solvent.jac_ineq},
    ]
    
    try:
        res = minimize(solver.fun, x0, method='SLSQP', jac=solver.grad,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return res.x[:n]
    except (TimeoutError, Exception):
        pass
    return None

def transfer_resolution(h_src, n_src, n_tgt):
    """Transfer a solution from one resolution to another."""
    if n_src == n_tgt:
        return h_src.copy()
    # Linear interpolation
    x_src = (np.arange(n_src) + 0.5) / n_src * 2.0
    x_tgt = (np.arange(n_tgt) + 0.5) / n_tgt * 2.0
    h_interp = np.interp(x_tgt, x_src, h_src)
    target_src = np.sum(h_src) * (2.0 / n_src)
    target_tgt = target_src * n_tgt / 2.0
    return project_box_sum(h_interp, target_tgt)

def perturb(h, n, rng, scale):
    """Add Gaussian noise and project."""
    noise = rng.standard_normal(n) * scale
    z = h + noise
    target = np.sum(h)
    return project_box_sum(z, target)
