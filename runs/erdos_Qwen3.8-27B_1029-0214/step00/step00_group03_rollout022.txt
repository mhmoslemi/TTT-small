    while np.sum(h_bin) > target:
        positions = np.where(h_bin == 1)[0]
        min_pos = positions[np.argmin(h_interp[positions])]
        h_bin[min_pos] = 0
    while np.sum(h_bin) < target:
        positions = np.where(h_bin == 0)[0]
        max_pos = positions[np.argmax(h_interp[positions])]
        h_bin[max_pos] = 1
    return h_bin

def random_binary(n, rng):
    h = np.zeros(n, dtype=np.int64)
    h[rng.choice(n, n // 2, replace=False)] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = np.full(45, 0.5)
    
    best_h = h_init.copy()
    n_pts = len(h_init)
    target_sum = n_pts / 2.0
    dx = 2.0 / n_pts
    best_c5 = compute_c5(best_h, dx)
    
    # Phase 1
    t1 = start_time + budget_s * 0.15
    h1, c5_1 = subgrad_nesterov(h_init.copy(), target_sum, dx, 100000, 0.1, t1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
    
    # Phase 2
    t2 = start_time + budget_s * 0.25
    for it in range(10000):
        if time.time() > t2:
            break
        h_pert = best_h + rng.randn(n_pts) * 0.003
        h_pert = project_onto_constraint(h_pert, target_sum)
        c5_pert = compute_c5(h_pert, dx)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert
    
    # Phase 3
    t3 = start_time + budget_s * 0.95
    n_candidates = [46, 44, 48, 50, 52]
    for n_try in n_candidates:
        if time.time() > t3:
            break
        dx_t = 2.0 / n_try
        h_bin = make_binary(best_h, n_pts, n_try)
        h_b, c5_b = binary_local_search(h_bin, dx_t, 500, t3)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.astype(float)
            n_pts = n_try
        if time.time() < t3 - 60:
            h_rand = random_binary(n_try, rng)
            h_r, c5_r = binary_local_search(h_rand, dx_t, 300, t3)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.astype(float)
                n_pts = n_try
    
    return best_h, best_c
