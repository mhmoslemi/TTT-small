            z[:] = target / n
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(z - lam, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-13:
        if residual > 0:
            order = np.argsort(-result)
            rem = residual
            for i in order:
                add = min(rem, 1.0 - result[i])
                if add <= 0:
                    break
                result[i] += add
                rem -= add
                if rem < 1e-14:
                    break
        else:
            order = np.argsort(result)
            rem = -residual
            for i in order:
                sub = min(rem, result[i])
                if sub <= 0:
                    break
                result[i] -= sub
                rem -= sub
                if rem < 1e-14:
                    break
    return result

def exact_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(raw))

def get_active_lags(h, tol=0.005):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    c5 = dx * np.max(raw)
    threshold = (c5 - tol) / dx
    active = np.where(raw >= threshold)[0]
    return active, c5, raw

def score_pairs_for_lags(h, active_lags, n, max_pairs=30):
    pairs = []
    scores = {}
    for lag_idx in active_lags:
        tau = lag_idx - (n - 1)
        if tau >= 0:
            L = n - tau
            for i in range(L):
                p = i + tau
                q = i
                key = (min(p, q), max(p, q))
                val = h[p] * (1.0 - h[q])
                if key not in scores:
                    scores[key] = 0.0
                scores[key] += val
        else:
            r = -tau
            L = n - r
            for i in range(L):
                p = i
                q = i + r
                key = (min(p, q), max(p, q))
                val = h[p] * (1.0 - h[q])
                if key not in scores:
                    scores[key] = 0.0
                scores[key] += val
    for (p, q), s in scores.items():
        t_lo = max(-h[p], h[q] - 1.0)
        t_hi = min(1.0 - h[p], h[q])
        if t_hi - t_lo > 1e-8:
            reach = t_hi - t_lo
            pairs.append((s * reach, p, q))
    pairs.sort(reverse=True)
    seen = set()
    result = []
    for s, p, q in pairs:
        key = (min(p, q), max(p, q))
        if key not in seen:
            seen.add(key)
            result.append((p, q))
        if len(result) >= max_pairs:
            break
    return result

def pair_line_search(h, p, q, n, max_iter=60):
    dx = 2.0 / n
    t_lo = max(-h[p], h[q] - 1.0)
    t_hi = min(1.0 - h[p], h[q])
    if t_hi - t_lo < 1e-12:
        return h.copy(), exact_c5(h)
    h_base = h.copy()
    def obj(t):
        h_t = h_base.copy()
        h_t[p] = h[p] + t
        h_t[q] = h[q] - t
        h_t[p] = np.clip(h_t[p], 0.0, 1.0)
        h_t[q] = np.clip(h_t[q], 0.0, 1.0)
        raw = np.correlate(h_t, 1.0 - h_t, mode="full")
        return float(dx * np.max(raw))
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    a = t_lo
    b = t_hi
    c_pt = b - invphi * (b - a)
    d_pt = a + invphi * (b - a)
    fc = obj(c_pt)
    fd = obj(d_pt)
    for _ in range(max_iter):
        if b - a < 1e-11:
            break
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - invphi * (b - a)
            fc = obj(c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + invphi * (b - a)
            fd = obj(d_pt)
    t_best = (a + b) / 2.0
    h_best = h_base.copy()
    h_best[p] = np.clip(h[p] + t_best, 0.0, 1.0)
    h_best[q] = np.clip(h[q] - t_best, 0.0, 1.0)
    c_best = exact_c5(h_best)
    return h_best, c_best

def local_polish_lagged(h, n, c_current, deadline, max_rounds=20, max_pairs=25, tol_lag=0.01):
    best_h = h.copy()
    best_c = c_current
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        active_lags, _, _ = get_active_lags(best_h, tol=tol_lag)
        pairs = score_pairs_for_lags(best_h, active_lags, n, max_pairs=max_pairs)
        if not pairs:
            order = np.argsort(best_h)
            for i in range(min(n // 3, max_pairs // 2)):
                p = order[i]
                q = order[n - 1 - i]
                if p != q:
                    pairs.append((p, q))
        if not pairs:
            break
        improved = False
        for p, q in pairs:
            if time.monotonic() > deadline:
                break
            h_new, c_new = pair_line_search(best_h, p, q, n, max_iter=55)
            if c_new < best_c - 1e-13:
                best_h = h_new
                best_c = c_new
                improved = True
                break
        if not improved:
            tol_lag = max(tol_lag * 0.7, 0.001)
            if tol_lag <= 0.001:
                break
    return best_h, best_c

def local_polish_general(h, n, c_current, deadline, max_rounds=15, max_pairs=20):
    best_h = h.copy()
    best_c = c_current
    rng_local = np.random.default_rng(123)
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        order = np.argsort(best_h)
        candidate_pairs = []
        for i in range(min(n // 3, max_pairs // 2)):
            p = order[i]
            q = order[n - 1 - i]
            if p != q:
                candidate_pairs.append((p, q))
        for _ in range(max_pairs // 3):
            i = rng_local.integers(0, n)
            j = rng_local.integers(0, n)
            if i != j:
                candidate_pairs.append((int(i), int(j)))
        seen = set()
        unique_cp = []
        for p, q in candidate_pairs:
            key = (min(p, q), max(p, q))
            if key not in seen:
                seen.add(key)
                unique_cp.append((p, q))
        for p, q in unique_cp[:max_pairs]:
            if time.monotonic() > deadline:
                break
            h_new, c_new = pair_line_search(best_h, p, q, n, max_iter=50)
            if c_new < best_c - 1e-13:
                best_h = h_new
                best_c = c_new
                improved = True
                break
        if not improved:
            break
    return best_h, best_c

def de_stage(h_init, n, c_init, deadline, pop_size=14, max_gens=25, seed_val=42):
    rng = np.random.default_rng(seed_val)
    pop = [h_init.copy()]
    c_vals = [c_init]
    for i in range(1, pop_size):
        noise_scale = 0.12 * (1.0 + 0.5 * rng.random())
        trial = h_init + noise_scale * rng.standard_normal(n)
        trial = project_feasible(trial, n)
        c_trial = exact_c5(trial)
        pop.append(trial)
        c_vals.append(c_trial)
    order = np.argsort(c_vals)
    pop = [pop[i] for i in order]
    c_vals = [c_vals[i] for i in order]
    best_h = pop[0].copy()
    best_c = c_vals[0]
    F = 0.6
    CR = 0.7
    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break
        F = 0.6 * (0.95 ** gen) + 0.2
        for i in range(pop_size):
            if time.monotonic() > deadline:
                break
            idx = rng.choice(pop_size, size=3, replace=False)
            a, b, c = int(idx[0]), int(idx[1]), int(idx[2])
            if a == i or b == i or c == i:
                continue
            mutant = pop[a] + F * (pop[b] - pop[c])
            trial = mutant.copy()
            cross_mask = rng.random(n) < CR
            j0 = int(rng.integers(0, n))
            cross_mask[j0] = True
            trial[cross_mask] = mutant[cross_mask]
            trial = project_feasible(trial, n)
            c_trial = exact_c5(trial)
            if c_trial < c_vals[i]:
                pop[i] = trial
                c_vals[i] = c_trial
                if c_trial < best_c:
                    best_c = c_trial
                    best_h = trial.copy()
        order = np.argsort(c_vals)
        pop = [pop[i] for i in order]
        c_vals = [c_vals[i] for i in order]
        if gen % 4 == 3:
            if time.monotonic() < deadline:
                ph, pc = local_polish_lagged(pop[0], n, c_vals[0], min(time.monotonic() + 3, deadline), max_rounds=3, max_pairs=8, tol_lag=0.008)
                if pc < best_c:
                    best_c = pc
                    best_h = ph.copy()
                    pop[0] = ph
                    c_vals[0] = pc
    return best_h, best_c

def refine_resolution(h_old, n_new):
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    old_centers = (np.arange(n_old) + 0.5) * dx_old
    new_centers = (np.arange(n_new) + 0.5) * dx_new
    h_new = np.interp(new_centers, old_centers, h_old, left=h_old[0], right=h_old[-1])
    h_new = project_feasible(h_new, n_new)
    c_new = exact_c5(h_new)
    return h_new, c_new

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.monotonic()
    deadline_total = start_time + budget_s * 0.93
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except (NameError, UnboundLocalError):
        parent = None
    if parent is not None and len(parent) >= 4:
        n = len(parent)
        parent = project_feasible(parent, n)
    else:
        n = 63
        parent = np.full(n, 0.5)
    best_h = parent.copy()
    best_c = exact_c5(best_h)
    if budget_s < 5:
        return best_h, best_c, n
    dl1 = start_time + budget_s * 0.30
    best_h, best_c = local_polish_lagged(best_h, n, best_c, dl1, max_rounds=20, max_pairs=30, tol_lag=0.01)
    dl2 = start_time + budget_s * 0.42
    if time.monotonic() < dl2:
        h_rev = parent[::-1].copy()
        c_rev = exact_c5(h_rev)
        h_rev, c_rev = local_polish_lagged(h_rev, n, c_rev, min(time.monotonic() + budget_s * 0.06, dl2), max_rounds=10, max_pairs=20, tol_lag=0.01)
        if c_rev < best_c:
            best_h = h_rev
            best_c = c_rev
    dl3 = start_time + budget_s * 0.65
    if time.monotonic() < dl3:
        pop_sz = max(10, min(16, int(80 / max(1, n // 10))))
        best_h, best_c = de_stage(best_h, n, best_c, dl3, pop_size=pop_sz, max_gens=40, seed_val=seed)
    dl4 = start_time + budget_s * 0.78
    if time.monotonic() < dl4:
        best_h, best_c = local_polish_lagged(best_h, n, best_c, dl4, max_rounds=20, max_pairs=25, tol_lag=0.005)
        best_h, best_c = local_polish_general(best_h, n, best_c, dl4, max_rounds=10, max_pairs=20)
    dl5 = start_time + budget_s * 0.88
    if time.monotonic() < dl5 and n <= 100:
        n2 = n * 2
        h2, c2 = refine_resolution(best_h, n2)
        if c2 < best_c:
            h2, c2 = local_polish_lagged(h2, n2, c2, dl5, max_rounds=12, max_pairs=20, tol_lag=0.008)
            if c2 < best_c:
                best_h = h2
                best_c = c2
                n = n2
    if time.monotonic() < start_time + budget_s * 0.93:
        best_h, best_c = local_polish_lagged(best_h, n, best_c, min(time.monotonic() + 5, start_time + budget_s * 0.93), max_rounds=5, max_pairs=15, tol_lag=0.004)
    best_h = project_feasible(best_h, len(best_h))
    final_c = exact_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c, n_final
