    while flo > 0 and lo > -1000:
        lo *= 2.0
        flo = np.sum(np.clip(v + lo, 0.0, 1.0)) - target_sum
    while fhi < 0 and hi < 1000:
        hi *= 2.0
        fhi = np.sum(np.clip(v + hi, 0.0, 1.0)) - target_sum
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        fmid = np.sum(np.clip(v + mid, 0.0, 1.0)) - target_sum
        if fmid > 0:
            hi = mid
        else:
            lo = mid
        if hi - lo < 1e-14:
            break
    return np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h):
    n = len(h)
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - target) > 1e-8:
        h = waterfill(h, target)
    h = np.clip(h, 0.0, 1.0)
    return h

def make_h_from_y(y, target_sum):
    lam = safe_solve_lambda(y, target_sum)
    h = expit(y + lam)
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - target_sum) > 1e-6:
        h = waterfill(h, target_sum)
    return h

def h_to_y(h):
    eps = 1e-7
    h_clip = np.clip(h, eps, 1.0 - eps)
    return np.log(h_clip / (1.0 - h_clip))

def upsample_block(h, n_target):
    n_src = len(h)
    if n_target == n_src:
        return h.copy()
    factor = n_target // n_src
    if factor * n_src != n_target:
        # Fallback to interpolation
        x_src = np.linspace(0.0, 1.0, n_src)
        x_tgt = np.linspace(0.0, 1.0, n_target)
        h_up = np.interp(x_tgt, x_src, h)
    else:
        h_up = np.repeat(h, factor)
    target = n_target / 2.0
    if abs(np.sum(h_up) - target) > 1e-10:
        h_up = waterfill(h_up, target)
    h_up = np.clip(h_up, 0.0, 1.0)
    return h_up

def upsample_interp(h, n_target):
    n_src = len(h)
    if n_target == n_src:
        return h.copy()
    x_src = np.linspace(0.0, 1.0, n_src)
    x_tgt = np.linspace(0.0, 1.0, n_target)
    h_up = np.interp(x_tgt, x_src, h)
    target = n_target / 2.0
    h_up = waterfill(h_up, target)
    return np.clip(h_up, 0.0, 1.0)

def compute_smooth_obj_grad(y, beta, target_sum):
    n = len(y)
    dx = 2.0 / n
    lam = safe_solve_lambda(y, target_sum)
    h = np.clip(expit(y + lam), 0.0, 1.0)
    
    n_lags = 2 * n - 1
    c = np.empty(n_lags)
    for s in range(-n + 1, n):
        L = n - abs(s)
        if s >= 0:
            c[s + n - 1] = dx * np.dot(h[s:s+L], 1.0 - h[:L])
        else:
            r = -s
            c[s + n - 1] = dx * np.dot(h[:L], 1.0 - h[r:r+L])
    
    c_max = np.max(c)
    shifted = beta * (c - c_max)
    shifted_clipped = np.maximum(shifted, -60.0)
    exp_shifted = np.exp(shifted_clipped)
    log_sum = np.log(np.sum(exp_shifted))
    F = c_max + log_sum / beta
    
    w = exp_shifted / np.sum(exp_shifted)
    
    G = np.zeros(n)
    for m in range(n_lags):
        s = m - (n - 1)
        wm = w[m]
        if wm < 1e-16:
            continue
        L = n - abs(s)
        if s >= 0:
            G[s:s+L] += wm * dx * (1.0 - h[:L])
            G[0:L] -= wm * dx * h[s:s+L]
        else:
            r = -s
            G[0:L] += wm * dx * (1.0 - h[r:r+L])
            G[r:r+L] -= wm * dx * h[0:L]
    
    W = h * (1.0 - h)
    S = np.sum(W)
    if S > 1e-15:
        grad_y = W * (G - np.dot(G, W) / S)
    else:
        grad_y = np.zeros(n)
    
    return F, grad_y

def smooth_fun(y, beta, target_sum):
    F, g = compute_smooth_obj_grad(y, beta, target_sum)
    if not np.isfinite(F):
        return 1e10, np.zeros_like(y)
    if not np.all(np.isfinite(g)):
        g = np.where(np.isfinite(g), g, 0.0)
    return F, g

def pair_local_search(h, max_sweeps=200):
    n = len(h)
    current_c = exact_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n - 1):
            hi_saved = h[i]
            hj_saved = h[i + 1]
            for delta in (0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.03, -0.03):
                new_hi = hi_saved + delta
                new_hj = hj_saved - delta
                if 0.0 <= new_hi <= 1.0 and 0.0 <= new_hj <= 1.0:
                    h[i] = new_hi
                    h[i + 1] = new_hj
                    c_new = exact_c5(h)
                    if c_new < current_c - 1e-12:
                        current_c = c_new
                        improved = True
                        hi_saved = new_hi
                        hj_saved = new_hj
                        break
            h[i] = hi_saved
            h[i + 1] = hj_saved
        if not improved:
            break
    return h, current_c

def try_candidate(h, best_h, best_c):
    h_rep = repair_h(h)
    c = exact_c5(h_rep)
    if np.isfinite(c) and c < best_c:
        return h_rep.copy(), c
    return best_h, best_c

def optimize_resolution(h_start, deadline, best_h, best_c, rng):
    n = len(h_start)
    target_sum = n / 2.0
    h_init = repair_h(h_start)
    y0 = h_to_y(h_init)
    
    betas = [20.0, 60.0, 200.0, 600.0, 1500.0, 4000.0]
    maxiters = [300, 250, 200, 150, 100, 80]
    
    n_attempts = 3
    
    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break
        
        y_cur = y0.copy()
        if attempt == 1:
            y_cur = y0 + rng.normal(0, 0.05, n)
        elif attempt == 2:
            y_cur = y0 + rng.normal(0, 0.15, n)
        
        for bi, (beta, maxit) in enumerate(zip(betas, maxiters)):
            if time.monotonic() > deadline:
                break
            
            try:
                res = minimize(
                    smooth_fun, y_cur,
                    args=(beta, target_sum),
                    jac=True,
                    method='L-BFGS-B',
                    options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-10, 'gtol': 1e-8}
                )
                if np.all(np.isfinite(res.x)):
                    y_cur = res.x
            except Exception:
                pass
            
            h_eval = make_h_from_y(y_cur, target_sum)
            h_eval = repair_h(h_eval)
            c_eval = exact_c5(h_eval)
            best_h, best_c = try_candidate(h_eval, best_h, best_c)
        
        # Pair local search
        if time.monotonic() < deadline - 30:
            h_ls = best_h.copy() if best_c < exact_c5(h_init) else h_eval.copy()
            # Use the current best at this resolution
            h_ls = repair_h(best_h)
            h_ls, c_ls = pair_local_search(h_ls, max_sweeps=50)
            best_h, best_c = try_candidate(h_ls, best_h, best_c)
    
    return best_h, best_c
