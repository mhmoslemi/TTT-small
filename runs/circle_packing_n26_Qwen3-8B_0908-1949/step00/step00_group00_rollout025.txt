import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define initial grid positions
    # 5x5 grid with a central circle (26 circles)
    grid_size = 5
    positions = []
    for i in range(grid_size):
        for j in range(grid_size):
            x = i / (grid_size - 1)
            y = j / (grid_size - 1)
            positions.append([x, y])
    # Add a central circle
    positions.append([0.5, 0.5])
    
    # Initial radii: uniform
    initial_radii = np.full(26, 0.03)
    
    # Define variables for optimization
    n = 26
    x = opt.Variable((n, 2))
    r = opt.Variable(n)
    
    # Objective: maximize sum of radii
    objective = opt.Maximize(opt.sum(r))
    
    # Constraints
    constraints = []
    
    # All centers must lie within [0,1]^2
    for i in range(n):
        constraints.append(x[i, 0] >= 0)
        constraints.append(x[i, 0] <= 1)
        constraints.append(x[i, 1] >= 0)
        constraints.append(x[i, 1] <= 1)
    
    # Radii must be non-negative
    for i in range(n):
        constraints.append(r[i] >= 0)
    
    # Non-overlap constraints: distance between centers >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            distance = opt.sqrt((x[i, 0] - x[j, 0]) ** 2 + (x[i, 1] - x[j, 1]) ** 2)
            constraints.append(distance >= r[i] + r[j])
    
    # Initial guess for optimization
    initial_guess = np.vstack((np.array(positions), initial_radii)).flatten()
    
    # Solve the optimization problem
    problem = opt.Problem(objective, constraints)
    result = problem.solve(method='SLSQP', maxiter=1000, tol=1e-8, initial_guess=initial_guess)
    
    # Extract results

    radii = np.reshape(result[n*2:], (n,))

    valid, message = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Packing invalid: {message}")
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
