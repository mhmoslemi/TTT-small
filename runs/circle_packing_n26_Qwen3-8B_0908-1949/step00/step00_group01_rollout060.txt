import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def objective_function(x, n_circles, centers_grid):
    radii = x[n_circles:]
    centers = x[:n_circles].reshape((n_circles, 2))
    return -np.sum(radii)  # Minimize negative sum to maximize sum

def constraints_func(x, n_circles, centers_grid):
    # Extract centers and radii
    centers = x[:n_circles].reshape((n_circles, 2))
    radii = x[n_circles:]
    
    # Boundary constraints
    for i in range(n_circles):
        x_i, y_i = centers[i]
        r_i = radii[i]
        if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
            return np.inf  # Penalize if outside the square

    # Non-overlapping constraints
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < radii[i] + radii[j] - 1e-12:
                return np.inf  # Penalize if overlapping

    # Ensure non-negative radii
    for i in range(n_circles):
        if radii[i] < 0:
            return np.inf  # Penalize if radius is negative

def run_packing():
    # Initial grid-based circle positions
    n_circles = 26
    grid_size = 5
    grid = np.array([[i / (grid_size - 1), j / (grid_size - 1)] for j in range(grid_size) for i in range(grid_size)])

    initial_radii = np.full(n_circles, 0.05)

    # Flatten centers and radii for the optimization
    x0 = np.concatenate([grid.flatten(), initial_radii])

    # Bounds: centers must be in [0, 1] and radii must be non-negative
    bounds = Bounds(np.zeros_like(x0), np.inf)

    # Optimization
    result = opt.minimize(
        fun=lambda x: objective_function(x, n_circles, grid),
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints={
            'type': 'ineq',
            'fun': lambda x: constraints_func(x, n_circles, grid)

        options={'disp': True, 'maxiter': 100}

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
