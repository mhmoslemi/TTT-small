    col_spacing = 1.0  # width between columns in hexagonal grid

    # We will place 7 circles in the first row
    # Then 6 circles in the next, and 5 in the next
    # Making a total of 7+6+5+4+3+2+1 = 28 circles, but need to adjust for 26

    # Reduce the number to 26 by adjusting the last row
    rows = 6  # 6 rows for 26 circles
    cols_per_row = [7, 6, 5, 4, 3, 2]

    idx = 0
    for i in range(rows):
        for j in range(cols_per_row[i]):
            # Adjust positions to fit within the square
            x = j * col_spacing / (rows + 1)
            y = i * row_spacing / (rows + 1)
            centers[idx] = [x, y]
            radii[idx] = 0.05  # initial radius
            idx += 1

    # Initial guess for radii
    initial_radii = np.full(n, 0.05)

    # Define the objective function
    def objective(x):
        return -np.sum(x)  # Minimize negative of sum of radii to maximize

    # Define the constraints
    def constraint_center(x):
        return np.concatenate([x[i*2:(i+1)*2] for i in range(n)])

    def constraint_boundary(x):
        return np.zeros(n)

    def constraint_overlap(x):
        return np.zeros(n)

    # Helper function to detect overlap
    def constraint_overlap_func(x):
        centers = np.reshape(x[:n*2], (n, 2))
        radii = x[n*2:]
        for i in range(n):

                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.array([-1e6])  # Penalize overlapping
        return np.array([0.0])

    # Optimization problem
    x0 = np.concatenate([centers.flatten(), initial_radii])

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
