import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():

    n = 26
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    
    # Optimal radius in hexagonal packing for a square of side length L
    # tau = sqrt(3)/2 * d where d is the side of each circle
    # In a square of size L x L, the maximum number of hexagons that can fit is roughly ((L / d) * (L / (d * sqrt(3)/2))) = (L^2) / (d^2 * sqrt(3)/2)
    # Solving for d, we get d = L / sqrt(2 * sqrt(3) * (n / L^2)) )

    # As a heuristic, estimate the packing based on hexagonal grid
    # Let's try to fit 26 circles in a hexagonal grid inside the unit square

    # Calculate initial grid parameters
    grid_side = 0.8  # Approximate size of the hexagonal grid to fit in the unit square
    h = np.sqrt(3) / 2 * grid_side  # Vertical spacing
    w = grid_side  # Horizontal spacing
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))

    # Generate hexagonal grid points
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * w + w / 2  # Shift to center the grid horizontally
            y = i * h
            if i % 2 == 1:
                x += w / 2  # Offset alternate rows
            centers.append([x, y])

    centers = np.array(centers[:n])
    radii = np.full(n, 0.0)

    # Initial guess for radii: based on grid spacing
    initial_r = grid_side / 2
    radii = np.full(n, initial_r)

    # Normalize to fit within unit square
    max_dist = np.max(np.linalg.norm(centers, axis=1))
    radii = radii * (1.0 / max_dist)

    # Optimization: maximize sum of radii with non-overlap and boundary constraints
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum to maximize

    def gradient(radii):
        return np.zeros_like(radii)

    def constraints(radii):
        # Boundary constraints
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - 1e-12})
        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: (r[i] + r[j]) - dist - 1e-12})
        return constraints

    # Optimization bounds
    bounds = Bounds(0, 1.0)

    # Run optimization
    result = optimize.minimize(
        objective,
        radii,
        jac=gradient,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 1000}
    )

    # Post-optimization: check if valid
    valid, msg = validate_packing(result.x.reshape(-1, 1), result.x)

    if not valid:
        raise ValueError(msg)

    sum_radii = np.sum(result.x)

    return centers, result.x, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
