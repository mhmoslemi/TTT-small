import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Step 1: Generate initial symmetric grid of 26 circles (hexagonal packing)
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Create a square grid of 5x5 = 25 points with one extra
    grid = np.array([[i/4, j/4] for i in range(5) for j in range(5)])
    centers[:25] = grid
    centers[25] = [0.5, 0.5]  # Add one more in the center for symmetry

    # Initial radius: same for all
    radii[:] = 0.05

    def objective(radii):
        return -np.sum(radii)

    def constraint_overlap(i, j, radii):
        x1, y1 = centers[i]
        x2, y2 = centers[j]
        dist = np.hypot(x1 - x2, y1 - y2)
        return dist - (radii[i] + radii[j]) + 1e-12

    def constraint_boundary(i, radii):
        x, y = centers[i]
        return (x - radii[i] + 1e-12)  # Ensures x - r >= 0
        # and (x + radii[i] - 1 - 1e-12)  # Ensures x + r <= 1

    # Define the constraints
    constraints = []

    for i in range(num_circles):
        # Add boundary constraints
        cons = {'type': 'ineq', 'fun': lambda r, i=i: constraint_boundary(i, r)}
        constraints.append(cons)

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            cons = {'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint_overlap(i, j, r)}
            constraints.append(cons)

    # Bounds on radii (non-negative)
    bounds = Bounds(0, np.inf)

    # Optimization
    result = opt.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-10}
    )

    # Extract results
    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
