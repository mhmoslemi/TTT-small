import numpy as np
import cvxpy as cp

def run_packing():
    # Initial positions: hexagonal grid in a unit square
    # We'll place the circles in a square with hexagonal packing
    # Parameters: number of rows and columns
    # Assuming we place 5 rows of hexagons to fit 26 circles (can be adjusted)
    num_rows = 5
    num_cols = 6
    spacing = 1.0 / (num_cols - 1)  # Spacing between centers in x-direction
    hexagon_height = np.sqrt(3) / 2 * spacing  # Height of a hexagon in y-direction

    # Generate initial positions
    centers = np.zeros((26, 2))
    idx = 0
    for row in range(num_rows):
        for col in range(num_cols):
            x = col * spacing
            y = row * hexagon_height
            if row % 2 == 1:
                x += spacing / 2  # Shift odd rows for hexagonal symmetry
            centers[idx] = [x, y]
            idx += 1

    # Initial radii: uniform
    radii = np.ones(26) * 0.05

    # Define optimization variables
    x = cp.Variable((26, 2))
    r = cp.Variable(26, non_negative=True)

    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []

    # Non-overlap constraints
    for i in range(26):
        for j in range(i + 1, 26):
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(dist >= r[i] + r[j] - 1e-6)  # Allow a small tolerance

    # Boundary constraints (within [0,1]^2)
    for i in range(26):
        constraints.append(x[i, 0] >= 0)
        constraints.append(x[i, 0] <= 1)
        constraints.append(x[i, 1] >= 0)
        constraints.append(x[i, 1] <= 1)

    # Initial guess
    x.value = centers
    r.value = radii

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-6)

    # Extract results
    centers_opt = x.value
    radii_opt = r.value
    sum_radii = np.sum(radii_opt)

    return centers_opt, radii_opt, sum_radii
