import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():

    # First, calculate how many rows we can fit in the square
    # We assume r is a starting value for configuration
    # Start by distributing circles in a hexagonal grid

    # Number of circles
    n = 26
    # Starting radius as a guess
    r = 0.08

    # Compute how many rows fit in a square: floor(sqrt(n))
    rows = int(np.sqrt(n))
    cols = (n + rows - 1) // rows

    # Initial centers
    centers = np.zeros((n, 2))
    for i in range(rows):
        for j in range(cols):
            idx = i * cols + j
            if idx >= n:
                break
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            centers[idx] = [x, y]

    # Initial radii
    radii = np.full(n, r)

    # Optimization: maximize sum of radii under constraints
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        # Boundary constraints: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
        x = centers[:, 0]
        y = centers[:, 1]
        r = radii
        return np.concatenate([
            x - r,
            1 - x - r,
            y - r,
            1 - y - r
        ])

    def inequality_constraints(radii):
        # Non-overlap constraints: distance between centers >= radii[i] + radii[j]
        x = centers[:, 0]
        y = centers[:, 1]
        r = radii
        dists = np.sqrt((x[:, np.newaxis] - x[np.newaxis, :]) ** 2 + (y[:, np.newaxis] - y[np.newaxis, :]) ** 2)
        return dists - (r[np.newaxis, :] + r[:, np.newaxis])

    # Set up bounds for optimization
    bounds = Bounds(np.zeros(n), np.inf)

    # Optimize
    result = optimize.minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=[optimize.LBFGSBConstraint(fun=constraints, bounds=bounds, type='ineq'),
                     optimize.LBFGSBConstraint(fun=inequality_constraints, bounds=bounds, type='ineq')],

    )

    # Validate the result
