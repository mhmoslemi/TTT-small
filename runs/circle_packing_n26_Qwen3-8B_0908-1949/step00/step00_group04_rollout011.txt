import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define initial centers (grid-based)
    n = 26
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    
    centers = np.zeros((n, 2))
    idx = 0
    for i in range(rows):
        for j in range(cols):
            if idx >= n:
                break

            centers[idx, 1] = i / (rows - 1)
            idx += 1

    # Initial guess for radii
    radii = np.full(n, 0.03)

    # Optimization bounds for radii (non-negative)
    bounds = [(0, 1) for _ in range(n)]

    def objective(x):
        return -np.sum(x)  # Negative for minimization

    # Constraints for boundary and non-overlap
    def constraint_boundary(x, centers):
        radii = x
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -1
        return 0

    def constraint_overlap(x, centers):

        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                dx = x1 - x2
                dy = y1 - y2
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': lambda x: constraint_boundary(x, centers)},
            {'type': 'ineq', 'fun': lambda x: constraint_overlap(x, centers)}
        ],
        method='SLSQP'
    )

    radii_opt = result.x

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
