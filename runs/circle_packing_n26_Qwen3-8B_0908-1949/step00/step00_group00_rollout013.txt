import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt

def run_packing():
    """
    Generates a configuration of 26 non-overlapping circles within the unit square [0,1]×[0,1]
    to maximize the sum of their radii.
    Returns: (centers, radii, sum_radii)
    """

    # Initial configuration: hexagonal grid with 5 rows and 6 columns
    # Number of rows = 5, number of columns = 6
    num_circles = 26
    rows = 5
    cols = 6
    centers = np.zeros((num_circles, 2))
    for i in range(rows):
        for j in range(cols):
            x = j / (cols - 1)
            y = i / (rows - 1)
            centers[i * cols + j] = [x, y]

    # Initial guess for radii (uniform radius)
    radii = np.ones(num_circles) * 0.1

    # Objective function to maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Negative for minimization

    # Bounds on radii: non-negative
    bounds = [(0, 1) for _ in range(num_circles)]

    # Constraints to ensure circles do not overlap and stay within the unit square
    def constraint_overlap(x, centers):
        # x is the radii vector
        # centers is the centers matrix
        n = centers.shape[0]
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < x[i] + x[j] - 1e-8:
                    return dist - x[i] - x[j] + 1e-8
        return 0

    def constraint_boundary(x, centers):
        # x is the radii vector
        # centers is the centers matrix
        n = centers.shape[0]
        for i in range(n):
            x_center, y_center = centers[i]
            r = x[i]
            if x_center - r < -1e-8 or x_center + r > 1 + 1e-8 or y_center - r < -1e-8 or y_center + r > 1 + 1e-8:
                return 1e-8 - (x_center - r) if x_center - r < -1e-8 else 0
        return 0

    # Define constraints
    cons = [
        {'type': 'ineq', 'fun': lambda x: constraint_overlap(x, centers)},
        {'type': 'ineq', 'fun': lambda x: constraint_boundary(x, centers)}
    ]

    # Run the optimization
    result = opt.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        tol=1e-8
    )

    return optimized_centers, optimized_radii, sum_radii


def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:

    return True, "ok"
