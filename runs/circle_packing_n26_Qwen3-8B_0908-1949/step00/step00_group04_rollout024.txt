import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Step 1: Initial Hexagonal Packing
    n = 26
    rows = 5
    cols_per_row = [6, 5, 6, 5, 6]
    centers = np.zeros((n, 2))
    idx = 0

    for row in range(rows):
        col_count = cols_per_row[row]
        for col in range(col_count):
            x = col + 0.5 * (1 - (row % 2))  # alternate row spacing
            y = row + 0.5 * (row % 2)         # staggered rows
            centers[idx] = [x, y]
            idx += 1

    # Step 2: Compute initial radius based on spacing
    # Maximum possible radius before overlapping
    max_radius = 0.15  # This is a reasonable initial guess based on spacing
    radii = np.full(n, max_radius)

    # Step 3: Define objective function (maximize sum of radii)
    def objective(radii):
        return -np.sum(radii)  # minimize negative of sum

    # Step 4: Define constraints
    def constraint_overlap(i, j, radii):
        def cons(x):
            r = x.copy()
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            return dist - (r[i] + r[j]) + 1e-12  # ensure non-overlapping
        return cons

    def constraint_boundary(i, radii):
        def cons(x):
            r = x.copy()
            x_c, y_c = centers[i]
            return (x_c - r[i] >= -1e-12 and
                    x_c + r[i] <= 1 + 1e-12 and
                    y_c - r[i] >= -1e-12 and
                    y_c + r[i] <= 1 + 1e-12)
        return cons

    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i, radii)})

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': constraint_overlap(i, j, radii)})

    # Step 5: Optimization
    result = optimize.minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=constraints,
        options={'ftol': 1e-9, 'maxiter': 1000}
    )

    # Step 6: Validate and return result
    centers = result.x.reshape(-1, 2)
    radii = result.x
    sum_radii = np.sum(radii)

    # Final validation
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Validation failed: {msg}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
