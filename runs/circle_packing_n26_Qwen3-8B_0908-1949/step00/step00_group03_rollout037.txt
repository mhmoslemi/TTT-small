import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial distribution of circles in a grid-like pattern
    num_circles = 26
    grid_rows = 5
    grid_cols = 6
    spacing_x = 1.0 / grid_cols
    spacing_y = 1.0 / grid_rows
    centers = np.zeros((num_circles, 2))
    radii = np.ones(num_circles) * 0.05

    for i in range(num_circles):
        col = i % grid_cols
        row = i // grid_cols
        x = col * spacing_x + 0.5 * spacing_x
        y = row * spacing_y + 0.5 * spacing_y
        centers[i] = [x, y]

    # Define the objective function to maximize
    def objective(x):
        # x contains the flattened centers and radii
        centers_flat = x[:num_circles * 2].reshape(num_circles, 2)
        radii_flat = x[num_circles * 2:].reshape(num_circles)
        return -np.sum(radii_flat)  # Minimize negative sum for maximization

    # Define constraints
    def constraint_center_bounds(x):
        centers_flat = x[:num_circles * 2].reshape(num_circles, 2)
        radii_flat = x[num_circles * 2:].reshape(num_circles)
        for i in range(num_circles):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
