import numpy as np
import scipy.optimize as optimize

def run_packing():

    n = 26

    # For simplicity, start with a regular hexagonal grid inside the unit square
    # We will place circles in rows. Let's use 6 rows for 26 circles
    # This is a very rough starting point, and we will refine it later
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // 6
        col = i % 6
        # Adjust coordinates to fit within [0, 1] and ensure spacing between circles
        y = 0.5 + row * (1 / (6 + 1))  # n rows with spacing

        if (x - radii[i] < 0) or (x + radii[i] > 1) or (y - radii[i] < 0) or (y + radii[i] > 1):
            radii[i] = min(x, 1 - x, y, 1 - y)

    # Define the objective function to maximize sum of radii
    def objective(x):

        # Flatten x for easier access
        flat_x = x.reshape(-1)

        radii_flat = np.maximum(radii_flat, 0)

            x, y = centers_flat[i]

    result = optimize.minimize(

        bounds=bounds,
        method='L-BFGS-B',
        tol=1e-6,
