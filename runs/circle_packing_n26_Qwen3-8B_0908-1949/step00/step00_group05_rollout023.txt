import numpy as np
import scipy.optimize as opt

def run_packing():

    num_circles = 26
    # Hexagonal packing with 2 rows
    rows = 2
    cols = 13  # Because 2 rows of hexagonal packing with 13 circles can fit in a unit square
    
    # Hexagonal grid parameters
    hex_side = 0.5 / np.sqrt(3)
    dx = hex_side * np.sqrt(3)
    dy = hex_side

    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    for i in range(rows):
        for j in range(cols):
            x = j * dx
            y = i * dy + (j % 2) * (dy / 2)
            centers[i * cols + j] = [x, y]
            radii[i * cols + j] = 0.05  # Initial radius

    # Function to compute the total area (sum of radii squared, negative for minimization)
    def objective(x):
        centers_flat = x[:num_circles * 2].reshape((num_circles, 2))
        radii_flat = x[num_circles * 2:]
        return -np.sum(radii_flat)

    # Constraints for boundary (centers within [0,1])
    def boundary_constraints(x):
        centers_flat = x[:num_circles * 2].reshape((num_circles, 2))
        radii_flat = x[num_circles * 2:]
        return np.concatenate([
            centers_flat[:, 0] - radii_flat,
            1 - centers_flat[:, 0] - radii_flat,
            centers_flat[:, 1] - radii_flat,
            1 - centers_flat[:, 1] - radii_flat
        ])

    # Constraints for no overlap between circles
    def no_overlap_constraints(x):
        centers_flat = x[:num_circles * 2].reshape((num_circles, 2))

        constraints = []

                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist_sq = dx ** 2 + dy ** 2
                dist = np.sqrt(dist_sq)
                constraint = dist - (radii_flat[i] + radii_flat[j])
                constraints.append(constraint)
        return np.array(constraints)

    # Set up bounds: centers must be within [0,1], radii ≥ 0
    bounds = []

        bounds.append((0, 1))

        bounds=bounds,

    return centers_opt, radii_opt, np.sum(radii_opt)
