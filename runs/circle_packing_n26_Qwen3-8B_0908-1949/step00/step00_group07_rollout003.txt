import numpy as np
import scipy.optimize as opt
from typing import Tuple, Any

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    def objective_function(x: np.ndarray, radii: np.ndarray) -> float:
        return -np.sum(radii)  # Maximize sum of radii by minimizing negative sum

    def constraints(x: np.ndarray, radii: np.ndarray) -> np.ndarray:
        # Constraints: distance between any two centers >= sum of radii
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = x[2*i] - x[2*j]
                dy = x[2*i + 1] - x[2*j + 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (radii[i] + radii[j]) + 1e-8)  # +1e-8 for tolerance

        # Boundary constraints: x_i - r_i >= 0 and x_i + r_i <= 1
        for i in range(n):
            constraints.append(x[2*i] - radii[i] + 1e-8)  # x_i - r_i >= 0
            constraints.append(1.0 - x[2*i] - radii[i] + 1e-8)  # 1 - x_i - r_i >= 0
            constraints.append(x[2*i + 1] - radii[i] + 1e-8)  # y_i - r_i >= 0
            constraints.append(1.0 - x[2*i + 1] - radii[i] + 1e-8)  # 1 - y_i - r_i >= 0

        return np.array(constraints)

    # Initial configuration: place circles in a hexagonal grid in unit square
    n = 26
    radius_initial = 0.1  # Initial guess for all radii
    centers_initial = np.zeros(2 * n)
    for i in range(n):
        # Hexagonal grid with vertical spacing: ~sqrt(3)/2 times horizontal spacing
        col = i % 5  # 5 columns
        row = i // 5  # 5 rows
        x = col * 0.2 + 0.1  # Horizontal spacing
        y = row * np.sqrt(3)/2 * 0.2 + 0.1  # Vertical spacing
        centers_initial[2*i] = x
        centers_initial[2*i + 1] = y

    # Randomly perturb the centers to avoid local optima
    np.random.seed(0)
    centers_initial += np.random.uniform(-0.05, 0.05, size=2 * n)

    # Initial radii guess
    radii_initial = np.full(n, radius_initial)

    # Optimization
    result = opt.minimize(
        fun=lambda x: objective_function(x, radii_initial),
        x0=centers_initial,
        method='SLSQP',
        constraints=opt.constraints.Constraint(
            fun=lambda x: constraints(x, radii_initial),
            type='ineq'
        ),
        bounds=[(0.0, 1.0) for _ in range(2 * n)]
    )

    # Extract results
    centers = result.x[:2*n].reshape(-1, 2)
    radii = np.full(n, radius_initial)

    # Check if the optimization succeeded
    if not result.success:
        raise ValueError("Optimization failed")

    # Validate the packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")

    # Return the result
    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
