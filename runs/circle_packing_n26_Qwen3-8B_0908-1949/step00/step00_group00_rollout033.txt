import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    num_circles = 26
    centers = np.zeros((num_circles, 2))
    # Create a grid layout with more spacing for better packing
    row_count = 4
    col_count = 7
    spacing = 1.0 / (col_count + 1)
    for i in range(row_count):
        for j in range(col_count):
            idx = i * col_count + j
            x = j * spacing + spacing / 2
            y = i * spacing + spacing / 2
            centers[idx] = [x, y]
    
    # Initial guess for radii: square root of area per circle

    def objective(radii_vec):
        return -np.sum(radii_vec)

    # Constraints
    def constraint_center_inside(radii_vec):

        centers_with_radii = np.vstack((centers, radii_vec))
        # Check if centers are within the unit square
        center_bounds = np.all((centers_with_radii[:, 0] >= -1e-12) & (centers_with_radii[:, 0] <= 1 + 1e-12))
