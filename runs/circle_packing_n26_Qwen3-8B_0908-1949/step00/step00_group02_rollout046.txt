import numpy as np
import cvxpy as cp

def run_packing():
    n = 26
    # Initial layout: dense grid in a triangle-like shape
    max_y = 1.0
    max_x = 1.0
    centers = np.zeros((n, 2))
    
    for i in range(n):
        row = i // 5  # 5 circles per row
        col = i % 5
        x = col / 4.0  # spread columns evenly across 0 to 1
        y = row / 4.0  # spread rows evenly
        centers[i] = [x, y]
    
    # Introduce slight random perturbation to avoid overlapping
    centers += np.random.uniform(-0.05, 0.05, size=centers.shape)
    
    # Estimate initial radii based on spacing
    radii = np.zeros(n)
    for i in range(n):
        dist_to_neighbors = []
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx ** 2 + dy ** 2)
            dist_to_neighbors.append(dist)
        min_dist = min(dist_to_neighbors)
        radii[i] = min_dist / 2.0
    
    # Define the optimization variables
    x = cp.Variable(n, 2)
    r = cp.Variable(n)
    
    # Constraints
    constraints = [
        # Boundary constraints
        [0 <= x[i, 0] + r[i], x[i, 0] - r[i] <= 1 for i in range(n)],
        [0 <= x[i, 1] + r[i], x[i, 1] - r[i] <= 1 for i in range(n)],
        # Non-overlapping constraints
        [cp.norm(x[i] - x[j]) >= r[i] + r[j] for i in range(n) for j in range(i + 1, n)]
    ]
    
    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(r))
    
    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver='SCS', eps=1e-6)
    
    # Extract the solution
    centers_opt = np.array(x.value)
    radii_opt = np.array(r.value)
    
    # Validate the solution
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise RuntimeError(f"Optimization failed with error: {msg}")
    
    return centers_opt, radii_opt, np.sum(radii_opt)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
