import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial guess for the centers of the circles in a hexagonal grid
    num_circles = 26
    row_count = 5  # 5 rows of circles
    col_count = 5  # 5 columns of circles
    circle_radius_initial = 0.1  # initial radius
    
    # Define the hexagonal grid
    centers = np.zeros((num_circles, 2))
    radius_guess = np.full(num_circles, circle_radius_initial)
    
    for i in range(row_count):
        for j in range(col_count):
            x = j * (2 * circle_radius_initial + 2 * circle_radius_initial * np.sqrt(3)/2)

            centers[idx] = [x, y]

    bounds_centers = [(0, 1) for _ in range(2 * num_circles)]
    bounds_radii = [(0, 1) for _ in range(num_circles)]

    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum to maximize
    
    # Constraints for non-overlapping
    def non_overlap_constraint(x):
        centers = x[:2*num_circles].reshape(num_circles, 2)
        radii = x[2*num_circles:]
        constraints = []
        for i in range(num_circles):

                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]

                constraints.append(dist - (radii[i] + radii[j]) + 1e-12)
        return constraints
    
    # Constraints for centers within the unit square
    def boundary_constraint(x):
        centers = x[:2*num_circles].reshape(num_circles, 2)
        radii = x[2*num_circles:]

            constraints.append(y_c - radii[i] + 1e-12)
            constraints.append(1 - y_c - radii[i] + 1e-12)
        return constraints

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
