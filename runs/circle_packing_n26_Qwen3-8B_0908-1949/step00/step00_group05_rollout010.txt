import scipy.optimize as optimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial hexagonal packing
    num_circles = 26
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    
    # Hexagon packing parameters
    spacing = 1.0 / (cols + 1)
    hex_radius = spacing * np.sqrt(3) / 2
    max_radius = spacing / 2
    
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    for i in range(num_circles):
        row = i // cols
        col = i % cols
        x = col * spacing + spacing / 2
        y = row * spacing * np.sqrt(3) + spacing / 2
        centers[i] = [x, y]
        radii[i] = min(max_radius, max(0.0, 1.0 - max(x, 1.0 - x, y, 1.0 - y)))

    x = np.concatenate([centers.reshape(-1), radii])
    x0 = x

    # Define objective function
    def objective(x):
        centers_opt = x[:2*num_circles].reshape(num_circles, 2)
        radii_opt = x[2*num_circles:]
        return -np.sum(radii_opt)

        centers_opt = x[:2*num_circles].reshape(num_circles, 2)
        radii_opt = x[2*num_circles:]
        dist = cdist(centers_opt, centers_opt)
        return np.sum((dist - (radii_opt[:, np.newaxis] + radii_opt[np.newaxis, :])) < 1e-6)

                return -1
        return 0

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
