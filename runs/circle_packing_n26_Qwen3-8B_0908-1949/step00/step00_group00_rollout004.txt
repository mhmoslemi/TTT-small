import numpy as np
import cvxpy as cp
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Number of circles
    n = 26

    # Initialize centers using a hexagonal grid and perturbations for diversity
    # 5 rows of 5 circles, slightly offset in each row
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 0.5 * row)
        y = row + 0.5 * (1 - 0.5 * col)
        x += np.random.uniform(-0.1, 0.1)
        y += np.random.uniform(-0.1, 0.1)
        centers[i] = np.array([x, y])

    # Define all variables
    r = cp.Variable(n, nonneg=True)
    x = cp.Variable(n)
    y = cp.Variable(n)

    # Objective: Maximize total sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []
    for i in range(n):
        # Boundary constraints
        constraints.append(x[i] - r[i] >= 0)
        constraints.append(x[i] + r[i] <= 1)
        constraints.append(y[i] - r[i] >= 0)
        constraints.append(y[i] + r[i] <= 1)

    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            constraints.append(dist >= r[i] + r[j] - 1e-5)

    # Solve the problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS)

    if prob.status not in ["optimal", "optimal_inaccurate"]:
        raise ValueError("Optimization failed to find a feasible solution.")

    # Validate the packing
    centers_opt = np.stack((x.value, y.value), axis=1)
    radii_opt = r.value

    # Check for nan values
    if np.isnan(centers_opt).any() or np.isnan(radii_opt).any():
        raise ValueError("NaN values found in the solution.")

    # Check if radii are non-negative
    if np.any(radii_opt < 0):
        raise ValueError("Negative radii values found in the solution.")

    # Validate the packing using the given validation function
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")

    return centers_opt, radii_opt, np.sum(radii_opt)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
