import scipy.optimize as optimize
from scipy.optimize import minimize

def validate_packing(centers, radii):

        return False, "NaN values present"

        if radii[i] < 0:

        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def initial_hexagonal_grid(n, max_radius=0.1):
    """
    Generate initial hexagonal grid of circles in a unit square.
    """
    centers = []
    radii = []
    col = 0
    row = 0
    while len(centers) < n:
        x = col * (2 * max_radius) + max_radius
        y = row * (np.sqrt(3) * max_radius) + max_radius
        if x <= 1 - max_radius and y <= 1 - max_radius:
            centers.append([x, y])
            radii.append(max_radius)
        col += 1
        if col % 2 == 0:
            row += 1
    return np.array(centers), np.array(radii)

def energy_function(params, centers, radii, n):

    Energy function to minimize.
    """
    if params.shape[0] != 2 * n:
        raise ValueError("Incorrect number of parameters")
    new_centers = params.reshape((n, 2))

    for i in range(n):
        x, y = new_centers[i]
        r = new_radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            constraints.append(1000)
        else:
            constraints.append(0)
    for i in range(n):
        for j in range(i + 1, n):
            dx = new_centers[i, 0] - new_centers[j, 0]

            if dist < new_radii[i] + new_radii[j] - 1e-12:
                constraints.append(1000)
            else:
                constraints.append(0)
    return np.sum(constraints)
