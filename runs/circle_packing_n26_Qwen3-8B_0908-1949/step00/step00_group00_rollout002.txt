import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt

def run_packing():
    # Step 1: Generate an initial hexagonal grid of circles
    n = 26
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    positions = []
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            positions.append((x, y))
    positions = np.array(positions)

    positions /= np.max(positions)
    
    # Step 3: Initialize radii
    radii = np.ones(n) * 0.05
    
    # Step 4: Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    # Step 5: Define constraints
    def constraints(x):
        # x has shape (n, 2) for centers and (n,) for radii
        centers, r = x[:n*2].reshape(n, 2), x[n*2:]
        
        # Boundary constraints
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12 or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12:
                return np.inf  # Violation of boundary constraints
        
        # Non-overlap constraints
        for i in range(n):

                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < r[i] + r[j] - 1e-12:
                    return np.inf  # Violation of non-overlap constraint
        
        return 0
    
    # Step 6: Set up optimization
    bounds = [(0, 1) for _ in range(n*2)] + [(0, 1) for _ in range(n)]
    x0 = np.concatenate([positions.flatten(), radii])
    
    # Perform optimization
    result = opt.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraints},
        method="SLSQP"
    )

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
