from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess for circle centers as a hexagonal grid
    n = 26
    centers = np.zeros((n, 2))
    
    # Generate a hexagonal grid pattern in the unit square
    # Start with 5 rows, up to 26 circles total
    # Find a hexagonal pattern that fits in the unit square
    
    # First, define a hexagonal-like arrangement
    num_rows = 5  # 5 rows for a hexagonal grid
    num_cols = 6  # 6 columns per row (changes in rows)
    total = num_rows * num_cols  # Total circles in hexagonal pattern
    if total < n:
        num_cols += 1  # Add one more column for extra circles
    
    # Place circles in a hexagonal pattern
    for i in range(n):
        row = i // num_cols
        col = i % num_cols
        x = col * (1.5 / num_cols)
        y = row * (np.sqrt(3) / 2) / num_cols
        # Adjust to fit in the square
        x = np.clip(x, 0, 1)
        y = np.clip(y, 0, 1)
        centers[i] = [x, y]
    
    # Initial guess for radii
    radii = np.full(n, 0.1)
    
    # Objective function: maximize the sum of radii
    # We minimize -sum(radii)
    def objective(x):
        return -x.sum()
    
    # Constraints:
    # 1. All circles must lie within the square
    # 2. No circle can have negative radius
    # 3. All circles must not overlap
    
    # Bounds: radii must be non-negative
    bounds = Bounds(np.zeros(n), np.inf)
    
    # Function to check if a circle at (x, y) with radius r is within the square
    def within_square(x, y, r):
        return all(0 <= x - r and x + r <= 1 for x, y in zip([x], [y]))
    
    # Function to compute the distance between two circles
    def distance(c1, c2):
        return np.sqrt(np.sum((c1 - c2) ** 2))
    
    # Constraints that circles do not overlap
    def constraint_overlap(x):
        # x is a vector of all variables
        # First n variables are centers, then next n are radii
        centers = x[:n*2].reshape(n, 2)
        radii = x[n*2:]
        for i in range(n):
            for j in range(i + 1, n):
                d = distance(centers[i], centers[j])
                if d < radii[i] + radii[j] - 1e-8:
                    return np.inf  # Penalize overlapping circles
        return 0
    
    # Add overlap constraints
    constraints = [LinearConstraint([], [], [0], [], [], [], fun=constraint_overlap)]
    
    # Add square boundary constraints
    def constraint_boundary(x):
        centers = x[:n*2].reshape(n, 2)
        radii = x[n*2:]
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:

        return 0
    
    # Add boundary constraints
    constraints.append(LinearConstraint([], [], [0], [], [], [], fun=constraint_boundary))
    
    # Run optimization

        fun=objective,
        x0=np.concatenate([centers.flatten(), radii]),

        bounds=bounds,

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
