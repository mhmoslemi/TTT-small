import scipy.optimize as optimize
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial guess: grid of 26 circles
    # 5 rows x 5 columns with 1 extra circle
    num_circles = 26
    rows, cols = 5, 5
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Step 1: Generate a grid-based initial layout
    for i in range(num_circles):
        row = i // cols
        col = i % cols
        x = col / (cols - 1)
        y = row / (rows - 1)
        centers[i] = [x, y]
        radii[i] = 0.04  # Initial radius (tentative value for optimization)

    # Step 2: Define objective function
    def objective(radii_vector):
        return -np.sum(radii_vector)  # Minimize negative sum (equivalent to maximizing sum)

    # Step 3: Define constraints
    def constraint_centers(x_vec):
        # Flatten and reshape to centers matrix
        centers_flat = x_vec[:num_circles * 2].reshape((num_circles, 2))
        radii_flat = x_vec[num_circles * 2:].reshape((num_circles,))
        # Check boundary constraints

            x, y = centers_flat[i]
            r = radii_flat[i]
            if x - r < 0 - 1e-12 or x + r > 1 + 1e-12 or y - r < 0 - 1e-12 or y + r > 1 + 1e-12:
                return np.inf
        # Check non-overlap

            for j in range(i + 1, num_circles):

                dy = centers_flat[i, 1] - centers_flat[j, 1]

                    return np.inf
        return 0

    # Step 4: Optimization
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [
        (0.0, 1.0) for _ in range(num_circles * 2)
    ] + [(0.0, 1.0) for _ in range(num_circles)]

    result = optimize.minimize(
        fun=objective,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=[

        options={'maxiter': 1000, 'ftol': 1e-9}

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
