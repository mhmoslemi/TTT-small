from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Number of circles
    n = 26

    # Initial guess for circle centers (hexagonal packing)
    # We'll generate a hexagonal grid within [0, 1] x [0, 1]
    # Starting with a few rows
    num_rows = 5
    num_cols = 5

    centers = np.zeros((n, 2))
    radius_guess = np.zeros(n)

    # Hexagonal grid parameters
    row_offset = np.sqrt(3) / 2 * 0.15  # Vertical spacing between rows
    col_offset = 0.15                   # Horizontal spacing between columns

    # Generate a hexagonal grid with infinite spacing
    idx = 0
    for i in range(num_rows):
        for j in range(num_cols):
            x = j * col_offset
            y = i * row_offset
            centers[idx] = [x, y]
            idx += 1

    # Initialize radii based on initial spacing
    radius_guess = np.full(n, 0.15)

    # Objective function to maximize total area
    def objective(radii):
        return -np.sum(np.pi * radii**2)

    # Constraints
    def constraint_center_in_square(radii, centers):
        return np.sum((centers[:, 0] - radii) < 0) + np.sum((centers[:, 0] + radii) > 1) + \
               np.sum((centers[:, 1] - radii) < 0) + np.sum((centers[:, 1] + radii) > 1)

    def constraint_no_overlap(radii, centers):
        dist = np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :]) ** 2, axis=2))
        return np.sum(dist < radii[:, np.newaxis] + radii[np.newaxis, :])

    # Optimization bounds: radii must be non-negative
    bounds = Bounds(0, 1)

    # Optimization settings
    result = minimize(
        fun=objective,
        x0=radius_guess,
        method='SLSQP',
        jac=None,
        bounds=bounds,
        constraints=[

            {'type': 'ineq', 'fun': lambda r: constraint_no_overlap(r, centers)}
        ],
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Extract the final result
    radii = result.x
