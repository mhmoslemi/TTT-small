import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define the number of circles
    n = 26

    # Initialize positions using a grid-like pattern with hexagonal spacing
    # Start with a base grid layout
    centers = np.zeros((n, 2))
    row_count = int(np.ceil(np.sqrt(n)))
    col_count = int(np.ceil(n / row_count))
    
    for i in range(n):
        row = i // col_count
        col = i % col_count
        x = col / (col_count - 1)
        y = row / (row_count - 1)
        centers[i] = [x, y]
    
    # Optimize the radii and positions to maximize total radius
    # Objective: maximize sum of radii
    # Constraints:
    # 1. All circles must be inside the unit square
    # 2. All circles must not overlap

    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum to maximize sum

    def constraint_boundary(radii):
        # Check if all circles are within the unit square
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return -1  # Violation
        return 0  # No violation

    def constraint_overlap(radii):
        # Check if any two circles overlap
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j]:
                    return -1  # Violation
        return 0  # No violation

    # Setup optimization
    bounds = [(0.0, 1.0) for _ in range(n)]  # Radii must be between 0 and 1

    # Multi-start optimization
    best_centers = centers.copy()
    best_radii = np.ones(n) * 0.01  # Initial guess
    best_sum = 0.0

        centers_perturbed = centers + perturbation

        # Optimize radii given the positions
        # Define optimization problem

            return objective(radii)

        cons = [
            {'type': 'ineq', 'fun': constraint_boundary},

        ]

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
