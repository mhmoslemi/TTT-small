import numpy as np
import scipy.optimize as optimize
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initialize circle positions near the center with some near the corners
    centers = np.zeros((26, 2))
    radii = np.zeros(26)

    # Create an initial guess with 4 rows of 6 and 5 circles
    for i in range(26):
        col = i % 6
        row = i // 6
        x = col * 0.2 + 0.1  # Adjust spacing to fit in the unit square
        y = row * 0.2 + 0.1
        centers[i] = [x, y]

    # Function to calculate the total sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraints: non-overlap and boundary
    def constraint_overlap(radii, centers):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-6:
                    return dist - (radii[i] + radii[j] - 1e-6)
        return 0.0

    # Boundary constraints
    def constraint_boundary(radii):
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-6 or x + radii[i] > 1 + 1e-6:
                return x - radii[i] + 1e-6
            if y - radii[i] < -1e-6 or y + radii[i] > 1 + 1e-6:
                return y - radii[i] + 1e-6
        return 0.0

    # Create bounds on radii (non-negative)
    bounds = [(0, 1) for _ in range(26)]

    # Define the optimization problem
    constraints = [
        {'type': 'ineq', 'fun': constraint_overlap, 'args': (radii, centers)},
        {'type': 'ineq', 'fun': constraint_boundary, 'args': (radii,)}
    ]

    # Run the optimization
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'ftol': 1e-8, 'eps': 1e-8}
    )

    # Extract results

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
