import scipy.optimize as optimize

def run_packing():

    n_circles = 26
    rows = 3  # Number of rows in hexagonal packing
    cols = 9  # Number of columns in hexagonal packing
    row_spacing = 2.0  # Horizontal distance between adjacent circles
    col_spacing = 2.0  # Vertical distance between adjacent rows

    # Estimate initial radius based on area limitations
    initial_radius = (1.0 / np.sqrt(np.pi)) * np.sqrt(n_circles / 3)  # Area per circle in hexagonal packing

    # Generate initial centers
    centers = np.zeros((n_circles, 2))
    idx = 0
    for i in range(rows):
        for j in range(cols):
            x = j * row_spacing
            y = i * col_spacing
            if i % 2 == 1:
                x += row_spacing / 2  # Shift for staggered rows
            centers[idx] = [x, y]
            idx += 1

    # Scale centers to fit in the unit square
    centers = centers / (np.max(centers[:, 0]) + np.max(centers[:, 1]))  # Normalize to [0,1]^2

    # Initial radii
    radii = np.full(n_circles, initial_radius)

        return -np.sum(radii)  # Minimize the negative sum

    # Define constraints

        for i in range(n_circles):
            for j in range(i + 1, n_circles):

                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return np.inf  # Penalize overlap
        return 0

    def constraint_bounds(radii):
        for r in radii:
            if r < 0:

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
