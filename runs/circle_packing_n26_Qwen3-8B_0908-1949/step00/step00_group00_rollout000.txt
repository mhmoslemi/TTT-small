import numpy as np
import cvxpy as cp

def generate_initial_grid(num_circles, grid_size=5):
    """Generate an initial hexagonal grid packing for a given number of circles."""
    # Create a hexagonal grid of circles
    grid = []
    for i in range(grid_size):
        for j in range(grid_size):
            x = (i + 0.5 * j) / (grid_size - 1) if j % 2 == 1 else i / (grid_size - 1)
            y = j / (grid_size - 1)
            grid.append([x, y])
    # Shuffle to avoid clustering and make it more realistic
    np.random.shuffle(grid)
    return np.array(grid)

def compute_overlap(centers, radii):
    """Compute pairwise distances and check for overlaps between circles."""
    n = len(centers)
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-8:
                return True
    return False

    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []
    for i in range(num_circles):

    # No overlap between circles

            dy = x[i, 1] - x[j, 1]
            dist_squared = dx**2 + dy**2
            constraints.append(dist_squared >= (r[i] + r[j])**2 - 1e-8)

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    result = problem.solve(solver=cp.SCS, verbose=False)

    if result is None:
        raise RuntimeError("Optimization failed")

    centers = x.value

    valid, message = validate_packing(centers, radii)
