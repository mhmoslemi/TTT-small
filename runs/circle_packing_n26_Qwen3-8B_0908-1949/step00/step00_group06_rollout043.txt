def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):

        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

def objective_function(x, n_circles):

def constraint_boundary(x, n_circles):
    centers = x[:n_circles * 2].reshape((n_circles, 2))
    radii = x[n_circles * 2:]
    for i in range(n_circles):
        x_i, y_i = centers[i]

            return -1e6
        if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
            return -1e6
    return 0

def constraint_non_overlap(x, n_circles):

    radii = x[n_circles * 2:]
    for i in range(n_circles):

            dist_sq = dx*dx + dy*dy

def initial_hexagonal(n_circles):
    row_count = int(np.ceil(np.sqrt(n_circles)))

        y = (i // col_count) * (1 + r) * np.sqrt(3) / 2
        centers[i] = np.array([x, y])
    return centers
