def objective_function(params, n_circles):

    Objective function to maximize the total sum of radii.

    return -np.sum(radii)  # Minimize negative sum to maximize sum

def boundary_penalty(params, n_circles, radius_penalty=1000.0):
    """
    Penalty function for circles going out of bounds.
    """
    radii = params[:n_circles]
    centers = params[n_circles:]
    centers = centers.reshape(n_circles, 2)

    return penalty

def overlap_penalty(params, n_circles, overlap_penalty=10000.0):
    """
    Penalty function for circle overlaps.
