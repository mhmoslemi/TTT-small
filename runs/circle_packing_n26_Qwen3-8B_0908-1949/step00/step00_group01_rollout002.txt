import scipy.optimize as opt

def run_packing():

    centers = np.zeros((26, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            centers[idx] = [j / 4, i / 4]
            idx += 1
    centers[idx] = [0.5, 0.5]  # center circle

    # Initial guess for radii (all equal for uniformity)
    radii = np.full(26, 0.1)

    # Objective function to maximize the sum of radii
    def objective(x):
        return -np.sum(x)  # Negative to convert to minimization

    # Constraints: non-overlapping and boundary
    def constraint_overlap(x):
        radii = x
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1000  # Violation penalty
        return 0

    def constraint_boundary(x):
        radii = x
        for i in range(26):
            xcen, ycen = centers[i]
            if (xcen - radii[i] < -1e-12 or xcen + radii[i] > 1 + 1e-12 or
                ycen - radii[i] < -1e-12 or ycen + radii[i] > 1 + 1e-12):
                return -1000  # Violation penalty
        return 0

    # Bounds: radii must be non-negative
    bounds = [(0, 1) for _ in range(26)]

    # Use SLSQP method with constraints
    result = opt.minimize(

        x0=radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[
