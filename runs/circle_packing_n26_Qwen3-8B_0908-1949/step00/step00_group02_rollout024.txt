import numpy as np
from scipy.optimize import minimize
from scipy.optimize import OptimizeResult

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initialize 26 centers as a hexagonal grid
    def initial_centers():
        n = 26
        centers = np.zeros((n, 2))
        for i in range(n):
            row = i // 6
            col = i % 6
            x = col - 2.5 + 0.5 * (np.sin(np.pi / 3) * row)
            y = row * (np.sqrt(3) / 2) + 0.5
            centers[i, 0] = x
            centers[i, 1] = y
        return centers

    # Scale initial grid to unit square
    centers = initial_centers() * 1.5  # Slightly larger than unit square for flexibility
    radii = np.full(26, 0.15)  # Initial guess for radii

    # Define the objective function
    def objective(x):
        return -np.sum(x[26:])  # Maximize radii, so minimize negative of sum

    # Define the variables: first 52 are x and y coordinates of centers, next 26 are radii
    x0 = np.concatenate((centers.flatten(), radii))

    # Define bounds for the variables
    bounds = []
    for i in range(26):
        bounds.append((0.0, 1.0))  # x coordinate
        bounds.append((0.0, 1.0))  # y coordinate
        bounds.append((0.0, 1.0))  # radius

    # Define constraints for boundaries (x_i - r_i >= 0, x_i + r_i <= 1)
    def boundary_constraints(x):
        constraints = []
        for i in range(26):
            idx = i * 3
            x_i = x[idx]
            y_i = x[idx + 1]
            r_i = x[idx + 2]
            # x_i - r_i >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i * 3] - x[i * 3 + 2]})
            # x_i + r_i <= 1
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i * 3] - x[i * 3 + 2]})
            # y_i - r_i >= 0
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i * 3 + 1] - x[i * 3 + 2]})
            # y_i + r_i <= 1
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i * 3 + 1] - x[i * 3 + 2]})
        return constraints

    # Define constraints for circle separation
    def separation_constraints(x):
        constraints = []
        for i in range(26):
            for j in range(i + 1, 26):
                idx_i = i * 3
                idx_j = j * 3
                x_i = x[idx_i]
                y_i = x[idx_i + 1]
                r_i = x[idx_i + 2]
                x_j = x[idx_j]

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
