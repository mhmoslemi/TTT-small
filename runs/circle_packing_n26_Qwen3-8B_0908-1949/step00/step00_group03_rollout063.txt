import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # Adjusting to fit within the unit square by scaling

    # Grid dimensions
    rows = 4
    cols = 6
    grid_centers = []
    for i in range(rows):
        for j in range(cols):
            x = j / (cols - 1)
            y = i / (rows - 1)
            grid_centers.append([x, y])
    grid_centers = np.array(grid_centers)

    grid_centers_hex = []
    for i in range(rows):

            x = j + i / 2
            y = i
            if x > cols - 1:
                x = cols - 1
            if y > rows - 1:
                y = rows - 1
            grid_centers_hex.append([x, y])
    grid_centers_hex = np.array(grid_centers_hex)

    initial_centers = grid_centers_hex[:26]
    initial_radii = np.full(26, 0.04)  # Start with small radii

    # Function to calculate the penalty for constraint violations
    def penalty(x):
        centers = x[:26 * 2].reshape(26, 2)
        radii = x[26 * 2:]
        penalty = 0
        # Boundary constraint violations
        for i in range(26):
            r = radii[i]
            if centers[i, 0] - r < 0 or centers[i, 0] + r > 1 or centers[i, 1] - r < 0 or centers[i, 1] + r > 1:
                penalty += 1e6
        # Overlap constraint violations
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                distance = np.sqrt(dx * dx + dy * dy)
                if distance < radii[i] + radii[j] - 1e-12:
                    penalty += 1e6
        return penalty

    # Objective function: maximize the sum of radii (minimize negative sum)
    def objective(x):
        centers = x[:26 * 2].reshape(26, 2)
        radii = x[26 * 2:]
        return -np.sum(radii)

    # Initial guess: centers = initial_centers, radii = initial_radii
    initial_guess = np.concatenate([initial_centers.reshape(26 * 2), initial_radii])

    bounds = []

        bounds.append((0, 1))  # Center coordinates must be between 0 and 1

        bounds.append((0, 0.5))  # Radii are constrained to avoid overlap

        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': lambda x: -penalty(x)}]
    )

    # Extract the results
    centers = result.x[:26 * 2].reshape(26, 2)
    radii = result.x[26 * 2:]
    sum_radii = np.sum(radii)

    is_valid, message = validate_packing(centers, radii)
    if not is_valid:

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
