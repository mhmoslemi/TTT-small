import numpy as np
import scipy.optimize as optimize
import scipy.spatial.distance as distance

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define number of circles
    n = 26

    # Initial seed: create a hexagonal grid in the unit square
    def initial_hexagonal_grid(n):
        # Calculate number of rows and columns in hexagonal grid
        num_rows = int(np.sqrt(n))
        num_cols = num_rows
        while num_cols * (num_cols + 1) // 2 < n:
            num_rows += 1
        num_cols = num_rows
        grid_centers = []
        for i in range(num_rows):
            for j in range(num_cols):
                row = i + (j % 2) * 0.5  # Shift every other row
                x = j * (1.0 / (num_cols - 1)) if num_cols > 1 else 0.5
                y = row * (1.0 / (num_rows - 1)) if num_rows > 1 else 0.5
                grid_centers.append([x, y])
        # Adjust to fit within unit square
        grid_centers = np.array(grid_centers)
        grid_centers = grid_centers / np.max(grid_centers, axis=0)
        return grid_centers

    # Initial guess for centers
    centers = initial_hexagonal_grid(n)
    # Initial guess for radii (all equal for now)
    radii = np.full(n, 0.01)

    # Define objective function to maximize sum of radii
    def objective(x):
        # x is a vector of (centers, radii)
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:].reshape(n)
        return -np.sum(radii_flat)  # Minimize negative sum of radii

    # Define constraints
    def constraints(x):
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:].reshape(n)
        constraints = []
        for i in range(n):
            # Boundary constraint
            if centers_flat[i, 0] - radii_flat[i] < 0 or centers_flat[i, 0] + radii_flat[i] > 1:
                constraints.append(0)
            if centers_flat[i, 1] - radii_flat[i] < 0 or centers_flat[i, 1] + radii_flat[i] > 1:
                constraints.append(0)
            # Non-negative constraint
            if radii_flat[i] < 0:
                constraints.append(0)
        # Non-overlapping constraint
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    constraints.append(0)
        return constraints

    # Set up optimization
    initial_guess = np.concatenate((centers.flatten(), radii))
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])  # x and y bounds
        bounds.extend([(0.0, 1.0)])  # radius bound
    result = optimize.minimize(

        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraints}],

    # Validate packing
    valid, reason = validate_packing(final_centers, final_radii)
    if not valid:
        raise ValueError(f"Packing invalid: {reason}")

    return final_centers, final_radii, final_sum

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
