import numpy as np
import scipy.optimize as optimize
from dataclasses import dataclass

@dataclass
class Circle:
    x: float
    y: float

def generate_hexagonal_grid(num_circles, grid_rows, grid_cols):
    """Generate positions for circles in a hexagonal grid pattern."""
    radius = 1.0 / np.sqrt(3)  # Closest packing spacing
    x_positions = np.linspace(0.5 * radius, 1 - 0.5 * radius, grid_cols)
    y_positions = np.linspace(radius, 1 - radius, grid_rows)
    centers = []
    for row in range(grid_rows):
        for col in range(grid_cols):
            x = x_positions[col]
            y = y_positions[row] + (row % 2) * radius * np.sqrt(3) / 2
            centers.append((x, y))
    return np.array(centers)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26
    # Starting with a hexagonal grid of 5 rows and 5 columns (total 25 circles)
    initial_centers = generate_hexagonal_grid(num_circles, 5, 5)

    extra_circle = np.array([[0.5, 0.5]])
    initial_centers = np.vstack((initial_centers, extra_circle))
    
    # Initial radii: all equal to the distance between adjacent circles
    initial_radii = np.full(num_circles, 1.0 / np.sqrt(3))
    
    # Define the objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum
    
    # Define the constraints
    def constraint_circles_within_bounds(radii):
        return np.all((initial_centers - radii[:, np.newaxis]) >= 0)
    
    def constraint_no_overlap(radii):

            for j in range(i + 1, num_circles):
                dx = initial_centers[i, 0] - initial_centers[j, 0]
                dy = initial_centers[i, 1] - initial_centers[j, 1]
                distance = np.sqrt(dx**2 + dy**2)
                if distance < radii[i] + radii[j] - 1e-8:
                    return np.inf  # Penalize overlap
        return 0
    
    # Set up the optimization problem
    bounds = [(0, 1) for _ in range(num_circles)]
    constraints = [
        {'type': 'ineq', 'fun': constraint_circles_within_bounds},
        {'type': 'ineq', 'fun': constraint_no_overlap}
    ]

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
