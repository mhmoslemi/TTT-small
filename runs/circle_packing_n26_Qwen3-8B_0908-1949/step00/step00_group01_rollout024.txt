import numpy as np
import scipy.optimize as opt
from numpy import linalg as LA

def run_packing():
    # Initial hexagonal grid configuration with small radius
    num_circles = 26
    radius_initial = 0.05
    centers = np.zeros((num_circles, 2))
    radii = np.full(num_circles, radius_initial)

    # Place in a hexagonal grid
    row = 0
    col = 0
    for i in range(num_circles):
        x = col * (1 + np.sqrt(3) * radius_initial)
        y = row * (1 + radius_initial)
        centers[i, 0] = x
        centers[i, 1] = y
        col += 1
        if col % 2 == 0:
            row += 1
        if i % 6 == 5:
            row += 1
            col = 0

    # Define objective function (maximize sum of radii)
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum

    # Define constraints for non-overlapping and boundary
    def constraints(x):
        # x is a vector of flattened centers and radii
        centers_flat = x[:num_circles * 2].reshape(num_circles, 2)
        radii_flat = x[num_circles * 2:]
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append(dist - (radii_flat[i] + radii_flat[j]) + 1e-12)
        for i in range(num_circles):
            x_center, y_center = centers_flat[i]
            r = radii_flat[i]
            constraints.append(x_center - r + 1e-12)
            constraints.append(x_center + r - 1.0 - 1e-12)
            constraints.append(y_center - r + 1e-12)
            constraints.append(y_center + r - 1.0 - 1e-12)
        return constraints

    # Initial values for optimization
    x0 = np.concatenate([centers.flatten(), radii])

    # Use SLSQP optimization
    bounds = [(-1e-6, 1.0) for _ in range(num_circles * 2)] + [(0, 1.0) for _ in range(num_circles)]

    result = opt.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints},
        tol=1e-8
    )

    # Extract results
    centers = result.x[:num_circles * 2].reshape(num_circles, 2)
    radii = result.x[num_circles * 2:]
