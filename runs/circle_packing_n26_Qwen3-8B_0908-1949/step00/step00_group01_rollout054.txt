import numpy as np
import scipy.optimize as opt

def run_packing():

    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Simple initial distribution: grid-like with some offset and randomness
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col / 4.0 + np.random.uniform(-0.1, 0.1)
        y = row / 4.0 + np.random.uniform(-0.1, 0.1)
        centers[i] = np.array([x, y])
        radii[i] = 0.035

    def objective(x):
        # x has shape (26*3, ) = (centers_x, centers_y, radii)
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:].reshape(n)
        # Objective: maximize sum of radii
        obj = -np.sum(radii_flat)  # negative for minimization

        # Penalize overlaps and boundary violations
        penalty = 0.0
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            # Boundary constraint
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                penalty += 1e4 * (max(0, x_i - r_i + 1e-12) + max(0, x_i + r_i - 1 - 1e-12)
                                 + max(0, y_i - r_i + 1e-12) + max(0, y_i + r_i - 1 - 1e-12))
            # Overlap constraint

                dx = centers_flat[i][0] - centers_flat[j][0]

                    # Quadratic penalty for overlap
                    penalty += 1e4 * (dist - (radii_flat[i] + radii_flat[j] - 1e-12)) ** 2

        return obj + penalty

    # Initial guess for x
    x0 = np.concatenate([centers.flatten(), radii])

    # Optimization settings
    bounds = []
    for _ in range(2*n):  # centers x and y
        bounds.append((0, 1))
    for _ in range(n):  # radii
        bounds.append((0, 1))

    # Use L-BFGS-B for bounded optimization
    result = opt.minimize(

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
