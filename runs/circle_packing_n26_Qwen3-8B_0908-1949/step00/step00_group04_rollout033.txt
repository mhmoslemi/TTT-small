import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt

def run_packing():
    """
    Returns (centers, radii, sum_radii) for 26 circles in the unit square.
    """
    # Initial guess: grid-based arrangement
    n = 26
    rows = int(np.sqrt(n))
    cols = (n + rows - 1) // rows

    # 1. Initial placement: grid with circles arranged in rows and columns
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Calculate spacing
    row_spacing = 1.0 / (rows + 1)
    col_spacing = 1.0 / (cols + 1)

    for i in range(n):
        row = i // cols
        col = i % cols
        x = col * col_spacing + col_spacing / 2
        y = row * row_spacing + row_spacing / 2
        centers[i] = [x, y]

    # Objective function to maximize sum of radii
    def objective(x):
        # x contains all centers and radii: x[:2*n] are centers, x[2*n:] are radii
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:]
        total_radii = np.sum(radii_flat)
        return -total_radii  # negate for minimization

    # Constraints:
    # 1. All circles must lie within [0, 1]^2
    # 2. Distances between circles must be at least sum of their radii
    # 3. All radii non-negative

    def constraint_center(x):
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:]
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if not (0 - 1e-12 <= x_i - r_i and x_i + r_i <= 1 + 1e-12 and
                    0 - 1e-12 <= y_i - r_i and y_i + r_i <= 1 + 1e-12):
                return np.inf
        return 0

    def constraint_overlap(x):
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:]
        for i in range(n):
            for j in range(i + 1, n):
                x_i, y_i = centers_flat[i]
                x_j, y_j = centers_flat[j]
                r_i = radii_flat[i]
                r_j = radii_flat[j]
                dist = np.hypot(x_i - x_j, y_i - y_j)
                if dist < r_i + r_j - 1e-12:
                    return np.inf
        return 0

    def constraint_non_negative(x):
        radii_flat = x[2*n:]
        if np.any(radii_flat < 0 - 1e-12):
            return np.inf
        return 0

    x0 = np.concatenate([centers.flatten(), np.ones(n) * 0.02])

    bounds = []

        bounds.append((0.0, 1.0))  # x-coordinate for center i
        bounds.append((0.0, 1.0))  # y-coordinate for center i
    for i in range(n):
        bounds.append((0.0, 1.0))  # radius for circle i

        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_non_negative}
    ]

    result = opt.minimize(

        constraints=cons,

    print(f"Optimized packing: sum of radii = {sum_radii:.4f}")
    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
