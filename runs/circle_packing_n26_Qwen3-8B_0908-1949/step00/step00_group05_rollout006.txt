import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define number of circles
    n = 26

    # Initial guess for centers using a hexagonal grid
    # We'll place the centers in a hexagonal grid within the unit square
    # to guide the optimization
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initial guess for radii: assume all circles have the same radius
    # This is just a rough starting point and will be optimized
    initial_radius = 0.08
    radii[:] = initial_radius

    # Define the objective function to maximize the sum of radii
    # We'll use the negative of the total area as the objective to minimize
    def objective(x):
        # x is a flat array of [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
        x_coords = x[::3]
        y_coords = x[1::3]
        radii_vals = x[2::3]
        total_area = np.sum(radii_vals)
        return -total_area  # Minimize the negative sum of radii

    # Define constraints
    # Box constraint: each circle is within [0,1]^2
    def box_constraint(x):
        x_coords = x[::3]
        y_coords = x[1::3]
        radii_vals = x[2::3]
        for i in range(n):
            if x_coords[i] - radii_vals[i] < -1e-12 or x_coords[i] + radii_vals[i] > 1 + 1e-12:
                return np.inf
            if y_coords[i] - radii_vals[i] < -1e-12 or y_coords[i] + radii_vals[i] > 1 + 1e-12:
                return np.inf
        return 0

    # Pairwise non-overlapping constraint
    def non_overlap_constraint(x):
        x_coords = x[::3]
        y_coords = x[1::3]
        radii_vals = x[2::3]
        for i in range(n):
            for j in range(i + 1, n):
                dx = x_coords[i] - x_coords[j]
                dy = y_coords[i] - y_coords[j]
                distance = np.sqrt(dx*dx + dy*dy)
                if distance < radii_vals[i] + radii_vals[j] - 1e-12:
                    return np.inf
        return 0

    # Combine all constraints
    constraints = (
        {'type': 'ineq', 'fun': box_constraint},

    x0 = np.empty(3 * n)

    # Perform the optimization
    result = optimize.minimize(
