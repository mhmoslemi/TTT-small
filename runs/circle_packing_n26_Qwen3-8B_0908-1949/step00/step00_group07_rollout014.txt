import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define the number of circles
    n = 26

    # Initial guess: hexagonal grid with equal radii
    grid_size = np.sqrt(n)
    x_coords = np.linspace(0.5, 1 - 0.5, int(grid_size))
    y_coords = np.linspace(0.5, 1 - 0.5, int(grid_size))
    centers = []
    for i in range(len(x_coords)):
        for j in range(len(y_coords)):
            centers.append([x_coords[i], y_coords[j]])

    # Flatten and convert to numpy array
    centers = np.array(centers)
    radii = np.full(n, 0.1)  # Initial small radius

    # Define the optimization problem
    def objective(radii_vec):
        return -np.sum(radii_vec)  # Minimize negative sum (i.e., maximize sum)

    # Define constraints: non-overlap and boundary placement
    def constraint_non_overlap(centers, radii_vec):
        for i in range(len(radii_vec)):
            for j in range(i + 1, len(radii_vec)):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii_vec[i] + radii_vec[j] - 1e-12:
                    return False
        return True

    def constraint_boundary(centers, radii_vec):
        for i in range(len(radii_vec)):
            x, y = centers[i]
            r = radii_vec[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
        return True

    # Create bounds for radii to be non-negative
    bounds = [(0.0, 0.5) for _ in range(n)]

    # Constraints in scipy format
    def constraint_wrapper(x):
        radii_vec = x
        if not constraint_non_overlap(centers, radii_vec):
            return np.inf
        if not constraint_boundary(centers, radii_vec):
            return np.inf
        return 0

    # Run the optimization
    result = optimize.minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        method='SLSQP',
        constraints={'type': 'ineq', 'fun': constraint_wrapper}
    )

    # Extract the optimal centers and radii

    optimal_radii = result.x

    # Validate the packing

    if not valid:
        raise ValueError(f"Packing validation failed: {msg}")

    sum_radii = np.sum(optimal_radii)

    return optimal_centers, optimal_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
