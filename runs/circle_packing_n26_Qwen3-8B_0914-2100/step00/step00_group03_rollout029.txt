from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    spacing = 0.25  # Spacing between centers to allow for some radius

    centers = []

        for j in range(cols):
            x = j * spacing

            centers.append([x, y])
    centers = np.array(centers)

    # Radii initialization: all with radius 0.05 (to be optimized)
    radii = np.full(len(centers), 0.05)

    # Optimization variables: radii as decision variables
    # Wrap into a function for optimization

    def objective(radii_vec):
        return -np.sum(radii_vec)  # Minimize negative sum (i.e., maximize sum)

    def constraint_overlap(i, j):
        def cons(radii_vec):
            r_i = radii_vec[i]
            r_j = radii_vec[j]
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            return dist - (radii_vec[i] + radii_vec[j]) + 1e-8
        return cons

    def constraint_boundary(i):
        def cons(radii_vec):
            r = radii_vec[i]
            x, y = centers[i]
            return (x - r) - 0.0  # Ensure circle does not go beyond left boundary
        return cons

    def constraint_boundary_right(i):

            r = radii_vec[i]
            x, y = centers[i]
            return (1.0 + 1e-8) - (x + r)  # Ensure circle does not go beyond right boundary
