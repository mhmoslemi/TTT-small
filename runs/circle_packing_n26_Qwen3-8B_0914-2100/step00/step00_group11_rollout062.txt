import numpy as np
from scipy.optimize import minimize
from itertools import combinations

def run_packing():

    num_rows = 5
    num_cols = 5
    total = num_rows * num_cols

    horizontal_spacing = 0.45
    vertical_spacing = 0.45 * np.sqrt(3) / 2  # Hexagonal grid spacing

    # Generate initial hexagonal grid
    centers = []
    for row in range(num_rows):
        for col in range(num_cols):
            x = col * horizontal_spacing
            y = row * vertical_spacing
            centers.append([x, y])

    centers = np.array(centers)
    radii = np.full_like(centers, 0.1)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

            r = radii[i]
            if not (0 <= x - r and x + r <= 1 and 0 <= y - r and y + r <= 1):
                return -1
        return 0

    def constraint_overlap(radii):

            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-8:
                return -1
        return 0

    # Optimization problem

    initial_guess = np.full(len(centers), 0.1)

    ]

    # Run optimization
    result = minimize(
