import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance
from scipy.optimize import Bounds

def run_packing():
    # Initial guess for circle centers
    # Use a hexagonal packing with 5 rows, 5 columns, with one circle at the bottom
    # This is a tunable seed; will be refined during optimization
    centers = []
    radii = []
    
    # First, create a hexagonal packing with 5 rows
    for row in range(5):
        for col in range(5):
            x = col + 0.5 * (row % 2)
            y = row * np.sqrt(3) / 2
            centers.append([x, y])
    
    # Adjust to fit within unit square
    # Max x is 4.5, Max y is (5 - 1) * sqrt(3)/2 = 4 * sqrt(3)/2 = 2 * sqrt(3) ≈ 3.464
    # Scale down by factor of 3.464 to fit into [0, 1]
    scale = 1 / (2 * np.sqrt(3))
    centers = np.array(centers) * scale
    
    # Add one additional circle near the bottom right
    centers = np.vstack((centers, [0.8, 0.1]))
    radii = np.array([0.05] * len(centers))
    
    # Define bounds for centers
    center_bounds = [(0, 1) for _ in range(2 * len(centers))]
    bounds = Bounds(center_bounds, True)
    
    # Define objective function: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative because we use minimization
    
    # Define constraints: non-overlapping circles, within square
    def non_overlap(radii):
        return []
    
    # Function to compute distances and ensure non-overlap
    def nonlinear_constraints(radii, centers):
        n = len(centers)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraint = dist - (radii[i] + radii[j]) >= -1e-12
                constraints.append(constraint)
        return constraints
    
    # Function to ensure circles are within the unit square
    def square_bounds(radii, centers):
        n = len(centers)
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            constraints.append(x - r >= -1e-12)
            constraints.append(x + r <= 1 + 1e-12)
            constraints.append(y - r >= -1e-12)
            constraints.append(y + r <= 1 + 1e-12)
        return constraints
    
    # Combine all constraints
    def constraints(radii):
        non_overlap_constraints = nonlinear_constraints(radii, centers)
        square_constraints = square_bounds(radii, centers)
        return non_overlap_constraints + square_constraints
    
    # Set up optimization problem
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 500}
    )
    
    # Extract results
    optimized_radii = result.x
    optimized_centers = centers
    
    # Ensure radii are non-negative
    optimized_radii[optimized_radii < 0] = 0
    
    # Validate the packing
    valid, msg = validate_packing(optimized_centers, optimized_radii)
    if not valid:
        raise ValueError(f"Packing validation failed: {msg}")
    
    sum_radii = np.sum(optimized_radii)
    
    return (optimized_centers, optimized_radii, sum_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
