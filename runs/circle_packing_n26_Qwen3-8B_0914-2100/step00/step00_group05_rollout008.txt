import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial grid-based packing for 26 circles
    # 5 rows and 5 columns with some offset to fit 26
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Set up initial positions in a grid
    for i in range(num_circles):
        row = i // 5
        col = i % 5
        centers[i, 0] = col / 4.0
        centers[i, 1] = row / 4.0

    # Objective function to maximize total radius
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize sum

    # Constraints: non-overlapping and within unit square
    cons = []

        for j in range(i + 1, num_circles):
            def constraint_overlap(x):
                idx = np.arange(num_circles)
                r_i = x[idx == i]
                r_j = x[idx == j]
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx**2 + dy**2
                if dist_sq < 1e-12:
                    return -1e6
                return np.sqrt(dist_sq) - r_i - r_j
            cons.append({'type': 'ineq', 'fun': constraint_overlap})

    for i in range(num_circles):
        def constraint_boundary(x, i):
            r = x[i]
            dx = np.abs(centers[i, 0]) - r

            return np.minimum(dx, dy)

    # Bounds: radii must be non-negative
    bounds = [(0, 1.0) for _ in range(num_circles)]

    # Run optimization
    result = opt.minimize(
