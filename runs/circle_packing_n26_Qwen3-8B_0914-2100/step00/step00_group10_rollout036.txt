def run_packing():
    # Hexagonal grid with 2 rows and 13 columns
    n_rows = 2
    n_cols = 13
    total_circles = n_rows * n_cols
    assert total_circles == 26, "Mismatch in number of circles"

    # Grid spacing: horizontal and vertical
    d = 1.0 / (n_cols - 1)  # horizontal spacing
    h = (np.sqrt(3) / 2) * d  # vertical spacing between rows

    # Create grid of circle centers
    centers = np.zeros((total_circles, 2))
    for i in range(n_rows):
        for j in range(n_cols):
            col = j / (n_cols - 1)  # x-coordinate
            row = i * h / 1.0  # y-coordinate
            centers[i * n_cols + j] = [col, row]

    # Compute the maximum radius for which there is no overlap
    # Assume all circles are the same size to simplify
    # The radius is constrained by the minimal distance between circles and boundary

    # Consider the minimal distance from circles to the boundary
    # Minimal distance is the minimum of:
    # - distance from circle i to the left/right edge
    # - distance from circle i to the top/bottom edge
    # - distance to neighboring circles (horizontal, diagonal)

    def get_radius(centers):
        min_dist = float('inf')
        for i in range(total_circles):
            x, y = centers[i]
            # Distance to left/right boundary
            left_dist = x
            right_dist = 1 - x
            # Distance to top/bottom boundary
            top_dist = 1 - y
            bottom_dist = y
            min_dist = min(min_dist, left_dist, right_dist, top_dist, bottom_dist)
        return min_dist

    # Starting radius based on minimal distance
    r = get_radius(centers)

    # Compute the maximum possible radius by taking the minimum of:
    # - distance to corners (diagonal distance)
    # - distance to neighboring circles (horizontal, vertical, diagonal)

    def get_max_radius(centers):
        min_dist = float('inf')
        for i in range(total_circles):
            x, y = centers[i]
            # Distance to corners
            dist_left_top = np.sqrt(x**2 + y**2)
            dist_right_top = np.sqrt((1 - x)**2 + y**2)
            dist_left_bottom = np.sqrt(x**2 + (1 - y)**2)
            dist_right_bottom = np.sqrt((1 - x)**2 + (1 - y)**2)
            min_dist = min(min_dist, dist_left_top, dist_right_top, dist_left_bottom, dist_right_bottom)

            # Distance to neighboring circles
            if i < total_circles - 1:
                dist = np.linalg.norm(centers[i] - centers[i + 1])
                min_dist = min(min_dist, dist)
            if i < n_cols - 1:

                    dist = np.linalg.norm(centers[i] - centers[j])

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
