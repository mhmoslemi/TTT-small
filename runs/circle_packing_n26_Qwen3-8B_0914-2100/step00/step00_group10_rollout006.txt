    # Row 5: 6 circles, horizontal with spacing adjusted
    row5 = np.array([[0.1, 0.9], [0.25, 0.9], [0.4, 0.9], [0.55, 0.9], [0.7, 0.9], [0.85, 0.9]])

    # Combine all rows into a single list of initial centers
    centers = np.vstack([row1, row2, row3, row4, row5])

    radii = np.zeros(26)
    for i in range(26):
        if i < 20:  # First 20 circles

    def objective(radii, centers, n):
        # Objective: maximize sum of radii
        return -np.sum(radii)

    def constraints(radii, centers, n):
        # Constraints:
        # - Circle radii > 0
        # - No overlap between circles
        # - Circle inside the square

        constraints = []
        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i]})

        for i in range(n):
            for j in range(i + 1, n):
                def constraint_func(r):
                    x1, y1 = centers[i]
                    x2, y2 = centers[j]
                    dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                    return dist - (r[i] + r[j])
                constraints.append({'type': 'ineq', 'fun': constraint_func})

        for i in range(n):

                r_i = r[i]
                if x - r_i < -1e-12 or x + r_i > 1 + 1e-12 or y - r_i < -1e-12 or y + r_i > 1 + 1e-12:
