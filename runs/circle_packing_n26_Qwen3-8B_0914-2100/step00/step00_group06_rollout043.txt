import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial grid-based layout with 26 points
    num_circles = 26
    grid_size = int(np.sqrt(num_circles))
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Create a grid that roughly spaces out the circles
    for i in range(num_circles):
        row = i // grid_size
        col = i % grid_size
        x = col / (grid_size - 1)
        y = row / (grid_size - 1)
        centers[i] = [x, y]

    # Function to calculate objective with constraints
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize

    def constraint_distances(radii):
        # Constraints: circles cannot overlap
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    def constraint_boundary(radii):
        # Constraints: circles must be inside [0, 1]^2
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e8:

        method='SLSQP',
        bounds=bounds,
        constraints=cons
