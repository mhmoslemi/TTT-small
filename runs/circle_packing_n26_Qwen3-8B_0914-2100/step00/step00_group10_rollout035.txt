```python
import numpy as np
import scipy.optimize as optimize

def run_packing():

    # Approximate packing density for hexagonal packing is ~0.9069

    # Start with a base hexagonal grid of 26 circles
    rows = 5
    cols = [5, 6, 6, 6, 5]  # Number of circles in each row (alternates)

    # Calculate positions for a hexagonal grid
    centers = []
    for i in range(rows):
        row = []
        for j in range(cols[i]):
            x = j + 0.5 * i
            y = i * np.sqrt(3) / 2
            row.append((x, y))
        centers.extend(row)

    # Scale positions to fit within [0, 1]
    x_coords = np.array([c[0] for c in centers])
    y_coords = np.array([c[1] for c in centers])
    x_min, x_max = np.min(x_coords), np.max(x_coords)
    y_min, y_max = np.min(y_coords), np.max(y_coords)
    scale = 1 / max(x_max - x_min, y_max - y_min)
    centers = [(x * scale, y * scale) for x, y in centers]

    # We will now optimize the radii while maintaining spacing and boundaries

    # Define variables for optimization
    n = len(centers)
    radii = np.full(n, 0.01)  # Initial small radius for optimization

    # Define the constraints for optimization
    def constraint_dist(i, j):
        def cons(x):
            # x is a vector of radii
            r_i = x[i]
            r_j = x[j]
            dist = np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
            return dist - (r_i + r_j)
        return cons

    # Set up the optimization constraints
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': constraint_dist(i, j)})

    # Add constraint for staying within the unit square
    def boundary_constraint(x):
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = x[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return -1
        return 0

    cons.append({'type': 'ineq', 'fun': boundary_constraint})

    # Set up the objective function to maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Negative because we minimize

    # Optimize the radii
    result = optimize.minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        constraints=cons,
        bounds=[(0, 1) for _ in range(n)]
    )

    radii = result.x
    sum_radii = np.sum(radii)
    centers = np.array(centers)

    # Validate the result
    is_valid, msg = validate_packing(centers, radii)
    if not is_valid:
        raise ValueError(f"Invalid packing: {msg}")

    return centers, radii, sum_radii
```

```python
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
