import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import warnings

def run_packing():
    # Initial arrangement: 7 columns, 4 rows with one circle removed
    positions = []
    for row in range(4):
        for col in range(7):
            x = col / 6.0
            y = row / 3.0
            positions.append([x, y])
    # Remove one circle from the middle to improve spacing
    positions.pop(2)
    centers = np.array(positions)
    n = len(centers)

    # Initial guess for radii
    radii = np.full(n, 0.1)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize total radii

    # Constraints
    def constraint_overlap(radii):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return np.sum(radii)

    def constraint_boundary(radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return -np.inf
        return np.sum(radii)

    # Set bounds for radii
    bounds = [(0, 1)] * n

    # Set up optimization
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}
        ],
        method='SLSQP'
    )

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    centers = np.array(positions)
    radii = result.x

    # Final validation
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-8 or x + r > 1 + 1e-8
            or y - r < -1e-8 or y + r > 1 + 1e-8):
            raise RuntimeError(f"Circle {i} is outside the unit square")

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-8:
                raise RuntimeError(f"Circles {i} and {j} overlap")

    total_sum = np.sum(radii)
    return (centers, radii, total_sum)
