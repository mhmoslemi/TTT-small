from scipy.optimize import minimize
from scipy.optimize.slsqp import StructuredOptimizationProblem

def generate_initial_hex_grid(n_circles, radius):
    """
    Generate initial hexagonal grid of circles with given radius.
    Returns a list of (x, y) centers.
    """
    rows = int(np.ceil(np.sqrt(n_circles)))
    cols = int(np.ceil(n_circles / rows))
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * 2 * radius
            y = i * radius * np.sqrt(3)
            if i % 2 == 1:
                x += radius
            if x > 1 - radius or y > 1 - radius:
                continue
            centers.append((x, y))
    return centers

            return False

        return True

    def objective(radii):

    # Define constraints for validity
    def constraint_func(x):
        return is_valid(x, centers)

    # Create optimization problem
    problem = StructuredOptimizationProblem(

    )

    result = minimize(
        fun=objective,
