import numpy as np
import cvxpy as cp

def run_packing():
    # Define the number of circles
    n = 26

    # Define variable for centers
    centers = cp.Variable((n, 2))

    # Define variable for radii
    radii = cp.Variable(n)

    # Objective: Maximize the sum of radii
    obj = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Non-negativity of radii
    constraints.append(radii >= 0)

    # Boundary constraint: each circle must be within the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r >= -1e-12)  # x - r >= 0
        constraints.append(x + r <= 1 + 1e-12)  # x + r <= 1
        constraints.append(y - r >= -1e-12)  # y - r >= 0
        constraints.append(y + r <= 1 + 1e-12)  # y + r <= 1

    # Non-overlapping constraint: distance between centers >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Create and solve the problem
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS)

    # Extract the results
    centers_np = centers.value
    radii_np = radii.value
    sum_radii = np.sum(radii_np)

    return (centers_np, radii_np, sum_radii)
