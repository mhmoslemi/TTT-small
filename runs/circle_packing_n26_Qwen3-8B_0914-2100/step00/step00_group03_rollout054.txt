    centers = []

    cols = 6

    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum

    # Define constraints
    def constraint_overlap(radii, centers):
        n = len(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx * dx + dy * dy)

                    return 1.0
        return 0.0

    def constraint_boundary(radii, centers):

    # Constraints: non-overlapping and within square

        {"type": "ineq", "fun": constraint_overlap, "args": (centers,)},
