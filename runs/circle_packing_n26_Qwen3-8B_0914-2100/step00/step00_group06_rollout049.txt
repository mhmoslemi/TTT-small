from scipy.optimize import Bounds

def run_packing():
    # Define initial centers assuming square packing
    num_circles = 26

    num_layers = 4
    num_per_layer = 6  # 6 circles per layer (for a total of 4 layers)

    initial_centers = np.zeros((num_circles, 2))
    for i in range(num_layers):

    bounds = Bounds(0, 1)

        return -np.sum(x)  # Minimize the negative sum to maximize the sum
