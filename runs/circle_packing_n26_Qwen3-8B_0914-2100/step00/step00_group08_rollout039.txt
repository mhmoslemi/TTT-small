import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial grid-based placement: 2 rows of 13 circles
    num_circles = 26
    n_rows, n_cols = 2, 13
    spacing = 1.0 / (n_cols - 1)

    # Initialize centers with a grid-like hexagonal pattern
    centers = np.zeros((num_circles, 2))
    for i in range(n_rows):
        for j in range(n_cols):
            x = j * spacing
            y = i * spacing + (i % 2) * spacing / 2  # Hexagonal offset
            centers[i * n_cols + j] = [x, y]

    # Initial radius based on spacing
    initial_radius = spacing / 2

    # Function to compute radius of each circle
    def compute_radii(centers, spacing):
        radii = np.zeros_like(centers[:, 0])
        for i in range(len(centers)):
            x, y = centers[i]
            # Initial radius based on spacing from nearest boundary
            r = min(x, 1 - x, y, 1 - y)
            # Adjust radius based on distance to neighbors
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                r = min(r, (dist - 1e-8) / 2)  # Ensure non-overlapping
            radii[i] = r
        return radii

    # Compute initial radii
    radii = compute_radii(centers, spacing)

    # Function to compute total sum of radii
    def total_sum(radii):
        return np.sum(radii)

    # Function to check constraints (boundary and overlap)
    def check_constraints(centers, radii):
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Optimization: Maximize total sum of radii
    result = optimize.minimize(
        fun=lambda x: -total_sum(x),
        x0=radii,
        bounds=[(0, 1)] * num_circles,
        constraints=[{
            'type': 'ineq',
            'fun': lambda x: check_constraints(centers, x)
        }],
        method='SLSQP'
    )

    radii_opt = result.x
    sum_radii = total_sum(radii_opt)

    return centers, radii_opt, sum_radii
