import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Grid-based initial packing
    def grid_packing(n_circles: int = 26):
        # Create an initial grid of points
        rows = int(np.sqrt(n_circles))
        cols = n_circles // rows if n_circles % rows == 0 else n_circles // rows + 1
        grid = np.zeros((n_circles, 2))
        for i in range(n_circles):
            row = i // cols
            col = i % cols
            grid[i, 0] = col / (cols - 1)
            grid[i, 1] = row / (rows - 1)
        return grid

    # Initial configuration
    centers = grid_packing()
    n = len(centers)
    
    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Nonlinear constraints: circle-circle and circle-boundary distance constraints
    def constraints(radii):
        constraints = []
        for i in range(n):
            # Boundary distance constraint: ensure circle lies within [0,1]^2
            x, y = centers[i]
            r = radii[i]
            # Left/right boundary
            if x - r < -1e-12:
                constraints.append((x - r + 1e-12) * 100)  # Penalize hard boundaries
            if x + r > 1 + 1e-12:
                constraints.append((x + r - 1 - 1e-12) * 100)
            # Top/bottom boundary
            if y - r < -1e-12:
                constraints.append((y - r + 1e-12) * 100)
            if y + r > 1 + 1e-12:
                constraints.append((y + r - 1 - 1e-12) * 100)
        
        # Circle-circle overlap constraint
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    constraints.append((dist - (radii[i] + radii[j] - 1e-12)) * 100)
        
        return np.array(constraints)

    # Bounds on radii: must be non-negative
    bounds = [(0, 1) for _ in range(n)]

    # Set up optimization
    res = opt.minimize(
        fun=objective,
        x0=np.ones(n),
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraints},

    )

    radii = res.x

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
