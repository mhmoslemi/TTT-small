import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial guess for centers: 2 circles per corner (8 total), 6 circles on edges, and 12 in center
    n = 26

    # Define corner circles: 2 per corner
    corners = [
        (0.1, 0.1), (0.9, 0.1), (0.1, 0.9), (0.9, 0.9),
        (0.1, 0.1), (0.9, 0.1), (0.1, 0.9), (0.9, 0.9)
    ]

    # Define edge circles: evenly spaced along the four sides
    edge_centers = []
    for x in np.linspace(0.1, 0.9, 5):
        edge_centers.append((x, 0.1))
        edge_centers.append((x, 0.9))
    for y in np.linspace(0.1, 0.9, 5):
        edge_centers.append((0.1, y))
        edge_centers.append((0.9, y))

    # Remove duplicates and ensure count is 6
    edge_centers = list(set(edge_centers))
    edge_centers = [tuple(c) for c in edge_centers]
    edge_centers = edge_centers[:6]  # Remove duplicate edge points

    # Define inner circles: a 3x3 grid in center
    inner_centers = [(0.3, 0.3), (0.5, 0.3), (0.7, 0.3),
                     (0.3, 0.5), (0.5, 0.5), (0.7, 0.5),
                     (0.3, 0.7), (0.5, 0.7), (0.7, 0.7)]

    # Combine all centers (4 corners + 6 edges + 12 inner)
    centers = np.array(corners + edge_centers + inner_centers)

    # Initial radii (arbitrary starting values)
    radii = np.random.uniform(0.01, 0.05, n)

    # Define objective function
    def objective(x):
        return -np.sum(x)

    # Define constraints
    def constraint_boundary(x, i):
        r = x[n + i]
        x_c, y_c = centers[i]
        return x_c - r >= -1e-12 and x_c + r <= 1 + 1e-12 and y_c - r >= -1e-12 and y_c + r <= 1 + 1e-12

    def constraint_overlap(x, i, j):
        r_i = x[n + i]
        r_j = x[n + j]
        dx = centers[i][0] - centers[j][0]
        dy = centers[i][1] - centers[j][1]
        dist = np.sqrt(dx**2 + dy**2)
        return dist >= r_i + r_j - 1e-12

    # Prepare variables
    x = np.zeros(2 * n)
    x[:n] = centers.flatten()
    x[n:] = radii

    # Build bounds
    bounds = []
    for i in range(n):
        bounds.append((0, 1))
        bounds.append((0, 1))

    # Build constraints
    cons = []
    for i in range(n):
        cons.append(('ineq', lambda x, i=i: constraint_boundary(x, i)))
    for i in range(n):
        for j in range(i + 1, n):
            cons.append(('ineq', lambda x, i=i, j=j: constraint_overlap(x, i, j)))

    # Run optimization
    result = optimize.minimize(
        objective,
        x,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-12, 'maxiter': 1000}
    )

    # Extract final centers and radii
    centers_final = result.x[:n].reshape(n, 2)
    radii_final = result.x[n:]

    # Compute sum of radii
    sum_radii = np.sum(radii_final)

    return (centers_final, radii_final, sum_radii)
