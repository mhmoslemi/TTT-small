import numpy as np
import cvxpy as cp
import math

def run_packing():
    # Initial grid-based packing with 6 rows and 5 columns (26 circles)
    n = 26
    cols = 5
    rows = 6
    spacing = 1.0 / (cols + 1)
    radius_initial = 0.05 * np.random.rand(n)

    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + spacing / 2
            y = i * spacing + spacing / 2
            centers.append([x, y])
    centers = np.array(centers)

    # Set up optimization variables
    radii = cp.Variable(n)
    obj = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []
    constraints.append(radii >= 0)

    for i in range(n):
        # Ensure circle i is within [0,1]^2
        x, y = centers[i]
        constraints.append(x - radii[i] >= -1e-12)
        constraints.append(x + radii[i] <= 1 + 1e-12)
        constraints.append(y - radii[i] >= -1e-12)
        constraints.append(y + radii[i] <= 1 + 1e-12)

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Solve the optimization
    problem = cp.Problem(obj, constraints)

    try:
        problem.solve(solver=cp.SCS, eps=1e-5)
    except:
        raise Exception("Optimization failed")

    # Extract results
    centers = np.array(centers)
    radii = np.array(radii.value)
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
