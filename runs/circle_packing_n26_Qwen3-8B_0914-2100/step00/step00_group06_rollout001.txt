```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Define initial layout: hexagonal packing with 5 rows
    num_circles = 26
    rows = 5
    cols = 5
    spacing = 0.5  # Adjust spacing to fit in unit square

    centers = []
    for row in range(rows):
        for col in range(cols):
            x = col * spacing
            y = row * (spacing * np.sqrt(3) / 2)
            if row % 2 == 1:
                x += spacing / 2
            centers.append([x, y])
    
    # Convert to numpy array
    centers = np.array(centers)
    radii = np.full(num_circles, 0.03)  # Initial small radius

    # Define objective function to maximize sum of radii
    def objective(x):
        # x is [flat array of centers and radii]
        flat = np.reshape(x, (-1, 2))
        centers_flat = flat[:, :2]
        radii_flat = flat[:, 2:]
        
        # Compute distances between all pairs of circles
        dists = cdist(centers_flat, centers_flat)
        overlap = (dists < radii_flat[:, np.newaxis] + radii_flat[np.newaxis, :]) | (dists < 1e-8)
        # Penalize overlap
        penalty = np.sum(overlap * 1e4)
        # Penalize going out of bounds
        bounds_penalty = np.sum(np.maximum(centers_flat[:, 0] - radii_flat, 0) + 
                               np.maximum(1 - centers_flat[:, 0] - radii_flat, 0) +
                               np.maximum(centers_flat[:, 1] - radii_flat, 0) +
                               np.maximum(1 - centers_flat[:, 1] - radii_flat, 0))
        
        return - (np.sum(radii_flat) + penalty + bounds_penalty)

    # Flatten centers and radii for optimization
    initial_flat = np.concatenate([centers.flatten(), radii.flatten()])
    
    # Optimization parameters
    bounds = []
    for i in range(num_circles):
        # x and y coordinates must be in [0,1]
        bounds.extend([(0, 1), (0, 1)])
        # Radius must be non-negative
        bounds.extend([(0, 1)])  # We limit radius to avoid going too large initially
    
    result = minimize(objective, initial_flat, bounds=bounds, method='L-BFGS-B', tol=1e-8, options={'maxiter': 500})
    
    # Extract results
    flat_result = result.x
    centers_opt = flat_result[:num_circles*2].reshape(-1, 2)
    radii_opt = flat_result[num_circles*2:].reshape(-1)
    
    # Validate final packing
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")

    sum_radii = np.sum(radii_opt)
    return (centers_opt, radii_opt, sum_radii)
```

```python
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
