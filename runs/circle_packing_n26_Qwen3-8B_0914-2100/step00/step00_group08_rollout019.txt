import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():

    n = 26
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        centers[i, 0] = col / 4.5
        centers[i, 1] = row / 4.5

    # Initial radii estimation: take the minimum distance to the boundary
    radii = np.zeros(n)

        radii[i] = dist_to_boundary / 2

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraints: non-overlapping and within boundaries
    def constraint_overlap(radii, centers):

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    def constraint_boundary(radii, centers):
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
        return True

    # Define bounds for radii
    bounds = [(0, 1) for _ in range(n)]

    # Optimization problem
    result = minimize(

        x0=radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': lambda r: constraint_overlap(r, centers)},
            {'type': 'ineq', 'fun': lambda r: constraint_boundary(r, centers)}
        ],
        method='L-BFGS-B'
    )

    radii = result.x
