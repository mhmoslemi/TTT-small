import scipy.optimize as optimize
from scipy.optimize import Bounds, NonlinearConstraint

def run_packing():
    # Initial layout: Hexagonal grid with 26 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05  # Initial guess for radii

    # Define the hexagonal grid: m rows, with increasing number of circles per row
    # Adjust rows and columns to fit 26 circles
    rows = 6
    cols = 5
    spacing = 1.0 / np.sqrt(3)

    # Hexagonal grid with rows alternated in offset
    for i in range(rows):
        for j in range(cols):
            x = j + i * 0.5
            y = i * spacing
            if i % 2 == 1:
                x += 0.5
            centers[i * cols + j] = [x, y]

    # Scale centers to fit within the unit square
    centers /= max(centers.max(), 1.0)

    # Define objective function: maximize sum of radii
    def objective(x, radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Define constraints: non-overlapping and within unit square
    def constraint_overlap(x, radii):
        # Check if all circles are within [0, 1] x [0, 1]
        in_bounds = np.all((x >= 1e-12) & (x <= 1 - 1e-12), axis=1)
        if not np.all(in_bounds):
            return -1  # Penalize for out-of-bounds
        # Check if any pair of circles overlaps
        for i in range(len(radii)):

                dist_sq = dx**2 + dy**2
                if dist_sq < (radii[i] + radii[j])**2 - 1e-12:

        return 0

    # Optimization bounds: x in [0, 1], y in [0, 1]
    bounds = Bounds(np.zeros((n, 2)), np.ones((n, 2)))

    def constraint_func(x):
