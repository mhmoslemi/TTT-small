import numpy as np
import scipy.optimize as optimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Initial grid-based packing
    num_circles = 26
    rows = 5
    cols = 6
    spacing = 0.25  # spacing between circles
    initial_centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + (spacing / 2)
            y = i * spacing + (spacing / 2)
            if i == 0 or i == rows - 1 or j == 0 or j == cols - 1:
                x = j * spacing
                y = i * spacing
            initial_centers.append([x, y])
    initial_centers = np.array(initial_centers)
    initial_radii = np.full(num_circles, 0.05)  # initial guess for radii

    # Optimization function to maximize sum of radii
    def objective(radii, centers):
        return -np.sum(radii)  # Use negative for maximization

    def constraints(radii, centers):
        # Ensure circles lie within the unit square
        for i in range(num_circles):
            if not (0 - 1e-12 <= centers[i, 0] - radii[i] <= 1 + 1e-12 and
                    0 - 1e-12 <= centers[i, 1] - radii[i] <= 1 + 1e-12):
                return False
        # Ensure no overlaps
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    bounds = [(0.0, 1.0) for _ in range(num_circles)]  # Radii must be non-negative

    # Optimization
    result = optimize.minimize(
        lambda x: objective(x, initial_centers),
        x0=initial_radii,
        bounds=bounds,
        constraints=optimize.LinearConstraint(
            A=np.eye(num_circles),
            lb=np.zeros(num_circles),
            ub=np.ones(num_circles)
        ),
        method="SLSQP"
    )

    radii = result.x

    valid, message = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Invalid packing: {message}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
