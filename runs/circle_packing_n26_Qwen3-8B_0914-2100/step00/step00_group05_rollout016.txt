import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d
from itertools import product

def run_packing():
    # Initial guess: place circles in a hexagonal grid with spacing
    num_circles = 26
    # Initial spacing to ensure non-overlapping
    initial_spacing = 0.25

    # Create a grid of points in the square
    x_coords = np.linspace(0, 1, 6)
    y_coords = np.linspace(0, 1, 6)
    grid_points = np.array(list(product(x_coords, y_coords)))

    # Convert to hexagonal packing
    centers = []
    for i in range(num_circles):
        col = i // 3
        row = i % 3
        x = col * initial_spacing + (row % 2) * initial_spacing / 2
        y = row * initial_spacing + (row % 2) * initial_spacing / 2
        # Map to [0,1] square
        x = x / 1.5
        y = y / 1.5
        centers.append([x, y])
    centers = np.array(centers)

    # Initial radii: set to spacing / 2 for non-overlap
    radii = np.full(num_circles, initial_spacing / 2)

    # Define helper function for optimization
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii, so minimize -sum

    def constraint_overlap(i, j):
        def cons(x):
            r_i = x[i]
            r_j = x[j]
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            return dist - (r_i + r_j)
        return cons

    # Set up constraints for non-overlapping
    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i, j=j: np.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2) - (x[i] + x[j])
            })

    # Set up constraints for boundary conditions
    for i in range(num_circles):
        def boundary_constraint(x, i=i):
            r = x[i]
            x_center = x[i, 0]
            y_center = x[i, 1]
            return x_center - r >= -1e-8 and x_center + r <= 1 + 1e-8 and y_center - r >= -1e-8 and y_center + r <= 1 + 1e8

        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[i, 0] - x[i] >= -1e-8
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[i, 0] + x[i] <= 1 + 1e-8
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[i, 1] - x[i] >= -1e-8
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[i, 1] + x[i] <= 1 + 1e-8
        })

    # Optimization variables: radii
    # The optimization variables are the radii, and the centers are fixed
    # So we pass the radii as the variables, and the centers are separate

    # Wrap the radii into a single array for optimization
    radii_array = radii.copy()

    # Define the optimization problem
    bounds = [(0, 1) for _ in range(num_circles)]

    result = minimize(
        objective,
        radii_array,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints
    )

    # Extract the best radii
    best_radii = result.x
    # Create centers with small perturbations to avoid exact overlaps
    perturb = 0.01
    centers_with_perturbation = centers.copy()
    for i in range(num_circles):
        centers_with_perturbation[i, 0] += np.random.uniform(-perturb, perturb)
        centers_with_perturbation[i, 1] += np.random.uniform(-perturb, perturb)

    # Final check for overlaps (optional, but recommended)
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dx = centers_with_perturbation[i, 0] - centers_with_perturbation[j, 0]
            dy = centers_with_perturbation[i, 1] - centers_with_perturbation[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < best_radii[i] + best_radii[j] - 1e-6:
                # Adjust radii to prevent overlap
                delta = (best_radii[i] + best_radii[j]) - dist - 1e-6
                best_radii[i] -= delta / 2
                best_radii[j] -= delta / 2

    return centers_with_perturbation, best_radii, np.sum(best_radii)
