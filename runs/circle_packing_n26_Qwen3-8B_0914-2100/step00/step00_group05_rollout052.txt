import scipy.optimize as optimize

def run_packing():

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize the sum

    # Constraints: Ensure circles do not overlap, and stay within square
    def constraint_overlap(centers, radii):
        n = len(centers)
        for i in range(n):

                x1, y1 = centers[i]
                x2, y2 = centers[j]
                dx = x1 - x2
                dy = y1 - y2
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    def constraint_bounds(centers, radii):
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1.0 + 1e-8 or y - r < -1e-8 or y + r > 1.0 + 1e-8:
                return False
        return True

    # Optimization setup
    bounds = [(0.0, 1.0) for _ in range(n * 2)]  # Bounds for x and y coordinates

        return [constraint_overlap(centers_flat, radii_flat), constraint_bounds(centers_flat, radii_flat)]
