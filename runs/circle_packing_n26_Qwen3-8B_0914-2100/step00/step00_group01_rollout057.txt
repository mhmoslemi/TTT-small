import cvxpy as cp

def run_packing():

    num_circles = 26
    # We use a hexagonal grid with reduced spacing to allow tighter packing
    # Experimental spacing factor for the grid
    grid_spacing = 1.5
    # We start placing circles with radius = 0.04 (initial guess)
    init_radius = 0.04

    # Start with a hexagonal grid (first row starts from bottom)
    centers = []
    radii = []

    for i in range(13):
        x = i * grid_spacing

        centers.append((x, y))
        radii.append(init_radius)

        centers.append((x, y))
        radii.append(init_radius)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Optimization problem
    # Variables
    r = cp.Variable(num_circles)

    # Objective function: Maximize sum of radii
    obj = cp.Maximize(cp.sum(r))

    # Constraints
    # All circles must be inside the unit square
    constraints = []
    for i in range(num_circles):
        x, y = centers[i]

        constraints.append(x + r[i] <= 1 + 1e-10)

    # Create and solve the problem
    problem = cp.Problem(obj, constraints)
    problem.solve(solver=cp.ECOS)

    final_radii = r.value

    # Return the result
    return final_centers, final_radii, np.sum(final_radii)
