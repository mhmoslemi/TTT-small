import numpy as np
import scipy.optimize as optimize

def run_packing():
    def objective(x):
        return -np.sum(x[1:])  # Maximize sum of radii

    def constraint_boundary(x):
        for i in range(26):
            x_i, y_i = x[2*i], x[2*i+1]
            r_i = x[26 + i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return -1
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -1
        return 0

    def constraint_overlap(x):
        for i in range(26):
            for j in range(i+1, 26):
                dx = x[2*i] - x[2*j]
                dy = x[2*i+1] - x[2*j+1]
                dist = np.sqrt(dx**2 + dy**2)
                r_i = x[26 + i]
                r_j = x[26 + j]
                if dist < r_i + r_j - 1e-12:
                    return -1
        return 0

    # Initial positions: small circles at corners and edges, large ones in the center
    # 8 small circles at corners (0.02 radius)
    corners = np.array([
        [0.02, 0.02], [0.98, 0.02], [0.98, 0.98], [0.02, 0.98]
    ])
    # 4 small circles on edges (0.02 radius)
    edge_centers = np.array([
        [0.02, 0.5], [0.5, 0.02], [0.98, 0.5], [0.5, 0.98]
    ])
    # 14 circles packed in hexagonal pattern near the center
    hex_centers = []
    for r in range(4):
        for c in range(7):
            x = 0.5 + (c - 3.5) * 0.2
            y = 0.5 + (r - 1.5) * 0.2
            if 0 <= x <= 1 and 0 <= y <= 1:
                hex_centers.append([x, y])

    # Flatten all centers
    centers = np.vstack((corners, edge_centers, hex_centers))
    # Initialize radii: 0.02 for corners and edges, and 0.1 for hexagonal ones
    radii = np.array([0.02]*4 + [0.02]*4 + [0.1]*14)

    # Flatten into a single vector for optimization
    x_initial = np.concatenate((centers.flatten(), radii))

    # Bounds: x and y coordinates must be in [0, 1], and radii must be non-negative
    bounds = []
    for _ in range(52):  # 26 circles * 2 coordinates
        bounds.append((0, 1))
    for _ in range(26):  # 26 radii
        bounds.append((0, 1))

    # Constraints: boundary and non-overlap
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Optimization
    result = optimize.minimize(
        objective,
        x_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    # Extract final centers and radii
    x_final = result.x
    centers = x_final[:52].reshape(26, 2)
    radii = x_final[52:]

    # Validate packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Packing validation failed: {msg}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
