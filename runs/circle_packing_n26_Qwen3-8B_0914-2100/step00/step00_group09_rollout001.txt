import random
from deap import base, creator, algorithms, tools

def run_packing():
    # Define the configuration for the genetic algorithm
    random.seed(42)  # For reproducibility

    # Define the problem
    creator.create("FitnessMax", base.Fitness, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMax)

    toolbox = base.Toolbox()

    # Define the individual as a tuple of (x, y, r) for each circle
    def random_individual():
        centers = np.random.rand(26, 2)
        # Ensure circles are slightly inside the unit square
        centers += np.random.uniform(-0.05, 0.05, size=(26, 2))
        radii = np.random.uniform(0.01, 0.1, size=26)
        return (centers, radii)

    toolbox.register("individual", tools.initIterate, list, random_individual)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    def evaluate(individual):
        centers, radii = individual
        # Validate the packing
        valid, message = validate_packing(centers, radii)
        if not valid:
            return 0.0,  # Invalid packing
        return float(np.sum(radii)),  # Fitness is the sum of radii

    toolbox.register("evaluate", evaluate)

    # Genetic operators: crossover and mutation
    def cx_multi_point(ind1, ind2):
        # Multi-point crossover
        ind1_centers, ind1_radii = ind1
        ind2_centers, ind2_radii = ind2
        new_centers = np.zeros((26, 2))
        new_radii = np.zeros(26)
        for i in range(26):

                new_centers[i] = ind1_centers[i]
                new_radii[i] = ind1_radi
            else:
                new_centers[i] = ind2_centers[i]
                new_radii[i] = ind2_radii[i]
        return (new_centers, new_radii), (ind1_centers, ind1_radii)

    def mut_gaussian(individual):
        # Gaussian mutation on the centers and radii
        centers, radii = individual
        for i in range(26):
            if random.random() < 0.1:

    toolbox.register("mutate", mut_gaussian)

    pop = toolbox.population(n=50)

    # Run the genetic algorithm
    algorithms.eaSimple(pop, toolbox, cxpb=0.5, mutpb=0.1, ngen=100, verbose=False)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
