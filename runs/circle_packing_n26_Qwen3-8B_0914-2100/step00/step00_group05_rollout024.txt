        radii = np.zeros(n)
        for i in range(n):
            angle = 2 * np.pi / n * i
            r = 0.1 + 0.05 * i
            x = 0.5 + r * np.cos(angle)
            y = 0.5 + r * np.sin(angle)

            radii[i] = r

    def constraint_non_overlap(centers, radii):

        dists = squareform(pdist(centers))

                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        return True
    
    def constraint_boundary(centers, radii):
        n = len(centers)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
        return True
    
    # Optimization
    constraints = [

        {'type': 'ineq', 'fun': lambda r: -r},

    ]
    
    result = opt.minimize(
        objective,

        method='SLSQP',
        bounds=[(0, 1)] * 26,

        raise ValueError("Optimization failed")

    final_radii = result.x
