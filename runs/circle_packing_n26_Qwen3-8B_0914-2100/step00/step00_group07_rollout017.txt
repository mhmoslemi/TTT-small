import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial configuration: 2 rows of 13 circles
    # Total of 26 circles

    # Initial guess: grid-based placing
    centers = np.zeros((26, 2))
    for i in range(26):
        row = i // 13
        col = i % 13
        centers[i, 0] = col / (13 - 1)  # x-coordinate
        centers[i, 1] = row / (2 - 1)  # y-coordinate

    # We'll now perform optimization to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints(radii):
        # All radii must be non-negative
        for r in radii:
            if r < 0:
                return False
        return True

    result = optimize.minimize(
        fun=objective,
        x0=np.ones(26),
        constraints={"type": "ineq", "fun": constraints},
        method="Nelder-Mead"
    )

    # Extract optimal radii
    radii = result.x

    # Check if optimization was successful
    if not result.success:
        raise ValueError("Optimization failed with message: {}".format(result.message))

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
