import numpy as np

def run_packing():
    # Parameters
    num_circles = 26
    spacing = 0.5  # Spacing between centers of adjacent circles

    # Hexagonal grid setup for square packing
    max_rows = 5  # Approximate number of rows that fit in the unit square
    max_cols = 5  # Approximate number of columns that fit in the unit square

    # Initialize centers
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Define the offset for alternating row alignment
    offset = spacing / 2

    current_row = 0
    current_col = 0

    for i in range(num_circles):
        # Calculate x and y based on hexagonal grid
        # Offset rows for staggered grid
        if i % 2 == 0:
            y = current_row * spacing
        else:
            y = current_row * spacing + offset
        x = current_col * spacing

        # Update the current row and column for next circle
        if i % 2 == 0:
            # Even rows: add next column
            if current_col + 1 < max_cols:
                current_col += 1
            else:
                # Move to next row
                current_row += 1
                current_col = 0
        else:
            # Odd rows: next column in the same row
            if current_col + 1 < max_cols:
                current_col += 1
            else:
                # Move to next row
                current_row += 1
                current_col = 0

        # Set center position
        centers[i, 0] = x
        centers[i, 1] = y

        # Calculate maximum possible radius
        # Find the minimum of:
        # - Distance to boundary (left, right, top, bottom)
        # - Distance to next circle (based on spacing)

        # Distance to left boundary
        left_dist = x
        # Distance to right boundary
        right_dist = 1 - x
        # Distance to bottom boundary
        bottom_dist = y
        # Distance to top boundary
        top_dist = 1 - y

        # Distance to right neighbor
        if i % 2 == 0:
            # Even rows, check to the next column
            if current_col + 1 < max_cols:
                right_neighbor_col = current_col + 1
                if current_row % 2 == 0:
                    # Same row
                    right_neighbor_y = y
                else:
                    # Shifted row
                    right_neighbor_y = y + offset
                right_neighbor_x = (current_col + 1) * spacing
                dist_to_right = np.sqrt((right_neighbor_x - x) ** 2 + (right_neighbor_y - y) ** 2)
            else:
                dist_to_right = float('inf')
        else:
            # Odd rows, check to the previous column
            if current_col - 1 >= 0:
                left_neighbor_col = current_col - 1

                    # Same row

                else:
                    # Shifted row

                left_neighbor_x = (current_col - 1) * spacing

                dist_to_right = float('inf')

        # Find the minimum distance to the boundaries or neighbors
        min_dist = np.min([left_dist, right_dist, bottom_dist, top_dist, dist_to_right])

    if not valid:

    sum_radii = np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
