import numpy as np
import random
from typing import Tuple, List, Union

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    def initial_grid_packing(n: int) -> Tuple[np.ndarray, np.ndarray]:
        grid_size = int(np.sqrt(n))
        if grid_size * grid_size < n:
            grid_size += 1
        max_radius = 0.5 / (grid_size + 1)
        centers = []
        radii = []
        for i in range(grid_size):
            for j in range(grid_size):
                x = i / (grid_size)
                y = j / (grid_size)
                centers.append([x, y])
                radii.append(max_radius)
        return np.array(centers), np.array(radii)

    def perturb_circle(centers: np.ndarray, radii: np.ndarray, index: int) -> Tuple[np.ndarray, np.ndarray]:
        x, y = centers[index]
        r = radii[index]
        dx = random.uniform(-0.01, 0.01)
        dy = random.uniform(-0.01, 0.01)
        new_x = x + dx
        new_y = y + dy
        new_r = r * (1 + random.uniform(-0.1, 0.1))
        new_r = max(new_r, 0.0)
        return np.array([[new_x, new_y]]), np.array([new_r])

    def validate_packing(centers: np.ndarray, radii: np.ndarray) -> bool:
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    def generate_initial_packing() -> Tuple[np.ndarray, np.ndarray]:
        centers, radii = initial_grid_packing(26)
        return centers, radii

    def run_reinforcement_learning(max_steps: int = 1000) -> Tuple[np.ndarray, np.ndarray]:
        centers, radii = generate_initial_packing()
        best_centers = centers.copy()
        best_radii = radii.copy()
        best_sum = np.sum(radii)

            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers_perturbed
                best_radii = radii_perturbed

        return best_centers, best_radii

    centers, radii = run_reinforcement_learning()
    total_radius = np.sum(radii)
    return centers, radii, total_radius
