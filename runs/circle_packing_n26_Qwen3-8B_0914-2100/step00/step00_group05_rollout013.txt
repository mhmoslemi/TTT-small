from scipy.optimize import minimize

    columns = 6

    horizontal_separation = 1.0 / (columns + 0.5)  # Horizontal spacing

    radii = []

    radii = np.array(radii)

    def within_bounds(x, y, r):

        return 0

    def non_overlap_constraints(c, r):
