import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial configuration: 4 large circles at the corners
    corners = np.array([
        [0.1, 0.1],
        [0.1, 0.9],
        [0.9, 0.9],
        [0.9, 0.1]
    ])

    # Place 5 equidistant circles along the horizontal and vertical axes
    horizontal = np.array([
        [0.2, 0.5],
        [0.4, 0.5],
        [0.6, 0.5],
        [0.8, 0.5]
    ])

    vertical = np.array([
        [0.5, 0.2],
        [0.5, 0.4],
        [0.5, 0.6],
        [0.5, 0.8]
    ])

    # Initial centers
    centers = np.vstack((corners, horizontal, vertical))

    # Initial guesses for radii (small circles around the center, larger at corners)
    radii = np.array([0.2] * len(centers))

    # Optimization problem setup
    def objective(x):
        return -np.sum(x)  # Maximize sum of radii

    def constraint_boundary(x, i):
        x_i, y_i = centers[i]
        r_i = x[i]
        if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
            return -1  # Violation of boundary constraint
        return 0

    def constraint_overlap(x, i, j):
        x_i, y_i = centers[i]
        x_j, y_j = centers[j]
        r_i = x[i]
        r_j = x[j]
        dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
        if dist < r_i + r_j - 1e-12:
            return -1  # Violation of non-overlap constraint
        return 0

    bounds = [(0, 1) for _ in range(len(centers))]

    constraints = []
    for i in range(len(centers)):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary(x, i)})
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint_overlap(x, i, j)})

    result = optimize.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints
    )

    # Extract the optimized radii and centers
    optimized_radii = result.x
    sum_radii = np.sum(optimized_radii)

    return (centers, optimized_radii, sum_radii)
