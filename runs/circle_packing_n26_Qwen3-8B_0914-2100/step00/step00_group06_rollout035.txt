import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial grid parameters
    n = 26
    grid_x = np.linspace(0.1, 0.9, 5)
    grid_y = np.linspace(0.1, 0.9, 5)

    # Define starting centers
    centers = []
    for i in range(5):
        for j in range(5):
            x = grid_x[i]
            y = grid_y[j]
            centers.append([x, y])
    
    # Try to fit a 26th circle in the center
    centers.append([0.5, 0.5])
    
    # Initial radii: all same size (for optimization)
    radii0 = [0.15] * n

    # Objective: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize

    # Constraint: ensure all circles are within bounds
    def bounds_func(radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
        return True

    # Constraint: ensure no circles overlap
    def overlap_func(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Combine bounds and constraints
    def constraint_func(radii):
        if not bounds_func(radii):
            return -np.inf
        if not overlap_func(radii):
            return -np.inf
        return 0

    # Run optimization
    bounds = [(0.0, 1.0)] * n
    result = optimize.minimize(
        objective,
        radii0,
        bounds=bounds,
        method='SLSQP',
        constraints={'type': 'ineq', 'fun': constraint_func}
    )

    if not result.success:
        raise RuntimeError("Optimization failed: " + result.message)

    radii = result.x
    sum_radii = np.sum(radii)
    return np.array(centers), radii, sum_radii
