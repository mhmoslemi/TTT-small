import numpy as np
import scipy.optimize as optimize
import random
import math

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def initial_greedy_packing(n_circles):
    circles = []
    for _ in range(n_circles):
        # Randomly choose a position and radius within the square
        while True:
            x = np.random.uniform(0.1, 0.9)
            y = np.random.uniform(0.1, 0.9)
            r = np.random.uniform(0.01, 0.15)
            # Check if it overlaps with existing circles
            overlap = False
            for (cx, cy, cr) in circles:
                dx = x - cx
                dy = y - cy
                dist_sq = dx**2 + dy**2
                if dist_sq < (cr + r)**2 - 1e-12:
                    overlap = True
                    break
            if not overlap:
                circles.append((x, y, r))
                break
    centers = np.array([ (x, y) for x, y, _ in circles ])
    radii = np.array([ r for _, _, r in circles ])
    return centers, radii

def define_objective(centers, radii):
    # We aim to maximize sum of radii, so we minimize -sum(radii)
    return -np.sum(radii)

def run_packing():

    centers, radii = initial_greedy_packing(n_circles)

    valid, msg = validate_packing(centers, radii)

        raise ValueError(f"Initial packing invalid: {msg}")

    # Step 2: Simulated Annealing to refine
    # We'll use a custom annealing function to perturb and update circles

    # Create a cost function to minimize
    def objective(x):
        # x is flattened: first n_circles*2 entries are centers, rest are radii
        centers_flat = x[:n_circles*2].reshape((n_circles, 2))

            return np.inf  # Penalize invalid configurations

    # Initial guess: centers and radii from initial placement
    initial_x = np.concatenate([
        centers.flatten(),

    # Run simulated annealing
    result = optimize.basinhopping(
