        if row % 2 == 1:
            x += 0.25 / 2
        centers.append((x, y))

    # Calculate minimum distance to the boundary
    def min_dist_to_edge(x, y):
        return min(x, 1 - x, y, 1 - y)

    def distance(p1, p2):
        return np.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)

        radii.append(r)
