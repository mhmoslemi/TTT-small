import cvxpy as cp
import math

    cols = 6

    # Step 2: Use optimization for improved packing
    # Define variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n)

    # Objective: Maximize the sum of radii
    obj = cp.Maximize(cp.sum(r))

    # Constraints:
    # 1. Non-overlap constraints: ||x_i - x_j|| >= r_i + r_j for all i < j
    # 2. Boundary constraints: for each circle, r_i <= min(x_i[0], 1 - x_i[0], x_i[1], 1 - x_i[1])

    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.norm(x[i] - x[j])
            constraints.append(dist >= r[i] + r[j] - 1e-6)

    # Solve the optimization
    prob = cp.Problem(obj, constraints)

    centers = np.round(x.value, 5)

    for i in range(n):
