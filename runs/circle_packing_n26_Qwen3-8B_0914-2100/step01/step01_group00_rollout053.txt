def run_packing():

    n = 26

    # First, define the grid structure
    # The pattern will be as follows:
    # Layer 0: center circle
    # Layer 1: 6 surrounding circles
    # Layer 2: 12 circles
    # Layer 3: 18 circles

    # Define the radius of the central circle
    r_center = 0.1

    # Generate the centers
    centers = []
    # Central circle
    centers.append((0.5, 0.5))

    angles = np.linspace(0, 2 * np.pi, 6, endpoint=False)
    for angle in angles:
        x = 0.5 + r_center * np.cos(angle)
        y = 0.5 + r_center * np.sin(angle)
        centers.append((x, y))

    # Layer 2: 12 circles
    r_layer2 = 0.2
    for i in range(6):
        angle = angles[i] + np.pi / 2
        x = 0.5 + r_layer2 * np.cos(angle)
        y = 0.5 + r_layer2 * np.sin(angle)
        centers.append((x, y))

    # Layer 3: 18 circles
    r_layer3 = 0.25
    for i in range(6):

        x = 0.5 + r_layer3 * np.cos(angle)
        y = 0.5 + r_layer3 * np.sin(angle)
        centers.append((x, y))

    # Compute radii
    radii = []
    for i in range(n):
        # Compute distance to the nearest boundary
        x, y = centers[i]
        dist_to_boundary = min(x, 1 - x, y, 1 - y)
