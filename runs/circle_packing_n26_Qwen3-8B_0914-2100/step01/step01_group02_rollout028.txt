    for row in range(grid_rows):
        for col in range(grid_cols):

                # Even rows are aligned to the left
                x = col * grid_spacing + 0.05
            else:
                # Odd rows are offset by half the spacing
                x = (col + 0.5) * grid_spacing + 0.05
            y = row * grid_spacing + 0.05
            centers.append((x, y))

    # Add small circles in unused gaps
    additional_centers = [

        (0.15, 0.15),

    ]

    # Now compute radii for each circle
    radii = []
    for i in range(len(centers)):
        x, y = centers[i]
        # Calculate the minimum distance to the boundaries
        min_dist_to_edge = min(x, 1 - x, y, 1 - y)
        # Calculate the minimum distance to other circles
        min_dist_to_neighbor = float('inf')
