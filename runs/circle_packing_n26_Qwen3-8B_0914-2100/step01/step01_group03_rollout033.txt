import numpy as np

def run_packing():
    # Hexagonal grid parameters
    num_rows = 5
    num_cols = 5
    row_spacing = 0.6 / (num_rows - 1)  # vertical spacing between rows
    col_spacing = 0.6 / (num_cols - 1)  # horizontal spacing between columns

    # Generate a hexagonal grid
    centers = []
    for row in range(num_rows):
        for col in range(num_cols):
            x = col * col_spacing + 0.1  # Bottom-left offset to avoid boundary issues
            y = row * row_spacing + 0.1
            centers.append((x, y))
    
    # Adjust for corner circles
    corner_circles = [
        (0.1, 0.1),
        (0.9, 0.1),
        (0.1, 0.9),
        (0.9, 0.9)
    ]
    centers += corner_circles

    # Ensure we have exactly 26 circles
    if len(centers) > 26:
        centers = centers[:26]
    elif len(centers) < 26:
        # Fill remaining spots with circles evenly distributed in the remaining area
        remaining = 26 - len(centers)
        for i in range(remaining):
            # Distribute evenly in the remaining area
            x = np.random.uniform(0.1, 0.9)
            y = np.random.uniform(0.1, 0.9)
            centers.append((x, y))

    # Compute radii: smallest distance to boundary or nearest neighbor
    radii = []
    for i in range(len(centers)):
        x, y = centers[i]
        
        # Distance to nearest boundary
        dist_to_boundary = min(x, 1 - x, y, 1 - y)
        
        # Distance to nearest neighbor
        min_dist_to_neighbor = float('inf')
        for j in range(len(centers)):
            if i != j:
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < min_dist_to_neighbor:
                    min_dist_to_neighbor = dist
        
        # Radius is half the minimum of these two distances
        r = min(dist_to_boundary, min_dist_to_neighbor / 2)
        radii.append(r)

    centers = np.array(centers)
    radii = np.array(radii)
    total_sum = np.sum(radii)

    return centers, radii, total_sum
