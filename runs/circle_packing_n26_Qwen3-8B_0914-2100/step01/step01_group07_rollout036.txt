    # This is a dynamic grid where circle spacing increases as we move toward the boundaries

    def generate_hexagonal_grid(max_radius=0.15):

        points = []

                x = r * np.cos(angle)
                y = r * np.sin(angle)
                points.append((x, y))
        points = np.array(points)
        points = points / np.max(np.abs(points))  # Normalize to [0,1]^2
        return points

        [0.5, 0.05],
        [0.5, 0.95],
        [0.05, 0.5],
        [0.95, 0.5],
    ])
    initial_points = np.vstack((initial_points, corner_points))

    unique_points = np.unique(initial_points, axis=0)

                continue

            dy = centers[i, 1] - centers[j, 1]
            dist = np.hypot(dx, dy)
