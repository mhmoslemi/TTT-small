import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Define initial approximate positions in hexagonal packing
    num_circles = 26
    positions = []
    for i in range(num_circles):
        row = i // 5
        col = i % 5
        x = 0.2 + col * 0.25
        y = 0.2 + row * 0.25
        positions.append((x, y))
    
    # Initial guess for radii
    radii = np.full(num_circles, 0.05)

    # Objective function: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraints: non-overlap and boundary conditions
    def constraint_non_overlap(radii):
        def cons(x):
            r = x[:num_circles]
            # Test non-overlap
            distances = cdist(positions, positions)
            for i in range(num_circles):
                for j in range(i + 1, num_circles):
                    if distances[i, j] < r[i] + r[j] - 1e-8:
                        return 1
            return 0
        return {'type': 'ineq', 'fun': cons}

    def constraint_boundary(radii):
        def cons(x):
            r = x[:num_circles]
            for i in range(num_circles):
                x_i, y_i = positions[i]
                if x_i - r[i] < 0 or x_i + r[i] > 1 or y_i - r[i] < 0 or y_i + r[i] > 1:
                    return 1
            return 0
        return {'type': 'ineq', 'fun': cons}

    # Optimization
    x0 = np.concatenate((np.array(positions).flatten(), radii))
    cons = [constraint_non_overlap(radii), constraint_boundary(radii)]
    result = minimize(objective, x0, constraints=cons)

    # Extract final positions and radii
    final_positions = result.x[:2 * num_circles].reshape((num_circles, 2))
    final_radii = result.x[2 * num_circles:]

    # Validate that all radii are non-negative
    if np.any(final_radii < 0):
        raise ValueError("Negative radii found")

    total_sum = np.sum(final_radii)
    return final_positions, final_radii, total_sum
