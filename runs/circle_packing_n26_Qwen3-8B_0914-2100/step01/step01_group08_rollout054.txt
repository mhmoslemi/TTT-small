import math
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    radii = np.random.rand(26) * 0.1  # Initial small radii for optimization

    # Helper function to compute total "energy" to minimize
    def objective(x: np.ndarray) -> float:
        # x is flattened array of [x0, y0, r0, x1, y1, r1, ...]
        # Reconstruct centers and radii
        x = x.reshape((26, 4))  # [x, y, r] for each of 26 circles
        centers = x[:, :2]
        radii = x[:, 2]

        total_energy = 0.0
        for i in range(26):
            # Ensure circle stays within the unit square
            x_i, y_i = centers[i]
            r_i = radii[i]
            if (x_i - r_i < -1e-6) or (x_i + r_i > 1 + 1e-6) or (y_i - r_i < -1e-6) or (y_i + r_i > 1 + 1e-6):
                total_energy += 10000.0  # Penalize for going outside
            # Penalize for small radii
            if r_i < 1e-6:
                total_energy += 10000.0

        # Check for overlaps
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx**2 + dy**2)
                r_sum = radii[i] + radii[j]
                if dist < r_sum - 1e-6:
                    total_energy += 10000.0  # Penalize overlaps

        return total_energy

        initial_guess[i * 4 : (i + 1) * 4] = [i / 6, 0.5, 0.1, 0.1]  # [x, y, r, dx] — dummy

    # Apply optimization

    centers = result.x.reshape((26, 2))
