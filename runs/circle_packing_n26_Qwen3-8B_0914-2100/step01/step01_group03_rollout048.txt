import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Define the number of circles
    n = 26

    # Create an initial guess for the circle centers using a hexagonal grid pattern
    # Start with a hexagonal grid of 6 rows (odd number for better packing)
    grid_centers = []
    for row in range(6):
        for col in range(6):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            grid_centers.append((x, y))
    
    # Convert to numpy array for easier manipulation
    grid_centers = np.array(grid_centers)
    grid_centers = (grid_centers - grid_centers.min(axis=0)) / (grid_centers.max(axis=0) - grid_centers.min(axis=0))
    
    # Add two more circles in the bottom-left and top-right corners
    grid_centers = np.vstack([grid_centers, (0.1, 0.1), (0.9, 0.9)])
    grid_centers = grid_centers[:n]  # Keep only first 26 circles

    def objective(radii):
        return -np.sum(radii)  # Maximize the sum of radii

    def constraints(radii):
        # Ensure non-negativity
        if np.any(radii < 0):
            return np.inf
        
        # Check boundaries
        for i in range(n):
            x, y = grid_centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return np.inf

        # Check circle-circle distances
        distances = cdist(grid_centers, grid_centers)
        for i in range(n):
            for j in range(i + 1, n):
                if distances[i, j] < radii[i] + radii[j] - 1e-12:
                    return np.inf

        return 0  # No constraints violated

    # Initial radii: all set to 0.1 as a starting point
    initial_radii = np.full(n, 0.1)

    # Optimization
    result = minimize(
        fun=objective,
        x0=initial_radii,
        method='SLSQP',
        jac=None,
        bounds=[(0, 1) for _ in range(n)],
        constraints=[{
            'type': 'ineq',
            'fun': constraints
        }],
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )

    centers = grid_centers
    radii = result.x
    total_sum = np.sum(radii)

    return centers, radii, total_sum
