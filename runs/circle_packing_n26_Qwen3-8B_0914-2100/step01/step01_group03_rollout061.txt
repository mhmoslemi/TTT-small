from scipy.optimize import minimize
from itertools import combinations

def run_packing():

    num_circles = 26
    initial_centers = np.random.rand(num_circles, 2)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize total

    # Define constraints:
    # 1. Each circle must be inside the unit square
    # 2. Each circle must be non-overlapping with others
    def constraint_inside(radii):
        return np.array([np.min(np.array([center[0] - radii[i], 1 - center[0] - radii[i],
                                          center[1] - radii[i], 1 - center[1] - radii[i]])) for i, center in enumerate(initial_centers)])
    
    def constraint_overlap(radii):
        dists = np.zeros((num_circles, num_circles))
        for i in range(num_circles):

                dx = initial_centers[i, 0] - initial_centers[j, 0]
                dy = initial_centers[i, 1] - initial_centers[j, 1]
                dists[i, j] = np.sqrt(dx**2 + dy**2)
        return np.array([dists[i, j] - (radii[i] + radii[j]) for i, j in combinations(range(num_circles), 2)])

    def all_constraints(radii):

        overlap = constraint_overlap(radii)
