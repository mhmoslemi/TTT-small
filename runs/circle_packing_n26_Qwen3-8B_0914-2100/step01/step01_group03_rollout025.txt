from scipy.optimize import minimize

def run_packing():

    rows = 6
    columns = 5
    
    # X and Y offsets for hexagonal grid
    x_offset = 0.5
    y_offset = np.sqrt(3) / 2

    # Define grid points in hexagonal packing
    centers = []
    for i in range(rows):
        for j in range(columns):
            x = j * (1 - x_offset) + x_offset * i
            y = i * y_offset
            centers.append((x, y))

    # Add an extra circle at a strategic location
    extra_circle = (0.75, 0.75)
    centers.append(extra_circle)

    # Convert to numpy arrays
    centers = np.array(centers)
    n = len(centers)

    # Define objective function
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Define constraints
    def constraint_boundary(radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return -np.inf
        return np.sum(radii)

    def constraint_overlap(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < (radii[i] + radii[j] - 1e-12):
                    return -np.inf
        return np.sum(radii)

    # Bounds: radii >= 0
    bounds = [(0, 1) for _ in range(n)]

    # Initial guess for radii (based on minimum distance to neighbors)
    initial_radii = []
    for i in range(n):

                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < min_dist:
                    min_dist = dist
        initial_radii.append(min_dist / 2)
    initial_radii = np.array(initial_radii)

    # Optimization
    result = minimize(

        x0=initial_radii,
        bounds=bounds,
        constraints=[
