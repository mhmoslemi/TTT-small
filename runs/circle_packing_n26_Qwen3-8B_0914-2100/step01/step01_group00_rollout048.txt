def run_packing():

    def generate_hex_grid(n_rows, max_radius):

        for row in range(n_rows):
            for col in range(n_rows + 1):
                x = col * (1 + max_radius) / n_rows
                y = row * (np.sqrt(3) * max_radius) / n_rows
                if x < 1 - 1e-8 and y < 1 - 1e-8:
                    grid.append((x, y))
        return grid

    n_rows = 5
    max_radius = 0.15

    grid = generate_hex_grid(n_rows, max_radius)

        raise ValueError("Hexagonal grid does not provide enough points")

    # Compute radii
    radii = []
    for i in range(len(grid)):
        x, y = grid[i]
        # Distance to boundaries
        min_dist_to_edge = min(x, 1 - x, y, 1 - y)
        # Distance to nearest neighbor
        min_dist_to_neighbor = float('inf')
        for j in range(len(grid)):
            if i != j:
                dx = grid[i][0] - grid[j][0]
                dy = grid[i][1] - grid[j][1]
                dist = np.sqrt(dx**2 + dy**2)
