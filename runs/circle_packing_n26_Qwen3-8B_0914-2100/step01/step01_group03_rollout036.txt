import numpy as np

def run_packing():
    grid_size = 6
    grid_spacing = 0.25  # Adjust spacing for better packing
    
    # Generate a hexagonal grid of 26 circles
    centers = []
    for i in range(grid_size):
        for j in range(grid_size):
            # Create a hexagonal grid
            angle = 2 * np.pi / 3
            x = i + (j % 2) * np.cos(angle) * (grid_spacing / 2)
            y = j * np.sin(angle) * (grid_spacing / 2)
            centers.append((x, y))
    
    # Check if we have exactly 26 circles
    if len(centers) != 26:
        raise ValueError("Expected 26 circles in the grid")
    
    # Compute radii
    radii = []
    for i in range(len(centers)):
        x, y = centers[i]
        
        # Minimum distance to the boundary
        min_dist_to_edge = min(x, 1 - x, y, 1 - y)
        
        # Minimum distance to other circles
        min_dist_to_other = float('inf')
        for j in range(len(centers)):
            if i != j:
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < min_dist_to_other:
                    min_dist_to_other = dist
        
        # Radius is the minimum of the two
        r = min(min_dist_to_edge, min_dist_to_other / 2)
        radii.append(r)
    
    centers = np.array(centers)
    radii = np.array(radii)
    total_sum = np.sum(radii)
    
    return centers, radii, total_sum
