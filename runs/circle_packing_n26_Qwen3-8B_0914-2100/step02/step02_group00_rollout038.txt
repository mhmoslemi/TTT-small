import numpy as np
import math
from scipy.optimize import minimize
import cvxpy as cp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    num_circles = 26
    # Use a hexagonal grid, slightly adjusted to fit into the unit square
    # We will place circles in two rows for better dense packing

    # Initial positions for hexagonal grid
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Place circles in a hexagonal grid pattern
    for i in range(num_circles):
        row = i // 6
        col = i % 6
        x = col + 0.5 * (row % 2)
        y = row * math.sqrt(3) / 2
        rad = 0.05
        # Scale to fit in [0, 1]^2
        x = x / 4.0
        y = y / 4.0
        centers[i] = [x, y]
        radii[i] = rad

    def is_inside(x, y, r):
        return (0.0 <= x - r < 1.0 + 1e-6 and 0.0 <= y - r < 1.0 + 1e-6)

    def is_overlap(x1, y1, r1, x2, y2, r2):
        dx = x1 - x2
        dy = y1 - y2
        dist_sq = dx**2 + dy**2
        return math.sqrt(dist_sq) < r1 + r2 - 1e-6

    # Optimization variables
    variables = []
    for i in range(num_circles):
        variables.append(cp.Variable(2))  # x and y
        variables.append(cp.Variable())   # radius

    # Objective: Maximize the sum of radii
    obj = cp.Maximize(sum(variables[i*2+1] for i in range(num_circles)))

    # Constraints: circles must not overlap and stay within [0, 1]^2
    constraints = []

    for i in range(num_circles):

        constraints.append(ri <= 0.5)  # Max radius to keep feasible

        for j in range(i+1, num_circles):
            xj, yj = variables[j*2], variables[j*2+1]
            rj = variables[j*2+2]
            constraints.append( (xi - xj)**2 + (yi - yj)**2 >= (ri + rj)**2 - 1e-6 )

    # Create the problem
    prob = cp.Problem(obj, constraints)

    # Solve the problem with a high precision
    try:
        prob.solve(solver=cp.SCS, eps=1e-4)

        return np.zeros((num_circles, 2)), np.zeros(num_circles), 0.0

    # Extract optimized centers and radii
    centers_opt = np.zeros((num_circles, 2))

            y - r < -1e-6 or y + r > 1 + 1e-6):
            radii_opt[i] = 0.0

    # Compute total sum of radii
    total_sum = np.sum(radii_opt)

    return centers_opt, radii_opt, total_sum
