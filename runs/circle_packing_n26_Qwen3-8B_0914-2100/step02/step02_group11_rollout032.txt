from scipy.optimize import minimize
import itertools

    centers.append((0.05, 0.95))
    centers.append((0.05, 0.05))

    def compute_total_radii(centers, n):

        for i in range(n):
            x, y = centers[i]
            # Compute distance to the boundaries
            min_dist_to_boundary = min(x, 1 - x, y, 1 - y)
            # Compute distance to other circles
            min_dist_to_other = float('inf')
            for j in range(n):
                if i != j:
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = np.sqrt(dx**2 + dy**2)
                    if dist < min_dist_to_other:

            # Radius is the minimum of the two

            radii.append(r)
