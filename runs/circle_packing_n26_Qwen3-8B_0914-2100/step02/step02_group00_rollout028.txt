import numpy as np
import math
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal grid configuration
    def generate_hexagonal_grid(n_circles, max_diameter=1.0):
        centers = []
        for i in range(n_circles):
            row = i // 2
            col = i % 2
            x = col * (max_diameter / np.sqrt(3))
            y = row * (max_diameter / 2)
            if x > 1 or y > 1:
                continue
            centers.append([x, y])
        return np.array(centers)

    centers = generate_hexagonal_grid(26)
    radii = np.ones(26) * 0.05

    # Define helper functions
    def is_inside(x, y, r):
        return 0.0 <= x - r and x + r <= 1.0 and 0.0 <= y - r and y + r <= 1.0

    def is_overlap(x1, y1, r1, x2, y2, r2):
        dx = x1 - x2
        dy = y1 - y2
        dist_sq = dx**2 + dy**2
        return math.sqrt(dist_sq) < r1 + r2 - 1e-6

    def objective_function(radii):
        return -np.sum(radii)

    def constraints(radii):
        n = len(centers)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                if is_overlap(centers[i][0], centers[i][1], radii[i],
                              centers[j][0], centers[j][1], radii[j]):
                    constraints.append((radii[i] + radii[j] - 1e-6, 0))
        return constraints

    def check_bounds(radii):
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            if not is_inside(x, y, r):
                return False
        return True

    # Define bounds for optimization
    bounds = [(0.0, 1.0) for _ in range(len(radii))]

    # Optimization using SLSQP
    result = minimize(
        fun=objective_function,
        x0=radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    # Final radius values
    final_radii = result.x

    # Check if the solution is feasible
    if not check_bounds(final_radii):
        raise RuntimeError("Optimized configuration violates boundary constraints")

    # Return the final result
    return centers, final_radii, np.sum(final_radii)
