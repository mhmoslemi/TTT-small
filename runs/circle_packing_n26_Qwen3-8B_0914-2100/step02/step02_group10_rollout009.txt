from scipy.optimize import minimize
import random

def run_packing():

    initial_radii = np.random.rand(n) * 0.1  # Small initial estimates

    def compute_total_radii(centers, n):
        radii = []
        for i in range(n):
            x, y = centers[i]
            # Distance to boundaries
            min_dist_to_boundary = min(x, 1 - x, y, 1 - y)
            # Distance to other circles
            min_dist_to_other = float('inf')
            for j in range(n):
                if i != j:
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = np.sqrt(dx**2 + dy**2)
                    if dist < min_dist_to_other:
                        min_dist_to_other = dist
            # Radius is the minimum of the two
            r = min(min_dist_to_boundary, min_dist_to_other / 2)
            radii.append(r)
        return radii

    def objective(x):
        centers = np.reshape(x, (n, 2))
        radii = compute_total_radii(centers, n)
        return -np.sum(radii)

    def constraint(x):

        for i in range(n):
            x_center, y_center = centers[i]
            if not (0 <= x_center <= 1) or not (0 <= y_center <= 1):
                return -1
        return 0

    # Optimization arguments
    bounds = [(0, 1) for _ in range(2 * n)]
    result = minimize(

        method='L-BFGS-B',
