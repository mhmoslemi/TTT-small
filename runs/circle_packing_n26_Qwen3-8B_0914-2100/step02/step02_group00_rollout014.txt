import math
from scipy.optimize import minimize
from scipy.optimize import LinearConstraint
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    num_circles = 26

    # Initial circle positions using a more interleaved, hexagonal grid pattern
    centers = np.zeros((num_circles, 2))
    radius = np.zeros(num_circles)

    # Step 1: Create a hexagonal grid of circle positions
    # Define the number of circles in rows
    rows = int(math.sqrt(num_circles))
    cols = (num_circles + rows - 1) // rows
    spacing = 0.35  # adjust spacing for hexagonal packing
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + spacing / 2
            y = i * (spacing * math.sqrt(3)) / 2 + spacing / 2
            if x < 1 - 1e-6 and y < 1 - 1e-6:
                idx = i * cols + j
                centers[idx] = [x, y]
                radius[idx] = 0.01  # start with small radius and optimize

    # Step 2: Define objective function and constraints
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

            r1 = radii[i]

            dy = y1 - y2

            dist = math.sqrt(dist_sq)

        return constraint
