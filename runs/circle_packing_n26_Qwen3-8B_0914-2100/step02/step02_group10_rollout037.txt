    # Top right corner
    centers.append((0.95, 0.95))
    # Top left corner
    centers.append((0.05, 0.95))
    # Bottom left corner
    centers.append((0.05, 0.05))

    centers.append((0.95, 0.05))
    # Add 2 more circles in the middle top and bottom regions
    centers.append((0.5, 0.75))
    centers.append((0.5, 0.25))

    # Convert centers to a flat array for optimization
    initial_guess = np.array([x for x in itertools.chain.from_iterable(centers)])
    
    # Define a function to compute the total sum of radii given centers
    def compute_total_radii(centers, n, BOUNDARY_TOLERANCE):
        radii = []
        for i in range(n):
            x, y = centers[i]
            # Compute distance to the right boundary
            right_dist = 1.0 - x - BOUNDARY_TOLERANCE
            # Compute distance to the left boundary
            left_dist = x - BOUNDARY_TOLERANCE
            # Compute distance to the top boundary
            top_dist = 1.0 - y - BOUNDARY_TOLERANCE
            # Compute distance to the bottom boundary
            bottom_dist = y - BOUNDARY_TOLERANCE
            min_dist_to_boundary = min(right_dist, left_dist, top_dist, bottom_dist)

            # Compute distance to other circles
            min_dist_to_other = float('inf')
            for j in range(n):
                if i != j:
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = np.sqrt(dx**2 + dy**2)

            # Radius is the minimum of the two

            radii.append(r)
        return radii

    def objective(x):
