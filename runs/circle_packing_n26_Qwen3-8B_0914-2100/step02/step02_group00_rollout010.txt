    grid_spacing = 0.3
    init_radius = grid_spacing / 2

    centers = np.zeros((26, 2))
    radii = np.zeros(26)

    for col in range(grid_cols):

    N = len(centers)

    def is_inside(x, y, r):

        return (x - r >= -1e-6 and x + r <= 1 + 1e-6 and
                y - r >= -1e-6 and y + r <= 1 + 1e-6)

    def get_bounds():
        """Bound the radii between 0 and 0.5."""
        return [(0, 0.5) for _ in range(N)]

    def objective(radii):
        """Objective: Maximize the sum of radii."""
        return -np.sum(radii)  # Negate for minimization
