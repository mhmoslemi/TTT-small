import numpy as np
import random
from scipy.optimize import differential_evolution

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Initialize centers with some spread to allow optimization
    centers = np.random.rand(n, 2)
    # Set the initial radii to 0.01 for a rough guess
    radii = np.full(n, 0.01)

    # Define the fitness function for differential evolution
    def fitness(x):
        # x is an array of size 2*n (x1, y1, r1, x2, y2, r2, ... xn, yn, rn)
        # Reconstruct centers and radii
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:].reshape(n)
        # Ensure radii are non-negative
        radii = np.maximum(radii, 0.0)
        # Check if any circle is out of bounds
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if not (0.0 - 1e-6 <= x - r <= 1.0 + 1e-6 and
                   0.0 - 1e-6 <= y - r <= 1.0 + 1e-6):
                return -np.inf
        # Compute total area as a fitness function
        total_area = np.sum(np.pi * radii ** 2)
        # Compute overlaps
        overlap_penalty = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                if dist < radii[i] + radii[j] - 1e-6:
                    overlap_penalty += (radii[i] + radii[j] - dist) ** 2
        # Return negative of total area to minimize overlap
        return -total_area - overlap_penalty

    # Bounds for each variable
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 0.5)])  # x, y, r

    # Run differential evolution with a large population and high iterations
    result = differential_evolution(
        func=fitness,
        bounds=bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=100,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True
    )

    # Extract the optimal positions and radii
    solution = result.x
    centers = solution[:2*n].reshape(n, 2)
    radii = solution[2*n:].reshape(n)

    # Return the centers, radii, and total sum of radii
    total_sum = np.sum(radii)
    return centers, radii, total_sum
