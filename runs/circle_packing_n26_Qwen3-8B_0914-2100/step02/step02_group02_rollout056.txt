import numpy as np
import math
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define grid parameters for initial placement
    grid_cols = 6
    grid_rows = 5
    grid_spacing = 0.2
    initial_radius = grid_spacing / 2

    # Initialize centers
    centers = np.zeros((26, 2))
    radii = np.zeros(26)

    # Place initial grid
    counter = 0
    for col in range(grid_cols):
        for row in range(grid_rows):
            x = col * grid_spacing + grid_spacing / 2
            y = row * grid_spacing + grid_spacing / 2
            if x < 1 - 1e-6 and y < 1 - 1e-6:
                centers[counter] = [x, y]
                radii[counter] = initial_radius
                counter += 1

    # Define helper functions
    def is_inside(x, y, r):
        return (0.0 <= x - r - 1e-12 and x + r + 1e-12 <= 1.0 and
                0.0 <= y - r - 1e-12 and y + r + 1e-12 <= 1.0)

    def is_overlap(x1, y1, r1, x2, y2, r2):
        dx = x1 - x2
        dy = y1 - y2
        dist = math.sqrt(dx**2 + dy**2)
        return dist < r1 + r2 - 1e-6

    # Objective function for optimization: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraints for optimization
    def constraints(x):
        # x is the concatenated vector of centers and radii
        # Extract centers and radii
        flat_x = x
        centers_curr = flat_x[:26*2].reshape(26, 2)
        radii_curr = flat_x[26*2:]

        # Ensure all circles are within the unit square
        for i in range(26):
            x_i, y_i = centers_curr[i]
            r_i = radii_curr[i]
            if not is_inside(x_i, y_i, r_i):
                return np.array([1.0])

        # Ensure no two circles overlap
        for i in range(26):
            for j in range(i + 1, 26):
                x_i, y_i = centers_curr[i]
                r_i = radii_curr[i]
                x_j, y_j = centers_curr[j]
                r_j = radii_curr[j]
                if is_overlap(x_i, y_i, r_i, x_j, y_j, r_j):
                    return np.array([1.0])

        return np.array([0.0])

    # Flatten centers and radii for optimization
    initial_flat = np.concatenate((centers.reshape(-1), radii))

    # Run optimization
    result = minimize(
        fun=objective,
        x0=initial_flat,
        constraints={"type": "ineq", "fun": constraints},
        method="SLSQP"
    )

    # Extract results
    optimized_flat = result.x
    centers_opt = optimized_flat[:26*2].reshape(26, 2)
    radii_opt = optimized_flat[26*2:]

    # Ensure circles are within bounds
    for i in range(26):
        x, y = centers_opt[i]
        r = radii_opt[i]
        if not is_inside(x, y, r):
            radii_opt[i] = 0.0

    total_sum = np.sum(radii_opt)
    return centers_opt, radii_opt, total_sum
