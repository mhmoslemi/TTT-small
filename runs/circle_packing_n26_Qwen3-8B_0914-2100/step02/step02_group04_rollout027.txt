import numpy as np
import random
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Number of circles
    n = 26
    # Set of initial positions: hexagonal grid, random, and corner placements
    positions = []

    # Generate a hexagonal grid inside the unit square
    spacing = 0.18
    col_count, row_count = 6, 5
    for col in range(col_count):
        for row in range(row_count):
            x = col * spacing + spacing / 2.0
            y = row * spacing + spacing / 2.0
            if x > 1.0 - 1e-6 or y > 1.0 - 1e-6:
                continue
            positions.append([x, y])

    # Generate random points inside unit square
    random_points = []
    for _ in range(n - len(positions)):
        x = np.random.uniform(0.1, 0.9)
        y = np.random.uniform(0.1, 0.9)
        random_points.append([x, y])
    positions += random_points

    # Generate corner and edge-based points
    corner_points = [
        [0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]
    ]
    positions += corner_points

    # Ensure unique positions (remove duplicates)
    unique_positions = []
    seen = set()
    for p in positions:
        t = tuple(p)
        if t not in seen:
            seen.add(t)
            unique_positions.append(p)
    unique_positions = unique_positions[:n]

    # Convert to numpy array
    centers = np.array(unique_positions)

    # Define radii as variables to be optimized
    radii = np.full(n, 0.05)  # Initial guess for radii

    # Define objective function for optimization
    def objective(radii):
        # Ensure non-negativity
        radii = np.maximum(radii, 0.0)

        # Check if circles are within the square
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-6 or x + r > 1.0 + 1e-6 or y - r < -1e-6 or y + r > 1.0 + 1e-6:
                return -np.inf

        # Check if circles overlap
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                distance = math.hypot(x1 - x2, y1 - y2)
                if distance < radii[i] + radii[j] - 1e-6:
                    return -np.inf

        # Return the negative sum of radii (to maximize sum)
        return -np.sum(radii)

    # Use simulated annealing to maximize the sum of radii
    result = minimize(
        fun=objective,
        x0=radii,
        bounds=[(0, 1.0) for _ in range(n)],
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    # Extract the optimized radii
    optimized_radii = result.x
    total_sum = np.sum(optimized_radii)

    return centers, optimized_radii, total_sum
