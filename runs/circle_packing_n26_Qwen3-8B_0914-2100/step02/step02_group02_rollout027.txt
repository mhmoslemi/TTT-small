def is_inside(x, y, eps=1e-6):
    return 0.0 - eps <= x - eps and x + eps <= 1.0 + eps and 0.0 - eps <= y - eps and y + eps <= 1.0 + eps

def is_overlap(x1, y1, r1, x2, y2, r2, eps=1e-6):
    dx = x1 - x2
    dy = y1 - y2
    dist_sq = dx**2 + dy**2
    return math.sqrt(dist_sq) < r1 + r2 - eps
