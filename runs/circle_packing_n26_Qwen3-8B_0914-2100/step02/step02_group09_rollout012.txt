import numpy as np
from scipy.optimize import minimize
import itertools
import random

def run_packing():
    n = 26
    # Hybrid initial setup: 
    # 1. 5 rows of circles in a hexagonal fashion (even and odd rows offset)
    # 2. 1 additional circle placed in a corner
    # 3. 1 circle added in between to fill a gap (can be optimized to get better total area)
    # 4. 1 circle placed in center to maximize space usage

    # Create a grid with 5 rows and 6 columns in a hexagonal pattern
    # with a spacing of 0.3 between adjacent circles
    centers = []

    for i in range(5):
        for j in range(6):
            if i % 2 == 0:
                # Even rows are aligned to the left
                x = j * 0.3 + 0.1
                y = i * np.sqrt(3) / 2 * 0.3 + 0.1
            else:
                # Odd rows are offset horizontally by half the horizontal spacing
                x = (j + 0.5) * 0.3 + 0.1
                y = i * np.sqrt(3) / 2 * 0.3 + 0.1
            centers.append((x, y))
    
    # Add the 26th circle near the top-right corner
    centers.append((0.9, 0.9))
    
    # Define a function to compute total radii given centers
    def compute_total_radii(centers, n):
        # Initial estimate of radii based on the minimum distance to surrounding circles and boundaries
        radii = []
        for i in range(n):
            x, y = centers[i]
            # Compute distance to the boundary
            min_dist_to_boundary = min(x, 1 - x, y, 1 - y)
            # Compute distance to other circles
            min_dist_to_other = float('inf')
            for j in range(n):
                if i != j:
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = np.sqrt(dx**2 + dy**2)
                    if dist < min_dist_to_other:
                        min_dist_to_other = dist
            # Radius is the minimum of the two
            r = min(min_dist_to_boundary, min_dist_to_other / 2)
            radii.append(r)
        return radii
    
    # Define the objective function to maximize the sum of radii
    def objective(x):
        # x is a flat array of all centers (flattened), with shape (2n,)
        # Convert to centers (n, 2)
        centers = np.reshape(x, (n, 2))
        radii = compute_total_radii(centers, n)
        # Return negative of sum of radii to minimize (since scipy.optimize.minimize minimizes)
        return -np.sum(radii)
    
    # Define constraints: all circles must be within [0,1]^2
    def constraint(x):
        centers = np.reshape(x, (n, 2))
        for i in range(n):
            x_center, y_center = centers[i]
            if not (0 <= x_center <= 1) or not (0 <= y_center <= 1):
                return -1  # Violate the constraint
        return 0
    
    # Initial guess is the grid centers
    initial_guess = np.array([x for x in itertools.chain.from_iterable(centers)])
    
    # Perform optimization
    result = minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(2 * n)],
        constraints={'type': 'ineq', 'fun': constraint}
    )
    
    # Extract the optimal centers and radii
    centers = np.reshape(result.x, (n, 2))
    radii = compute_total_radii(centers, n)
    
    total_sum = np.sum(radii)
    
    return centers, radii, total_sum
