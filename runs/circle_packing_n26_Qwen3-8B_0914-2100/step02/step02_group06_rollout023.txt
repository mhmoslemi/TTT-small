# genetic_algorithm.py
import numpy as np
import random
import math
from typing import Callable

class GeneticAlgorithm:
    def __init__(
        self,
        population_size: int,
        generations: int,
        mutation_rate: float,
        crossover_rate: float,
        n_cities: int,
        bounds: list,
        objective_function: Callable,
        fitness_function: Callable
    ) -> None:
        self.n_cities = n_cities
        self.bounds = bounds
        self.pop_size = population_size
        self.generations = generations
        self.mutation_rate = mutation_rate
        self.crossover_rate = crossover_rate
        self.objective_function = objective_function
        self.fitness_function = fitness_function

    def initialize_population(self) -> np.ndarray:
        pop_size, n_params = self.pop_size, self.n_cities * 2
        population = np.random.uniform(
            low=[b[0] for b in self.bounds],
            high=[b[1] for b in self.bounds],
            size=(pop_size, n_params)
        )
        return population

    def evaluate(self, population: np.ndarray) -> np.ndarray:
        fitness = []
        for ind in population:
            # Split into centers and radii
            centers = ind[:self.n_cities * 2].reshape((self.n_cities, 2))
            radii = ind[self.n_cities * 2:]
            fitness_val = self.fitness_function(radii)
            fitness.append(fitness_val)
        return np.array(fitness)

    def select_parents(self, population: np.ndarray, fitness: np.ndarray) -> np.ndarray:
        # Tournament selection
        indices = np.random.choice(len(population), size=2 * len(population), replace=False)
        tournament = population[indices[:len(population)]]
        fitness_tournament = fitness[indices[:len(population)]]
        selected = []
        for _ in range(len(population)):
            a, b = np.random.choice(len(tournament), size=2, replace=False)
            if fitness_tournament[a] > fitness_tournament[b]:
                selected.append(tournament[a])
            else:
                selected.append(tournament[b])
        return np.array(selected)

    def crossover(self, parent1: np.ndarray, parent2: np.ndarray) -> np.ndarray:
        if random.random() < self.crossover_rate:
            # One-point crossover
            point = random.randint(1, len(parent1) - 1)
            child = np.concatenate((parent1[:point], parent2[point:]))
            return child
        else:
            return parent1

    def mutate(self, individual: np.ndarray) -> np.ndarray:
        n_params = len(individual)
        for i in range(n_params):
            if random.random() < self.mutation_rate:
                # Mutate with small random value
                delta = np.random.uniform(-0.1, 0.1)
                individual[i] += delta
                # Clamp to bounds
                individual[i] = np.clip(individual[i], self.bounds[i][0], self.bounds[i][1])
        return individual

    def optimize(self) -> tuple[np.ndarray, np.ndarray]:
        population = self.initialize_population()
        best_fitness = -np.inf
        best_individual = population[0]
        for gen in range(self.generations):
            fitness = self.evaluate(population)
            parents = self.select_parents(population, fitness)
            next_population = []
            for i in range(len(parents) // 2):
                parent1 = parents[i]
                parent2 = parents[len(parents) // 2 + i]
                child = self.crossover(parent1, parent2)
                child = self.mutate(child)
                next_population.append(child)
            population = np.array(next_population)
            current_best_fitness = np.max(fitness)
            if current_best_fitness > best_fitness:
                best_fitness = current_best_fitness
                best_individual = population[np.argmax(fitness)]
        return best_individual[:self.n_cities * 2].reshape((self.n_cities, 2)), best_individual[self.n_cities * 2:]
