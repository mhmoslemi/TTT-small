import math
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    grid_cols = 6
    grid_rows = 5
    grid_spacing = 0.2
    max_radius = 0.25

    # Initialize centers and radii
    centers = np.zeros((26, 2))
    radii = np.zeros(26)

    # Place initial hexagonal grid
    for col in range(grid_cols):
        for row in range(grid_rows):
            x = col * grid_spacing + grid_spacing / 2
            y = row * grid_spacing + grid_spacing / 2 + (row % 2) * grid_spacing / 2
            if x < 1 - 1e-6 and y < 1 - 1e-6:
                idx = col * grid_rows + row
                centers[idx] = [x, y]
                radii[idx] = max_radius / 2  # Initial radius based on spacing

    # Define helper functions
    def is_inside(x, y, r):
        return (x - r >= 0 - 1e-6 and x + r <= 1 + 1e-6 and
                y - r >= 0 - 1e-6 and y + r <= 1 + 1e-6)

    def is_overlap(x1, y1, r1, x2, y2, r2):
        dx = x1 - x2
        dy = y1 - y2
        dist_sq = dx**2 + dy**2
        return math.sqrt(dist_sq) < r1 + r2 - 1e-6

    # Define the objective function (to maximize sum of radii)
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum

    # Define constraints
    def constraints(radii):
        n = len(radii)
        constraints = []

        # Boundary constraint
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if not is_inside(x, y, r):
                return False  # Violates boundary constraint

        # Non-overlap constraint
        for i in range(n):
            for j in range(i + 1, n):
                if is_overlap(centers[i][0], centers[i][1], radii[i],
                              centers[j][0], centers[j][1], radii[j]):
                    return False  # Violates non-overlap constraint

        return True

    # Flatten radii for optimization
    radii_flat = radii.flatten()

    # Initial guess for optimization
    initial_guess = radii_flat

    # Set bounds for optimization
    bounds = Bounds(lb=0.0, ub=np.inf)

    # Define objective and constraints
    res = minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints={

        r = radii[i]

    # Compute sum of radii
    total_sum = np.sum(radii)

    return centers, radii, total_sum
