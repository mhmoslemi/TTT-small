    centers = list(set(centers))

            # Fill remaining with random points
            x = np.random.uniform(0.2, 0.8)

            centers.append((x, y))

    # Wrap the centers as a numpy array
    centers = np.array(centers)

    def compute_total_radii(centers, n):
        # Initial estimate of radii based on minimum distance to surrounding circles and boundaries
        radii = []
        for i in range(n):
            x, y = centers[i]
            min_dist_to_boundary = min(x, 1 - x, y, 1 - y)
            min_dist_to_other = float('inf')
            for j in range(n):
                if i != j:
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = np.sqrt(dx**2 + dy**2)
                    if dist < min_dist_to_other:
                        min_dist_to_other = dist
            # Radius is the minimum of the two
            r = min(min_dist_to_boundary, min_dist_to_other / 2)
            radii.append(r)
        return radii

    # Define the objective function to maximize the sum of radii
    def objective(x):
        # x is a flat array of all centers (flattened), with shape (2n,)
