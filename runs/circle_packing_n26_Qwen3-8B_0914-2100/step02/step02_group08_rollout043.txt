import numpy as np
import random
from deap import base, creator, tools

def run_packing():
    n = 26

    # Define creator for fitness and individual
    creator.create("FitnessMax", float, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMax)

    # Initialize toolbox
    toolbox = base.Toolbox()

    # Define possible positions for circles
    def generate_positions():
        # We'll use a hybrid approach: (1) dense columns, (2) clusters, (3) corners
        positions = []

        # Generate 6 rows with 4 circles each
        for i in range(6):
            for j in range(4):
                x = j + i * 0.2  # slight staggered rows
                y = i * 0.35 + 0.1
                positions.append((x, y))

        # Add more positions near corners
        positions.append((0.05, 0.05))
        positions.append((0.05, 0.95))
        positions.append((0.95, 0.05))
        positions.append((0.95, 0.95))

        # Ensure not more than 26 positions
        positions = positions[:26]

        return positions

    # Initialize the toolbox with random positions
    toolbox.register("generate_pos", generate_positions)

    # Define function to compute total radii
    def compute_total_radii(positions):
        radii = []
        for i in range(len(positions)):
            x, y = positions[i]
            # Compute distance to boundaries
            min_dist_to_boundary = min(x, 1 - x, y, 1 - y)
            # Compute distance to other circles
            min_dist_to_other = float('inf')
            for j in range(len(positions)):
                if i != j:
                    dx = positions[i][0] - positions[j][0]
                    dy = positions[i][1] - positions[j][1]
                    dist = np.sqrt(dx**2 + dy**2)
                    if dist < min_dist_to_other:
                        min_dist_to_other = dist
            # Radius is the minimum of the two
            r = min(min_dist_to_boundary, min_dist_to_other / 2)
            radii.append(r)
        return np.array(radii)

    # Create the fitness function
    def evaluate(individual):
        # individual is the set of positions (x, y) of circles
        positions = np.array(individual)
        radii = compute_total_radii(positions)
        total_sum = np.sum(radii)
        return total_sum,

    # Genetic Algorithm setup
    toolbox.register("individual", tools.initIterate, list, tools.stochastic_uniform, low=0, up=1)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("mate", tools.cxTwoPoint)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.05, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    toolbox.register("evaluate", evaluate)

    # Create population
    pop = toolbox.population(n=50)
    stats = tools.Statistics(lambda ind: ind.fitness.values)
    stats.register("avg", np.mean)
    stats.register("std", np.std)
    stats.register("min", np.min)
    stats.register("max", np.max)

    # Run genetic algorithm
    algorithms = [
        ("ga", tools.eaSimple, {"cxpb": 0.5, "mutpb": 0.2, "ngen": 100}),
    ]

    best_ind = None
    best_fitness = -np.inf

    for name, algorithm, params in algorithms:
        random.seed(42)
        pop = toolbox.population(n=50)
        for gen in range(100):
            offspring = algorithm(pop, **params)
            pop = tools.merge(pop, offspring)
            fits = list(map(toolbox.evaluate, pop))
            for fit, ind in zip(fits, pop):
                ind.fitness.values = fit
            pop = tools.selBest(pop, len(pop))
            record = stats.compile(pop)
            if record["max"] > best_fitness:
                best_fitness = record["max"]
                best_ind = pop[0]

    # Get the best individual
    best_positions = np.array(best_ind)
    best_radii = compute_total_radii(best_positions)

    return best_positions, best_radii, np.sum(best_radii)
