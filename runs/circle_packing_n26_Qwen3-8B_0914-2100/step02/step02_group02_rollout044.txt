import math
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    grid_padding = 0.15
    initial_circle_radius = 0.15

    radii = np.zeros(n_circles)
    circle_index = 0

            x = col + 0.5 * (col % 2)  # Adjusted for hexagonal grid

                radii[circle_index] = initial_circle_radius

        centers[i] = [x, y]
        radii[i] = initial_circle_radius

    def is_inside(x, y, r):
        return (x - r >= 0) and (x + r <= 1) and (y - r >= 0) and (y + r <= 1)

    def is_overlap(x1, y1, r1, x2, y2, r2):
        dx = x1 - x2
        dy = y1 - y2
        dist_sq = dx**2 + dy**2
        return math.sqrt(dist_sq) < r1 + r2 - 1e-6

    def objective_func(radii):
        # Objective function: maximize sum of radii
        return -np.sum(radii)

    def constraint_func(radii):
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                if is_overlap(centers[i][0], centers[i][1], radii[i],
                              centers[j][0], centers[j][1], radii[j]):
                    constraints.append((radii[i] + radii[j] - 1e-6, 0))
        return constraints

    # Build radius bounds for Nelder-Mead
    radius_bounds = [(0.0, 0.5) for _ in range(n_circles)]
