from scipy.optimize import minimize
import math

def run_packing():

    # We'll use two compact clusters and one sparse tail to maximize radii

    # Define a grid that is compact in the left and right part, but sparse on the edges
    centers = np.array([
        [0.02, 0.02], [0.02, 0.25], [0.02, 0.5], [0.02, 0.75], [0.02, 0.98],
        [0.25, 0.02], [0.25, 0.25], [0.25, 0.5], [0.25, 0.75], [0.25, 0.98],
        [0.5, 0.02], [0.5, 0.25], [0.5, 0.5], [0.5, 0.75], [0.5, 0.98],
        [0.75, 0.02], [0.75, 0.25], [0.75, 0.5], [0.75, 0.75], [0.75, 0.98],
        [0.98, 0.02], [0.98, 0.25], [0.98, 0.5], [0.98, 0.75], [0.98, 0.98]
    ])

    # Define a function to compute total radii given centers
    def compute_total_radii(centers, n):
        radii = []
        for i in range(n):
            x, y = centers[i]
            # Distance to wall
            min_dist_to_wall = min(x, 1 - x, y, 1 - y)
            # Distance to other circles
            min_dist_to_other = float('inf')
            for j in range(n):
                if i != j:
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = math.hypot(dx, dy)
                    if dist < min_dist_to_other:
                        min_dist_to_other = dist
            # Radius is the minimum of the two
            r = min(min_dist_to_wall, min_dist_to_other / 2)
            radii.append(r)
        return radii

    # Define the objective function to maximize the sum of radii
    def objective(x):
        centers = x.reshape((n, 2))
        radii = compute_total_radii(centers, n)
        return -np.sum(radii)

    # Define constraints to ensure that all centers are in [0,1]^2
    def constraint(x):
        centers = x.reshape((n, 2))
        for i in range(n):
            x_center, y_center = centers[i]
            if not (0 <= x_center <= 1) or not (0 <= y_center <= 1):
                return -1
        return 0

    # Initial guess
    x0 = centers.reshape(-1)

    # Optimization
    result = minimize(
        objective,
        x0,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(2 * n)],
        constraints={'type': 'ineq', 'fun': constraint},
        tol=1e-8,
        options={'maxiter': 1000}
    )

    # Extract the optimal centers and radii
    centers = result.x.reshape((n, 2))
    radii = compute_total_radii(centers, n)
    total_sum = np.sum(radii)

    return centers, radii, total_sum
