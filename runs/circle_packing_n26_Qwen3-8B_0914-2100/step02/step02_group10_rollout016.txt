from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    n = 26

    # Create a more flexible initial configuration
    # We will place 26 circles in a more balanced configuration across the unit square
    # Inspired by Peter L. Boyer and Ulrich Betz's work on circle packing

    # Create random initial points near the center with some spread
    centers = np.random.rand(n, 2)
    # Shift them to place about half near the center and half near the boundaries
    for i in range(n):
        if np.random.rand() < 0.5:
            # Put near boundary
            if np.random.rand() < 0.5:
                centers[i, 0] = np.random.rand() * 0.2
            else:
                centers[i, 1] = np.random.rand() * 0.2
        else:
            # Put near the center
            centers[i, 0] = 0.5 + np.random.rand() * 0.4
            centers[i, 1] = 0.5 + np.random.rand() * 0.4

    # Define a function to compute the total sum of radii given centers
    def compute_total_radii(centers, n):
        radii = []
        for i in range(n):
            x, y = centers[i]
            # Compute distance to the boundary
            min_dist_to_boundary = min(x, 1 - x, y, 1 - y)
            # Compute distance to other circles
            min_dist_to_other = float('inf')
            for j in range(n):
                if i != j:
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = np.sqrt(dx**2 + dy**2)
                    if dist < min_dist_to_other:
                        min_dist_to_other = dist
            # Radius is the minimum of the two
            r = min(min_dist_to_boundary, min_dist_to_other / 2)
            radii.append(r)
        return radii

    # Define the objective function to maximize the sum of radii
    def objective(x):
        # x is a flat array of all centers (flattened), with shape (2n,)
        # Convert to centers (n, 2)
        centers = np.reshape(x, (n, 2))
        radii = compute_total_radii(centers, n)
        # Return negative of sum of radii to minimize (since scipy.optimize.minimize minimizes)
        return -np.sum(radii)

    # Define bounds
    bounds = [(0, 1) for _ in range(2 * n)]

    # Define the soft constraint for overlap
    def constraint(x):
        centers = np.reshape(x, (n, 2))
        for i in range(n):
            x_center, y_center = centers[i]
            if not (0 <= x_center <= 1) or not (0 <= y_center <= 1):
                return -1000  # Penalize boundary violation
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < 1e-6:
                    return -1000  # Penalize overlap
        return 0

    # Initial guess is the random centers
    initial_guess = np.array([x for x in np.reshape(centers, (2 * n,))])

    # Perform optimization
    result = minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=Bounds(bounds),
        constraints={'type': 'ineq', 'fun': constraint}
    )

    # Extract the optimal centers and radii
    centers = np.reshape(result.x, (n, 2))
    radii = compute_total_radii(centers, n)

    total_sum = np.sum(radii)

    return centers, radii, total_sum
