from scipy.optimize import differential_evolution, minimize
import time

def upscale_h(initial_h, n_new):
    n_old = len(initial_h)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    x_new = np.linspace(dx_new/2, 2 - dx_new/2, n_new)
    indices = (x_new / dx_old).astype(int)
    h_unscaled = initial_h[indices]
    scaled = (50.0 / h_unscaled.sum()) * h_unscaled
    return np.clip(scaled, 0, 1)

def repair_h(h_candidate):
    h_clipped = np.clip(h_candidate, 0, 1)
    s = h_clipped.sum()
    if s == 0:
        return h_clipped
    return (50.0 / s) * h_clipped

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    max_val = 0.0
    for k in range(n):
        if k == 0:
