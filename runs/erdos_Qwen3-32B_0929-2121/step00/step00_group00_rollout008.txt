    h_new = h_new * (target_sum / h_new.sum()) if h_new.sum() != 0 else h_new
    best_h = h_new.copy()
    
    def compute_c5(h):
        dx = 2.0 / n_points
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx

    def clip_scale(x, target_sum):

        s = x_clipped.sum()
        if s == 0:

        scaled = x_clipped * (target_sum / s)
        # Ensure no values exceed bounds after scaling

        overflow = target_sum - scaled[scaled < 1].sum()
        under_1 = scaled < 1
        scaled[under_1] += overflow / under_1.sum() if under_1.sum() > 0 else 0
        return scaled

    def objective(x):
