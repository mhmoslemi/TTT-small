        x = np.linspace(0, 2, n_points)
        f_interp = interp1d(x_initial, initial_h, kind='linear', fill_value='extrapolate')
        h_initial = f_interp(x)
    except:
        h_initial = np.zeros(n_points)
    h_initial = repair_h(h_initial)

    bounds = [(0, 1)] * n_points
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)

    # Differential Evolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        de_gen = 50
        de_pop = 10
        def de_objective(x):
            return compute_c5(repair_h(x))
        try:
            de_result = differential_evolution(de_objective, bounds, popsize=de_pop, maxiter=de_gen, polish=False, seed=seed, tol=1e-6)
            if de_result.success:
                candidate = repair_h(de_result.x)
                c5 = compute_c5(candidate)
                if c5 < best_c5:
                    best_h, best_c5 = candidate, c5
        except:
            pass

    # L-BFGS-B refinement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 10:
        def lbfgs_objective(x):
            return compute_c5(repair_h(x))
        try:
            lbfgs_result = minimize(lbfgs_objective, x0=best_h, bounds=bounds, method='L-BFGS-B', options={'maxiter': 500})
            if lbfgs_result.success:
                candidate = repair_h(lbfgs_result.x)
                c5 = compute_c5(candidate)
                if c5 < best_c5:
                    best_h, best_c5 = candidate, c5
        except:
            pass

    best_h = repair_h(best_h)
    best_c5 = compute_c5(best_h)

    return (best_h.tolist(), best_c5, n_points)
