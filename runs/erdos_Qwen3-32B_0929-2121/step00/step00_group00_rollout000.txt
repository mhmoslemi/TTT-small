from scipy.optimize import differential_evolution, minimize
from scipy.interpolate import interp1d
import time

            scaled = np.full_like(h_clipped, 0.5)
        else:
            scaled = h_clipped * (50.0 / sum_h)

        correlate = np.correlate(scaled, 1 - scaled, mode='full')
