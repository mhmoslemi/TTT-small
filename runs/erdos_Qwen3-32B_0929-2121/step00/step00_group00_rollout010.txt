def evaluate_c5(h):
    dx = 2.0 / len(h)
    correl = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correl) * dx
    return c5

def repair(u):
    u_clipped = np.clip(u, 0, 1)
    s = np.sum(u_clipped)
    if np.isclose(s, 50, atol=1e-8):
        return u_clipped
    delta = 50 - s
    if delta > 0:
        indices = np.where(u_clipped < 1)[0]
        for i in indices:
            add = min(delta, 1 - u_clipped[i])
            u_clipped[i] += add
            delta -= add
            if np.isclose(delta, 0):

        indices = np.where(u_clipped > 0)[0]
        for i in indices:
            subtract = min(delta, u_clipped[i])
            u_clipped[i] -= subtract
            delta -= subtract
            if np.isclose(delta, 0):
                break
    return u_clipped
