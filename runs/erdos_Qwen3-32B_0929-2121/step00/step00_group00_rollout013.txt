import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.interpolate import interp1d
import time

def run(seed=42, budget_s=80, **kwargs):
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    required_sum = n_points / 2.0  # since ∫h dx = 1

    # Load initial_h_values if available
    has_initial = 'initial_h_values' in globals()
    if has_initial:
        original_x = np.linspace(0, 2, len(initial_h_values))
        new_x = np.linspace(0, 2, n_points)
        f = interp1d(original_x, initial_h_values, kind='nearest')
        h_start = f(new_x)
    else:
        h_start = np.zeros(n_points)
        mid = n_points // 2
        h_start[mid:] = 1.0

    # Ensure sum matches required_sum and values are in [0,1]
    def project_to_feasible(h):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if s <= 1e-8:
            return np.zeros_like(h)
        scale = required_sum / s
        h = np.clip(h * scale, 0.0, 1.0)
        return h

    h_start = project_to_feasible(h_start)

    # Define objective for DE
    def objective(h_vec):
        h = np.clip(h_vec, 0.0, 1.0)
        s = h.sum()
        if s <= 1e-8:
            return 1e6
        h = project_to_feasible(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points

    # Differential Evolution phase
    de_popsize = 10
    de_max_gen = 50
    de_result = differential_evolution(
        objective, 
        bounds,
        popsize=de_popsize,
        strategy='best1bin',
        maxiter=de_max_gen,
        mutation=(0.5, 0.8),
        recombination=0.9,
        seed=seed,
        polish=False,
        disp=False,
        update_interval=100,
    )
    best_de = de_result.x
    best_de = project_to_feasible(best_de)

    # Local refinement with L-BFGS-B
    def lbfgs_objective(h_vec):
        return objective(h_vec)

    lbfgs_bounds = [(0.0, 1.0)] * n_points
    lbfgs_result = minimize(
        lbfgs_objective,
        best_de,
        method='L-BFGS-B',
        bounds=lbfgs_bounds,
        options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6},
    )
    best_lbfgs = lbfgs_result.x
    best_lbfgs = project_to_feasible(best_lbfgs)

    # Select the best solution found so far
    def evaluate(h):
        h = project_to_feasible(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    candidates = [h_start, best_de, best_lbfgs]
    evaluations = [(evaluate(h), h) for h in candidates]
    c5_min, h_opt = min(evaluations)

    # Final check before timeout
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return h_opt, c5_min, n_points

    # Perform additional LBFGS steps if time allows

        lbfgs_objective,
        h_opt,
        method='L-BFGS-B',
        bounds=lbfgs_bounds,
        options=lbfgs_options,
    )
    h_opt = project_to_feasible(lbfgs_result.x)

    return h_opt, evaluate(h_opt), n_points
