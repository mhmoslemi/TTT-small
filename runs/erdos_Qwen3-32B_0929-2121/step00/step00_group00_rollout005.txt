from scipy.optimize import differential_evolution, minimize
import time

def upscale_h(original_h, original_n, new_n):
    original_x = np.linspace(0, 2, original_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    bins = np.append(original_x, 2.0)
    indices = np.digitize(new_x, bins) - 1
    indices = np.clip(indices, 0, original_n - 1)
    return original_h[indices]

def project_to_sum(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if np.allclose(current_sum, target_sum):
        return h_clipped
    scale = target_sum / current_sum
    h_scaled = h_clipped * scale
    h_scaled_clipped = np.clip(h_scaled, 0, 1)
    return h_scaled_clipped

def objective(h, dx, n_points):
    h_proj = project_to_sum(np.copy(h), n_points / 2)
    cross_corr = np.correlate(h_proj, 1 - h_proj, mode='full')
    return np.max(cross_corr * dx)

import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def upscale_h(original_h, original_n, new_n):
    original_x = np.linspace(0, 2, original_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    bins = np.append(original_x, 2.0)
    indices = np.digitize(new_x, bins) - 1
    indices = np.clip(indices, 0, original_n - 1)
    return original_h[indices]

def project_to_sum(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if np.allclose(current_sum, target_sum):
        return h_clipped
    scale = target_sum / current_sum
    h_scaled = h_clipped * scale
    h_scaled_clipped = np.clip(h_scaled, 0, 1)
    return h_scaled_clipped

def evaluate_objective(h, dx, n_points):
    h_proj = project_to_sum(h, n_points / 2)
    cross_corr = np.correlate(h_proj, 1 - h_proj, mode='full')
    return np.max(cross_corr * dx)

def run(seed=42, budget_s=80, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    initial_h = np.random.rand(n_points)
    if 'initial_h_values' in globals():
        try:
            original_h = initial_h_values
            original_n = len(original_h)
            initial_h = upscale_h(original_h, original_n, n_points)
            initial_h = project_to_sum(initial_h, target_sum)
        except Exception:
            pass

    bounds = [(0, 1) for _ in range(n_points)]

    de_popsize = 10
    de_init = [initial_h.copy()]
    while len(de_init) < de_popsize:
        new_h = np.random.rand(n_points)
        new_h = project_to_sum(new_h, target_sum)
        de_init.append(new_h)
    de_init_array = np.array(de_init)

    start_time = time.time()
    result_de = differential_evolution(
        lambda h: evaluate_objective(h, dx, n_points),
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=de_popsize,
        tol=1e-6,
        recombination=0.9,
        mutation=0.8,
        polish=False,
        init=de_init_array,
        workers=1,
        seed=seed
    )
    best_h = result_de.x

    best_h = project_to_sum(best_h, target_sum)
    c5_de = evaluate_objective(best_h, dx, n_points)

    if time.time() - start_time < budget_s * 0.8:
        def objective_for_bfgs(h_flat):
            h = h_flat.copy()
            h = project_to_sum(h, target_sum)
            return evaluate_objective(h, dx, n_points)

        result_bfgs = minimize(
            objective_for_bfgs,
            best_h.copy(),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 500, 'ftol': 1e-12}
        )
        best_h = result_bfgs.x
        best_h = project_to_sum(best_h, target_sum)
        c5_best = evaluate_objective(best_h, dx, n_points)
    else:
        c5_best = c5_de

    best_h = project_to_sum(best_h, target_sum)
    return (best_h.tolist(), c5_best, n_points)
