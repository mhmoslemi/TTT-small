import scipy.optimize as opt
import time

def run(seed=42, budget_s=80, **kwargs):
    start_time = time.time()
    n_points = 100

    original_h = initial_h_values.copy()
    original_n = len(original_h)
    x_original = np.linspace(0, 2, original_n, endpoint=False)
    x_new = np.linspace(0, 2, n_points, endpoint=False)
    h_initial = np.interp(x_new, x_original, original_h)
    h_initial = np.clip(h_initial, 0.0, 1.0)
    h_initial *= 50.0 / np.sum(h_initial)
    h_initial = np.clip(h_initial, 0.0, 1.0)

    bounds = [(0.0, 1.0)] * n_points

    def objective(h_vec):
        h = np.clip(h_vec, 0.0, 1.0)
        current_sum = np.sum(h)
        if current_sum < 1e-6:
            return float('inf')
        h = h * (50.0 / current_sum)
        h = np.clip(h, 0.0, 1.0)
        dx = 2.0 / n_points
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)

    try:
        result = opt.differential_evolution(
            objective, bounds, strategy='best1bin', maxiter=40,
            popsize=10, mutation=(0.5, 1.0), recombination=0.9,
            seed=seed, polish=False, init='random', tol=1e-6,
            workers=1, disp=False
        )
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_h = result.x.copy()
            best_c5 = current_c5
    except Exception as e:
        pass

    elapsed = time.time() - start_time
    if elapsed < budget_s - 5:
        h0 = np.clip(best_h, 0.0, 1.0)
        h0 *= 50.0 / np.sum(h0)
        h0 = np.clip(h0, 0.0, 1.0)
        try:
            result_lb = opt.minimize(
                objective, x0=h0, method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 400, 'ftol': 1e-6}
            )
            current_c5 = objective(result_lb.x)
            if current_c5 < best_c5:
                best_h = result_lb.x.copy()
                best_c5 = current_c5
        except Exception as e:
            pass

    best_h = np.clip(best_h, 0.0, 1.0)
    total = np.sum(best_h)
    best_h = best_h * (50.0 / total)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = objective(best_h)
    
    return (best_h, final_c5, n_points)
