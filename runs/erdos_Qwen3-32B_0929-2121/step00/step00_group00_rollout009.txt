def project_h(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if np.isclose(current_sum, 0):
        return h_clipped
    scaling_factor = target_sum / current_sum
    h_scaled = h_clipped * scaling_factor
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled
