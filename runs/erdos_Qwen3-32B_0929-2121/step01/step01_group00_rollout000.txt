import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def run(seed=42, budget_s=80, **kwargs):
    n_points = 200
    dx = 2.0 / n_points
    
    # Interpolate initial_h_values if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        initial_h = np.array(initial_h_values)
        n_initial = len(initial_h)
        x_initial = np.linspace(0, 2, n_initial, endpoint=True)
        x_new = np.linspace(0, 2, n_points, endpoint=True)
        f_interpolate = interp1d(x_initial, initial_h, kind='linear', fill_value='extrapolate')
        h0 = f_interpolate(x_new)
    else:
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
    
    # Normalize to satisfy the sum constraint
    current_sum = np.sum(np.clip(h0, 0, 1))
    if current_sum != 0:
        h0 = np.clip(h0 * (100.0 / current_sum), 0, 1)
    else:
        h0 = np.full_like(h0, 0.5)
    
    # Surrogate objective using log-sum-exp regularization
    def objective_surrogate(h_flat, tau=0.1):
        h = h_flat.reshape(-1)
        h_clipped = np.clip(h, 0, 1)
        correlate = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        c5_values = correlate * dx
        max_c5 = np.max(c5_values)
        exp_vals = np.exp(tau * (c5_values - max_c5))
        exp_sum = np.sum(exp_vals)
        return (1.0 / tau) * np.log(exp_sum) + max_c5
    
    # Equality constraint: sum(h) = 100
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - 100.0})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization with adaptive learning via smoothed objective
    start_time = time.time()
    result = minimize(
        objective_surrogate,
        x0=h0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 150, 'ftol': 1e-7, 'disp': False}
    )
    opt_duration = time.time() - start_time
    
    # Final solution processing
    best_h = np.clip(result.x, 0, 1)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - 100.0) > 1e-8:
        best_h = best_h * (100.0 / best_h_sum)
        best_h = np.clip(best_h, 0, 1)
    
    # Evaluate final C5 using original max objective
    def true_c5(h):
        correlate = np.correlate(h, 1 - h, mode='full')
        return np.max(correlate) * dx
    
    final_c5 = true_c5(best_h)
    
    return best_h, final_c5, n_points
