import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def run(seed=42, budget_s=80, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    tau = 0.1

    if 'initial_h_values' in globals() and initial_h_values is not None:
        initial_h = np.array(initial_h_values)
        n_initial = len(initial_h)
        x_initial = np.linspace(0, 2, n_initial, endpoint=True)
        x_new = np.linspace(0, 2, n_points, endpoint=True)
        f = interp1d(x_initial, initial_h, kind='nearest', fill_value='extrapolate')
        h0 = f(x_new)
    else:
        h0 = np.zeros(n_points)
        h0[:int(n_points/2)] = 1.0

    h0 = np.clip(h0, 0, 1)
    sum_h0 = np.sum(h0)
    if sum_h0 != 0:
        h0 = h0 * (n_points / 2 / sum_h0)
    h0 = np.clip(h0, 0, 1)

    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        correlate = np.correlate(h, 1 - h, mode='full')
        overlap = correlate * dx
        log_sum_exp = np.log(np.sum(np.exp(tau * overlap))) / tau
        return log_sum_exp

    bounds = [(0, 1)] * n_points
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2})

    start_time = time.time()
    result_opt = minimize(
        objective,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    opt_duration = time.time() - start_time

    best_h = np.clip(result_opt.x, 0, 1)
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        best_h = best_h * ((n_points / 2) / sum_h)
        best_h = np.clip(best_h, 0, 1)

    correlate = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(correlate) * dx

    return best_h, c5_bound, n_points
