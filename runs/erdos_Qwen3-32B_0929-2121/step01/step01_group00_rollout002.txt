        return h0, objective(h0), n_points

    # Validate and repair solution
    best_h = np.clip(result.x, 0.0, 1.0)
    best_sum = np.sum(best_h)
    if abs(best_sum - 100.0) > 1e-8:
        best_h = best_h * (100.0 / best_sum)
        best_h = np.clip(best_h, 0.0, 1.0)

    # Evaluate final C5
    correlate = np.correlate(best_h, 1 - best_h, mode='full')
    c5 = np.max(correlate) * dx

    return best_h, c5, n_points
