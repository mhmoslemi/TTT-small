import numpy as np
from scipy.optimize import minimize
import time
from scipy.interpolate import interp1d

def run(seed=42, budget_s=80, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial h construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        initial_h = np.array(initial_h_values)
        n_initial = len(initial_h)
        x_initial = np.linspace(0, 2, n_initial, endpoint=True)
        x_new = np.linspace(0, 2, n_points, endpoint=True)
        f_interpolate = interp1d(x_initial, initial_h, kind='linear', fill_value='extrapolate')
        h0 = f_interpolate(x_new)
    else:
        h0 = np.zeros(n_points)
        h0[:int(n_points/2)] = 1.0
    
    # Normalize initial guess
    h0 = np.clip(h0, 0, 1)
    h0_sum = np.sum(h0)
    if h0_sum != 0:
        h0 = (h0 * (50.0 / h0_sum)).clip(0, 1)
    
    # Surrogate objective with temperature parameter
    tau = 0.1
    def objective(h):
        h = np.clip(h, 0, 1)
        correlate = np.correlate(h, 1 - h, mode='full') * dx
        exp_vals = np.exp(tau * correlate)
        return np.log(exp_vals.sum()) / tau  # Differentiable approximation of max
    
    # Constraints and bounds
    bounds = [(0.0, 1.0)] * n_points
    constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) - 50.0}
    
    # Optimization using SLSQP with surrogate objective
    start_time = time.time()
    result = minimize(
        objective, h0, method='SLSQP', bounds=bounds, constraints=constraints,
        options={'maxiter': 300, 'ftol': 1e-8, 'disp': False}
    )
    
    # Process result and compute actual C5
    best_h = np.clip(result.x, 0, 1).astype(np.float64)
    best_h_sum = np.sum(best_h)
    if abs(best_h_sum - 50.0) > 1e-8:
        best_h = (best_h * (50.0 / best_h_sum)).clip(0, 1)
    
    c5_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(c5_values)
    
    return best_h, c5_bound, n_points
