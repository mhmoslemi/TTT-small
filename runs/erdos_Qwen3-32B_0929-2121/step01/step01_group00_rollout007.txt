        return np.max(correlate) * dx
    
    # Local optimization with SLSQP
    constraints_slsqp = ({'type': 'eq', 'fun': lambda h: np.sum(h) - (n_points / 2)})
    bounds_slsqp = [(0.0, 1.0) for _ in range(n_points)]
    
    remaining_time = budget_s - de_duration
    if remaining_time < 1:
        # Not enough time for local optimization
        best_h = h_de_scaled
        best_h_sum = np.sum(best_h)
        if not np.isclose(best_h_sum, n_points / 2, atol=1e-6):
            best_h = best_h * (n_points / 2 / best_h_sum)
            best_h = np.clip(best_h, 0, 1)
        return best_h, objective_final(best_h), n_points
    
    iter_estimate = int(remaining_time * 1.5)  # Adjust based on time budget
    options_slsqp = {'maxiter': iter_estimate, 'ftol': 1e-8, 'disp': False}
    result_slsqp = minimize(
        objective_final, h_de_scaled, method='SLSQP', bounds=bounds_slsqp,
        constraints=constraints_slsqp, options=options_slsqp
    )
    
    # Select best solution
    if result_slsqp.success:
        best_h = np.clip(result_slsqp.x, 0, 1)
        best_h_sum = np.sum(best_h)
        if not np.isclose(best_h_sum, n_points / 2, atol=1e-6):
            best_h = best_h * (n_points / 2 / best_h_sum)
            best_h = np.clip(best_h, 0, 1)
        final_c5 = objective_final(best_h)
    else:
        best_h = h_de_scaled
        best_h_sum = np.sum(best_h)
        if not np.isclose(best_h_sum, n_points / 2, atol=1e-6):
            best_h = best_h * (n_points / 2 / best_h_sum)
            best_h = np.clip(best_h, 0, 1)
        final_c5 = objective_final(best_h)
    
    return best_h, final_c5, n_points
