    )

    h_opt = np.clip(h_opt, 0, 1)
    
    # Final constraint repair
    h_sum = np.sum(h_opt)
    if abs(h_sum - n_points/2) > 1e-5:
        h_opt = h_opt * (n_points/2) / h_sum
        h_opt = np.clip(h_opt, 0, 1)
    
    # Final evaluation with original objective
    correlate = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5_bound = np.max(correlate) * dx
    
    return h_opt, c5_bound, n_points
