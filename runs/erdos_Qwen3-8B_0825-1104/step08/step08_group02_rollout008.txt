import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with a moderate number of points
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize with a smooth, periodic function
    x = np.linspace(0, 2, n_points)
    h_initial = np.sin(np.pi * x) / 2 + 0.5
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        correlation = np.correlate(h, one_minus_h, mode="full")
        overlap = correlation * dx
        return np.max(overlap)
    
    # Function to enforce the area constraint
    def area_constraint(h):
        return np.sum(h) * dx - 1.0
    
    # Optimization bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial optimization with multiple perturbations
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    
    # Generate and optimize several perturbed starting points
    for _ in range(10):
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = h_initial + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        
        # Scale to satisfy the area constraint
        area_perturbed = np.sum(perturbed_h) * dx
        if area_perturbed != 0:
            scale_factor = 1.0 / area_perturbed
            perturbed_h *= scale_factor
        
        # Local optimization
        result = optimize.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': area_constraint}],
            options={'maxiter': 50, 'disp': False}
        )
        
        if objective(result.x) < best_c5:
            best_h = result.x
            best_c5 = objective(best_h)
    
    # Refine with higher resolution
    n_points_refine = 400
    dx_refine = 2.0 / n_points_refine
    
    # Interpolate the best solution to higher resolution
    x_refine = np.linspace(0, 2, n_points_refine)
    x_initial = np.linspace(0, 2, n_points)
    best_h_refine = np.interp(x_refine, x_initial, best_h)
    best_h_refine = np.clip(best_h_refine, 0, 1)
    best_h_refine = best_h_refine * (n_points_refine / 2) / np.sum(best_h_refine * dx_refine)
    
    # Refine the solution
    result_refine = optimize.minimize(
        objective,
        best_h_refine,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_points_refine)],
        constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) * dx_refine - 1.0}],
        options={'maxiter': 50, 'disp': False}
    )
    
    # Interpolate back to original resolution
    x_final = np.linspace(0, 2, n_points)
    best_h_final = np.interp(x_final, x_refine, result_refine.x)
    best_h_final = np.clip(best_h_final, 0, 1)
    best_h_final = best_h_final * (n_points / 2) / np.sum(best_h_final * dx)
    
    # Final evaluation
    final_c5 = objective(best_h_final)
    
    # Early termination if time is running out
    elapsed_time = time.time() - time.time()
    if elapsed_time > budget_s - 5:
        return best_h_final, final_c5, n_points
    
    # Further refinement with diverse perturbations
    for _ in range(5):
        perturbation = np.random.normal(0, 0.03, size=n_points)
        perturbed_h = best_h_final + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        
        # Scale to satisfy the area constraint
        area_perturbed = np.sum(perturbed_h) * dx
        if area_perturbed != 0:
            scale_factor = 1.0 / area_perturbed
            perturbed_h *= scale_factor
        
        # Local optimization
        result_perturb = optimize.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': area_constraint}],
            options={'maxiter': 50, 'disp': False}
        )
        
        if objective(result_perturb.x) < final_c5:
            best_h_final = result_perturb.x
            final_c5 = objective(best_h_final)
    
    return best_h_final, final_c5, n_points
