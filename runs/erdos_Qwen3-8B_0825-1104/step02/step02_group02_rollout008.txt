import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):

                                 0.7722, 0.7701, 0.7652, 0.7578, 0.7482, 0.7367, 0.7235, 0.7086, 0.6923, 0.6746,
                                 0.6559, 0.6364, 0.6162, 0.5954, 0.574, 0.5522, 0.5301, 0.5076, 0.4847, 0.4616,

                                 0.193, 0.168, 0.143, 0.118, 0.093, 0.068, 0.043, 0.018, 0.0])

    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    time_limit = budget_s
    start_time = time.time()

    def compute_c5(h):
        h = np.array(h)
        shifted_h = np.concatenate([h[-n_points + 1:], h], axis=0)
        one_minus_h = 1 - h
        corr = correlate(shifted_h, one_minus_h, mode='full')
        return np.max(corr) * dx

    # Initial best solution
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)

    # Generate a few random starting points
    random_starts = [np.random.uniform(0, 1, n_points) for _ in range(5)]
    for start in random_starts:

        start = start * (n_points / 2) / np.sum(start)

        if c5 < best_c5:
            best_c5 = c5
            best_h = start.copy()

    # Define a function to minimize
    def objective_function(x):
        return compute_c5(x)

    # Use L-BFGS-B to refine the best solution
    result = optimize.minimize(
        objective_function,
        best_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)]
    )
    if result.success:
        h_candidate = result.x

        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()

    h = cp.Variable(n_points)
    constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]

    # Define the objective: minimize the maximum overlap integral
    def minimize_max_overlap(h_values):
        h_values = np.array(h_values)
        shifted_h = np.concatenate([h_values[-n_points + 1:], h_values], axis=0)
        one_minus_h = 1 - h_values
        corr = correlate(shifted_h, one_minus_h, mode='full')
        return np.max(corr) * dx

    # Solve the convex optimization problem
    try:
        prob = cp.Problem(cp.Minimize(minimize_max_overlap(h)), constraints)
        prob.solve(solver=cp.CPLEX, verbose=False)
        if prob.status == 'optimal':
            best_h = h.value
            best_c5 = compute_c5(best_h)
    except:
        pass

    return best_h, best_c5, n_points
