import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
import matplotlib.pyplot as plt

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assuming this is a pre-loaded initial solution

def run(seed=42, budget_s=60, **kwargs):
    # Start timing
    start_time = time.time()

    # Initial parameters
    n_points = 200  # Start with a manageable number of samples
    dx = 2.0 / n_points
    max_iterations = 100
    tolerance = 1e-4
    
    # Initial guess from the provided initial solution
    h_values = np.copy(initial_h_values)
    if len(h_values) > n_points:
        h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_values)), h_values)
    elif len(h_values) < n_points:
        h_values = np.zeros(n_points)
        h_values[:len(h_values)] = h_values
    h_values = np.clip(h_values, 0, 1)

    # Rescale to satisfy the integral constraint
    h_values = h_values / (np.sum(h_values) * dx) * (n_points / 2)
    h_values = np.clip(h_values, 0, 1)

    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        return np.max(np.correlate(h, 1 - h, mode="full")) * dx

    # Function to evaluate the objective and constraints
    def objective_function(h):
        return evaluate_overlap(h)

    # Optimizer settings
    optimizer = opt.SPSA(maxiter=max_iterations, eps=tolerance, niter=100)
    best_h = h_values
    best_c5 = evaluate_overlap(h_values)

    # Optimization loop
    for _ in range(5):  # Try multiple starting points for diversity
        # Create a small perturbation
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed_h = h_values + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)

        # Rescale to meet the integral constraint
        perturbed_h = perturbed_h / (np.sum(perturbed_h) * dx) * (n_points / 2)
        perturbed_h = np.clip(perturbed_h, 0, 1)

        # Optimize
        result = optimizer.minimize(objective_function, x0=perturbed_h)
        if result.success:
            current_h = result.x
            current_c5 = evaluate_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5

        # Check budget
        if time.time() - start_time > budget_s - 1:
            break

    # Final check with a smaller number of points
    n_points = 100
    dx = 2.0 / n_points
    h_values = np.zeros(n_points)
    h_values[:int(n_points / 2)] = 1.0
    h_values[int(n_points / 2):] = 0.0
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values / (np.sum(h_values) * dx) * (n_points / 2)
    h_values = np.clip(h_values, 0, 1)

    # Final evaluation
    final_c5 = evaluate_overlap(h_values)
    if final_c5 < best_c5:
        best_h = h_values
        best_c5 = final_c5

    # Return the best solution found
    return (best_h, best_c5, n_points)
