import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial number of points
    n_points = 100  # Start with 100 points, can increase if time allows
    
    # Initial guess from provided h_values
    h_values = initial_h_values[:n_points]
    h_values = np.array(h_values)
    
    # Ensure the integral condition is satisfied
    integral = np.sum(h_values) * (2.0 / n_points)
    if np.abs(integral - 1.0) > 1e-6:
        h_values = (h_values / integral) * 1.0
    
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        h_extended = np.concatenate([h, np.zeros_like(h)])
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization: minimize C5 using a gradient-based approach
    def objective_func(x):
        h = x.copy()
        h = np.clip(h, 0.0, 1.0)
        integral = np.sum(h) * (2.0 / len(h))
        if np.abs(integral - 1.0) > 1e-6:
            h = (h / integral) * 1.0
        return compute_c5(h)
    
    def constraint_func(x):
        h = x.copy()
        integral = np.sum(h) * (2.0 / len(h))
        if np.abs(integral - 1.0) > 1e-6:
            return np.abs(integral - 1.0)
        return 0.0
    
    # Set up bounds for each h[i] (0 to 1)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    options = {
        'gtol': 1e-5,
        'ftol': 1e-5,
        'maxiter': 1000,
        'disp': False
    }
    
    # Use L-BFGS-B for optimization
    result = opt.minimize(
        fun=objective_func,
        x0=h_values,
        bounds=bounds,
        method='L-BFGS-B',
        options=options
    )
    
    # If optimization converges, use the result
    if result.success:
        h_values = result.x
        h_values = np.clip(h_values, 0.0, 1.0)
        integral = np.sum(h_values) * (2.0 / len(h_values))
        if np.abs(integral - 1.0) > 1e-6:
            h_values = (h_values / integral) * 1.0
    
    # Check time
    elapsed = time.time() - start_time
    if elapsed > budget_s - 5:
        # If time is tight, use the current solution
        pass
    else:
        # Optionally, try increasing n_points if time allows
        n_points = min(200, int(2.0 / (2.0 / n_points)))
        h_values = np.zeros(n_points)
        h_values[:n_points // 2] = 1.0
        h_values = np.concatenate([h_values, np.zeros_like(h_values)])
        h_values = h_values[:n_points]
        h_values = np.clip(h_values, 0.0, 1.0)
        integral = np.sum(h_values) * (2.0 / len(h_values))
        if np.abs(integral - 1.0) > 1e-6:
            h_values = (h_values / integral) * 1.0
    
    # Final evaluation
    c5_bound = compute_c5(h_values)
    
    return h_values, c5_bound, n_points
