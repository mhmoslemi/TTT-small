import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup based on the provided initial_h_values
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2):
        h = h / sum_h * (n_points / 2)
    
    # Function to compute the overlap integral
    def compute_c5(h_values):
        h = np.array(h_values)
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Function to compute C5 for the current h
    def objective(h_values):
        return compute_c5(h_values)
    
    # Function to compute the maximum overlap integral
    def evaluate(h_values):
        return compute_c5(h_values)
    
    # Optimization with simulated annealing
    def optimize_h(h_initial):
        n_points = len(h_initial)
        h = h_initial.copy()
        best_h = h.copy()
        best_c5 = evaluate(best_h)
        temp = 100.0
        cooling_rate = 0.995
        iterations = 1000
        
        for _ in range(iterations):
            if time.time() - start_time > budget_s:
                break
            # Perturb the h values
            idx = np.random.randint(n_points)
            delta = np.random.uniform(-0.1, 0.1)
            h[idx] = np.clip(h[idx] + delta, 0, 1)
            
            # Check the constraints
            if np.sum(h) < n_points / 2 - 1e-6 or np.sum(h) > n_points / 2 + 1e-6:
                h = best_h.copy()
                continue
            
            # Compute the new C5
            new_c5 = evaluate(h)
            
            # Acceptance probability
            if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / temp):
                best_h = h.copy()
                best_c5 = new_c5
                temp *= cooling_rate
        
        return best_h, best_c5
    
    # Use the initial guess
    best_h, best_c5 = optimize_h(h)
    
    # Try a few different starting points if time remains
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        # Generate a random starting point
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        best_h_rand, best_c5_rand = optimize_h(h_rand)
        if best_c5_rand < best_c5:
            best_h, best_c5 = best_h_rand, best_c5_rand
    
    if remaining_time > 10:
        # Generate another random starting point
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        best_h_rand, best_c5_rand = optimize_h(h_rand)
        if best_c5_rand < best_c5:
            best_h, best_c5 = best_h_rand, best_c5_rand
    
    return best_h, best_c5, n_points
