import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Define the number of points and basic parameters
    n_points = 100  # Start with 100 points for a balance between speed and accuracy
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points], dtype=float)
    h /= np.sum(h) * dx  # Normalize to meet the integral constraint
    
    def compute_c5(h):
        # Ensure h meets constraints
        h = np.clip(h, 0, 1)
        # Rescale to meet the integral constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute the overlap integral using convolution
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def objective(h):
        return compute_c5(h)
    
    def objective_with_constraints(h):
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Rescale to meet the integral constraint
        h = h * (n_points / 2) / np.sum(h)
        return compute_c5(h)
    
    # Define bounds for the optimization variables
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization: Use a local minimizer to refine the initial guess
    result = opt.minimize(objective, h, bounds=bounds, method='L-BFGS-B', options={'maxiter': 100})
    best_h = result.x
    
    # Ensure the final h satisfies the constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    c5_bound = compute_c5(best_h)
    
    # Try a few random perturbations to escape local minima
    for _ in range(3):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        c5 = compute_c5(perturbed_h)
        if c5 < c5_bound:
            best_h = perturbed_h
            c5_bound = c5
    
    # Final check to ensure we stay within the time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return (best_h.tolist(), c5_bound, n_points)
    
    # Return the best solution found
    return (best_h.tolist(), c5_bound, n_points)

# Start the timer
start_time = time.time()
