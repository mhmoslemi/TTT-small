def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial point from provided initial_h_values
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    sum_h = np.sum(h) * dx
    if abs(sum_h - 1.0) > 1e-6:
        h = h / sum_h * 1.0
    h = np.clip(h, 0.0, 1.0)
    
    def compute_c5(h):
        n = len(h)
        h_extended = np.concatenate([h, np.zeros(n)])
        corr = np.correlate(h_extended, 1.0 - h, mode="full")
        corr = corr[n-1:2*n-1]  # Extract relevant part
        c5 = np.max(corr * dx)
        return c5
    
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h) * dx
        if abs(sum_h - 1.0) > 1e-6:
            h = h / sum_h * 1.0
        return compute_c5(h)
    
    def constraint(x):
        h = x.copy()
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h) * dx
        if abs(sum_h - 1.0) > 1e-6:
            h = h / sum_h * 1.0
        return sum_h - 1.0
    
    # Local search with simulated annealing
    def local_search(h, budget_s):
        start_time = time.time()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        temp = 1.0
        cooling_rate = 0.995
        iterations = 0
        while time.time() - start_time < budget_s:
            new_h = h.copy()
            idx = np.random.randint(0, n_points)
            new_h[idx] = np.random.uniform(0.0, 1.0)
            new_h = np.clip(new_h, 0.0, 1.0)
            sum_h = np.sum(new_h) * dx
            if abs(sum_h - 1.0) > 1e-6:
                new_h = new_h / sum_h * 1.0
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / temp):
                h = new_h
                best_c5 = new_c5
                if best_c5 < 0.380876:
                    best_h = h.copy()
            temp *= cooling_rate
            iterations += 1
        return best_h, best_c5
    
    best_h, best_c5 = local_search(h, budget_s)
    return best_h.tolist(), best_c5, n_points
