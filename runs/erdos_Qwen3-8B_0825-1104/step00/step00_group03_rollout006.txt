import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported helper functions (simulated)
initial_h_values = np.random.rand(70) * 2 - 1  # Simulated initial values
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:] = h_values[:]
    # Pad h to double the length for full convolution
    h_padded = np.zeros(2 * n_points)
    h_padded[:n_points] = h
    # Compute (1 - h) for the convolution
    one_minus_h = 1 - h
    # Compute correlation
    corr = signal.correlate(h_padded, one_minus_h, mode='full')
    # Extract relevant portion
    overlap = corr[n_points - 1 : 2 * n_points - 1]
    # Compute the max value scaled by dx
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial values
    n_points = 100  # Start with 100 points, can try smaller if needed
    h_initial = initial_h_values[:n_points].copy()
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)  # Normalize to meet the integral constraint
    
    # Function to evaluate overlap
    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Ensure integral constraint is satisfied
        corr = signal.correlate(h, 1 - h, mode='full')
        overlap = corr[n_points - 1 : 2 * n_points - 1]
        return np.max(overlap) * (2.0 / n_points)
    
    # Optimization: minimize the maximum overlap
    # We'll use a gradient descent approach with line search
    h = h_initial.copy()
    learning_rate = 0.01
    max_iterations = 1000
    best_c5 = float('inf')
    best_h = h.copy()
    
    for _ in range(max_iterations):
        # Compute current C5
        current_c5 = evaluate_overlap(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient (approximate using finite differences)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_overlap(h_plus)
            h_minus = h.copy()
            h_minus[i] -= 1e-5
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            c5_minus = evaluate_overlap(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)
        
        # Update h
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Try a few more points with different configurations
    for n_points in [150, 200]:
        h = best_h.copy()
        h = h * (n_points / 2) / np.sum(h)
        h = np.random.permutation(h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = evaluate_overlap(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Final check with a different random initialization
    h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    current_c5 = evaluate_overlap(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Final evaluation
    final_c5 = evaluate_overlap(best_h)
    return best_h, final_c5, n_points
