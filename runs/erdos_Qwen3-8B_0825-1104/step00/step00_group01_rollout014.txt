```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Assume this is loaded with n_points=47

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial number of points
    n_points = 47
    h = initial_h_values.copy()
    
    # Function to compute the C5 value
    def compute_c5(h):
        return evaluate_erdos_solution(h)
    
    # Function to generate a perturbed version of h
    def perturb_h(h, perturb_ratio=0.1):
        h_perturbed = h.copy()
        indices = np.random.choice(len(h), int(len(h) * perturb_ratio), replace=False)
        h_perturbed[indices] = np.clip(h_perturbed[indices] + np.random.uniform(-0.2, 0.2, len(indices)), 0, 1)
        return h_perturbed
    
    # Function to optimize h with a smaller number of points
    def optimize_h(n_points, initial_h=None):
        if initial_h is None:
            initial_h = np.random.uniform(0, 1, n_points)
            total = np.sum(initial_h)
            initial_h = initial_h * (n_points / 2) / total
        else:
            initial_h = initial_h.copy()
            total = np.sum(initial_h)
            initial_h = initial_h * (n_points / 2) / total
        
        # Try a local search
        for _ in range(100):
            h_candidate = perturb_h(initial_h)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < compute_c5(initial_h):
                initial_h = h_candidate
        
        return initial_h, compute_c5(initial_h)
    
    # Try with fewer points
    n_points_small = 32
    h_small, c5_small = optimize_h(n_points_small)
    
    # Try with medium points
    n_points_medium = 64
    h_medium, c5_medium = optimize_h(n_points_medium)
    
    # Try with initial setup
    h_initial, c5_initial = optimize_h(n_points, initial_h=initial_h_values)
    
    # Select the best result
    best_h = h_initial
    best_c5 = c5_initial
    
    # Evaluate the small and medium point solutions
    best_h = h_small if c5_small < best_c5 else best_h
    best_c5 = c5_small if c5_small < best_c5 else best_c5
    best_h = h_medium if c5_medium < best_c5 else best_h
    best_c5 = c5_medium if c5_medium < best_c5 else best_c5
    
    # Refine the best solution using a simple optimization
    if time.time() - start_time < budget_s - 5:
        h_refined = best_h.copy()
        for _ in range(50):
            h_candidate = perturb_h(h_refined)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < compute_c5(h_refined):
                h_refined = h_candidate
        best_h = h_refined
        best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
```
