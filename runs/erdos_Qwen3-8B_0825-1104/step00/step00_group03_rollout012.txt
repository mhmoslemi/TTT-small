```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Try a few different starting points
    initial_points = [initial_h_values]
    for _ in range(3):
        # Create a random perturbation to the initial point
        perturbation = np.random.uniform(-0.1, 0.1, size=len(initial_h_values))
        perturbed = np.clip(initial_h_values + perturbation, 0, 1)
        initial_points.append(perturbed)
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    for h_initial in initial_points:
        n_points = len(h_initial)
        if n_points < 100:
            continue  # Skip small points to save time

        # Define the function to minimize
        def objective(x):
            h = x.reshape((n_points,))
            # Ensure h is in [0,1] and has integral 1
            h = np.clip(h, 0, 1)
            # Compute the overlap integral
            h_vec = np.array(h)
            h_shifted = np.zeros_like(h_vec)
            for k in range(1, n_points):
                h_shifted[k:] = h_vec[:n_points - k]
            overlap = np.sum(h_vec * (1 - h_shifted)) * (2.0 / n_points)
            # We want to minimize the maximum overlap
            return np.max(overlap)
        
        # Set up bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [cp.sum(h) == n_points / 2]
        
        # Try using a constrained optimization solver
        try:
            # Use scipy.optimize with bounds
            result = opt.minimize(objective, h_initial, bounds=bounds, method='L-BFGS-B', tol=1e-4, options={'maxiter': 100})
            h_opt = result.x
            c5 = objective(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n_points = n_points
        except:
            pass
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break
    
    # If no solution found, use the initial one
    if best_h is None:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, best_n_points)
```
