import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions and initial construction
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.pad(h, (0, 1), mode='constant')  # To handle full correlation
        corr = np.correlate(h, 1 - h, mode="full")
        dx = 2.0 / n_points
        return np.max(corr * dx)
    
    # Function to compute the integral constraint
    def integral_constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization: Use a direct search method for faster exploration
    # We use a simple gradient descent-like approach with random perturbations
    
    max_iterations = 100
    learning_rate = 0.01
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for _ in range(max_iterations):
        # Random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = np.clip(h + perturbation, 0, 1)
        
        # Enforce the integral constraint: sum(h) = n_points / 2
        h_perturbed = h_perturbed - (np.sum(h_perturbed) - n_points / 2) / n_points
        
        # Compute C5 for the perturbed h
        current_c5 = compute_c5(h_perturbed)
        
        # Accept the perturbation if it improves the result
        if current_c5 < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = current_c5
    
    # Check if we have enough time to refine with a more precise method
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.1:
        # Use a simple convex optimization with CVXPY to refine the result
        x = cp.Variable(n_points)
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]
        # Objective: minimize the maximum overlap integral
        # This is a difficult problem, so we approximate it by minimizing the maximum of several overlaps
        overlap_k = []
        for k in range(1, 6):  # Check overlaps for k = 1 to 5
            overlap_k.append(cp.Max(cp.sum(x * (1 - x[k:]))) * (2.0 / n_points))
        objective = cp.Minimize(cp.Max(overlap_k))
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_time=remaining_time)
            if prob.status == 'OPTIMAL':
                best_h = x.value
                best_c5 = compute_c5(best_h)
        except:
            pass
    
    return best_h.tolist(), best_c5, n_points
