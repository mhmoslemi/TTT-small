import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    # Initial construction from the global variable
    global initial_h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    time_limit = budget_s
    start_time = time.time()

    def compute_c5(h):
        h = np.array(h)
        shifted_h = np.concatenate([h[-n_points + 1:], h], axis=0)
        one_minus_h = 1 - h
        corr = correlate(shifted_h, one_minus_h, mode='full')
        return np.max(corr) * dx

    # Initialize with the best known construction
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)

    # Use a more sophisticated optimization strategy
    for _ in range(10):  # Perform multiple optimization iterations
        # Define the optimization problem
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
        objective = cp.Minimize(0)  # Dummy objective to start optimization

        # Use a different solver for better performance
        result = optimize.minimize(
            compute_c5,
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100}
        )
        if result.success:
            h_candidate = result.x
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()

        # Add a random perturbation to avoid local minima
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        c5_new = compute_c5(new_h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h.copy()

        # Check for time limit
        if time.time() - start_time >= time_limit:
            break

    # Final optimization with CVXPy to refine the solution
    h = cp.Variable(n_points)
    constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
    def objective_function(h_var):
        shifted_h = np.concatenate([h_var[-n_points + 1:], h_var], axis=0)
        one_minus_h = 1 - h_var
        corr = correlate(shifted_h, one_minus_h, mode='full')
        return np.max(corr)

    try:
        result = cp.Problem(cp.Minimize(objective_function(h)), constraints)
        result.solve(solver='SLSQP', eps=1e-6)
        if result.status == 'optimal':
            best_h = h.value
            best_c5 = compute_c5(best_h)
    except:
        pass

    # Return the best solution found
    return best_h, best_c5, n_points
