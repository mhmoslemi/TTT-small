import scipy.optimize as optimize

    def objective(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        correlation = np.correlate(h, one_minus_h, mode="full")
        overlap = correlation * dx
        return np.max(overlap)
    
    # Bounds for h: [0,1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint: integral of h must be 1
    def constraint(h):
        return np.sum(h) * dx - 1
    
    # Optimization with SLSQP
    def optimize_solution(initial_guess):
        result = optimize.minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],

        )
        return result.x
    
    # Refine the solution with local optimization
    def refine_solution(h):
        result = optimize.minimize(

            method='SLSQP',

        )
        return result.x
    
    # Start with initial guess
    best_h = optimize_solution(initial_h)
    best_c5 = objective(best_h)
    
    # Perturb and optimize
    for _ in range(10):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)

        scale_factor = 1.0 / area if area != 0 else 1.0

            best_c5 = objective(best_h)

    elapsed_time = time.time() - time.time()
