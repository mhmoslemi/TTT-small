import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):

    random.seed(seed)

    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)

    def objective(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        correlation = np.correlate(h, one_minus_h, mode="full")
        overlap = correlation * dx
        return np.max(overlap)
    
    # Bounds for h: [0,1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint: integral of h must be 1
    def constraint(h):
        return np.sum(h) * dx - 1
    
    # Optimization with SLSQP
    def optimize_solution(initial_guess):
        result = optimize.minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'maxiter': 50, 'disp': False}
        )
        return result.x
    
    # Genetic algorithm with improved diversity and selection
    def genetic_algorithm(population_size=50, generations=20, mutation_rate=0.1, elite_ratio=0.1):
        population = np.random.uniform(0, 1, (population_size, n_points))
        for i in range(population_size):
            population[i] = np.clip(population[i], 0, 1)
            area = np.sum(population[i]) * dx
            scale_factor = 1.0 / area if area != 0 else 1.0
            population[i] *= scale_factor
        
        for _ in range(generations):
            fitness = np.array([objective(ind) for ind in population])
            elite_indices = np.argsort(fitness)[:int(elite_ratio * population_size)]
            elites = population[elite_indices]
            offspring = []
            for _ in range(population_size - len(elite_indices)):
                parent1 = elites[np.random.choice(len(elite_indices))]
                parent2 = elites[np.random.choice(len(elite_indices))]

                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])

                child[mask] = np.random.uniform(0, 1, np.sum(mask))
                child = np.clip(child, 0, 1)
                area = np.sum(child) * dx
                scale_factor = 1.0 / area if area != 0 else 1.0
                child *= scale_factor
                offspring.append(child)
            population = np.vstack([elites, offspring])
        
        best_individual = population[np.argmin(fitness)]
        return best_individual

    best_h = optimize_solution(initial_h)

        options={'maxiter': 50, 'disp': False}
    )

    best_c5 = objective(best_h)

    for _ in range(5):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = best_h + perturbation

        perturbed_h *= scale_factor

            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            options={'maxiter': 50, 'disp': False}
        )

            best_h = result_perturb.x
            best_c5 = objective(best_h)
    
    # Early termination if time is running out
    elapsed_time = time.time() - time.time()
    if elapsed_time > budget_s - 5:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
