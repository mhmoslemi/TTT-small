import scipy.optimize as optimize
import scipy.signal as signal
from scipy.optimize import Bounds, differential_evolution

def run(seed=42, budget_s=60, **kwargs):

        initial_h
    )

    initial_h_resampled = initial_h_resampled * (n_points / 2) / np.sum(initial_h_resampled)

    def objective(h):
        # Ensure h is in [0, 1] and has sum n_points / 2
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        # Compute the overlap integral
        corr = signal.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx

    # Bounds for h (each value between 0 and 1)
    bounds = Bounds(0, 1)

    # Initial optimization with L-BFGS-B
    result = optimize.minimize(
        objective,
        initial_h_resampled,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 500, 'ftol': 1e-8}
    )

    # Perturb and refine
    h_values = result.x
    best_c5 = result.fun
    best_h = h_values

    # Use differential evolution for global search
    def differential_evolution_objective(h):
        return objective(h)

    # Define bounds for differential evolution
    de_bounds = [(0, 1) for _ in range(n_points)]

    # Run differential evolution
    de_result = differential_evolution(
        differential_evolution_objective,
        de_bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=50,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )

    # Evaluate DE result
    de_h = de_result.x
    de_c5 = objective(de_h)
    if de_c5 < best_c5:
        best_c5 = de_c5
        best_h = de_h

    # Refine with L-BFGS-B on DE result
    refined_result = optimize.minimize(
        objective,
        de_h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    if refined_result.fun < best_c5:
        best_c5 = refined_result.fun
        best_h = refined_result.x

    # Final refinement with random perturbation
    for _ in range(5):
        perturbation = np.random.normal(0, 0.03, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        perturbed_result = optimize.minimize(
            objective,
            perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if perturbed_result.fun < best_c5:
            best_c5 = perturbed_result.fun

                method='L-BFGS-B',

            )
            if perturbed_result.fun < best_c5:
                best_c5 = perturbed_result.fun
                best_h = perturbed_result.x
