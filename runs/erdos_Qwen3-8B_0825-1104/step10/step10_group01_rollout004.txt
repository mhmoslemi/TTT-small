import scipy.optimize as optimize
import time

    def objective(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        correlation = np.correlate(h, one_minus_h, mode="full")
        overlap = correlation * dx
        return np.max(overlap)
    
    # Area constraint function
    def area_constraint(h):
        return np.sum(h) * dx - 1.0
    
    # Optimization bounds: [0, 1] for all points
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimizer settings
    options = {'maxiter': 200, 'ftol': 1e-8, 'disp': False}
    
    # Initial optimization with multiple perturbations
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    for _ in range(5):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = initial_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        
        # Ensure area constraint is satisfied
        area_perturbed = np.sum(perturbed_h) * dx
        scale_factor = 1.0 / area_perturbed
        perturbed_h *= scale_factor
        
        # Local optimization
        result = optimize.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': area_constraint}],
            options=options
        )
        
        # Update best solution
        if objective(result.x) < best_c5:
            best_h = result.x
            best_c5 = objective(best_h)
