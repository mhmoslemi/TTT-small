import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a moderate number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Use the initial construction as a starting point
    initial_h = np.array(initial_h_values)
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        correlation = np.correlate(h, one_minus_h, mode="full")
        overlap = correlation * dx
        return np.max(overlap)
    
    # Area constraint function
    def area_constraint(h):
        return np.sum(h) * dx - 1.0
    
    # Optimization bounds: [0, 1] for all points
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimizer settings
    options = {'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    
    # Initialize best solution
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Use a more systematic approach with genetic algorithm
    # Genetic algorithm parameters
    pop_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Function to create a random valid individual
    def create_individual():
        h = np.random.rand(n_points)
        area = np.sum(h) * dx
        scale_factor = 1.0 / area
        return h * scale_factor
    
    # Function to evaluate a population
    def evaluate_population(population):
        return [objective(ind) for ind in population]
    
    # Function to perform crossover
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(n_points):
            if np.random.rand() < crossover_rate:
                child[i] = parent2[i]
        return child
    
    # Function to perform mutation
    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        area = np.sum(individual) * dx
        scale_factor = 1.0 / area
        return individual * scale_factor
    
    # Initialize population
    population = [create_individual() for _ in range(pop_size)]
    
    # Genetic algorithm optimization
    start_time = time.time()
    for generation in range(generations):
        # Evaluate population
        fitness = evaluate_population(population)
        
        # Select the best individuals
        best_indices = np.argsort(fitness)
        selected = [population[i] for i in best_indices[:pop_size//2]]
        
        # Create new population through crossover and mutation
        new_population = []
        for _ in range(pop_size):
            parent1 = random.choice(selected)
            parent2 = random.choice(selected)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
        
        # Early termination if time is running out
        if time.time() - start_time > budget_s - 5:
            break
    
    # Find the best individual from the final population
    best_individual = min(population, key=objective)
    best_c5 = objective(best_individual)
    
    # Refine the best solution with local optimization
    result = optimize.minimize(
        objective,
        best_individual,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': area_constraint}],
        options=options
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Return the best solution found
    return best_h, best_c5, n_points
