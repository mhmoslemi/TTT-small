import scipy.optimize as optimize
import time

    def objective(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        correlation = np.correlate(h, one_minus_h, mode="full")
        overlap = correlation * dx
        return np.max(overlap)
    
    # Area constraint function
    def area_constraint(h):
        return np.sum(h) * dx - 1.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    result = optimize.minimize(

        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': area_constraint}],
        options=options
    )

    best_h = result.x
