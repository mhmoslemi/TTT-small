import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    # Initial construction (pre-loaded)
    initial_h_values = np.array([0.4925, 0.5833, 0.6458, 0.6865, 0.7135, 0.7342, 0.7495, 0.7602, 0.7673, 0.7713,
                                 0.7722, 0.7701, 0.7652, 0.7578, 0.7482, 0.7367, 0.7235, 0.7086, 0.6923, 0.6746,
                                 0.6559, 0.6364, 0.6162, 0.5954, 0.574, 0.5522, 0.5301, 0.5076, 0.4847, 0.4616,
                                 0.4382, 0.4145, 0.3905, 0.3663, 0.3419, 0.3173, 0.2926, 0.2677, 0.2428, 0.2179,
                                 0.193, 0.168, 0.143, 0.118, 0.093, 0.068, 0.043, 0.018, 0.0])

    # Try different n_points values (smaller is faster, but larger can yield better results)
    n_points_list = [49, 64, 80, 96, 112, 128]  # These values are chosen to balance speed and accuracy
    best_c5 = 1.0
    best_h = initial_h_values
    best_n_points = 49

    # Seed for reproducibility
    np.random.seed(seed)

    for n_points in n_points_list:
        # Ensure the number of points is even to simplify the problem
        if n_points % 2 != 0:
            continue

        # Construct initial sample points
        dx = 2.0 / n_points
        h_initial = np.full(n_points, 0.5) if n_points == 49 else np.random.uniform(0, 1, n_points)

        # Normalize the initial h to satisfy the integral constraint
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)

        # Compute initial C5
        h_initial = np.clip(h_initial, 0, 1)
        def compute_c5(h):
            h = np.array(h)
            shifted_h = np.concatenate([h[-n_points + 1:], h], axis=0)
            one_minus_h = 1 - h
            corr = correlate(shifted_h, one_minus_h, mode='full')
            return np.max(corr) * dx

        # Use gradient-based optimization
        def objective(x):
            return compute_c5(x)

        result = optimize.minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-5}
        )

        if result.success:
            h_candidate = np.clip(result.x, 0, 1)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
                best_n_points = n_points

        # Add random perturbation to explore new solutions
        for _ in range(3):
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            new_h = np.clip(result.x + perturbation, 0, 1)
            c5_new = compute_c5(new_h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = new_h
                best_n_points = n_points

        # Check time budget
        if time.time() - time.time() >= budget_s:
            break

    return best_h, best_c5, best_n_points
