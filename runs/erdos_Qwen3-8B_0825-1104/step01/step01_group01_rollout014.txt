def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points

    corr = np.zeros(n_points)
    for k in range(n_points):
        corr[k] = np.sum(h * (1 - h_shifted[k, :])) * dx
    return np.max(corr)
