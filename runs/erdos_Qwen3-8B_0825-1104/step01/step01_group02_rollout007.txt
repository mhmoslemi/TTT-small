import numpy as np
import scipy.optimize as optimize
import time
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import eigs
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Determine the number of points. Let's start with 100 points for efficiency.
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial guess from the pre-defined starting point scaled to n_points
    h = np.zeros(n_points)
    initial_h = initial_h_values
    for i in range(n_points):
        h[i] = initial_h[int(i * len(initial_h) / n_points)]
    
    # Normalize the initial guess to satisfy the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Define the overlap integral function
    def compute_c5(h_values):
        h = np.array(h_values)
        # Create shifted versions of h
        h_shifted = []
        for k in range(1, len(h) - 1):
            shifted_h = np.roll(h, k)
            h_shifted.append(shifted_h)
        
        # Compute the maximum overlap integral
        max_overlap = 0
        for shifted_h in h_shifted:
            overlap = np.sum(h * (1 - shifted_h)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    # Define the objective function for optimization
    def objective(x):
        return compute_c5(x)
    
    # Define bounds for the variables
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a gradient-based optimization method
    result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
    
    # If optimization improved the result, use that; otherwise, use the initial guess
    if result.success and result.fun < compute_c5(h):
        h_result = result.x
    else:
        h_result = h
    
    # Compute the final C5 value
    c5_bound = compute_c5(h_result)
    
    # Stop if time is up
    if time.time() - start_time > budget_s:
        return (h_result.tolist(), c5_bound, n_points)
    
    # Try another optimization with different initial points to avoid local optima
    # Perturb the current solution slightly
    h_perturbed = h_result + 0.01 * np.random.rand(n_points) - 0.005
    h_perturbed = np.clip(h_perturbed, 0, 1)
    h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
    
    result_perturbed = optimize.minimize(objective, h_perturbed, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
    
    if result_perturbed.success and result_perturbed.fun < compute_c5(h_result):
        h_result = result_perturbed.x
    
    c5_bound = compute_c5(h_result)
    
    # Stop if time is up
    if time.time() - start_time > budget_s:
        return (h_result.tolist(), c5_bound, n_points)
    
    # Try a genetic algorithm for global optimization
    def genetic_algorithm(population_size=20, generations=20, mutation_rate=0.1):
        population = np.random.rand(population_size, n_points)
        population = np.clip(population, 0, 1)
        population = population * (n_points / 2) / np.sum(population, axis=1, keepdims=True)
        
        for _ in range(generations):
            fitness = np.array([compute_c5(ind) for ind in population])
            best_idx = np.argmin(fitness)
            best = population[best_idx]
            
            # Create offspring
            offspring = []
            for _ in range(population_size // 2):
                parent1 = population[np.random.choice(population_size)]
                parent2 = population[np.random.choice(population_size)]
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                offspring.append(child)
            
            # Mutation
            for i in range(len(offspring)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        offspring[i][j] += np.random.rand() - 0.5
                        offspring[i][j] = np.clip(offspring[i][j], 0, 1)
                        offspring[i][j] = offspring[i][j] * (n_points / 2) / np.sum(offspring[i])
            
            population = np.vstack((population, offspring))
            population = np.unique(population, axis=0)
            population = population[:population_size]
        
        return population[np.argmin([compute_c5(ind) for ind in population])]
    
    genetic_result = genetic_algorithm(population_size=20, generations=20, mutation_rate=0.1)
    c5_genetic = compute_c5(genetic_result)
    
    if c5_genetic < c5_bound:
        h_result = genetic_result
        c5_bound = c5_genetic
    
    # Final check if time is up
    if time.time() - start_time > budget_s:
        return (h_result.tolist(), c5_bound, n_points)
    
    # If time allows, refine the solution with a more precise method
    # Using CVXPY for constrained optimization
    x = cp.Variable(n_points)
    constraints = [
        cp.sum(x) == n_points / 2,
        0 <= x,
        x <= 1
    ]
    objective = cp.Maximize(cp.sum(x * (1 - cp.roll(x, k)) * dx) for k in range(1, n_points - 1))
    prob = cp.Problem(objective, constraints)
    result_cvxpy = prob.solve(solver=cp.SCS)
    
    if result_cvxpy is not None:
        h_cvxpy = x.value
        c5_cvxpy = compute_c5(h_cvxpy)
        if c5_cvxpy < c5_bound:
            h_result = h_cvxpy
            c5_bound = c5_cvxpy
    
    return (h_result.tolist(), c5_bound, n_points)
