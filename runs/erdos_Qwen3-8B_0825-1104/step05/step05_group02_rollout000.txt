    def objective(h):

    result = optimize.minimize(

        method='SLSQP',

    )
    
    # Best solution so far
    best_h = result.x
    
    # Perturb and optimize to escape local minima
    for _ in range(5):

        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        result_perturbed = optimize.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,

        )

        return np.max(overlap)

    result_refine = optimize.minimize(
