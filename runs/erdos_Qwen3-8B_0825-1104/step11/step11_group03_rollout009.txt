import scipy.optimize as optimize
import time

def run(seed=42, budget_s=60, **kwargs):

    def objective(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        correlation = np.correlate(h, one_minus_h, mode="full")
        overlap = correlation * dx
        return np.max(overlap)
    
    def area_constraint(h):
        return np.sum(h) * dx - 1.0
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    options = {'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Initial optimization
    result = optimize.minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': area_constraint}],
        options=options
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Refinement with multiple perturbations
    for _ in range(3):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        area_perturbed = np.sum(perturbed_h) * dx
        scale_factor = 1.0 / area_perturbed
        perturbed_h *= scale_factor
        
        result = optimize.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': area_constraint}],
            options=options
        )

        result = optimize.minimize(
            objective,
            random_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': area_constraint}],
            options=options
        )

        options=options
    )
