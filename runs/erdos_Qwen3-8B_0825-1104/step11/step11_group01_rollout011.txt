import scipy.optimize as optimize
import time

    def objective(h):

        one_minus_h = 1 - h
        correlation = np.correlate(h, one_minus_h, mode="full")
        overlap = correlation * dx
        return np.max(overlap)

    def area_constraint(h):
        return np.sum(h) * dx - 1.0
    
    # Optimization bounds: [0, 1] for all points
    bounds = [(0.0, 1.0) for _ in range(n_points)]

        perturbed_solutions = [np.random.uniform(0.0, 1.0, size=n_points) for _ in range(5)]

            # Adjust to satisfy area constraint
            area_perturbed = np.sum(perturbed) * dx
            scale_factor = 1.0 / area_perturbed
            perturbed *= scale_factor
            
            # Local optimization
            result = optimize.minimize(
                objective,
                perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': area_constraint}],

            )
