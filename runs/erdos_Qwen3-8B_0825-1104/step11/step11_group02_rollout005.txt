import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial construction
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Initialize the best solution
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Define the objective function to minimize the maximum overlap
    def objective(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        correlation = np.correlate(h, one_minus_h, mode="full")
        overlap = correlation * dx
        return np.max(overlap)
    
    # Define the area constraint function
    def area_constraint(h):
        return np.sum(h) * dx - 1.0
    
    # Initial bounds for all points
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    options = {'maxiter': 300, 'ftol': 1e-9, 'disp': False}
    
    # Global optimization with multiple starting points
    for _ in range(3):
        # Generate a random perturbation
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = initial_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        
        # Ensure the area constraint is satisfied
        area_perturbed = np.sum(perturbed_h) * dx
        scale_factor = 1.0 / area_perturbed
        perturbed_h *= scale_factor
        
        # Perform local optimization
        result = optimize.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': area_constraint}],
            options=options
        )
        
        # Update the best solution if improved
        if objective(result.x) < best_c5:
            best_h = result.x
            best_c5 = objective(best_h)
    
    # Refine the solution with a higher resolution grid
    n_points_refine = 200
    dx_refine = 2.0 / n_points_refine
    best_h_refine = np.interp(np.linspace(0, 2, n_points_refine), np.linspace(0, 2, n_points), best_h)
    best_h_refine = np.clip(best_h_refine, 0, 1)
    
    # Refine the solution further
    result_refine = optimize.minimize(
        objective,
        best_h_refine,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_points_refine)],
        constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) * dx_refine - 1.0}],
        options=options
    )
    
    # Interpolate back to the original grid
    best_h_final = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points_refine), result_refine.x)

    # Ensure the area constraint is satisfied
    area_final = np.sum(best_h_final) * dx
    scale_factor_final = 1.0 / area_final
    best_h_final *= scale_factor_final
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h_final)
    
    # Early termination if time is running out
    elapsed_time = time.time() - time.time()
    if elapsed_time > budget_s - 5:
        return best_h_final, final_c5, n_points

    for _ in range(3):
        perturbation = np.random.uniform(-0.03, 0.03, size=n_points)
        perturbed_h = best_h_final + perturbation

        # Ensure the area constraint is satisfied
        area_perturbed = np.sum(perturbed_h) * dx
        scale_factor_perturbed = 1.0 / area_perturbed
        perturbed_h *= scale_factor_perturbed
        
        # Perform local optimization
        result_perturb = optimize.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': area_constraint}],
            options=options
        )
        
        # Update the best solution if improved
        if objective(result_perturb.x) < final_c5:
            best_h_final = result_perturb.x
            final_c5 = objective(best_h_final)
    
    return best_h_final, final_c5, n_points
