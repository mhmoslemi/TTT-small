import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------
# Basic utilities
# -------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for step function h (dx = 2/len(h))."""
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Project v onto the box [lower,upper] while forcing sum(v)=target."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    # tiny correction for floating errors
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible(n, rng):
    """Generate a random feasible vector (0<=h<=1, sum = n/2)."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


def perturb_h(h, rng, scale=0.2):
    """Move a random amount from one coordinate to another, preserving sum."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple hill‑climb with decreasing step size."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# -------------------------------------------------
# Smooth‑max gradient descent (log‑sum‑exp surrogate)
# -------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP pass using the smooth‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x


def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)               # scaled overlaps
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.zeros_like(w)
            else:
                w /= w_sum                      # soft‑max weights

            # gradient of the smooth max (raw, before scaling)
            g_raw = np.ones(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:]    -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw

            new_h = h - step * grad
            new_h = project_onto_box_sum(new_h, n / 2.0)

            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
            else:
                # occasional uphill acceptance to escape shallow traps
                if rng.random() < 0.01:
                    h = new_h
            step *= 0.9995
    return best_h, best_c

def upsample_linear(h, new_n):
    """Upsample h to length new_n by linear interpolation on the interval [0,2)."""
    n = h.size
    old_x = np.linspace(0.0, 2.0, n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    up = np.interp(new_x, old_x, h)
    up = np.clip(up, 0.0, 1.0)
    up = project_onto_box_sum(up, new_n / 2.0)
    return up
