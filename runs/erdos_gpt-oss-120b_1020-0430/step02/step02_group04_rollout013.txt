    N = h.size

        step = 0.5

import time
import numpy as np

# ------------------------------------------------------------
# Basic utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Discrete C5 for step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v: np.ndarray, target: float,
                         lower: float = 0.0, upper: float = 1.0,
                         tol: float = 1e-12, max_iter: int = 60) -> np.ndarray:
    """Project v onto [lower,upper] with exact sum = target (box‑sum projection)."""
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def enforce_symmetry(h: np.ndarray) -> np.ndarray:
    """Make h symmetric and re‑project onto the feasible set."""
    h = (h + h[::-1]) * 0.5
    h = project_onto_box_sum(h, h.size / 2.0)
    return h


def random_feasible(N: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible vector (0≤h≤1, Σh = N/2)."""
    v = rng.random(N)
    v = project_onto_box_sum(v, N / 2.0)
    v = enforce_symmetry(v)
    return v


def perturb_h(h: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Swap a small amount of mass between two random positions."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start: np.ndarray, rng: np.random.Generator,
               max_iters: int = 2000, start_step: float = 0.5) -> tuple:
    """Simple hill‑climb using pairwise mass swaps."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        cand = enforce_symmetry(cand)
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_h = cand.copy()
            h = cand
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def gradient_descent_optimize(h0: np.ndarray, beta_sched,
                              steps_per_beta: int,
                              rng: np.random.Generator) -> tuple:
    """
    Projected gradient descent on the log‑sum‑exp surrogate.
    Returns the best vector and its C5 value.
    """
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(best_h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1 - h, mode="full")
            C = corr * (2.0 / n)               # scaled overlaps
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w_sum = w.sum()
            if w_sum == 0:
                w[:] = 1.0 / w.size
            else:
                w /= w_sum

            # gradient of the smooth max (before scaling by dx)
            g_raw = np.ones(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:]     -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw

            new_h = h - step * grad
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = project_onto_box_sum(new_h, n / 2.0)
            new_h = enforce_symmetry(new_h)

            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c = c_new
                best_h = new_h.copy()
                h = new_h
            else:
                # occasional uphill move to avoid stagnation
                if rng.random() < 0.01:
                    h = new_h
            step *= 0.9995
    return best_h, best_c


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs) -> tuple:
    rng = np.random.default_rng(seed)
    start = time.time()

    # --------------------------------------------------------
    # Initialise a feasible starter vector
    # --------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        N0 = h0.size
        target = N0 / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, target)
        h0 = enforce_symmetry(h0)

        # Upsample to a modest size (≥300) for better flexibility
        N_target = max(300, N0)
        if N_target > N0:
            old_x = np.linspace(0, 2, N0, endpoint=False)
            new_x = np.linspace(0, 2, N_target, endpoint=False)
            h0 = np.interp(new_x, old_x, h0)
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = project_onto_box_sum(h0, N_target / 2.0)
            h0 = enforce_symmetry(h0)
        N = h0.size
    else:
        N = 300
        h0 = random_feasible(N, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # --------------------------------------------------------
    # Quick hill‑climb from the parent (if time permits)
    # --------------------------------------------------------
    if time.time() - start < budget_s * 0.18:
        h_tmp, c_tmp = hill_climb(best_h, rng, max_iters=2500, start_step=0.5)
        if c_tmp < best_c:
            best_h, best_c = h_tmp, c_tmp

    # --------------------------------------------------------
    # Main optimisation loop (gradient descent + local hill climb)
    # --------------------------------------------------------
    betas = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0]
    steps_per_beta = 250
    restarts = 0
    max_restarts = 12

    while time.time() - start < budget_s * 0.80 and restarts < max_restarts:
        restarts += 1

        # -----------------------------------------------------------------
        # Initialise a candidate (sometimes a noisy perturbation of the best)
        # -----------------------------------------------------------------
        if rng.random() < 0.4:
            cand = best_h + rng.normal(scale=0.02, size=N)
            cand = np.clip(cand, 0.0, 1.0)
            cand = project_onto_box_sum(cand, N / 2.0)
            cand = enforce_symmetry(cand)
        else:
            cand = random_feasible(N, rng)

        # ----------------------------------------------------
        # Gradient‑descent phase
        # ----------------------------------------------------
        cand, c_cand = gradient_descent_optimize(cand, betas, steps_per_beta, rng)

        # ----------------------------------------------------
        # Light hill‑climb for final polishing
        # ----------------------------------------------------
        if time.time() - start < budget_s * 0.75:
            cand_hc, c_hc = hill_climb(cand, rng, max_iters=1200, start_step=0.3)
            if c_hc < c_cand:
                cand, c_cand = cand_hc, c_hc

        # ----------------------------------------------------
        # Update global best
        # ----------------------------------------------------
        if c_cand < best_c:
            best_h, best_c = cand, c_cand

    # --------------------------------------------------------
    # Final high‑beta refinement if time permits
    # --------------------------------------------------------
    if time.time() - start < budget_s * 0.93:
        refine, c_ref = gradient_descent_optimize(
            best_h, [2000.0, 5000.0], 400, rng
        )
        c_ref = compute_c5(refine)
        if c_ref < best_c:
            best_h, best_c = refine, c_ref

    # --------------------------------------------------------
    # Clean‑up and return
    # --------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_box_sum(best_h, best_h.size / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(best_h.size)
