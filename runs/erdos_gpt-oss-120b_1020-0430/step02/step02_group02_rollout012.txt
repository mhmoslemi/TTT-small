def c5_fft(h):
    n = h.size
    m = 2*n - 1
    nfft = 1 << (m-1).bit_length() # next power of 2 >= m
    H = np.fft.rfft(h, nfft)

    conv = np.fft.irfft(H * np.conj(G), nfft)[:m]
    max_corr = np.max(conv)
    return max_corr * (2.0 / n)

def simulated_annealing(h, time_limit, target=None):
    start = time.time()
    cur_h = h.copy()
    cur_c = c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    T0 = 0.05 # initial temperature maybe based on cur_c? Since objective <0.5.
    # schedule: T = T0 * (1 - elapsed/total) 
    while time.time() < time_limit:
        elapsed = time.time() - start
        remaining = time_limit - time.time()
        # compute temperature proportional to remaining fraction
        frac = elapsed / (time_limit - start)
        T = T0 * (1 - frac)  # linear cooling.
        # propose move:
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        # choose eps magnitude ~ sigma * random
        sigma = 0.1  # step size scaling
        eps = sigma * np.random.randn()
        # bound eps
        if eps > 0:
            eps = min(eps, cur_h[i], 1.0 - cur_h[j])
        else:
            eps = max(eps, -min(1.0 - cur_h[i], cur_h[j]))
        if eps == 0: continue
        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or np.exp(-delta / (T + 1e-12)) > np.random.rand():
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
        # optional occasional local refinement:
        if np.random.rand() < 0.001: # 0.1% chance
            # run SLSQP from cur_h small steps maybe 50 iter

            if opt_c < best_c:
                best_h, best_c = opt_h.copy(), opt_c
                cur_h, cur_c = opt_h.copy(), opt_c
    return best_h, best_c

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 using a water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # Tiny correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


def _corr_full(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1-h) using FFT (full length 2n‑1)."""
    n = h.size
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())  # next power of two
    H = np.fft.rfft(h, nfft)
    G = np.fft.rfft(1.0 - h, nfft)
    conv = np.fft.irfft(H * np.conj(G), nfft)[:m]
    # tiny imag part may appear; we keep real component
    return conv


def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a feasible h (integral already enforced)."""
    n = h.size
    max_corr = np.max(_corr_full(h))
    return max_corr * (2.0 / n)


# ----------------------------------------------------------------------
# Local SLSQP optimiser (exact constraints)
# ----------------------------------------------------------------------
def _ineq_constraint(x: np.ndarray) -> np.ndarray:
    n = x.size - 1
    h = x[:n]
    t = x[-1]
    return t - _corr_full(h)


def _eq_constraint(x: np.ndarray) -> float:
    n = x.size - 1
    return np.sum(x[:n]) - (n / 2.0)


def _solve_h(initial_h: np.ndarray, max_iter: int = 400) -> tuple:
    """Run one SLSQP minimise‑max‑correlation from a feasible start."""
    n = initial_h.size
    h0 = _project_to_feasible(initial_h, n)
    t0 = np.max(_corr_full(h0))
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constraints = [
        {"type": "ineq", "fun": _ineq_constraint},
        {"type": "eq", "fun": _eq_constraint},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9, "disp": False}
    res = optimize.minimize(
        lambda x: x[-1],
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=opts,
    )
    h_opt = res.x[:n] if res.success else h0
    h_opt = _project_to_feasible(h_opt, n)
    return h_opt, _c5_from_h(h_opt)


def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

# Simulated‑annealing style global search
# ----------------------------------------------------------------------
def _simulate_anneal(start_h: np.ndarray, time_limit: float) -> tuple:
    """Pair‑exchange SA with linear cooling; returns best (h,c5)."""
    n = start_h.size
    cur_h = start_h.copy()
    cur_c = _c5_from_h(cur_h)
    best_h, best_c = cur_h.copy(), cur_c

    start = time.time()
    total = max(time_limit - start, 0.0)
    if total <= 0.0:
        return best_h, best_c

    max_sigma = 0.15
    min_sigma = 0.005

    while True:
        now = time.time()
        if now >= time_limit:
            break
        elapsed = now - start
        frac = elapsed / total if total > 0 else 1.0
        sigma = max_sigma * (1 - frac) + min_sigma * frac

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        eps = sigma * np.random.randn()
        if eps > 0:
            eps = min(eps, cur_h[i], 1.0 - cur_h[j])
        else:
            eps = -min(-eps, 1.0 - cur_h[i], cur_h[j])
        if abs(eps) < 1e-12:

        cand = cur_h.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_c = _c5_from_h(cand)

        delta = cand_c - cur_c
        T = max(1e-5, 0.02 * (1 - frac))   # linear temperature decay
