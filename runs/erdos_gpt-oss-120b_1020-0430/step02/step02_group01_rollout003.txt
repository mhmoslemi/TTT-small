def simulated_annealing(h, rng, time_limit, start_temp, end_temp, max_step):
    n = h.size
    target = n/2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    start = time.time()
    while time.time() - start < time_limit:
        # compute temperature based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / time_limit
        temp = start_temp * (1 - frac) + end_temp * frac
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # decide direction randomly
        if rng.random() < 0.5:
            i_src, i_dst = i, j
        else:
            i_src, i_dst = j, i
        max_eps = min(cur_h[i_src], 1.0 - cur_h[i_dst])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * max_step
        if eps == 0:
            continue
        new_h = cur_h.copy()
        new_h[i_src] -= eps
        new_h[i_dst] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c

def hill_climb(h, rng, max_iters=1000, start_step=0.5):
    cur_h = h.copy()
    best_h = cur_h.copy()
    best_c = compute_c5(cur_h)
    step = start_step
    n = cur_h.size
    decay_every = max(1, max_iters // 5)
    for it in range(max_iters):
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # direction random
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_eps = min(cur_h[src], 1.0 - cur_h[dst])
        if max_eps <= 0:
            continue
        eps = (rng.random() * 2 - 1) * max_eps * step
        # but we need eps >=0 for transfer direction? Let's enforce eps >=0: transfer from src to dst.
        # Actually we the direction is already set, so eps is positive.
        eps = rng.random() * max_eps * step
        new_h = cur_h.copy()
        new_h[src] -= eps
        new_h[dst] += eps
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
            cur_h = new_h

            step = max(step*0.9, 1e-4)
    return best_h, best_c

def random_feasible(n, rng):
    # approach: sample n uniform random numbers, scale to sum target, and then clip and adjust.
    h = rng.random(n)
    target = n/2.0
    # simple scaling
    h = h * (target / h.sum())
    # now ensure within [0,1]; if any exceed 1, clip and fix sum.
    h = np.clip(h, 0.0, 1.0)
    # adjust sum difference iteratively
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h

CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1 - x, mode='full')
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log-sum-exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, rng, max_iter=300):

import time
import numpy as np
from scipy.optimize import minimize

# globals for soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def repair_sum(h, target, rng):
    """Adjust h in‑place to satisfy Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()


def random_feasible(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    h *= (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    repair_sum(h, target, rng)
    return h


def perturb_move(h, rng, scale):
    """Transfer ε from one index to another, preserving sum."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    # choose direction randomly
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    max_eps = min(h[src], 1.0 - h[dst])
    if max_eps <= 0.0:
        return h
    eps = rng.random() * max_eps * scale
    new_h = h.copy()
    new_h[src] -= eps
    new_h[dst] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (keep only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_every = max(1, max_iters // 5)
    for it in range(max_iters):
        new_h = perturb_move(h, rng, step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h, h = c, new_h.copy(), new_h
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
    return best_h, best_c


def simulated_annealing(h0, rng, time_limit, start_temp=1.0, end_temp=1e-4, max_step=0.5):
    """Simple SA that respects the sum constraint by transferring mass."""
    n = h0.size
    target = n / 2.0
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h, best_c = cur_h.copy(), cur_c
    start = time.time()
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        temp = start_temp * (1 - frac) + end_temp * frac
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # random direction
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_eps = min(cur_h[src], 1.0 - cur_h[dst])
        if max_eps <= 0.0:
            continue
        eps = rng.random() * max_eps * max_step
        new_h = cur_h.copy()
        new_h[src] -= eps
        new_h[dst] += eps
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=6):
    """Few gradient steps targeting the worst shift."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]          # + (1 - h[i+k]) for i< n-k
            g[k:] -= h[:upper]                # - h[i] for i>=k
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            repair_sum(h, target, rng)
    return h


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run one SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # exact sum correction
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        repair_sum(x, target, rng)
    return x
