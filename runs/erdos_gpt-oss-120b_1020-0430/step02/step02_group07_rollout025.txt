import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Random vector in [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h = np.zeros(n)
        idx = rng.choice(n, int(np.floor(target)), replace=False)
        h[idx] = 1.0
    else:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def compute_c5(h):
    """Exact discrete C5 for step function h on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.3):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.6, 1e-5)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=3000,
                        init_temp=0.04, decay=0.998, scale=0.2):
    """Stochastic hill‑climb with occasional uphill moves."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / max(temp, 1e-12)):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c

CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1-x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=1000):
    """Run SLSQP on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # repair possible numeric drift
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def refine_h(h, rng, betas=(5.0, 20.0, 80.0, 200.0, 500.0),
            slsqp_iter=1000, hill_iters=2000, sa_iters=3000):
    """Intensive local refinement."""
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in betas:
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=slsqp_iter)
        except Exception:
            continue
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
    # deterministic hill‑climb
    trial_h, c_h = hill_climb(best_h, rng, max_iters=hill_iters, start_step=0.3)
    if c_h < best_c:
        best_h, best_c = trial_h, c_h
    # simulated annealing
    trial_sa, c_sa = simulated_annealing(best_h, rng, max_iters=sa_iters,
                                        init_temp=0.04, decay=0.998, scale=0.2)
    if c_sa < best_c:
        best_h, best_c = trial_sa, c_sa
    return best_h, best_c


def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, preserving the integral."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up

                                slsqp_iter=600,
                                hill_iters=1200,
                                sa_iters=1800)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to higher resolution and deep refinement
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.80:
        h_up = upsample_h(best_h, factor=2, rng=rng)
        n_up = h_up.size
        h_up_ref, c_up_ref = refine_h(h_up, rng,
                                      betas=(5.0, 20.0, 80.0, 200.0, 500.0),
                                      slsqp_iter=1200,
                                      hill_iters=2500,
                                      sa_iters=4000)
        if c_up_ref < best_c:
            best_h, best_c = h_up_ref, c_up_ref
            n = n_up

    # optional second up‑sampling if time remains
    if time.time() - start_time < budget_s * 0.95 and n * 2 <= 800:
        h_up2 = upsample_h(best_h, factor=2, rng=rng)
        n_up2 = h_up2.size
        h_up2_ref, c_up2_ref = refine_h(h_up2, rng,
                                        betas=(5.0, 20.0, 80.0, 200.0, 500.0),
                                        slsqp_iter=1500,
                                        hill_iters=3000,
                                        sa_iters=5000)
        if c_up2_ref < best_c:
            best_h, best_c = h_up2_ref, c_up2_ref
            n = n_up2

    # ------------------------------------------------------------------
    # Final clean‑up
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
