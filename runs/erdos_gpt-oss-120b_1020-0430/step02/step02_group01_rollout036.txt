def softmax_corr_and_weights(h, beta):
    corr = np.correlate(h, 1 - h, mode='full')
    # compute stable softmax
    maxc = np.max(corr)
    exp_vals = np.exp(beta * (corr - maxc))
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    return corr, weights, maxc + np.log(sum_exp) / beta

def gradient_step(h, beta, step):
    corr, w, _ = softmax_corr_and_weights(h, beta)
    n = h.size
    term = np.zeros_like(h)
    # w has length 2n-1, shifts from -(n-1) .. n-1
    shifts = np.arange(-(n-1), n)
    # compute forward and backward contributions
    # using vectorized approach: for each shift, add w * rolled h
    # We can loop (fast enough)
    for wk, k in zip(w, shifts):
        if wk == 0:
            continue
        term += wk * np.roll(h, -k)  # shift left
        term += wk * np.roll(h, k)   # shift right
    grad = 1 - term
    # gradient descent: h_new = h - step * grad
    h_new = h - step * grad
    # project to [0,1] then adjust sum
    h_new = np.clip(h_new, 0.0, 1.0)
    # apply sum repair
    target = n/2.0
    diff = target - h_new.sum()
    if np.abs(diff) > 1e-12:
        # add diff over indices not at bounds
        # pick indices where h_new not at bound for adjustment direction
        # attempt random distribution
        for _ in range(1000):
            if np.abs(diff) < 1e-12:
                break
            idx = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h_new[idx])
            else:
                add = max(diff, -h_new[idx])
            h_new[idx] += add
            diff = target - h_new.sum()
    return h_new

def gradient_descent_step(h, beta, rng, step):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    maxc = np.max(corr)
    exp_vals = np.exp(beta * (corr - maxc))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    # gradient term: sum_k w_k * (np.roll(h, -k) + np.roll(h, k))
    term = np.zeros_like(h)
    shifts = np.arange(-(n-1), n)
    for wk, k in zip(w, shifts):
        if wk == 0.0:
            continue
        term += wk * np.roll(h, -k) + wk * np.roll(h, k)
    grad = 1.0 - term
    h_new = h - step * grad
    h_new = make_feasible(h_new, n/2.0, rng)
    # compute surrogate objective new
    corr_new = np.correlate(h_new, 1 - h_new, mode='full')
    maxc_new = np.max(corr_new)
    exp_new = np.exp(beta * (corr_new - maxc_new))
    obj_new = maxc_new + np.log(np.sum(exp_new)) / beta
    # compute old objective
    obj = maxc + np.log(sum_exp) / beta
    improved = obj_new < obj
    return h_new, improved, obj_new

import time

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def make_feasible(h, target, rng):
    """Clip to [0,1] and adjust sum to the required target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(2000):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, h.size)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    h = h * (target / h.sum())
    h = np.clip(h, 0.0, 1.0)
    return make_feasible(h, target, rng)


def upsample_h(h, n_new):
    """Linear interpolation of a periodic step function."""
    n_old = h.size
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    # periodised extension for interpolation at the right endpoint
    x_ext = np.concatenate((x_old, [2.0]))
    h_ext = np.concatenate((h, [h[0]]))
    return np.interp(x_new, x_ext, h_ext)

# Gradient descent on the smooth max‑overlap surrogate
# --------------------------------------------------------------
def gradient_descent_continuation(h, beta, rng, iterations=15):
    """Projected gradient descent on the log‑sum‑exp surrogate."""
    step = 0.5
    target = h.size / 2.0
    n = h.size
    shifts = np.arange(-(n - 1), n)          # all possible shifts
    for _ in range(iterations):
        # surrogate and its weights
        corr = np.correlate(h, 1 - h, mode="full")
        maxc = np.max(corr)
        exp_vals = np.exp(beta * (corr - maxc))
        sum_exp = np.sum(exp_vals)
        w = exp_vals / sum_exp

        # gradient = 1 - Σ_k w_k (h_{i+k}+h_{i-k})
        term = np.zeros_like(h)
        for wk, k in zip(w, shifts):
            if wk == 0.0:
                continue
            term += wk * np.roll(h, -k) + wk * np.roll(h, k)
        grad = 1.0 - term

        # trial step
        h_new = h - step * grad
        h_new = make_feasible(h_new, target, rng)

        # surrogate value after the step
        corr_new = np.correlate(h_new, 1 - h_new, mode="full")
        maxc_new = np.max(corr_new)
        exp_new = np.exp(beta * (corr_new - maxc_new))
        obj_new = maxc_new + np.log(np.sum(exp_new)) / beta
        obj = maxc + np.log(sum_exp) / beta

        if obj_new < obj:
            h = h_new
            step = min(step * 1.05, 1.0)
        else:
            step *= 0.7
    return h


def gradient_refine_worst(h, rng, iterations=8, step=0.02):
    """Fine‑tune using the true worst shift (sub‑gradient of the max)."""
    n = h.size

def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple deterministic hill‑climb (transfer mass between two coordinates)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    n = h.size
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):

        j = rng.integers(0, n)
        if i == j:

        if max_eps <= 0.0:
