import time, numpy as np
from scipy import optimize

def project_to_feasible(x, target, tol=1e-9, max_iter=200):
    # Clip first.
    x = np.clip(x, 0.0, 1.0)
    # water-filling iterative:
    for _ in range(max_iter):
        s = x.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            # need to decrease sum
            idx = x > 0
        else:
            idx = x < 1
        if not np.any(idx):
            break
        delta = diff / np.count_nonzero(idx)
        x[idx] -= delta
        x = np.clip(x, 0.0, 1.0)
    # final adjust residual
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        # distribute greedily
        if residual > 0:
            # add to elements <1
            idx = np.where(x < 1)[0]
            for i in idx:
                add = min(residual, 1 - x[i])
                x[i] += add
                residual -= add
                if residual <= 0:
                    break
        else:
            idx = np.where(x > 0)[0]
            for i in idx:
                sub = min(-residual, x[i])
                x[i] -= sub
                residual += sub
                if residual >= 0:
                    break
    return np.clip(x, 0.0, 1.0)

def c5_of_h(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def upscale_h(coarse_h, n_fine):
    n_coarse = len(coarse_h)
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)

import time
import numpy as np

def _project_to_feasible(x, target, tol=1e-9, max_iter=200):
    """Clip to [0,1] and adjust sum to target by water‑filling."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        s = x.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            idx = x > 0
        else:
            idx = x < 1
        if not np.any(idx):
            break
        delta = diff / np.count_nonzero(idx)
        x[idx] -= delta
        x = np.clip(x, 0.0, 1.0)
    # tiny residual correction
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(x < 1)[0]
            for i in idx:
                add = min(residual, 1.0 - x[i])
                x[i] += add
                residual -= add
                if residual <= 0:
                    break
        else:
            idx = np.where(x > 0)[0]
            for i in idx:
                sub = min(-residual, x[i])
                x[i] -= sub
                residual += sub
                if residual >= 0:
                    break
    return np.clip(x, 0.0, 1.0)


def _c5_of_h(h):
    """Return C5 for a discretised step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _tournament(pop, scores, k=3):
    """Return a copy of the best individual among k random contestants."""
    idxs = np.random.choice(len(pop), size=k, replace=False)
    best_idx = idxs[np.argmin(scores[idxs])]
    return pop[best_idx].copy()


def _upscale_h(coarse_h, n_fine):
    """Linear interpolation of a coarse step function onto a finer grid."""
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


def _local_refine(h, target, time_check, budget, start_time):
    """Simple stochastic hill‑climbing around h."""
    best_h = h.copy()
    best_c5 = _c5_of_h(best_h)
    attempts = 15
    for _ in range(attempts):
        if budget - (time.time() - start_time) < 1.0:
            break
        trial = best_h + 0.015 * np.random.randn(h.size)
        trial = _project_to_feasible(trial, target)
        c5 = _c5_of_h(trial)
        if c5 < best_c5:
            best_h, best_c5 = trial, c5
    return best_h, best_c5
