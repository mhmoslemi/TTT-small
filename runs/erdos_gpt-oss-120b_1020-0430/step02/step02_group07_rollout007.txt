def equalize_worst_shift(h, k, step, rng):
    n = len(h)
    if k == 0:
        return h
    if k > 0:
        i = np.arange(n - k)
        j = i + k
    else:
        k = -k
        i = np.arange(k, n)
        j = i - k
    # compute diff
    diff = h[i] - h[j]
    # move in direction to equalize
    delta = step * diff * 0.5
    # apply
    h[i] = np.clip(h[i] - delta, 0.0, 1.0)
    h[j] = np.clip(h[j] + delta, 0.0, 1.0)

    return h

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders for the SLSQP surrogate
# ----------------------------------------------------------------------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) via log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2] with n points."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0.0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    # ensure bounds
    return np.clip(h, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Generate a random vector with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    h = rng.random(n)
    s = h.sum()
    if s > 0.0:
        h = h * (target / s)
    else:
        h = np.zeros(n)
    h = np.clip(h, 0.0, 1.0)
    h = enforce_sum(h, target, rng)
    return h


def slsqp_optimize(h0, beta, max_iter=300):
    """One SLSQP run on the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # Repair any small drift in the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = len(x)
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0.0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def local_equalization(h, rng, max_iters=500, start_step=0.5):
    """
    Deterministic refinement: for the worst shift k (largest overlap)
    move h_i and h_{i+k} toward each other.
    """
    n = h.size
    target = n / 2.0
    step = start_step
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    for it in range(max_iters):
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        k = np.argmax(corr) - (n - 1)  # shift producing the max overlap
        new_h = cur_h.copy()
        if k != 0:
            if k > 0:
                i_idx = np.arange(n - k)
                j_idx = i_idx + k
            else:
                kk = -k
                i_idx = np.arange(kk, n)
                j_idx = i_idx - kk
            diff = new_h[i_idx] - new_h[j_idx]
            delta = step * diff * 0.5
            new_h[i_idx] = np.clip(new_h[i_idx] - delta, 0.0, 1.0)
            new_h[j_idx] = np.clip(new_h[j_idx] + delta, 0.0, 1.0)
        else:
            # random exchange move when the worst shift is zero
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_eps = min(new_h[i], 1.0 - new_h[j])
                if max_eps > 0.0:
                    eps = (rng.random() - 0.5) * 2.0 * max_eps * step
                    new_h[i] -= eps
                    new_h[j] += eps
        new_h = enforce_sum(new_h, target, rng)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            cur_h = new_h
        if (it + 1) % (max_iters // 10 + 1) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h


def refine_solution(h, rng, betas, slsqp_iter=200, eq_iters=500, eq_step=0.5):
    """Sequential SLSQP on a ladder of β, then local equalization."""
    for beta in betas:
        try:
            h = slsqp_optimize(h, beta=beta, max_iter=slsqp_iter)
        except Exception:
            continue
    h = local_equalization(h, rng, max_iters=eq_iters, start_step=eq_step)
    return h, compute_c5(h)


def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, preserving the integral."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    h_up = enforce_sum(h_up, target_up, rng)
    return h_up


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise from the supplied parent solution (if any)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        n = h0.size
        h0 = enforce_sum(h0, n / 2.0, rng)
    else:
        n = 120
        h0 = random_feasible_vector(n, rng)

    # quick refinement of the initial candidate
    quick_betas = (5.0, 20.0, 80.0)
    best_h, best_c = refine_solution(
        h0, rng, betas=quick_betas, slsqp_iter=150, eq_iters=400, eq_step=0.4
    )

    # ------------------------------------------------------------------
    # Phase 1 – many random restarts with cheap screening
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.55:
        h_rand = random_feasible_vector(n, rng)
        c_rand = compute_c5(h_rand)
        if c_rand < best_c:
            # refine only promising candidates
            h_ref, c_ref = refine_solution(
                h_rand,
                rng,
                betas=quick_betas,
                slsqp_iter=150,
                eq_iters=400,
                eq_step=0.4,
            )
            if c_ref < best_c:
                best_h, best_c = h_ref, c_ref

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample and deep refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() - start_time < budget_s * 0.85 and n * 2 <= 800:
        h_up = upsample_h(best_h, factor=2, rng=rng)
        n_up = h_up.size
        deep_betas = (5.0, 20.0, 80.0, 200.0, 500.0, 2000.0)
        h_up_ref, c_up_ref = refine_solution(
            h_up,
            rng,
            betas=deep_betas,
            slsqp_iter=250,
            eq_iters=800,
            eq_step=0.3,
        )
        if c_up_ref < best_c:
            best_h, best_c = h_up_ref, c_up_ref
            n = n_up

    # Optional second up‑sample
    if time.time() - start_time < budget_s * 0.95 and n * 2 <= 1000:
        h_up2 = upsample_h(best_h, factor=2, rng=rng)
        n_up2 = h_up2.size
        deep_betas = (5.0, 20.0, 80.0, 200.0, 500.0, 2000.0)
        h_up2_ref, c_up2_ref = refine_solution(
            h_up2,
            rng,
            betas=deep_betas,
            slsqp_iter=250,
            eq_iters=800,
            eq_step=0.3,
        )
        if c_up2_ref < best_c:
            best_h, best_c = h_up2_ref, c_up2_ref
            n = n_up2

    # ------------------------------------------------------------------
    # Final clean‑up (exact feasibility)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
