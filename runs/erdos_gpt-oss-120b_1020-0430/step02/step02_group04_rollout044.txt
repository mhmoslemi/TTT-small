def simulated_annealing(h0, steps, T0, T_end, rng):

    h = h0.copy()
    cur_c = compute_c5_fast(h)
    best_h_local = h.copy()
    best_c_local = cur_c
    for t in range(steps):
        # linear temperature schedule
        T = T0 + (T_end - T0) * (t / steps)
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2 * max_eps * 0.4  # scale 40% of allowable
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_c = compute_c5_fast(new_h)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, cur_c = new_h, new_c
            if cur_c < best_c_local:
                best_c_local, best_h_local = cur_c, h.copy()
    return best_h_local, best_c_local

import time
import numpy as np
from scipy.optimize import minimize

# ---------- projection onto [lower,upper] with exact sum ----------
def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    lo = lower - np.max(v) - 1.0
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj

# ---------- fast C5 evaluation via FFT ----------
def compute_c5_fast(h):
    """Maximum overlap for step function h on [0,2] (dx = 2/len(h))."""
    n = h.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    a = np.fft.rfft(h, m)
    b = np.fft.rfft((1.0 - h)[::-1], m)
    conv = np.fft.irfft(a * b, m)
    corr = conv[: 2 * n - 1]                     # linear convolution = correlation
    return np.max(corr) * (2.0 / n)

# ---------- simple feasible generator ----------
def random_feasible(n, rng):
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)

# ---------- simulated annealing on the exact C5 ----------
def simulated_annealing(h0, steps, T0, T_end, rng):
    h = h0.copy()
    cur_c = compute_c5_fast(h)
    best_h = h.copy()
    best_c = cur_c
    n = h.size
    for it in range(steps):
        # exponential cooling
        T = T0 * (T_end / T0) ** (it / steps)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # choose direction
        if rng.random() < 0.5:
            max_move = min(h[i], 1.0 - h[j])
            if max_move <= 0.0:
                continue
            eps = max_move * rng.random() * (T / T0)
            if eps <= 0.0:
                continue
            new_h = h.copy()
            new_h[i] -= eps
            new_h[j] += eps
        else:
            max_move = min(h[j], 1.0 - h[i])
            if max_move <= 0.0:
                continue
            eps = max_move * rng.random() * (T / T0)
            if eps <= 0.0:
                continue
            new_h = h.copy()
            new_h[j] -= eps
            new_h[i] += eps
        new_c = compute_c5_fast(new_h)
        delta = new_c - cur_c
        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
    return best_h, best_c

# ---------- SLSQP refinement using a smooth‑max surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None

def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET

def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp stable version
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA

def slsqp_optimize(h0, beta, max_iter=200):
    """Few SLSQP iterations starting from h0."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    x = project_onto_box_sum(x, n / 2.0)
    return x
