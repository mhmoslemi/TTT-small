def subgradient_descent(h0, max_iters, step0, decay, rng, sym=False):
    h = h0.copy()
    n = h.size
    target = n/2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for it in range(max_iters):
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = int(np.argmax(corr))
        max_val = corr[max_idx]
        shift = max_idx - (n - 1)
        # compute gradient
        g = np.zeros_like(h)
        if shift >= 0:
            end = n - shift
            # first factor
            g[:end] = 1.0 - h[shift:]
            # second factor
            g[shift:] -= h[:end]
        else:
            s = -shift
            start = s
            end = n
            # first factor
            g[start:end] = 1.0 - h[:n - s]
            # second factor
            g[:n - s] -= h[s:]
        # update
        h = h - step * g
        # clip and project
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_box_sum(h, target)
        if sym:
            h = (h + h[::-1]) * 0.5
            h = project_onto_box_sum(h, target)
        step *= decay

        if (it % 20) == 0:
            c = max_val * (2.0 / n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# ---------- basic utilities ----------
def compute_c5(h):
    """Discrete C5 for step vector h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_box_sum(v, target, lower=0.0, upper=1.0, tol=1e-12, max_iter=60):
    """Projection of v onto [lower,upper] with exact sum = target."""
    lo, hi = -1e6, 1e6
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, lower, upper)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) * 0.5, lower, upper)
    diff = target - proj.sum()
    if abs(diff) > tol:
        mask = (proj > lower + 1e-12) & (proj < upper - 1e-12)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def random_feasible(n, rng):
    """Uniform random feasible vector (box + sum)."""
    v = rng.random(n)
    return project_onto_box_sum(v, n / 2.0)


# ---------- hill‑climb (swap move) ----------
def perturb_h(h, rng, scale=0.2):
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=1500, start_step=0.4):
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c, best_h = c, new_h.copy()
            h = new_h
        if (it + 1) % max(1, max_iters // 8) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


# ---------- smooth‑max gradient descent ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Single SLSQP pass using soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return project_onto_box_sum(x, n / 2.0)


def grad_descent_optimize(h0, beta_sched, steps_per_beta, rng):
    """Projected GD on log‑sum‑exp surrogate."""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    n = h.size
    for beta in beta_sched:
        step = 0.5
        for _ in range(steps_per_beta):
            corr = np.correlate(h, 1.0 - h, mode="full")
            C = corr * (2.0 / n)                     # scaled overlaps
            bc = beta * C
            m = np.max(bc)
            w = np.exp(bc - m)
            w /= w.sum()
            # gradient of smooth max before scaling
            g_raw = np.ones(n)
            for idx, wk in enumerate(w):
                if wk == 0.0:
                    continue
                k = idx - (n - 1)
                if k >= 0:
                    g_raw[: n - k] -= wk * h[k:]
                    g_raw[k:]      -= wk * h[: n - k]
                else:
                    kk = -k
                    g_raw[kk:]    -= wk * h[: n - kk]
                    g_raw[: n - kk] -= wk * h[kk:]
            grad = (2.0 / n) * g_raw
            new_h = h - step * grad
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = project_onto_box_sum(new_h, n / 2.0)
            c_new = compute_c5(new_h)
            if c_new < best_c:
                best_c, best_h = c_new, new_h.copy()
                h = new_h
            else:
                if rng.random() < 0.01:
                    h = new_h
            step *= 0.9995
    return best_h, best_c


# ---------- subgradient descent on exact max ----------
def subgradient_descent(h0, max_iters, step0, decay, rng, sym=False, deadline=None):
    """Projected sub‑gradient descent on the exact max overlap."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    step = step0
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        max_val = corr[max_idx]
        shift = max_idx - (n - 1)

        # gradient for this shift (sub‑gradient of max)
        g = np.zeros_like(h)
        if shift >= 0:
            end = n - shift
            g[:end] = 1.0 - h[shift:]
            g[shift:] -= h[:end]
        else:
            s = -shift
            g[s:] = 1.0 - h[: n - s]
            g[: n - s] -= h[s:]

        # projected step
        h = h - step * g
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_box_sum(h, target)

        if sym:
            h = (h + h[::-1]) * 0.5
            h = project_onto_box_sum(h, target)

        step *= decay

        if (it & 31) == 0:  # every 32 iterations
            c = max_val * (2.0 / n)
            if c < best_c:
                best_c, best_h = c, h.copy()
    return best_h, best_c
