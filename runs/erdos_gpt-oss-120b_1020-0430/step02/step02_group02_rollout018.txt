    # clip
    h = np.clip(h, 0.0, 1.0)
    # compute sum
    s = h.sum()
    target = target_sum
    # if sum close, done
    if abs(s - target) < 1e-12:
        return h
    # compute needed adjustment diff = s - target
    # We'll adjust by moving values within bounds.
    # Using simple approach: while diff not small:
    #    free = np.where((diff > 0 and h > 0) or (diff < 0 and h < 1))[0]
    #    if no free: break
    #    delta = diff / len(free)
    #    h[free] -= np.sign(diff) * delta (or adjust accordingly)
    #    clip again
    # to avoid infinite loops due to precision.
    # Let's implement with a few iterations (like 100) max.
    for _ in range(100):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-10:
            break
        if diff > 0:  # need to reduce sum
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= np.sign(diff) * delta
        h = np.clip(h, 0.0, 1.0)
    # final residual adjust single element
    res = target - h.sum()
    if abs(res) > 1e-12:
        idx = np.argmin(h) if res > 0 else np.argmax(h)
        add = min(res, 1 - h[idx]) if res > 0 else max(res, -h[idx])
        h[idx] += add
    return np.clip(h, 0.0, 1.0)

def _optimize_pgdc(n, initial_h, time_limit):
    # gradient descent with softmax.
    h = _project_to_sum_and_bounds(initial_h)
    best_h = h.copy()
    best_c = _c5_from_h(h)
    start = time.time()
    # parameters
    alpha = 5.0  # initial
    step = 0.5  # initial step size
    # schedule
    while time.time() < time_limit:
        # compute corr and g
        corr = np.correlate(h, 1.0 - h, mode='full')
        g_vals = corr * (2.0 / n)  # dx
        max_g = np.max(g_vals)
        # compute softmax weights
        # optionally increase alpha gradually
        weights = np.exp(alpha * (g_vals - max_g))
        p = weights / weights.sum()
        # gradient
        grad_raw = np.convolve(p, (1.0 - 2.0 * h), mode='full')
        grad = grad_raw[n-1:n-1+n] * (2.0 / n)  # multiply by dx
        # step update
        h_new = h - step * grad
        # project
        h_new = _project_to_sum_and_bounds(h_new)
        # evaluate
        c_new = _c5_from_h(h_new)
        if c_new < best_c - 1e-9:
            best_h, best_c = h_new.copy(), c_new
            # maybe increase step if improving
            step = min(step * 1.05, 2.0)
        else:
            # maybe revert step
            step = max(step * 0.5, 1e-4)
        h = h_new
        # adjust alpha gradually
        alpha = min(alpha + 0.1, 200.0)  # increase to sharpen max
        # early exit if best_c < target
        if best_c < 0.3808:
            break
    return best_h, best_c

def _project_to_feasible(h, target_sum, tol=1e-9, max_iter=200):
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    diff = total - target_sum
    for _ in range(max_iter):
        if abs(diff) <= tol:
            break
        if diff > 0:
            # need to subtract diff
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= np.sign(diff) * delta
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target_sum
    # final tiny correction
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # adjust one element if possible
        idx = np.argmin(h) if residual > 0 else np.argmax(h)
        if residual > 0:
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
        else:
            sub = min(-residual, h[idx])
            h[idx] -= sub
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np

def _c5_from_h(h: np.ndarray) -> float:
    """C5 value for a step function h (integral already fixed)."""
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _project_to_feasible(h: np.ndarray, target_sum: float,
                         tol: float = 1e-9, max_iter: int = 200) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=target_sum by a water‑filling scheme."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target_sum
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= np.sign(diff) * delta
        h = np.clip(h, 0.0, 1.0)
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        idx = np.argmin(h) if residual > 0 else np.argmax(h)
        if residual > 0:
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
        else:
            sub = min(-residual, h[idx])
            h[idx] -= sub
    return np.clip(h, 0.0, 1.0)


def _interpolate_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    """Linear interpolation of a coarse step vector onto a finer grid."""
    n_coarse = coarse_h.size
    xc = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    xf = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(xf, xc, coarse_h)


def _optimize_pgdc(initial_h: np.ndarray, time_limit: float,
                   max_iter: int = 2000) -> tuple:
    """Projected gradient descent with a soft‑max approximation of the max."""
    n = initial_h.size
    target = n / 2.0
    h = _project_to_feasible(initial_h, target)
    best_h = h.copy()
    best_c = _c5_from_h(h)
    dx = 2.0 / n
    alpha = 5.0
    step = 0.5

    for it in range(max_iter):
        if time.time() > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        g = corr * dx
        maxg = np.max(g)

        # soft‑max weights (avoids overflow)
        w = np.exp(alpha * (g - maxg))
        p = w / w.sum()

        # gradient of the smoothed objective
        a = 1.0 - 2.0 * h
        conv = np.convolve(p, a, mode="full")
        grad = conv[n - 1:n - 1 + n] * dx

        h_new = h - step * grad
        h_new = _project_to_feasible(h_new, target)

        c_new = _c5_from_h(h_new)
        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new
            step = min(step * 1.05, 2.0)
        else:
            step = max(step * 0.5, 1e-4)

        h = h_new
        alpha = min(alpha + 0.2, 300.0)

        if best_c < 0.3808:
            break

    return best_h, best_c


def _hill_climb(h: np.ndarray, max_steps: int, time_limit: float) -> tuple:
    """Small random exchanges between two coordinates, accepting improvements."""
    n = h.size
    best_h = h.copy()
    best_c = _c5_from_h(best_h)
    for _ in range(max_steps):
        if time.time() > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_down = best_h[i]
        max_up = 1.0 - best_h[j]
        eps = np.random.rand() * min(max_down, max_up) * 0.5
        if eps <= 0:
            continue
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        c = _c5_from_h(cand)
        if c < best_c:
            best_h, best_c = cand, c
    return best_h, best_c
