def hill_climb_sa(h0, rng, max_iters, start_step, T0, cooling_rate):
    h = h0.copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    step = start_step
    for it in range(max_iters):
        # optionally decay step size
        if it % (max_iters // 10 + 1) == 0:
            step = max(step * 0.9, 1e-4)

        new_h = perturb_h(h, rng, scale=step)
        new_c = compute_c5(new_h)
        delta = new_c - c
        accept = False
        if delta < 0:
            accept = True
        else:
            T = T0 * (cooling_rate ** it)
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            h = new_h
            c = new_c
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

    return corr.max() * (2.0 / h.size)

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables for soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp trick
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function h (dx = 2/len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def random_feasible_vector(n, rng):
    """Generate a random feasible h (0≤h_i≤1, Σh_i=n/2)."""
    h = rng.random(n)
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
    return h


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb_sa(h0, rng, max_iters=3000, start_step=0.5,
                  T0=0.01, cooling_rate=0.999):
    """Simulated‑annealing hill‑climb (pair transfers, accepts occasional worsening)."""
    h = h0.copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    step = start_step
    decay_every = max(1, max_iters // 10)
    for it in range(max_iters):
        if (it + 1) % decay_every == 0:
            step = max(step * 0.9, 1e-4)
        new_h = perturb_h(h, rng, scale=step)
        new_c = compute_c5(new_h)
        delta = new_c - c
        accept = False
        if delta < 0:
            accept = True
        else:
            T = T0 * (cooling_rate ** it)
            if T > 0 and rng.random() < np.exp(-delta / T):
                accept = True
        if accept:
            h = new_h
            c = new_c
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c


def gradient_refine(h, rng, step=0.02, max_iter=8):
    """Few gradient‑descent steps targeting the worst shift."""
    n = h.size
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr) - (n - 1)      # shift index (negative allowed)
        k = abs(idx)
        g = np.zeros_like(h)
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            upper = n - k
            g[:upper] += 1.0 - h[k:]          # + (1 - h[i+k])
            g[k:] -= h[:upper]                # - h[i]
        step_sz = step * (0.5 ** it)
        h = h - step_sz * g
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            for _ in range(500):
                if abs(diff) < 1e-12:
                    break
                i = rng.integers(0, n)
                if diff > 0:
                    add = min(diff, 1.0 - h[i])
                else:
                    add = max(diff, -h[i])
                h[i] += add
                diff = target - h.sum()
    return h


def slsqp_optimize(h0, beta, rng, max_iter=300):
    """Run a single SLSQP optimisation with the soft‑max surrogate."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    target = CURRENT_TARGET
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = target - x.sum()
    return x

    """Project h onto the hyperplane Σh = len(h)/2 while staying in [0,1]."""

        diff = target - h.sum()
