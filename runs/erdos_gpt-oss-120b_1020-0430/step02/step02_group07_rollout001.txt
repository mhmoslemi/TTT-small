def binary_refine(h, rng, max_iters=2000):
    n = h.shape[0]
    target = n // 2
    # threshold
    bin_h = (h > 0.5).astype(float)
    # adjust sum
    cur_sum = int(bin_h.sum())
    if cur_sum > target:
        # turn some 1's to 0
        idx1 = np.where(bin_h == 1)[0]
        to_zero = rng.choice(idx1, cur_sum - target, replace=False)
        bin_h[to_zero] = 0
    elif cur_sum < target:
        idx0 = np.where(bin_h == 0)[0]
        to_one = rng.choice(idx0, target - cur_sum, replace=False)
        bin_h[to_one] = 1

    best_h = bin_h.copy()
    best_c = compute_c5(best_h)
    for it in range(max_iters):
        # pick random 1 and 0
        idx_one = np.where(best_h == 1)[0]
        idx_zero = np.where(best_h == 0)[0]
        if idx_one.size == 0 or idx_zero.size == 0:
            break
        i = rng.choice(idx_one)
        j = rng.choice(idx_zero)
        # propose swap
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_h = cand
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Global placeholders for the soft‑max surrogate used by SLSQP
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth approximation of max_k Σ_i x_i (1 - x_{i+k}) using log‑sum‑exp."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp (stable)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C5 for a step function on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng, max_iter=5000):
    """Adjust h in‑place so that Σh = target while staying in [0,1]."""
    diff = target - h.sum()
    n = h.size
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        i = rng.integers(0, n)
        if diff > 0:
            add = min(diff, 1.0 - h[i])
        else:
            add = max(diff, -h[i])
        h[i] += add
        diff = target - h.sum()
    h[:] = np.clip(h, 0.0, 1.0)
    return h


def random_feasible_vector(n, rng):
    """Generate a random vector with 0≤h_i≤1 and Σh_i = n/2."""
    target = n / 2.0
    # start from Dirichlet-like scaling
    h = rng.random(n)
    s = h.sum()
    if s == 0:
        h[:] = 0.0
    else:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, target, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        # gradually shrink step size
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def simulated_annealing(h_start, rng, max_iters=2000, init_temp=0.02,
                        decay=0.999, scale=0.3):
    """Stochastic hill‑climb with occasional uphill moves."""
    h = h_start.copy()
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    temp = init_temp
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=scale)
        new_c = compute_c5(new_h)
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        if new_c < cur_c or rng.random() < np.exp((cur_c - new_c) / temp):
            h = new_h
            cur_c = new_c
        temp *= decay
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """SLSQP on the soft‑max surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # repair possible drift in the sum
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = len(x)
        for _ in range(2000):
            if abs(diff) < 1e-12:
                break
            i = rng.integers(0, n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x


def quick_refine(h, rng):
    """Fast (few β) refinement with hill climbing."""
    best_h, best_c = h.copy(), compute_c5(h)
    for beta in (5.0, 20.0, 80.0):
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=200)
        except Exception:
            continue
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
    # follow with a modest hill‑climb
    trial_h, c_h = hill_climb(best_h, rng, max_iters=1200, start_step=0.4)
    if c_h < best_c:
        best_h, best_c = trial_h, c_h
    return best_h, best_c


def binary_refine(h, rng, max_iters=3000):
    """Round to binary (0/1) with exact sum, then improve by swapping."""
    n = h.size
    target = n // 2
    bin_h = (h > 0.5).astype(float)
    cur = int(bin_h.sum())
    if cur > target:
        idx = np.where(bin_h == 1)[0]
        to_zero = rng.choice(idx, cur - target, replace=False)
        bin_h[to_zero] = 0.0
    elif cur < target:
        idx = np.where(bin_h == 0)[0]
        to_one = rng.choice(idx, target - cur, replace=False)
        bin_h[to_one] = 1.0
    best_h = bin_h.copy()
    best_c = compute_c5(best_h)
    # pre‑compute index pools for speed
    for _ in range(max_iters):
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_h = cand
    return best_h, best_c


def upsample_h(h, factor, rng):
    """Linear up‑sampling by an integer factor, preserving the integral."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    h_up = np.clip(h_up, 0.0, 1.0)
    target_up = (n * factor) / 2.0
    return enforce_sum(h_up, target_up, rng)


def deep_refine(h, rng, time_limit=None):
    """Intensive refinement: multi‑β SLSQP, hill climb, annealing, binary fix."""
    start = time.time()
    best_h, best_c = h.copy(), compute_c5(h)
    betas = (5.0, 20.0, 80.0, 200.0, 500.0, 1000.0)
    for beta in betas:
        if time_limit is not None and time.time() - start > time_limit:
            break
        try:
            trial = slsqp_optimize(best_h, beta=beta, max_iter=300)
        except Exception:
            continue
        c_trial = compute_c5(trial)
        if c_trial < best_c:
            best_h, best_c = trial, c_trial
        # local hill climb
        trial_h, trial_c = hill_climb(best_h, rng, max_iters=2000, start_step=0.3)
        if trial_c < best_c:
            best_h, best_c = trial_h, trial_c
    # simulated annealing
    if time_limit is None or time.time() - start < time_limit:
        trial_sa, c_sa = simulated_annealing(best_h, rng, max_iters=4000,
                                            init_temp=0.03, decay=0.998, scale=0.2)
        if c_sa < best_c:
            best_h, best_c = trial_sa, c_sa
    # binary fix & swap improvement
    if time_limit is None or time.time() - start < time_limit:
        bin_h, bin_c = binary_refine(best_h, rng, max_iters=4000)
        if bin_c < best_c:
            best_h, best_c = bin_h, bin_c
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise from the supplied parent solution (if any)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        h0 = enforce_sum(h0, target, rng)
    else:
        n = 180
        h0 = random_feasible_vector(n, rng)

    best_h, best_c = quick_refine(h0, rng)

    # ------------------------------------------------------------------
    # Phase 1 – many random restarts with quick refinement
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.60:
        h_start = random_feasible_vector(n, rng)
        h_ref, c_ref = quick_refine(h_start, rng)
        if c_ref < best_c:
            best_h, best_c = h_ref, c_ref
        # binary improvement after quick refine
        bin_h, bin_c = binary_refine(h_ref, rng, max_iters=2000)
        if bin_c < best_c:
            best_h, best_c = bin_h, bin_c
        if best_c < 0.3808:  # early exit if we beat the target
            break

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to higher resolution and deep refinement
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.90 and n * 2 <= 400:
        h_up = upsample_h(best_h, factor=2, rng=rng)
        remaining = budget_s * 0.90 - elapsed
        h_up_ref, c_up_ref = deep_refine(h_up, rng,
                                         time_limit=remaining * 0.95)
        if c_up_ref < best_c:
            best_h, best_c = h_up_ref, c_up_ref
            n = h_up_ref.size

    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.97 and n * 2 <= 400:
        h_up2 = upsample_h(best_h, factor=2, rng=rng)
        remaining = budget_s * 0.97 - elapsed
        h_up2_ref, c_up2_ref = deep_refine(h_up2, rng,
                                           time_limit=remaining * 0.95)
        if c_up2_ref < best_c:
            best_h, best_c = h_up2_ref, c_up2_ref
            n = h_up2_ref.size

    # ------------------------------------------------------------------
    # Final clean‑up (exact feasibility)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = enforce_sum(best_h, n / 2.0, rng)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
