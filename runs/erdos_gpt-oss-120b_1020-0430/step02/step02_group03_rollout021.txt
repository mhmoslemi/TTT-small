def enforce_sum_bounds(h, target, rng):
    # first clip
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        # pick random index
        i = rng.integers(0, n)
        # capacity for addition or subtraction
        if diff > 0:
            cap = 1.0 - h[i]
            delta = min(cap, diff)
        else:
            cap = -h[i]
            delta = max(cap, diff)
        h[i] += delta
        diff = target - h.sum()
    return h

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    """Equality constraint Σ x_i = CURRENT_TARGET."""
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth log‑sum‑exp surrogate of max_k Σ_i x_i (1‑x_{i+k})."""
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def compute_c5(h):
    """Exact discrete C₅ for step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def enforce_sum(h, target, rng):
    """Adjust h so that Σh = target while respecting 0≤h≤1."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    n = h.size
    # distribute the remaining deficit / surplus over entries that can move
    while abs(diff) > 1e-12:
        if diff > 0:
            # need to add mass
            free = np.where(h < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            i = rng.choice(free)
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
        else:
            # need to remove mass
            free = np.where(h > 1e-12)[0]
            if free.size == 0:
                break
            i = rng.choice(free)
            dec = min(-diff, h[i])
            h[i] -= dec
        diff = target - h.sum()
    return h


def random_feasible_vector(n, rng, target=None):
    """Generate a random feasible h (0≤h≤1, Σh = target)."""
    if target is None:
        target = n / 2.0
    v = rng.random(n)
    return enforce_sum(v, target, rng)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount ε from one coordinate to another (preserves sum)."""
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return h.copy()
    # direction i→j
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0:
        # try opposite direction
        max_eps = min(h[j], 1.0 - h[i])
        if max_eps <= 0:
            return h.copy()
        i, j = j, i
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_h = perturb_h(h, rng, scale=step)
        c = compute_c5(new_h)
        if c < best_c:
            best_c = c
            best_h = new_h.copy()
            h = new_h
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_h, best_c


def slsqp_optimize(h0, beta, max_iter=300):
    """Run SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny sum correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        rng = np.random.default_rng()
        n = x.size
        while abs(diff) > 1e-12:
            i = rng.integers(n)
            if diff > 0:
                add = min(diff, 1.0 - x[i])
            else:
                add = max(diff, -x[i])
            x[i] += add
            diff = CURRENT_TARGET - x.sum()
    return x

    scores = [compute_c5(h) for h in pop]

    best_h = pop[best_idx].copy()

    # ---- evolutionary search (simple (μ,λ)-ES) ----
    sigma = 0.25
    gen = 0
    no_improve = 0

        idx = np.argsort(combined_scores)[:pop_size]

            best_h = pop[0].copy()

                scores[i] = compute_c5(pop[i])
            no_improve = 0
