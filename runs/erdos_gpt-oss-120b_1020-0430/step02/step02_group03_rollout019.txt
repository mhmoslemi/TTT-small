def perturb_v(v, rng, scale=0.2):
    n = v.size
    i = rng.integers(0,n)
    j = rng.integers(0,n)
    if i==j: return v.copy()
    max_eps = min(v[i], 1.0 - v[j])
    if max_eps <= 0: return v.copy()
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    new_v = v.copy()
    new_v[i] -= eps
    new_v[j] += eps
    return new_v
```

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# globals used by the surrogate
GLOBAL_N = None          # total number of points (set in run)
CURRENT_BETA = None      # temperature for soft‑max surrogate


def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function h (dx = 2 / len(h))."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def symmetrize(v: np.ndarray, n: int) -> np.ndarray:
    """
    Build a symmetric step function h on [0,2] from a half‑vector v.
    For even n, h[i] = v[i] for i < n/2 and h[i] = 1 - v[n-1-i] otherwise.
    This guarantees Σ h = n/2 (i.e. ∫ h = 1) and 0 ≤ h ≤ 1.
    """
    m = n // 2
    v = np.clip(v, 0.0, 1.0)
    h = np.empty(n, dtype=float)
    h[:m] = v
    if n % 2 == 0:
        h[m:] = 1.0 - v[::-1]
    else:  # odd n: middle point forced to 0.5
        h[m] = 0.5
        h[m + 1 :] = 1.0 - v[::-1]
    return h


def softmax_obj_v(v: np.ndarray) -> float:
    """Smooth max surrogate (log‑sum‑exp) for the C5 objective."""
    n = GLOBAL_N
    h = symmetrize(v, n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    # log‑sum‑exp, numerically stable
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def perturb_v(v: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount between two coordinates while staying in [0,1]."""
    n = v.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return v.copy()
    max_eps = min(v[i], 1.0 - v[j])
    if max_eps <= 0.0:
        return v.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_v = v.copy()
    new_v[i] -= eps
    new_v[j] += eps
    return new_v


def hill_climb(
    start_v: np.ndarray,
    rng: np.random.Generator,
    max_iters: int = 2000,
    start_step: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Simple deterministic hill‑climb (accept only improvements)."""
    v = start_v.copy()
    best_v = v.copy()
    best_c = compute_c5(symmetrize(v, GLOBAL_N))
    step = start_step
    decay_interval = max(1, max_iters // 10)

    for it in range(max_iters):
        new_v = perturb_v(v, rng, scale=step)
        c = compute_c5(symmetrize(new_v, GLOBAL_N))
        if c < best_c:
            best_c = c
            best_v = new_v.copy()
            v = new_v
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)

    return best_v, best_c


def slsqp_optimize_v(v0: np.ndarray, beta: float, max_iter: int = 300) -> np.ndarray:
    """Refine v with SLSQP on the smooth surrogate."""
    global CURRENT_BETA
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * v0.size
    # No equality constraint – symmetry guarantees the integral.
    res = minimize(
        softmax_obj_v,
        v0,
        method="SLSQP",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    v = np.clip(res.x, 0.0, 1.0)
    return v


class ObjC5:
    """Callable wrapper for differential_evolution."""

    def __init__(self, n: int):
        self.n = n

    def __call__(self, v: np.ndarray) -> float:
        return compute_c5(symmetrize(v, self.n))


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # -------------------------------------------------------------
    # 1. Determine dimension
    # -------------------------------------------------------------
    if "initial_h_values" in globals():
        h_full = np.array(initial_h_values, dtype=float)
        n = h_full.size
    else:
        n = 180
        h_full = np.full(n, 0.5, dtype=float)

    # enforce even length for the simple symmetry construction
    if n % 2 == 1:
        # drop the last sample to make it even (integral stays ≈1)
        h_full = h_full[:-1]
        n -= 1

    global GLOBAL_N
    GLOBAL_N = n
    m = n // 2

    # -------------------------------------------------------------
    # 2. Build initial symmetric candidates
    # -------------------------------------------------------------
    # symmetric version of the provided initial function
    v0 = (h_full[:m] + (1.0 - h_full[-m:][::-1])) * 0.5
    v0 = np.clip(v0, 0.0, 1.0)

    candidates = [v0]
    for _ in range(4):
        candidates.append(rng.random(m))                     # uniform random
        # a simple linear ramp (0→1) on the first half

    best_v = None

    for v in candidates:

    time_elapsed = time.time() - start_time

    # -------------------------------------------------------------
    beta_list = [5.0, 20.0, 80.0, 200.0]

            global_best_v = v_opt.copy()

    # -------------------------------------------------------------
    # 5. Final clean‑up and output
    # -------------------------------------------------------------
    final_h = symmetrize(global_best_v, n).tolist()
    final_c = float(compute_c5(symmetrize(global_best_v, n)))
    return final_h, final_c, int(n)
