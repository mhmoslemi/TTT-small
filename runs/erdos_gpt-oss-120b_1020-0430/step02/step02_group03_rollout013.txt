```
def softmax_obj_p(p, beta):
    h = p_to_h(p)
    corr = np.correlate(h, 1 - h, mode='full')
    bc = beta * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta
```

SLSQP optimizer:

```
def slsqp_optimize_p(p0, beta, max_iter=400):
    bounds = [(0.0,1.0)] * p0.size
    # no equality constraints.
    def obj(x):
        return softmax_obj_p(x, beta)
    res = minimize(obj, p0, method='SLSQP', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = np.clip(res.x, 0.0, 1.0)
    return x
```

```
def perturb_p(p, rng, scale):
    m = p.size
    i = rng.integers(0,m)
    j = rng.integers(0,m)
    if i == j:
        return p.copy()
    max_eps = min(p[i], 1.0 - p[j])
    if max_eps <= 0:
        return p.copy()
    eps = (rng.random() - 0.5) * 2 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_p = p.copy()
    new_p[i] -= eps
    new_p[j] += eps
    return new_p
```

```
def hill_climb_p(p_start, rng, max_iters=2000, start_step=0.5):
    p = p_start.copy()
    best_p = p.copy()
    best_c = compute_c5_from_p(p)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_p = perturb_p(p, rng, scale=step)
        c = compute_c5_from_p(new_p)

            best_p = new_p.copy()
            p = new_p
```

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper functions working with the symmetric representation
# ------------------------------------------------------------

def p_to_h(p: np.ndarray) -> np.ndarray:
    """Map half‑vector p (size m) to full step function h (size 2*m) satisfying
    h(x) + h(2-x) = 1."""
    m = p.shape[0]
    h = np.empty(2 * m, dtype=float)
    h[:m] = p
    h[m:] = 1.0 - p[::-1]
    return h


def compute_c5_from_p(p: np.ndarray) -> float:
    """Exact discrete C5 for a given half‑vector p."""
    h = p_to_h(p)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def softmax_obj_p(p: np.ndarray, beta: float) -> float:
    """Smooth log‑sum‑exp surrogate of max_k Σ_i h_i (1‑h_{i+k})."""
    h = p_to_h(p)
    corr = np.correlate(h, 1.0 - h, mode="full")
    bc = beta * corr
    m = np.max(bc)
    # stable log‑sum‑exp
    return (m + np.log(np.sum(np.exp(bc - m)))) / beta


def slsqp_optimize_p(p0: np.ndarray, beta: float, max_iter: int = 400) -> np.ndarray:
    """Refine p using SLSQP on the smooth surrogate."""
    bounds = [(0.0, 1.0)] * p0.size

    def obj(x):
        return softmax_obj_p(x, beta)

    res = minimize(
        obj,
        p0,
        method="SLSQP",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    return x


def perturb_p(p: np.ndarray, rng: np.random.Generator, scale: float = 0.2) -> np.ndarray:
    """Transfer a small amount ε from p[i] to p[j] (preserves total sum)."""
    m = p.size
    i = rng.integers(0, m)
    j = rng.integers(0, m)
    if i == j:
        return p.copy()
    max_eps = min(p[i], 1.0 - p[j])
    if max_eps <= 0.0:
        return p.copy()
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    eps = max(-max_eps, min(eps, max_eps))
    new_p = p.copy()
    new_p[i] -= eps
    new_p[j] += eps
    return new_p


def hill_climb_p(p_start: np.ndarray, rng: np.random.Generator,
                 max_iters: int = 1500, start_step: float = 0.5):
    """Simple deterministic hill‑climb on the p‑space."""
    p = p_start.copy()
    best_p = p.copy()
    best_c = compute_c5_from_p(p)
    step = start_step
    decay_interval = max(1, max_iters // 10)
    for it in range(max_iters):
        new_p = perturb_p(p, rng, scale=step)
        c = compute_c5_from_p(new_p)
        if c < best_c:
            best_c = c
            best_p = new_p.copy()
            p = new_p
        if (it + 1) % decay_interval == 0:
            step = max(step * 0.7, 1e-4)
    return best_p, best_c
