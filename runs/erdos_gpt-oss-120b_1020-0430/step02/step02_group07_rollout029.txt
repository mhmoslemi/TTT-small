def project_to_feasible(x, target, tol=1e-8, max_iter=100):
    lo = np.min(x) - 1.0
    hi = np.max(x)
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if np.abs(s - target) <= tol:
            return y
        if s > target:
            # need larger lambda to reduce sum
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo+hi)/2.0, 0.0, 1.0)

def pgd_optimize(init_h, schedule, step, rng):
    h = init_h.copy()
    target = h.size/2.0
    for beta, iters in schedule:
        for i in range(iters):
            # compute correlation
            corr_full = np.correlate(h, 1 - h, mode='full')
            # compute weight
            w = np.exp(beta * corr_full - np.max(beta * corr_full))
            w /= w.sum()
            # compute gradient
            # compute FFT length
            n = h.size
            len_w = 2*n - 1
            N = 1 << ((len_w + n - 1).bit_length())
            # FFTs
            w_rev = w[::-1]
            # Use rfft for speed
            w_fft = np.fft.rfft(w, n=N)
            w_rev_fft = np.fft.rfft(w_rev, n=N)
            h_fft = np.fft.rfft(h, n=N)
            one_minus_h_fft = np.fft.rfft(1 - h, n=N)
            conv1 = np.fft.irfft(w_rev_fft * one_minus_h_fft, n=N)
            conv2 = np.fft.irfft(w_fft * h_fft, n=N)
            # slices
            term1 = conv1[len_w - 1 : len_w - 1 + n]  # len_w-1 = 2n-2
            term2 = conv2[n - 1 : n - 1 + n]
            grad = term1 - term2
            # step
            h_new = h - step * grad
            # projection
            h = project_onto_simplex_interval(h_new, target)
    return h

def hill_climb(h_start, rng, max_iters=2000, step_scale=0.2):
    # start with h = h_start copy
    best_h = h_start.copy()
    best_c = compute_c5(best_h)
    h = best_h.copy()
    for it in range(max_iters):
        i, j = rng.integers(0, h.size, size=2)
        if i == j:
            continue
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 0:
            continue
        eps = (rng.random() - 0.5) * 2 * max_eps * step_scale
        new_h = h.copy()
        new_h[i] -= eps
        new_h[j] += eps

import time
import numpy as np

def compute_c5(h):
    """Exact discrete C5 for a step function h on [0,2]."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_onto_simplex_interval(v, target, tol=1e-12, max_iter=200):
    """Project v onto {0≤x≤1, sum x = target} by water‑filling."""
    lo = np.min(v) - 1.0   # makes all entries clamp to 1
    hi = np.max(v)         # makes all entries clamp to 0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        y = np.clip(v - mid, 0.0, 1.0)
        s = y.sum()
        if abs(s - target) <= tol:
            return y
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def random_feasible_vector(n, rng):
    """Uniform random point in the feasible polytope."""
    target = n / 2.0
    v = rng.random(n)
    return project_onto_simplex_interval(v, target)


def perturb_h(h, rng, scale=0.2):
    """Transfer a small amount between two coordinates (preserves sum)."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h
    eps = (rng.random() - 0.5) * 2.0 * max_eps * scale
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def hill_climb(h_start, rng, max_iters=2000, start_step=0.5):
    """Simple deterministic hill‑climb (accept only improvements)."""
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = start_step
    for it in range(max_iters):
        cand = perturb_h(h, rng, scale=step)
        c = compute_c5(cand)
        if c < best_c:
            best_c = c
            best_h = cand.copy()
            h = cand
        # slowly shrink step size
        if (it + 1) % (max_iters // 8 + 1) == 0:
            step = max(step * 0.7, 1e-5)
    return best_h, best_c


def pgd_optimize(h0, schedule, step, target_sum, rng, end_time):
    """
    Projected gradient descent on the log‑sum‑exp surrogate.
    `schedule` is an iterable of (beta, iters) pairs.
    """
    h = h0.copy()
    n = h.size
    len_w = 2 * n - 1
    # pre‑allocate FFT length (power of two ≥ 3n‑2)
    N = 1 << ((len_w + n - 1).bit_length())
    for beta, iters in schedule:
        for _ in range(iters):
            if time.time() > end_time:
                return h
            # correlation f·(1−f) (full, zero‑padded)
            corr_full = np.correlate(h, 1 - h, mode="full")
            # smooth max weights
            z = beta * corr_full
            z -= np.max(z)                # prevent overflow
            w = np.exp(z)
            w /= w.sum()
            # FFT based convolutions for gradient
            w_fft = np.fft.rfft(w, n=N)
            w_rev_fft = np.fft.rfft(w[::-1], n=N)
            h_fft = np.fft.rfft(h, n=N)
            om_fft = np.fft.rfft(1 - h, n=N)
            conv1 = np.fft.irfft(w_rev_fft * om_fft, n=N)
            conv2 = np.fft.irfft(w_fft * h_fft, n=N)
            term1 = conv1[len_w - 1 : len_w - 1 + n]
            term2 = conv2[n - 1 : n - 1 + n]
            grad = term1 - term2
            # gradient step + projection
            h = project_onto_simplex_interval(h - step * grad, target_sum)
    return h


def upsample_h(h, factor):
    """Linear up‑sampling by an integer factor, preserving the integral."""
    n = h.size
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n * factor, endpoint=False)
    h_up = np.interp(x_new, x_old, h)
    target = (n * factor) / 2.0
    return project_onto_simplex_interval(h_up, target)
