def project_onto_bounds_sum(v, target, max_iter=100):
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper utilities ----------
def compute_c5(h):
    """Exact discrete C₅ for a step function h (dx = 2/n)."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def project_sum_bounds(v, target, max_iter=60):
    """
    Project vector v onto the set {x | 0≤x_i≤1, sum(x)=target}.
    Uses bisection on the Lagrange multiplier λ for the box‑constrained simplex.
    """
    lo = -np.max(v) - 1.0          # ensures v+λ ≤ 0 for all i
    hi = 1.0 - np.min(v) + 1.0     # ensures v+λ ≥ 1 for all i
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def enforce_symmetry(h):
    """Replace h by (h + reversed h)/2 (preserves sum)."""
    return (h + h[::-1]) * 0.5


def compute_descent_dir(h, beta):
    """
    Direction that (approximately) reduces the smooth max of the overlap.
    The direction equals sum_k p_k * (h shifted forward + h shifted backward),
    where p_k are softmax weights derived from the current beta.
    """
    n = h.size
    # self‑correlation of h (no 1‑h term, because constant part drops out)
    corr_full = np.correlate(h, h, mode="full")
    corr = corr_full[n - 1 :]                # length n, shift 0..n‑1
    corr = np.append(corr, 0.0)              # shift n gives zero overlap
    f = -corr                                # f_k = –corr_k
    max_f = np.max(f)
    w = np.exp(beta * (f - max_f))
    p = w / w.sum()

    dir_vec = np.zeros_like(h)
    # shift 0 contributes 2*h
    dir_vec += p[0] * 2.0 * h

    # shifts 1 … n‑1
    for k in range(1, n):
        wk = p[k]
        if wk == 0.0:
            continue
        # forward part: h[i+k] added to position i
        dir_vec[: n - k] += wk * h[k:]
        # backward part: h[i‑k] added to position i
        dir_vec[k:] += wk * h[: n - k]
    # shift n gives no contribution
    return dir_vec


# ---------- SLSQP surrogate ----------
CURRENT_TARGET = None
CURRENT_BETA = None


def sum_eq_constraint(x):
    return np.sum(x) - CURRENT_TARGET


def softmax_obj(x):
    """Smooth surrogate of max_k ∫ h(1‑h_shift)."""
    global CURRENT_BETA
    corr = np.correlate(x, 1 - x, mode="full")
    bc = CURRENT_BETA * corr
    m = np.max(bc)
    return (m + np.log(np.sum(np.exp(bc - m)))) / CURRENT_BETA


def slsqp_optimize(h0, beta, max_iter=300):
    """Refine h0 with SLSQP on the smooth surrogate for a given β."""
    global CURRENT_TARGET, CURRENT_BETA
    n = h0.size
    CURRENT_TARGET = n / 2.0
    CURRENT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_eq_constraint}
    res = minimize(
        softmax_obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    x = np.clip(res.x, 0.0, 1.0)
    # tiny drift correction
    diff = CURRENT_TARGET - x.sum()
    if abs(diff) > 1e-12:
        x = project_sum_bounds(x, CURRENT_TARGET)
    return x


# ---------- random symmetric construction ----------
def random_symmetric_vector(rng, half, half_target):
    """Generate a random feasible symmetric vector of length 2*half."""
    v = rng.random(half)
    v = project_sum_bounds(v, half_target)
    h = np.concatenate([v, v[::-1]])
    return h

    def perturb_h(h, scale=0.2):

    best_h = enforce_symmetry(best_h)
    best_h = project_sum_bounds(best_h, target_sum)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), int(n)
