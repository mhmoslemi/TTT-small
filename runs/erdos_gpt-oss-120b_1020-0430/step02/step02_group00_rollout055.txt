def subgradient_descent(h_init, max_iter=2000, max_time=None):
    h = h_init.copy()
    n = len(h)
    target_sum = n/2
    step = 0.02 (initial)
    best_h = h.copy()
    best_val = c5
    start = time.time()
    for it in range(max_iter):
        if max_time and time.time() - start > max_time:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(c)
        lag = idx - (n - 1)
        s = abs(lag)
        # compute gradient

            g = 1 - 2*h
        else:
            left = np.concatenate([h[s:], np.zeros(s)])
            right = np.concatenate([np.zeros(s), h[:-s]])
            g = 1 - left - right
        # step size schedule: step = init_step / sqrt(it+1)

            best_h = h_new.copy()
        # Dampen step if no improvement for many iterations?

        h = h_new
    return best_h, best_val

def gradient_descent(h, time_allow, init_eta=0.05):

    n = len(h)
    best_h = h.copy()
    best_val = _c5_from_h(h)
    start = time.time()
    it = 0
    while time.time() - start < time_allow:
        c = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(c)
        lag = idx - (n - 1)
        s = abs(lag)
        # gradient
        if s == 0:
            g = 1 - 2 * h
        else:
            left = np.concatenate([h[s:], np.zeros(s)])
            right = np.concatenate([np.zeros(s), h[:-s]])
            g = 1 - left - right
        eta = init_eta / np.sqrt(it + 1)

        if val < best_val - 1e-9:

        h = h_new
        it += 1

import time
import numpy as np

# --------------------------------------------------------------
# Feasibility projection (box [0,1] and sum = n/2)
# --------------------------------------------------------------
def _project_to_feasible(h: np.ndarray, n: int, tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        s = h.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        if diff > 0:
            free = np.where(h > 0)[0]
        else:
            free = np.where(h < 1)[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] -= delta
        h = np.clip(h, 0.0, 1.0)
    # tiny correction (rare)
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        for i in range(n):
            if residual > 0 and h[i] < 1 - 1e-12:
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
            elif residual < 0 and h[i] > 1e-12:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)


# --------------------------------------------------------------
# C5 evaluation for a step function
# --------------------------------------------------------------
def _c5_from_h(h: np.ndarray) -> float:
    n = h.size
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


# --------------------------------------------------------------
# Upscaling via linear interpolation
# --------------------------------------------------------------
def _upscale_h(coarse_h: np.ndarray, n_fine: int) -> np.ndarray:
    n_coarse = coarse_h.size
    x_coarse = (np.arange(n_coarse) + 0.5) * (2.0 / n_coarse)
    x_fine = (np.arange(n_fine) + 0.5) * (2.0 / n_fine)
    return np.interp(x_fine, x_coarse, coarse_h)


# --------------------------------------------------------------
# Gradient descent on the worst shift (subgradient method)
# --------------------------------------------------------------
def _gradient_descent(h0: np.ndarray, time_allow: float, init_eta: float = 0.1) -> tuple:
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_val = _c5_from_h(h)
    start = time.time()
    it = 0
    while time.time() - start < time_allow:
        # correlation (unscaled, just to find worst shift)
        c = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(c)
        lag = idx - (n - 1)
        s = lag if lag >= 0 else -lag
        # gradient of the worst shift
        if s == 0:
            g = 1.0 - 2.0 * h
        else:
            left = np.concatenate([h[s:], np.zeros(s)])
            right = np.concatenate([np.zeros(s), h[:-s]])
            g = 1.0 - left - right
        eta = init_eta / np.sqrt(it + 1)
        h_new = h - eta * g
        h_new = _project_to_feasible(h_new, n)
        val = _c5_from_h(h_new)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_val


# --------------------------------------------------------------
# Simulated annealing with sum‑preserving swaps
# --------------------------------------------------------------
def _simulated_annealing(h0: np.ndarray, time_allow: float,
                         init_T: float = 0.02, final_T: float = 0.001,
                         init_step: float = 0.2, final_step: float = 0.02) -> tuple:
    cur_h = h0.copy()
    n = cur_h.size
    cur_val = _c5_from_h(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    start = time.time()
    while time.time() - start < time_allow:
        elapsed = time.time() - start
        frac = elapsed / time_allow
        T = init_T * (1.0 - frac) + final_T * frac
        step = init_step * (1.0 - frac) + final_step * frac
        i, j = np.random.choice(n, size=2, replace=False)
        max_dec = cur_h[i]
        max_inc = 1.0 - cur_h[j]
        if max_dec == 0.0 or max_inc == 0.0:
            continue
        delta_max = min(max_dec, max_inc, step)
        delta = (np.random.rand() * 2.0 - 1.0) * delta_max
        delta = np.clip(delta, -max_dec, max_inc)
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_val = _c5_from_h(new_h)
        dval = new_val - cur_val
        if dval <= 0.0 or np.random.rand() < np.exp(-dval / (T + 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    return best_h, best_val


# --------------------------------------------------------------
# Combine the two local refinements
# --------------------------------------------------------------
def _improve_h(h_start: np.ndarray, n: int, budget: float) -> tuple:
    h_start = _project_to_feasible(h_start, n)
    gd_time = budget * 0.4
    sa_time = budget * 0.6
    h_gd, val_gd = _gradient_descent(h_start, gd_time)
    h_sa, val_sa = _simulated_annealing(h_gd, sa_time)
    if val_sa < val_gd:
        return h_sa, val_sa
    return h_gd, val_gd


# --------------------------------------------------------------
# Main driver
# --------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    # ------------------------------------------------------------------
    # Load optional initial construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except NameError:
        init_h = None

    best_h = None
    best_val = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Helper to possibly update the best global solution
    # ------------------------------------------------------------------
    def _update(candidate_h, candidate_val, n):
        nonlocal best_h, best_val, best_n
        if candidate_val < best_val - 1e-12:
            best_h = candidate_h.copy()
            best_val = float(candidate_val)
            best_n = n

    # ------------------------------------------------------------------
    # Phase 1 – polish the provided construction (if any)
    # ------------------------------------------------------------------
    if init_h is not None:
        n0 = init_h.size
        budget_phase = min(time_left() * 0.2, 180.0)
        if budget_phase > 5.0:
            h_opt, v_opt = _improve_h(init_h, n0, budget_phase)
            _update(h_opt, v_opt, n0)

    # ------------------------------------------------------------------
    # Phase 2 – random restarts at the base resolution (256)
    # ------------------------------------------------------------------
    base_n = 256
    if best_h is None:
        # start from a random feasible vector
        rand_h = np.random.rand(base_n)
        rand_h = _project_to_feasible(rand_h, base_n)
        best_h = rand_h
        best_val = _c5_from_h(best_h)
        best_n = base_n

    while time_left() > 30.0:
        chunk = min(30.0, time_left() * 0.25)
        # random feasible start
        start_h = np.random.rand(base_n)
        start_h = _project_to_feasible(start_h, base_n)
        h_opt, v_opt = _improve_h(start_h, base_n, chunk)
        _update(h_opt, v_opt, base_n)
        if best_val < 0.3808:
            break

    # ------------------------------------------------------------------
    # Phase 3 – upscale to a finer grid and refine further
    # ------------------------------------------------------------------
    if time_left() > 30.0:
        fine_n = 512
        up_h = _upscale_h(best_h, fine_n)
        budget_fine = min(time_left() * 0.7, 400.0)
        if budget_fine > 5.0:
            h_opt, v_opt = _improve_h(up_h, fine_n, budget_fine)
            _update(h_opt, v_opt, fine_n)

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    best_h = _project_to_feasible(best_h, best_n)
    final_val = _c5_from_h(best_h)
    best_val = float(final_val)
    return best_h.tolist(), best_val, int(best_n)
